    % --- Imagen adaptativa ---
    % \imagenfit[alto]{fichero}{título}{fuente}: la imagen queda fija entre los párrafos donde se escribe (no flota), con su
    % título en el índice de figuras y la fuente debajo. Sin [alto]: las figuras PDF (TikZ) salen al tamaño con que se
    % dibujaron y solo se reducen si no caben en el ancho o en el alto de la página; las PNG/JPG ocupan el ancho del texto
    % hasta \imgaltoras del alto de la página. Con [alto] (fracción del alto de la página; [1] = página entera) se fija el
    % tamaño de esa imagen. Si no cabe en lo que queda de página, pasa entera a la siguiente.
    % \imagenfithorizontal{fichero}{título}{fuente}: la imagen en una página girada (apaisada), en ese mismo punto del texto.
    % Reglas: main/GUIA_TEMAS.md, §4 (Figuras).
    \newdimen\imgcap
    \newdimen\imgh
    \newdimen\imgw
    \newdimen\imgneed
    \newsavebox\imgcaja
    \newcommand{\imgaltoras}{0.30}% alto por defecto de las PNG/JPG (fracción del alto de la página)
    \makeatletter
    \newcommand{\imagenfit}[4][]{%
      \addvspace{10pt}%
      \par\begingroup
        \imgcap=3.5\baselineskip% alto aproximado de título + fuente
        \imgw=\linewidth
        \filename@parse{#2}\def\img@pdf{pdf}%
        \if\relax\detokenize{#1}\relax
          \ifx\filename@ext\img@pdf
            \sbox\imgcaja{\includegraphics{\detokenize{#2}}}% tamaño natural
            \ifdim\wd\imgcaja<\imgw \imgw=\wd\imgcaja \fi
            \imgh=\ht\imgcaja
          \else
            \imgh=\imgaltoras\textheight
          \fi
        \else
          \imgh=#1\textheight
        \fi
        \dimen@=\textheight \advance\dimen@ by -\imgcap
        \ifdim\imgh>\dimen@ \imgh=\dimen@ \fi
        \sbox\imgcaja{\includegraphics[width=\imgw,height=\imgh,keepaspectratio]{\detokenize{#2}}}%
        \imgneed=\ht\imgcaja \advance\imgneed by \imgcap
        \Needspace{\imgneed}%
        \noindent\begin{minipage}{\textwidth}
          \centering
          \captionof{figure}{\textemdash\ #3}\par\vspace{0.3em}% título arriba
          \usebox\imgcaja\par
          \vspace{0.3em}\small\textit{Fuente: }#4% fuente abajo
        \end{minipage}%
      \endgroup
      \par\addvspace{10pt}\RequestRestore
    }
    \newcommand{\imagenfithorizontal}[3]{%
      \begin{landscape}
        \centering
        \captionof{figure}{\textemdash\ #2}\par\vspace{0.3em}%
        \includegraphics[width=\linewidth,height=\dimexpr\textheight-4\baselineskip\relax,keepaspectratio]{\detokenize{#1}}\par
        \vspace{0.3em}{\small\textit{Fuente: }#3}%
      \end{landscape}%
    }
    \makeatother

    % \anexo{Título}: un anexo equivale a un \section (su título sale del mismo tamaño). Dentro se usan \subsection,
    % \subsubsection y \paragraph, nunca \section: se numeran I.1, I.1.1, I.1.1.1 y las páginas del anexo, I-1, I-2…
    % En el índice de contenido inicial solo figura el nombre del anexo. Sus apartados forman el índice del anexo, que sale
    % solo si el anexo los tiene, justo después de la introducción del anexo (antes del primer \subsection).
    % \anexo*{Título}: el mismo anexo, sin ese índice. Reglas: main/GUIA_TEMAS.md, §6.

    \newcommand{\anx@empezar}[2]{%
      \clearpage
      \refstepcounter{anexo}%
      \global\EnAnexotrue
      \setcounter{page}{1}\gdef\thepage{\theanexo-\arabic{page}}%
      \setcounter{section}{\numexpr\value{anexo}-1\relax}\stepcounter{section}% el anexo hace de \section
      \gdef\thesection{\Roman{section}}\gdef\theHsection{anexo.\Roman{section}}%
      \phantomsection
      \protected@xdef\leq@current@section{Anexo \theanexo.- \unexpanded{#2}}\gdef\leq@current@subsection{}% índice de ecuaciones
      \addtocontents{stc}{\protect\contentsline{section}{Anexo \theanexo.- #2}{}{\@currentHref}}% índice inicial: solo el nombre, sin página
      \addcontentsline{toc}{section}{Anexo \theanexo.- #2}% marcadores del PDF
      {\noindent\normalfont\Large\bfseries Anexo \theanexo.- #2\par}\nobreak\vspace{1.5ex \@plus .2ex}%
      #1\@ifundefined{anx@lista@\theanexo}{\global\anx@pendientefalse}{}%
    }
    % Índice del anexo: cada apartado escribe una línea en el fichero .atc (anexo, nivel, texto, página, enlace); al
    % empezar el documento se leen las de la compilación anterior y se guardan por anexo (dos pasadas, como cualquier índice).
    % En ese índice los apartados suben un nivel, porque el anexo hace de \section.
    \def\anx@entrada#1#2#3#4#5{\@ifundefined{anx@lista@#1}{\global\@namedef{anx@lista@#1}{}}{}%
      \expandafter\g@addto@macro\csname anx@lista@#1\endcsname{\contentsline{#2}{#3}{#4}{#5}}}
    \tcbuselibrary{breakable}
    \def\anx@indice{\ifanx@pendiente\global\anx@pendientefalse

            before upper={\parindent=0pt\parskip=2pt}]
          \small
          % números más anchos que en el índice inicial (I.1, I.1.1, I.1.1.1; páginas I-1)
          \def\@pnumwidth{2.6em}\def\@tocrmarg{3.4em}%
          \def\l@section##1##2{\addpenalty\@secpenalty\addvspace{1.0em \@plus\p@}\setlength\@tempdima{2.6em}%
            \begingroup\parindent\z@\rightskip\@pnumwidth\parfillskip-\@pnumwidth\leavevmode\bfseries
            \advance\leftskip\@tempdima\hskip-\leftskip##1\nobreak\hfil\nobreak\hb@xt@\@pnumwidth{\hss##2\kern-\p@\kern\p@}\par\endgroup}%

          \csname anx@lista@\theanexo\endcsname

      \fi}
    \def\anx@escribir#1#2#3{\ifEnAnexo

    % margen superior, así que no cambia las páginas del tema. Se enciende con \pagenumbering{arabic} y se apaga con

    \cs_new_protected:Npn \cab_guardar:n #1
      {
        \tl_gset:Nn \g_cab_titulo_tl {#1}
        \regex_replace_once:nnN { \A \s* Tema \s+ \S+ \s* \-\-? \s* } { } \g_cab_titulo_tl % sin «Tema X.Y.N --»
        \regex_replace_once:nnN { \.? \s* \c{\\} (\[ [^\]]* \])? \s* (\c{large} \s*)? } { .\  } \g_cab_titulo_tl % título. subtítulo
        \regex_replace_once:nnN { [\.\s]* \Z } { } \g_cab_titulo_tl % sin subtítulo: nada detrás
      }

\section{La Síntesis Neoclásica: el consenso de posguerra}\label{b1}
\puntosclave{
La Síntesis Neoclásica fue el paradigma dominante desde la posguerra hasta los años setenta: keynesiana a corto plazo (precios y salarios lentos, políticas de demanda con prioridad fiscal) y neoclásica a largo (vaciado de mercados, equilibrio general, crecimiento). Nace de la lectura hidráulica de la Teoría General que formalizan HICKS (1937) y MODIGLIANI (1944).

Su creencia nuclear es que la economía es medible y estabilizable: econometría estructural para estimar las funciones, expectativa estática para los precios y ajuste fino de la demanda. Su supuesto más hondo es un mundo ergódico, cuyo equilibrio de largo plazo no depende de la historia.

El aparato es el IS-LM(-BP), leído con una IS inelástica (la inversión va por animal spirits) y una LM elástica: la política fiscal rinde y la monetaria apenas. La curva de PHILLIPS completa el sistema como ecuación perdida y se lee como un menú entre inflación y paro.

Las grietas estaban dentro: la esquizofrenia entre el corto plazo keynesiano y el largo plazo neoclásico, y una inflación que desde mediados de los sesenta deja de autocorregirse. Sobre ellas golpeará el monetarismo.
}

El cuarto de siglo posterior a la Segunda Guerra Mundial perteneció a una sola escuela. La Síntesis Neoclásica dio a los gobiernos occidentales su lenguaje económico, sus modelos de previsión y su confianza en que las recesiones eran evitables; entender qué creían sus autores, de dónde venían esas creencias y cómo se proyectaron sobre su método y su aparato es la mitad de este tema, y la condición para entender al rival que lo desmontó.

\subsection{El origen: ¿qué hacer con la \textit{Teoría General}?}

La escuela nace de un problema de lectura. La \textit{Teoría General} (\authorfont{KEYNES}, 1936) es una obra ensayística, sin modelo formal, y la experiencia de leerla es, en palabras de \authorfont{MANKIW}, tan estimulante como frustrante: las piezas del puzle no encajan solas.\footnote{%
    En el capítulo 12, \authorfont{KEYNES} declara que el mensaje central del libro es la incertidumbre que rodea a las decisiones de inversión; en el resto de los capítulos, por simplicidad, trabaja como si la información fuera buena. Qué pieza se toma por central decide qué escuela se funda.
} En los meses siguientes a su publicación, \authorfont{MEADE}, \authorfont{HARROD} y \authorfont{HICKS} emprendieron en paralelo la misma tarea, traducirla a un sistema de ecuaciones comparable con la teoría clásica, y del diagrama de \authorfont{HICKS} (1937) salió el embrión del IS-LM ([\authorfont{Ver Anexo \ref{anx:aparato}}]). Pero la traducción no era neutral: cada lectura de la obra fundó una escuela distinta, y conviene tener el mapa delante desde el principio, porque vertebra los cuatro temas siguientes del programa.

\begin{center}\begin{minipage}{\linewidth}\centering
{\small\textbf{Las lecturas de la \textit{Teoría General} y la genealogía de las escuelas}}\par\vspace{0.3em}
\includegraphics{b1_mapa.pdf}\par\vspace{0.3em}
{\small\textit{Fuente: }Elaboración propia.}
\end{minipage}\end{center}

La Síntesis es la lectura \textbf{hidráulica}: la obra entendida como un sistema de funciones agregadas estables (consumo, inversión, demanda de dinero) entre las que fluye el gasto y que la política puede manipular, dejando fuera la incertidumbre radical que reivindicarán los fundamentalistas y el análisis del desequilibrio ([\authorfont{Ver Tema 3.A.4}]; la taxonomía completa de las lecturas, en el Anexo \ref{anx:autores}).

¿Conciliación o resistencia? La pregunta incómoda es si esta lectura continúa a \authorfont{KEYNES} o lo domestica. La Síntesis surge, a priori, como una crítica frontal a la consideración de la \textit{Teoría General} como una formulación innovadora: el gesto fundacional de \authorfont{HICKS}, apenas un año después de publicarse la obra, consiste precisamente en reducirla a un caso particular de un modelo más general en el que también cabe la teoría clásica. Hay razones para leer en ello el conservadurismo intelectual con el que las ideas establecidas resisten a las revoluciones científicas, una dinámica que no es exclusiva de la economía;\footnote{%
    El paralelismo clásico es la física: la oposición frontal de \authorfont{EINSTEIN} al desarrollo pionero de la física cuántica, con toda la autoridad de su figura, pudo haber retrasado años de investigación; por suerte, muchos de sus colegas concluyeron que, a la postre, \href{https://www.nationalgeographic.com.es/ciencia/einstein-contra-cuantica-dios-no-juega-a-dados-universo_24976}{Dios sí parecía jugar a los dados}. La resistencia del paradigma establecido es el precio inevitable de la potencia de las ideas, también cuando retrasa lo nuevo.
} y hay razones para la lectura amable, porque el proyecto de integración era genuino. \authorfont{MODIGLIANI} lo dejó dicho al repasar su propia trayectoria\textcolor{orange}{\textsuperscript{[Probable]}}:

\begin{cita}[MODIGLIANI]
Uno de los aspectos principales que ha dominado mi investigación científica ha sido integrar los principales postulados de la Teoría General con la metodología establecida de la ciencia económica, la cual descansa en el comportamiento optimizador de los agentes económicos.
\end{cita}

El término mismo lo acuñó \authorfont{SAMUELSON} en su manual, describiendo el consenso de los economistas americanos de los años cincuenta\textcolor{orange}{\textsuperscript{[Probable]}}:

\begin{cita}[SAMUELSON][1955][Economics, 3.ª ed.]
En los años recientes, el 90 por ciento de los economistas americanos han dejado de llamarse economistas keynesianos o economistas antikeynesianos. En cambio, han trabajado juntos hacia una síntesis entre los postulados neoclásicos y las teorías keynesianas de determinación de la renta. El resultado podría llamarse síntesis neoclásica, y es compartido por todos salvo por un 5 por ciento que se sitúa en los extremos.
\end{cita}

Las dos lecturas no se excluyen: la Síntesis fue a la vez la vía por la que las ideas de \authorfont{KEYNES} conquistaron la academia y la política económica, y el molde que las recortó a lo que cabía en el equilibrio.\footnote{%
    De ahí que la historiografía discuta hasta el nombre: \authorfont{DE VROEY} considera más preciso hablar de \textit{macroeconomía keynesiana}, el \textit{Institute for New Economic Thinking} emplea \textit{neokeynesianismo} y \authorfont{MANKIW} propone \textit{síntesis neoclásico-keynesiana}. El debate sobre qué dijo de verdad \authorfont{KEYNES} y quién lo heredó se desarrolla en el Anexo \ref{anx:autores}.
}

\subsection{El sistema de creencias: ¿por qué keynesianos a corto plazo y neoclásicos a largo?}

El credo de la escuela cabe en dos compromisos keynesianos y una lealtad neoclásica. Los compromisos: \textbf{los precios y salarios no se ajustan con la rapidez necesaria para vaciar los mercados}, de modo que la estabilidad del sistema es lenta y el desempleo involuntario puede persistir; y, en consecuencia, \textbf{las políticas de demanda pueden y deben estabilizar el ciclo}, con prioridad de la fiscal sobre la monetaria. La lealtad: a largo plazo los precios son flexibles, los mercados se vacían y rige la teoría del equilibrio general competitivo, de modo que la microeconomía, el comercio internacional y el crecimiento se analizan con instrumental neoclásico, como hicieron los propios \authorfont{HICKS}, \authorfont{SAMUELSON} o \authorfont{SOLOW}. El keynesianismo queda así acotado a un dominio: el corto plazo macroeconómico de una economía dirigida por la demanda.

Bajo los compromisos hay axiomas compartidos: una \textbf{epistemología instrumental} (la teoría se formula en modelos matemáticos y se valida midiendo), el \textbf{individualismo metodológico} (la macroeconomía debe poder reconstruirse agregando agentes, el programa que \authorfont{MODIGLIANI} llamaba integrar a \authorfont{KEYNES} con el comportamiento optimizador), la \textbf{racionalidad} de unos agentes que maximizan su bienestar y una economía concebida como \textbf{intercambio bajo escasez}.\footnote{%
    La fe en la racionalidad no era ciega: especialmente entre los autores más keynesianos de la escuela, los \textit{animal spirits} se percibían como determinante principal de la inversión, y la incertidumbre que los rodea, la que \authorfont{KEYNES} y antes \authorfont{KNIGHT} distinguían del riesgo calculable, quedaba reconocida pero sin formalizar. Esa pieza no digerida es la bandera de los postkeynesianos ([\authorfont{Ver Tema 3.A.4}]) y el núcleo de la teoría de la decisión bajo incertidumbre ([\authorfont{Ver Tema 3.A.10}]).
} Y bajo los axiomas, la apuesta más honda, la que casi nunca se enuncia: \textbf{el mundo económico se trata como ergódico}, un sistema cuyo equilibrio de largo plazo existe, es único y no depende de las condiciones iniciales, de modo que el pasado estadístico informa sobre el futuro como la trayectoria pasada de un cometa informa de su regreso. \authorfont{SAMUELSON} llegó a presentar la hipótesis ergódica como el requisito para que la economía fuera una ciencia dura y saliera del terreno de la mera historia, según recuerda la crítica postkeynesiana de \authorfont{DAVIDSON} (1991)\textcolor{orange}{\textsuperscript{[Probable]}}.\footnote{%
    La objeción de \authorfont{DAVIDSON}: si los procesos económicos son creativos y no estacionarios (cada decisión altera el escenario sobre el que se decidió), los promedios del pasado no son la esperanza matemática del futuro, y una economía sin historia es una economía sin su objeto. La crítica heterodoxa contemporánea, en su versión divulgativa de \authorfont{ASTARITA} (2011), añade la carga ideológica: tratar el equilibrio y la distribución como leyes naturales independientes del punto de partida convierte lo existente en inevitable. La discusión completa del supuesto corresponde a los temas de incertidumbre y de las escuelas heterodoxas ([\authorfont{Ver Tema 3.A.10}]; [\authorfont{Ver Tema 3.A.4}]).
} La tensión interna de este credo tiene nombre propio: la \textit{esquizofrenia} de enseñar un largo plazo donde la política de demanda es inocua y practicar un corto plazo donde lo es todo, con la frontera entre ambos sin teoría propia ([\authorfont{Ver Anexo \ref{anx:aparato}}]).

\subsection{Método y aparato: ¿cómo se gobierna una economía?}

Del sistema de creencias sale el método. Si las funciones agregadas son estables, se pueden \textbf{estimar}; si se pueden estimar, la política económica es un problema de ingeniería. La escuela construyó para ello la econometría estructural de sistemas de ecuaciones simultáneas, del primer modelo nacional de \authorfont{TINBERGEN} al modelo MPS de \authorfont{MODIGLIANI} para la Reserva Federal, con 334 ecuaciones (la historia completa, con su batalla metodológica, se desarrolla en el Anexo \ref{anx:autores}). El tratamiento de las expectativas delata la misma confianza: para el corto plazo, la \textbf{expectativa estática} sobre los precios,

$$p^{e}_{t}=p_{t-1}$$

es decir, el nivel de precios esperado es el del periodo anterior y el ajuste de corto plazo se produce en cantidades; para plazos mayores, la flexibilidad neoclásica, articulada en las tres curvas de oferta del modelo completo ([\authorfont{Ver Anexo \ref{anx:aparato}}]). Que los agentes no aprendieran sistemáticamente de la inflación pasada no inquietaba a nadie mientras la inflación fue episódica; se verá enseguida por qué dejó de serlo, y más adelante lo que el monetarismo construyó exactamente sobre esa grieta.

El aparato es el modelo IS-LM(-BP), cuya construcción completa, estática comparativa y multiplicadores quedan para el Anexo \ref{anx:aparato}; aquí importa cómo lo leía la escuela, porque \textbf{las pendientes de las curvas son creencias cuantificadas}. En la lectura de la Síntesis, la inversión depende de los volátiles estados de ánimo empresariales mucho más que del tipo de interés ($b$ pequeño: IS inelástica) y la demanda de dinero es muy sensible al interés ($h$ grande: LM elástica, con el extremo de la trampa de la liquidez).

\begin{center}\begin{minipage}{\linewidth}\centering
{\small\textbf{Las elasticidades como creencias: la lectura de la Síntesis}}\par\vspace{0.3em}
\includegraphics{b1_islm.pdf}\par\vspace{0.3em}
{\small\textit{Fuente: }Elaboración propia.}
\end{minipage}\end{center}

Con esas pendientes, el veredicto de política está decidido antes de calcular nada: la expansión fiscal desplaza la IS y gana mucha renta con poco coste de interés ($E_0\to E_F$), mientras que la expansión monetaria desplaza la LM a lo largo de una IS casi vertical y apenas rasca renta ($E_0\to E_M$). \textbf{La política fiscal es el instrumento; la monetaria, el acompañamiento.} La culminación práctica fue el \textit{ajuste fino} (\textit{fine tuning}): la gestión continua de la demanda agregada para mantener la economía en su senda, con la rebaja fiscal estadounidense de 1964 como pieza maestra\footnote{%
    Diseñada por el Consejo de Asesores Económicos de \authorfont{KENNEDY} presidido por \authorfont{HELLER}, que popularizó el término \textit{fine tuning}, y aprobada con \authorfont{JOHNSON}; el episodio se consideró la demostración empírica del multiplicador en acción\textcolor{orange}{\textsuperscript{[Probable]}}.
} y una confianza creciente hasta la arrogancia: a finales de los años sesenta la profesión discutía en serio si el ciclo económico era ya cosa del pasado.\footnote{%
    La pregunta dio título a una conferencia y a su volumen, \textit{Is the Business Cycle Obsolete?} (\authorfont{BRONFENBRENNER}, ed., 1969)\textcolor{orange}{\textsuperscript{[Probable]}}: se creía posible evitar las recesiones como se evitan los accidentes aéreos, con instrumentos, medición y pericia. La década siguiente respondió por su cuenta ([\authorfont{Ver Tema 3.A.42}]).
}

\subsection{La ecuación perdida y el menú: ¿qué hacer con los precios?}

El modelo tenía un hueco: con precios rígidos, no explicaba la inflación. El hueco fue tolerable mientras la inflación se comportó como un fenómeno episódico que se autocorregía; dejó de serlo cuando se volvió persistente.

\begin{center}\begin{minipage}{\linewidth}\centering
{\small\textbf{La inflación deja de autocorregirse: Reino Unido y Estados Unidos}}\par\vspace{0.3em}
\includegraphics{b1_inflacion.pdf}\par\vspace{0.3em}
{\small\textit{Fuente: }Elaboración propia con datos del Banco de Inglaterra (\textit{A Millennium of Macroeconomic Data for the UK}) y del Bureau of Labor Statistics, vía FRED (libro de datos del tema).}
\end{minipage}\end{center}

Durante siglo y medio, las subidas de precios británicas y americanas habían sido rachas ligadas a guerras y malas cosechas, seguidas de deflaciones que devolvían el nivel: variaciones que oscilan alrededor de cero. Desde mediados de los años sesenta la serie cambia de naturaleza: la inflación pasa a ser positiva, persistente y creciente, y la formación de los precios se convierte en la pregunta obligada. La respuesta de la Síntesis fue la curva de \authorfont{PHILLIPS} (1958): la evidencia de un siglo de datos británicos de que la variación de los salarios nominales se relaciona inversa y no linealmente con la tasa de paro, $\dot{W}/W=f(U)$ con $f'<0$, fundamentada por \authorfont{LIPSEY} (1960) y transformada por \authorfont{SAMUELSON} y \authorfont{SOLOW} (1960) en una relación entre inflación y paro para Estados Unidos (derivación, ajuste y cifras, en el Anexo \ref{anx:aparato}). La escuela la adoptó como su \textit{ecuación perdida}: la pieza que faltaba para repartir entre precios y cantidades los efectos de la demanda.

\begin{center}\begin{minipage}{\linewidth}\centering
{\small\textbf{El menú de \authorfont{SAMUELSON} y \authorfont{SOLOW}: la inflación como precio del empleo}}\par\vspace{0.3em}
\includegraphics{b1_menu.pdf}\par\vspace{0.3em}
{\small\textit{Fuente: }Elaboración propia a partir de \authorfont{SAMUELSON} y \authorfont{SOLOW} (1960).}
\end{minipage}\end{center}

Y la leyó como se leen las herramientas: como un \textbf{menú de política económica}. Si la curva es estable, la sociedad puede elegir su combinación de inflación y paro, pagar algo de inflación por acercarse al pleno empleo (el punto $B$) o algo de paro por la estabilidad de precios (el punto $A$), y la política de demanda ejecuta la elección. Ese salto, de regularidad empírica a menú estable explotable, es la apuesta intelectual más arriesgada de la Síntesis, y los propios autores habían dejado por escrito la advertencia de que la curva podía desplazarse. Sobre esa apuesta, y sobre la expectativa estática que la sostenía, golpeará el monetarismo; el duelo se libra más adelante.

\clearpage
\section{El Monetarismo: la contrarrevolución}\label{b2}
\puntosclave{
El Monetarismo, encabezado por FRIEDMAN desde Chicago, critica a la Síntesis en su propio terreno y con su mismo instrumental. Sus creencias: el dinero es el principal determinante de la demanda global (money matters), el sector privado es intrínsecamente estable y el gobierno estabiliza mal (información escasa, retardos largos y variables, incentivos torcidos).

Por el lado de la demanda, la renta permanente y la demanda de dinero estable invierten el veredicto de la Síntesis: la política fiscal pierde su multiplicador (lo transitorio se ahorra) y las variaciones de la cantidad de dinero dominan la renta nominal.

Por el lado de la oferta, la curva de PHILLIPS aumentada por las expectativas sustituye el menú por una tasa natural de paro y la tesis aceleracionista: no hay intercambio explotable entre inflación y paro a largo plazo. La estanflación de los setenta confirmó la predicción.

Implicación de política: reglas frente a discrecionalidad, con la regla del k por ciento como emblema, control de cantidades y tipos de cambio flexibles; nada de ajuste fino.
}

El contexto de la escuela es el mismo que el de la Síntesis, los años posteriores a la Segunda Guerra Mundial, y precisamente el fuerte crecimiento de aquellos años, gobernados por el paradigma rival, la mantuvo en posición secundaria: sus postulados circulaban en los círculos académicos, pero no cobraron peso público hasta los shocks de oferta de la década de 1970, ante la incapacidad del modelo de la Síntesis de explicar de manera satisfactoria la estanflación. \authorfont{H. JOHNSON} (1971) acuñó para el episodio el nombre que ha quedado: una \textit{contrarrevolución}, dispuesta a discutir la práctica totalidad de las tesis del keynesianismo de la Síntesis utilizando su mismo marco teórico de referencia.

\subsection{Chicago: ¿qué creían los monetaristas?}

El término \textit{monetarismo}, introducido por \authorfont{BRUNNER} (1968), condensa la primera creencia en una expresión: \textit{money matters}. El dinero es el principal determinante de la demanda global, y las raíces de esa convicción están en la teoría cuantitativa, pilar de la teoría monetaria clásica con formulaciones que se remontan a Azpilcueta, Bodin o Hume ([\authorfont{Ver Tema 3.A.3}]): a largo plazo la velocidad de circulación varía poco y el producto real lo determinan factores reales, de modo que los cambios monetarios acaban en los precios, y la correlación secular entre dinero y precios respalda la frase que \authorfont{FRIEDMAN} convirtió en lema: \textit{la inflación es siempre y en todo lugar un fenómeno monetario} (la reconstrucción completa de esta herencia, de \authorfont{FISHER} y Cambridge a la reformulación de 1956, se desarrolla en el Anexo \ref{anx:contra}).

La segunda creencia es la \textbf{estabilidad intrínseca de la economía de libre mercado}: las fluctuaciones graves no nacen del sector privado, sino de la gestión del dinero. La tercera completa el credo: el gobierno estabiliza mal, porque sabe poco (no conoce el reparto de sus impulsos entre precios y cantidades), llega tarde (los retardos de sus instrumentos son largos y variables) y quiere otra cosa (está sometido al ciclo electoral). El énfasis se desplaza así de las limitaciones del mercado, el terreno de la Síntesis, a las \textbf{limitaciones de la intervención del Estado}. En el método, en cambio, no hay ruptura: los monetaristas critican a la Síntesis en su mismo terreno y con su mismo lenguaje, los componentes de la demanda agregada, la demanda de dinero, el IS-LM y la curva de \authorfont{PHILLIPS}, aunque con un estilo empírico propio, el enfoque marshalliano de ecuaciones simples pegadas a los datos frente a los grandes sistemas estructurales (la metodología de \authorfont{FRIEDMAN}, su falsacionismo y la tesis de la irrelevancia de los supuestos se desarrollan en el Anexo \ref{anx:autores} y en [\authorfont{Ver Tema 3.A.1}]).\footnote{%
    Serán los sucesores, la Nueva Macroeconomía Clásica, quienes repudien la mayor parte del instrumental keynesiano y vuelvan a conceptos puramente neoclásicos ([\authorfont{Ver Tema 3.A.6}]). La cabeza visible de la escuela fue \authorfont{FRIEDMAN}, Nobel en 1976; su perfil, el círculo de Chicago y su faceta de divulgador político se tratan en el Anexo \ref{anx:autores}.
} El credo cabe en siete proposiciones:
\begin{lnum}
    \item La flexibilidad de precios y salarios es una aproximación aceptable para modelizar la realidad;
    \item Los individuos deciden su consumo en función de su renta permanente, no de la corriente;
    \item La demanda de dinero es estable;
    \item Las fluctuaciones de la cantidad de dinero provocan las fluctuaciones de la demanda a corto plazo (y no a la inversa), y son la causa principal de los ciclos ([\authorfont{Ver Tema 3.A.42}]);
    \item La inflación la causan únicamente los aumentos de la oferta monetaria por encima de lo que exige el crecimiento de la renta; se rechazan las explicaciones por presión de costes;
    \item El mercado de trabajo está en equilibrio, con una tasa natural de desempleo; y,
    \item La política monetaria debe ser reglada.
\end{lnum}

\subsection{La demanda: ¿por qué importa el dinero y la política fiscal no?}

La artillería analítica empieza por la función de consumo. Para \authorfont{FRIEDMAN} (1957), el consumidor racional planifica sobre su \textbf{renta permanente} $Y_p$, el flujo que puede mantener indefinidamente sin menoscabar su riqueza, y deja pasar lo transitorio:

$$C_p=k\!\left(r,\ \frac{W_H}{W},\ \theta\right)\cdot Y_p$$

donde $k$ depende del tipo de interés, del peso de la riqueza humana y de las preferencias (construcción completa y reconciliación de la evidencia, en el Anexo \ref{anx:contra}). La implicación de política es inmediata y demoledora: si los hogares solo consumen lo permanente, el multiplicador keynesiano se queda sin materia prima.

\begin{center}\begin{minipage}{\linewidth}\centering
{\small\textbf{El consumo responde a la renta permanente: el impulso transitorio se ahorra}}\par\vspace{0.3em}
\includegraphics{b2_consumo.pdf}\par\vspace{0.3em}
{\small\textit{Fuente: }Elaboración propia a partir de \authorfont{FRIEDMAN} (1957).}
\end{minipage}\end{center}

Un impulso fiscal transitorio desplaza la renta corriente, pero el consumo apenas se mueve a lo largo de la recta corta ($E\to T$), lejos del punto $P$ que el multiplicador esperaba sobre el rayo de largo plazo: \textbf{la renta transitoria se ahorra}, la tesis del estancamiento queda desacreditada y la política fiscal de ajuste fino pierde su palanca antes incluso de contar la expulsión financiera. A cambio, el consumo queda enganchado a la riqueza ($Y_p=r\,W$), y por esa puerta entran los \textbf{efectos riqueza} que amplifican a la política monetaria y debilitan más aún a la fiscal financiada con deuda (el detalle sobre el IS-LM, en el Anexo \ref{anx:contra}).

La segunda pieza es la \textbf{demanda de dinero}, reformulada en 1956 como la demanda de un activo más dentro de la cartera:

$$\frac{M^{d}}{P}=g(r,\ \pi^{e})\cdot Y_p$$

Depende de la renta permanente, que es suave, y poco del tipo de interés: es, por tanto, \textbf{estable}, justo lo que la Síntesis negaba con su motivo especulación y su trampa de la liquidez. Con demanda estable, la velocidad es una función estable, no un residuo caprichoso, y la ecuación cuantitativa recupera su causalidad: las variaciones de $M$ se trasladan al gasto nominal, $\Delta M\to\Delta(P\,Y)$, a través de un mecanismo de transmisión mucho más ancho que el keynesiano, el reequilibrio de toda la cartera (bonos, acciones, bienes duraderos, capital), que estimula el gasto directamente y hace de los tipos de interés observables una guía engañosa del tono de la política (cadenas de transmisión comparadas y función completa, en el Anexo \ref{anx:contra}). En el tablero del IS-LM, la lectura de las pendientes se invierte punto por punto respecto a la de la Síntesis: inversión sensible al interés (IS tendida) y demanda de dinero insensible (LM empinada, vertical en el límite), con la expulsión total del gasto privado como resultado de la expansión fiscal ([\authorfont{Ver Anexo \ref{anx:aparato}}]).

La munición empírica la puso la \textit{Historia monetaria} (\authorfont{FRIEDMAN} y \authorfont{SCHWARTZ}, 1963): un siglo de datos en el que el dinero precede a la renta nominal y, como pieza central, la reinterpretación de la Gran Depresión como \textit{Gran Contracción}, con la cantidad de dinero cayendo más de un tercio ante la pasividad de la Reserva Federal. La lección invierte el mito fundacional del intervencionismo: la Depresión no probó la impotencia del dinero sino su poder, trágicamente ejercido en dirección contractiva ([\authorfont{Ver Tema 3.A.42}]).\footnote{%
    La batalla empírica completa, el duelo de correlaciones de \authorfont{FRIEDMAN} y \authorfont{MEISELMAN}, la ecuación de San Luis y las réplicas de \authorfont{TOBIN} (la causalidad inversa, la caja negra), se desarrolla en el Anexo \ref{anx:contra}.
}

\subsection{La oferta: ¿por qué el menú era una ilusión?}

El golpe decisivo fue contra la curva de \authorfont{PHILLIPS} leída como menú. El argumento de \authorfont{FRIEDMAN} (1968), anticipado de forma independiente por \authorfont{PHELPS} (1967), no discute los datos sino la lógica: \textbf{el menú exige ilusión monetaria sistemática}. Para sostener indefinidamente un paro inferior al de equilibrio hace falta que los trabajadores crean cobrar un salario real alto mientras cobran uno bajo, es decir, que se dejen engañar siempre; y un supuesto de irracionalidad permanente no puede fundar una política permanente. Lo que sí puede ocurrir es el engaño transitorio: los trabajadores negocian salarios nominales con una expectativa de precios que forman mirando al pasado, según la hipótesis de las expectativas adaptativas,

$$\pi^{e}_{t+1}=\pi^{e}_{t}+\lambda\,(\pi_{t}-\pi^{e}_{t})$$

y mientras la inflación efectiva supere a la esperada ofrecerán más trabajo del que querrían (mercado de trabajo con errores de predicción y equivalencia de la hipótesis con la memoria geométrica, en el Anexo \ref{anx:contra}).

\begin{center}\begin{minipage}{\linewidth}\centering
{\small\textbf{La demolición del menú: una curva por cada expectativa y una vertical al fondo}}\par\vspace{0.3em}
\includegraphics{b2_phillips.pdf}\par\vspace{0.3em}
{\small\textit{Fuente: }Elaboración propia a partir de \authorfont{FRIEDMAN} (1968) y \authorfont{PHELPS} (1967).}
\end{minipage}\end{center}

La curva de \authorfont{PHILLIPS} sobrevive, pero aumentada con el término omitido, $\pi_t=\pi^{e}_{t}-\alpha\,(U_t-U_n)$: hay una curva de corto plazo por cada nivel de expectativas y el intercambio solo existe mientras dura la sorpresa. La expansión lleva la economía de $A$ a $B$; cuando las expectativas alcanzan a la inflación, la curva entera sube y el paro vuelve en $C$ a la \textbf{tasa natural} $U_n$, que depende solo de factores reales (fricciones, búsqueda, estructura del mercado de trabajo) y es inmune a la política de demanda. Repetir la jugada exige acelerar: es la \textbf{tesis aceleracionista de la inflación}, $\Delta\pi_t=\alpha\,(U_n-U_t)$, y a largo plazo la curva es vertical. La década siguiente hizo el contraste fuera de muestra: la combinación de inflación y paro crecientes que hundió a la Síntesis, imposible sobre el menú, es exactamente lo que la curva aumentada predice para una demanda sobreestimulada cuyo público ha dejado de dejarse sorprender (tasa natural frente a NAIRU, y el caso de inflación alta con pendiente positiva, en el Anexo \ref{anx:contra}).

\subsection{La política: ¿por qué reglas y no discreción?}

Las conclusiones de política convergen desde cuatro direcciones. Primera, la \textbf{ignorancia}: no se conoce la ecuación que reparte a corto plazo una expansión entre precios y renta real, la \textit{ecuación perdida}, de modo que toda sorpresa monetaria es un experimento a ciegas. Segunda, la \textbf{tasa natural}: no hay nada permanente que ganar por el lado real. Tercera, los \textbf{retardos largos y variables} de la propia política monetaria, que convierten el activismo en una fuente probable de ciclo, la medicina administrada a destiempo. Y cuarta, la \textbf{economía política}: una autoridad discrecional sometida a elecciones explotará una y otra vez la curva de corto plazo.\footnote{%
    El argumento, enunciado de manera informal, lo formalizará la literatura de la inconsistencia temporal como sesgo inflacionista de la política discrecional ([\authorfont{Ver Tema 3.A.6}]). Los retardos, el álgebra de la estabilización de \authorfont{FRIEDMAN} (1953) y el antecedente de 1948 se desarrollan en el Anexo \ref{anx:contra}.
} La respuesta es sustituir la discreción por una \textbf{regla}: el crecimiento constante de la cantidad de dinero, el \textit{k por ciento} propuesto en \textit{A Program for Monetary Stability} (\authorfont{FRIEDMAN}, 1960), instrumentado controlando cantidades, no tipos de interés, mediante operaciones de mercado abierto; y, como condición exterior de la regla, \textbf{tipos de cambio flexibles} que devuelvan a cada país el control de su dinero, el argumento de \authorfont{FRIEDMAN} (1953) contra Bretton Woods que la historia terminó atendiendo en 1971-1973 ([\authorfont{Ver Tema 3.B.20}]). El resto es intervención mínima: confianza en la estabilidad del sector privado y vuelta al \textit{laissez-faire} en todo lo que no sea la provisión de un marco monetario estable (los experimentos de los años ochenta, el gradualismo frente al choque y el paradójico final de la doctrina se tratan en el Anexo \ref{anx:autores}).

El impacto desborda el marcador del duelo. Al monetarismo se le debe haber resucitado el mercado monetario y la curva LM, arrinconados en los años cuarenta y cincuenta por los multiplicadores y aceleradores; haber devuelto al primer plano el lado de la oferta y el mercado de trabajo; y haber incorporado a la macroeconomía los efectos riqueza y un abanico amplio de activos en la demanda de dinero, piezas todas de la disciplina actual. Tampoco fue el único asaltante: en los mismos años sesenta y setenta la Síntesis recibía críticas desde el propio campo keynesiano, en particular de los neokeynesianos del desequilibrio ([\authorfont{Ver Tema 3.A.4}]). Quién ganó qué, exactamente, se dirime a continuación.

\clearpage
\section{El duelo: dos sistemas de creencias frente a frente}\label{b3}
\puntosclave{
El duelo nunca fue por las herramientas, que ambas escuelas compartían, sino por dos parejas de creencias: mercado inestable y gobierno capaz (Síntesis) frente a mercado estable y gobierno torpe (monetarismo). Las elasticidades del IS-LM son esas creencias cuantificadas, y las guerras empíricas de los sesenta no bastaron para zanjarlas.

La estanflación de los setenta arbitró a favor de la crítica monetarista (tasa natural, aceleracionismo); los ochenta humillaron su receta operativa (inestabilidad de la demanda de dinero, ley de GOODHART). El consenso resultante es monetarista en los fines (estabilidad de precios, independencia, reglas) y keynesiano en las urgencias.

Las herencias cierran el mapa: la Nueva Macroeconomía Clásica lleva el monetarismo al límite de las expectativas racionales; la NEK-2 funde las rigideces keynesianas microfundamentadas con la tasa natural y las reglas: una nueva síntesis.

Vistas desde fuera, ambas escuelas comparten lo esencial (racionalidad, equilibrio, mundo ergódico, la distribución como dato técnico): ahí apunta la crítica heterodoxa.
}

Visto de cerca, el enfrentamiento tiene algo de pelea de familia: los contendientes comparten el modelo, el lenguaje, la fe en la medición y hasta los axiomas. Lo que los separa es anterior a toda ecuación, dos imágenes del mundo: una economía de mercado propensa a atascarse que un Estado competente puede gobernar, o una economía de mercado intrínsecamente estable a la que un Estado torpe solo puede estorbar. Este bloque enfrenta las dos imágenes pieza a pieza, deja que la historia arbitre y reparte las herencias.

\subsection{El tablero común: ¿qué deciden las elasticidades?}

El mismo modelo contiene los dos mundos: basta girar los parámetros. Con la IS inelástica y la LM elástica de la Síntesis, la política fiscal rinde ($E_0\to E_F$) y la monetaria apenas ($E_0\to E_M$); con la IS elástica y la LM inelástica del monetarismo, la monetaria manda y la fiscal solo sustituye gasto privado por público.

Pero la estimación no la zanjó, porque cada bando midió con su método (sistemas estructurales contra ecuaciones reducidas) y encontró lo que su método podía encontrar, y porque bajo las pendientes había una segunda capa menos medible: la anchura del mecanismo de transmisión y los efectos riqueza, que hacen potente a la política monetaria incluso donde la geometría keynesiana la da por muerta (las batallas empíricas y la guerra de métodos se desarrollan en los Anexos \ref{anx:contra} y \ref{anx:autores}). \subsection{El árbitro: ¿qué decidió la estanflación?}

La historia arbitró en dos asaltos de signo contrario. El primero lo ganó el monetarismo por la vía regia de la ciencia: una predicción incómoda formulada en 1967-1968, cuando el menú vivía su apogeo, y confirmada fuera de muestra por la década siguiente. La persistencia inflacionaria que ya asomaba en las series de precios desembocó, tras los shocks del petróleo, en la combinación de inflación y paro crecientes que el menú declaraba imposible y que la curva aumentada predice con naturalidad; la Síntesis se quedó sin ecuación y sin relato, el Nobel de 1976 consagró al vencedor y el giro práctico llegó con \authorfont{VOLCKER} y \authorfont{THATCHER}. El segundo asalto, menos contado, lo perdió el vencedor: en cuanto los agregados monetarios se convirtieron en objetivo oficial, la demanda de dinero estable dejó de serlo (el \textit{dinero desaparecido}, la ley de \authorfont{GOODHART}) y a mediados de los ochenta la receta operativa estaba abandonada (el relato completo de los experimentos, en el Anexo \ref{anx:autores}).

El saldo es un consenso híbrido que conviene saber descomponer: \textbf{monetarista en los fines y en el marco} (la estabilidad de precios como prioridad, bancos centrales independientes del ciclo electoral, preferencia por reglas, la tasa natural como frontera de la demanda) y \textbf{keynesiano en las urgencias} (gestión activa de la demanda cuando la recesión es profunda y el tipo de interés toca suelo). La frontera entre reglas y discrecionalidad no quedó cerrada: es constitutiva, porque resume la pregunta previa de a quién se teme más, al mercado o al gobierno, y cada gran crisis la reabre.\footnote{%
    La Gran Recesión de 2008 lo ilustra: tipos en cero, retorno masivo de la política fiscal y un debate sobre reglas, discrecionalidad y estímulo que reprodujo casi literalmente el duelo de este tema ([\authorfont{Ver Tema 3.A.7}]; [\authorfont{Ver Tema 3.A.42}]).
}

\subsection{Las herencias: ¿quién ganó la posguerra intelectual?}

Las dos escuelas sobreviven en sus hijos. Del monetarismo desciende la \textbf{Nueva Macroeconomía Clásica}, que lo lleva al límite lógico: si los errores de expectativas no pueden ser sistemáticos, la hipótesis adaptativa es una irracionalidad más, las expectativas se vuelven racionales y ni siquiera el corto plazo sobrevive a la política anticipada; la crítica de \authorfont{LUCAS} remata, de paso, a los modelos estructurales de la Síntesis ([\authorfont{Ver Tema 3.A.6}]).\footnote{%
    La crítica heterodoxa lee este paso como el cierre del último resquicio: si con \authorfont{FRIEDMAN} la política solo funcionaba mientras engañaba, con las expectativas racionales ni eso, y el mensaje implícito es aceptar lo dado (\authorfont{ASTARITA}, 2011). Es una lectura ideológica del resultado, no del teorema, pero señala bien qué se juega en el supuesto de expectativas.
} De la Síntesis desciende la \textbf{Nueva Economía Keynesiana}, que acepta el esqueleto de equilibrio, la tasa natural y las reglas, y microfundamenta las rigideces que la vieja escuela se limitaba a suponer; su fusión con el instrumental de los nuevos clásicos en la \textit{nueva síntesis neoclásica} repite, una generación después y con otro utillaje, el gesto integrador de 1937 ([\authorfont{Ver Tema 3.A.7}]). El mapa del principio del tema queda así cerrado: el monetarismo es la bisagra entre las dos síntesis.

Queda lo que el duelo nunca discutió, y que se ve mejor desde fuera. Ambas escuelas comparten la racionalidad optimizadora, el equilibrio como estado de referencia, el mundo ergódico cuyo largo plazo no depende del camino y el tratamiento de la tasa natural y de la distribución marginalista de la renta como datos técnicos, no como resultados institucionales. Sobre ese perímetro común carga la crítica heterodoxa: la postkeynesiana, con la incertidumbre radical que ya apareció antes; y la marxista, para la que presentar la producción y la distribución como regidas por leyes eternas, independientes de la historia, equivale a introducir subrepticiamente las relaciones sociales existentes como inmutables (\authorfont{MARX}, 1859), con lecturas contemporáneas que ven en la propia tasa natural un dispositivo que naturaliza el desempleo disciplinante (\authorfont{ASTARITA}, 2011).\footnote{%
    El desarrollo de estas escuelas y de su crítica corresponde a [\authorfont{Ver Tema 3.A.4}]; aquí cumplen la función de marcar el perímetro del tablero: todo el duelo de este tema se jugó dentro de él.
} Para el conjunto del temario, esa es la utilidad final de este tema: las creencias aquí enfrentadas, sobre la estabilidad del mercado, la capacidad del gobierno y la forma de las expectativas, son exactamente las variables que las escuelas de los temas siguientes recombinan.

\anexo{El aparato analítico de la Síntesis: el IS-LM(-BP) y la curva de \authorfont{PHILLIPS}}\label{anx:aparato}

Este anexo es el manual del instrumental que la Síntesis Neoclásica construyó para pensar y gobernar la demanda agregada: parte de las funciones de comportamiento más elementales y levanta con ellas, paso a paso, el modelo IS-LM de economía cerrada, su estática comparativa y sus multiplicadores, la extensión a la economía abierta de \authorfont{MUNDELL} y \authorfont{FLEMING}, el cierre nominal del modelo (la demanda agregada, las tres curvas de oferta y el efecto saldos reales que sostiene teóricamente la palabra \textit{síntesis}) y, por último, la curva de \authorfont{PHILLIPS} en su versión original, la \textit{ecuación perdida} que completó el sistema. El cuerpo del tema presenta estas piezas y las usa para entender el sistema de creencias de la escuela; aquí se construyen enteras, de la definición más básica a la derivación completa, de modo que el lector pueda reproducir cualquier resultado desde cero. La demolición monetarista de este aparato se desarrolla en el Anexo II.

\subsection{Del sistema de \authorfont{KEYNES} al diagrama de \authorfont{HICKS} (1937)}\label{anx1:hicks}

La primera pieza del aparato no es una ecuación, sino una traducción. La \textit{Teoría General} (1936) es una obra ensayística, sin un modelo formal explícito, y en los meses posteriores a su publicación varios economistas emprendieron en paralelo la misma tarea: extraer de ella un sistema de ecuaciones que permitiera compararla con la teoría que pretendía desbancar. De esa tarea, emprendida simultáneamente por \authorfont{MEADE} (1937), \authorfont{HARROD} (1937) y \authorfont{HICKS} (1937),\footnote{%
    Los tres trabajos se presentaron en el encuentro de la Econometric Society celebrado en Oxford en septiembre de 1936\textcolor{orange}{\textsuperscript{[Probable]}} y se publicaron al año siguiente. \authorfont{MEADE} formuló la determinación del nivel de empleo en forma algebraica (\textit{A Simplified Model of Mr. Keynes' System}); \authorfont{HARROD} subrayó la determinación conjunta del tipo de interés; la versión que hizo fortuna fue la de \authorfont{HICKS}, porque añadió a las ecuaciones un instrumento gráfico.
} salió el artículo más influyente de la historia de la macroeconomía: \textit{Mr. Keynes and the Classics; A Suggested Interpretation}.

\subsubsection{El problema: qué teoría clásica se está refutando}

El punto de partida de \authorfont{HICKS} es una objeción metodológica: \authorfont{KEYNES} dirige su ataque contra la \textit{teoría clásica} tomando como referente la teoría del mercado de trabajo de \authorfont{PIGOU} (\textit{The Theory of Unemployment}, 1933),\footnote{%
    La interpretación que el propio \authorfont{KEYNES} hace de la obra de \authorfont{PIGOU}, aclarada, puede leerse en el \href{https://www.marxists.org/reference/subject/economics/keynes/general-theory/ch19a.htm}{apéndice al capítulo 19 de la \textit{Teoría General}}.
} pero ese referente, argumenta \authorfont{HICKS}, no es representativo: muchos economistas neoclásicos habrían analizado el mercado de trabajo en términos de salarios nominales dados (variable exógena) y no de salarios reales determinados dentro del sistema. Como en la literatura no existía un modelo clásico centrado en salarios nominales, \authorfont{HICKS} lo construye él mismo, para poder comparar ambas teorías sobre un terreno común: mismo horizonte (el corto plazo), mismo supuesto de salario nominal dado y misma pregunta (qué determina la renta y el empleo).

\subsubsection{Los dos sistemas de tres ecuaciones}

Con salario nominal dado, \authorfont{HICKS} reduce cada teoría a tres ecuaciones, escritas en su notación original, donde $I$ es la renta total, $I_x$ la inversión, $i$ el tipo de interés y $M$ la cantidad de dinero. La teoría clásica queda, según \textit{Mr. Keynes and the Classics} (\authorfont{HICKS}, 1937):

$$M=k\,I,\qquad I_x=C(i),\qquad I_x=S(i,I)$$

donde la primera ecuación es la teoría cuantitativa en su versión de Cambridge (la demanda de dinero es una proporción estable $k$ de la renta nominal), la segunda hace depender la inversión del tipo de interés y la tercera iguala inversión y ahorro, que depende del interés y de la renta. En este sistema la primera ecuación determina por sí sola la renta nominal: el dinero es lo esencial, y las otras dos ecuaciones reparten esa renta entre interés e inversión. La teoría de \authorfont{KEYNES}, en cambio, queda:

$$M=L(I,i),\qquad I_x=C(i),\qquad I_x=S(I)$$

Las diferencias son exactamente dos, y \authorfont{HICKS} las subraya: la demanda de dinero deja de ser proporcional a la renta e incorpora el tipo de interés (la \textit{preferencia por la liquidez}), y el ahorro deja de depender del tipo de interés para depender solo de la renta. Con ellas la lógica causal se invierte: la renta ya no la fija la ecuación monetaria, sino el encuentro entre ahorro e inversión, y el dinero influye a través del tipo de interés.

\subsubsection{El diagrama IS-LL y la generalización}

Para comparar los sistemas, \authorfont{HICKS} condensa cada uno en dos curvas sobre el plano renta-interés: la curva IS recoge las combinaciones de $I$ e $i$ que igualan ahorro e inversión (las ecuaciones segunda y tercera), y la curva LL las que equilibran el mercado de dinero (la primera). El equilibrio simultáneo es el punto de corte.

La aportación decisiva del diagrama es la forma de la curva LL: plana a la izquierda y creciente a la derecha. El tramo plano recoge la situación en la que el tipo de interés ha bajado tanto que el público prefiere atesorar cualquier cantidad adicional de dinero antes que comprar títulos (la elasticidad de la demanda de dinero al interés se hace infinita): es la \textit{trampa de la liquidez}. \authorfont{HICKS} localiza ahí la diferencia operativa entre las dos teorías: si la curva IS corta a la LL en su tramo creciente, el sistema se comporta de manera clásica y la expansión monetaria funciona; si la corta en el tramo plano, como cabía esperar en la Gran Depresión, el dinero adicional se atesora, la política monetaria pierde tracción y solo el desplazamiento de la IS (la política fiscal) puede elevar la renta. De ahí su conclusión, deliberadamente reductora: la \textit{Teoría General} sería, en rigor, la \textit{Economics of Depression}, la economía de la depresión, válida en ese tramo singular. En la parte final del artículo, \authorfont{HICKS} propone además el sistema generalizado $M=L(I,i)$, $I_x=C(I,i)$, $I_x=S(I,i)$, en el que caben como casos particulares tanto la teoría clásica como la keynesiana: el gesto que convierte una disputa doctrinal en una discusión sobre valores de parámetros, y que anticipa toda la estrategia intelectual de la Síntesis.

\subsubsection{Lo que el artículo deja fuera}

El esqueleto de 1937 no es todavía el modelo canónico. No contiene mercado de trabajo ni función de producción (el salario nominal es un dato y el empleo queda implícito), no distingue entre renta real y nominal porque los precios no se modelizan, y el propio \authorfont{HICKS} admitió décadas después las limitaciones del artefacto, en particular la incomodidad de mezclar en un mismo diagrama un equilibrio de flujo (el mercado de bienes) y uno de stock (el de dinero).\footnote{%
    \authorfont{HICKS} (1980-81), \textit{IS-LM: An Explanation}, en el \textit{Journal of Post Keynesian Economics}: el propio autor califica allí el diagrama de artefacto de aula que sacrifica el tiempo y la incertidumbre de la \textit{Teoría General}.\textcolor{orange}{\textsuperscript{[Probable]}}
} Cerrar el modelo por el lado del trabajo y de los precios fue la contribución de \authorfont{MODIGLIANI}.

\subsection{\authorfont{MODIGLIANI} (1944): el mercado de trabajo y el cierre del modelo}\label{anx1:modigliani}

El artículo \textit{Liquidity Preference and the Theory of Interest and Money} (\authorfont{MODIGLIANI}, 1944, en \textit{Econometrica}) completa el sistema de \authorfont{HICKS} con las piezas que le faltaban: una función de producción, un mercado de trabajo y una regla de formación del salario. Con ellas el modelo puede por fin pronunciarse sobre la pregunta central del debate: ¿de qué depende que exista desempleo en equilibrio?

La respuesta de \authorfont{MODIGLIANI} distingue dos regímenes. Si el salario nominal es flexible y los mercados se vacían, el sistema completo reproduce los resultados clásicos: el empleo lo determina el mercado de trabajo, la producción la función de producción, y el dinero solo fija la escala nominal (los precios), con una única excepción, la trampa de la liquidez, en la que ni siquiera la flexibilidad salarial restaura el pleno empleo porque el tipo de interés no puede bajar hasta el nivel requerido. Si el salario nominal es rígido a la baja, en cambio, el nivel de empleo pasa a depender de la demanda agregada y, a través de ella, de la cantidad de dinero: la política monetaria tiene efectos reales, y el desempleo es compatible con un equilibrio estable del sistema. De ahí la lectura que la literatura posterior consagró: el desempleo keynesiano sería, en lo esencial, un fenómeno de rigidez salarial, y la economía keynesiana, el caso particular de una teoría clásica más general, con la trampa de la liquidez como única excepción pura.\footnote{%
    La historiografía reciente matiza esa lectura: el propio \authorfont{MODIGLIANI} no presentó la rigidez salarial como una concesión para reconciliar a \authorfont{KEYNES} con los clásicos, sino como un supuesto realista y un dispositivo analítico para mostrar el funcionamiento de una economía monetaria, en la que las variables reales dependen de la cantidad de dinero, y para evitar que el \textit{caso keynesiano} degenerase en una deflación indefinida. El debate sobre la génesis y la intención del artículo se desarrolla en el Anexo \ref{anx:autores}.
} El artículo pasó así a la literatura como la contribución decisiva, junto a la de \authorfont{HICKS}, a la construcción del modelo IS-LM canónico y como un primer paso hacia la restauración de la visión prekeynesiana.

La difusión del aparato fue obra, sobre todo, de \authorfont{HANSEN}, el gran introductor de \authorfont{KEYNES} en Estados Unidos, que en \textit{Monetary Theory and Fiscal Policy} (1949) y \textit{A Guide to Keynes} (1953) popularizó el diagrama en la forma y con los rótulos que lo harían universal (de ahí que el modelo se conozca también como diagrama de \authorfont{HICKS}-\authorfont{HANSEN}), y de los manuales que lo adoptaron como lengua franca de la macroeconomía de posguerra.

\subsection{El modelo canónico de economía cerrada}\label{anx1:canonico}

Esta sección construye el modelo en su forma de manual: primero cada mercado por separado, con sus funciones de comportamiento y la curva de equilibrio que genera; después el equilibrio conjunto, resuelto analíticamente, y los multiplicadores que gobiernan toda la discusión de política económica; por último, los dos casos extremos que delimitan el campo de batalla doctrinal. La notación es la moderna: $Y$ renta real, $r$ tipo de interés, $P$ nivel de precios (fijo en el corto plazo del modelo), $M$ oferta monetaria.\footnote{%
    Aunque el IS-LM es en esencia un modelo de demanda, descansa en dos supuestos implícitos de oferta que acotan su validez al corto plazo: un stock de capital fijo y unos salarios nominales exógenos, no determinados por los mecanismos de ajuste del propio sistema. Con ellos, la demanda determina la producción solo a corto plazo (\textit{demand-driven economy}); a largo plazo, levantados ambos, rige la Ley de \authorfont{SAY}.
}

\subsubsection{El mercado de bienes: la curva IS}

La curva IS recoge las combinaciones de renta y tipo de interés que equilibran el mercado de bienes, es decir, que igualan la producción a la demanda agregada planeada. Sus tres componentes, en la especificación lineal de referencia del modelo IS-LM canónico, son:

$$C=C_0+c\,(Y-T),\qquad I=I_0-b\,r,\qquad G=\overline{G}$$

El consumo depende de un componente autónomo $C_0$ y de la renta disponible $Y-T$, con una propensión marginal a consumir $0<c<1$: ante un aumento de la renta, el consumo aumenta en proporción menor, que es el contenido de la ley psicológica fundamental de \authorfont{KEYNES}; esa sensibilidad del consumo a la renta corriente es, además, la puerta por la que la política fiscal afecta a la demanda. La inversión tiene un componente autónomo $I_0$, dominado por el estado de las expectativas empresariales (los \textit{animal spirits}, que la Síntesis considera el determinante más inestable de la demanda), y responde negativamente al tipo de interés con sensibilidad $b$. El gasto público $G$ es discrecional. La condición de equilibrio $Y=C+I+G$ define la curva:

$$IS:\quad Y=C_0+c\,(Y-T)+I_0-b\,r+\overline{G}$$

Despejando, la pendiente es $dr/dY\rvert_{IS}=-(1-c)/b<0$: la curva es decreciente porque un tipo de interés menor estimula la inversión y, a través del multiplicador, la renta de equilibrio. Es tanto más inclinada (rígida) cuanto menor es la sensibilidad de la inversión al interés ($b$) y menor la propensión a consumir; tanto más plana cuanto mayores son ambas. La posición de la curva la fijan los componentes autónomos: aumentos de $G$, de $I_0$ o de $C_0$, o bajadas de $T$, la desplazan a la derecha.

\subsubsection{El mercado de dinero: la curva LM}

La curva LM recoge las combinaciones de renta y tipo de interés que equilibran el mercado de dinero. La demanda de dinero del modelo traduce los motivos de \authorfont{KEYNES}: los motivos transacción y precaución la ligan positivamente a la renta (parámetro $k$) y el motivo especulación la liga negativamente al tipo de interés (parámetro $h$), porque el interés es el coste de oportunidad de mantener dinero en lugar de bonos.\footnote{%
    La introducción explícita del tipo de interés en la función de demanda de dinero es la huella directa de \authorfont{KEYNES} en el modelo: lo que se renuncia a ganar por no invertir los saldos en bonos es el precio de la liquidez. En la teoría cuantitativa clásica, $M=k\,P\,Y$, esa variable sencillamente no aparecía ([\authorfont{Ver Tema 3.A.3}]).
} Con oferta monetaria $M$ y precios $P$ dados, el equilibrio exige que la oferta de saldos reales iguale a su demanda, según la formulación canónica del modelo IS-LM:

$$LM:\quad \frac{M}{P}=k\,Y-h\,r$$

La pendiente, $dr/dY\rvert_{LM}=k/h>0$, es creciente: si la renta aumenta, la demanda de dinero por transacciones sube y, con una oferta dada, el tipo de interés debe subir para que la demanda especulativa libere los saldos necesarios. La curva es tanto más plana cuanto mayor es $h$ (una demanda de dinero muy sensible al interés absorbe los ajustes con variaciones mínimas de $r$) y tanto más vertical cuanto menor es $h$. Aumentos de $M$, o caídas de $P$, la desplazan a la derecha.

Como muestra el panel derecho, la versión completa de la curva hereda de \authorfont{HICKS} sus tres tramos: uno keynesiano, plano, al nivel mínimo del tipo de interés (la trampa de la liquidez, donde $h\to\infty$); uno intermedio, creciente; y uno clásico, vertical, donde la demanda especulativa se ha agotado y todo el dinero se demanda por transacciones ($h\to 0$). En qué tramo corta la IS deja de ser un detalle gráfico: es, como se verá, la frontera entre mundos doctrinales.

\subsubsection{El equilibrio y la forma reducida: los multiplicadores}\label{anx1:multiplicadores}

El corte de las dos curvas determina simultáneamente la renta y el tipo de interés: el tipo de interés, al ser determinante a la vez de la demanda de dinero y de la inversión, conecta ambos mercados y rompe la dicotomía clásica entre lo real y lo monetario. Al ser dos ecuaciones lineales con dos incógnitas, el sistema se resuelve por sustitución: despejando $r$ de la LM, $r=(k\,Y-M/P)/h$, y llevándolo a la IS, se obtiene la forma reducida de la renta de equilibrio del modelo IS-LM canónico:

$$Y^{*}=\frac{C_0-c\,T+I_0+\overline{G}+\dfrac{b}{h}\,\dfrac{M}{P}}{1-c+\dfrac{b\,k}{h}}$$

De la forma reducida salen los dos multiplicadores que organizan toda la discusión de política económica:

$$\frac{dY}{dG}=\frac{1}{1-c+\dfrac{b\,k}{h}}\qquad\text{y}\qquad \frac{dY}{d(M/P)}=\frac{b/h}{1-c+\dfrac{b\,k}{h}}$$

El multiplicador fiscal es el del aspa keynesiana, $1/(1-c)$, amortiguado por el término $b\,k/h$: la expansión eleva la demanda de dinero, el tipo de interés sube y expulsa inversión privada. El multiplicador monetario depende del cociente $b/h$: la política monetaria es eficaz si la inversión responde al interés ($b$ alto) y la demanda de dinero no lo absorbe todo ($h$ bajo). Los dos casos límite condensan medio siglo de debate: si $h\to\infty$, el multiplicador fiscal recupera su valor máximo $1/(1-c)$ y el monetario se anula (trampa de la liquidez); si $h\to 0$, el fiscal se anula y el monetario se hace $1/k$, es decir, $\Delta Y=\Delta(M/P)/k$, la teoría cuantitativa en estado puro. Toda la controversia entre keynesianos y monetaristas sobre la eficacia relativa de las políticas cabe, formalmente, en una discusión sobre los valores de $b$ y $h$.

\subsubsection{La política fiscal: multiplicador, expulsión y financiación}

Una expansión fiscal desplaza la IS a la derecha: para cada tipo de interés, la renta que equilibra el mercado de bienes es mayor. El desplazamiento horizontal es el efecto multiplicador completo del aspa, $\Delta G/(1-c)$; el equilibrio final queda por debajo porque el mayor gasto eleva la demanda de dinero y, con oferta monetaria dada, el tipo de interés, que contrae la inversión privada. Esa diferencia entre el efecto a tipo constante y el efecto final es la expulsión o \textit{crowding out}, que el modelo permite descomponer: la expulsión transaccional, por el aumento de la demanda de dinero que acompaña al mayor gasto, y la expulsión financiera, porque el déficit financiado con deuda obliga al mercado a absorber un exceso de oferta de bonos, cuyo precio baja, es decir, cuyo tipo de interés sube.\footnote{%
    El precio de un bono y su tipo de interés son las dos caras de la misma magnitud: si el bono perpetuo paga un cupón $c_B$, su precio es $c_B/r$, de modo que absorber más bonos a cupón dado exige un precio menor y, por tanto, un tipo mayor.
}

La financiación del gasto no es neutral, y el modelo ordena los tres casos. Si el gasto se financia con impuestos, opera el teorema de \authorfont{HAAVELMO} (1945): incluso con presupuesto equilibrado, el aumento del gasto es expansivo, porque el multiplicador del gasto supera al de los impuestos (una parte de la renta detraída por el impuesto habría ido al ahorro, mientras que el gasto público entra entero en la demanda); en el aspa keynesiana el multiplicador del presupuesto equilibrado es exactamente la unidad, y en el IS-LM, positivo pero menor, por la respuesta del tipo de interés. Si se financia con deuda, a la expulsión transaccional se suma la financiera. Si se financia con creación de dinero (monetización), el aumento de la demanda de dinero queda absorbido por la mayor oferta, el tipo de interés no sube y la expansión alcanza su efecto máximo: bajo supuestos keynesianos, el déficit monetizado es más expansivo que el financiado con deuda.

\subsubsection{La política monetaria: la transmisión keynesiana}

Una expansión monetaria desplaza la LM a la derecha: con más saldos reales, cada nivel de renta es compatible con un tipo de interés menor. La cadena de transmisión es indirecta y pasa entera por el tipo de interés: el exceso de saldos se emplea en comprar bonos, el precio de los bonos sube y el tipo baja, la inversión responde y el multiplicador hace el resto ($\Delta M \to \nabla r \to \Delta I \to \Delta Y$). La eficacia depende por tanto de dos eslabones que pueden fallar: que el tipo de interés baje (no lo hace en la trampa de la liquidez) y que la inversión responda (no lo hace si $b$ es pequeño, como la Síntesis tendía a creer). Esta concepción estrecha del mecanismo de transmisión, un único precio relevante, el de los bonos, será el blanco central de la crítica monetarista, que opone a ella un ajuste de carteras sobre todo el espectro de activos (Anexo II).

\subsubsection{Los casos extremos: trampa de la liquidez y caso clásico}\label{anx1:extremos}

Los dos límites del multiplicador tienen su traducción gráfica y su carga doctrinal. En la trampa de la liquidez (corte en el tramo plano de la LM), la expansión monetaria es impotente: la nueva liquidez se atesora en lugar de gastarse, porque al tipo mínimo el público prefiere dinero a bonos en cualquier cantidad; la política fiscal, en cambio, despliega su multiplicador completo sin subir el tipo de interés ni expulsar inversión. \authorfont{HICKS} señaló que la posibilidad de esta situación es el supuesto genuinamente distintivo de una economía keynesiana frente a una neoclásica, y que caracteriza las crisis profundas: de ahí que la política fiscal sea, para la Síntesis, la herramienta de las depresiones. En el caso clásico (LM vertical), la demanda de dinero no responde al interés, el nivel de renta compatible con el equilibrio monetario es único y la política fiscal solo recompone la demanda: cada unidad de gasto público expulsa exactamente una de gasto privado (expulsión total), mientras que la política monetaria traslada íntegro su impulso a la renta nominal. Conviene retener este segundo caso: una LM vertical es la forma gráfica de la teoría cuantitativa, y el monetarismo se dejará representar, simplificadamente, como la convicción de que el mundo se parece más a ese tramo que al keynesiano (Anexo II y bloque final del cuerpo).

\subsection{La economía abierta: el modelo \authorfont{MUNDELL}-\authorfont{FLEMING}}\label{anx1:mf}

La extensión del IS-LM a una economía abierta, desarrollada en paralelo por \authorfont{MUNDELL} (1963) y \authorfont{FLEMING} (1962) desde el FMI,\footnote{%
    \authorfont{R. MUNDELL} (1932-2021), Premio Nobel en 1999 por este análisis y por la teoría de las áreas monetarias óptimas ([\authorfont{Ver Tema 3.B.16}]), y \authorfont{J. M. FLEMING} (1911-1976), entonces subdirector del departamento de estudios del FMI. Los perfiles de ambos se recogen en el Anexo \ref{anx:autores}.
} añade al modelo dos mercados: el de bienes incorpora exportaciones e importaciones, y aparece la cuenta financiera con el exterior. La demanda agregada pasa a depender también del tipo de cambio $e$ y de la renta exterior $Y^{*}$, y la curva IS se desplaza con las exportaciones netas. El equilibrio externo se recoge en una tercera curva.

\subsubsection{La curva BP y la movilidad del capital}

La balanza de pagos está en equilibrio cuando el saldo por cuenta corriente y las entradas netas de capital se compensan. Con exportaciones que dependen de la renta exterior y del tipo de cambio, importaciones que crecen con la renta interior (propensión marginal a importar $m$) y unos flujos de capital que responden al diferencial de interés con sensibilidad $\kappa$, la condición de equilibrio externo del modelo \authorfont{MUNDELL}-\authorfont{FLEMING} es:

$$BP:\quad \underbrace{X(Y^{*},e)-M(Y,e)}_{\text{cuenta corriente}}+\underbrace{\kappa\,(r-r^{*})}_{\text{entradas netas de capital}}=0$$

Su pendiente en el plano $(Y,r)$ es $dr/dY\rvert_{BP}=m/\kappa\geq 0$: un aumento de la renta deteriora la cuenta corriente vía importaciones y exige un tipo de interés mayor que atraiga capital. El parámetro $\kappa\in[0,\infty)$ gradúa la movilidad del capital y da los dos casos extremos: con movilidad perfecta ($\kappa\to\infty$) la BP es horizontal en $r=r^{*}$, porque cualquier diferencial de interés desata flujos ilimitados; sin movilidad ($\kappa=0$) la BP es vertical, y el equilibrio externo lo decide solo la cuenta corriente.

\subsubsection{La política económica según el régimen cambiario}

La eficacia de las políticas deja de depender únicamente de las pendientes internas ($b$, $h$) y pasa a depender del régimen cambiario y del grado de movilidad del capital. El mecanismo nuevo es el tipo de cambio: con tipo flexible, los desequilibrios externos se corrigen con depreciaciones o apreciaciones que desplazan la IS; con tipo fijo, los corrige la variación de reservas, que convierte la oferta monetaria en endógena y desplaza la LM. Los dos resultados canónicos, con movilidad perfecta, se muestran en la figura: la política monetaria con tipo flexible es plenamente eficaz (la caída del interés provoca salidas de capital, la moneda se deprecia y las exportaciones netas rematan la expansión: más eficaz incluso que en economía cerrada), mientras que la política fiscal con tipo flexible es impotente (la presión alcista sobre el interés atrae capital, aprecia la moneda y la pérdida de exportaciones netas devuelve la renta a su punto de partida: expulsión por el canal exterior). Con tipo fijo los papeles se invierten exactamente: la política monetaria queda anulada, porque la defensa de la paridad obliga a deshacer la expansión (pérdida de reservas), y la política fiscal alcanza su eficacia máxima, porque la entrada de capital obliga al banco central a expandir la oferta monetaria y el tipo de interés no llega a subir.

\subsection{El cierre nominal: la demanda agregada, las tres ofertas y el efecto saldos reales}\label{anx1:oada}

El IS-LM determina la renta con precios dados: es un modelo de demanda. Para pronunciarse sobre los precios, y para fundamentar la tesis central de la escuela (keynesiana a corto plazo, neoclásica a largo), la Síntesis necesita dos piezas más: una curva de demanda agregada en el plano renta-precios y una teoría de la oferta que diga cómo se reparte el ajuste entre cantidades y precios en cada horizonte.

\subsubsection{La demanda agregada desde el IS-LM}

La curva de demanda agregada se obtiene dejando variar el nivel de precios en el propio IS-LM: una subida de $P$ reduce los saldos reales $M/P$, desplaza la LM a la izquierda, eleva el tipo de interés, contrae la inversión y reduce la renta de equilibrio. Ese canal, precios $\to$ saldos reales $\to$ interés $\to$ inversión, se conoce como efecto \authorfont{KEYNES}, y genera una relación decreciente entre $P$ e $Y$: cada punto de la DA es un equilibrio IS-LM completo.

\subsubsection{Las tres curvas de oferta y la partición temporal}

Del lado de la oferta, la Síntesis reduce el corto plazo al supuesto de rigidez exógena: $p_t=p_{t-1}$, la expectativa estática que el cuerpo del tema examina como seña metodológica. En la versión más sencilla del modelo agregado, eso se traduce en tres curvas de oferta,\footnote{%
    Nótese la similitud con la partición temporal de los periodos de mercado, corto y largo plazo en la teoría de \authorfont{MARSHALL} ([\authorfont{Ver Tema 3.A.3}]).
} que el panel (b) de la figura superpone: una de muy corto plazo, perfectamente elástica (precios dados: todo el ajuste recae en las cantidades, y rigen las conclusiones keynesianas del IS-LM); una de corto plazo con pendiente positiva, en la que cantidades y precios comparten el ajuste; y una de largo plazo, perfectamente inelástica en la renta de pleno empleo $\bar{Y}$, donde el ajuste es íntegramente de precios y rigen las conclusiones neoclásicas.\footnote{%
    La convergencia de la curva de muy corto plazo hacia la de largo plazo, es decir, la dinámica del tránsito entre horizontes, fue estudiada por autores como \authorfont{DORNBUSCH}; véase el modelo de sobrerreacción del tipo de cambio (1976), que explota justamente la diferencia de velocidad entre un mercado de ajuste instantáneo y otro de ajuste lento ([\authorfont{Ver Tema 3.B.14}]): \href{https://en.wikipedia.org/wiki/Overshooting_model}{Wikipedia.org} o \href{https://people.ucsc.edu/~hutch/Econ241a/Articles/Dornbusch_overshooting.pdf}{\textit{Expectations and Exchange Rate Dynamics} (1976)}.
} La partición temporal es, así, la traducción operativa del armisticio doctrinal: qué teoría rige depende de cuánto tiempo se deja a los precios para ajustarse.

\subsubsection{El efecto saldos reales: \authorfont{PIGOU}, \authorfont{PATINKIN} y la síntesis como teorema}\label{anx1:saldos}

Queda la pregunta incómoda: ¿garantiza la flexibilidad de precios, por sí sola, la vuelta al pleno empleo? El efecto \authorfont{KEYNES} no basta para asegurarlo, porque tiene los mismos límites que la política monetaria: en la trampa de la liquidez la caída de precios reduce el interés hasta el suelo y ahí se detiene, y si la inversión es insensible al interés el estímulo no llega a la demanda. La respuesta afirmativa llegó por otro canal. \authorfont{PIGOU} (1943) observó que la deflación eleva el valor real de los saldos monetarios y de la deuda pública en manos del público, es decir, su riqueza, y que esa mayor riqueza estimula directamente el consumo sin pasar por el tipo de interés: es el efecto saldos reales o efecto \authorfont{PIGOU}, inmune a la trampa de la liquidez. \authorfont{PATINKIN} (\textit{Money, Interest, and Prices}, 1956) convirtió la observación en teoría general: integró el efecto saldos reales en todas las funciones de demanda de un sistema de equilibrio general walrasiano, demostrando que, con precios y salarios plenamente flexibles, el sistema posee un equilibrio de pleno empleo y los mecanismos para alcanzarlo, y que el desempleo keynesiano debe entenderse como un fenómeno de desequilibrio, un tránsito, no como un equilibrio con paro.

La conclusión fija la doctrina de la escuela con precisión quirúrgica: a \authorfont{KEYNES} no se le concede la razón en la teoría, se le concede la razón en la práctica. En la teoría, el largo plazo clásico queda restaurado como teorema; en la práctica, el mecanismo que lo sostiene es cuantitativamente débil y peligrosamente lento, porque exige una deflación sostenida cuyos efectos colaterales pueden ser desestabilizadores,\footnote{%
    \textbf{OJO.-} El argumento contra la deflación como terapia es anterior: \authorfont{KEYNES} lo expone en el capítulo 19 de la \textit{Teoría General} (la caída de salarios nominales deprime la demanda por la vía de las expectativas y de la redistribución) y \authorfont{FISHER} (1933) en la teoría de la deflación de deuda: la caída de precios eleva la carga real de las deudas y puede contraer la demanda en lugar de expandirla, un canal desarrollado en el análisis del ciclo financiero ([\authorfont{Ver Tema 3.A.42}]).
} de modo que la política de estabilización no contradice la teoría: la abrevia. Esta arquitectura, un núcleo walrasiano de largo plazo conviviendo con una macroeconomía keynesiana de corto plazo sin un puente microeconómico entre ambos, es exactamente lo que la literatura posterior llamará la esquizofrenia de la Síntesis, y la grieta por la que entrarán tanto la teoría del desequilibrio ([\authorfont{Ver Tema 3.A.4}]) como el programa de microfundamentación de la Nueva Macroeconomía Clásica ([\authorfont{Ver Tema 3.A.6}]).

\subsection{La curva de \authorfont{PHILLIPS}: la ecuación perdida}\label{anx1:phillips}

Al aparato le faltaba una pieza: nada en el IS-LM con precios dados explica la inflación. Mientras la inflación fue un fenómeno episódico, la carencia resultó tolerable; cuando a finales de los años cincuenta y en los sesenta se volvió persistente, se hizo insostenible, y la pieza que faltaba, la ecuación perdida del sistema, llegó de la evidencia empírica antes que de la teoría.

\subsubsection{El hallazgo empírico de \authorfont{PHILLIPS} (1958)}

El economista neozelandés \authorfont{A. W. PHILLIPS} publicó en 1958, en \textit{Economica}, un estudio estadístico de la relación entre la tasa de paro y la tasa de variación de los salarios nominales en el Reino Unido entre 1861 y 1957. Su resultado: una relación inversa, no lineal y notablemente estable a lo largo de casi un siglo. Ajustada sobre el periodo 1861-1913, la curva estimada responde a la forma $\dot{W}/W=-0{,}9+9{,}638\,U^{-1{,}394}$,\textcolor{orange}{\textsuperscript{[Probable]}} que corta el eje en un paro en torno al $5{,}5\%$ (el nivel que estabiliza los salarios) y se vuelve casi vertical en niveles de paro muy bajos.

{\small\textbf{La curva de \authorfont{PHILLIPS} original: forma ajustada y bucles cíclicos}}\par\vspace{0.3em}
\includegraphics{anx1_phillips.pdf}\par\vspace{0.3em}

Aunque el análisis es empírico, el artículo avanza las hipótesis económicas de la relación. La pendiente negativa respondería a las fuerzas del mercado: cuando la demanda de trabajo es alta y quedan pocos desempleados, las empresas pujan por los trabajadores elevando los salarios. La no linealidad respondería a una asimetría: los trabajadores se resisten a aceptar rebajas nominales cuando el paro es alto, de modo que los salarios suben deprisa en las expansiones pero bajan despacio en las depresiones. \authorfont{PHILLIPS} documentó además los bucles que el ciclo describe alrededor de la curva, en sentido antihorario: con el paro cayendo los salarios aceleran más de lo que la curva predice, y al revés, un matiz dinámico que la literatura posterior olvidaría durante décadas.

\subsubsection{La fundamentación teórica de \authorfont{LIPSEY} (1960)}

La curva nació sin teoría, y el primer fundamento sistemático se lo dio \authorfont{LIPSEY} (1960) con dos piezas. La primera es una función de ajuste del mercado de trabajo, según la formalización de \authorfont{LIPSEY} (1960) de la curva de \authorfont{PHILLIPS}: el crecimiento del salario nominal responde linealmente al exceso de demanda de trabajo,

$$\frac{\dot W}{W}=\alpha\, X_L=\alpha\,\left(\frac{L^{D}-L^{S}}{L^{S}}\right)$$

La segunda es la relación, negativa y no lineal, entre ese exceso de demanda, que no es observable, y la tasa de paro, que sí lo es:

$$X_L=g(U),\qquad g'(U)<0$$

donde la no linealidad recoge las fricciones del mercado de trabajo: aunque el exceso de demanda agregado sea nulo, subsiste un paro friccional positivo, y reducciones adicionales del paro exigen excesos de demanda crecientes. Componiendo ambas se obtiene la curva observada, $\dot W/W=f(U)$ con $f'<0$, que hereda la convexidad de $g$. \authorfont{LIPSEY} añadió un resultado de agregación con consecuencias duraderas: si los mercados de trabajo sectoriales tienen curvas propias, la curva agregada depende de la dispersión del paro entre sectores, de modo que cambios en la composición del desempleo desplazan la relación agregada aunque ninguna curva sectorial se mueva.\textcolor{orange}{\textsuperscript{[Probable]}}

\subsubsection{De salarios a precios: \authorfont{SAMUELSON} y \authorfont{SOLOW} (1960) y el menú de política}\label{anx1:menu}

El salto de una regularidad sobre salarios británicos a un instrumento de política económica lo dieron \authorfont{SAMUELSON} y \authorfont{SOLOW} en \textit{Analytical Aspects of Anti-Inflation Policy} (1960). El puente analítico es la formación de precios por margen sobre costes: si los precios se fijan como margen sobre el coste laboral unitario, la inflación es aproximadamente el crecimiento salarial menos el de la productividad, $\dot P/P\approx \dot W/W-\dot x/x$, y la curva de salarios se convierte en una curva de inflación sin más que restar la productividad tendencial. Estimada para Estados Unidos, los autores la leyeron como la primera cuantificación del dilema: la estabilidad de precios parecía exigir un paro del $5$-$6\%$, mientras que un paro del $3\%$ costaría una inflación del orden del $4$-$5\%$ anual.\textcolor{orange}{\textsuperscript{[Probable]}}

La reinterpretación decisiva no fue estadística sino conceptual: la curva pasó a leerse como un menú de elección de política económica, un conjunto de combinaciones alcanzables de inflación y paro entre las que la sociedad, a través de sus políticas de demanda, elige su punto preferido. Conviene dejar constancia de que los propios autores advirtieron de que la relación podía desplazarse con el tiempo, entre otras razones por el papel de las expectativas de inflación, una cautela que la práctica posterior olvidó y que la crítica monetarista convertiría en demolición.\textcolor{orange}{\textsuperscript{[Probable]}}

\subsubsection{La curva en acción y sus primeras grietas}

Con la curva de \authorfont{PHILLIPS}, el sistema quedó cerrado: el IS-LM determinaba la demanda y el nivel de actividad, y la curva traducía la presión de la demanda en salarios y precios. Los grandes modelos macroeconométricos de los años sesenta incorporaron la relación como bloque de salarios y precios, y la política económica la usó como lo que decía ser: un menú. La elección del punto, más empleo a cambio de algo más de inflación, encajaba con las preferencias del momento y pareció funcionar durante un decenio. Las grietas aparecieron pronto, sin embargo: a finales de los años sesenta la inflación estadounidense empezó a acelerarse sin la mejora correspondiente del empleo, desplazando la nube de puntos hacia fuera, exactamente el síntoma que cabría esperar si la posición de la curva dependiera de las expectativas de inflación. La crítica que convirtió ese síntoma en teorema, la curva aumentada por expectativas y la tasa natural de paro de \authorfont{FRIEDMAN} y \authorfont{PHELPS}, se desarrolla en el Anexo II, y el desenlace empírico de los años setenta, en el bloque final del cuerpo.

\subsection{Evaluación: qué puede y qué no puede este aparato}\label{anx1:eval}

El aparato completo, IS-LM, su extensión abierta, el cierre OA-DA y la curva de \authorfont{PHILLIPS}, explica por sí solo el dominio intelectual de la Síntesis durante un cuarto de siglo. Sus virtudes son las de una buena herramienta: condensa la determinación simultánea de los mercados de bienes y de dinero en un gráfico de dos curvas; convierte cualquier debate de política económica en una discusión disciplinada sobre desplazamientos y pendientes, es decir, sobre magnitudes en principio estimables; y se deja estimar, de modo que el esqueleto teórico se prolongó con naturalidad en los grandes modelos macroeconométricos que las administraciones y los bancos centrales usaron durante décadas (la historia de esa econometría estructural se desarrolla en el Anexo \ref{anx:autores}). Esa versatilidad explica su extraordinaria supervivencia: el modelo se sigue enseñando en los estudios de grado de todo el mundo por su simplicidad y su utilidad para identificar mecanismos de transmisión, y la mayoría de los modelos aplicados de la OCDE, el FMI o el BCE confirman efectos desbordamiento de las políticas análogos a los que predice, aunque en la investigación de frontera y en las instituciones el relevo lo tomaron los modelos de equilibrio general dinámico estocástico ([\authorfont{Ver Tema 3.A.6}]).

Sus límites son igual de nítidos, y conviene enumerarlos porque cada uno tiene heredero. Es un modelo estático, sin dinámica explícita de ajuste ni acumulación: el stock de capital está fijo aunque haya inversión. Sus expectativas son exógenas cuando no inexistentes: nada en el modelo dice cómo revisan los agentes sus previsiones cuando la política cambia, el flanco por el que atacarán primero el monetarismo con las expectativas adaptativas (Anexo II) y después \authorfont{LUCAS} con las racionales ([\authorfont{Ver Tema 3.A.6}]). Sus funciones de comportamiento son \textit{ad hoc}: postuladas, no derivadas de la optimización de ningún agente, el déficit que alimentará todo el programa de microfundamentación ([\authorfont{Ver Tema 3.A.6, Tema 3.A.7}]). Su curva de \authorfont{PHILLIPS} es una regularidad empírica sin teoría de las expectativas, condenada a desplazarse en cuanto la inflación dejara de ser una sorpresa. Y su arquitectura entera convive con la esquizofrenia ya señalada: un largo plazo walrasiano y un corto plazo keynesiano sin puente entre ambos. Nada de esto era invisible para sus autores; la apuesta de la escuela fue que los defectos eran el precio razonable de un instrumento útil. La primera escuela que recogió el guante, discutiendo el aparato desde dentro y en su propio lenguaje, fue el monetarismo, cuyo arsenal analítico se construye en el Anexo II.







\anexo{La contrarrevolución analítica: demanda de dinero, renta permanente y la curva de \authorfont{PHILLIPS} aumentada}\label{anx:contra}

Este anexo es el manual del aparato analítico que el monetarismo construyó para desmontar, pieza a pieza, el edificio del Anexo \ref{anx:aparato}. Su lógica interna es una sola idea desplegada: si la demanda de dinero es una función estable de pocas variables, entonces el dinero, y no el gasto autónomo, es el determinante dominante de la renta nominal, y toda la jerarquía de políticas de la Síntesis se invierte. El recorrido va de lo más básico a lo más complejo: la herencia cuantitativa que \authorfont{FRIEDMAN} reivindica, la reformulación de la demanda de dinero de 1956, el modelo de \authorfont{CAGAN} y las expectativas adaptativas, la función de consumo de la renta permanente, los mecanismos de transmisión y los efectos riqueza, la batalla empírica de los años sesenta y, como culminación, la curva de \authorfont{PHILLIPS} aumentada y el corolario de las reglas. Las batallas metodológicas y los protagonistas se tratan en el Anexo \ref{anx:autores}.

\subsection{La herencia cuantitativa: \authorfont{FISHER}, Cambridge y el ataque de \authorfont{KEYNES}}\label{anx2:tcd}

Las raíces del monetarismo residen en la teoría cuantitativa del dinero, uno de los pilares de la teoría monetaria clásica, con formulaciones tempranas en Azpilcueta, Bodin o Hume ([\authorfont{Ver Tema 3.A.3}]). Su enunciado moderno de referencia es el de \authorfont{FISHER} (1911), que parte de una identidad contable entre el dinero que circula y el valor de lo que se paga con él:

$$M\,V\equiv P\,T$$

donde $M$ es la cantidad de dinero, $V$ su velocidad de circulación, $P$ el nivel de precios y $T$ el volumen de transacciones (en las versiones modernas, la renta real $Y$). La identidad se convierte en teoría al añadir tres hipótesis: $V$ viene determinada por los hábitos e instituciones de pago y varía despacio; $T$ viene determinado por factores reales y, a largo plazo, por el pleno empleo; y $M$ la fija la autoridad monetaria. Con $V$ y $T$ anclados, toda variación de $M$ se traslada proporcionalmente a $P$, y la causalidad corre de izquierda a derecha de la ecuación. La escuela de Cambridge reescribió la misma aritmética como una función de demanda: según la formulación de \authorfont{MARSHALL} (1923) y \authorfont{PIGOU} (1917)\textcolor{orange}{\textsuperscript{[Probable]}},

$$M^{d}=k\,P\,Y$$

donde $k$ es la fracción de la renta nominal que el público desea mantener en forma de dinero (aritméticamente, $k=1/V$). El desplazamiento parece menor y es decisivo: si $k$ es algo que el público \textit{desea}, la teoría cuantitativa se convierte en una teoría de la elección de cartera, y la pregunta pasa a ser de qué depende ese deseo. Por esa puerta entró \authorfont{KEYNES} (1936), alumno de \authorfont{MARSHALL}: al añadir el motivo especulación, el tipo de interés se coló en la función ($L(Y,i)$) y la velocidad dejó de ser un dato institucional para volverse endógena y volátil, $V=Y/L(Y,i)$; en una depresión con trampa de la liquidez, $V$ se desploma y absorbe cualquier expansión de $M$ (\authorfont{Ver} Anexo \ref{anx1:extremos}). La teoría cuantitativa quedó arrumbada como reliquia: el reto exacto que la contrarrevolución tenía que responder no era defender que $V$ fuese constante, que no lo es, sino demostrar que es una función \textit{estable} de pocas variables. De ese programa sale la conclusión que \authorfont{FRIEDMAN} convirtió en lema en sus conferencias de 1963: \textit{la inflación es siempre y en todo lugar un fenómeno monetario} (\authorfont{FRIEDMAN}, 1963)\textcolor{orange}{\textsuperscript{[Probable]}}.

\subsection{La reformulación de 1956: la demanda de dinero como elección de cartera}\label{anx2:restatement}

La pieza fundacional es el ensayo que abre el volumen colectivo del taller de dinero y banca de Chicago, \textit{Studies in the Quantity Theory of Money} (\authorfont{FRIEDMAN}, 1956). \authorfont{FRIEDMAN} lo presentó no como una teoría nueva, sino como la destilación de una supuesta tradición oral de Chicago, una genealogía que \authorfont{PATINKIN} discutiría después documentadamente (la polémica se desarrolla en el Anexo \ref{anx:autores}). La premisa es que el dinero es un activo que presta servicios (liquidez, seguridad, poder de compra disponible) y reporta utilidad, de modo que su demanda ha de tratarse como la de cualquier otro bien duradero: con una restricción de riqueza, unos rendimientos relativos de los activos alternativos y unas preferencias.\footnote{%
    \authorfont{FRIEDMAN} retoma así la tradición de \authorfont{MARSHALL}, \authorfont{FISHER} y \authorfont{PIGOU}, pero acepta las motivaciones de \authorfont{KEYNES} para demandar dinero (transacción, precaución, especulación) y las lleva más allá: en lugar de tres motivos separados, una sola decisión de cartera sobre toda la riqueza.
} El dinero tendrá utilidad marginal decreciente y los agentes lo sustituirán por otros activos cuando estos les reporten mayor utilidad marginal: bonos, acciones y participaciones de capital, e incluso activos reales, en particular bienes de consumo duradero. La utilidad de esos otros activos es el coste de oportunidad de mantener dinero, y el resultado es la función de demanda nominal de \authorfont{FRIEDMAN} (1956):

$$M^{d}=f\!\left(W,\ r_b,\ r_k,\ \pi^{e},\ \frac{W_H}{W}\right)\cdot P$$

donde $W$ es la riqueza total, $r_b$ el rendimiento de los bonos, $r_k$ el de las acciones y participaciones de capital, $\pi^{e}$ la inflación esperada y $W_H/W$ el peso de la riqueza humana en la riqueza total. Sus propiedades:
\begin{lnum}
    \item $\dfrac{\partial f}{\partial W}>0$. La riqueza es la restricción presupuestaria de la decisión: a mayor riqueza, más dinero se demanda para financiar transacciones y como depósito de valor, y la propia riqueza limita cuánto dinero se puede demandar;
    \item $\dfrac{\partial f}{\partial r_b}<0$ y $\dfrac{\partial f}{\partial r_k}<0$. A mayor rentabilidad de los activos alternativos, mayor coste de oportunidad de los saldos monetarios y menor demanda de dinero;
    \item $\dfrac{\partial f}{\partial \pi^{e}}<0$. La tasa esperada de variación de los precios\footnote{%
        $\pi^{e}=E\!\left[\dfrac{\partial P}{\partial t}\dfrac{1}{P}\right]$. Es el rendimiento (negativo) propio del dinero: la inflación esperada erosiona su poder de compra y hace comparativamente más atractivos los demás activos financieros y los bienes duraderos.
    } reduce la utilidad esperada de la tenencia de dinero y, con ella, su demanda;
    \item $\dfrac{\partial f}{\partial (W_H/W)}>0$. Cuanto mayor es el peso de las rentas del trabajo, más inciertas por lo general que las de los activos financieros y no pignorables, más dinero se desea mantener por motivo precaución; y,
    \item La demanda es homogénea de grado uno en $P$: duplicar todos los precios duplica la demanda nominal y deja intacta la demanda real. El nivel absoluto de precios determina la demanda nominal, pero no la real.\footnote{%
        Este supuesto conlleva que los agentes están libres de ilusión monetaria, algo que la rigidez salarial nominal de \authorfont{KEYNES} necesitaba implícitamente negar, e introduce un grado mayor de racionalidad en la conducta modelizada: una semilla de lo que después exigirá la Nueva Macroeconomía Clásica ([\authorfont{Ver Tema 3.A.6}]).
    }
\end{lnum}

Dividiendo por $P$ y sustituyendo la riqueza por su equivalente anualizado, la renta permanente $Y_p$ (la relación $Y_p=r\cdot W$ se desarrolla en \ref{anx2:pih}), queda la demanda de saldos reales:

$$\frac{M^{d}}{P}=f\!\left(Y_p,\ r_b,\ r_k,\ \pi^{e},\ \frac{W_H}{W}\right)$$

Es determinante que la función dependa de la renta permanente y no de la corriente: $Y_p$ es, por construcción, mucho menos volátil que $Y$, con lo que la demanda de dinero será mucho más estable de lo que la Síntesis suponía.\footnote{%
    La estabilidad no es un axioma sino la hipótesis empírica central del programa: \authorfont{FRIEDMAN} (1959) estimó la función con datos estadounidenses y encontró una demanda estable, con elasticidad renta superior a la unidad (el dinero como bien de lujo) y una sensibilidad al tipo de interés tan pequeña que llegó a tratarla como despreciable\textcolor{orange}{\textsuperscript{[Probable]}}. La literatura posterior confirmó elasticidades al interés pequeñas pero significativas, lo que matiza, sin anularla, la conclusión.
} Agrupando los rendimientos de los activos alternativos en un tipo de interés $r$, la forma compacta es

$$\frac{M^{d}}{P}=g(r,\ \pi^{e})\cdot Y_p$$

y la velocidad de circulación resulta ser $V=Y/(M/P)$, que a largo plazo (cuando $Y=Y_p$) es $V=1/g(r,\pi^{e})$: no una constante, como en la caricatura de \authorfont{FISHER}, sino una \textit{función estable}, que se mueve poco y de forma predecible. Esa es toda la diferencia con \authorfont{KEYNES}: donde la Síntesis veía una $L(Y,i)$ dominada por un motivo especulación caprichoso, capaz de absorber en la trampa cualquier expansión, \authorfont{FRIEDMAN} ve una demanda anclada en la renta permanente, sin tramo plano relevante. Reinsertada en la ecuación cuantitativa, con la causalidad corriendo del dinero a la renta nominal,

$$\underbrace{M\,V=P\,Y}_{\text{\scriptsize Causalidad: }\Rightarrow}\quad\longrightarrow\quad \underbrace{\frac{M}{P}=g(r,\pi^{e})\,Y_p}_{\text{\scriptsize Equilibrio del mercado monetario}}$$

la reformulación entrega una teoría de la \textit{renta nominal}: con $g$ estable, las variaciones de $M$ se trasladan a $P\,Y$.\footnote{%
    Según la definición de \authorfont{FRIEDMAN}, la renta corriente tiene un componente permanente y uno transitorio, $Y=Y_p+Y_{tr}$, con el transitorio aleatorio y de media cero: a largo plazo $Y=Y_p$. A largo plazo, además, $Y_p$ solo depende de factores reales, con lo que la variación monetaria acaba íntegramente en precios: es una manera más elaborada de llegar a la conclusión de \authorfont{FISHER}. La diferencia está en el corto plazo, donde \authorfont{FRIEDMAN}, a diferencia de \authorfont{FISHER}, admite efectos reales: cómo se reparte $\Delta(PY)$ entre $P$ e $Y$ es el problema de la ecuación perdida (\ref{anx2:evidencia}).
}

\subsection{\authorfont{CAGAN} (1956): hiperinflaciones y expectativas adaptativas}\label{anx2:cagan}

En el mismo volumen de 1956, \authorfont{CAGAN} llevó la nueva demanda de dinero a su banco de pruebas extremo: siete hiperinflaciones europeas de entreguerras y posguerra (Austria, Alemania, Grecia, Polonia, Rusia y Hungría en dos ocasiones)\textcolor{orange}{\textsuperscript{[Probable]}}. Su pregunta es exactamente la que deja abierta la reformulación: ¿bajo qué condiciones deja de ser estable la demanda de dinero? En una hiperinflación todos los rendimientos reales quedan empequeñecidos por la variable que \authorfont{FRIEDMAN} había añadido a la función: la inflación esperada pasa a ser el único coste de oportunidad relevante, y la demanda de saldos reales de \authorfont{CAGAN} (1956) se reduce a

$$\frac{M^{d}}{P}=e^{-\alpha\,\pi^{e}},\qquad \alpha>0$$

donde $\alpha$ es la semielasticidad de la demanda respecto a la inflación esperada: cada punto adicional de $\pi^{e}$ reduce los saldos reales deseados en una proporción $\alpha$, constante.\footnote{%
    La forma completa es $M^{d}/P=e^{\gamma-\alpha\pi^{e}}$, con $\gamma$ recogiendo la escala (renta real y demás argumentos, congelados durante el episodio). La forma exponencial no es un capricho: con saldos reales que caen a una milésima de su nivel inicial, solo una semielasticidad constante ajusta los datos de principio a fin del episodio.
}

Para hacerla operativa, \authorfont{CAGAN} necesitaba una regla observable de formación de $\pi^{e}$, y la que propuso se convirtió en la hipótesis de las expectativas adaptativas (HEA), la aportación metodológica más duradera del episodio. En su forma de corrección de errores,

$$\pi^{e}_{t+1}=\pi^{e}_{t}+\lambda\,(\pi_{t}-\pi^{e}_{t}),\qquad 0<\lambda\le 1$$

las expectativas futuras son las pasadas corregidas en una fracción $\lambda$ del error cometido: $\lambda$ es la velocidad de aprendizaje y, si $\lambda=1$, se espera simplemente la inflación del último periodo. Sustituyendo recursivamente la misma regla hacia atrás se obtiene la forma equivalente de retardos distribuidos,

$$\pi^{e}_{t+1}=\lambda\sum_{j=0}^{\infty}(1-\lambda)^{j}\,\pi_{t-j}$$

una media de toda la inflación pasada con ponderaciones geométricamente decrecientes: los periodos más recientes pesan más y las ponderaciones suman uno. Esta aritmética de la memoria geométrica es uno de los sellos técnicos del monetarismo y reaparece, con otros nombres, en la estimación de la renta permanente (\ref{anx2:pih}) y en la curva de \authorfont{PHILLIPS} aumentada (\ref{anx2:phillips}).

La dinámica conjunta de las dos ecuaciones responde a la pregunta inicial. Si la inflación se acelera, $\pi^{e}$ sube con retardo y el público huye del dinero ($M^{d}/P$ cae), lo que, para una misma senda de emisión, acelera aún más los precios: la huida alimenta la inflación que la provoca. El sistema solo es estable si la reacción es suficientemente amortiguada, esto es, si $\alpha\,\lambda<1$: una demanda muy sensible ($\alpha$ alto) combinada con expectativas que se revisan deprisa ($\lambda$ alto) desancla el mercado monetario y la hiperinflación puede autoalimentarse. La estimación de \authorfont{CAGAN} encontró en casi todos los episodios parámetros dentro de la región estable: incluso en las hiperinflaciones la demanda de dinero se comporta como una función estable y la causa del desastre no es un pánico autorrealizado, sino la emisión\textcolor{orange}{\textsuperscript{[Probable]}}.\footnote{%
    De ahí la implicación de política que el monetarismo extrajo del modelo: la estabilidad de la demanda de dinero no es incondicional, sino que presupone una política monetaria que mantenga ancladas las expectativas de inflación. Una política antiinflacionaria es el prerrequisito para que la premisa central del propio monetarismo se cumpla. El modelo de \authorfont{CAGAN} reaparece como bloque de los modelos monetarios del tipo de cambio ([\authorfont{Ver Tema 3.B.14}]) y en el análisis de la inflación ([\authorfont{Ver Tema 3.A.35}]).
}

El modelo explica también por qué se emite hasta ese extremo: la inflación es un impuesto. En estado estacionario, el señoreaje real que recauda el gobierno es la base (los saldos reales) por el tipo impositivo (la inflación):

$$S=\pi\cdot\frac{M^{d}}{P}=\pi\,e^{-\alpha\pi}$$

que tiene forma de campana: crece hasta $\pi^{*}=1/\alpha$ y decrece después, porque más allá de ese punto la base huye más deprisa de lo que el tipo sube. Las hiperinflaciones son, en última instancia, fenómenos fiscales: gobiernos sin acceso a impuestos ordinarios ni a deuda que exprimen el impuesto inflacionario más allá de su máximo recaudatorio\textcolor{orange}{\textsuperscript{[Probable]}}.\footnote{%
    El coste de bienestar de financiarse así lo analizó \authorfont{BAILEY} (1956) con el instrumental del excedente del consumidor aplicado a la demanda de saldos reales\textcolor{orange}{\textsuperscript{[Probable]}}.
}

\subsection{La renta permanente (1957): la otra función de consumo}\label{anx2:pih}

El segundo pilar de la contrarrevolución ataca la función de comportamiento más querida del keynesianismo. Para \authorfont{KEYNES} (1936), el consumo corriente depende de la renta corriente,

$$C_t=c_0+c'\,Y_t$$

con $c_0>0$ un consumo de subsistencia y $c'$ la propensión marginal a consumir, constante y menor que la unidad. La implicación está en la propensión \textit{media}: $C/Y=c_0/Y+c'$ es decreciente con la renta, de modo que, a medida que la economía se enriquece, consume una fracción cada vez menor de su producto y el hueco de demanda que hay que llenar con inversión o gasto público crece sin límite. Es la \textit{tesis del estancamiento secular}, que en los años cuarenta justificó la expectativa de una recaída en la depresión al acabar el esfuerzo bélico y un papel permanente y creciente del Estado como gastador de último recurso. La evidencia la desmintió por partida doble, y de forma desconcertante: los presupuestos familiares y las series cortas mostraban, en efecto, una propensión media decreciente con la renta, pero las series largas de contabilidad nacional reconstruidas por \authorfont{KUZNETS} (1946) mostraban una propensión media aproximadamente constante durante siete décadas\textcolor{orange}{\textsuperscript{[Probable]}}. Dos evidencias sólidas y contradictorias: el \textit{enigma del consumo}.

La solución de \authorfont{FRIEDMAN} (1957) en \textit{A Theory of the Consumption Function} es redefinir la variable de la que depende el consumo. La renta corriente se descompone en un componente permanente y otro transitorio, y lo mismo el consumo:

$$Y=Y_p+Y_{tr},\qquad C=C_p+C_{tr}$$

\textbf{Definición.-} (Renta permanente) Es el flujo de renta que puede mantenerse indefinidamente en el tiempo y consumirse íntegramente sin menoscabar la fuente de la que deriva. Está ligada a la riqueza total por la relación $Y_p=r\cdot W$, donde $r$ es el rendimiento interno esperado de la riqueza y $W$ incluye el valor actualizado de los bonos,\footnote{%
    Si el bono es perpetuo con cupón a tipo uniforme $r$, su valor actualizado es, por definición, $B/r$, siendo $B$ su renta nominal.
} los saldos monetarios, el capital y la riqueza humana (el valor actualizado de las rentas del trabajo). La hipótesis pretende describir cómo los agentes distribuyen el consumo a lo largo de su vida: el consumo de hoy no depende solo de los ingresos de hoy, sino de la expectativa de los ingresos de los próximos años.

El consumidor racional planifica sobre lo permanente y deja pasar lo transitorio: si percibe que un incremento de su renta es efímero, no altera su plan de consumo, sino su ahorro. La hipótesis central es que el consumo permanente es proporcional a la renta permanente,

$$C_p=k\!\left(r,\ \frac{W_H}{W},\ \theta\right)\cdot Y_p$$

donde $k\in(0,1)$ no es una constante universal sino una función de las características del consumidor:\footnote{%
    Tres letras $k$ conviven en este tema por accidente histórico y no guardan relación: la fracción de Cambridge (\ref{anx2:tcd}), esta propensión $k$ de la renta permanente y la regla del $k$ por ciento (\ref{anx2:regla}).
} depende inversamente del tipo de interés, que es el precio relativo del consumo presente; inversamente del peso de la riqueza humana $W_H/W$, porque las rentas del trabajo son más inciertas y el capital humano no puede pignorarse como garantía, lo que obliga a ahorrar más por precaución; y de un factor $\theta$ que recoge gustos, edad, incertidumbre y posición social. Los componentes transitorios, por su parte, se suponen aleatorios, de media cero y sin correlación con los permanentes ni entre sí: la renta transitoria apenas se consume.\footnote{%
    La crítica clásica a la hipótesis es que la renta permanente no es observable: ¿cómo va a decidir nadie su consumo respecto a algo que no conoce? La respuesta de \authorfont{FRIEDMAN} es que el individuo racional sí estima sus rentas futuras, y que $Y_p$ es solo el artefacto que sintetiza esa estimación. Para medirla propuso exactamente la aritmética de \ref{anx2:cagan}: una media de las rentas pasadas con ponderaciones geométricamente decrecientes, es decir, una expectativa adaptativa de la renta.
}

La hipótesis disuelve el enigma con un solo mecanismo, recogido en la figura. En las series largas, la renta observada es esencialmente permanente ($Y\approx Y_p$) y el cociente $C/Y=k$ es constante: es el rayo de \authorfont{KUZNETS}, con los puntos $A$ y $B$ marcando el crecimiento secular. En un corte transversal o en el ciclo, en cambio, los hogares (o los años) de renta alta incluyen de media rentas transitorias positivas que no se consumen, y los de renta baja, negativas: la recta ajustada ($C^{\,1}_{cp}$, $C^{\,2}_{cp}$) es más plana que el rayo y la propensión media parece caer con la renta, como en el punto $T$, situado bajo el rayo. La función de consumo keynesiana no era más que la fotografía sesgada de un comportamiento proporcional.

Las implicaciones desbordan la teoría del consumo, porque la función keynesiana sostenía media Síntesis:
\begin{lnum}
    \item \textit{No hay riesgo de subconsumo}. La participación del consumo en la demanda se mantiene constante al crecer la renta: la tesis del estancamiento queda desacreditada y, con ella, la justificación de un gasto público secularmente creciente;
    \item \textit{El multiplicador se encoge y se vuelve esquivo}. Si los hogares solo consumen lo permanente, una expansión fiscal transitoria (o un impuesto transitorio) apenas mueve el consumo: el individuo reparte la renta extra entre todos sus periodos. La propensión marginal sobre la renta transitoria es próxima a cero, y la potencia de la política fiscal de ajuste fino queda minada en su base;\footnote{%
        La predicción tuvo su contraste célebre: el recargo impositivo transitorio estadounidense de 1968, diseñado para enfriar el consumo, apenas lo redujo, tal y como la hipótesis anticipaba\textcolor{orange}{\textsuperscript{[Probable]}}.
    } y,
    \item \textit{Los efectos riqueza entran en la función de consumo}. Al depender el consumo de la riqueza a través de $Y_p=r\,W$, los cambios en la cantidad de dinero, en los precios de los activos o en los tipos de interés alteran el consumo directamente: el canal que la Síntesis había dejado en los márgenes (\authorfont{Ver} Anexo \ref{anx1:saldos}) pasa al centro del mecanismo de transmisión (\ref{anx2:transmision}).
\end{lnum}

La hipótesis, mucho más fundamentada microeconómicamente que la función keynesiana, tiene una hermana contemporánea e independiente, la hipótesis del ciclo vital de \authorfont{MODIGLIANI}, con la que comparte la lógica de suavización intertemporal ([\authorfont{Ver Tema 3.A.29}]).

\subsection{La transmisión monetaria: dos mecanismos y los efectos riqueza}\label{anx2:transmision}

Con la demanda de dinero y la función de consumo reconstruidas, la pregunta decisiva es cómo afecta el dinero al gasto. Las dos escuelas no discuten la existencia de un mecanismo de transmisión, sino su anchura, y esa diferencia resume todo el desacuerdo.

En la cadena keynesiana de la Síntesis, el dinero actúa a través de un solo mercado y un solo precio: el exceso de saldos se dirige al mercado de bonos, baja el tipo de interés y, si la inversión es sensible ($b$ alto), el gasto sube y el multiplicador hace el resto. La cadena tiene dos eslabones débiles reconocidos por el propio modelo: la caída de $r$ eleva la demanda de dinero (la velocidad cae y amortigua el impulso de $M$) y, en la trampa de la liquidez, el primer eslabón se rompe del todo. En la cadena monetarista, en cambio, el exceso de saldos se derrama sobre \textit{todo} el abanico de activos: bonos, acciones, bienes duraderos, capital real e incluso capital humano. Los precios de todos ellos suben, sus rendimientos (explícitos e implícitos) bajan y el gasto en bienes se estimula de forma directa, sin pasar necesariamente por el tipo de interés observable:

$$\Delta M\ \rightarrow\ M^{s}>M^{d}\ \text{(dinero abundante)}\ \rightarrow\ \Delta DA\ \text{(activos financieros y reales)}\ \rightarrow\ \Delta (P\,Y)$$

De ahí que los ciclos tengan para \authorfont{FRIEDMAN} un origen esencialmente monetario ([\authorfont{Ver Tema 3.A.42}]) y que la potencia de la política monetaria sea mucho mayor de lo que la Síntesis le concedía: la velocidad es estable y no amortigua el impulso, y los tipos de interés de mercado son una guía engañosa del tono de la política, porque la mayor parte de la sustitución ocurre en rendimientos que ningún mercado cotiza.\footnote{%
    Los monetaristas no niegan el mecanismo keynesiano: sostienen que es un caso particular y menor del suyo, la sustitución entre dinero y un único activo. La controversia sobre cuál de los dos domina alimentó la batalla empírica de \ref{anx2:evidencia}, con resultado ambiguo.
}

\includegraphics{anx2_transmision.pdf}\par\vspace{0.3em}

Sobre el propio tablero del IS-LM, la artillería monetarista se concreta en los \textit{efectos riqueza}, que operan a través de la función de consumo de \ref{anx2:pih}:
\begin{lnum}
    \item \textit{Efecto riqueza directo} (efecto \authorfont{PIGOU}). Al aumentar la cantidad de dinero (o de bonos), aumenta la riqueza de los agentes y con ella el consumo: la IS se desplaza a la derecha,
    $$\Delta M\ (\Delta B)\ \rightarrow\ \Delta W\ \rightarrow\ \Delta C$$
    \item \textit{Efecto riqueza indirecto}. Las variaciones del tipo de interés alteran el valor de mercado de los bonos y demás activos: una subida de $r$ reduce la riqueza y el consumo (y una bajada los eleva), desplazando la IS en el sentido correspondiente,
    $$\Delta r\ \rightarrow\ \nabla W\ \rightarrow\ \nabla C$$
    \item \textit{Efecto reajuste de cartera}. Si el Estado financia su déficit emitiendo deuda, los bonos se vuelven relativamente abundantes frente al dinero y los agentes recomponen su cartera demandando más saldos: la LM se desplaza a la izquierda,
    $$\Delta B\ \rightarrow\ \Delta M^{d}$$
\end{lnum}

La figura recoge el saldo de los tres efectos. Para una política monetaria expansiva, al desplazamiento habitual de la LM se suma el efecto riqueza directo sobre la IS y el indirecto de la caída de los tipos: la economía va de $E_0$ a $E_1$, más allá del resultado de la Síntesis ($E_{SNC}$). Para una política fiscal expansiva financiada con deuda ocurre lo contrario: el efecto \authorfont{PIGOU} de los bonos emitidos queda contrarrestado por el efecto indirecto de la subida de $r$ y por el reajuste de cartera que desplaza la LM a la izquierda, de modo que $E_1$ apenas mejora a $E_0$ en renta y la carga recae sobre el tipo de interés. Incluso sin efectos riqueza, la lectura monetarista de las pendientes invierte el veredicto de la Síntesis: con una inversión sensible al interés (IS tendida) y una demanda de dinero insensible (LM empinada, en el límite vertical), la geometría es la del caso clásico (\authorfont{Ver} Anexo \ref{anx1:extremos}) y la política fiscal solo sustituye gasto privado por público, el \textit{crowding-out} total, con el agravante probable de que el gasto público en inversión sea menos eficiente que el privado.

\subsection{La evidencia y los debates: de la \textit{Historia monetaria} a la ecuación perdida}\label{anx2:evidencia}

La contrarrevolución no se libró en el terreno de los axiomas sino en el de los datos, y su pieza mayor es \textit{A Monetary History of the United States, 1867-1960} (\authorfont{FRIEDMAN} y \authorfont{SCHWARTZ}, 1963): un siglo de evidencia de que los grandes movimientos de la renta nominal estadounidense vienen precedidos por movimientos de la cantidad de dinero de origen identificablemente autónomo. Su capítulo central reinterpreta la Gran Depresión como la \textit{Gran Contracción}: entre 1929 y 1933 la cantidad de dinero cayó más de un tercio, en medio de oleadas de quiebras bancarias a las que la Reserva Federal, nacida precisamente para evitarlas, asistió sin ejercer de prestamista de última instancia. Para los autores esa contracción de la liquidez fue la causa principal, no la única, del desastre (el otro factor determinante habría sido la ola proteccionista que hundió el comercio mundial), y la lección invierte la de la Síntesis: la Depresión no demuestra la impotencia del dinero, sino su poder, trágicamente ejercido en dirección contractiva ([\authorfont{Ver Tema 3.A.42}]).\footnote{%
    La implicación historiográfica es incendiaria: si la Depresión fue un fracaso del gobierno (del banco central) y no del mercado, el episodio fundacional que justificaba toda la arquitectura intervencionista de posguerra queda reescrito como el mejor argumento contra la discrecionalidad.
}

En paralelo, \authorfont{FRIEDMAN} y \authorfont{MEISELMAN} (1963) plantearon la batalla en forma de duelo de correlaciones: si la teoría keynesiana acierta, el multiplicador (la relación entre consumo y gasto autónomo) debe ser más estable que la velocidad (la relación entre renta y dinero); si acierta la cuantitativa, al revés. Su resultado, la velocidad gana en casi todos los subperiodos salvo los años treinta, provocó una réplica metodológica en toda regla, con \authorfont{ANDO} y \authorfont{MODIGLIANI} (1965) y \authorfont{DEPRANO} y \authorfont{MAYER} (1965) discutiendo la definición de gasto autónomo y la simultaneidad de las ecuaciones\textcolor{orange}{\textsuperscript{[Probable]}}. El asalto siguiente lo dio el banco de la Reserva Federal de San Luis, cuartel general monetarista: la ecuación de \authorfont{ANDERSEN} y \authorfont{JORDAN} (1968) regresa el crecimiento de la renta nominal sobre retardos del crecimiento monetario y de la acción fiscal, y encuentra que los coeficientes del dinero son grandes, significativos y duraderos, mientras que los fiscales son pequeños y se desvanecen en pocos trimestres\textcolor{orange}{\textsuperscript{[Probable]}}.\footnote{%
    La guerra era tanto de resultados como de método: ecuaciones reducidas de una sola línea contra los grandes modelos estructurales de tradición \authorfont{COWLES} que encarnaba el modelo MPS (la historia de esa econometría se desarrolla en el Anexo \ref{anx:autores}). La propia ecuación de San Luis acabó víctima de su éxito: la inestabilidad de la demanda de dinero de los setenta y la crítica de \authorfont{LUCAS} a las formas reducidas minaron su fiabilidad ([\authorfont{Ver Tema 3.A.6}]).
}

La contraofensiva teórica llegó de \authorfont{TOBIN} (1970), con título latino deliberadamente burlón: \textit{post hoc ergo propter hoc}. Que el dinero preceda a la renta no prueba que la cause: \authorfont{TOBIN} construye un modelo enteramente keynesiano en el que la oferta monetaria, endógena al sistema bancario, se anticipa sistemáticamente a la renta sin causarla. A ello se sumó la acusación de la \textit{caja negra}: la transmisión monetarista enumera sustituciones de cartera pero no exhibe una estructura contrastable eslabón a eslabón. Obligado a formalizarse, \authorfont{FRIEDMAN} (1970) presentó su marco teórico para el análisis monetario, y el debate subsiguiente con sus críticos, recogido en el volumen editado por \authorfont{GORDON} (1974), dejó a la vista la carencia que el propio \authorfont{FRIEDMAN} admitía: su teoría determina la renta nominal $P\,Y$, pero no contiene ninguna ecuación que reparta a corto plazo esa variación entre precios y cantidades. Es la famosa \textit{ecuación perdida} del monetarismo. La Síntesis había tapado ese mismo hueco con la curva de \authorfont{PHILLIPS} estable (\authorfont{Ver} Anexo \ref{anx1:phillips}); \authorfont{FRIEDMAN} sostiene que tal reparto no es cognoscible a priori, lo que convierte la ignorancia en argumento de política: si no se sabe cuánto de una expansión acabará en precios y cuánto en renta real, mejor no introducir perturbaciones monetarias. Y su respuesta constructiva, la única ecuación de reparto compatible con la racionalidad de los agentes, es la curva aumentada del epígrafe siguiente.

\subsection{La curva de \authorfont{PHILLIPS} aumentada: la tasa natural y el aceleracionismo}\label{anx2:phillips}

El blanco final de la contrarrevolución fue la pieza más usada del aparato de la Síntesis: el menú estable entre inflación y paro (\authorfont{Ver} Anexo \ref{anx1:menu}). La crítica de \authorfont{FRIEDMAN} (1968), anticipada de forma independiente y con microfundamentos más explícitos por \authorfont{PHELPS} (1967, 1968), no discute el ajuste empírico de la curva original, sino su lógica: el menú confunde variables nominales con reales y solo se sostiene si una de las dos partes del mercado de trabajo padece ilusión monetaria sistemática, un supuesto de irracionalidad inadmisible como fundamento de una política permanente.

El argumento se construye en el mercado de trabajo.\footnote{%
    Advertencia de notación: hasta aquí $W$ ha designado la riqueza; en este epígrafe, como es usual, $W$ es el salario nominal y $w=W/P$ el salario real.
} Los trabajadores ofrecen trabajo en función del salario real que \textit{esperan} obtener, porque lo que negocian es el salario nominal y el nivel de precios al que lo gastarán aún no se conoce: su variable es $W/P^{e}$. Las empresas, en cambio, conocen el precio de su propio producto y se sitúan siempre sobre su curva de demanda $L^{d}$, donde el salario real iguala la productividad marginal. Esa asimetría de información es todo el secreto de la curva de \authorfont{PHILLIPS}. Si el gobierno expande la demanda y la inflación supera a la esperada ($\pi>\pi^{e}$), los trabajadores sobreestiman su salario real y ofrecen más trabajo, mientras que las empresas, que ven caer el salario real efectivo, demandan más: el empleo sube de $L^{*}$ a $L_1$ en la figura, de $E^{*}$ a $E_1$, con un salario real percibido alto y uno efectivo bajo. Ambas partes están satisfechas solo porque una de ellas se equivoca.

\includegraphics{anx2_trabajo.pdf}\par\vspace{0.3em}
{\small\textit{Fuente: }Elaboración propia a partir de \authorfont{FRIEDMAN} (1968).}

El error no puede durar: los trabajadores forman sus expectativas según la HEA (\ref{anx2:cagan}) y las revisan cada periodo. En cuanto $\pi^{e}$ alcanza a $\pi$, la oferta de trabajo vuelve a su posición y el empleo regresa a $L^{*}$, al que corresponde una tasa de paro $U_n$ que \authorfont{FRIEDMAN} (1968) bautiza como \textit{tasa natural de desempleo}, en eco deliberado del tipo de interés natural de \authorfont{WICKSELL}\textcolor{orange}{\textsuperscript{[Probable]}}: la que genera el equilibrio del mercado de trabajo con sus fricciones reales (búsqueda, rotación, estructura del mercado, productividad, preferencias), independiente por completo de la política de demanda.\footnote{%
    Si $L_T$ es la población activa y $L$ el empleo, $U=(L_T-L)/L_T$ es la tasa de paro y $U_n$ la correspondiente a $L^{*}$, que incluye al menos un componente friccional. La literatura posterior acuñó términos vecinos pero no idénticos: la NIRU de \authorfont{MODIGLIANI} y \authorfont{PAPADEMOS} (1975) y, después, la NAIRU de la Nueva Economía Keynesiana, definidas como la tasa que no acelera la inflación, sin el compromiso de \authorfont{FRIEDMAN} con que esa tasa sea un equilibrio competitivo de mercado\textcolor{orange}{\textsuperscript{[Probable]}} ([\authorfont{Ver Tema 3.A.7}]). En Europa el término NAIRU suele asociarse a la NEK y en Estados Unidos al monetarismo.
} La relación empírica de \authorfont{PHILLIPS} sobrevive, pero aumentada con el término que sus usuarios habían omitido:

$$\pi_t=\pi^{e}_{t}-\alpha\,(U_t-U_n),\qquad \alpha>0$$

según la formulación de \authorfont{FRIEDMAN} (1968) y \authorfont{PHELPS} (1967). Hay una curva de corto plazo por cada nivel de expectativas, decreciente y transitable mientras el público se equivoca, y una curva de largo plazo vertical en $U_n$, porque a largo plazo $\pi^{e}=\pi$ y el término de sorpresa desaparece. La figura recoge la danza completa: la expansión sorprende y la economía recorre la curva de $A$ a $B$ (paso 1); las expectativas alcanzan a la inflación y la curva entera se desplaza hacia arriba, devolviendo el paro a $U_n$ en $C$ (paso 2); repetir la jugada exige más inflación ($C\to D\to E$), y la escalera sube por la vertical. De ahí la \textit{teoría aceleracionista de la inflación}: con expectativas adaptativas y $\lambda=1$,

$$\pi_t=\pi_{t-1}+\alpha\,(U_n-U_t)$$

mantener el paro por debajo de la tasa natural no exige inflación alta, sino inflación \textit{creciente}: cada periodo hereda la inflación anterior y añade una sorpresa nueva, como muestra el panel (b). El intercambio del menú existe, pero es transitorio (dura lo que tardan las expectativas en aprender) y su precio es acumulativo.\footnote{%
    A partir de ciertos niveles, además, la inflación deja de ser un simple numerario que sube: su volatilidad destruye la información de los precios relativos y el dinero empieza a incumplir sus funciones, hasta el punto de que la curva puede adquirir pendiente positiva, con más inflación y más paro a la vez, como argumentó \authorfont{FRIEDMAN} (1977) en su lección Nobel ([\authorfont{Ver Tema 3.A.35}]).
}

\includegraphics[width=\linewidth]{anx2_phillips.pdf}\par\vspace{0.3em}

La predicción se formuló en 1967-1968, con la curva original aún en su apogeo empírico, y la estanflación de los años setenta la confirmó fuera de muestra: la combinación de inflación y paro crecientes, imposible sobre el menú de la Síntesis, es exactamente lo que la curva aumentada predice para una demanda sobreestimulada cuyo público ha dejado de dejarse sorprender. Fue la victoria intelectual más limpia del monetarismo, y también el punto de partida de sus herederos: si los errores de expectativas son lo único que mueve el paro, la pregunta inmediata es por qué habrían de ser sistemáticos, y la respuesta de las expectativas racionales disolverá la curva de corto plazo entera ([\authorfont{Ver Tema 3.A.6}]).

\subsection{El corolario: retardos, reglas y el $k$ por ciento}\label{anx2:regla}

Las piezas anteriores (demanda estable, transmisión potente, tasa natural) no bastan por sí solas para condenar la política de estabilización: un timonel con un barco que responde podría, en principio, gobernarlo. El último argumento es temporal. Según los estudios de \authorfont{WARBURTON} (1966), \authorfont{FRIEDMAN} (1961) y \authorfont{FRIEDMAN} y \authorfont{SCHWARTZ} (1963), los efectos de la política monetaria llegan con retardos que pueden alcanzar los dieciocho meses y, lo que es peor, son muy variables de un episodio a otro. Una política contracíclica con retardos largos y variables corre el riesgo de aplicar el estímulo cuando la recesión ya pasó y la restricción cuando ya empezó: de ser, ella misma, la perturbación.\footnote{%
    \authorfont{FRIEDMAN} formalizó el punto en sus \textit{Essays in Positive Economics} (1953): si $x$ es el ciclo sin política y $p$ el efecto de la política, la varianza total es $\sigma^{2}_{x+p}=\sigma^{2}_{x}+\sigma^{2}_{p}+2\rho\,\sigma_{x}\sigma_{p}$, y la política solo estabiliza ($\sigma^{2}_{x+p}<\sigma^{2}_{x}$) si su correlación con el ciclo es suficientemente negativa: $\rho<-\sigma_{p}/(2\sigma_{x})$\textcolor{orange}{\textsuperscript{[Probable]}}. Con retardos variables, lograr esa sincronía negativa es precisamente lo difícil.
}

El corolario completo suma cuatro argumentos convergentes: la ignorancia del reparto a corto plazo (la ecuación perdida, \ref{anx2:evidencia}), la inexistencia de un intercambio explotable a largo plazo (la tasa natural, \ref{anx2:phillips}), los retardos largos y variables, y la economía política de la discrecionalidad (autoridades sometidas a presiones electorales que explotarán una y otra vez la curva de corto plazo).\footnote{%
    Este último argumento quedó enunciado de manera informal; su formalización como sesgo inflacionista de la política discrecional llegará con la inconsistencia temporal ([\authorfont{Ver Tema 3.A.6}]).
} De ahí la recomendación que condensa el monetarismo entero, propuesta por \authorfont{FRIEDMAN} en \textit{A Program for Monetary Stability} (1960)\textcolor{orange}{\textsuperscript{[Probable]}}: sustituir la discrecionalidad por una regla de crecimiento constante de la cantidad de dinero, el $k$ por ciento anual, calibrado para acomodar el crecimiento tendencial de la renta real y la deriva tendencial de la velocidad, de modo que la inflación tendencial sea nula.\footnote{%
    La regla tuvo un antecedente más intervencionista en el marco monetario-fiscal automático de \authorfont{FRIEDMAN} (1948), que la evolución posterior del autor fue despojando de componentes contracíclicos hasta dejar solo el crecimiento constante\textcolor{orange}{\textsuperscript{[Probable]}}. En el diseño operativo, los monetaristas recomiendan controlar la cantidad de dinero y no el tipo de interés, cuyo nivel es una guía engañosa del tono de la política (\ref{anx2:transmision}), e instrumentar ese control con operaciones de mercado abierto.
} La defensa de la regla no descansa en que el mercado sea perfecto, sino en una comparación de fallos: frente a un sector privado intrínsecamente estable (la lección conjunta de \ref{anx2:pih} y de la \textit{Historia monetaria}), un gobierno con poca información, instrumentos lentos e incentivos torcidos hace, en media, más daño que bien. La misma lógica llevó a \authorfont{FRIEDMAN} (1953), en su ensayo sobre los tipos de cambio flexibles, a abogar por la libre flotación de las monedas: solo con tipo de cambio flexible conserva un país el control de su cantidad de dinero para someterla a la regla, y el sistema de Bretton Woods, que lo impedía, debía terminar ([\authorfont{Ver Tema 3.B.20}]). El consejo encontró a la historia de su parte: la convertibilidad del dólar se suspendió en agosto de 1971 y la flotación generalizada llegó en marzo de 1973 ([\authorfont{Ver Tema 3.B.14}]). El paso de la regla a la práctica, los experimentos monetaristas de \authorfont{VOLCKER} y \authorfont{THATCHER} y su desenlace, se desarrolla en el Anexo \ref{anx:autores}.

\anexo{Autores, método y batallas: las dos escuelas por dentro}\label{anx:autores}

Los dos anexos anteriores construyen y desmontan un aparato analítico; este tercero se ocupa de las personas que lo hicieron, del método con el que trabajaron y de las batallas, metodológicas, historiográficas y de política práctica, en las que el duelo se libró de verdad. El orden va de los protagonistas a las ideas sobre el conocimiento (la economía positiva y la econometría), de ahí a las disputas sobre la propia historia de la macroeconomía (qué dijo \authorfont{KEYNES}, qué hizo \authorfont{MODIGLIANI} en 1944, si existió la tradición oral de Chicago) y termina donde las doctrinas se juegan el prestigio: los experimentos monetaristas de los años ochenta y su paradójico final.

\subsubsection{La Síntesis: un consenso sin líder}

A diferencia del monetarismo o de la Nueva Macroeconomía Clásica, en la Síntesis la existencia de un líder es problemática: hubo pioneros (\authorfont{HICKS} y, en los años inmediatamente posteriores a la \textit{Teoría General}, \authorfont{MODIGLIANI}) y hubo un apóstol, \authorfont{HANSEN}, cuyo seminario de política fiscal en Harvard formó a la generación que gobernaría la macroeconomía americana y cuyos manuales (\authorfont{HANSEN}, 1949, 1953) popularizaron el IS-LM como caballo de batalla de la escuela; pero no un jefe de filas, sino economistas que compartieron el objetivo de reconstrucción y acabaron compartiendo postulados. El consenso tuvo su vehículo de masas en el manual \textit{Economics} de \authorfont{SAMUELSON} (1948), el libro de texto dominante durante décadas, y su eje institucional en el MIT, Yale y la Comisión \authorfont{COWLES}. Los perfiles principales, limitados a los premios Nobel:
\begin{lnum}
    \item \authorfont{SAMUELSON}. Sus \textit{Foundations of Economic Analysis} (1947) fundaron la formalización matemática de la economía moderna. Su contribución específica a la Síntesis fue doble: el término mismo, acuñado en su manual, y la incorporación de la curva de \authorfont{PHILLIPS} al instrumental de política con \authorfont{SOLOW} en 1960 (\authorfont{Ver} Anexo \ref{anx1:menu}). Es el mejor ejemplo de la fusión que define a la escuela, microeconomía neoclásica para el largo plazo y keynesianismo para el corto: suyas son la preferencia revelada (con la que reconstruyó por vía inductiva el mismo conjunto axiomático de la teoría neoclásica de la demanda: la labor conciliadora, también en microeconomía), la condición de eficiencia en la provisión de bienes públicos, el modelo de consumo-ahorro con generaciones y aportaciones a la teoría de los mercados financieros eficientes y a la teoría neoclásica del comercio internacional;
    \item \authorfont{SOLOW}. Figura central de la Síntesis y de la teoría del crecimiento: su modelo de 1956 reformuló el de \authorfont{HARROD}-\authorfont{DOMAR} con supuestos neoclásicos, y sus estudios sobre las contribuciones del capital y la tecnología al crecimiento fundaron la contabilidad del crecimiento;
    \item \authorfont{HICKS}. Además del artículo de 1937, teorizó los equilibrios generales temporales, una suerte de microfundamentación de inspiración keynesiana de modelos no walrasianos; en microeconomía, la descomposición renta-sustitución de la demanda y el excedente del consumidor; en el mercado de trabajo, la \textit{Theory of Wages} (1932), base de la teoría neoclásica de la demanda de trabajo; en bienestar, los criterios de compensación; y en el ciclo, el modelo del acelerador ([\authorfont{Ver Tema 3.A.42}]);
    \item \authorfont{MODIGLIANI}. Su artículo de 1944 fue decisivo en el IS-LM canónico (\authorfont{Ver} Anexo \ref{anx1:modigliani}); dirigió después el modelo macroeconométrico más ambicioso de la tradición, el MPS, con 334 ecuaciones; y firmó el teorema de \authorfont{MODIGLIANI}-\authorfont{MILLER} sobre el valor de la empresa y la hipótesis del ciclo vital del consumo ([\authorfont{Ver Tema 3.A.29}]);
    \item \authorfont{TOBIN}. Su artículo de 1958 sobre la preferencia por la liquidez como conducta frente al riesgo dio microfundamentos a la demanda de dinero y de activos de la Síntesis, e influyó en el desarrollo del modelo de valoración de activos financieros ([\authorfont{Ver Tema 3.B.23}]); suyas son también la $q$ de \authorfont{TOBIN} y la propuesta de imposición sobre las conversiones de divisas para desincentivar los flujos especulativos;
    \item \authorfont{MEADE}. Autor de una de las formalizaciones de 1937 (\authorfont{Ver} Anexo \ref{anx1:hicks}), destacó después en la economía internacional, extendiendo el modelo keynesiano a la economía abierta y analizando los efectos de las políticas comerciales sobre el bienestar; y,
    \item \authorfont{KLEIN}. Nobel en 1980 por la construcción de modelos econométricos de coyuntura y su aplicación al análisis de la política económica; su modelo con \authorfont{GOLDBERGER} (1955) fue la primera representación empírica del sistema keynesiano (\ref{anx3:econometria}).
\end{lnum}

\subsubsection{Chicago y sus aliados}

El monetarismo sí tuvo cabeza visible. \authorfont{FRIEDMAN} empezó su carrera en la estadística y la reflexión metodológica, y desde su cátedra en Chicago\footnote{%
    Junto con \authorfont{STIGLER} y \authorfont{BECKER}, \authorfont{FRIEDMAN} cuenta incluso con un pequeño museo dentro del edificio de la facultad de economía de la Universidad de Chicago, de la que sigue siendo uno de los profesores más reconocidos.
} organizó el taller de dinero y banca del que salió el volumen de 1956; en 1957 publicó la teoría de la renta permanente, en los años posteriores completó con otros autores la contrarrevolución, y el reconocimiento llegó con el Nobel de 1976. En torno a él, \authorfont{SCHWARTZ} (coautora de la \textit{Historia monetaria} desde el NBER), \authorfont{CAGAN}, \authorfont{BRUNNER} (que acuñó el término \textit{monetarismo} en 1968) y \authorfont{MELTZER} (con \authorfont{BRUNNER}, fundador del comité en la sombra que auditaba públicamente a la Reserva Federal)\textcolor{orange}{\textsuperscript{[Probable]}}, \authorfont{H. JOHNSON} (que bautizó la \textit{contrarrevolución} y fue a la vez su crítico más incisivo, \ref{anx3:batallas}) y, como difusores de la segunda generación, \authorfont{LAIDLER} (demanda de dinero) y \authorfont{FRENKEL} (enfoque monetario de la balanza de pagos, [\authorfont{Ver Tema 3.B.11}]). Conviene deshacer un equívoco frecuente: \authorfont{PHELPS}, que suele colarse en estas listas por su papel en la tasa natural, no perteneció a Chicago ni fue monetarista; llegó al resultado de forma independiente y con una microfundamentación explícita de la información y la búsqueda que, con el tiempo, alimentaría antes a la Nueva Economía Keynesiana que al monetarismo ([\authorfont{Ver Tema 3.A.7}]).

\subsection{La economía positiva: F53 y la controversia \authorfont{FRIEDMAN}-\authorfont{MYRDAL}}\label{anx3:positiva}

La batalla por el método precede a la batalla por los modelos. \authorfont{FRIEDMAN} fijó su posición en el ensayo que abre sus \textit{Essays in Positive Economics} (1953), conocido como F53: el propósito de la ciencia económica es formular hipótesis que produzcan predicciones contrastables sobre problemas concretos, y las proposiciones refutadas empíricamente deben rechazarse, una traslación a la economía del criterio de \authorfont{POPPER}. De ahí su tesis más polémica, la \textit{irrelevancia de los supuestos}: una teoría se juzga por el poder predictivo de sus conclusiones y no por el realismo de sus supuestos.\footnote{%
    El ejemplo clásico de \authorfont{FRIEDMAN} es la maximización de beneficios: aunque ninguna empresa calcule literalmente ingresos y costes marginales, el economista puede trabajar \textit{como si} lo hicieran, porque la competencia elimina a las que se comportan de otro modo. La réplica más citada es la de \authorfont{SAMUELSON}, que motejó el argumento de \textit{F-twist}: si los supuestos son falsos, lo que la teoría predice bien lo predice por accidente\textcolor{orange}{\textsuperscript{[Probable]}}. El desarrollo completo de F53 y de sus críticas corresponde al tema de metodología ([\authorfont{Ver Tema 3.A.1}]).
} Sobre el estatuto de la disciplina sostuvo la controversia \authorfont{FRIEDMAN}-\authorfont{MYRDAL}, que marcó el punto álgido del debate entre economía positiva y normativa:
\begin{la}
    \item Para \authorfont{MYRDAL} (1953), los principales conceptos económicos están cargados implícitamente de juicios de valor, y la pretensión de neutralidad es ella misma una posición política\textcolor{orange}{\textsuperscript{[Probable]}}; mientras que,
    \item \authorfont{FRIEDMAN} defendió la prevalencia de la economía positiva, independiente de cualquier postura ética, y supeditó a sus resultados el único papel legítimo de la normativa.\footnote{%
        El propio \authorfont{FRIEDMAN} ejemplifica la secuencia: su defensa normativa de que las autoridades mantengan controlado el crecimiento de la oferta monetaria se deriva de sus estudios positivos sobre la primacía del dinero como determinante de la inflación.
    }
\end{la}

Hay una capa más honda que el falsacionismo, señalada por la historiografía: la divisoria entre una concepción \textit{marshalliana} de la teoría (un motor de análisis parcial, pegado al problema y a los datos, la de \authorfont{FRIEDMAN} y el NBER) y una concepción \textit{walrasiana} (el sistema completo de ecuaciones simultáneas, la de \authorfont{COWLES} y la Síntesis), según la lectura de \authorfont{DE VROEY} (2016)\textcolor{orange}{\textsuperscript{[Probable]}}. Esa divisoria explica mejor que ninguna otra por qué cada bando consideró el método del contrario no ya erróneo sino inadmisible, y reaparece en cada batalla de este anexo. La ironía es que F53 sirve a ambos: el \textit{como si} que licencia la maximización licencia igualmente las funciones agregadas estables del IS-LM.

\subsection{La econometría estructural y su contramodelo}\label{anx3:econometria}

Los economistas de la Síntesis emplearon mayor formalización matemática que los monetaristas y las demás escuelas coetáneas, y entre los años cuarenta y setenta esa apuesta cristalizó en la econometría, la herramienta que prometía refutar teorías. La econometría puede considerarse la confluencia de dos corrientes, los economistas con inclinación matemática y los estadísticos con aficiones económicas, y a partir de los años treinta arraigó en Estados Unidos, donde la mentalidad pragmática le ofreció campo propicio.\footnote{%
    Entre los precursores destacaron los ingleses \authorfont{PETTY}, \authorfont{GREGORY KING}, \authorfont{JEVONS}, \authorfont{EDGEWORTH} y \authorfont{MARSHALL}; los franceses \authorfont{QUESNAY} y \authorfont{COURNOT}; el suizo \authorfont{DANIEL BERNOULLI}; el italiano \authorfont{PARETO}; el alemán \authorfont{VON THÜNEN}; y el norteamericano \authorfont{FISHER}.
} Además de los fundadores del NBER y de la Sociedad de Econometría, se distinguieron como económetras \authorfont{MOORE} (1869-1958),\footnote{%
    Sobre todo por la deducción estadística de curvas de demanda y sus elasticidades. Aunque otros autores habían trabajado con curvas matemáticas de oferta y demanda, \authorfont{SCHUMPETER} (1954) opina que \authorfont{CHEYSSON} en 1887, \authorfont{LEHFELDT} en \textit{La elasticidad de la demanda del trigo} (1914) y \authorfont{MOORE} fueron los primeros que, tras un paréntesis de unos doscientos años, continuaron la tarea emprendida por \authorfont{GREGORY KING}. Sus obras más importantes: \textit{Las leyes de los salarios} (1911), \textit{Ciclos económicos: su ley y su causa} (1914), \textit{Predicción de la cosecha y el precio del algodón} (1917), \textit{Génesis de ciclos económicos} (1923) y \textit{Economía sintética} (1929).
} \authorfont{SCHULTZ} (1893-1938),\footnote{%
    Alumno de \authorfont{MOORE} en Columbia, continuó la labor de su maestro en colaboración con estadísticos: \textit{Leyes estadísticas de demanda y oferta} (1936), \textit{Teoría y medición de la demanda} (1938) y su artículo sobre la interrelación entre demanda, precios y renta.
} \authorfont{DOUGLAS} (1892-1976)\footnote{%
    Estudiante en Columbia en la época de \authorfont{MOORE} y \authorfont{J. B. CLARK}, ejerció en Chicago, donde siguió las investigaciones de \authorfont{MOORE} sobre la distribución de la renta y la comprobación estadística de las productividades marginales; su obra econométrica mayor es \textit{La teoría de los salarios} (1934). Con el matemático \authorfont{COBB} había publicado \textit{Una teoría de la producción} (1928), usando una variante de la función de producción social de \authorfont{WICKSELL} con dos factores, la función \authorfont{COBB}-\authorfont{DOUGLAS}, en particular la de coeficientes $\beta=1-\alpha$, con una distribución estimada de la renta de un tercio para el capital y dos tercios para el trabajo.
} y \authorfont{LEONTIEF} (1906-1999).\footnote{%
    Profesor de Harvard e investigador del NBER a su llegada a Estados Unidos, imprescindible por sus tablas sectoriales input-output; a partir de la Segunda Guerra Mundial los avances econométricos se aprovecharon para la investigación estadística de los coeficientes de sus tablas.
}

Sobre ese suelo se levantaron dos programas rivales de investigación empírica, y la divisoria de \ref{anx3:positiva} se volvió institucional. El programa que heredó la Síntesis fue el de la Comisión \authorfont{COWLES}: \authorfont{HAAVELMO} (1944) dio a la econometría su fundamento probabilístico y la comisión desarrolló la identificación y estimación de \textit{sistemas de ecuaciones simultáneas} estructurales, el molde matemático exacto del IS-LM. El programa rival era el del NBER de \authorfont{BURNS} y \authorfont{MITCHELL} (1946), medición minuciosa de ciclos sin modelo formal previo, y el choque quedó retratado en la reseña de \authorfont{KOOPMANS} (1947), que acusó al NBER de \textit{medir sin teoría}. \authorfont{FRIEDMAN}, formado precisamente en el NBER, quedó del lado de los acusados, y toda su obra empírica posterior, de la \textit{Historia monetaria} a las ecuaciones de una sola línea de San Luis (\authorfont{Ver} Anexo \ref{anx2:evidencia}), es la contestación marshalliana a la econometría estructural walrasiana.

La vía estructural produjo sus frutos en cadena: \authorfont{TINBERGEN} construyó el primer modelo nacional, para Holanda, en 1936\textcolor{orange}{\textsuperscript{[Probable]}}, y lo extendió a Estados Unidos y Reino Unido en sus trabajos para la Sociedad de Naciones (\authorfont{TINBERGEN}, 1939); \authorfont{KLEIN} y \authorfont{GOLDBERGER} (1955) crearon, con 63 variables, la primera representación empírica del sistema keynesiano; y los modelos se ampliaron hasta incluir cientos de ecuaciones, cada una desarrollando un aspecto estructural o el comportamiento de un agente, con cumbre en el modelo MPS de \authorfont{MODIGLIANI} y sus colaboradores para la Reserva Federal, con 334 ecuaciones. La popularidad de estos modelos se derrumbó en los años setenta por un triple golpe: los shocks del petróleo invalidaron sus predicciones; la crítica de \authorfont{LUCAS} mostró que sus coeficientes estimados no eran parámetros exógenos sino variables endógenas a la política, al no recoger el comportamiento optimizador de los agentes; y los modelos de series temporales de la familia ARMA demostraron mayor capacidad predictiva con una fracción del coste ([\authorfont{Ver Tema 3.A.6}]).\footnote{%
    Esto no implica que la corriente naciente rechazara las matemáticas, nada más lejos: lo que exigió fue un elemento nuevo, la microfundamentación de los modelos, formalizada también matemáticamente.
}

\subsection{Las batallas por la historia}\label{anx3:batallas}

El duelo se libró también sobre el pasado: quién era el verdadero heredero de \authorfont{KEYNES}, qué pretendía el artículo fundacional de la Síntesis y si la teoría de \authorfont{FRIEDMAN} tenía los ancestros que alegaba. Las tres disputas comparten moraleja: en macroeconomía, las genealogías son armas.

\subsubsection{Qué dijo \authorfont{KEYNES}: el reparto de la herencia}

La \textit{Teoría General} admite lecturas rivales, y la taxonomía clásica de las interpretaciones es la de \authorfont{CODDINGTON} (1976)\textcolor{orange}{\textsuperscript{[Probable]}}: el \textit{keynesianismo hidráulico}, que lee la obra como un sistema de funciones agregadas estables entre las que fluye el gasto y que la política puede manipular, y que es exactamente la lectura de la Síntesis y de su aparato (Anexo \ref{anx:aparato}); el \textit{fundamentalismo}, de \authorfont{ROBINSON} o \authorfont{SHACKLE}, para el que el corazón del libro es la incertidumbre radical del capítulo 12 y toda formalización de equilibrio lo traiciona, lectura de la que descienden los postkeynesianos ([\authorfont{Ver Tema 3.A.4}]); y el \textit{reduccionismo reconstituido} de \authorfont{CLOWER} y \authorfont{LEIJONHUFVUD}, que busca los microfundamentos del desequilibrio, con procesos de intercambio fuera del equilibrio walrasiano ([\authorfont{Ver Tema 3.A.4}]). El manifiesto de esta tercera vía, \authorfont{LEIJONHUFVUD} (1968), popularizó la distinción entre la \textit{economía keynesiana} (lo que la Síntesis enseñaba en su nombre) y la \textit{economía de \authorfont{KEYNES}} (lo que el libro decía), con la tesis incómoda de que ambas cosas llevaban treinta años divergiendo.

\subsubsection{Qué hizo \authorfont{MODIGLIANI} en 1944: la génesis revisada}

La lectura canónica del artículo de 1944 (\authorfont{Ver} Anexo \ref{anx1:modigliani}) lo presenta como el primer paso de la restauración prekeynesiana: la rigidez salarial como concesión que reconcilia a \authorfont{KEYNES} con los clásicos. La reconstrucción historiográfica reciente de su génesis, debida a \authorfont{RANCAN}\textcolor{orange}{\textsuperscript{[Suposición]}}, sostiene otra cosa: \authorfont{MODIGLIANI} no introdujo la rigidez para conciliar los dos sistemas, sino como un dispositivo analítico conveniente y un supuesto realista con el que mostrar el funcionamiento de una economía monetaria, en la que las variables reales dependen de la cantidad de dinero; y la usó además para demostrar la coherencia del desempleo con una posición de equilibrio estable, evitando que el \textit{caso keynesiano} de preferencia por la liquidez degenerase en una deflación indefinida. Bajo esta lectura, el artículo no se centra en la relación entre rigidez y desempleo, sino en el papel crucial del dinero en la determinación de las variables reales, y su atención al origen monetario del desempleo contrasta con el descuido del aspecto monetario, y la inclinación fiscal, de la Síntesis que lo adoptó como fundación. La celebrada \textit{esquizofrenia} entre desequilibrio a corto y equilibrio a largo (\authorfont{Ver} Anexo \ref{anx1:saldos}) tampoco está en el texto de 1944: es una relación que \authorfont{MODIGLIANI} no discute entonces, aunque en interpretaciones posteriores de su propio artículo presentara el desempleo como característica sistemática de una economía que depende del dinero para realizar transacciones.

\subsubsection{La tradición oral de Chicago: \authorfont{PATINKIN} contra \authorfont{FRIEDMAN}}

El \textit{Restatement} de 1956 se presentó como destilación de una tradición oral enseñada en Chicago en los años treinta (\authorfont{Ver} Anexo \ref{anx2:restatement}). \authorfont{PATINKIN} (1969), que estudió en Chicago en aquella década, cotejó la alegación con los apuntes y textos de la época y concluyó que la teoría cuantitativa que allí se enseñaba era fisheriana, centrada en la velocidad y partidaria incluso de la política monetaria contracíclica, y no contenía nada parecido a una teoría de la demanda de dinero como elección de cartera: eso, sostiene, procede de \authorfont{KEYNES} y de la teoría de la preferencia por la liquidez que el propio \authorfont{FRIEDMAN} generalizó. La acusación es elegante y venenosa: el \textit{Restatement} sería análisis keynesiano navegando bajo bandera cuantitativista, y la tradición oral, una genealogía inventada para dotar de pedigrí ortodoxo a la innovación. \authorfont{JOHNSON} (1971) completó la anatomía del episodio en su conferencia sobre la revolución keynesiana y la contrarrevolución monetarista: para triunfar, explicó, una (contra)revolución necesita atacar la proposición central del consenso vigente, ofrecer una apariencia de continuidad respetable con la ortodoxia anterior y proporcionar a los investigadores jóvenes una metodología nueva con la que producir resultados propios; el monetarismo, concluyó, cumplió el guion punto por punto. Que el autor del diagnóstico militara en el bando monetarista da la medida de la independencia intelectual del personaje, y de hasta qué punto la escuela fue consciente de su propia retórica.

\subsection{El divulgador y el político}\label{anx3:divulgador}

\authorfont{FRIEDMAN} fue tanto un economista académico como un divulgador al gran público de las virtudes del capitalismo y del liberalismo económico: entre 1935 y 2005 publicó 18 libros técnicos y 14 para el público general, 104 artículos técnicos y 192 artículos, audiencias y conferencias destinadas al gran público, sin contar sus innumerables columnas en prensa. \authorfont{DE VROEY} propone separar su carrera en tres periodos: hasta mediados de los cincuenta, metodología, estadística y primeras contribuciones monetarias; hasta los setenta, los años más creativos, del \textit{Restatement} a la \textit{Historia monetaria}; y desde entonces, la divulgación, con \textit{Capitalism and Freedom} (\authorfont{FRIEDMAN}, 1962), las columnas quincenales en \textit{Newsweek} y la serie de televisión y libro \textit{Free to Choose} (\authorfont{FRIEDMAN} y \authorfont{FRIEDMAN}, 1980). Esa segunda faceta es indudablemente controvertida y, en opinión de muchos economistas coetáneos, ensombreció su legado y afectó a su credibilidad: su ardiente defensa del libre mercado, en un contexto de expansión del gasto público y de inflación entre 1965 y 1991 aproximadamente, le llevó a posiciones muy cercanas al ala más conservadora del Partido Republicano estadounidense y a jugar un papel, a través de sus discípulos, en el régimen militar de \authorfont{PINOCHET} en Chile; en ciencia política se le considera uno de los padres del \textit{neoliberalismo}. El contraste con la Síntesis es en sí mismo instructivo: sus rivales (\authorfont{SAMUELSON} en su manual y en su columna, \authorfont{TOBIN} como consejero de \authorfont{KENNEDY}) ejercieron una influencia pública igual de deliberada, pero desde dentro de las instituciones del consenso; \authorfont{FRIEDMAN} la ejerció como insurgente, y la historia del péndulo ideológico de los setenta es, en parte, la historia de esa diferencia de posición.

\subsection{El monetarismo en la práctica: experimentos y final}\label{anx3:practica}

La estanflación dio al monetarismo la razón teórica (\authorfont{Ver} Anexo \ref{anx2:phillips}) y los años ochenta le dieron la oportunidad práctica; el resultado fue una victoria doctrinal conseguida al precio de la doctrina misma.

Dentro de la escuela convivían dos recetas de desinflación: el gradualismo de \authorfont{FRIEDMAN}, una reducción anunciada y paulatina del crecimiento monetario que diera tiempo a las expectativas a ajustarse\textcolor{orange}{\textsuperscript{[Probable]}}, y la contracción de choque, defendida en los setenta como mal menor frente a una inflación enquistada: resolver el problema de una vez, aun a costa de provocar una recesión, para permitir después un crecimiento sostenible. La práctica fue más dura que ambas recetas. En Estados Unidos, \authorfont{VOLCKER} anunció en octubre de 1979 un cambio de procedimiento operativo de inspiración monetarista, el control de las reservas no prestadas en lugar del tipo de interés, y la desinflación resultante llevó la inflación de cifras de dos dígitos en 1980 a alrededor del 3\% en 1983, a costa de la recesión de 1981-82, la más profunda desde los años treinta, y de una volatilidad extrema de los tipos; los propios agregados monetarios, convertidos en objetivo, se volvieron erráticos durante el experimento\textcolor{orange}{\textsuperscript{[Probable]}}. En el Reino Unido, la estrategia financiera a medio plazo de \authorfont{THATCHER} (1980) fijó sendas objetivo para el agregado amplio, que se desbordaron sistemáticamente mientras la economía se hundía y la libra se apreciaba, con el paro por encima de los tres millones\textcolor{orange}{\textsuperscript{[Probable]}}.

La explicación del fiasco instrumental la habían dado, por adelantado, un banquero central y un econométra. \authorfont{GOODHART} (1975) formuló la ley que lleva su nombre: toda regularidad estadística observada tiende a desmoronarse en cuanto se la somete a presión con fines de control, porque los agentes y las instituciones financieras se reorganizan alrededor del objetivo\textcolor{orange}{\textsuperscript{[Probable]}}. Y \authorfont{GOLDFELD} (1976) documentó el \textit{caso del dinero desaparecido}: las funciones de demanda de dinero estimadas, estables durante décadas, empezaron a sobrepredecir sistemáticamente los saldos desde mediados de los setenta, cuando la innovación financiera y la desregulación multiplicaron los sustitutos del dinero. La premisa empírica central del monetarismo, la estabilidad de la demanda de dinero (\authorfont{Ver} Anexo \ref{anx2:restatement}), falló exactamente cuando más se la necesitaba, y a mediados de los ochenta los bancos centrales fueron abandonando los objetivos de agregados\textcolor{orange}{\textsuperscript{[Probable]}}.

El final es paradójico y conviene enunciarlo con precisión: el monetarismo murió como doctrina operativa y venció como consenso. De la contrarrevolución quedaron la prioridad de la estabilidad de precios, la independencia de los bancos centrales frente al ciclo político, la preferencia por reglas frente a discrecionalidad y la tasa natural como frontera de lo que la demanda puede conseguir; lo que desapareció fue el dinero mismo como variable de control, sustituido por objetivos directos de inflación instrumentados con el tipo de interés. Casi todos sus miembros engrosaron las filas de la Nueva Macroeconomía Clásica, que repudió el instrumental keynesiano que el monetarismo aún compartía ([\authorfont{Ver Tema 3.A.6}]), y no hay aportaciones significativas de tono monetarista estricto al debate del siglo XXI: la escuela se disolvió en su propia victoria.
