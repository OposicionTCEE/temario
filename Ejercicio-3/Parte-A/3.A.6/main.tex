    % --- Imagen adaptativa ---
    % \imagenfit[alto]{fichero}{título}{fuente}: la imagen queda fija entre los párrafos donde se escribe (no flota), con su
    % título en el índice de figuras y la fuente debajo. Sin [alto]: las figuras PDF (TikZ) salen al tamaño con que se
    % dibujaron y solo se reducen si no caben en el ancho o en el alto de la página; las PNG/JPG ocupan el ancho del texto
    % hasta \imgaltoras del alto de la página. Con [alto] (fracción del alto de la página; [1] = página entera) se fija el
    % tamaño de esa imagen. Si no cabe en lo que queda de página, pasa entera a la siguiente.
    % \imagenfithorizontal{fichero}{título}{fuente}: la imagen en una página girada (apaisada), en ese mismo punto del texto.
    % Reglas: main/GUIA_TEMAS.md, §4 (Figuras).
    \newdimen\imgcap
    \newdimen\imgh
    \newdimen\imgw
    \newdimen\imgneed
    \newsavebox\imgcaja
    \newcommand{\imgaltoras}{0.30}% alto por defecto de las PNG/JPG (fracción del alto de la página)
    \makeatletter
    \newcommand{\imagenfit}[4][]{%
      \addvspace{10pt}%
      \par\begingroup
        \imgcap=3.5\baselineskip% alto aproximado de título + fuente
        \imgw=\linewidth
        \filename@parse{#2}\def\img@pdf{pdf}%
        \if\relax\detokenize{#1}\relax
          \ifx\filename@ext\img@pdf
            \sbox\imgcaja{\includegraphics{\detokenize{#2}}}% tamaño natural
            \ifdim\wd\imgcaja<\imgw \imgw=\wd\imgcaja \fi
            \imgh=\ht\imgcaja
          \else
            \imgh=\imgaltoras\textheight
          \fi
        \else
          \imgh=#1\textheight
        \fi
        \dimen@=\textheight \advance\dimen@ by -\imgcap
        \ifdim\imgh>\dimen@ \imgh=\dimen@ \fi
        \sbox\imgcaja{\includegraphics[width=\imgw,height=\imgh,keepaspectratio]{\detokenize{#2}}}%
        \imgneed=\ht\imgcaja \advance\imgneed by \imgcap
        \Needspace{\imgneed}%
        \noindent\begin{minipage}{\textwidth}
          \centering
          \captionof{figure}{\textemdash\ #3}\par\vspace{0.3em}% título arriba
          \usebox\imgcaja\par
          \vspace{0.3em}\small\textit{Fuente: }#4% fuente abajo
        \end{minipage}%
      \endgroup
      \par\addvspace{10pt}\RequestRestore
    }
    \newcommand{\imagenfithorizontal}[3]{%
      \begin{landscape}
        \centering
        \captionof{figure}{\textemdash\ #2}\par\vspace{0.3em}%
        \includegraphics[width=\linewidth,height=\dimexpr\textheight-4\baselineskip\relax,keepaspectratio]{\detokenize{#1}}\par
        \vspace{0.3em}{\small\textit{Fuente: }#3}%
      \end{landscape}%
    }
    \makeatother

    % \anexo{Título}: un anexo equivale a un \section (su título sale del mismo tamaño). Dentro se usan \subsection,
    % \subsubsection y \paragraph, nunca \section: se numeran I.1, I.1.1, I.1.1.1 y las páginas del anexo, I-1, I-2…
    % En el índice de contenido inicial solo figura el nombre del anexo. Sus apartados forman el índice del anexo, que sale
    % solo si el anexo los tiene, justo después de la introducción del anexo (antes del primer \subsection).
    % \anexo*{Título}: el mismo anexo, sin ese índice. Reglas: main/GUIA_TEMAS.md, §6.

    \newcommand{\anx@empezar}[2]{%
      \clearpage
      \refstepcounter{anexo}%
      \global\EnAnexotrue
      \setcounter{page}{1}\gdef\thepage{\theanexo-\arabic{page}}%
      \setcounter{section}{\numexpr\value{anexo}-1\relax}\stepcounter{section}% el anexo hace de \section
      \gdef\thesection{\Roman{section}}\gdef\theHsection{anexo.\Roman{section}}%
      \phantomsection
      \protected@xdef\leq@current@section{Anexo \theanexo.- \unexpanded{#2}}\gdef\leq@current@subsection{}% índice de ecuaciones
      \addtocontents{stc}{\protect\contentsline{section}{Anexo \theanexo.- #2}{}{\@currentHref}}% índice inicial: solo el nombre, sin página
      \addcontentsline{toc}{section}{Anexo \theanexo.- #2}% marcadores del PDF
      {\noindent\normalfont\Large\bfseries Anexo \theanexo.- #2\par}\nobreak\vspace{1.5ex \@plus .2ex}%
      #1\@ifundefined{anx@lista@\theanexo}{\global\anx@pendientefalse}{}%
    }
    % Índice del anexo: cada apartado escribe una línea en el fichero .atc (anexo, nivel, texto, página, enlace); al
    % empezar el documento se leen las de la compilación anterior y se guardan por anexo (dos pasadas, como cualquier índice).
    % En ese índice los apartados suben un nivel, porque el anexo hace de \section.
    \def\anx@entrada#1#2#3#4#5{\@ifundefined{anx@lista@#1}{\global\@namedef{anx@lista@#1}{}}{}%
      \expandafter\g@addto@macro\csname anx@lista@#1\endcsname{\contentsline{#2}{#3}{#4}{#5}}}
    \tcbuselibrary{breakable}
    \def\anx@indice{\ifanx@pendiente\global\anx@pendientefalse

            before upper={\parindent=0pt\parskip=2pt}]
          \small
          % números más anchos que en el índice inicial (I.1, I.1.1, I.1.1.1; páginas I-1)
          \def\@pnumwidth{2.6em}\def\@tocrmarg{3.4em}%
          \def\l@section##1##2{\addpenalty\@secpenalty\addvspace{1.0em \@plus\p@}\setlength\@tempdima{2.6em}%
            \begingroup\parindent\z@\rightskip\@pnumwidth\parfillskip-\@pnumwidth\leavevmode\bfseries
            \advance\leftskip\@tempdima\hskip-\leftskip##1\nobreak\hfil\nobreak\hb@xt@\@pnumwidth{\hss##2\kern-\p@\kern\p@}\par\endgroup}%

          \csname anx@lista@\theanexo\endcsname

      \fi}
    \def\anx@escribir#1#2#3{\ifEnAnexo

    % margen superior, así que no cambia las páginas del tema. Se enciende con \pagenumbering{arabic} y se apaga con

    \cs_new_protected:Npn \cab_guardar:n #1
      {
        \tl_gset:Nn \g_cab_titulo_tl {#1}
        \regex_replace_once:nnN { \A \s* Tema \s+ \S+ \s* \-\-? \s* } { } \g_cab_titulo_tl % sin «Tema X.Y.N --»
        \regex_replace_once:nnN { \.? \s* \c{\\} (\[ [^\]]* \])? \s* (\c{large} \s*)? } { .\  } \g_cab_titulo_tl % título. subtítulo
        \regex_replace_once:nnN { [\.\s]* \Z } { } \g_cab_titulo_tl % sin subtítulo: nada detrás
      }

\section{La revolución: creencias y método}\label{b1}

\puntosclave{
    La estanflación y el fracaso predictivo de los grandes modelos macroeconométricos cierran el resquicio que el monetarismo había dejado abierto: con expectativas que solo miran hacia atrás, la política vivía del error ajeno, y en los setenta el error dejó de cooperar.

    Tres creencias definen la escuela: los agentes optimizan siempre, los mercados se vacían siempre y el mundo es aprendible. La hipótesis de las expectativas racionales (MUTH, 1961) exige coherencia, no clarividencia: el error de previsión existe, pero no puede ser sistemático, de modo que una política que se repite deja de sorprender.

    El método convierte las creencias en un formato: microfundamentos, expectativas racionales y vaciado continuo, el molde DSGE en que se escribe la macroeconomía desde entonces. La escuela vence completando el programa walrasiano que la propia Síntesis decía profesar.
}

\subsection{El resquicio y la estanflación: ¿por qué se rompe el consenso?}\label{b1:resquicio}

El duelo entre la Síntesis Neoclásica y el monetarismo se cerró con una victoria parcial de \authorfont{FRIEDMAN}: la curva de \authorfont{PHILLIPS} no es un menú estable y, a largo plazo, el paro vuelve a su tasa natural ([\authorfont{Ver Tema 3.A.5}]). Pero esa victoria dejó una rendija abierta, y la rendija estaba en el supuesto sobre el que se apoyaba: las \textbf{expectativas adaptativas}. Si los agentes aprenden mirando hacia atrás, el gobierno puede ir siempre un paso por delante; mientras acelere la inflación, la sorpresa se renueva. La política perdía el largo plazo y conservaba el corto, que es donde se gobierna.

Los años setenta cerraron esa rendija por la vía de los hechos, antes de que nadie la cerrara con un argumento.\footnote{%
    El telón de fondo lo pusieron el embargo petrolero de la OPEP de 1973, que cuadruplicó el precio del crudo, y el segundo shock de la Revolución iraní de 1979; la inflación estadounidense tocó su máximo en 1980. La descomposición del sistema de Bretton Woods, en 1971, había añadido la volatilidad de los tipos de cambio, y aquel contexto alimentó también la traslación política hacia el monetarismo ([\authorfont{Ver Tema 3.A.5}]).
}

\imagenfit{b1_estanflacion.pdf}{Inflación y paro en Estados Unidos, 1960-1985}{Elaboración propia con datos de la Reserva Federal de San Luis (IPC y tasa de paro, Oficina de Estadísticas Laborales).}

Durante los años sesenta la disyuntiva pareció una regularidad firme: nueve observaciones que se alinean sobre una curva descendente y que invitaban a leerla como un menú de política.\footnote{%
    El ajuste de trazos es una regresión de la inflación sobre la inversa de la tasa de paro con los datos de 1961 a 1969, que explica el $81\%$ de su variación; con datos anuales y nueve observaciones, la relación resultaba difícil de discutir. Las series son el IPC de todos los consumidores urbanos y la tasa de paro civil, ambas en media anual, y el libro de datos del tema recoge los valores y el tratamiento.
} A partir de 1970 el recorrido se desordena, gira y acaba lejos, arriba y a la derecha: \textbf{inflación y paro suben a la vez}. No es que la disyuntiva se hubiera vuelto menos favorable; es que había dejado de existir como relación utilizable.

El segundo golpe fue predictivo. Los grandes modelos macroeconométricos, que eran el instrumento de gobierno de la Síntesis, fallaron justo cuando más falta hacían, y lo hicieron de forma sistemática: predecían una caída de la inflación que no llegaba.\footnote{%
    Su arquitectura tenía cuatro debilidades reconocidas. Carecían de \textit{principios de comportamiento}: pese a los esfuerzos de los años sesenta por microfundamentar por separado el consumo (\authorfont{ANDO} y \authorfont{MODIGLIANI}, 1963), la inversión (\authorfont{JORGENSON}, 1967) y la demanda de dinero (\authorfont{BAUMOL}, 1952; \authorfont{TOBIN}, 1956), el conjunto seguía siendo un sistema de ecuaciones \textit{ad hoc} incompatible con un comportamiento optimizador general. Eran \textit{estáticos}: la dimensión temporal de las decisiones se añadía después, con retardos elegidos por su ajuste estadístico. Trataban las \textit{expectativas} como miopes o adaptativas, es decir, como algo que no responde a la política. Y razonaban en \textit{equilibrio parcial}, con la cláusula \textit{ceteris paribus} aplicada a todo lo que no se estaba mirando. Una referencia en español de la época sobre aquellos modelos: \href{https://repositorio.uam.es/entities/publication/256ba000-dcff-41e2-8801-40b03ad037f7}{\authorfont{ESPASA}\textcolor{orange}{\textsuperscript{[Probable]}} (1973), \textit{Modelos macroeconométricos: simultaneidad y recursividad, estimación y observaciones escasas}}.
}

De esos dos fracasos nace una pregunta que no es empírica, sino de principio: \textbf{¿qué queda de la política económica si los agentes no pueden equivocarse siempre en la misma dirección?} Y, detrás de ella, la que organiza toda la escuela: ¿puede explicarse el ciclo económico sin salir del equilibrio?

\subsection{El sistema de creencias: ¿qué hay que aceptar para ser un nuevo clásico?}\label{b1:creencias}

Antes que un conjunto de modelos, la nueva macroeconomía clásica es un conjunto de creencias sobre cómo funciona una economía. Son tres, y conviene enunciarlas sin suavizarlas, porque de ellas se sigue todo lo demás.

La primera es que \textbf{los agentes optimizan siempre}. No hay conducta rutinaria, ni reglas aproximadas, ni comportamiento institucional: hay problemas de maximización sujetos a restricciones. De ahí el \textit{individualismo metodológico}, que es su consecuencia directa: si el comportamiento individual es optimizador, el comportamiento agregado se obtiene sumando individuos, y la macroeconomía no es más que microeconomía agregada.\footnote{%
    El supuesto implícito es que la economía no tiene \textit{propiedades emergentes}, es decir, rasgos del conjunto que no se deducen de la suma de las partes. Un ejemplo ayuda a ver lo que está en juego. Una neurona tiene propiedades individuales que pueden describirse; un cerebro tiene ochenta mil millones. ¿Es el cerebro la mera agregación de sus neuronas, o aparecen al agregarlas propiedades que ninguna de ellas posee? Un cerebro de elefante tiene $257$ mil millones de neuronas, tres veces más, pero distribuidas de otro modo, con el $97{,}5\%$ concentradas en el cerebelo: el número no basta, la organización importa. Esta es la objeción central de la economía de la complejidad y de buena parte de la heterodoxia ([\authorfont{Ver Tema 3.A.4}]).
}

La segunda es que \textbf{los mercados se vacían siempre}. Los precios y los salarios son flexibles y se ajustan hasta que no queda intercambio ventajoso sin hacer, de modo que la economía está en equilibrio en todo momento, también durante una recesión. La consecuencia es incómoda y la escuela la asume sin rodeos: no existe desempleo involuntario, y quien no trabaja ha preferido el ocio al salario vigente.

La tercera, la más profunda y la menos visible, es que \textbf{el mundo es aprendible}. Detrás de los datos hay una estructura estable, con una distribución de probabilidad que no cambia, y los agentes, que viven en ella, pueden conocerla. Sobre esa creencia descansa la pieza que da nombre a la revolución, la \textbf{hipótesis de las expectativas racionales} (\authorfont{MUTH}, 1961): la previsión subjetiva de los agentes coincide con la esperanza matemática que se deduce del propio modelo,
$$\pi^{e}_{t}=\mathbb{E}\left[\pi_{t}\mid\Omega_{t-1}\right]\equiv\mathbb{E}_{t-1}[\pi_{t}]$$
donde $\Omega_{t-1}$ es toda la información disponible, incluida la regla de política.

La hipótesis no atribuye clarividencia a nadie, y ahí está su fuerza retórica: exige \textbf{coherencia}, no omnisciencia. Un agente que optimiza al decidir no puede ser ingenuo al prever, porque prever es parte de decidir.

\imagenfit{b1_expectativas.pdf}{Un error sistemático y un error que se compensa}{Elaboración propia a partir de \authorfont{MUTH} (1961).}

Lo decisivo no es, por tanto, que los agentes acierten, sino que \textbf{no se equivoquen siempre en el mismo sentido}. El error de previsión tiene media cero y es ortogonal a cuanto se sabía al decidir, de modo que ninguna variable conocida permite anticiparlo. Y con eso basta para cerrar la rendija que el monetarismo había dejado abierta: \textbf{una política que se repite deja de sorprender}, porque repetirse es, precisamente, volverse previsible.\footnote{%
    Que el mundo sea aprendible es un supuesto, no un hecho, y es el que la crítica poskeynesiana discute con más fuerza: \authorfont{DAVIDSON} (1991) sostiene que elimina la incertidumbre que \authorfont{KEYNES} y \authorfont{KNIGHT} consideraban decisiva, la que no se deja reducir a riesgo calculable. El alcance de esa objeción se desarrolla en [\authorfont{Ver Anexo \ref{anx:politica}}] y [\authorfont{Ver Tema 3.A.4}]; la definición formal de la hipótesis y las propiedades de su error, en [\authorfont{Ver Anexo \ref{anx:islas}}].
}

\subsection{El método: ¿cómo se hace macroeconomía desde 1972?}\label{b1:metodo}

Las tres creencias se convirtieron en un procedimiento de trabajo, y ahí reside la verdadera revolución: no en lo que la escuela sostuvo, sino en lo que exigió para admitir que algo era un argumento macroeconómico. A partir de \authorfont{LUCAS}, un modelo solo cuenta si parte de hogares y empresas que resuelven un problema de optimización, si sus agentes forman expectativas racionales y si los mercados se vacían; y, además, debe poder resolverse, ponerse números y simularse. Esa es la receta de los \textbf{modelos dinámicos estocásticos de equilibrio general}, el formato en que se escribe la macroeconomía desde entonces.

\imagenfit{b1_mapa.pdf}{De dónde viene la escuela, en qué se divide y qué deja}{Elaboración propia.}

La escuela recibe de la Síntesis el problema, los modelos estimados que no resistían un cambio de política, y del monetarismo la tesis, la tasa natural; y responde con un instrumental nuevo que se bifurca casi de inmediato en dos motores del ciclo, el de las sorpresas monetarias y el de las perturbaciones de productividad. Lo que deja, una década después, es el armazón sobre el que trabaja incluso quien la contradice.

Conviene medir bien la ironía de ese resultado. La nueva macroeconomía clásica no derrota a la Síntesis desde fuera, sino \textbf{completando el programa que la propia Síntesis decía profesar}: el del equilibrio general con agentes que optimizan, que \authorfont{HICKS} y \authorfont{MODIGLIANI} habían prometido y nunca llevaron hasta el final.\footnote{%
    \authorfont{DE VROEY} (2016) describe el cambio como el paso de una macroeconomía marshalliana, de equilibrios parciales y medición, en la que todavía se mueve \authorfont{FRIEDMAN}, a una walrasiana, formal y general, que es la de \authorfont{LUCAS}. El método, sus autores y la disciplina del equilibrio se desarrollan en [\authorfont{Ver Anexo \ref{anx:politica}}].
}

El precio de esa disciplina es alto y la escuela lo pagó sin discutirlo: \textbf{lo que no puede escribirse como un problema de optimización con expectativas racionales deja de ser macroeconomía admisible}. Esa exigencia es a la vez su fuerza, porque obliga a declarar todos los supuestos, y su límite, porque lo que no cabe en el formato, sencillamente, no se estudia.

\clearpage
\section{Los desarrollos: dos motores para un ciclo de equilibrio}\label{b2}

\puntosclave{
    Islas de LUCAS: con información imperfecta, solo la parte imprevista de la política monetaria mueve la producción; la parte esperada va entera a los precios. La curva de PHILLIPS existe en los datos, porque la sorpresa mueve a la vez inflación y producción, pero no ofrece ningún menú.

    Crítica de LUCAS: los parámetros estimados resumen reglas de decisión que cambian con la política, de modo que los modelos econométricos sirven para predecir dentro de un régimen, no para comparar regímenes. El remedio es microfundamentar, y la pendiente del propio modelo de islas, que cae cuando se intenta explotarla, es su mejor demostración.

    El ciclo real conserva el credo entero y cambia el impulso: perturbaciones de productividad propagadas por la sustitución intertemporal del trabajo y la acumulación de capital. El ciclo pasa a ser la respuesta óptima de la economía, no un fallo, y el crecimiento y el ciclo se explican con el mismo modelo.
}

\subsection{Las islas: ¿puede el dinero mover la producción sin romper el equilibrio?}\label{b2:islas}

El credo deja a \authorfont{LUCAS} ante un problema difícil. Los datos dicen que el dinero mueve la producción, pero si los mercados se vacían en todo momento y nadie se equivoca de forma sistemática, no hay rigidez ni ilusión monetaria a la que atribuir ese efecto. La explicación tendrá que ser de otra naturaleza, y la que encuentra es \textbf{informativa}: lo que falla no es el ajuste de los precios, sino lo que los agentes saben de ellos.

La parábola con la que lo plantea es la de una economía repartida en islas, tomada de \authorfont{PHELPS} (1970). Cada productor conoce bien el precio de su mercado y mal el nivel general de precios, que solo se publica con retraso. Cuando ve subir su precio, se enfrenta a una pregunta que no puede responder con certeza: ¿se ha encarecido su bien frente a los demás, en cuyo caso conviene trabajar más y producir más, o están subiendo todos los precios a la vez, en cuyo caso no conviene nada?

La respuesta racional no es elegir una de las dos, sino \textbf{repartir la culpa}, y repartirla según lo que la experiencia enseñe sobre cuál de las dos perturbaciones es más frecuente. De ahí sale la curva de oferta agregada de la escuela (\authorfont{LUCAS}, 1973):
$$y_{t}=y^{g}_{t}+\psi\left(p_{t}-\mathbb{E}_{t-1}[p_{t}]\right),\qquad \psi=\varepsilon^{F}\,\frac{\sigma^{2}_{\mathrm{rel}}}{\sigma^{2}_{\mathrm{rel}}+\sigma^{2}_{p}}$$
donde la producción solo se separa de su tendencia, $y^{g}_{t}$, cuando el nivel de precios se separa de lo que se esperaba, y lo hace con una intensidad $\psi$ que mide cuánta parte de cada subida se atribuye, en promedio, al precio relativo propio.\footnote{%
    La derivación completa, con el problema de extracción de la señal resuelto como una proyección, está en [\authorfont{Ver Anexo \ref{anx:islas}}]. El peso depende de las varianzas relativas: si los precios relativos se mueven mucho ($\sigma^{2}_{\mathrm{rel}}$ grande) y el nivel general poco ($\sigma^{2}_{p}$ pequeño), el productor atribuye la subida a su mercado y responde; en el caso contrario, la descuenta como inflación y no se mueve. La elasticidad $\varepsilon^{F}$, la de \authorfont{FRISCH}, mide cuánto responde su esfuerzo a un precio relativo más alto.
}

Añadida una demanda agregada, que en su versión más simple es la teoría cuantitativa del dinero, el modelo entrega su resultado: \textbf{solo la parte imprevista de la política monetaria mueve la producción}; la parte que se esperaba se traslada entera a los precios.

\imagenfit{b2_islas.pdf}{Una expansión monetaria imprevista y la misma expansión anunciada}{Elaboración propia a partir de \authorfont{LUCAS} (1973).}

Lo que esta construcción logra es notable, y conviene no pasarlo por alto: \textbf{explica el ciclo sin salir del equilibrio}. Nadie se comporta de forma irracional, ningún mercado deja de vaciarse y, sin embargo, el dinero mueve la producción. El ciclo no es un fallo del sistema, sino lo que ocurre cuando agentes que optimizan reciben información imperfecta.

De ahí se sigue, por partida doble, el desmantelamiento del menú de política. La relación entre inflación y producción \textbf{existe en los datos}, porque ambas suben cuando hay sorpresa, de modo que un econometrista la encontrará; pero \textbf{no puede aprovecharse}, porque las sorpresas no se fabrican a voluntad. Y si la autoridad sigue una regla conocida, su política entera está dentro de las expectativas y no mueve nada real: es la proposición de ineficacia de \authorfont{SARGENT} y \authorfont{WALLACE} (1975), que se desarrolla en [\authorfont{Ver Anexo \ref{anx:politica}}].

\subsection{La crítica de \authorfont{LUCAS}: ¿por qué los modelos de la Síntesis no sirven para evaluar la política?}\label{b2:critica}

El mismo razonamiento, aplicado no a la economía sino a quienes la modelizan, produce la aportación más influyente de la escuela y la única que hoy nadie discute. \authorfont{LUCAS} (1976) la enuncia como un silogismo de tres líneas: los parámetros de un modelo econométrico resumen \textbf{reglas de decisión} de los agentes; esas reglas son óptimas dadas las reglas del juego, es decir, dada la política; luego, si la política cambia, los parámetros cambian con ella.

La consecuencia es demoledora para el uso que se hacía de aquellos modelos. Un modelo estimado puede predecir razonablemente bien mientras el régimen de política no cambie, y por eso servía para los pronósticos a corto plazo; pero \textbf{no sirve para comparar regímenes}, que es exactamente para lo que se estaba utilizando. Las simulaciones de políticas alternativas con parámetros estimados bajo una política distinta no son aproximadas: son inválidas.

De ahí la distinción que la escuela impone a cualquier modelo. Hay \textbf{parámetros profundos}, que describen preferencias y tecnología, como la impaciencia o la aversión al riesgo, y son invariantes frente a la política; y hay \textbf{parámetros que dependen del régimen}, porque resumen cómo se forman las expectativas en él. Solo los primeros pueden tomarse como datos al evaluar una política, y solo se llega a ellos escribiendo el problema de optimización de los agentes: la crítica no es únicamente destructiva, porque señala el remedio, que es la microfundamentación.\footnote{%
    Los modelos cuyos coeficientes mezclan las dos familias, como el IS-LM estimado, quedan atrapados en medio: la política altera la posición y la pendiente de sus curvas a través de las expectativas incorporadas en los parámetros. El propio \authorfont{LUCAS} reconoció que había poco en su ensayo que no estuviera implícito en \authorfont{FRIEDMAN}, \authorfont{MUTH} y, antes, en \authorfont{KNIGHT}; los extractos del artículo se recogen en [\authorfont{Ver Anexo \ref{anx:politica}}].
}

La elegancia del argumento está en que el propio modelo de islas lo demuestra. La pendiente $\psi$ depende de la varianza del nivel de precios, y esa varianza la fabrica la política monetaria: un gobierno que intente explotar la curva de \authorfont{PHILLIPS} con expansiones repetidas vuelve los precios más erráticos, los productores se fían menos de su propio precio y $\psi$ cae, hasta que la curva se vuelve vertical. \textbf{La disyuntiva desaparece justo cuando se intenta aprovecharla}, y no por mala suerte, sino porque la relación nunca fue estructural.\footnote{%
    La contrastación internacional de \authorfont{LUCAS} (1973) con dieciocho países fue la primera evidencia a favor: donde la demanda nominal es estable, la relación entre inflación y producción aparece generosa; donde es volátil, con Argentina como caso extremo, la curva es casi vertical. El desarrollo formal y la ampliación moderna de esa evidencia, en [\authorfont{Ver Anexo \ref{anx:islas}}] y [\authorfont{Ver Tema 3.A.42}].
}

\subsection{El ciclo real: ¿y si el ciclo no necesita al dinero?}\label{b2:ciclo}

El primer motor duró poco. Las sorpresas no pueden encadenarse, porque dejarían de serlo, de modo que el modelo solo genera desviaciones de un periodo cuando los ciclos duran años; los datos monetarios se publican pronto, lo que hace difícil sostener la confusión; y la contrastación mostró que las perturbaciones monetarias explican una parte pequeña de las fluctuaciones. La salida la señalaba el primero de esos fallos: \textbf{si la propagación del ciclo ha de ser real, el impulso también puede serlo}.

Ese es el paso que dan \authorfont{KYDLAND} y \authorfont{PRESCOTT} (1982) y \authorfont{LONG} y \authorfont{PLOSSER} (1983). Conservan íntegro el credo (optimización, expectativas racionales, mercados que se vacían) y cambian solo la fuente de las fluctuaciones: donde había sorpresas monetarias hay ahora \textbf{perturbaciones de la productividad}, que son reales y no necesitan que nadie se confunda.

El mecanismo descansa en dos decisiones del hogar. La primera es \textbf{cuándo trabajar}: como una mejora de la productividad eleva el salario de hoy frente al de mañana, conviene trabajar más ahora y descansar después, de modo que la producción sube más que la propia productividad. La segunda es \textbf{cuándo consumir}: como la renta es transitoriamente alta, el hogar no la gasta, la invierte, y por eso la inversión oscila mucho más que el consumo.\footnote{%
    El modelo completo, con el hogar, la empresa, la acumulación de capital, las condiciones de primer orden y el estado estacionario, se desarrolla en [\authorfont{Ver Anexo \ref{anx:ciclo}}]; su versión cuantitativa, con la calibración y las simulaciones, en [\authorfont{Ver Tema 3.A.42}].
}

Con la producción decidida por la tecnología y las preferencias, la oferta agregada se vuelve vertical, y el papel del dinero se desvanece del todo.

\imagenfit{b2_ciclo_real.pdf}{Una mejora de la productividad y una relajación monetaria con oferta vertical}{Elaboración propia a partir de \authorfont{KYDLAND} y \authorfont{PRESCOTT} (1982).}

De aquí salen las dos tesis que hacen de este modelo el más radical de la escuela. La primera es que \textbf{el ciclo es eficiente}: como los mercados son competitivos y no hay distorsión alguna, el equilibrio coincide con lo que elegiría un planificador benévolo, de modo que una recesión no es un fallo que corregir, sino el periodo en que conviene trabajar e invertir menos porque los factores rinden menos. Estabilizar la producción no mejoraría el bienestar: lo reduciría.

La segunda es de otro orden y explica la ambición del programa: \textbf{el crecimiento y el ciclo dejan de ser dos teorías}. El modelo que explica la tendencia a largo plazo es el mismo que explica las oscilaciones a corto, sometido a perturbaciones de distinto tamaño y duración, y por primera vez la macroeconomía dispone de un único aparato para los dos fenómenos.

El precio de ambas tesis es alto, y será el flanco por el que la escuela reciba los golpes: el modelo necesita una oferta de trabajo mucho más elástica que la que miden los microdatos, y exige explicar las recesiones como episodios de \textit{retroceso} tecnológico, algo que cuesta aceptar incluso a quienes admiten lo demás.\footnote{%
    Con la utilidad habitual del modelo, la elasticidad de \authorfont{FRISCH} implícita es del orden de $2$, frente a valores muy inferiores a la unidad en la evidencia microeconómica; es la paradoja que \authorfont{ROGERSON} (1988) y \authorfont{HANSEN} (1985) intentaron resolver con la indivisibilidad del trabajo. El estado de esta discusión y la medida de las perturbaciones tecnológicas se recogen en [\authorfont{Ver Anexo \ref{anx:politica}}] y [\authorfont{Ver Tema 3.A.42}].
}

\section{La política y el balance}\label{b3}

\puntosclave{
    Cara monetaria: la política sistemática no puede estabilizar, porque lo que responde a información pública ya está descontado (SARGENT y WALLACE, 1975); y la discreción de una autoridad benévola acaba en sesgo inflacionista sin ganancia de empleo, $\pi^{d}=\pi^{*}+\alpha\beta k$ (KYDLAND y PRESCOTT, 1977). El problema es de credibilidad, no de objetivos.

    Cara fiscal: con agentes que conocen la restricción presupuestaria del Estado, la financiación del gasto es indiferente y la deuda no es riqueza neta (equivalencia ricardiana, BARRO, 1974); y si la política fiscal domina, la monetaria no controla la inflación de forma indefinida (aritmética monetarista desagradable).

    Las soluciones son institucionales, no técnicas: reglas, delegación, reputación y contratos, el guion que copió la arquitectura actual (bancos centrales independientes, reglas fiscales). Balance: la escuela perdió las discusiones sustantivas y ganó la metodológica; sus adversarios la discuten con sus propias herramientas.
}

\subsection{La cara monetaria: ¿puede estabilizar el banco central?}\label{b3:monetaria}

Las implicaciones de política de la escuela no son un apéndice de sus modelos: son la razón por la que se discutieron tanto. Y en el terreno monetario llegan encadenadas, porque el segundo resultado muerde justo donde el primero deja la herida abierta.

El primero es la \textbf{proposición de ineficacia} de \authorfont{SARGENT} y \authorfont{WALLACE} (1975). Si el banco central sigue una regla conocida, por sofisticada que sea, esa regla está dentro de las expectativas de todo el mundo antes de aplicarse, y lo que se espera se traslada a los precios sin tocar la producción. Conviene medir el alcance del resultado, que suele quedarse corto en el enunciado habitual: no es solo que la política sistemática no pueda elevar la producción media, es que \textbf{tampoco puede reducir su variabilidad}, que era el objetivo declarado de cualquier política de estabilización. A la autoridad le queda únicamente la parte imprevisible de su conducta, y explotarla exigiría disponer de información que el público no tenga.

El segundo va más al fondo, porque no depende del modelo de islas ni de ninguna teoría particular del ciclo. \authorfont{KYDLAND} y \authorfont{PRESCOTT} (1977) muestran que una autoridad \textit{benévola}, que persigue exactamente lo que la sociedad quiere, llega a un resultado peor por el solo hecho de poder elegir cada periodo. El motivo es que las expectativas privadas se forman antes, y una vez formadas se convierten en una restricción: lo que ayer era una variable que influir, hoy es un dato heredado que invita a incumplir lo prometido.

\imagenfit{b3_kp.pdf}{La inconsistencia temporal: por qué un anuncio creíble no lo es}{Elaboración propia a partir de \authorfont{KYDLAND} y \authorfont{PRESCOTT} (1977).}

El equilibrio del juego tiene una forma sencilla de recordar. Si $\alpha$ mide cuánto reduce el paro una inflación no esperada, $\beta$ el peso que la autoridad da al empleo frente a la inflación y $k$ lo ambicioso que es su objetivo de paro frente a la tasa natural, la inflación de equilibrio con discreción es (\authorfont{KYDLAND} y \authorfont{PRESCOTT}, 1977, en la versión cuadrática de \authorfont{BARRO} y \authorfont{GORDON}, 1983):
$$\pi^{d}=\pi^{*}+\alpha\beta k$$
es decir, \textbf{más inflación de la deseada y exactamente el mismo paro}. El sesgo crece con la ambición del gobierno y con su afán por el empleo, y no compra nada a cambio.\footnote{%
    La derivación, la comparación con el resultado bajo compromiso y el recorrido completo del juego están en [\authorfont{Ver Anexo \ref{anx:politica}}]. La misma expresión, escrita en producción en lugar de en paro, es la que usa [\authorfont{Ver Tema 3.B.16}] para explicar por qué un país con sesgo inflacionista gana credibilidad al entrar en una unión monetaria.
}

De las dos piezas sale una conclusión que cambió la práctica de los bancos centrales: \textbf{el problema de la política monetaria no está en sus objetivos, sino en su credibilidad}. No se corrige con mejores previsiones ni con funcionarios más competentes, porque no es un problema de información ni de voluntad, sino de incentivos.

\subsection{La cara fiscal: ¿puede estimular el gobierno?}\label{b3:fiscal}

La política monetaria no fue el único blanco. \authorfont{BARRO} (1974) aplicó el mismo principio, el de agentes que miran al futuro y conocen la restricción presupuestaria del Estado, a la política fiscal, y obtuvo un resultado igual de incómodo para la tradición keynesiana.

El argumento cabe en una sustitución. Si el Estado se financia con deuda, tendrá que devolverla con impuestos futuros de idéntico valor actual, de modo que al meter su restricción dentro de la del hogar los impuestos desaparecen y solo queda el gasto (\authorfont{BARRO}, 1974, en su versión de dos periodos):
$$c_{1}+\frac{c_{2}}{1+r}=Y_{1}+\frac{Y_{2}}{1+r}-\left(G_{1}+\frac{G_{2}}{1+r}\right)$$

\imagenfit{b3_fisher.pdf}{La equivalencia ricardiana: la dotación se desliza y el óptimo no se mueve}{Elaboración propia a partir de \authorfont{BARRO} (1974).}

Una rebaja de impuestos financiada con deuda no hace más rico a nadie: anuncia impuestos futuros del mismo valor y el hogar la ahorra entera para pagarlos. De ahí que las tres formas de financiar un mismo gasto, deuda, impuestos o emisión de dinero, sean equivalentes, y que lo relevante sea \textbf{cuánto gasta el Estado y en qué, no cómo lo financia}: la deuda pública no es riqueza neta.

El resultado es un teorema y vale lo que valgan sus supuestos, que son exigentes: horizontes infinitos o herederos a los que se quiere dejar con qué pagar, certidumbre sobre los impuestos futuros, impuestos de suma fija, mercados de capitales sin restricciones y gasto improductivo. Donde esos supuestos fallan, y fallan a menudo, la equivalencia es solo parcial.\footnote{%
    Los hogares que no pueden endeudarse gastan lo que reciben, de modo que una transferencia financiada con deuda sí eleva su consumo; los impuestos reales distorsionan, porque el único neutral es el de suma fija; y la evidencia encuentra compensaciones parciales del ahorro privado, no completas. Los cinco supuestos y sus grietas se desarrollan en [\authorfont{Ver Anexo \ref{anx:politica}}].
} Conserva, en cambio, todo su valor como referencia, y de él se siguió una tesis que la política europea haría célebre: la de los \textbf{efectos no keynesianos} de la política fiscal, según la cual una consolidación creíble puede resultar expansiva, porque anticipa menores impuestos futuros y reduce la prima de riesgo.

Las dos caras acaban reuniéndose, y lo hacen por donde era previsible, por la restricción presupuestaria del Estado. Si la política fiscal es \textit{dominante}, es decir, si los déficits se deciden al margen de lo que haga el banco central, la autoridad monetaria no puede controlar la inflación de manera indefinida: podrá no emitir hoy, pero entonces la deuda crece y exigirá emitir más mañana. Apretar hoy significa, en ese régimen, más inflación después, y es lo que \authorfont{SARGENT} y \authorfont{WALLACE} (1981) llamaron aritmética monetarista desagradable\textcolor{orange}{\textsuperscript{[Probable]}}. \textbf{La independencia del banco central no basta si la política fiscal no acompaña}.

\subsection{Las soluciones: ¿cómo se atan las manos un gobierno?}\label{b3:soluciones}

Si el mal nace de la discrecionalidad, el remedio consiste en restringirla, y la literatura posterior ensayó cuatro vías. La primera son las \textbf{reglas} anunciadas de antemano, que permiten a los agentes deducir las reglas del juego, pero que por sí solas no resuelven nada mientras la autoridad conserve la capacidad de incumplirlas. La segunda es la \textbf{delegación} en un banco central independiente y más duro con la inflación que la sociedad, el banquero conservador de \authorfont{ROGOFF} (1985), que reduce el sesgo al precio de estabilizar peor las perturbaciones reales. La tercera es la \textbf{reputación}: en un juego repetido, la autoridad puede sacrificar la ganancia de hoy para que la crean mañana, como hizo la Reserva Federal de \authorfont{VOLCKER}. Y la cuarta son los \textbf{contratos} que penalizan a la autoridad por desviarse, cuyo equivalente práctico es encomendarle un objetivo de inflación más exigente que el socialmente óptimo.\footnote{%
    El desarrollo de las cuatro, con la crítica de \authorfont{POSEN} (1993) a la lectura causal de la independencia de los bancos centrales, en [\authorfont{Ver Anexo \ref{anx:politica}}].
}

Lo notable es que la arquitectura institucional que rige hoy copió el guion en las dos caras: bancos centrales independientes con un objetivo de inflación explícito, y reglas fiscales vigiladas por autoridades independientes, que es la estructura de la Unión Económica y Monetaria y de su Pacto de Estabilidad y Crecimiento ([\authorfont{Ver Tema 3.B.16, Tema 3.A.23}]). La escuela consiguió así algo que ninguna de sus predecesoras logró: convertir la política macroeconómica en un \textbf{problema de diseño institucional} y no de manejo de instrumentos.

\subsection{El balance: ¿qué quedó de todo esto?}\label{b3:balance}

El balance tiene dos columnas que apuntan en direcciones opuestas, y de su conjunto sale la posición que la escuela ocupa hoy.

En lo \textbf{sustantivo perdió casi todas las discusiones}. La evidencia sobre los efectos reales de las perturbaciones monetarias contradice la neutralidad; el vaciado continuo de los mercados y el desempleo voluntario resultan difíciles de sostener; la oferta de trabajo que el ciclo real necesita es mucho más elástica que la que miden los microdatos; y explicar las recesiones como retrocesos tecnológicos convenció a pocos.

En lo \textbf{metodológico ganó sin discusión}. Hoy no se publica un modelo macroeconómico sin microfundamentos explícitos, sin expectativas racionales y sin capacidad de ser cuantificado, y esa exigencia la impuso ella. La prueba más clara es que \textbf{quienes la contradicen lo hacen con sus herramientas}: la nueva economía keynesiana recupera el papel del dinero y de la política añadiendo competencia imperfecta y rigideces nominales a un modelo que, por lo demás, es el del ciclo real ([\authorfont{Ver Tema 3.A.7}]).

Lo que distingue a esta es que su método resultó tan exigente que sus propios adversarios tuvieron que aceptarlo para poder discutirle, y por eso puede decirse que la macroeconomía posterior a 1972 se escribe entera, incluso cuando la contradice, en el idioma que ella impuso.

\anexo{Las expectativas racionales y el modelo de islas de \authorfont{LUCAS}}\label{anx:islas}

Este anexo es el manual del primer motor de la nueva macroeconomía clásica. Construye el aparato de abajo arriba: primero la hipótesis de las expectativas racionales, que es la pieza que lo sostiene todo; después la economía de islas en la que LUCAS la puso a trabajar; luego la extracción de la señal, la oferta y la demanda agregadas y el equilibrio, resuelto con el método de los coeficientes indeterminados hasta las formas reducidas de la producción y de la inflación; y, al final, lo que el modelo enseña: una curva de \authorfont{PHILLIPS} que existe y no puede usarse, la ineficacia de la política sistemática y la crítica de \authorfont{LUCAS} como corolario, junto con las razones por las que el programa se agotó y dio paso al ciclo real.

Dos convenciones valen para todo el anexo. Las letras minúsculas son logaritmos, de modo que las diferencias entre ellas se leen como porcentajes; y $\mathbb{E}_{t-1}[\cdot]$ es la esperanza condicionada a la información disponible al cerrar el periodo anterior, $\Omega_{t-1}$, que incluye el propio modelo, la regla de política y toda la historia de la economía. La notación es la común con el 3.A.42, que usa la curva resultante para estudiar el ciclo y remite aquí para su derivación ([\authorfont{Ver Tema 3.A.42}]).

\subsection{La hipótesis de las expectativas racionales}\label{anx1:her}

Toda decisión intertemporal depende de variables que aún no existen: el precio al que se venderá la cosecha, la inflación que erosionará un salario pactado hoy. Cómo forman los agentes esas expectativas no es un detalle técnico, sino la bisagra de cualquier modelo macroeconómico: de ella depende cuánto tiempo puede una política alimentarse del error de quienes la sufren. Este apartado presenta las tres respuestas que la teoría ha dado a esa pregunta y las propiedades que distinguen a la tercera.

\subsubsection{Tres maneras de mirar al futuro}

La respuesta más simple son las \textit{expectativas estáticas}, habituales en los modelos de la Síntesis ([\authorfont{Ver Tema 3.A.5}]): el futuro repetirá el presente,
$$p^{e}_{t}=p_{t-1}$$
de modo que el agente es miope y cualquier cambio le sorprende entero. Un paso más allá están las \textit{expectativas adaptativas}, popularizadas por \authorfont{CAGAN} (1956) y \authorfont{FRIEDMAN} (1968) y centrales en el monetarismo ([\authorfont{Ver Tema 3.A.5}]): la previsión se corrige con una fracción $\lambda$ del último error cometido,
$$p^{e}_{t}=p^{e}_{t-1}+\lambda\,\left(p_{t-1}-p^{e}_{t-1}\right),\qquad 0<\lambda\leq 1$$
con lo que el agente aprende, pero solo mirando hacia atrás.\footnote{%
    Dentro de la familia adaptativa se distinguen tres variantes. Con \textit{aprendizaje}, la recogida en la ecuación: se incorpora el error cometido en la previsión anterior, la diferencia entre el valor realizado y el esperado. \textit{Extrapolativas}: la previsión pondera los valores pasados, con pesos que decrecen a medida que se consideran observaciones más lejanas. \textit{Regresivas}: además de los valores pasados ponderados, los agentes tienen un ancla o valor de largo plazo hacia el que esperan que la variable regrese.
}

El fallo común de ambas es que el error puede volverse \textit{sistemático}. Si la cantidad de dinero crece de forma sostenida, quien prevé adaptativamente infravalora la inflación un periodo tras otro, siempre en el mismo sentido, y no corrige nunca del todo: la regla de formación de expectativas no usa algo que el agente conoce, la propia política.\footnote{%
    \authorfont{LUCAS} y \authorfont{PRESCOTT} (1971) lo formularon con precisión en un problema de inversión: si las perturbaciones de demanda tienen un carácter estocástico regular, las expectativas adaptativas generan una distribución de precios distinta con cada innovación, y el error predictivo resultante es persistente y sistemático. Impedir ese desajuste es justo lo que motiva su propuesta de expectativas racionales: los agentes conocen la estructura de la economía, actúan como si conocieran las reglas de política futuras, consideran que los mercados se vacían, y todo ese conocimiento es común.
} Sobre esa grieta se construye la contrarrevolución: un modelo cuyos agentes se equivocan siempre igual atribuye a la política un poder que solo dura mientras dura la ingenuidad.

\subsubsection{La definición de \authorfont{MUTH}}

\authorfont{MUTH} (1961), estudiando los mercados agrícolas, propuso cerrar la grieta por el camino más directo: las expectativas subjetivas de los agentes coinciden con la esperanza matemática que se deduce del propio modelo,
$$p^{e}_{t}=\mathbb{E}\left[\,p_{t}\mid \Omega_{t-1}\,\right]\equiv \mathbb{E}_{t-1}[p_{t}]$$
donde $\Omega_{t-1}$ reúne toda la información relevante disponible: la estructura de la economía, las reglas de política y la historia observada. La hipótesis no atribuye a los agentes clarividencia, sino coherencia: no pueden ser racionales al decidir y crédulos al prever, porque prever es parte de decidir. La primera aplicación macroeconómica llegó una década después, con \authorfont{LUCAS} y \authorfont{PRESCOTT} (1971) en un contexto de decisiones de inversión, y fue \authorfont{LUCAS} (1972) quien la convirtió en el cimiento de una teoría del ciclo y de la política monetaria.

\subsubsection{Las propiedades del error de previsión}

Definido el error como $\varepsilon_{t}=p_{t}-\mathbb{E}_{t-1}[p_{t}]$, la hipótesis le impone tres propiedades (\authorfont{MUTH}, 1961):
$$\mathbb{E}_{t-1}[\varepsilon_{t}]=0,\qquad \mathrm{cov}\left(\varepsilon_{t},\,x\right)=0\;\;\forall\, x\in\Omega_{t-1},\qquad \varepsilon_{t}\sim N\!\left(0,\,\sigma^{2}_{\varepsilon}\right),\;\;\sigma^{2}_{\varepsilon}<\infty$$
La primera dice que el agente puede equivocarse en cualquier periodo, pero no se espera que lo haga: el error medio es cero. La segunda, la \textit{ortogonalidad}, es la más exigente y la más contrastable: ninguna variable conocida en $t-1$ ayuda a predecir el error, porque si ayudara, el agente ya la habría usado. La tercera convierte al error en un ruido blanco de varianza finita. En conjunto, la hipótesis no elimina la incertidumbre: elimina la posibilidad de explotarla de forma sistemática, que es exactamente lo que una política de sorpresas repetidas necesitaría.

\subsection{La economía de islas}\label{anx1:economia}

La hipótesis necesita un escenario donde la información incompleta haga trabajo económico, y ese escenario lo aportó la parábola de las islas de \authorfont{PHELPS} (1970): una economía de mercados descentralizados en la que cada participante conoce bien lo que ocurre en el suyo y mal lo que ocurre en los demás. \authorfont{LUCAS} (1972, 1973) la convirtió en un modelo completo, y de sus supuestos sale todo lo demás.\footnote{%
    La curva de oferta resultante se presenta de maneras algo distintas en los artículos de \authorfont{LUCAS}: la versión de 1972 es un modelo de equilibrio general con generaciones solapadas, y la de 1973, la que sigue este anexo, una formulación directamente estimable. Todas comparten los mismos supuestos de fondo y conducen a las mismas conclusiones.
}

\subsubsection{Los supuestos}

\begin{lnum}
    \item \textit{Muchos mercados, agentes repartidos}. La economía se compone de un gran número de mercados o islas, indexados por $z$, entre los que los productores se distribuyen de forma aleatoria. Cada uno vende su producción en su isla, al precio local, y consume una cesta comprada en el conjunto de la economía, al nivel general de precios;
    \item \textit{Información local completa, información agregada incompleta}. Al decidir, cada productor observa el precio de su isla, $p_{t}(z)$, pero no el nivel general de precios del periodo, $p_{t}$, del que solo tiene la información común $\Omega_{t-1}$;
    \item \textit{Oferta por sustitución intertemporal}. La decisión de producción nace de la elección entre trabajo y ocio a lo largo del tiempo (\authorfont{LUCAS} y \authorfont{RAPPING}, 1969): conviene trabajar y producir más en los periodos en que el precio propio es alto en términos relativos, y descansar cuando es bajo, porque el esfuerzo rinde más cuando mejor se paga ([\authorfont{Ver Tema 3.A.25}]); y,
    \item \textit{Expectativas racionales}. Sobre el nivel general de precios, los productores forman la mejor previsión posible con la información de que disponen.
\end{lnum}

\subsubsection{La oferta del productor}

Con esos ingredientes, la oferta de la isla $z$ responde al precio relativo \textit{percibido}: la diferencia entre el precio local observado y lo que el productor cree que vale el nivel general de precios una vez vista su isla (\authorfont{LUCAS}, 1973),
$$y_{t}(z)=\varepsilon^{F}\left(p_{t}(z)-\mathbb{E}^{z}_{t}[p_{t}]\right)$$
donde $\mathbb{E}^{z}_{t}[p_{t}]$ es la expectativa del nivel general que forman los productores de $z$ después de observar $p_{t}(z)$, y $\varepsilon^{F}$ es la elasticidad de la oferta: cuánto más se trabaja y produce cuando el precio relativo percibido sube un uno por ciento. Es la elasticidad de \authorfont{FRISCH} de la oferta de trabajo, la misma que reaparecerá como pieza central del ciclo real.\footnote{%
    La magnitud de esa elasticidad sigue siendo uno de los debates empíricos abiertos que heredó toda la macroeconomía posterior. Desde el punto de vista macroeconómico se considera elevada, sobre la base de las fluctuaciones de tipos impositivos y horas totales trabajadas entre países y a lo largo del tiempo; además, los modelos de equilibrio general nacidos de la nueva macroeconomía clásica, del ciclo real a los neokeynesianos, necesitan una elasticidad alta para reproducir los hechos estilizados del ciclo. La evidencia microeconómica, en cambio, apunta a elasticidades muy por debajo de la unidad para los hombres del tramo central de edad (25 a 54 años), aunque los estudios que incluyen a mujeres o a trabajadores jóvenes tienden a elevarlas. La respuesta de \authorfont{ROGERSON} y \authorfont{HANSEN} a esta tensión, la indivisibilidad del trabajo, se estudia en [\authorfont{Ver Tema 3.A.25}].
}

\subsection{La extracción de la señal}\label{anx1:senal}

El productor que ve subir su precio afronta un problema de inferencia: la subida puede deberse a que su bien se ha encarecido frente a los demás, un cambio real que invita a producir más, o a que todos los precios están subiendo, un cambio nominal que no invita a nada. Decidir cuánto de lo observado es señal y cuánto es ruido es el corazón del modelo, y la respuesta tiene la forma de una regresión.

\subsubsection{La estructura de la información}

El precio local se desvía del nivel general por una perturbación específica de la isla (\authorfont{LUCAS}, 1973),
$$p_{t}(z)=p_{t}+z_{t},\qquad z_{t}\sim N\!\left(0,\,\sigma^{2}_{\mathrm{rel}}\right)$$
donde $z_{t}$ recoge la suerte relativa del mercado $z$, con media cero y varianza $\sigma^{2}_{\mathrm{rel}}$; al sumar sobre todas las islas, las perturbaciones específicas se cancelan. Antes de observar nada, la información común sitúa al nivel general de precios en
$$p_{t}\sim N\!\left(\mathbb{E}_{t-1}[p_{t}],\,\sigma^{2}_{p}\right)$$
donde $\sigma^{2}_{p}$ es la varianza de la parte del nivel de precios que la información común no anticipa. Esa varianza no es un dato de la naturaleza: la genera, como se verá, la propia política monetaria, y de esa circularidad saldrá la crítica de \authorfont{LUCAS}.

\subsubsection{La proyección, derivada}

El productor combina lo que creía con lo que ve. Para variables conjuntamente normales, la esperanza condicionada es la \textit{proyección lineal}, la regresión poblacional de la variable que interesa sobre la observada:
$$\mathbb{E}\left[p_{t}\mid p_{t}(z)\right]=\mathbb{E}_{t-1}[p_{t}]+\frac{\mathrm{cov}\left(p_{t},\,p_{t}(z)\right)}{\mathrm{var}\left(p_{t}(z)\right)}\left(p_{t}(z)-\mathbb{E}_{t-1}[p_{t}(z)]\right)$$
Los tres ingredientes se calculan directamente de la estructura anterior: como $z_{t}$ es independiente del nivel general, $\mathrm{cov}(p_{t},p_{t}(z))=\sigma^{2}_{p}$; la varianza de lo observado es $\mathrm{var}(p_{t}(z))=\sigma^{2}_{p}+\sigma^{2}_{\mathrm{rel}}$; y, al tener $z_{t}$ media cero, $\mathbb{E}_{t-1}[p_{t}(z)]=\mathbb{E}_{t-1}[p_{t}]$. Sustituyendo,
$$\mathbb{E}^{z}_{t}[p_{t}]=\left(1-\theta\right)\mathbb{E}_{t-1}[p_{t}]+\theta\,p_{t}(z),\qquad \theta=\frac{\sigma^{2}_{p}}{\sigma^{2}_{p}+\sigma^{2}_{\mathrm{rel}}}$$
La creencia revisada es una media ponderada de la creencia previa y del precio observado, y el peso $\theta$ es el cociente de varianzas: la fracción de la variabilidad del precio local que, en promedio, procede del nivel general.\footnote{%
    El mismo resultado se obtiene aplicando la regla de \authorfont{BAYES} a las densidades normales: la distribución posterior de $p_{t}$ dado $p_{t}(z)$ es normal, y su media es exactamente esta media ponderada. La proyección lineal y la actualización bayesiana coinciden bajo normalidad, y por eso la literatura presenta la extracción de la señal indistintamente de las dos maneras.
}

El peso se lee como un grado de confianza en la propia isla como espejo de la economía. Si la política monetaria es estable, $\sigma^{2}_{p}$ es pequeña, $\theta$ es bajo y el productor apenas actualiza: atribuye la subida a su suerte relativa y produce. Si la política es volátil, $\sigma^{2}_{p}$ es grande, $\theta$ es alto y la creencia sigue al precio observado: la subida se descuenta como inflación general y la producción no responde.

\subsubsection{La oferta tras la señal}

Sustituyendo la creencia revisada en la oferta del productor, el precio relativo percibido resulta ser una fracción fija de la sorpresa local:
$$p_{t}(z)-\mathbb{E}^{z}_{t}[p_{t}]=\left(1-\theta\right)\left(p_{t}(z)-\mathbb{E}_{t-1}[p_{t}]\right)\quad\Longrightarrow\quad y_{t}(z)=\varepsilon^{F}\left(1-\theta\right)\left(p_{t}(z)-\mathbb{E}_{t-1}[p_{t}]\right)$$
Cada productor responde solo a la parte de su precio que la información común no explica, y la intensidad de la respuesta ya no depende únicamente de sus preferencias ($\varepsilon^{F}$), sino del entorno informativo en que vive ($\theta$).

\subsection{La oferta y la demanda agregadas}\label{anx1:oada}

El paso del productor a la economía exige agregar las islas y añadir una demanda. Las dos piezas son deliberadamente simples: toda la novedad del modelo está en la oferta, y la demanda más austera posible basta para cerrar el sistema.

\subsubsection{La agregación y el componente tendencial}

Al promediar sobre las islas, las perturbaciones específicas $z_{t}$ se cancelan y el precio local medio coincide con el nivel general. La suma de las ofertas individuales da la \textit{función de oferta agregada de \authorfont{LUCAS}}, que en su versión completa incluye el componente tendencial de la producción (\authorfont{LUCAS}, 1973):
$$y_{t}=y^{g}_{t}+\psi\left(p_{t}-\mathbb{E}_{t-1}[p_{t}]\right),\qquad \psi=\varepsilon^{F}\left(1-\theta\right)=\varepsilon^{F}\,\frac{\sigma^{2}_{\mathrm{rel}}}{\sigma^{2}_{\mathrm{rel}}+\sigma^{2}_{p}}$$
El término $y^{g}_{t}$ es la senda normal de la producción, la que dibujan la población, el capital y la tecnología; la producción solo se separa de ella cuando el nivel de precios se separa de lo que se esperaba. En lo que sigue, la producción se mide en desviaciones sobre esa tendencia, lo que equivale a normalizar $y^{g}_{t}=0$ sin perder nada.\footnote{%
    En su contrastación, \authorfont{LUCAS} (1973) estimó la curva añadiendo además un término de persistencia en la desviación. La estimación, su lectura y la evidencia internacional ampliada se estudian en [\authorfont{Ver Tema 3.A.42}].
}

\subsubsection{La demanda cuantitativa}

La demanda sale de la teoría cuantitativa del dinero ([\authorfont{Ver Tema 3.A.5}]): en logaritmos, $m_{t}+v_{t}=p_{t}+y_{t}$, y con una velocidad de circulación constante y normalizada a cero,
$$y_{t}=m_{t}-p_{t}$$
Para una cantidad de dinero dada, la demanda tiene pendiente negativa en el plano de la producción y los precios, porque el mismo dinero compra más producción cuanto más bajos son los precios, y una expansión monetaria la desplaza en paralelo hacia fuera.\footnote{%
    En rigor, $m_{t}$ no es estrictamente la oferta monetaria, sino cualquier variable que gobierne la demanda nominal: en la formulación original de \authorfont{LUCAS} (1973) el papel lo hace la renta nominal. Leerla como cantidad de dinero es la interpretación habitual y la que conecta el modelo con el debate sobre la política monetaria.
}

\subsection{El equilibrio con expectativas racionales}\label{anx1:equilibrio}

En un modelo con expectativas racionales, la expectativa no es un dato que se añade al final: es parte de la solución, porque los precios dependen de lo que se espera y lo que se espera depende de cómo se forman los precios. El equilibrio es el punto fijo de ese círculo, y el procedimiento estándar para encontrarlo es el método de los coeficientes indeterminados.

\subsubsection{El precio de equilibrio y la clave del modelo}

Igualando oferta y demanda, $\psi\left(p_{t}-\mathbb{E}_{t-1}[p_{t}]\right)=m_{t}-p_{t}$, y despejando el nivel de precios:
$$p_{t}=\frac{1}{1+\psi}\,m_{t}+\frac{\psi}{1+\psi}\,\mathbb{E}_{t-1}[p_{t}]$$
Tomando esperanzas condicionadas a $\Omega_{t-1}$ en los dos lados, la expectativa del precio se despeja de sí misma y queda la igualdad que resume todo el enfoque:
$$\mathbb{E}_{t-1}[p_{t}]=\mathbb{E}_{t-1}[m_{t}]$$
La previsión racional del nivel de precios es, exactamente, la previsión de la política monetaria. Sustituyendo arriba, precios y producción quedan en función de la parte esperada y la parte sorpresiva del dinero:
$$p_{t}=\mathbb{E}_{t-1}[m_{t}]+\frac{1}{1+\psi}\left(m_{t}-\mathbb{E}_{t-1}[m_{t}]\right),\qquad y_{t}=\frac{\psi}{1+\psi}\left(m_{t}-\mathbb{E}_{t-1}[m_{t}]\right)$$
El dinero esperado se traslada a los precios uno a uno y no toca la producción; la sorpresa monetaria se reparte, una fracción $1/(1+\psi)$ a los precios y una fracción $\psi/(1+\psi)$ a la producción. Solo lo imprevisto tiene efectos reales.

\subsubsection{La regla monetaria y el método de los coeficientes indeterminados}

Para cerrar el modelo falta decir cómo se comporta el dinero. La regla más sencilla con contenido es un paseo aleatorio con deriva (\authorfont{LUCAS}, 1973),
$$m_{t}=m_{t-1}+\mu+u_{t},\qquad u_{t}\sim N\!\left(0,\,\sigma^{2}_{u}\right)$$
con tres componentes: la dependencia del valor pasado, una deriva $\mu$ que hace de tendencia de la política, y un ruido blanco $u_{t}$ que es su componente imprevisible. Con esta regla, $\mathbb{E}_{t-1}[m_{t}]=m_{t-1}+\mu$.

El método de los coeficientes indeterminados consiste en \textit{conjeturar} que el precio de equilibrio es una función lineal de las variables de estado, con coeficientes aún por determinar,
$$p_{t}=\varphi_{0}+\varphi_{1}\,m_{t-1}+\varphi_{2}\,u_{t}$$
calcular con esa conjetura la expectativa, $\mathbb{E}_{t-1}[p_{t}]=\varphi_{0}+\varphi_{1}\,m_{t-1}$, sustituir ambas en la ecuación del precio e igualar los coeficientes de cada variable a ambos lados. La constante exige $\varphi_{0}=\frac{\mu}{1+\psi}+\frac{\psi}{1+\psi}\varphi_{0}$, de donde $\varphi_{0}=\mu$; el término en $m_{t-1}$ exige $\varphi_{1}=\frac{1}{1+\psi}+\frac{\psi}{1+\psi}\varphi_{1}$, de donde $\varphi_{1}=1$; y el término en $u_{t}$ da directamente $\varphi_{2}=\frac{1}{1+\psi}$. La conjetura se verifica a sí misma, que es precisamente lo que define un equilibrio con expectativas racionales, y el precio queda
$$p_{t}=m_{t-1}+\mu+\frac{1}{1+\psi}\,u_{t}$$

\subsubsection{Las formas reducidas de la producción y de la inflación}

De la solución anterior salen las dos formas reducidas del modelo. La producción solo carga con la sorpresa del periodo,
$$y_{t}=\frac{\psi}{1+\psi}\,u_{t}$$
y la inflación, restando el precio de ayer y usando que $m_{t-1}-m_{t-2}=\mu+u_{t-1}$,
$$\pi_{t}=\mu+\frac{1}{1+\psi}\,u_{t}+\frac{\psi}{1+\psi}\,u_{t-1}$$
La lectura separa con nitidez los dos componentes de la política. El componente \textit{sistemático}, la deriva $\mu$ y todo lo que la regla hace esperar, vive solo en los precios y en la inflación: es neutral. El componente \textit{no sistemático}, $u_{t}$, es el único con efectos reales, y los tiene una sola vez: la fracción de la sorpresa de ayer que ayer no pasó a los precios pasa hoy, que es lo que recoge el término en $u_{t-1}$ de la inflación.

Queda una pieza que cierra el círculo anunciado: la varianza del nivel de precios que los productores no anticipan se deduce de la propia solución,
$$\sigma^{2}_{p}=\mathrm{var}\left(p_{t}-\mathbb{E}_{t-1}[p_{t}]\right)=\frac{\sigma^{2}_{u}}{\left(1+\psi\right)^{2}}$$
de modo que $\sigma^{2}_{p}$ depende de la volatilidad de la política, $\sigma^{2}_{u}$, y a la vez $\psi$ depende de $\sigma^{2}_{p}$: el equilibrio es un punto fijo también en las varianzas. La pendiente de la oferta no está dada de antemano: la fabrica, sin querer, el régimen de política vigente.

\subsection{Lo que el modelo enseña}\label{anx1:lectura}

Del aparato anterior salen tres lecciones, que el cuerpo del tema recoge y que la política económica de los años ochenta convirtió en doctrina, y un obituario: las razones por las que el propio programa de los ciclos monetarios se agotó en menos de una década.

\subsubsection{Una curva de \authorfont{PHILLIPS} que existe y no puede usarse}

Las formas reducidas comparten el término $u_{t}$: cuando hay sorpresa monetaria positiva, suben a la vez la producción y la inflación. Un econometrista que mire los datos encontrará, por tanto, una correlación positiva entre inflación y actividad, una curva de \authorfont{PHILLIPS} con la forma observable de la curva aumentada de \authorfont{FRIEDMAN} ([\authorfont{Ver Tema 3.A.5}]), aunque el mecanismo sea otro: donde el monetarismo ponía ilusión monetaria transitoria, \authorfont{LUCAS} pone un problema de información en agentes plenamente racionales. La diferencia de mecanismo tiene una consecuencia brutal para el uso de la curva: como solo las sorpresas generan la correlación, y las sorpresas no pueden fabricarse de forma sistemática, la curva existe estadísticamente y no ofrece ningún menú de elección al gobierno.\footnote{%
    Los primeros contrastes respaldaron esta lectura: \authorfont{BARRO} (1977) encontró que los cambios \textit{no anticipados} del crecimiento monetario, contemporáneos y con un retardo, tienen un alto poder explicativo sobre las variaciones del desempleo estadounidense, mientras que los anticipados carecen de él.
}

\subsubsection{La ineficacia de la política sistemática}

Si la regla de política es conocida, su parte sistemática está dentro de $\mathbb{E}_{t-1}[m_{t}]$ y, por la clave del modelo, dentro de $\mathbb{E}_{t-1}[p_{t}]$: los agentes la incorporan antes de que actúe, y solo mueve precios. Es la \textit{proposición de ineficacia} de \authorfont{SARGENT} y \authorfont{WALLACE} (1975): ninguna regla de estabilización sistemática, por sofisticada que sea, afecta a la producción ni al empleo, porque lo que la regla responde a la información pública ya está descontado por quienes comparten esa información. A la autoridad solo le queda el componente $u_{t}$, y usarlo deliberadamente exigiría disponer de información relevante que los agentes no tienen; de lo contrario, estabilizar es imposible por construcción.\footnote{%
    Siguiendo a \authorfont{SARGENT} y \authorfont{WALLACE} (1975) puede hablarse incluso de \textit{superneutralidad} del dinero: los cambios en la variable monetaria carecen de efectos reales con independencia de si afectan a su nivel o a su tasa de variación, puesto que una senda de crecimiento del dinero conocida está, toda ella, dentro de las expectativas.
} Las consecuencias de este resultado para el diseño institucional, de las reglas a la independencia de los bancos centrales, se desarrollan en el anexo de política económica.

\subsubsection{La crítica de \authorfont{LUCAS} como corolario}

La pendiente $\psi$ contiene $\sigma^{2}_{p}$, y $\sigma^{2}_{p}$ la fabrica el régimen de política. Eso convierte a $\psi$ en todo lo contrario de un parámetro estructural: es una forma reducida que mezcla preferencias ($\varepsilon^{F}$) con el entorno de política en que se estimó. Un gobierno que, viendo una curva de oferta generosa, decida explotarla con expansiones repetidas, eleva $\sigma^{2}_{u}$ y con ella $\sigma^{2}_{p}$; los productores, que viven en un mundo más ruidoso, se fían menos de sus precios, $\theta$ sube, y $\psi$ cae: la relación que se quería explotar se aplana por el propio intento de explotarla.

\imagenfit{anx1_pendiente.pdf}{La pendiente $\psi$ depende del régimen: la crítica de \authorfont{LUCAS} dentro del propio modelo}{Elaboración propia a partir de \authorfont{LUCAS} (1973, 1976).}

Es el caso particular que \authorfont{LUCAS} (1976) elevó a principio general: los parámetros de los modelos econométricos estimados bajo un régimen no sobreviven al cambio de régimen, porque no son constantes de la naturaleza sino reglas de decisión óptimas de los agentes, y las reglas óptimas cambian cuando cambian las reglas del juego. De ahí su clasificación de los parámetros en dos familias: los \textit{profundos} o estructurales, que definen el comportamiento de los agentes y son invariantes a la política, como la preferencia temporal, la aversión al riesgo o la propia $\varepsilon^{F}$; y los que dependen de la formación de expectativas y, con ella, del régimen de política, como $\theta$ y $\psi$.\footnote{%
    Los modelos cuyos coeficientes mezclan las dos familias, como el IS-LM estimado, quedan atrapados en medio: las medidas de política alteran la posición y la inclinación de sus curvas a través de las expectativas incorporadas en los parámetros, y por eso sirven para predecir a corto plazo dentro de un régimen, pero no para comparar regímenes.
} La propia contrastación internacional de \authorfont{LUCAS} (1973) con dieciocho países funcionó como primera evidencia: en los países de demanda nominal estable la relación entre inflación y producción aparece generosa, y en los de demanda volátil, con ARGENTINA y PARAGUAY como casos extremos, la curva es casi vertical. La ampliación moderna de esa evidencia se estudia en [\authorfont{Ver Tema 3.A.42}].

\subsubsection{Por qué el programa se agotó}

El modelo de los ciclos monetarios murió joven, y por heridas bien identificadas. La primera es la \textit{persistencia}: las sorpresas $u_{t}$ no pueden estar autocorrelacionadas, porque dejarían de ser sorpresas, de modo que el modelo solo genera desviaciones de un periodo, cuando los ciclos observados duran años. Los intentos de alargar la respuesta añadiendo acumulación de capital y de existencias (\authorfont{LUCAS}, 1975, 1977; \authorfont{BLINDER} y \authorfont{FISCHER}, 1981) trasladaban la persistencia a mecanismos reales, lo que apuntaba ya fuera del programa ([\authorfont{Ver Tema 3.A.42}]). La segunda es la \textit{información}: el problema de extracción de señal exige que el nivel general de precios sea difícil de conocer, y los agregados monetarios y los índices de precios se publican pronto, con lo que sostener la confusión más allá de unas semanas resulta poco creíble. La tercera es la \textit{contrastación}: los trabajos de \authorfont{SIMS} (1980) y \authorfont{NELSON} y \authorfont{PLOSSER} (1982), y la revisión del propio \authorfont{BARRO} (1989), mostraron que las perturbaciones monetarias solo explican una parte pequeña de la variabilidad del desempleo y de la producción; la calibración de un modelo a la manera de \authorfont{LUCAS} con datos estadounidenses tampoco reproduce ni la volatilidad ni la persistencia observadas (\authorfont{COOLEY} y \authorfont{HANSEN})\textcolor{orange}{\textsuperscript{[Probable]}}.\footnote{%
    El propio \authorfont{LUCAS} lo reconoció sin rodeos años después, en una entrevista con \authorfont{SNOWDON} y \authorfont{VANE} (1998): \textit{that 75 paper was a dead end (...) it was an attempt to introduce some kind of usual dynamics into my 72 paper, and it did not work. I think that \authorfont{KYDLAND} and \authorfont{PRESCOTT}'s 1982 followed from that}.
}

La salida quedó señalada por la primera herida: si la propagación del ciclo ha de ser real, el impulso también puede serlo. Ese paso, sustituir la sorpresa monetaria por la perturbación tecnológica conservando intacto el resto del credo, es el ciclo real, y su manual ocupa el anexo siguiente.

\clearpage
\anexo{El manual del ciclo real}\label{anx:ciclo}

El anexo anterior terminó con un diagnóstico: si la propagación del ciclo ha de ser real, el impulso también puede serlo. Este anexo es el manual del modelo que dio ese paso y se convirtió en el segundo motor de la nueva macroeconomía clásica. Construye el modelo de abajo arriba: primero el giro que lo separa de los ciclos monetarios; después la intuición del mecanismo en su versión más pequeña, de dos periodos; luego el modelo completo, con el hogar, la empresa, la perturbación tecnológica y la acumulación del capital; sus condiciones de primer orden una a una; el equilibrio competitivo y su equivalencia con el problema de un planificador benévolo; el estado estacionario; y, al final, cómo se propaga por la economía una mejora de la productividad.

Tres convenciones valen para todo el anexo. Las variables van en niveles, no en logaritmos como en el anexo anterior, salvo $z_t$, que es el logaritmo de la productividad, y las variables con acento circunflejo, que son desviaciones porcentuales respecto al estado estacionario. El factor de descuento es $\delta$, como en el 3.A.42, y la tasa de depreciación del capital se escribe $\delta_K$ para distinguirla de aquel. Y el tiempo disponible se normaliza a uno, de modo que $h_t$ son las horas trabajadas y $L_t=1-h_t$ el ocio, la misma convención del tema de mercado de trabajo ([\authorfont{Ver Tema 3.A.25}]).

La frontera con el tema del ciclo es nítida: aquí están el modelo, sus condiciones y su estado estacionario; su solución estocástica, su calibración, las respuestas al impulso, la contrastación con los hechos estilizados, las extensiones y el coste de bienestar de las fluctuaciones se estudian en [\authorfont{Ver Tema 3.A.42}].

\subsection{El giro: del impulso monetario al impulso real}\label{anx2:giro}

\authorfont{KYDLAND} y \authorfont{PRESCOTT} (1982) y \authorfont{LONG} y \authorfont{PLOSSER} (1983) propusieron explicar el ciclo sin dinero y sin fricciones, como la respuesta óptima de una economía competitiva a cambios aleatorios de la productividad. La formulación canónica que sigue este anexo, ya depurada, es la de \authorfont{KING}, \authorfont{PLOSSER} y \authorfont{REBELO} (1988).\footnote{%
    El título del trabajo de \authorfont{KYDLAND} y \authorfont{PRESCOTT} alude a uno de sus ingredientes: la inversión tarda varios trimestres en convertirse en capital productivo, un retardo de construcción que ayuda a generar la forma del ciclo. \authorfont{LONG} y \authorfont{PLOSSER}, por su parte, trabajaron con varios sectores y con supuestos que permiten resolver el modelo con lápiz y papel, entre ellos la depreciación total del capital en cada periodo y la utilidad logarítmica.
}

Del credo de la escuela no se abandona nada. Los agentes siguen optimizando, las expectativas siguen siendo racionales, los mercados siguen vaciándose en todo momento y el ciclo sigue siendo un fenómeno de equilibrio. Lo que cambia son las dos piezas que el modelo de islas no conseguía sostener. El \textbf{impulso} deja de ser la sorpresa monetaria y pasa a ser una perturbación de la productividad, que es real y no necesita que nadie se confunda. Y la \textbf{propagación} deja de descansar en un error de información, que por su propia naturaleza no puede durar, y pasa a descansar en dos decisiones que sí duran: acumular capital y repartir el trabajo en el tiempo.

El cambio de impulso trae consigo una apuesta metodológica mucho más ambiciosa que la de \authorfont{LUCAS}: \textbf{un solo modelo para el crecimiento y para el ciclo}. La base no es un aparato construido para explicar fluctuaciones, sino el modelo neoclásico de crecimiento de \authorfont{RAMSEY} (1928), \authorfont{CASS} (1965) y \authorfont{KOOPMANS} (1965), al que basta añadir la elección entre trabajo y ocio y una perturbación estocástica en la tecnología. La tendencia y las oscilaciones alrededor de la tendencia dejan de ser dos objetos con dos teorías: son la misma ecuación con perturbaciones de distinto tamaño.\footnote{%
    De ahí el título programático de \authorfont{PRESCOTT} (1986), \textit{Theory ahead of business cycle measurement}: la teoría del crecimiento, ya disponible, se declaraba suficiente para medir el ciclo. La perturbación se identifica en la práctica con el residuo de la contabilidad del crecimiento de \authorfont{SOLOW} (1957), la productividad total de los factores, y los problemas de esa identificación se estudian en [\authorfont{Ver Tema 3.A.42}].
}

\subsection{La intuición en dos periodos: cuándo conviene trabajar}\label{anx2:dos}

Antes del modelo completo conviene ver el mecanismo en su versión más pequeña. Un hogar que vive dos periodos, consume $c_1$ y $c_2$, disfruta de un ocio $L_1$ y $L_2$ y trabaja el resto del tiempo a los salarios reales $w_1$ y $w_2$, puede prestar y pedir prestado al tipo de interés real $r$. Su problema es
$$\max_{\{c_1,\,c_2,\,L_1,\,L_2\}}\ u(c_1,L_1)+\delta\,u(c_2,L_2)\qquad \text{sujeto a}\qquad c_1+\frac{c_2}{1+r}=w_1\left(1-L_1\right)+\frac{w_2\left(1-L_2\right)}{1+r}$$
donde la restricción iguala el valor actual del consumo al valor actual de las rentas del trabajo. Llamando $\lambda$ al multiplicador de la restricción, las cuatro condiciones de primer orden son
$$u_{c}(1)=\lambda,\qquad \delta\,u_{c}(2)=\frac{\lambda}{1+r},\qquad u_{L}(1)=\lambda\,w_{1},\qquad \delta\,u_{L}(2)=\frac{\lambda\,w_{2}}{1+r}$$
y de sus cocientes salen las tres relaciones que gobernarán también el modelo completo:
$$\underbrace{u_{c}(1)=\delta\left(1+r\right)u_{c}(2)}_{\text{\authorfont{EULER}: reparto del consumo}}\qquad \underbrace{\frac{u_{L}(1)}{u_{c}(1)}=w_{1}}_{\text{intratemporal: consumo y ocio}}\qquad \underbrace{\frac{u_{L}(1)}{u_{L}(2)}=\delta\left(1+r\right)\frac{w_{1}}{w_{2}}}_{\text{intertemporal: reparto del trabajo}}$$

La tercera es el mecanismo del ciclo real, y conviene leerla despacio. El hogar no decide cuánto trabajar mirando el \textit{nivel} del salario, sino el salario de hoy frente al de mañana: \textbf{trabaja más cuando el salario es transitoriamente alto}, porque entonces el ocio resulta caro, y también cuando el tipo de interés es alto, porque lo que gane hoy rendirá más si lo ahorra. Es la sustitución intertemporal del trabajo de \authorfont{LUCAS} y \authorfont{RAPPING} (1969), y la intensidad con que opera es la \textbf{elasticidad de \authorfont{FRISCH}}, $\varepsilon^{F}$: en cuánto por ciento aumentan las horas de hoy frente a las de mañana cuando el salario de hoy sube un uno por ciento frente al de mañana.\footnote{%
    La elasticidad de \authorfont{FRISCH} mantiene constante la utilidad marginal de la riqueza, es decir, aísla el efecto sustitución del efecto renta, y por eso es la que aparece en las fluctuaciones. No debe confundirse con la elasticidad de sustitución intertemporal del \textit{consumo}, que mide cuánto se desplaza el consumo en el tiempo cuando cambia el tipo de interés. La oferta de trabajo estática, con sus efectos renta y sustitución y su posible tramo hacia atrás, se estudia en [\authorfont{Ver Tema 3.A.25}], y la tensión entre la elasticidad que el modelo necesita y la que miden los microdatos, en [\authorfont{Ver Tema 3.A.42}].
}

La distinción entre lo transitorio y lo permanente es, por tanto, decisiva. Una subida \textit{transitoria} del salario eleva $w_1/w_2$ y el hogar trabaja más hoy y menos mañana: la respuesta es grande, y se produce a lo largo de la misma curva de oferta. Una subida \textit{permanente} no altera ese cociente y solo hace al hogar más rico, de modo que el efecto renta compensa al de sustitución y las horas apenas cambian: la curva de oferta se desplaza.

\imagenfit{anx2_sustitucion.pdf}{La sustitución intertemporal del trabajo: una subida transitoria y una permanente del salario}{Elaboración propia a partir de \authorfont{LUCAS} y \authorfont{RAPPING} (1969).}

Ahí está, en miniatura, por qué el ciclo real necesita perturbaciones que no sean ni permanentes ni instantáneas, sino \textbf{persistentes pero transitorias}: solo esas mueven el salario de hoy frente al de mañana lo bastante como para que las horas trabajadas oscilen.

\subsection{El modelo completo}\label{anx2:modelo}

El modelo canónico tiene tres piezas: un hogar representativo que decide cuánto consumir, cuánto trabajar y cuánto acumular; una empresa representativa que produce con capital y trabajo y recibe cada periodo una perturbación en su productividad; y una tecnología de acumulación que convierte la inversión de hoy en el capital de mañana.

\subsubsection{El hogar}

El hogar representativo maximiza el valor esperado de su utilidad a lo largo de un horizonte infinito,
$$\max\ \mathbb{E}_{0}\sum_{t=0}^{\infty}\delta^{t}\,u\!\left(c_{t},L_{t}\right),\qquad L_{t}=1-h_{t},\qquad 0<\delta<1$$
donde $\delta$ mide su impaciencia. Es propietario del capital, que alquila a la empresa al precio $R_{t}$, de modo que su restricción presupuestaria en cada periodo es
$$c_{t}+K_{t+1}=w_{t}h_{t}+\left(1+r_{t}\right)K_{t},\qquad 1+r_{t}=R_{t}+1-\delta_{K}$$
es decir, lo que consume y ahorra en forma de capital nuevo sale de lo que gana trabajando y de lo que le renta el capital que ya tenía, una vez descontada la parte que se ha desgastado.\footnote{%
    La utilidad se supone creciente y cóncava en el consumo y en el ocio, y aditivamente separable en el tiempo. El supuesto de agente representativo, que permite tratar a todos los hogares como uno solo, es una de las piezas que la crítica discutirá: supone implícitamente que la distribución de la renta y de la riqueza no afecta al comportamiento agregado.
}

\subsubsection{La empresa y la perturbación tecnológica}

La empresa produce con una función de rendimientos constantes a escala,
$$Y_{t}=A_{t}K_{t}^{\alpha}h_{t}^{1-\alpha},\qquad 0<\alpha<1$$
donde $\alpha$ es la participación del capital en la renta. La productividad total de los factores, $A_{t}=e^{z_{t}}$, es la única fuente de perturbaciones del modelo, y su logaritmo sigue un proceso autorregresivo de primer orden:
$$z_{t}=\rho_{z}\,z_{t-1}+\nu^{z}_{t},\qquad 0<\rho_{z}<1,\qquad \nu^{z}_{t}\sim N\!\left(0,\,\sigma^{2}_{\nu}\right)$$
El parámetro $\rho_{z}$ mide la persistencia: una mejora no desaparece al periodo siguiente, sino que se desvanece poco a poco, que es justo lo que la sustitución intertemporal necesita. La perturbación es \textbf{exógena}: el modelo no explica de dónde vienen las mejoras y los empeoramientos de la productividad, y en ese punto se concentrará buena parte de la crítica posterior. Las versiones ampliadas añaden otras perturbaciones reales, como las del gasto público, y se estudian en [\authorfont{Ver Tema 3.A.42}].

\subsubsection{La acumulación del capital y la restricción de recursos}

El capital se deprecia a una tasa $\delta_{K}$ y se repone con la inversión,
$$K_{t+1}=\left(1-\delta_{K}\right)K_{t}+I_{t}$$
y la economía, al no comerciar con el exterior ni tener sector público, dedica todo lo que produce a consumir o a invertir:
$$c_{t}+I_{t}=Y_{t}$$
Estas dos ecuaciones son la segunda fuente de propagación del modelo: el capital enlaza unos periodos con otros, de modo que una perturbación de hoy sigue presente mañana en forma de maquinaria instalada, y ahí reside la persistencia que el modelo de islas no podía generar.

\subsection{Las condiciones de primer orden}\label{anx2:cpo}

Las decisiones del hogar y de la empresa se resumen en cuatro condiciones. Las tres primeras son las del hogar, y son las mismas que aparecieron en la versión de dos periodos, ahora con incertidumbre y horizonte infinito; la cuarta es la de la empresa y determina los precios de los factores.

\subsubsection{La ecuación de \authorfont{EULER}: el reparto del consumo}

Sacrificar una unidad de consumo hoy para invertirla cuesta $u_{c}(c_{t},L_{t})$ y rinde mañana $1+r_{t+1}$ unidades, valoradas con la utilidad marginal de entonces y descontadas. En el óptimo, el coste iguala al beneficio esperado:
$$u_{c}\!\left(c_{t},L_{t}\right)=\delta\,\mathbb{E}_{t}\!\left[u_{c}\!\left(c_{t+1},L_{t+1}\right)\left(1+r_{t+1}\right)\right]$$
De aquí sale el \textbf{reparto del consumo en el tiempo}: como la utilidad marginal decrece, el hogar prefiere un consumo estable a uno que siga a la renta, y por eso ahorra la renta transitoriamente alta en lugar de gastarla.

\subsubsection{La condición intratemporal entre consumo y ocio}

Dentro de cada periodo, la última hora de ocio debe valer lo mismo que lo que podría comprarse trabajándola:
$$\frac{u_{L}\!\left(c_{t},L_{t}\right)}{u_{c}\!\left(c_{t},L_{t}\right)}=w_{t}$$
es decir, la relación marginal de sustitución entre ocio y consumo iguala al salario real. Esta condición es la que convierte al empleo en una decisión y no en un residuo: en el modelo no hay desempleo involuntario, porque quien no trabaja es porque, al salario vigente, prefiere el ocio.

\subsubsection{La condición intertemporal del trabajo}

Combinando las dos anteriores en dos periodos consecutivos se obtiene la condición que da al modelo su mecanismo característico:
$$\frac{u_{L}\!\left(c_{t},L_{t}\right)}{u_{L}\!\left(c_{t+1},L_{t+1}\right)}=\delta\left(1+r_{t+1}\right)\frac{w_{t}}{w_{t+1}}$$
En desviaciones porcentuales respecto al estado estacionario, y prescindiendo de la incertidumbre para verla con claridad, esta condición adopta la forma en que la usa el tema del ciclo:
$$\hat h_{t}-\hat h_{t+1}=\varepsilon^{F}\left[\hat w_{t}-\hat w_{t+1}+\left(r_{t+1}-\bar r\right)\right]$$
donde $\hat h_{t}$ y $\hat w_{t}$ son las desviaciones de las horas y del salario real y $\bar r$ el tipo de interés del estado estacionario ([\authorfont{Ver Tema 3.A.42}]). Con incertidumbre, las magnitudes de mañana se sustituyen por sus valores esperados.

\subsubsection{Las demandas de factores}

La empresa, que toma los precios como dados, contrata trabajo y capital hasta igualar sus productividades marginales a sus precios:
$$w_{t}=\left(1-\alpha\right)\frac{Y_{t}}{h_{t}},\qquad R_{t}=\alpha\,\frac{Y_{t}}{K_{t}},\qquad r_{t}=R_{t}-\delta_{K}$$
Conviene reparar en lo que estas dos igualdades significan para el conjunto del modelo: \textbf{los precios de los factores no son datos, sino que los mueve la propia perturbación}. Una mejora de la productividad eleva a la vez el salario real y la rentabilidad del capital, y por eso el modelo genera un salario real procíclico sin necesidad de rigidez alguna, que era justo lo que las teorías con precios rígidos no conseguían.

\subsection{El equilibrio competitivo y el planificador}\label{anx2:equilibrio}

Con las cuatro condiciones, las restricciones y el proceso de la perturbación, el modelo está cerrado. Queda definir en qué consiste su solución y señalar la propiedad que le da su carga normativa: que coincide con lo que elegiría un planificador benévolo.

\subsubsection{La definición de equilibrio}

Un \textbf{equilibrio competitivo} es un conjunto de sendas de cantidades, $\{c_{t},h_{t},I_{t},K_{t+1}\}$, y de precios, $\{w_{t},R_{t}\}$, tales que, dado el estado inicial del capital y el proceso de la perturbación: las cantidades resuelven el problema del hogar a los precios vigentes; los precios son los que resuelven el problema de la empresa; los mercados de bienes, de trabajo y de capital se vacían en todos los periodos; y las expectativas sobre las variables futuras son racionales, es decir, se forman con el propio modelo. El equilibrio no es un punto, sino una \textbf{senda estocástica}: una regla que dice qué hace la economía en cada estado posible.

\subsubsection{La equivalencia con el problema del planificador}

Considérese ahora un planificador benévolo que maneja directamente las cantidades, sin precios de por medio, con las preferencias del hogar y la tecnología de la economía:
$$\max\ \mathbb{E}_{0}\sum_{t=0}^{\infty}\delta^{t}u\!\left(c_{t},1-h_{t}\right)\qquad \text{sujeto a}\qquad c_{t}+K_{t+1}-\left(1-\delta_{K}\right)K_{t}=A_{t}K_{t}^{\alpha}h_{t}^{1-\alpha}$$
Sus condiciones de primer orden son exactamente las del equilibrio competitivo, con los precios sombra ocupando el lugar de los precios de mercado: la utilidad marginal del consumo hace de multiplicador, la productividad marginal del trabajo hace de salario y la del capital, de rentabilidad. La razón es el primer teorema fundamental de la economía del bienestar: en una economía de competencia perfecta, con mercados completos y sin distorsiones ni externalidades, el equilibrio es un óptimo de \authorfont{PARETO}.\footnote{%
    El marco de fondo es el equilibrio general de \authorfont{ARROW} y \authorfont{DEBREU}, cuyas aportaciones, poco anteriores, dieron a la escuela la arquitectura sobre la que montar el vaciado continuo. La equivalencia se rompe en cuanto se introducen impuestos distorsionantes, poder de mercado, externalidades o información asimétrica, y ese es precisamente el camino que tomó la nueva economía keynesiana para recuperar un papel para la política ([\authorfont{Ver Tema 3.A.7}]).
}

De esa equivalencia salen dos consecuencias de muy distinto alcance. La primera es \textbf{técnica}: como el problema del planificador no tiene precios, es mucho más fácil de resolver, y la práctica habitual consiste en resolverlo y recuperar después los precios con las productividades marginales. La segunda es \textbf{sustantiva} y explica la fama del modelo: si el equilibrio es óptimo en todo momento, \textbf{el ciclo es eficiente}. Una recesión no es un fallo que corregir, sino el periodo en que conviene trabajar e invertir menos porque los factores rinden temporalmente menos, y una política que la combatiera reduciría el bienestar. Además, el dinero no aparece por ninguna parte: las variables reales se determinan sin él y se mantiene la \textbf{dicotomía clásica}.

\subsection{El estado estacionario}\label{anx2:estacionario}

El estado estacionario es el punto al que la economía converge cuando las perturbaciones cesan ($z_{t}=0$, $A_{t}=1$) y las variables dejan de cambiar. Importa por dos razones: es el centro alrededor del cual oscila el ciclo, de modo que las fluctuaciones se miden como desviaciones respecto a él, y es lo que permite poner números al modelo, porque sus relaciones ligan los parámetros con proporciones observables de la contabilidad nacional.

\subsubsection{Las condiciones del estado estacionario}

En el estado estacionario el consumo se repite, de modo que la ecuación de \authorfont{EULER} se reduce a
$$1=\delta\left(1+\bar r\right)\qquad\Longrightarrow\qquad \bar r=\frac{1}{\delta}-1$$
un resultado notable: \textbf{el tipo de interés real de largo plazo lo fija la impaciencia}, no la tecnología. La tecnología determina, en cambio, cuánto capital se acumula, porque la rentabilidad del capital debe igualar ese tipo más la depreciación:
$$\alpha\,\frac{\bar Y}{\bar K}=\bar R=\bar r+\delta_{K}=\frac{1}{\delta}-1+\delta_{K}\qquad\Longrightarrow\qquad \frac{\bar K}{\bar Y}=\frac{\alpha}{\frac{1}{\delta}-1+\delta_{K}}$$

\imagenfit{anx2_estacionario.pdf}{El estado estacionario: el capital se acumula hasta que su rendimiento iguala la impaciencia}{Elaboración propia a partir de \authorfont{KING}, \authorfont{PLOSSER} y \authorfont{REBELO} (1988).}

Como el capital ni crece ni mengua, la inversión se limita a reponer lo que se desgasta, y de ahí salen las proporciones del gasto:
$$\bar I=\delta_{K}\bar K\qquad\Longrightarrow\qquad \frac{\bar I}{\bar Y}=\delta_{K}\frac{\bar K}{\bar Y},\qquad \frac{\bar c}{\bar Y}=1-\delta_{K}\frac{\bar K}{\bar Y}$$
Las horas trabajadas, por su parte, salen de la condición intratemporal. Con la utilidad logarítmica habitual, $u=\ln c+\chi\ln L$, esa condición se escribe $\chi\,\bar c/\bar L=\bar w=(1-\alpha)\bar Y/\bar h$, y el parámetro $\chi$ se elige de modo que el hogar dedique al trabajo la fracción del tiempo que se observa. Con esas preferencias, además, la elasticidad de \authorfont{FRISCH} queda determinada por las propias horas, $\varepsilon^{F}=\bar L/\bar h=(1-\bar h)/\bar h$, de manera que trabajar un tercio del tiempo implica una elasticidad de $2$: un valor muy alto frente al que estiman los microdatos, y ese contraste será el talón de Aquiles del modelo ([\authorfont{Ver Tema 3.A.42}]).

\subsubsection{Las grandes proporciones y el crecimiento equilibrado}

Las relaciones anteriores no son solo un punto de partida para medir el ciclo: son también las que hacen del modelo una teoría del crecimiento. Si la tecnología crece a una tasa constante, la economía recorre una \textbf{senda de crecimiento equilibrado} en la que la producción, el consumo, la inversión y el capital crecen al mismo ritmo, las proporciones $\bar c/\bar Y$, $\bar I/\bar Y$ y $\bar K/\bar Y$ permanecen constantes, el tipo de interés real no tiene tendencia y las horas trabajadas tampoco. \authorfont{KING}, \authorfont{PLOSSER} y \authorfont{REBELO} (1988) mostraron qué exige eso de las preferencias: que el efecto renta y el efecto sustitución de una subida permanente del salario se cancelen exactamente, lo que restringe la utilidad a una familia concreta, de la que la logarítmica es el caso más usado.\footnote{%
    Es la misma condición que en la versión de dos periodos hacía que una subida permanente del salario no moviera las horas. Sin ella, el modelo predeciría que las horas trabajadas por persona han caído o crecido sin límite a lo largo de dos siglos de salarios crecientes, lo que contradice la observación.
}

Que las mismas ecuaciones describan la tendencia y las oscilaciones es la pretensión más ambiciosa de esta literatura: \textbf{el crecimiento y el ciclo dejan de ser dos teorías} y pasan a ser el mismo modelo sometido a perturbaciones de distinto tamaño y duración.

\subsection{La propagación de una mejora de la productividad}\label{anx2:propagacion}

Reunidas las piezas, puede seguirse el recorrido de una mejora transitoria pero persistente de la productividad. En el impacto, la empresa produce más con los mismos factores y, como la productividad marginal del trabajo sube, la demanda de trabajo se desplaza y con ella el salario real.

\imagenfit{anx2_tecnologia.pdf}{Una mejora de la productividad en el mercado de trabajo: sin oferta elástica no hay ciclo}{Elaboración propia.}

Como la mejora es transitoria, el salario de hoy sube frente al de mañana y el hogar responde trabajando más, de modo que \textbf{la producción aumenta más que la propia productividad}: la oferta de trabajo amplifica el impulso. Ahí se ve también la condición que el modelo necesita y que tanto le costará defender: si la oferta de trabajo fuera poco elástica, la mejora se traduciría en salarios y no en empleo, y el ciclo se apagaría.

El segundo canal es el del ahorro. Como la renta es transitoriamente alta, el hogar no la consume: la invierte. Por eso el modelo reproduce sin esfuerzo uno de los hechos estilizados más robustos del ciclo, que \textbf{la inversión oscila mucho más que el consumo}, y por eso la recesión aparece como un periodo en que conviene sembrar menos porque la tierra rinde menos.

El tercer elemento es la persistencia, y aquí el modelo es más débil de lo que parece. Una parte la hereda de la propia perturbación, a través de $\rho_{z}$, y otra la aporta el capital acumulado, que mantiene el efecto vivo mientras se desgasta; pero si la mejora durase un solo periodo, la inversión de ese periodo apenas cambiaría el capital y la economía volvería casi de inmediato a su posición. El modelo básico, en otras palabras, \textbf{toma prestada del impulso buena parte de la persistencia que debería generar él mismo}.

Hasta aquí el modelo. Lo que sigue, que es lo que convirtió esta construcción en un programa de investigación y, más tarde, en el blanco de sus críticos, pertenece al tema del ciclo: cómo se resuelve el modelo estocástico, cómo se calibran sus parámetros, qué respuestas al impulso genera, hasta qué punto reproduce los hechos estilizados, qué extensiones hicieron falta para que lo hiciera y cuánto vale, en términos de bienestar, eliminar las fluctuaciones ([\authorfont{Ver Tema 3.A.42}]).

\anexo{La política, los autores y las batallas}\label{anx:politica}

Los dos anexos anteriores han construido los modelos. Este reúne lo que la escuela dijo sobre la política económica y lo que la escuela fue como comunidad científica. Empieza por lo examinable: la ineficacia de la política monetaria sistemática, la inconsistencia temporal y las soluciones institucionales que inspiró, y la cara fiscal del mismo argumento, la equivalencia ricardiana. Sigue con los autores, las universidades y el método que los unía. Recoge después la crítica de \authorfont{LUCAS} en sus propias palabras. Y termina con las batallas que libró la escuela: quién la combatió, con qué argumentos y cómo acabó cada frente.

En los dos primeros apartados, $\alpha$ mide la respuesta de la actividad a la inflación no esperada, $\beta$ el peso que la autoridad da al objetivo de actividad frente al de inflación y $k$ la distancia entre el objetivo de actividad y el nivel natural, que es la notación con la que el tema de integración monetaria remite aquí ([\authorfont{Ver Tema 3.B.16}]).\footnote{%
    En el anexo del ciclo real, $\alpha$ era la participación del capital en la renta. Son dos literaturas distintas y cada una conserva la letra con la que se escribe habitualmente; dentro de cada anexo, el símbolo tiene un solo significado.
}

\subsection{La ineficacia de la política sistemática}\label{anx3:ineficacia}

La proposición de ineficacia es la primera consecuencia de política que la escuela extrajo de su aparato, y la más radical. No dice que la política monetaria sea inútil en general, sino algo más incómodo: que la parte de ella que puede planificarse es, precisamente, la que no sirve para estabilizar.

\subsubsection{El resultado de \authorfont{SARGENT} y \authorfont{WALLACE} (1975)}

Tómese el modelo del anexo de las islas ([\authorfont{Ver Anexo \ref{anx:islas}}]) y supóngase que la autoridad monetaria, lejos de actuar al azar, sigue una regla de estabilización que responde al estado de la economía con toda la información disponible,
$$m_{t}=g\!\left(\Omega_{t-1}\right)+u_{t}$$
donde $g(\cdot)$ puede ser tan sofisticada como se quiera: expandir cuando la producción del periodo anterior cayó, contraer cuando se recalentó, responder a cuantos indicadores se deseen. Como esa regla es una función de información pasada, los agentes la conocen y la incorporan, de modo que $\mathbb{E}_{t-1}[m_{t}]=g(\Omega_{t-1})$ y la producción queda
$$y_{t}=\frac{\psi}{1+\psi}\left(m_{t}-\mathbb{E}_{t-1}[m_{t}]\right)=\frac{\psi}{1+\psi}\,u_{t}$$
\textbf{La función $g$ ha desaparecido}. Y conviene medir bien el alcance de esa desaparición: no es solo que la política no pueda elevar la producción media, es que tampoco puede reducir su \textit{variabilidad}, que era el objetivo declarado de toda política de estabilización. La distribución entera de la producción es independiente de la regla elegida.\footnote{%
    Por eso puede hablarse de \textit{superneutralidad} del dinero: los cambios en la variable monetaria carecen de efectos reales con independencia de si afectan a su nivel o a su tasa de variación, pues una senda de crecimiento del dinero conocida está, toda ella, dentro de las expectativas.
}

El corolario para el debate entre reglas y discreción es inmediato, aunque paradójico: si la estabilización activa no sirve, más vale una regla simple y anunciada, que al menos reduce la incertidumbre sobre la política y, con ella, la varianza de los precios de la que depende la propia pendiente de la oferta.

\subsubsection{Las grietas del resultado}

La proposición no tardó en recibir el mismo trato que su autor había dado a los modelos de la Síntesis: se buscaron los supuestos de los que dependía, y resultaron ser tres.

El primero es la \textbf{igualdad de información}. Si la autoridad sabe algo que el público no sabe, o lo sabe antes, le queda margen para actuar; el propio resultado lo admite, aunque sus autores dudaran de que un banco central tenga ventaja informativa duradera sobre los mercados.

El segundo, y el que abrió la puerta a toda una literatura, es la \textbf{flexibilidad total de precios y salarios}. \authorfont{FISCHER} (1977) y \authorfont{TAYLOR} (1979) mostraron que basta con introducir contratos nominales que duran más de un periodo, sin tocar las expectativas racionales, para que la política sistemática recupere sus efectos reales: la autoridad puede reaccionar a perturbaciones ocurridas después de firmado el contrato y antes de que pueda renegociarse. Agentes plenamente racionales y política eficaz resultaban, pues, compatibles, y sobre esa compatibilidad se construyó la primera oleada neokeynesiana ([\authorfont{Ver Tema 3.A.7}]).

El tercero lo señaló \authorfont{TOBIN} con una observación que el tiempo confirmó: de los dos pilares de la escuela, las expectativas racionales y el vaciado continuo de los mercados, \textbf{el que tiene consecuencias de política no es el primero, sino el segundo}. Puede aceptarse la racionalidad de las expectativas y seguir defendiendo la intervención, con tal de admitir que los mercados no se vacían instantáneamente.

\subsection{La inconsistencia temporal}\label{anx3:kp}

La segunda aportación es más profunda que la primera, porque no depende del modelo de islas ni de ninguna teoría particular del ciclo, y porque su alcance desborda la macroeconomía: se aplica a cualquier situación en la que alguien decide hoy sobre mañana sabiendo que mañana tendrá la tentación de cambiar de idea.

\subsubsection{El problema en palabras}

Una decisión es \textbf{temporalmente inconsistente} cuando el plan que hoy se considera óptimo para mañana deja de serlo cuando llega mañana, sin que hayan cambiado ni los objetivos de quien decide ni la información disponible. Lo único que ha cambiado es que las expectativas privadas ya se han formado y se han convertido en un dato: lo que era una variable que influir se ha vuelto una restricción heredada.

El estudio moderno de las preferencias dinámicamente inconsistentes arranca de \authorfont{STROTZ} (1955), y fue \authorfont{KYDLAND} y \authorfont{PRESCOTT} (1977) quienes lo llevaron a la política económica, en el trabajo que les valdría el Nobel casi treinta años después. Su tesis es demoledora para la práctica discrecional: el problema de una autoridad que no puede comprometerse no es que tenga malos objetivos, sino que \textbf{su propia libertad de elegir la perjudica}.

\subsubsection{El modelo}

Sea una economía en la que la inflación no esperada reduce el paro, según la curva de \authorfont{PHILLIPS} aumentada con expectativas ([\authorfont{Ver Tema 3.A.5}]),
$$U_{t}=U_{n}-\alpha\left(\pi_{t}-\pi^{e}_{t}\right),\qquad \alpha>0$$
y una autoridad benévola que quiere acercarse a un objetivo de inflación, $\pi^{*}$, y a un objetivo de paro, $U^{*}=U_{n}-k$, con $k>0$, es decir, \textbf{más ambicioso que la tasa natural}. Su pérdida es cuadrática en las dos desviaciones:
$$\mathcal{L}=\left(\pi_{t}-\pi^{*}\right)^{2}+\beta\left(U_{t}-U^{*}\right)^{2}$$
donde $\beta$ mide cuánto pesa el objetivo de paro frente al de inflación: una autoridad dura con la inflación, lo que la literatura llama \textit{hard-nosed}, tiene un $\beta$ pequeño.\footnote{%
    Que el objetivo de paro sea inferior al natural, $k>0$, es el supuesto que lo pone todo en marcha, y tiene dos lecturas. Una, benévola: el paro natural no es eficiente, porque los impuestos, el poder de negociación o las prestaciones lo elevan por encima de lo deseable, de modo que la autoridad persigue legítimamente un empleo mayor. Otra, política: la presión de los agentes sociales y el ciclo electoral empujan a cualquier gobierno a querer menos paro del que la economía puede sostener. El resultado del modelo es el mismo con las dos.
}

Con \textbf{compromiso}, la autoridad elige la inflación sabiendo que los agentes la anticiparán, de modo que $\pi^{e}=\pi$ y el paro será el natural haga lo que haga; su pérdida se reduce entonces a $(\pi-\pi^{*})^{2}+\beta k^{2}$, que se minimiza anunciando y cumpliendo $\pi=\pi^{*}$.

Con \textbf{discreción}, en cambio, la autoridad elige una vez que las expectativas están dadas, y entonces sí puede comprar empleo con inflación. Derivando la pérdida respecto a $\pi$ con $\pi^{e}$ fijo, y usando que $\partial U/\partial\pi=-\alpha$,
$$\left(\pi_{t}-\pi^{*}\right)-\alpha\beta\left(U_{t}-U^{*}\right)=0$$
En equilibrio, los agentes no se equivocan sistemáticamente, de modo que $\pi^{e}=\pi$, el paro vuelve a ser el natural y $U_{t}-U^{*}=k$. Sustituyendo, el resultado es el \textbf{sesgo inflacionista}:
$$\pi^{d}=\pi^{*}+\alpha\beta k,\qquad U_{t}=U_{n}$$
Es decir, \textbf{más inflación que la deseada y exactamente el mismo paro}. La discreción no compra empleo: solo inflación. Y el sesgo es tanto mayor cuanto más responde el paro a la sorpresa ($\alpha$), cuanto más pesa el empleo en las preferencias de la autoridad ($\beta$) y cuanto más ambicioso es su objetivo ($k$).\footnote{%
    La versión que se escribe en producción en lugar de en paro, $y_{t}=\bar y+\alpha(\pi_{t}-\pi^{e}_{t})$ con un objetivo $\bar y+k$, da exactamente la misma expresión, y es la que usa [\authorfont{Ver Tema 3.B.16}] al estudiar la credibilidad importada en una unión monetaria, siguiendo la presentación de \authorfont{BARRO} y \authorfont{GORDON} (1983). Con un objetivo de inflación nulo, el sesgo se reduce a $\pi^{d}=\alpha\beta k$.
}

\subsubsection{El juego y el equilibrio}

El resultado anterior se entiende mejor como el desenlace de un juego entre la autoridad y los agentes, que es como lo presentaron sus autores.

\imagenfit{anx3_kp.pdf}{El juego de \authorfont{KYDLAND} y \authorfont{PRESCOTT}: del anuncio no creíble al sesgo inflacionista}{Elaboración propia a partir de \authorfont{KYDLAND} y \authorfont{PRESCOTT} (1977) y \authorfont{BARRO} y \authorfont{GORDON} (1983).}

Partiendo de $A$, donde la autoridad ha anunciado inflación cero y los agentes la han creído, el anuncio no es creíble precisamente porque, una vez creído, a la autoridad le conviene incumplirlo: con $\pi^{e}=0$ prefiere $B$, que está sobre la misma curva de corto plazo pero en una \textbf{curva de pérdida menor}, es decir, mejor para ella. Los agentes, que conocen sus preferencias, anticipan esa tentación y elevan sus expectativas, con lo que la curva de corto plazo se desplaza y la economía pasa a $C$: la misma inflación que en $B$, pero con el paro de vuelta en su nivel natural. Repetido el razonamiento, el proceso avanza hasta $E$, donde la tentación se agota porque la curva de pérdida es tangente a la curva de corto plazo justo sobre la vertical de largo plazo.

$E$ es un equilibrio de \authorfont{NASH}: dadas las expectativas, la autoridad no quiere desviarse; dada la conducta de la autoridad, las expectativas son correctas. Y es un equilibrio peor que $A$ para todos, porque tiene la misma tasa de paro y más inflación. \textbf{El problema no son los objetivos de la autoridad, que es benévola, sino su incapacidad de comprometerse}, y de ahí que la solución haya de buscarse en instituciones y no en intenciones.

\subsection{Las soluciones: cómo se atan las manos un gobierno}\label{anx3:soluciones}

Si el mal nace de la discrecionalidad, el remedio consiste en restringirla. La literatura posterior ensayó cuatro vías, cada una con su coste, y las cuatro acabaron reconociéndose en la arquitectura institucional que rige hoy.

\subsubsection{Reglas}

La primera respuesta es fijar de antemano la regla de política, de modo que los agentes puedan deducir las reglas del juego y formar sus expectativas sobre ellas. Pero una regla \textit{anunciada} no resuelve nada por sí sola: mientras la autoridad conserve la capacidad de incumplirla, la tentación permanece y los agentes lo saben, de modo que hace falta algún mecanismo que haga costoso el incumplimiento.\footnote{%
    El equilibrio del modelo incluye tres piezas: una regla de decisión de los agentes privados, dada su información; una función que determine sus expectativas; y una regla de política que especifique las acciones de la autoridad. Para que el resultado sea un equilibrio racional, la regla privada debe ser óptima dadas las expectativas, y debe ser óptimo para la autoridad actuar conforme a esas expectativas, conociendo la regla privada. Como la autoridad elige de forma secuencial, una vez formadas las expectativas, la coincidencia entre lo esperado y lo realizado es una propiedad del equilibrio, no algo garantizado de antemano.
}\textsuperscript{,}\footnote{%
    Las reglas tienen además un coste de rigidez: para cubrir eventualidades imprevistas tienden a complicarse, y al complicarse pierden lo que las hacía útiles, que era ser verificables por cualquiera.
}

\subsubsection{Delegación}

La segunda es encargar la política monetaria a un banco central independiente y de preferencias más duras que las de la sociedad, es decir, con un $\beta$ menor que el del gobierno: el \textbf{banquero conservador} de \authorfont{ROGOFF} (1985). El sesgo se reduce en la misma proporción en que se reduce $\beta$, pero aparece un coste: ese banquero también estabilizará menos de lo deseable cuando lleguen perturbaciones reales, porque le importa menos el empleo. La delegación compra credibilidad al precio de una peor estabilización.

Desde finales de los años ochenta se acumuló evidencia de una relación negativa estrecha entre la independencia de los bancos centrales y la inflación. \authorfont{POSEN} (1993) opuso a esa lectura una objeción difícil de refutar: la correlación no prueba causalidad, porque las instituciones independientes pueden ser el \textit{producto} de sociedades particularmente aversas a la inflación, como la alemana tras las hiperinflaciones de entreguerras, y no su causa; a lo que se añade que los propios índices de independencia incorporan, entre sus componentes, que el control de la inflación figure en el mandato de la institución.

\subsubsection{Reputación}

La tercera vía no cambia las instituciones, sino el horizonte. Si el juego se repite, \authorfont{BARRO} y \authorfont{GORDON} (1983) muestran que la autoridad puede sostener una inflación baja sacrificando la ganancia de corto plazo a cambio de una reputación que le rinde después: los agentes mantienen expectativas contenidas mientras la autoridad cumple, y la castigan con expectativas altas en cuanto incumple. El resultado depende de la paciencia: cuanto más peso dé la autoridad al futuro, más sostenible es la inflación baja. El ejemplo que siempre se cita es la Reserva Federal de \authorfont{PAUL VOLCKER} entre 1979 y 1987, que pagó una recesión severa por adquirir esa reputación.

\subsubsection{Contratos}

La cuarta vía escribe el incentivo en un contrato. \authorfont{WALSH} (1995) y \authorfont{PERSSON} y \authorfont{TABELLINI} (1993) añaden a la función de pérdida del banco central un término de coste pactado con el gobierno, cuya cuantía óptima es exactamente el sesgo que generaría la discreción, de modo que la tentación se neutraliza sin sacrificar la respuesta a las perturbaciones. \authorfont{SVENSSON} (1997) demostró una equivalencia elegante: el mismo resultado se obtiene encomendando a la autoridad un objetivo de inflación \textit{por debajo} del socialmente óptimo.

\subsubsection{El paralelo fiscal}

La misma lógica se trasladó al terreno presupuestario desde los años noventa. Si los gobiernos tienen un sesgo deficitario análogo al sesgo inflacionista, la respuesta son reglas fiscales y autoridades fiscales independientes que las vigilen, que es la arquitectura de la Unión Económica y Monetaria y de su Pacto de Estabilidad y Crecimiento ([\authorfont{Ver Tema 3.B.16}]).\footnote{%
    Una revisión amplia de esta literatura, en \authorfont{CALMFORS} y \authorfont{WREN-LEWIS} (2011). Entre las referencias habituales figuran también \authorfont{CASTELLANI} y \authorfont{DEBRUN} (2005), \authorfont{DEBRUN} y \authorfont{KUMAR} (2007) y \authorfont{BEETSMA} y \authorfont{DEBRUN} (2016)\textcolor{orange}{\textsuperscript{[Probable]}}. El sesgo deficitario como fallo del sector público, con el modelo de economía política de \authorfont{ALESINA} y \authorfont{TABELLINI}, se estudia en [\authorfont{Ver Tema 3.A.23}].
}

\subsection{La cara fiscal: la equivalencia ricardiana}\label{anx3:barro}

La política monetaria no fue el único blanco. \authorfont{BARRO} (1974) aplicó el mismo principio, agentes racionales que miran al futuro y conocen la restricción presupuestaria del Estado, a la política fiscal, y obtuvo un resultado igual de incómodo para la tradición keynesiana.

\subsubsection{El resultado}

Considérese un gobierno que ha decidido un gasto $G_{1}$ hoy y $G_{2}$ mañana, y que puede financiarlo con impuestos hoy, $T_{1}$, o emitiendo deuda $D$ que habrá de devolver mañana con impuestos $T_{2}$. Su restricción intertemporal obliga a que el valor actual de los impuestos iguale al del gasto:
$$T_{1}+\frac{T_{2}}{1+r}=G_{1}+\frac{G_{2}}{1+r}$$
El hogar, por su parte, consume con cargo a su renta después de impuestos,
$$c_{1}+\frac{c_{2}}{1+r}=\left(Y_{1}-T_{1}\right)+\frac{Y_{2}-T_{2}}{1+r}$$
y basta sustituir la primera ecuación en la segunda para que los impuestos desaparezcan:
$$c_{1}+\frac{c_{2}}{1+r}=Y_{1}+\frac{Y_{2}}{1+r}-\left(G_{1}+\frac{G_{2}}{1+r}\right)$$
La riqueza del hogar depende del \textbf{valor actual del gasto público}, y no del calendario de impuestos con que se financie. Una rebaja de impuestos financiada con deuda no hace más rico a nadie: anuncia impuestos futuros de igual valor actual, y el hogar ahorra íntegramente la rebaja para pagarlos. \textbf{La deuda pública no es riqueza neta}.

\imagenfit{anx3_barro.pdf}{La equivalencia ricardiana: tres formas de financiar el mismo gasto, un solo consumo}{Elaboración propia a partir de \authorfont{BARRO} (1974).}

\subsubsection{Los supuestos}

El resultado es un teorema, y como todo teorema vale lo que valgan sus supuestos, que son cinco:
\begin{lnum}
    \item \textit{Agentes racionales con horizonte efectivamente infinito}. O bien viven para siempre, o bien se preocupan por sus herederos lo bastante como para dejarles con qué pagar la deuda, que es la vía del legado por la que \authorfont{BARRO} resolvió el problema de los horizontes finitos;
    \item \textit{Certidumbre sobre los impuestos futuros}. Los agentes saben que la deuda se pagará con impuestos y cuándo;
    \item \textit{Impuestos de suma fija}, neutrales en las decisiones de consumo y ocio;\footnote{%
        Es el supuesto más frágil, porque el único impuesto neutral es el de suma fija, y la primera norma de la hacienda pública es que esos impuestos no se usan. Cualquier figura real, del IVA al impuesto sobre la renta, altera la restricción del agente y modifica su elección entre consumo y ocio, de modo que el calendario de impuestos sí importa: distribuir la carga en el tiempo cambia las distorsiones acumuladas.
    }
    \item \textit{Mercados de capitales perfectos}, sin restricciones de liquidez, de modo que el hogar pueda prestar y pedir prestado al mismo tipo que el Estado; y,
    \item \textit{Gasto no productivo}. El gasto financiado es gasto corriente y no una inversión que eleve el crecimiento potencial y se pague a sí misma con mayores ingresos futuros.
\end{lnum}

\subsubsection{Las tres vías de financiación y los efectos no keynesianos}

Bajo esos supuestos, las tres formas de financiar el mismo gasto son equivalentes. Si se emite \textit{deuda}, el ahorro privado aumenta para hacer frente al impuesto futuro; si se \textit{reducen impuestos} hoy, la rebaja se ahorra entera; y si se \textit{monetiza}, la expectativa de emisión continuada genera ahorro en previsión del impuesto inflacionario. Lo que importa, por tanto, es cuánto gasta el Estado y en qué, no cómo lo financia.

De aquí sale una conclusión que la política europea de la última década hizo célebre: la existencia de \textbf{efectos no keynesianos} de la política fiscal. Con agentes que conocen la restricción presupuestaria del Estado, una consolidación creíble puede resultar expansiva, porque la expectativa de menores impuestos futuros anima el gasto privado y porque la reducción de la prima de riesgo abarata la financiación de todos.

\subsubsection{Dónde falla}

Las grietas están en los propios supuestos. Los hogares con restricciones de liquidez consumen lo que reciben, de modo que una transferencia financiada con deuda sí eleva su consumo; los impuestos reales distorsionan; la incertidumbre sobre quién pagará los impuestos futuros rompe la equivalencia incluso con agentes altruistas; y la evidencia empírica encuentra, en general, compensaciones parciales del ahorro privado, no completas. El resultado conserva, sin embargo, su valor como \textbf{referencia}: no describe la economía, pero señala con precisión qué hay que suponer para que el déficit tenga los efectos que la tradición keynesiana le atribuía.

\subsubsection{La aritmética monetarista desagradable}

Las dos caras de la política acaban reunidas por la restricción presupuestaria del Estado, y de esa reunión sale un resultado que los propios \authorfont{SARGENT} y \authorfont{WALLACE} (1981) llamaron aritmética desagradable\textcolor{orange}{\textsuperscript{[Probable]}}. Si la política fiscal es \textit{dominante}, es decir, si los déficits se deciden con independencia de lo que haga el banco central, la autoridad monetaria no puede controlar la inflación de manera indefinida: podrá no monetizar hoy, pero entonces la deuda crece, y cuanto mayor sea la deuda acumulada mayor habrá de ser la emisión futura para sostenerla. Apretar hoy significa, en ese régimen, más inflación mañana.

La lección es la que une este anexo: la independencia del banco central y la mejor de las reglas monetarias no bastan si la política fiscal no acompaña, porque \textbf{la inflación es, en último término, un fenómeno de las dos políticas a la vez}.

\subsection{Los autores y las universidades}\label{anx3:autores}

La escuela fue, además de un conjunto de modelos, una generación concreta en un puñado de departamentos. Conviene tener presentes sus nombres, sus fechas y qué aportó cada uno, porque la cronología de la revolución es también su argumento: cada pieza responde a la anterior.

\subsubsection{Chicago}

\paragraph{\authorfont{ROBERT LUCAS}}
Nacido en 1937 en Yakima, Washington, en una familia modesta golpeada por la Gran Depresión, estudió Historia y se licenció en 1959 en la Universidad de Chicago. Tras un breve paso por Berkeley volvió a Chicago, donde se doctoró en Economía en 1964 con una tesis sobre la sustitución entre trabajo y capital en la industria estadounidense entre 1929 y 1958, dirigida por \authorfont{H. GREGG LEWIS} y \authorfont{DALE JORGENSON}. Enseñó en Carnegie Mellon hasta 1975 y pasó el resto de su carrera en Chicago.

Su aportación de 1972 mostró que, si los agentes anticipan la política monetaria, el dinero es neutral salvo sorpresa; la crítica de 1976 convirtió esa idea en un estándar metodológico para toda la profesión. Trabajó además sobre crecimiento, estructura urbana, comercio internacional, activos financieros y consumo. En 1995 recibió el Premio Nobel \textit{por haber desarrollado y aplicado la hipótesis de las expectativas racionales, transformando así el análisis macroeconómico y profundizando nuestra comprensión de la política económica}.

\paragraph{\authorfont{THOMAS SARGENT}}
Nacido en 1943 en Pasadena, California, se licenció con distinción en Berkeley en 1964 y se doctoró en Harvard en 1968, con una tesis sobre la estructura de los tipos de interés dirigida por \authorfont{JOHN R. MEYER}. Pasó por Pennsylvania, Minnesota, Chicago y Stanford, y es hoy profesor en la Universidad de Nueva York. Con \authorfont{WALLACE} formuló en 1975 la proposición de ineficacia, y dedicó su carrera a construir las herramientas econométricas que permiten trabajar con expectativas racionales en modelos dinámicos, recogidas en sus manuales de teoría macroeconómica dinámica. En 2011 compartió el Nobel con \authorfont{SIMS} \textit{por su investigación empírica sobre causa y efecto en la macroeconomía}.

\paragraph{\authorfont{NEIL WALLACE}}
Coautor con \authorfont{SARGENT} de los dos resultados que enmarcan la política de la escuela: la ineficacia de la política sistemática (1975) y la aritmética monetarista desagradable (1981). Su obra, más técnica que pública, sentó buena parte de las bases formales del programa y de la práctica posterior de las reglas monetarias, de la que la regla de \authorfont{TAYLOR} es heredera.

\subsubsection{Minnesota y Carnegie Mellon}

\paragraph{\authorfont{JOHN MUTH}}
Formuló la hipótesis de las expectativas racionales en 1961, en Carnegie Mellon y a propósito de los mercados agrícolas. No participó en la revolución macroeconómica que su hipótesis hizo posible, y durante una década larga su artículo apenas fue citado: la pieza que organizó la escuela la escribió alguien que nunca perteneció a ella.

\paragraph{\authorfont{EDWARD PRESCOTT}}
Doctorado en Carnegie Mellon y profesor en Minnesota, fue la figura central de la segunda ola. Con \authorfont{KYDLAND} firmó los dos trabajos que la definen: el de 1977 sobre la inconsistencia temporal y el de 1982 que fundó la teoría del ciclo real. Su programa desplazó el centro de la escuela del dinero y las expectativas hacia el lado real, la tecnología y las instituciones.

\paragraph{\authorfont{FINN KYDLAND}}
Nacido en 1943 en una zona rural de Noruega, en una familia de agricultores, se graduó en la Norwegian School of Economics en 1968 y se doctoró en Carnegie Mellon en 1973 con una tesis sobre planificación macroeconómica descentralizada. Tras un paso breve por Noruega volvió a Carnegie Mellon. En 2004 compartió con \authorfont{PRESCOTT} el Nobel \textit{por sus contribuciones a la macroeconomía dinámica: la consistencia en el tiempo de la política económica y las fuerzas impulsoras detrás del ciclo económico}.

\paragraph{\authorfont{CHRISTOPHER SIMS}}
Desarrolló los modelos de vectores autorregresivos, que permiten identificar los efectos de las perturbaciones con poca teoría impuesta de antemano. Su trabajo dio a la macroeconomía cuantitativa un método para separar causa y efecto, y le sirvió además para discutir desde dentro el exceso de teoría del programa de \authorfont{LUCAS}. Compartió el Nobel de 2011 con \authorfont{SARGENT}.

A la vista del mapa, \authorfont{ROBERT HALL} acuñó el término \textit{freshwater economics}, economía de agua dulce, por la situación geográfica de Chicago, Minnesota, Carnegie Mellon y Rochester frente a los departamentos de las dos costas, el agua salada. La etiqueta se solapa solo parcialmente con la nueva macroeconomía clásica, pero describe bien la geografía de la disputa.\footnote{%
    Cualquier clasificación de este tipo es una aproximación: sirve para ver las influencias y las relaciones entre los autores más señalados, no para encerrar una trayectoria. Ninguno de ellos dedicó su carrera entera a los desarrollos por los que aquí se los recuerda.
}

\subsection{El método}\label{anx3:metodo}

Si se le pregunta a la escuela qué fue lo que hizo, su respuesta no habla de resultados, sino de método. Así lo formuló \authorfont{PRESCOTT} al recibir el Nobel:

\begin{cita}[\authorfont{PRESCOTT}][2004][Discurso del Premio Nobel]
\textit{What I am going to describe to you is a revolution in macroeconomics, a transformation in methodology that has reshaped how we conduct our science.}
\end{cita}

Esa transformación tiene tres caras. La primera es la \textbf{culminación de un programa ajeno}. La nueva macroeconomía clásica no derrota a la Síntesis desde fuera, sino desde dentro: completando el programa walrasiano, el de un equilibrio general con agentes que optimizan, que la Síntesis decía profesar y nunca llevó hasta el final. \authorfont{DE VROEY} (2016) ha descrito el cambio como el paso de una macroeconomía marshalliana, hecha de equilibrios parciales, medición y predicción, en la que todavía se mueve \authorfont{FRIEDMAN}, a una macroeconomía walrasiana, formal y general, que es la de \authorfont{LUCAS}. El monetarismo y la Síntesis compartían instrumental; lo que \authorfont{LUCAS} cambió fue el criterio de lo que cuenta como explicación.

La segunda es la \textbf{disciplina del equilibrio}. Un modelo deja de ser un sistema de ecuaciones ajustado a los datos y pasa a ser, en palabras de \authorfont{LUCAS} (1980), un sistema explícito de instrucciones capaz de imitar el comportamiento de una economía real: algo que se construye, se calibra y se somete a prueba, no algo que se estima. De ahí que la escuela exija a cualquier modelo tres credenciales: microfundamentos explícitos, expectativas racionales y capacidad de ser cuantificado.

La tercera es la \textbf{continuidad con la tradición clásica}. La escuela preserva intacta la figura del agente optimizador que recorre el pensamiento dominante desde los \authorfont{MILL}, y en ese sentido puede leerse como el cénit de la visión clásica de la macroeconomía, más que como una ruptura: lo que rompe es con el intervalo keynesiano, no con la línea larga.

\subsection{La crítica en palabras de \authorfont{LUCAS}}\label{anx3:extractos}

El artículo de 1976 que da el pistoletazo de salida a la escuela merece leerse en sus propios términos. Empieza recordando una observación empírica tan vieja como el ciclo:

\begin{cita}[\authorfont{LUCAS}][1976][Econometric Policy Evaluation: A Critique]
\textit{El hecho de que los precios y salarios nominales tiendan a aumentar con mayor rapidez en el punto máximo del ciclo económico que en el mínimo ha sido bien reconocido desde que el ciclo fue percibido por primera vez como un fenómeno distinto. La inferencia de que una inflación permanente inducirá, por lo tanto, una expansión económica permanente es sin duda igual de antigua; sin embargo, solo recientemente esta noción ha sufrido la misteriosa transformación de una falacia obvia a convertirse en una piedra angular de la teoría de la política económica.}

\textit{Esta transformación no surgió de nuevos desarrollos en la teoría económica. (...) Surgió, en cambio, de la tradición más reciente de los modelos econométricos de predicción y del compromiso, por parte de una gran fracción de economistas, con el uso de estos modelos para la evaluación cuantitativa de políticas. (...) Este conflicto tan claro entre dos tradiciones justamente respetadas, la teórica y la econométrica, tomó por sorpresa a quienes veíamos ambas como complementarias de manera armoniosa.}

\textit{(...) La tesis de este ensayo es que es la tradición econométrica, o más precisamente, la teoría de la política económica basada en esta tradición, la que necesita una revisión profunda. Más concretamente, sostendré que las características que conducen al éxito en la predicción a corto plazo no están relacionadas con la evaluación cuantitativa de políticas, que los principales modelos econométricos están diseñados (y bien diseñados) únicamente para realizar la primera tarea, y que las simulaciones utilizando estos modelos no pueden, en principio, proporcionar ninguna información útil sobre las consecuencias reales de políticas económicas alternativas.}

\textit{Hay poco en este ensayo que no esté implícito (y quizá, para lectores más perspicaces, explícito) en \authorfont{FRIEDMAN}, \authorfont{MUTH} y, aún antes, en \authorfont{KNIGHT}.}
\end{cita}

El razonamiento es un silogismo de tres líneas. Si la estructura y los parámetros de un modelo econométrico son el resultado de reglas de decisión óptimas de los agentes, y esas reglas cambian cuando cambian las reglas de política relevantes, entonces los cambios de política alteran la estructura y los parámetros del modelo. Nótese que el conflicto que \authorfont{LUCAS} describe no es entre escuelas, sino entre \textbf{dos tradiciones que él mismo consideraba complementarias}: la teoría económica, que exige comportamiento optimizador, y la práctica econométrica, que estimaba relaciones estables y las usaba para comparar políticas. Lo que el ensayo sostiene es que la segunda, tal como se practicaba, era incompatible con la primera.

De ahí la consecuencia práctica, que es la que recogieron los bancos centrales:

\begin{cita}[\authorfont{LUCAS}][1976][Econometric Policy Evaluation: A Critique]
\textit{En cuanto a la cuestión de la capacidad de los modelos econométricos para realizar predicciones o seguimientos a corto plazo, esto tiene una importancia solamente ocasional. En contraste, para asuntos relacionados con la evaluación de políticas es fundamental, porque implica que las comparaciones entre los efectos de reglas de política alternativas utilizando los modelos macroeconométricos actuales son inválidas, independientemente del rendimiento de estos modelos durante el período muestral o en las predicciones ex ante a corto plazo.}

\textit{Parece que los responsables de la política económica, si desean prever la respuesta de los ciudadanos, deben tomar a estos últimos en su confianza. Esta conclusión, aunque poco adecuada para la práctica econométrica actual, parece concordar bien con una preferencia por la toma de decisiones democrática.}
\end{cita}

La última frase merece subrayarse, porque suele pasar inadvertida: de la crítica se sigue que la política debe ser \textbf{transparente y anunciada}, ya que un gobierno que pretenda prever la reacción de los ciudadanos no tiene más remedio que contarles qué va a hacer. El resultado más técnico de la escuela desemboca en una recomendación institucional de signo democrático.

\subsection{Las batallas}\label{anx3:batallas}

La irrupción fue radical y muchas veces vehemente. En una intervención de 1979 titulada \textit{The death of Keynesian economics}\textcolor{orange}{\textsuperscript{[Probable]}}, \authorfont{LUCAS} llegó a sostener que la economía keynesiana estaba muerta, que no era posible encontrar un buen economista menor de cuarenta años que se identificara como keynesiano, que las revistas punteras ya no aceptaban artículos keynesianos y que la teoría keynesiana no se tomaba en serio en los seminarios de investigación. La reacción no se hizo esperar, y llegó por seis frentes distintos.

\subsubsection{La reacción keynesiana}

\authorfont{TOBIN} y \authorfont{MODIGLIANI} rechazaron el marco de plano, con el argumento de que sustituía la verdad desordenada contenida en \authorfont{KEYNES} por el error preciso de los modelos à la \authorfont{LUCAS}\textcolor{orange}{\textsuperscript{[Probable]}}. Frente a esa posición de rechazo, los economistas neokeynesianos de la primera oleada optaron por aceptar las armas del adversario, microfundamentos y expectativas racionales, para rehabilitar las conclusiones de \authorfont{KEYNES}: el desempleo involuntario, la no neutralidad del dinero y la lentitud del ajuste.\footnote{%
    No hubo un programa único, sino un conjunto de aportaciones con dos proposiciones comunes, tal como las recogen \authorfont{MANKIW} y \authorfont{ROMER} (1991): que el dinero no es neutral, porque las rigideces hacen que las perturbaciones nominales tengan efectos reales, y que las imperfecciones de mercado y la rigidez de precios y salarios son centrales para explicar el ciclo. Se trataba de modelos de equilibrio parcial, generalmente estáticos, con competencia imperfecta: contratos implícitos (\authorfont{BAILY}, 1974; \authorfont{GORDON}, 1974; \authorfont{AZARIADIS}, 1975), salarios escalonados (\authorfont{FISCHER}, 1977; \authorfont{TAYLOR}, 1979), salarios de eficiencia (\authorfont{SALOP}, 1979; \authorfont{SHAPIRO} y \authorfont{STIGLITZ}, 1984) y costes de menú (\authorfont{MANKIW}, 1985; \authorfont{AKERLOF} y \authorfont{YELLEN}, 1985). La segunda oleada, ya con equilibrio general dinámico, se estudia en [\authorfont{Ver Tema 3.A.7}].
}

Esta primera oleada no detuvo el avance de la escuela, cuya agenda se juzgaba por su validez empírica y ya se había desplazado hacia el ciclo real; hubo que esperar una década a que la segunda oleada neokeynesiana adoptara el instrumental completo para que el debate se reanudara en pie de igualdad.

\subsubsection{La evidencia contra la neutralidad}

El frente empírico fue el más dañino, porque atacaba la predicción central. Cuatro resultados lo resumen. La \textbf{ecuación de San Luis} (\authorfont{ANDERSEN} y \authorfont{JORDAN}, 1968) regresaba la variación del PIB real sobre los cambios del stock monetario y encontraba un impacto significativo, de alrededor de $0{,}25$ puntos por cada punto de aumento de la cantidad de dinero.\footnote{%
    El ejercicio adolece de dos problemas. El primero es que la correlación puede no ser causalidad: los agregados monetarios responden también a la demanda de dinero de empresas y hogares, que a su vez crece cuando crece la economía. El segundo, más conceptual, es que si la política monetaria fuese eficaz como estabilizador, los cambios en la oferta monetaria podrían no aparecer asociados a variaciones de la producción, precisamente porque habrían servido para evitarlas.
} El \textbf{análisis histórico} de \authorfont{FRIEDMAN} y \authorfont{SCHWARTZ} (1963) mostró que la mayoría de los movimientos monetarios desde la guerra civil tuvieron origen en el propio sector monetario y fueron seguidos de movimientos reales en la misma dirección. \authorfont{ROMER} y \authorfont{ROMER} (1989) identificaron seis episodios de posguerra en que los tipos se movieron por voluntad de reducir la inflación y no por la marcha de la economía real, y encontraron efectos reales; la selección de los episodios se ha discutido, pero el experimento se acepta en general como evidencia contraria a la neutralidad. Y los \textbf{VAR estructurales} (\authorfont{SIMS}, 1992; \authorfont{CHRISTIANO}, \authorfont{EICHENBAUM} y \authorfont{EVANS}, 1996) situaron el análisis en un marco multivariante, identificando la política con movimientos de los tipos de interés.\footnote%
{   Estos modelos no superan por sí solos la objeción sobre el origen de las perturbaciones monetarias: los movimientos de tipos pueden estar respondiendo a información que el banco central tiene sobre el futuro, como ocurrió con las bajadas de 2007. \authorfont{ROMER} y \authorfont{ROMER} (2004) combinan los dos métodos, atendiendo al origen de los cambios de política y aprovechando a la vez la potencia estadística del VAR, y estiman efectos mayores y más inmediatos sobre la producción y los precios. Cómo se aísla el efecto de la política monetaria de su reacción al ciclo se estudia en [\authorfont{Ver Tema 3.A.42}].
}

Conviene señalar que el propio marco de la escuela admitió después el dinero sin abandonar sus supuestos: \authorfont{COOLEY} y \authorfont{HANSEN} (1989) introdujeron restricciones de pago en efectivo en un modelo de ciclo real, y la predicción de que el dinero importa poco se mantuvo, tal y como recogen \authorfont{KING} y \authorfont{REBELO} (1999) en su revisión del programa.

\subsubsection{La disidencia interna}

\authorfont{SIMS} discrepó desde dentro y desde el principio: su objeción no era a las expectativas racionales, sino al exceso de teoría impuesta a los datos. Sostuvo además que la crítica de \authorfont{LUCAS} había tenido una influencia desproporcionada, porque los cambios de política que se observan en la práctica suelen ser modificaciones \textit{dentro} de una regla establecida, y no cambios de régimen: una variación de tipos o de tipos impositivos no altera las reglas del juego hasta el punto de invalidar todo análisis con modelos tradicionales.

\subsubsection{El vaciado de los mercados}

El supuesto de vaciado continuo implica que no existe desempleo involuntario: quien no trabaja ha elegido no hacerlo al salario vigente. Además de chocar con la intuición, choca con los datos, porque la mayor parte de las fluctuaciones de las horas trabajadas agregadas procede de entradas y salidas del empleo y no de cambios en las horas de quienes ya trabajan. \authorfont{ROGERSON} (1988) y \authorfont{HANSEN} (1985) ofrecieron una salida que conservaba el marco: si el trabajo es \textbf{indivisible}, de modo que se trabaja a jornada completa o no se trabaja, el ajuste ocurre en el margen extensivo, la elasticidad individual deja de ser relevante y el modelo genera a la vez una oferta agregada muy elástica y la existencia de desempleo ([\authorfont{Ver Tema 3.A.25}]).

\subsubsection{Las expectativas racionales}

Sobre la hipótesis misma se formularon tres objeciones. La primera, que los agentes pueden no comportarse racionalmente, como muestran la economía experimental y las aportaciones que introducen el altruismo o la equidad en la decisión individual.\footnote{%
    En la misma línea se sitúan los trabajos sobre las bases neurológicas del comportamiento económico, como los de \authorfont{PAUL GLIMCHER}, que asignan un papel central a las emociones. El tratamiento de la racionalidad limitada y de las desviaciones sistemáticas corresponde a [\authorfont{Ver Tema 3.A.10}].
} La segunda, de \authorfont{TOBIN}, \authorfont{SHILLER} y el propio \authorfont{FRIEDMAN}, que la hipótesis atribuye a los agentes un conocimiento desmesurado, tanto del modelo subyacente como de las reglas de política. La tercera, de \authorfont{MODIGLIANI}, que su validez empírica se limita a mercados especulativos y no se sostiene en mercados como el laboral.

Hay además una objeción de fondo, de otro orden. La hipótesis supone que el futuro es, en lo esencial, como el pasado: que existe una distribución de probabilidad estable que los agentes pueden aprender. Es el supuesto de \textbf{ergodicidad}, y \authorfont{DAVIDSON} (1991) sostuvo que elimina precisamente lo que \authorfont{KEYNES} y \authorfont{KNIGHT} consideraban decisivo, la incertidumbre que no puede reducirse a riesgo calculable, y que no hay motivo para esperar que las decisiones cruciales, las que cambian el entorno en que se toman, pertenezcan a un mundo ergódico. \authorfont{ASTARITA} (2011) añade a esa lectura una consecuencia política: si los agentes conocen el modelo verdadero y la economía vuelve siempre a su equilibrio natural, las crisis son errores de cálculo y no rasgos del sistema, de modo que la hipótesis acaba funcionando como un dispositivo que \textbf{saca a la economía de la historia} y naturaliza lo dado.\footnote{%
    Es el reverso de lo que el propio marco presume como virtud. Que la tasa natural de paro se llame natural no es inocente: convierte en dato técnico un nivel de desempleo que depende de instituciones, y el episodio argentino de 2001 ilustra hasta qué punto las expectativas de los agentes pueden estar gobernadas por rupturas institucionales que ninguna distribución estable anticipa. La crítica a la ergodicidad desde la economía poskeynesiana se estudia en [\authorfont{Ver Tema 3.A.4}].
}

Con el tiempo, la controversia sobre la hipótesis se apagó por donde \authorfont{TOBIN} había anticipado: lo verdaderamente decisivo no era ella, sino el vaciado de los mercados. Y el supuesto de información imperfecta quedó desarmado por los hechos, porque la difusión de estadísticas hace difícil sostener que los agentes no puedan estimar los precios agregados con precisión y a bajo coste.

\subsubsection{La exogeneidad de las perturbaciones}

El último frente apunta al ciclo real y a su impulso. Que las perturbaciones sean exógenas significa que el modelo no explica lo que lo mueve todo, y la identificación de esa perturbación con la productividad total de los factores tiene tres problemas. El primero es de \textbf{medida}: la productividad medida se ve afectada por el acaparamiento de trabajo, por la utilización variable del capital o por el gasto militar, de modo que no es una perturbación tecnológica pura (\authorfont{BURNSIDE}, \authorfont{EICHENBAUM} y \authorfont{REBELO}, 1993). El segundo es de \textbf{interpretación}: hay acuerdo en que el progreso técnico genera expansiones, pero explicar las recesiones como episodios de regresión tecnológica resulta difícil de aceptar.\footnote{%
    La salida habitual consiste en entender la perturbación negativa en sentido amplio, incluyendo todo lo que reduce la productividad marginal del trabajo: el precio de la energía, el clima, los cambios regulatorios.
} El tercero es \textbf{cualitativo}: \authorfont{GALÍ} (1999), identificando con un VAR las perturbaciones tecnológicas como única fuente de fluctuación a largo plazo, encontró que las horas trabajadas \textit{caen} a corto plazo tras una mejora tecnológica, justo lo contrario de lo que predice el ciclo real.\footnote{%
    Los resultados de \authorfont{GALÍ} han sido discutidos por no ser robustos a la especificación de las series (\authorfont{CHRISTIANO}, \authorfont{EICHENBAUM} y \authorfont{VIGFUSSON}, 2003) y por posibles problemas de identificación (\authorfont{CHARI}, \authorfont{KEHOE} y \authorfont{McGRATTAN}, 2008). El detalle de esta contrastación se estudia en [\authorfont{Ver Tema 3.A.42}].
}

El balance de las seis batallas es, sin embargo, menos adverso de lo que la enumeración sugiere, y explica la posición que la escuela ocupa hoy: perdió casi todas las discusiones sustantivas, sobre la neutralidad, sobre el vaciado y sobre el origen de las perturbaciones, y ganó la metodológica, porque sus adversarios acabaron discutiéndole con sus propias herramientas.
