    % --- Imagen adaptativa ---
    % \imagenfit[alto]{fichero}{título}{fuente}: la imagen queda fija entre los párrafos donde se escribe (no flota), con su
    % título en el índice de figuras y la fuente debajo. Sin [alto]: las figuras PDF (TikZ) salen al tamaño con que se
    % dibujaron y solo se reducen si no caben en el ancho o en el alto de la página; las PNG/JPG ocupan el ancho del texto
    % hasta \imgaltoras del alto de la página. Con [alto] (fracción del alto de la página; [1] = página entera) se fija el
    % tamaño de esa imagen. Si no cabe en lo que queda de página, pasa entera a la siguiente.
    % \imagenfithorizontal{fichero}{título}{fuente}: la imagen en una página girada (apaisada), en ese mismo punto del texto.
    % Reglas: main/GUIA_TEMAS.md, §4 (Figuras).
    \newdimen\imgcap
    \newdimen\imgh
    \newdimen\imgw
    \newdimen\imgneed
    \newsavebox\imgcaja
    \newcommand{\imgaltoras}{0.30}% alto por defecto de las PNG/JPG (fracción del alto de la página)
    \makeatletter
    \newcommand{\imagenfit}[4][]{%
      \addvspace{10pt}%
      \par\begingroup
        \imgcap=3.5\baselineskip% alto aproximado de título + fuente
        \imgw=\linewidth
        \filename@parse{#2}\def\img@pdf{pdf}%
        \if\relax\detokenize{#1}\relax
          \ifx\filename@ext\img@pdf
            \sbox\imgcaja{\includegraphics{\detokenize{#2}}}% tamaño natural
            \ifdim\wd\imgcaja<\imgw \imgw=\wd\imgcaja \fi
            \imgh=\ht\imgcaja
          \else
            \imgh=\imgaltoras\textheight
          \fi
        \else
          \imgh=#1\textheight
        \fi
        \dimen@=\textheight \advance\dimen@ by -\imgcap
        \ifdim\imgh>\dimen@ \imgh=\dimen@ \fi
        \sbox\imgcaja{\includegraphics[width=\imgw,height=\imgh,keepaspectratio]{\detokenize{#2}}}%
        \imgneed=\ht\imgcaja \advance\imgneed by \imgcap
        \Needspace{\imgneed}%
        \noindent\begin{minipage}{\textwidth}
          \centering
          \captionof{figure}{\textemdash\ #3}\par\vspace{0.3em}% título arriba
          \usebox\imgcaja\par
          \vspace{0.3em}\small\textit{Fuente: }#4% fuente abajo
        \end{minipage}%
      \endgroup
      \par\addvspace{10pt}\RequestRestore
    }
    \newcommand{\imagenfithorizontal}[3]{%
      \begin{landscape}
        \centering
        \captionof{figure}{\textemdash\ #2}\par\vspace{0.3em}%
        \includegraphics[width=\linewidth,height=\dimexpr\textheight-4\baselineskip\relax,keepaspectratio]{\detokenize{#1}}\par
        \vspace{0.3em}{\small\textit{Fuente: }#3}%
      \end{landscape}%
    }
    \makeatother

    % \anexo{Título}: un anexo equivale a un \section (su título sale del mismo tamaño). Dentro se usan \subsection,
    % \subsubsection y \paragraph, nunca \section: se numeran I.1, I.1.1, I.1.1.1 y las páginas del anexo, I-1, I-2…
    % En el índice de contenido inicial solo figura el nombre del anexo. Sus apartados forman el índice del anexo, que sale
    % solo si el anexo los tiene, justo después de la introducción del anexo (antes del primer \subsection).
    % \anexo*{Título}: el mismo anexo, sin ese índice. Reglas: main/GUIA_TEMAS.md, §6.
    \makeatletter
    \newcounter{anexo}
    \renewcommand{\theanexo}{\Roman{anexo}}
    \newif\ifEnAnexo \EnAnexofalse
    \newif\ifanx@pendiente \anx@pendientefalse
    \newcommand{\anexo}{\@ifstar{\anx@empezar\anx@pendientefalse}{\anx@empezar\anx@pendientetrue}}
    \newcommand{\anx@empezar}[2]{%
      \clearpage
      \refstepcounter{anexo}%
      \global\EnAnexotrue
      \setcounter{page}{1}\gdef\thepage{\theanexo-\arabic{page}}%
      \setcounter{section}{\numexpr\value{anexo}-1\relax}\stepcounter{section}% el anexo hace de \section
      \gdef\thesection{\Roman{section}}\gdef\theHsection{anexo.\Roman{section}}%
      \phantomsection
      \protected@xdef\leq@current@section{Anexo \theanexo.- \unexpanded{#2}}\gdef\leq@current@subsection{}% índice de ecuaciones
      \addtocontents{stc}{\protect\contentsline{section}{Anexo \theanexo.- #2}{}{\@currentHref}}% índice inicial: solo el nombre, sin página
      \addcontentsline{toc}{section}{Anexo \theanexo.- #2}% marcadores del PDF
      {\noindent\normalfont\Large\bfseries Anexo \theanexo.- #2\par}\nobreak\vspace{1.5ex \@plus .2ex}%
      #1\@ifundefined{anx@lista@\theanexo}{\global\anx@pendientefalse}{}%
    }
    % Índice del anexo: cada apartado escribe una línea en el fichero .atc (anexo, nivel, texto, página, enlace); al
    % empezar el documento se leen las de la compilación anterior y se guardan por anexo (dos pasadas, como cualquier índice).
    % En ese índice los apartados suben un nivel, porque el anexo hace de \section.
    \def\anx@entrada#1#2#3#4#5{\@ifundefined{anx@lista@#1}{\global\@namedef{anx@lista@#1}{}}{}%
      \expandafter\g@addto@macro\csname anx@lista@#1\endcsname{\contentsline{#2}{#3}{#4}{#5}}}
    \tcbuselibrary{breakable}
    \def\anx@indice{\ifanx@pendiente\global\anx@pendientefalse
        \begin{tcolorbox}[breakable, colback=white, colframe=black!35, title={Contenido del anexo}, fonttitle=\small\bfseries,
            before upper={\parindent=0pt\parskip=2pt}]
          \small
          % números más anchos que en el índice inicial (I.1, I.1.1, I.1.1.1; páginas I-1)
          \def\@pnumwidth{2.6em}\def\@tocrmarg{3.4em}%
          \def\l@section##1##2{\addpenalty\@secpenalty\addvspace{1.0em \@plus\p@}\setlength\@tempdima{2.6em}%
            \begingroup\parindent\z@\rightskip\@pnumwidth\parfillskip-\@pnumwidth\leavevmode\bfseries
            \advance\leftskip\@tempdima\hskip-\leftskip##1\nobreak\hfil\nobreak\hb@xt@\@pnumwidth{\hss##2\kern-\p@\kern\p@}\par\endgroup}%
          \def\l@subsection{\@dottedtocline{2}{2.6em}{3.2em}}\def\l@subsubsection{\@dottedtocline{3}{5.8em}{4.1em}}%
          \csname anx@lista@\theanexo\endcsname
        \end{tcolorbox}\par\medskip
      \fi}
    \def\anx@escribir#1#2#3{\ifEnAnexo
        \addtocontents{atc}{\protect\anx@entrada{\theanexo}{#2}{\protect\numberline{\csname the#1\endcsname}#3}{\thepage}{\@currentHref}}%
      \else\addcontentsline{stc}{#1}{\protect\numberline{\csname the#1\endcsname}#3}\fi}
    % Encabezados: dentro de un anexo escriben en su índice (y el primero saca antes ese índice); fuera, en el índice de contenido.
    % Se activan al empezar el documento, cuando ya están definidos los del índice propio (\MyTOC, más abajo).
    \def\anx@subsection[#1]#2{\ifEnAnexo\anx@indice\fi\MyTOC@orig@subsection[{#1}]{#2}\anx@escribir{subsection}{section}{#2}}
    \def\anx@subsubsection[#1]#2{\ifEnAnexo\anx@indice\fi\MyTOC@orig@subsubsection[{#1}]{#2}\anx@escribir{subsubsection}{subsection}{#2}}
    \def\anx@paragraph[#1]#2{\ifEnAnexo\anx@indice\fi\MyTOC@orig@paragraph[{#1}]{#2}\anx@escribir{paragraph}{subsubsection}{#2}}
    \AtBeginDocument{\@starttoc{atc}%
      \let\MyTOC@subsection@nostar\anx@subsection
      \let\MyTOC@subsubsection@nostar\anx@subsubsection
      \let\MyTOC@paragraph@nostar\anx@paragraph}
    \makeatother

    % En las páginas numeradas del tema (de la Introducción a la Conclusión), arriba, el título completo del tema tal como está en
    % \title, sin «Tema X.Y.N --»: título y subtítulo seguidos. En letra pequeña; más pequeña aún si no cabe en dos líneas. Va en el
    % margen superior, así que no cambia las páginas del tema. Se enciende con \pagenumbering{arabic} y se apaga con
    % \pagenumbering{gobble} (bibliografía): índices, bibliografía y anexos quedan sin cabecera. Reglas: main/GUIA_TEMAS.md, §4.

    \geometry{headheight=28pt}% hasta tres líneas de letra pequeña
    \ExplSyntaxOn
    \tl_new:N \g_cab_titulo_tl
    \cs_new_protected:Npn \cab_guardar:n #1
      {
        \tl_gset:Nn \g_cab_titulo_tl {#1}
        \regex_replace_once:nnN { \A \s* Tema \s+ \S+ \s* \-\-? \s* } { } \g_cab_titulo_tl % sin «Tema X.Y.N --»
        \regex_replace_once:nnN { \.? \s* \c{\\} (\[ [^\]]* \])? \s* (\c{large} \s*)? } { .\  } \g_cab_titulo_tl % título. subtítulo
        \regex_replace_once:nnN { [\.\s]* \Z } { } \g_cab_titulo_tl % sin subtítulo: nada detrás
      }
    \cs_new_eq:NN \cabGuardarTitulo \cab_guardar:n
    \cs_new:Npn \cabTitulo { \tl_use:N \g_cab_titulo_tl }
    \ExplSyntaxOff

\section{Función de costes: Análisis de dualidad}
\puntosclave{
	El problema dual consiste en minimizar el coste condicionado a un nivel de producción.
    
	Del problema dual se obtiene la demanda condicionada de factores que refleja la variación en la cantidad demandada de los factores ante alteraciones en sus precios, manteniendo el nivel de producción constante.
    
	Evaluando la demanda condicionada de factores en la función objetivo del problema dual se obtiene la función de costes. La función de coste cumple una serie de propiedades: continua, homogenea de grado uno y cóncava en los precios de los inputs, no decreciente respecto a la producción y cumple el Lema de Hotelling
    
	La propiedad de integrabilidad permite recuperar la tecnología de producción de la empresa a partir de la función de costes
    }


\textcolor{magenta}{Lo primero que tenemos que explicar es: ¿cuál es el objeto de este tema? Más en particular, lo que tenemos que dejar claro es la complementariedad (supongo) que existe entre el 3.A.11 y el presente. ¿Qué explicamos en 3.A.11 y por qué lo que explicamos no resulta suficiente para comprender el funcionamiento de la producción y de los conceptos que de ésta se desprenden (muy en particular, la eficiencia)? ¿Qué es lo que justifica el análisis de los costes como la cara opuesta del mismo análisis de la "Producción" y por qué es necesaria?

Además, una forma bonita de acabar el "Marco conceptual de los costes", por decirlo de alguna manera, es explicando porque el concepto principal que se presenta es el "Beneficio", que emerge de la producción teniendo en cuenta los costes, y que actúa como driving factor en la toma de decisiones de los agentes en el ámbito de la producción. }

La eficiencia técnica no es suficiente para la empresa. Sobre la base de criterios técnicos, podemos descartar procesos o combinaciones productivas técnicamente ineficientes, pero hay muchas formas técnicamente eficientes de utilizar los factores productivos, recogidas en la función de producción. El análisis de los costes introduce la eficiencia económica para seleccionar la forma más barata de acceder a un determinado nivel de producción.

A partir de la teoría de la producción, es decir, del análisis técnico del comportamiento del producto, podemos obtener las isocuantas como fronteras del conjunto de combinaciones de factores que son técnicamente factibles [ver tema 3.A.11]. 

Para ello necesitaremos información económica: los precios de los factores productivos. Dentro de un conjunto de combinaciones técnicamente eficientes (puntos de la isocuanta) resultarán más atractivos para el productor aquéllas combinaciones más intensivas en el factor comparativamente más barato, por lo que tendrá implicaciones esenciales respecto a la demanda de factores productivos por parte de la empresa: resolverá el problema de elegir una combinación de factores que permita alcanzar un nivel de producción determinado tal que su coste sea el más bajo posible.

De este modo, siguiendo a JULIO SEGURA (1996), el problema de optimización del productor se puede plantear de dos maneras:
\begin{la}
    \item Maximización de la producción sujeto a un nivel de costes máximo: $\max_{\{Q(\bar C)\}}$; y,
    \item Minimización del coste sujeto a un nivel de producción mínima: $\min_{\{C(\bar{Q})\}}$.
\end{la}

Por todo ello, los enfoques de maximización y minimización (problemas Duales) parten del mismo conjunto de hipótesis y elementos, pues el problema subyacente es el mismo, sólo cambia la manera de abordarlo. El enfoque dual requiere exactamente la misma especificación del conjunto tecnológico, función de producción y costes que el enfoque primal.

\imagenfit{TeoriaCostes_EsquemaDualidad.png}{Especificación necesaria del problema según el enfoque en el que se plantea. Análisis de dualidad}{Apuntes Víctor Gutiérrez. Tema 3.B.12}

Partimos de un conjunto tecnológico definido en el siguiente marco axiomático o normativo:
\begin{lnum}
    \item \textbf{Axioma 1}. Se trata de un conjunto no vacío;
    \item \textbf{Axioma 2}. Cerrado;
    \item \textbf{Axaioma 3}. Con eliminación gratuita;
    \item \textbf{Axioma 4}. Propiedad de \textit{no free-lunch}; y,
    \item \textbf{Axioma 5}. Posibilidad de inacción
\end{lnum}

Por ahora no definiremos ningún caso concreto de rendimientos a escala, por generalizar el resultado lo máximo posible. En caso de deber ser impuesto, será mencionado exprésamente.

Esto dará lugar a una función de producción, cardinal, de buen comportamiento (lo cual es útil para aplicar calculo diferencial, permitiéndonos plantear el problema del consumidor como un problema de optimización), que tendrá las siguientes características:\footnote{%
    Partimos de la base de que solo incorpora las combinaciones técnicamente eficientes pues consideraremos ahora la Función de Producción con la frontera del Conjunto de Posibilidades de Producción. La eficiencia técnica implica que no existe otro proceso de producción capaz de producir igual cantidad de output utilizando menor cantidad de inputs o mayor cantidad de output utilizando igual cantidad de inputs.
}
\begin{lnum}
    \item No-decreciente en factores productivos: eliminación gratuita;
    \item También será conveniente suponer que existe cierta complementariedad entre los factores productivos, dando lugar a unas isocuantas convexas (estríctamente o no).\footnote{%
        De forma intuitiva, si suponemos dos factores productivos, trabajo y capital, son complementarios en cierto grado si solo podemos producir poco output si uno de los factores es escaso aunque el otro sea abundante. En este sentido, ambos factores productivos son importantes para la producción y la media de dos vectores de producción extremos (uno con capital elevado y trabajo reducido y otro al revés) producirá estrictamente más producto que al menos uno de los dos vectores iniciales extremos (JEHLE y RENY, 2011). ADAM SMITH ya hablaba de complementariedad: los trabajadores son más productivos cuanto más capital haya. Por ejemplo, en una función de producción de tipo Cobb-Douglas existe la idea de complementariedad (la productividad marginal del capital aumenta cuanto aumenta el capital).
    }
\end{lnum}

Dicho esto, hay que tener en cuenta que dicha función será:
\begin{lnum}
    \item Objeto de maximización en el Problema de Maximización de la Producción; y,
    \item Considerada como la restricción en su “Dual”, el Problema de Minimización del Coste.
\end{lnum}

Por otro lado, la empresa hace frente a unos costes, derivados de que los precios de los factores productivos son positivos:\footnote{%
    Vamos a partir siempre de la consideración del coste como coste económico, que tiene en cuenta el coste de oportunidad, en contraposición al coste contable, que no lo tiene en cuenta. Debemos definir qué es el coste de un factor para la empresa. Trabajaremos con el concepto de coste marginal de oportunidad de un factor, que es el valor de la alternativa a la que renuncia la empresa por el uso de una unidad adicional de ese factor. Si la empresa no dispone en este momento de la unidad adicional, tendrá que comprarla (alquilarla) y el coste marginal de oportunidad será el precio de mercado o de alquiler de dicho factor. Si la empresa ya dispone en este momento de la unidad adicional, no tendrá que realizar un desembolso, pero, dado que dicha unidad podía haberse vendido en el mercado, este precio de mercado será el valor de la alternativa que se ha perdido. En resumen, el coste marginal de oportunidad es el precio de mercado.
}
\begin{lnum}
    \item Que serán la restricción del $\max_{\{Q(\bar C)\}}$; y,
    \item Y objeto de minimización en el $\min_{\{C(\bar Q)\}}$
\end{lnum}

NOTA.- Existen varias “acepciones” en la aproximación a la Relación Dual. Por ello, introduciremos primero los tres problemas y luego estableceremos las diferencias entre ellos y las relaciones de dualidad que presentan. Con ello, se podrá escoger la que más convenga para la exposición.

\subsection{Problemas de Maximización}
Cuando estamos maximizando la función de producción, es decir, llevando a cabo el PMaxP, lo que en realidad queremos contestar, dada la naturaleza de la empresa es: ¿cuál será el nivel de producción que maximice nuestros beneficios?

Esta pregunta nos supone un enorme problema, puesto que, como sabemos, para establecer las relaciones de dualidad, las aproximaciones de minimización y maximización deben desarrollarse en el mismo marco normativo y debe asegurarse un trasvase correcto de información entre uno y otro. Ahora bien, el beneficio es un elemento que no está en el $\min_{\{C(\bar Q)\}}$ y por ello deberemos tratar el $\max_{\{Q(\bar C)\}}$ con especial cuidado. 
Hay dos posibles vías de resolución para el Problema de Maximización del Beneficio:\footnote{%
    Nótese que hemos cambiado el nombre para aclarar nuevamente los problemas que supone este elemento en las relaciones de dualidad.
}
\begin{la}
    \item La Maximización del Beneficio en una etapa; y,
    \item La Maximización del Beneficio en dos etapas. 
\end{la}

Ahora bien, estas dos distinciones son relevantes si tenemos en cuenta el marco operativo de la empresa. Es decir, la maximización del beneficio no será igual en un mercado que opera en competencia-perfecta, en un mercado donde opera como oligopolio o en un mercado que opera como proveedor monopolístico, entre otras. De hecho, como más adelante veremos, se puede demostrar que en un mercado que opera en competencia-perfecta (empresa precio-aceptante) las relaciones de dualidad se cumplen en cualquiera de los casos. 

Por otro lado, introduciremos el propuesto por JULIO SEGURA que si es, verdaderamente un problema que presenta relaciones de dualidad directas con el $\min_{\{C(\bar Q)\}}$.

\imagenfit{Dif_ProbMAX_PRODUCCION.png}{Enfoques alternativos del problema de maximización del productor}{Elaboración propia}

\subsubsection{$\max_{\{B^o\}}$: Una etapa}
Vamos a considerar que el objetivo final de la empresa es la maximización de sus beneficios, esto es:

\eqblock{
\begin{aligned}
    &\max_{B^o}\;\: \Pi=\underbrace{I(Q)}_{\text{\scriptsize Ingresos}}-\underbrace{CT(Q)}_{\text{\scriptsize Costes Totales}}\\[1em]
    &\underset{\text{\scriptsize Supuestos}}{\Longrightarrow}\quad \boxed{\max_{0\leq \vec z}\;\;\Pi(\vec z)=p\cdot f(\vec z)-\vec w\cdot \vec z}
\end{aligned}
}{Problema de maximización del productor: Una etapa (maximización de beneficios). Función de producción: Análisis de dualidad}
 
Introducimos, no sólo el concepto de costes, sino también el de ingresos: la estructura de ingresos de la empresa corresponde a la de un mercado perfectamente competitivo de bienes, esto es, consideraremos que vende su output (q) a un precio paramétrico (fijo).\footnote{%
    Debemos definir que es el coste de un factor para la empresa. Trabajaremos con coste marginal de oportunidad de un factor, es decir, el valor de la alternativa a la que renuncia la empresa por el uso de una unidad adicional de ese factor:
\begin{la}
    \item Si la empresa no dispone en este momento de la unidad adicional tendrá que comprarla o alquilarla y el coste marginal de oportunidad será el precio de mercado; y,
    \item Si la empresa ya dispone en este momento de la unidad adicional, no tendrá que realizar un desembolso, pero, dado que dcha unidad podría haberse vendido en el mercado, el valor de la alternativa que se ha perdido será el precio de mercado.
\end{la}
} Partiremos de los siguientes supuestos:
\begin{lnum}
    \item Contexto estático con información perfecta y empresa precio-aceptante en ambos mercados (competencia-perfecta);
    \item Tecnología con sus propiedades [\authorfont{Ver Tema 3.A.11}] representable mediante una función de producción de buen comportamiento monoproducto:
    $$q=f(\vec z)$$
    \item De momento supondremos que la empresa opera en el largo plazo, de modo que puede combinar libremente los factores, que serán variables; y,
    \item Producto homogéneo.
\end{lnum}

El empresario ha de elegir el vector de inputs que resuelve directamente el siguiente problema:\footnote{En resumen, el coste marginal de oportunidad es igual al precio de mercado.
}

$$\max_{0\leq \vec z}\;\;\Pi(\vec z)=p\cdot f(\vec z)-\vec w\cdot \vec z$$
 
\paragraph{Condiciones necesarias (CPO’s)}
Dado que se trata de un problema de optimización condicionada con restricciones de desigualdad utilizaremos el Método de los multiplicadores de Lagrange y veremos si el resultado cumple con las condiciones necesarias (Khun-Tucker) y las condiciones suficientes (Hessiana). Se definen como condiciones necesarias (CPO’s):

\eqblock{
\begin{aligned}
    &\text{CPO's}:\\[1em]
    &\left\{\begin{aligned}
        &\text{Estacionariedad}&&\Longrightarrow\;&&\boxed{\frac{\partial \Pi(\vec z^*)}{\partial z_k}=p\cdot f_k(\vec z^*)-w_k\leq 0}\\[0.5em]
        &\text{No-negatividad}&&\Longrightarrow\;&&\boxed{0\leq z_k^*}\\[0.5em]
        &\text{Holgura complementaria}&&\Longrightarrow\;&&\boxed{z_k^*\cdot [p\cdot f_k(\vec z^*)-w_k]=0}
    \end{aligned}\right\}\Longrightarrow\;\forall k=1,2,\dots, n
\end{aligned}
}{Problema del productor (maximización del beneficio): Condiciones de primer orden. Función de producción, análisis de dualidad}

Las condiciones de primer orden implican:
\begin{lnum}
    \item \textbf{Condición de estacionariedad}. La derivada de la función respecto a cada uno de los outputs será igual al beneficio de este por el valor de producido menos el precio de este en el mercado, siempre menor o igual a cero (en el óptimo, se pretende que esto sea igual a cero para todos);
    \item \textbf{Conición de no-negatividad}. Que los outputs sean positivos o iguales a cero (que haya habido producción o no, pero irreversibilidad es imposible y no existe la producción negativa); y,
    \item \textbf{Condición de holgura complementaria}. Que cada output por la diferencia entre el valor adquirido en ingresos (beneficio por output) y el precio pagado por el sea igual a cero.
\end{lnum}

Por tanto, en el óptimo, la empresa contratará un factor productivo hasta que el valor de su PMg sea igual al ingreso que obtiene gracias a su uso. Además, trabajando con las CPO’s obtendremos que la Relación Marginal de Sustitución Técnica (RMST) se iguala al cociente de precios de estos dos factores productivos en el mercado. Esto implica que, estando en el óptimo, si la empresa desea aumentar marginalmente su producción (intensiva), le será indiferente qué factor productivo contratar, puesto que la PMg de todos ponderadas por el coste es la misma en todos los casos. 

\paragraph{Condiciones suficientes (CSO’s)}
Sin embargo, las condiciones necesarias no son suficientes para garantizar que el resultado sea solución al problema. Para que el resultado del problema de optimización sea un resultado \textbf{factible}, este debe cumplir las condiciones suficientes, que garantizan que nuestro vector es la solución a este problema:

\eqblock{
\begin{aligned}
    H=\left(\frac{\partial^2\Pi(\vec z^*)}{\partial z_k\partial z_l}\right)_{\forall k,l}=\left(p\cdot \frac{\partial^2 I(\vec z^*)}{\partial z_k\partial z_l}\right)_{\forall k,l}=(p\cdot f_{k,l})_{\forall k,l}=p^n\cdot 
    \underbrace{\left(\begin{matrix}
        f_{11}&\cdots &f_{1,n}\\
        \vdots&\ddots&\vdots\\
        f_{n1}&\cdots&f_{nn}
    \end{matrix}\right)}_{F\equiv \text{Función de Producción}}
\end{aligned}
}{Problema del productor (maximización del beneficio): Condiciones Sufucientes. Función de producción, análisis de dualidad}

Es decir, que la matriz Hessiana sea semidifinida negativa. Como los precios son estríctamente positivos, es necesario que $F$ sea semidefinida negativa. En otras palabras, es necesario que la función de producción sea estríctamente cóncava, condición que hemos impuesto de partida.\footnote{%
    Como sobre este problema de optimización no se han impuesto restricciones, si la función de producción fuese convexa, el valor máximo de la función de beneficios se obtendrá cuando tienda la función de producción a infinito, en cuyo caso obtendrá también un beneficio infinito.
}  

Veamos una explicación más detallada sobre esto: las condiciones suficientes requieren estricta concavidad (rendimientos decrecientes a escala) de la función de producción, por lo menos:

\eqblock{
\begin{aligned}
    &F\to\text{CSO}(\checkmark)  \iff&&\underbrace{\text{o,}\quad \frac{\partial CMg}{\partial q}>0;\quad \text{o,}
    \begin{cases}
        \frac{\partial {PMg}_L}{\partial L}<0\\[0.5em]
        \frac{\partial {PMg}_K}{\partial K}<0
    \end{cases}
    }_{\text{\scriptsize Alternativo. Garantiza concavidad función de producción $\sim$ RDecrecientes a Escala}}
\end{aligned}
}{Problema del productor (maximización del beneficio): Propiedades de la Función de Producción y relación con la CSO's. Función de producción, análisis de dualidad}

Para garantizar estas condiciones, los requisitos son más estrictos que para el problema $\min_{\{C(\bar Q)\}}$. En efecto, en diversas situaciones una función de costes no basta para asegurar una solución finita a la maximización de beneficios:
\begin{lnum}
    \item Si el CMe es siempre inferior al precio, los beneficios crecen continuamente con el output;
    \item Si los CMe son decrecientes, y por tanto el CMg está por debajo del medio, igualar precio a coste marginal implica necesariamente pérdidas. Pero esta presencia de economías de escala es una poderosa fuerza hacia posiciones de poder de mercado (en el límite tendríamos el monopolio natural) y por lo tanto dificilmente encajaría en un contexto competitivo;
    \item Si el CMg es decreciente, igualar precio a coste marginal implicaría minimizar beneficios, para una empresa precio-aceptante;
    \item Los rendimientos constantes a escala plantean tres posibilidades:
    \begin{la}
        \item El precio es mayor al CMe (igual que el caso 1), con beneficios positivos, que crecen indefinidamente con el nivel de producción. A nivel de empresa plantean la pregunta de por qué no entrarán más empresas, pero si todas tienen acceso a la misma tecnología, la entrada sólo trasladaría el problema a nivel industria. Realmente la cuestión radica en si los rendimientos pueden mantenerse constantes ilimitadamente, pese a la posibilidad de “replicar” las empresas los procesos seguidos;
        \item Si el precio es inferior al CMe, los beneficios se maximizan con un nivel de output nulo; o,
        \item Si el precio es igual al CMe, con beneficios nulos para cualquier nivel de output: a nivel de industria la demanda de mercado puede resolver la indeterminación, pero quedaría pendiente como la producción se distribuye entre un número de empresas a priori también indeterminado; el tamaño del mercado delimita el resultado de multiplicar el output de cada empresa por el número de empresas, pero no puede decidir cada uno de los factores de esta multiplicación.
    \end{la}
\end{lnum}

En todo caso, en la práctica, un criterio adicional a considerar es la no-negatividad de los  beneficios, ni siquiera a C/p para la cobertura de los costes variables. Por eso hay que añadir a las condiciones anteriores la restricción adicional de que los beneficios (económicos) sean mayores o iguales a cero.  

\paragraph{Sistema Completo de Ecuaciones de Demanda de Inputs No Condicionadas (SCEDINC)}
A partir de este problema de optimización, siempre y cuando tenga solución, podemos obtener el SCDEINC es el conjunto de funciones que resultan del problema de maximización de beneficios de una etapa, donde la empresa elige simultáneamente qué producir y cuánto usar de cada factor. El nivel de output se determina por tanto exógenamente al maximizar los beneficios. Por tanto, es el conjunto de demandas de factores derivadas del problema de maximización y tiene la siguiente forma:

\eqblock{
\begin{aligned}
    \vec Z^*(\vec w, p)=
    \left(\begin{matrix}
        z_1(w_1, w_2, \dots, w_n, p)\\
        \vdots\\
        z_{n}(w_1,w_2, \dots, w_n, p)
    \end{matrix}\right)
\end{aligned}
}{Sistema Completo de Ecuaciones de Demanda de Inputs No Condicionadas (SCEDINC). Función de Producción: Análisis de dualidad}
 
Cada una expresa la cantidad óptima del insumo $i$ como función del precio de los factores y del precio del producto. Posee las siguientes propiedades:
\begin{lnum}
    \item Las funciones de demanda son continuas y homogeneas de grado cero en precios y costes (p, w);\footnote{%
        Si los precios se multiplican por cualquier escalar, la demanda óptima de insumos no varía. Solo las relaciones relativas de precios importan, no sus niveles absolutos.
    } y,
    \item El SCEDINC es, con cierta generalidad, diferenciable tanto en coste de insumos como en precios de producto. 
\end{lnum}

    La pendiente de la recta de isobeneficios es $w/p$ porque así se deriva de la ecuación que hemos maximizado. \paragraph{La Función de Oferta (no condicionada) de output}
La función de oferta no condicionada de output es la ecuación que pone la cantidad de output ofertado en función del coste de los insumos y el precio del producto, es decir, con ella podremos obtener fácilmente el output en el óptimo.  

\eqblock{
\begin{aligned}
    &\text{Función de oferta no condicionada de output:}&&\boxed{q^*(\vec w, p)=f(\vec z^*)}
\end{aligned}
}{Función de Oferta (no condicionada de output). Función de Producción: Análisis de dualidad}
 
Esta es continua y homogenea de grado cero tanto en precios como en costes ($p, w$).

\paragraph{Función Valor de Beneficios}
Se trata de la función de valor máximo que obtenemos al sustituir las soluciones del problema anterior (es decir, al sustituir las demandas no condicionadas de los factores y la función de oferta no condicionada de output en la función objetivo de beneficios):

\eqblock{
\begin{aligned}
    &\text{Función Valor}&&\Pi^*=p\cdot f(\vec z)-\vec w\cdot \vec z^*=\Pi^*(\vec w, p)
\end{aligned}
}{Función Valor de beneficios. Función de Producción: Análisis de dualidad}

La función de beneficios posee las siguientes propiedades:
\begin{lnum}
    \item Es continua y homogénea de grado cero tanto en precios como en costes ($p,w$);
    \item Es estríctamente creciente en precios;
    \item Es no-creciente en costes; y,
    \item Es convexa tanto en precios como en costes ($p,w$); y,
    \item Cumple el Lema de Hotelling:
    $$\frac{\partial\Pi(p,w)}{\partial p}=q(p,w)\qquad \text{y}\qquad -\frac{\partial \Pi(p,w)}{\partial w_i}=z_i(p,w)$$
    La derivada del beneficio máximo respecto al precio del producto da la oferta óptima de consumo; la derivada del beneficio máximo respecto al coste de un factor (con signo negativo) da la demanda óptima de ese factor. Es decir, las funciones de oferta de output y demanda de inputs pueden obtenerse directamente a partir de la función de beneficio máximo.\footnote{%
        Como la función de beneficio máximo ya incorpora el comportamiento óptimo de la empresa;un pequeño cambio en precios o en coste de factores, solo afecta al beneficio a través de cómo cambio el ingreso o el costo en el punto óptimo, no por variaciones en las cantidades (que ya están optimizadas).
    }
\end{lnum}

Estas dos últimas propiedades llevan a reconocer ciertas propiedades de la matriz de sustitución de la función de beneficios, la matriz Hessiana de la función de beneficios respecto de todos los precios y costes:

\eqblock{
\begin{aligned}
    &\text{Matriz de sustitución:}&&\Xi\equiv
    \left(\begin{matrix}
        \partial^2\Pi/\partial p^2&\partial^2\Pi/\partial p\partial w_1&\cdots &\partial^2\Pi/\partial p\partial w_1\\
        \partial^2\Pi/\partial w_1\partial p&\partial^2\Pi/\partial w_1^2&\cdots &\partial^2\Pi/\partial w_1\partial w_n\\
        \vdots &\vdots &\ddots &\vdots\\
        \partial^2\Pi/w_n\partial p&\partial^2\Pi/\partial w_n\partial w_1&\cdots &\partial^2\Pi/\partial w_n^2\\
    \end{matrix}\right)\\[1em]
    &\underset{\substack{\text{\scriptsize Por el Lema}\\\text{\scriptsize de HOTELLING}}}{\Longrightarrow} && \Xi=
    \left(\begin{matrix}
        \partial q^*/\partial p&\partial q^*/\partial w_1&\cdots &\partial q^*/\partial w_n\\
        -(\partial z_1^*/\partial p)&-(\partial z_1^*/\partial w_1)&\cdots &-\partial (z_n^*/\partial w_n)\\
        \vdots &\vdots &\ddots &\vdots\\
        -(\partial z_n^*/\partial p)&-(\partial z_n^*/\partial w_1)&\cdots &-(\partial z_n^*/\partial w_n)\\
    \end{matrix}\right)
\end{aligned}
}{Matriz de sustictución de la Función de Beneficios: Matriz Hessiana de $\Pi$ respesto $(p,w_i)_{\forall i}$. Función de Producción: Análisis de dualidad}
 
Por tanto, la matriz de sustitución será:
\begin{lnum}
    \item Simétrica, de modo que el orden de derivación no importa, lo que implica que:
    \begin{la}
        \item Las demandas de inputs son simétricas en el precio de los factores:
        $$\partial^2\Pi/\partial w_k\partial w_l=\partial^2\Pi/\partial w_l\partial w_k\iff \partial z_k^*/\partial w_l=\partial z_l^*/\partial w_k$$
        \item El comportamiento de la oferta de output es simétrico (pero opuesto) al de cada una de las demandas de input:
        $$\partial^2\Pi/\partial w_k\partial p=\partial^2\Pi/\partial p\partial w_k\iff-(\partial z_k^*/\partial p)=\partial q^*/\partial w_k$$
    \end{la}
    \item Semidefinida negativa (por la convexidad de la función de beneficios), por lo que los elementos de su diagonal principal son no-negativos, esto es:
    \begin{la}
        \item La curva de oferta de output es no-decreciente en precios:
        $$\partial q^*/\partial p\ge0$$
        \item Las curvas de demanda de inputs son no-crecientes con respecto a su precio:
        $$\partial z_k^*/\partial w_k\leq0$$
    \end{la}
\end{lnum}

\subsection{Problemas de Minimización}
El Problema de Minimización de Costes fue introducido por PAUL SAMUELSON (1947) y analizado con mayor detalle por RONALD SHEPHARD (1953). 
Conociendo el precio de los factores productivos, podemos encontrar la combinación de factores productivos (eficiencia técnica) capaz de minimizar el coste de producir (eficiencia económica) una determinada cantidad de output para la empresa.

\subsubsection{Supuestos}
Nos encontramos bajo el siguiente marco normativo:
\begin{lnum}
    \item Contexto estático con información perfecta y empresa precio-aceptante en el mercado de inputs (sólo minimiza costes, la estructura de mercado del mercado del output es irrelevante, ya que el precio del bien final no aparece en el problema de optimización);
    \item Tecnología con sus propiedades [Tema 3.A.11] representable mediante una función de producción de buen comportamiento monoproducto:
    $$q=f(\vec z)$$
    \item Supondremos que la empresa opera al largo plazo, de modo que puede combinar libremente los factores; y,
    \item Producto homogéneo.
\end{lnum}

\subsubsection{Desarrollo }
Matemáticamente el problema se puede especificar de la siguiente manera:

\eqblock{
\begin{aligned}
    &\boxed{
    \begin{aligned}
        \min_{\vec z}\quad C(\vec z)=\vec w\cdot \vec z\\
        \text{s.a.}\quad 
        \begin{cases}
            \bar Q\leq f(\vec z)\\
            0\leq \vec z
        \end{cases}
    \end{aligned}
    }\\[2em]
    &\Longrightarrow&&\mathcal L=\vec w\cdot \vec z+\mu(\bar Q-f(\vec z))
\end{aligned}
}{Problema de minimización del gasto. Función de producción: Análisis de dualidad}
 
Dado que se trata de un problema de optimización condicionada con restricciones de desigualdad, nos apoyaremos en el Método de los Multiplicadores de Lagrange para su resolución. El multiplicador nos informa sobre la sensibilidad de la función objetivo (i.e. el coste) ante cambios en la restricción (i.e. en la producción). En concreto, representa lo que varía el coste al variar marginalmente la producción objetivo y su ssentido económico se interpreta como el coste marginal de la producción (${CMg}_Q$).

\subsubsection{Condiciones necesarias (CPOS’s, Khun-Thucker)}
Las condiciones de primer orden o condiciones necesarias para que el vector óptimo de demanda de factores sea la solución a este problema son las siguientes:

\eqblock{
\begin{aligned}
    &\text{CPO's}:\\[1em]
    &\left\{\begin{aligned}
        &\text{Estacionariedad}&&\Longrightarrow\;&&\boxed{\frac{\partial C(\vec z^*)}{\partial z_k}=w_k-\mu^*\cdot f_k(\vec z)\ge 0}\\[0.5em]
        &\text{No-negatividad}&&\Longrightarrow\;&&\boxed{0\leq z_k^*}\\[0.5em]
        &\text{Restricción producción}&&\Longrightarrow\;&&\boxed{f(\vec z^*)=\bar Q}\\[0.5em]
        &\text{Holgura complementaria}&&\Longrightarrow\;&&\boxed{z_k^*\cdot [w_k-\mu^*f_k(\vec z^*)]=0}
    \end{aligned}\right\}\Longrightarrow\;\forall k=1,2,\dots, n
\end{aligned}
}{Minimización del gasto: Condiciones de primer orden. Función de producción, análisis de dualidad}
 
En el óptimo, el multiplicador de Lagrange sería igual al cociente entre el coste del bien y su $PMg$.\footnote{%
    Nótese que se prescinde de la notación extendida, esto es, la PMg del bien (esto es, la derivada de la función de producción respecto al bien) utiliza la notación $f_{k}(z^{*})$
} Es decir, la medida en cómo se traslada la producción a costes depende del coste del factor, ponderado por su productividad. Esto implica que la empresa contrata un factor productivo hasta que su coste marginal es igual a su productividad marginal, cumpliendo la igualdad siguiente:

$$\frac{w_k}{f_k(\vec z^*)}=\mu^*\qquad \forall k=1,2,\dots, n$$
2
 
Y, estableciendo relaciones entre los diferentes factores productivos y el cociente entre sus precios, obtenemos una conclusión ya esperada: (suponiendo únicamente como factores productivos capital y trabajo)

$$\frac{w_K}{f_K(\vec z^*)}=\mu^*=\frac{w_L}{f_L(\vec z^*)}\;\Longrightarrow\;\underbrace{\frac{f_L(\vec z^*)}{f_K(\vec z^*)}}_{|{RMST}_L^K|}=\frac{w_L}{w_K}$$
 
Esto es, trabajando con las CPO’s, obtenemos que la RMST se iguala al cociente de precios de los factores productivos.\footnote{%
    [Tema 3.A.11] La cantidad de un factor productivo al que puede renunciar una empresa si aumenta la cantidad de otro factor productivo en una unidad para mantener el nivel de producción constante.
} Por lo que, en el óptimo, si la empresa desea aumentar marginalmente su producción, le será indiferente contratar más capital que más trabajo porque la PMg ponderada por el CMg es el mismo en ambos casos.

\subsubsection{Condiciones suficientes (CSO’s)}
Para garantizar que el vector de demanda óptimo de insumos es el vector solución a este problema es necesario que se den las CSO’s (condiciones suficientes), consistentes en que los menores del Hessiano orlado correspondiente sean negativos.\footnote{%
    El Hessiano orlado es una matriz que se utiliza para analizar la segunda derivada condicionada de una función objetivo sujeta a una o varias restricciones. Geométricamente, el Hessiano orlado describe la curvatura de la función objetivo restringida a la superficie $f(z) = 0$ [Restricción igual a cero]. En el punto crítico, mide cómo cambia la función objetivo a lo largo de las direcciones tangentes a la restricción. Si el Hessiano orlado es una matriz:
\begin{la}
    \item \textbf{Definida negativa}. Curvatura hacia abajo. La función objetivo decrece en todas las direcciones tangentes por lo que es un máximo restrinigido (local); y,
    \item \textbf{Definida negativa}. Curvatura hacia arriba. La función objetivo crece en todas las direcciones tangentes a la restricción por lo que es un mínimo restringido (local)
\end{la}
} En nuestro caso, como el Lagrangiano está compuesto por una única restricción y la función objetivo es una función compuesto de diferentes inputs utilizados para la producción, el Hessiano orlado vendrá definido por:

\eqblock{
\begin{aligned}
    &\mathcal L=\vec w\cdot \vec z+\mu[\bar Q-f(\vec z)]\\[1em]
    &\underset{\substack{\text{\scriptsize Hessiano}\\\text{\scriptsize orlado}}}{\widehat H}=\left[\begin{matrix}
        0&-\left(\frac{\partial f}{\partial z_1}(z_1^*,z_2^*, \dots, z_n^*,\mu^*)\right)&\cdots & -\left(\frac{\partial f}{\partial z_n}(z_1^*,z_2^*, \dots, z_n^*,\mu^*)\right)\\
        -\left(\frac{\partial f}{\partial z_1}(z_1^*,z_2^*, \dots, z_n^*,\mu^*)\right)&-\left(\frac{\partial^2 \mathcal L}{\partial^2 z_1}(z_1^*,z_2^*, \dots, z_n^*,\mu^*)\right)&\cdots &-\left(\frac{\partial^2 \mathcal L}{\partial z_1\partial z_n}(z_1^*,z_2^*, \dots, z_n^*,\mu^*)\right)\\
        \vdots &\vdots &\ddots&\vdots \\
        -\left(\frac{\partial f}{\partial z_n}(z_1^*,z_2^*, \dots, z_n^*,\mu^*)\right)&-\left(\frac{\partial^2 \mathcal L}{\partial z_n\partial z_1}(z_1^*,z_2^*, \dots, z_n^*,\mu^*)\right)&\cdots &-\left(\frac{\partial^2 \mathcal L}{\partial^2 z_n}(z_1^*,z_2^*, \dots, z_n^*,\mu^*)\right)    
    \end{matrix}\right]\\[1em]
    &\Longrightarrow\qquad\boxed{\det{\widehat H}<0}
\end{aligned}
}{Condiciones suficientes: Propiedades del Hessiano orlado. Función de producción: Análisis de dualidad}

Si el determinante del Hessiano orlado es inferior a cero, entonces podemos garantizar que nos encontramos ante un mínimo.\footnote{%
    En la derivación que proponemos en las Notas finales veremos como la condición es contraria, se exige que sea mayor que cero para un máximo. Esto no es sustancial ya que varía en función de cómo formulemos el Lagrangiano:
$$\mathcal{L}=w\cdot z+\mu(\bar Q-f(z))\qquad \text{o,}\qquad \mathcal L'=w\cdot z-\lambda(f(z)-\bar Q)$$
} Lo más importante es el significado geométrico o matemático: para poder optimizar la producción mediante la resolución del problema de minimización del gasto, la función de producción debe tener alguna región o subespacio cuya estructura a escala sea decreciente o constante, es decir, la función de producción debe tener una región de concavidad (no estricta) para que pueda existir una solución interior óptima.
Supondremos a partir de ahora que la Función de Producción es estríctamente cuasi-cóncava (para facilitar el análisis).
El Hessiano orlado indicaría que el Lagrangiano es convexo, por lo que cualquier desplazamiento factible que mantenga la producción supnodría un aumento del costo. Es decir, el plano isocosto es tangente inferior a la superficie de producción, es tangente en un punto donde la isocuanta es curvada hacia arriba respecto al plano isocosto. 

\textbf{\textcolor{red}{[IMPRESCINDIBLE LA REPRESENTACIÓN GRÁFICA DEL HESSIANO ORLADO PARA PODER ENTENDER EL PROBLEMA]}}

\subsubsection{Desarrollo gráfico}
Gráficamente, para el caso de dos inputs, representaríamos:
\begin{la}
    \item La isocuanta del nivel mínimo de producción que queremos alcanzar; y,
    \item Un mapa de curvas isocostes, que unan los puntos en los que distintas combinaciones de inputs tengan el mismo coste.
\end{la}

\subsubsection{Solución}
El equilibrio se encuentra en el punto de tangencia entre ambas, que supondrá la isocoste más cercana al origen sujeta a la restricción. En este punto, la RMST, es decir, el cociente de productividades marginales de los factores (pendiente de la curva isocuanta) se iguala al cociente del precio de los factores (pendiente de la curva isocoste).

\paragraph{Sistema Completo de Ecuaciones de Demanda de Inputs Condicionada: SCEDIC}
El SCEDIC describe cómo varían las demandas óptimas de factores productivos (inputs) que una empresa emplea, dado que produce un nivel determinado de output, cuando cambian los precios de los factores:

\eqblock{
\begin{aligned}
    \vec h^*(\vec w, \bar Q)=\left(\begin{matrix}
        h_1(w_1,w_2, \dots, w_n,\bar Q)\\
        \vdots\\
        h_n(w_1,w_2, \dots, w_n,\bar Q)\\
    \end{matrix}\right)
\end{aligned}
}{Problema de Minimización del Gasto: Sistema Completo de Ecuaciones de Inputs Condicionada (SCEDIC). Función de producción: ANálisis de dualidad}

Este Sistema de ecuaciones tiene las siguientes propiedades:
\begin{lnum}
    \item Las funciones de demanda son contínuas y homogéneas de grado cero tanto en costes como en cantidad de output $(p,Q)$; y,
    \item Es, con cierta generalidad, diferenciable tanto en costes como en output
\end{lnum}

\paragraph{Función Valor de Costes}
La Función Valor de costes es la función de valor mínimo que obtenemos al sustituir las soluciones del problema anterior (es decir, al sustituir las demandas condicionadas de los factores en la función objetivo de costes). Es de gran relevancia ya que presupone:
\begin{lnum}
    \item Eficiencia técnica, ya que hemos impuesto que cumpla la función de producción; y,
    \item Eficiencia económica, ya que, al introducir en el análisis el precio de los factores (coste), obtenemos el mínimo al que producir un nivel óptimo. 
\end{lnum}

\eqblock{
\begin{aligned}
    C^*(\vec w, \bar Q)=\vec w\cdot \vec h^*=\vec w\cdot \vec h^*(\vec w, \bar Q)
\end{aligned}
}{Problema de Minimización del Gasto: Función Valor de costes. Función de producción: Análisis de dualidad}

Además, posee las siguientes propiedades:
\begin{lnum}
    \item Es contínua tanto en costes como en cantidades de output;
    \item Es homogenea de grado uno en costes de los factores (w);
    \item Es estríctamente creciente en cantidades de output;
    \item Es no-decreciente, y cóncava, en coste de los factores. Esto se explica por la sustitución entre factores productivos ante variaciones en sus precios (coste). Si el precio de un factor aumenta, el coste aumentaría de forma proporcional si la empresa no modifica la combinación de factores. Sin embargo, cabe esperar que la empresa reaccione y el aumento en los costes aumente menos que proporcionalmente;\footnote{%
        Evidentemente, la tasa en la que se incrementa el coste viene definida por el grado de sustituibilidad de los factores, lo que se recoge en el concepto de elasticidad de sustitución de los factores [ver Tema 3.A.11]
    } y,
    \item Cumple el Lema de Shepard:
    $$h_K^*(\vec w, \bar Q)=\frac{\partial C^*(\vec w, \bar Q)}{\partial w_K}\qquad \forall k$$
\end{lnum}

Estas dos últimas propiedades llevan a reconocer ciertas propiedades de la matriz de sustitución de la función de costes, definida como la matriz Hessiana de la función de costes respecto de todos los precios:

\eqblock{
\begin{aligned}
    \Xi=\left(\begin{matrix}
        \xi_{11}&\cdots &\xi_{1n}\\
        \vdots &\ddots &\vdots \\
        \xi_{n1}&\cdots &\xi_{nn}
    \end{matrix}\right)
    \qquad \text{donde,}\qquad
    \begin{aligned}
        &\xi_{ki}\equiv \frac{\partial C^*(\vec w, \bar Q)}{\partial w_k, \partial w_i}\overbrace{=}^{\substack{\text{\scriptsize Por el Lema}\\\text{\scriptsize de SHEPARD}}}\frac{\partial h_K^*(\vec w, \bar Q)}{\partial w_i}\\[0.5em]
        &\hspace{5em}\text{\scriptsize$\forall k,i=1, 2, \dots, n\;\;\;\;k\neq i$}
    \end{aligned}   
\end{aligned}
}{Problema de Minimización del Gasto: Matriz Hessiana (Condiciones suficientes). Función de Producción: Análisis de dualidad}

La matriz de sustitución es:
\begin{lnum}
    \item Simétrica, de modo que el orden de derivación no importa, es decir, los efectos precio-cruzado para dos factores cualesquiera en las demandas condicionadas son idénticos; y,
    \item Semidefinida negativa (por la concavidad de la función de costes), lo que implica que los elementos de su diagonal principal son no-negativos, es decir, los efectos precio propios en las demandas hicksianas sean no-positivos.\footnote{%
        Esto es así porque todos los elementos de la diagonal principal de una matriz semidefinida negativa deben ser no positivos (nulos o negativos). Y estos elementos de la matriz de sustitución son los efectos precio propios de modo que los efectos precio-propios son negativos, lo que implica que si aumenta el precio de un factor, su demanda no condicionada se reduce o se mantiene constante, pero en ningún caso aumenta.
    }
\end{lnum}

Al igual que hicimos en la teoría del consumo [ver tema 3.A.9], vamos a introducir y sintetizar una serie de resultados que nos permitirán obtener toda Función obtenida en la Teoría de la Producción a partir de cualquier otra.

\subsubsection{Dualidad entre funciones de costes y la función de producción}
Una aplicación muy útil de la teoría de la dualidad es que ésta nos permite asegurar que la función de costes proviene de la optimización de una función de producción. Es decir, que la función de costes contiene esencialmente la misma información que la función de producción. En particular, si disponemos de la función de costes es posible recuperar la función de producción siguiendo un procedimiento análogo al utilizado en la teoría del consumo [ver tema 3.A.9]:

$$C^*(\vec w, \bar Q)\;\underset{\substack{\text{\scriptsize Invirtiendo, y}\\\bar C\equiv C^*}}{\Longrightarrow}\;q^*=\zeta(\vec w,\bar C)$$
 
La función de la derecha nos informa sobre el máximo output que es posible producir dados los precios de los factores y un nivel de coste dado. A partir de ella puede recuperarse la tecnología original mediante el siguiente problema:\footnote{De la misma forma que hicimos para la integrabilidad de las preferencias. 
}

$$\begin{aligned}
    &\arg \min_{\vec z}\quad \zeta(\vec w, \bar C)\\
    &\text{s.a.}\quad \vec w\cdot \vec z\leq \bar C
\end{aligned}$$

\subsubsection{Dualidad entre la función de beneficio y la de producción}
La tecnología puede representarse mediante el conjunto productivo ($\mathcal Y$), espacio delimitado por la función $f$ y definido como:

$$\underset{(2)}{\\[2em]\mathcal Y=\{(q, \vec z)>0\;;\;q<f(\vec z)\}}$$

Pero, de forma equivalente, la función de Beneficios es la función de soporte de $\mathcal Y$, es decir:

$$\mathcal Y=\{(q,\vec z)>0\;:\;p\cdot q-\vec w\cdot \vec z<\Pi(\vec w, p)\}$$

Gráficamente:

\imagenfit{Rel_DUALES_ConTECNOLOGICO.png}{Conjunto tecnológico y fronteras. Relaciones duales beneficio-gasto}{Apuntes Víctor Gutiérez. Tema 3.A.12}

Como la función de beneficios es homogénea de grado uno en el vector de precios y que éste es estríctamente mayor que cero, entonces:

$$\underset{(3)}{\\[2em]\Pi\left(\frac{1}{p}\cdot p,\frac{1}{p}w\right)=\frac{1}{p}\Pi(p,w)\;\underset{\text{\scriptsize Reales}}{\Longrightarrow}\;\Pi(1, \omega)\quad \text{con}\;\;\;\omega=\frac{w}{p}}$$

Utilizando estas relaciones (2 y 3), dividiendo por $p$ y reordenando tenemos que:

$$\underset{(4)}{\\[1em]\mathcal Y\equiv \{(q, \vec z)>0\;:\;q\leq \min{[\omega\cdot \vec z+\Pi(1, \omega)]\;\;;\;\;\text{s.a.}\;\;\omega \gg0}\}}$$

La expresión a minimizar es convexa (dado que es la suma de dos funciones convexa). Las condiciones necesarias de mínimo nos permiten encontrar sus valores óptimos ($\omega^*$) a partir de las siguientes ecuaciones:

$$\frac{\partial}{\partial w_k}[\omega\cdot \vec z+\Pi(1,\omega)]=z_k+\frac{\partial \Pi(1, \omega)}{\partial w_k}=0;\qquad \forall k=1, 2, \dots, n$$
 
Por el Lema de Hotelling -que la derivada parcial de la función de beneficio respecto a cada uno de los costes de cada factor es igual a la cantidad demandada de éste para el nivel óptimo, cambiado de signo, sustituyendo tendremos que:

$$z_k=z_k(1,\omega);\qquad \forall k= 1, 2, \dots, n$$

De donde será posible obtener el vector de soluciones:

$$w_k^*=w_k(\vec z)\qquad \forall k=1,2,\dots, n$$
 
Sustituyendo en la función objetivo (4):

$$\mathcal Y\equiv \{(q, \vec z)>0\;:\;q\leq w(\vec z)\cdot \vec z+\Pi(1, w(\vec z))\}$$

De acuerdo con la definición de función de beneficios, se verificará que $\Pi(1,\omega)=q-w\cdot \vec z$, por lo que:

$$\underset{(6)}{\\[1em]w(\vec z)\cdot \vec z+\Pi(1, w(\vec z))=q(w(\vec z))}$$

Y, comparando (6) y (1) vemos que:

$$f(\vec z)=q(w(\vec z))$$

Por lo que habremos recuperado la función de producción a partir de la solución del problema de minimización del coste. De manera sintetizada:

\eqblock{
\begin{aligned}
    &\hspace{0.5em}\Pi(p,w)&&\Rightarrow \hspace{2em}-\frac{\Pi(p,\vec w)}{\partial w_k}=z_k^*=z_k(p,\vec w)=z_k(1, \omega) &&\Longrightarrow \hspace{3em} w_k^*=w_k(\vec z)\\[1em]
    &\hspace{1.7em}\Downarrow&& &&\hspace{7em}\Downarrow \\[1em]
    &\frac{\partial \Pi(p, \vec w)}{\partial p}&&\Rightarrow\hspace{5.5em} q^*=q(p,\vec w)=q(1,\omega) &&\Longrightarrow\hspace{2em}\boxed{f(\vec z)=q(1, \omega(\vec z))}
\end{aligned}
}{\textcolor{red}{Poner título a esta ecuación}}
 
\subsection{Teorema básico de la dualidad}
Los axiomas anteriores nos permiten representar la tecnología mediante una función de producción, que se define como una relación puramente técnica entre insumos productivos y volúmenes de producción incluyendo todos los métodos de producción.\footnote{%
    En realidad, podemos distinguir dos categorías de axiomas:
\begin{la}
    \item Necesarios (garantizan que la tecnología pueda representarse mediante una función de producción): (i) no vacío; (ii) cerrado; (iii) eliminación gratuita; y, (iv) imposibilidad deproducción gratuita; y,
    \item Deseables (dan pie a una tecnología de buen comportamiento y, por ende, a una función neoclásica de buen comportamiento, por lo que son condiciones indispensables para la competencia perfecta): (i) posibilidad de inacción, se puede elegir si producir o no; y, (ii) convexidad, La convexidad implica la posibilidad de divisibilidad de escala (i.e. se puede reducir la escala de la producción). Además implica rendimientos a escala no crecientes. Los procesos extremos son menos productivos que los promediados. En el caso de un solo output este axioma implica que la función de producción es cóncava. Para maximizar el beneficio esta condición es esencial.
\end{la}
}

\clearpage
\section{Dimensión temporal de la función de costes}
\puntosclave{
    La teoría económica distingue entre los costes a corto plazo y los costes a largo plazo. Los costes a corto plazo son aquellos en los que algún factor de producción (generalmente los activos de capital y la dirección de la empresa) son fijos.

    
	Los costes a largo plazo son aquellos a través de un periodo de tiempo lo suficientemente largo como para permitir el cambio de todos los factores de producción. En el largo plazo todos los factores son variables.\footnote{%
	    Se puede entender también del siguiente modo. Los costes de corto plazo son aquellos con los que la empresa opera en cualquier periodo. Los costes a largo plazo son costes planeados o costes ex ante en tanto que representan las posibilidades óptimas de expansión de la produccióny de esta forma ayudan al emprendedor a planear sus actividades futuras. De esta forma, previo a cualquier inversión, el empresario se encuentra en una situación de largo plazo en el sentido de que éste puede elegir una alternativa de entre un amplio conjunto de inversiones posibles definidas por el estado de la tecnología. Una vez que la decisión de inversión se ha tomado y los fondos son atados en activos de capital fijo, el empresario opera bajo condiciones de corto plazo: ahora está en una curva de coste de corto plazo.
	}

	De la representación de las curvas de costes en el corto plazo se puede observar que al mínimo del coste medio total se le denomina óptimo de explotación, ya que, si el precio es igual al mínimo, se cubren con los igresos todos los costes
    
	La curva de coste medio a largo plazo representa el coste medio mínimo para obtener cada nivel de producción, cuando se dispone del tiempo necesario para ajustar óptimamente los factores de producción y no tiene por qué coincidir con el mínimo de los costes medios a corto plazo. Al tamaño de planta para el que coincide el mínimo del coste medio a corto plazo con el mínimo del coste medio a largo plazo se le denomina dimensión óptima de la empresa o tamaño óptimo de planta
    
	Según el Teorema de LE CHÂTELIER-SAMUELSON las funciones de oferta de la empresa competitiva a corto plazo son más inelásticas que la de largo plazo y tanto más cuanto mayor es el número de factores considerados fijos.\footnote{Demostración en SEGURA (1988), pag. 196
    }
    
	Las economías de escala ocurren cuando los costes crecen menos que proporcionalmente que la producción, X, o de forma equivalente, cuando los costes medios a largo plazo son decrecientes en el nivel de producción. Los rendimientos a escala crecientes implican economías de escala, pero no al revés.
}
	
En el apartado anterior hemos visto cómo del problema de minimización de costes obtenemos dos resultados: un SCEDIC y la Función Valor de Costes.
Estos resultados se derivan de las funciones de producción que describen los métodos de producción técnicamente eficientes. Pero además, en este problema se incorporan también los precios, por lo que la función de costes garantiza además la eficiencia económica.

Supongamos dos factores productivos, capital ($K$) y trabajo ($L$). En este caso, el problema de minimización del coste de la empresa se puede especificar de la siguiente manera:

\eqblock{
\begin{aligned}
    &\boxed{
    \begin{aligned}
        &\min_{\{K,L\}}\quad C(K,L)=w_L\cdot L+w_K\cdot K\\
        &\text{s.a.}\quad
        \begin{cases}
            \bar Q\leq f(K,L)\\[0.5em]
            \{K,L\}\in \mathbb R^+
        \end{cases}
    \end{aligned}
    }\\[2em]
    &\text{SCEDIC:}\quad
    \underset{\substack{\text{\scriptsize Sustituimos en la Función Objetivo}\\\text{\scriptsize y obtenemos Función de Costes}}}{
    \begin{cases}
        L^*(\vec w, \bar Q)\\
        K^*(\vec w, \bar Q)
    \end{cases}\;\Longrightarrow}\;\boxed{C(\vec w, \bar Q)=\vec w\cdot \vec h(\vec w, \bar Q)=w_L\cdot L^*(\vec w, \bar Q)+K^*(\vec w, \bar Q)}
\end{aligned}
}{Análisis temporal de la producción: Problema de minimización. Dimensión temporal de la producción: Punto de partida}
 
A partir de la resolución del problema, obtenemos las funciones de demanda de inputs condicionada, que luego utilizamos para sustituir en la función objetivo con el fin de obtener la Función de Costes para el caso de dos factores productivos. Ahora vamos a ver tanto gráficamente como analíticamente como varían estas funciones cuando varían los parámetros de los que dependen, es decir, cuando varían los precios de los factores productivos y cuando varía la cantidad producida. Será, por tanto, análogo a lo visto en la Teoría de la demanda neoclásica del consumidor [ver Tema 3.A.9].

\subsubsection{Variaciones en el nivel de producción (Q): Perspectiva de demanda de factores}
Veremos a continuación dos formas de analizar las variaciones sobre la demanda de factores productivos atendiendo a una variación en el nivel de producción. Estas dos perspectivas, complementarias, son fundamentalmente: el análisis gráfico y el análisis en base a elasticidades.
Trabajaremos sobre las funciones de demanda de factores que forman el SCEDIC: $\vec h^*(\vec w,\bar C)$

\paragraph{Análisis gráfico: Curva producción-demanda de factores productivos (Senda de Expansión)}
La trayectoria o senda de expansión óptima es el lugar geométrico de los puntos o combinaciones óptimas de factores que se obtienen al ir aumentando los factores y, por consiguiente la producción, manteniendo los precios de los factores constantes. 

\imagenfit{Senda_Expansion.png}{Senda de expansión. Análisis largo plazo}{Apuntes Víctor Gutiérrez. Tema 3.A.12}

A lo largo de la trayectoria de expansión óptima, la RMST es igual a la razón de los precios y, como asumimos que los precios no varían, la RMST permanece constante. Como consecuencia, la trayectoria de expansión óptima es una isoclina [\authorfont{Ver Tema 3.A.11}] a lo largo de la cual aumenta la producción, permaneciendo constante el precio de los factores de producción e indica cómo cambian las proporciones de los factores al modificarse el nivel de producción, manteniéndose constante el coste de éstos. Si la función de producción es homotética, en el caso de dos factores, el cociente de productividades marginales solo depende de la cantidad relativa de los factores, y, por tanto, dada una combinación inicial de factores, mientras se mantenga constante la relación entre el precio de los factores cualquier punto que pase por el origen y por dicha combinación pertenece a la senda de expansión.\footnote{%
    La senda de expansión se puede volver hacia atrás. En este caso hablaríamos de inferioridad del factor trabajo: elegimos menos trabajo cuando la producción aumenta por encima de un nivel de producción. De la misma manera, si la curva tuviera pendiente negativa en el sentido opuesto en algún tramo, el factor inferior sería el capital.
} En este caso, la senda de expansión es un radio que pasa por el origen de coordenadas.

\paragraph{Análisis gráfico: Curva de ENGEL del factor productivo}
De la senda de expansión óptima se puede obtener la curva de ENGEL, que relaciona la cantidad demandada de un factor de producción con el nivel de producción, es decir, nos trasladamos desde un plano cartesiano de factores a un plano cartesiano de factor-producción. A partir del análisis de la curva de ENGEL diremos que:
\begin{la}
    \item Un factor productivo es normal en coste y producción ($w,Q$) si la derivada parcial de la Función Valor de Demanda del Factor (función del factor $i$ del SCEDIC, $h_{i}$) en función de coste y producción respecto al output es mayor que cero, esto es si la demanda es creciente en la cantidad de capital (riqueza) y, por lo tanto, la pendiente de su curva de Engel es positiva;\footnote{%
        Si la función de producción es homotética, la senda de expansión es una recta de pendiente positiva y por lo tanto ambos factores productivos son normales.
    } o,
    \item Si por el contrario el efecto riqueza es negativo y su curva de ENGEL tiene pendiente negativa, se dice que el bien es inferior en ($w,Q$). Podría suceder que la pendiente dependa del tramo en el que nos encontremos (del nivel inicial de producción / riqueza)
\end{la}

\imagenfit{Curva_ENGEL_FFPPs.png}{Curva de ENGEL sobre factores productivas. Tramos de la Curva}{Apuntes Víctor Gutiérrez. Tema 3.A.12}
 
\paragraph{Análisis de elasticidades: elasticidad-producto-factor }
Además, analíticamente podemos obtener una medida de la sensibilidad de la demanda de factores productivos ante variaciones en la cantidad producida: la elasticidad de la demanda del factor productivo-cantidad producida. La elasticidad de la demanda del factor productivo respecto a la cantidad producida mide la variación porcentual en la cantidad demandada de un factor productivo ante una variación del 1\% de la cantidad producida:

\eqblock{
\begin{aligned}
    &\text{Elasticidad $\bar Q$ - factor}:&& \varepsilon_{h_i,\bar Q}=\frac{\partial h_i(\vec w, \bar Q)}{\partial \bar Q}\cdot\frac{\bar Q}{h_i(\vec w,\bar Q)}
\end{aligned}
}{Elasticidad producto factor. Dimensión temporal (largo plazo): Variaciones en el nivel de producción}

De este modo, podemos clasificar los factores productivos de la siguiente manera:
\begin{la}
    \item Factores productivos normales. Son aquellos para los que la elasticidad-producto-demanda es mayor a cero. Lo que implica que a mayor cantidad a producir, mayor cantidad demandada del factor productivo;
    \item Factores productivos frontera. Son aquellos para los que su elasticidad-producto-demanda. Lo que implica que si varía la cantidad a producir, la cantidad demandada del factor productivo no variará; y,
    \item Factores productivos inferiores. Son aquellos para los que su elasticidad-producto-demanda es inferior a cero. Lo que implica que a mayor cantidad a producir, menor cantidad demandada del factor productivo.
\end{la}

Como es normal, la elasticidad podría depender del nivel de producción de partida, y por lo tanto un factor productivo puede ser normal para ciertos niveles de producción e inferior para otros.

\subsubsection{Variaciones en el nivel de producción ($Q$): Perspectiva de costes}
\textbf{NOTA.-} Entender que lo que estamos haciendo es en el mismo gráfico que antes, requerir una curva isocuanta más alejada del origen de coordenadas, por lo que el coste total aumenta a medida que aumenta $\bar Q$, pero la forma que adopte la función de Coste Total (cóncava o convexa) dependerá de la forma funcional de la función de producción (rendimientos crecientes o decrecientes a escala). Hacer aquí todo el análisis de largo plazo, hablar de coste marginal y coste medio, de economías de escala (internas y externas) y hacer el análisis de elasticidades. \paragraph{Análisis gráfico: las Curvas de Costes, $CT(Q)$, $CMe(Q)$, $CMg(Q)$}
Si suponemos un vector de costes constante, podemos representar la Función Valor de Costes solo en función del output C(Q) sobre el plano cartesiano.\footnote{%
    Supondremos que se cumple (para esta función de producción) la Ley de proporciones variables, según la cual -menteniendo un nivel fijo de un factor productivo- si incrementamos la cantidad del otro factor productiva, llegará un momento en el que su PMg disminuirá, de tal forma que llegue incluso a ser negativa a partir de un determinado punto. Sin embargo, si la función deproducción es neoclásica de buen comportamiento y convexa en todos sus tramos (p.e. tipo Cobb-Douglas), no se dará este caso. Esta Ley también se conoce como la Ley de Rendimientos Marginales decrecientes.
}

\imagenfit{CurvaCostes_VarProduccion.png}{Curva de costes para un mismo nivel de producción}{Apuntes Víctor Gutiérrez. Tema 3.A.12}

Los Costes Medios ($CMe$) se definen como los $CT(Q)$ divididos entre la cantidad producida (o como la pendiente de la secante que parte del origen de coordenadas).\footnote{%
    Para representar el coste medio hay que fijarse en la pendiente de la secante que parte desde el origen de coordenadas. Los CMe decrecerán hasta que la recta secante a la curva de Costes Totales sea tangente a la curva Costes Totales y a partir de ese punto comienza a crecer.
} También podemos definir los costes marginales como el coste de producir una cantidad marginalmente mayor y se calcula como la pendiente de los costes totales en cada punto (o como la pendiente de la tangente en cada punto).\footnote{%
    Para representar el coste marginal hay que fijarse en la curvatura de la función de costes totales. Los CMg decrecerán cuando la función Costes Totales sea cóncava (la pendiente decrece a medida que aumenta la producción) y crecerá cuando la función Costes Totales sea convexa. El mínimo del coste marginal se da en el punto de inflexión de la curva de costes totales.
}

\eqblock{
\begin{aligned}
    CMe=\frac{C(\vec w,\bar Q)}{\bar Q}\qquad \text{y}\qquad CMg=\frac{\partial C(\vec w, \bar Q)}{\partial \bar Q}
\end{aligned}
}{Definición Costes Medios y Costes Marginales. Dimensión temporal de la producción: Variaciones en el nivel de proónducci}

La curva de $CMe(Q)$ a la que da lugar esta función de Costes Totales tendrá, a largo plazo, forma de U.\footnote{%
    WICKSELL fue el primer economista que señaló que podrían existir rendimientos mixtos a escala, es decir, para todo nivel de producción es posible que la empresa pudiera pasar por los tres rendimientos: al principio de la expansión serían crecientes, pero pasarían a ser decrecientes y en el punto de inflexión serían constante.
} Como el coste del factor se mantiene idéntico para todos los niveles de producción, cuando la PMg comience a decrecer, afectará negativamente a los costes (los aumentará) y a partir de ese punto la función de Costes Totales será convexa, es decir, crece aceleradamente.\footnote{%
    En este caso estamos suponiendo que dichos rendimientos marginales decrecientes aparecen en un cierto punto, es decir, la PMg del factor elástico / variable aumenta hasta un cierto punto, a partir del cual al aumentar la cantidad del factor variable su PMg comienza a reducirse. Ello es lo que origina que el CMe aumente a partir de un determinado punto.
}\textsuperscript{,}\footnote{%
    Esta es la clave de que los rendimientos marginales crecientes (PMg creciente) implique economías de escala pero que las economías de escala no necesariamente impliquen rendimientos marginales crecientes (PMg puede ser creciente o decrecientes).
}

La curva de CMe será decreciente hasta el punto donde se igualan con los CMg. A partir de este punto, será creciente, teniendo por tanto forma de U:
\begin{la}
    \item En el tramos de CMe decrecientes, al aumentar el nivel de producción los costes aumentan menos que proporcionalmente, por lo tanto diremos que estamos ante economías de escala; y, 
    \item Por otro lado, en el tramos en que los CMe son crecientes, los costes aumentarán proporcionalmente más que la producción, por lo que diremos que estamos ante deseconomías de escala.
\end{la}

\paragraph{Análisis de elasticidades: elasticidad-coste-producto}
Podemos medir el concepto de las economías de escala mediante la elasticidad del coste respecto a la cantidad producida. La elasticidad del coste respecto a la cantidad producida mide la variación porcentual en el coste ante una variación porcentual de la cantidad producida:

\eqblock{
\begin{aligned}
    &\varepsilon_{C,\bar Q}=\overbrace{\frac{\partial C(\vec w, \bar Q)}{\partial \bar Q}}^{\equiv CMg(\bar Q)}\cdot \underbrace{\frac{\bar Q}{C(\vec w, \bar Q)}}_{\equiv\frac{1}{CMe(\bar Q)}}=\frac{CMg(\bar Q)}{CMe(\bar Q)}\\[2em]
    &\text{y, si}\quad 
    \left\{\begin{aligned}
        &\varepsilon_{C,\bar Q}<1\Longrightarrow &&\underset{\text{\scriptsize $\Rightarrow CMe$ decrecientes}}{CMg(\bar Q)<CMe(\bar Q)}\Longrightarrow&&\text{Economías de escala}\\[1em]
        &\varepsilon_{C,\bar Q}=1\Longrightarrow &&\underset{\text{\scriptsize $\Rightarrow CMe$ constantes}}{CMg(\bar Q)=CMe(\bar Q)}\Longrightarrow&&\text{Ausencia economías de escala}\\[1em]
        &\varepsilon_{C,\bar Q}>1\Longrightarrow &&\underset{\text{\scriptsize $\Rightarrow CMe$ crecientes}}{CMg(\bar Q)>CMe(\bar Q)}\Longrightarrow&&\text{Deseconomías de escala}
    \end{aligned}\right.
\end{aligned}
}{Elasticidad coste producto. Dimensión temporal (largo plazo): Variaciones en el nivel de producción}
 
De hecho, podemos ver, con ayuda del Gráfico de Costes Totales como una misma función puede presentar economías mixtas (la Función de Producción presenta zonas de concavidad, convexidad y escalabilidad constante, rendimientos mixtos). En el sentido de que puede iniciar presentando economías de escala, tener un periodo con ausencia de economías de escala y luego presentar deseconomías de escala.

\paragraph{Relaciones: elasticidad-escala y elasticidad-coste-producto}
Aunque resulta tentador, el concepto de economías de escala [Tema 3.A.12] no puede ser confundido con el concepto de rendimientos de escala [\authorfont{Tema 3.A.11}].\footnote{%
    La elasticidad-escala mide la variación porcentual en el nivel de producción ante una variación del porcentual equiproporcional de todos los factores productivos:
$$\varepsilon_{Q,\lambda}=\frac{\mathrm dQ}{\mathrm d\lambda}\cdot\frac{\lambda}{Q}=\underset{\varepsilon_{Q,L}}{\frac{{PMg}_L}{{PMe}_L}}+\underset{\varepsilon_{Q,K}}{\frac{{PMg}_K}{{PMe}_K}}$$
} Las economías de escala son un concepto más amplio que los rendimientos a escala, dado que no exigen que la variación que experimenta la producción tenga su causa en una variación equiproporcional de los factores productiva (salto de isocuantas).

De esta forma, aunque resulta tentador pensar que las economías de escala surgirán únicamente cuando la función de producción exhiba rendimientos crecientes a escala y que habrá deseconomías de escala cuando la función de producción exhiba rendimientos decrecientes a escala esto no siempre será así:
\begin{lnum}
    \item En el corto plazo, podría darse el caso que una función de producción, la función de costes exhibiese economías de escala incluso cuando la función de producción exhibiese rendimientos a escala decrecientes pues la empresa puede obtener economías de escala con el aumento de un solo factor productivo (si así se experimenta);\footnote{%
        Por ejemplo, imaginémos que los precios de los factores no son constantes (por descuentos, aprendizaje, indivisibilidades, mejor utilización de equipos fijos, etc.), la empresa puede lograr economías de escala en costes aunque su tecnología tenga rendimientos decrecientes a escala.
    }
    \item En el largo plazo, rendimientos decrecientes sí implican deseconomías de escala salvo que se rompa alguna de las hipótesis estándar del largo plazo. Es decir, sí podría suceder, pero solo si:
    \begin{la}
        \item Los precios de los factores no son constantes (por descuentos, poder de mercado o aprendizaje); o,
        \item Existen eficiencias externas o tecnológicas que reducen los costes al crecer la escala (por ejemplo, economías externas o de red). En ese caso, la función de costes puede mostrar economías de escala aparentes, aunque la tecnología interna tenga RDE.
    \end{la}
    Pero eso ya no proviene de la tecnología de producción, sino de factores externos o de mercado.
\end{lnum}

En cambio, si la función de producción es homotética, la relación entre economías de escala y rendimientos a escala si será biunívoca. Concretamente, teniendo en cuenta que los rendimientos a escala de una función de producción se pueden medir mediante la elasticidad-escala, la relación entre ambos conceptos en el caso de funciones homotéticas vendrá dada por:

\eqblock{
\begin{aligned}
    \underset{\text{\scriptsize Función producción homotética}}{\text{Caso particular:}}&&\varepsilon_{C,\bar Q}=\frac{1}{\varepsilon_{Q,\lambda}}
\end{aligned}
}{Relación: Elasticidad-escala y Rendimientos a escala. Dimensión temporal: Variaciones en el nivel de producción}
 
\textbf{NOTA.-} Por ejemplo, cuando el factor productivo variable (en nuestro caso L) exhibe rendimientos marginales crecientes los costes marginales son decrecientes. Del mismo modo, cuando los rendimientos marginales son decrecientes, los costes marginales serán crecientes, formándose una curva de costes marginales en forma de U.

\subsubsection{Variaciones en el coste de los inputs ($\vec w$): Perspectiva de costes}
Habiendo visto como varían las demandas de factores productivos ante cambios en sus precios, pasamos a realizar el mismo análisis para la función de costes:

$$C(\vec w, \bar Q)=w_L\cdot L^*(\vec w,\bar Q)+w_K\cdot K^*(\vec w, \bar Q)$$
 
El efecto que tendrá un aumento proporcional en el coste de los inputs sobre el coste total de la empresa es directo. Esto es así dado que, como ya hemos dicho, la función de costes es homogénea de grado uno en coste de los factores:

$$C^*(\theta\cdot\vec w,\bar Q)=\theta^1\cdot C^*(\vec w, \bar Q)=\theta C^*(\vec w,\bar Q)$$
 
Los efectos que tiene el cambio del precio de un solo factor sobre el coste total también son bastante directos. Esto lo podemos ver tanto analíticamente como gráficamente. Desde un punto de vista gráfico, variaciones en el precio de los factores productivos se traducirán en desplazamientos de la curva de costes totales en el plano ($C,\bar Q$), aumentando del mismo modo los costes medios para todo nivel de producción $\bar Q$.

\paragraph{Análisis de elasticidades: elasticidad-precio-propio}
Además, aplicando el lema de Shephard, la elasticidad del coste respecto al precio propio se puede escribir como:

$$\varepsilon_{C,w_i}=\frac{\partial C(\vec w,\bar Q)}{\partial w_i}\cdot \frac{w_i}{C(\vec w,\bar Q)}=\frac{h_i(\vec w,\bar Q)\cdot w_i}{C(\vec w,\bar Q)}$$
 
La sensibilidad del coste a un cambio en el precio de un factor individual es igual a la proporción que representa el gasto de ese factor sobre el coste total.\footnote{%
    Dado que el coste medio se mantiene constante, para determinar el efecto de la variación del coste sobre el coste medio sería necesario calcular la elasticidad del coste medio respecto a al coste del factor ($i$) que también es igual a la participación en el gasto del factor $i$:
$$\varepsilon_{{CMe}_i,w_i}=\frac{\partial C(\vec w,\bar Q)/\bar Q}{\partial w_i}\cdot\frac{w_i}{C(\vec w,\bar Q)/\bar Q}=\frac{h_i(\vec w,\bar Q)}{\bar Q}\cdot\frac{w_i}{C(\vec w_i,\bar Q)/\bar Q}=\frac{h_i(\vec w,\bar Q)\cdot w_i}{C(\vec w,\bar Q)}$$
El efecto de un aumento dado en el coste del factor ($w_{i}$) sobre las curvas de costes totales y costes medios será desplazar estas curvas verticalmente hacia arriba en una cantidad que dependerá de la proporción que suponga el gasto en el factor productivo $i$ sobre el gasto total. Esto no significa que las curvas se desplacen en la misma proporción para todos los niveles de producción, dado que es posible que la proporción de lo que supone el gasto en ese factor productivo no sea constante con el nivel de producción.
}

\subsection{Corto plazo y Largo plazo}

\subsubsection{Corto plazo: C/p}
Empezaremos trabajando sobre el corto plazo pues debemos resolver un problema de optimización adicional para trabajar sobre este. Consiste en restringir lo expuesto inicialmente para las relaciones de dualidad al supuesto de que solo un factor productivo es variable (o varios, lo importante es que haya algunos fijos y otros variables). Esto es la esencia del corto plazo.


\paragraph{Desarrollo}
Entonces, para dos factores productivos, suponiendo que el factor capital ($K$) es fijo y el factor trabajo ($L$) es variable, tendríamos:

$$\begin{aligned}
    \min_{L}\;C(L)=\overbrace{w_L\cdot L}^{\text{\scriptsize Costes variables}}+\underbrace{w_K\cdot K}_{\text{\scriptsize Costes fijos}}\\[1em]
    \text{s.a.}\quad
    \left\{\begin{aligned}
    &\bar Q\leq f(\bar K,L)\\[0.5em]
    &0\leq L
    \end{aligned}\right.
\end{aligned}$$

Vemos que el problema de minimización del coste a corto plazo es idéntico al problema general de largo plazo descrito anteriormente, salvo por el hecho de incorporar una restricción adicional correspondiente a la limitación de los factores fijos (i.e. tamaño de planta). Para resolver este problema tendremos que elegir la cantidad de trabajo óptima para una determinada producción, es decir, la demanda condicionada del factor trabajo.
En el óptimo, la empresa utilizará el capital fijado y contratará trabajo de acuerdo a una función de demanda condicionada de trabajo. También podremos expresar la función de valor del problema (función de costes) como la suma de los costes totales a corto plazo (i.e. suma de los costes variables y los costes fijos):

$$\begin{aligned}
    L^*(\vec w,\bar K,\bar Q)\;\Longrightarrow\;\boxed{C^*(\vec w,\bar Q,\bar K)=w_L\cdot L^*(\vec w,\bar Q,\bar K)+w_K\cdot K}
\end{aligned}$$
 
\paragraph{Costes Totales: C/p vs. L/p}
Por la limitación del factor fijo, los costes de corto plazo asociados a la producción de un cierto nivel de output no pueden ser menores que los correspondientes costes de largo plazo. El gráfico de la senda de expansión permite comprobar como a corto plazo los costes van a ser mayores o iguales que a largo plazo.

\imagenfit{CT_Produccion_Senda.png}{Comparativa corto y largo plazo. Costes de producción}{Apuntes Víctor Gutiérrez. Tema 3.A.12}

La senda de expansión óptima (objetivo al que llegar a corto plazo) es el lugar geométrico de los puntos o combinaciones óptimas de factores productivos que se obtienen al ir incrementando la producción manteniendo los precios de los factores constantes (por combinaciones óptimas, nos referimos a la combinación óptima de largo plazo cuando todos los factores son variables, es decir, la tangencia entre isocoste e isocuanta).
Para producir la primera isocuanta ($\bar Q_{1}$), lo ideal es situarse en el punto $A$. Pero si el capital está fijo en $\bar K$, para producir esa misma cantidad nos tenemos que situar en una isocoste mayor, concretamente en el punto $B$. En cambio, para producir la segunda isocuanta ($\bar Q_{2}$) lo ideal es situarse en el punto $C$ cuya dotación de capital coincide con el capital fija, por lo que los costes a corto plazo son iguales que los costes a largo plazo y la asignación de capital es óptima: es decir, estamos en el tamaño de planta óptimo.

En conclusión, como la empresa a corto plazo no puede redimensionarse en todos los factores para elegir el tamaño de planta óptimo conjunto, la solución al problema de optimización a corto plazo coincidirá con la solución al problema a largo plazo, pues se situará en la dotación de factores que forme parte del conjunto de soluciones a L/p.

De este modo, la empresa a corto plazo siempre producirá de forma técnicamente eficiente y a largo plazo producirá también de forma económicamente eficiente. Sin embargo, a corto plazo no podemos asegurar que esté produciendo de forma ecnómicamente eficiente, ya que sólo será así si el nivel del factor productivo fijo coincide con el nivel de factor productivo que contrataría si pudiera elegir qué cantidad de este usar. 

\paragraph{Representación gráfica: Desagregación de costes a corto plazo}
Al analizar los costes en el corto plazo, lo primero que se debe saber es que existen costes fijos y costes variables:
\begin{la}
    \item \textbf{Costes Fijos}. Incluyen principalmente factores que requieren tiempo para cambiar (p.ej. el alquiler, la tecnología, etc.), y a menudo están asociados con los costes indirectos de producción (todo lo que no es un factor de producción directo). No dependen del nivel de producción y no se pueden ajustar de acuerdo a este; y,
    \item \textbf{Costes Variables}. Dependen del nivel de producción y pueden incluir costes asociados a materias primas, la mano de obra (sin incluir salarios fijos), etc.\footnote{%
        Sólo se incluirán aquellos salarios que guarden relación con la cantidad producida o las horas trabajadas para realizar producto (output)
    }
\end{la}
	
Analíticamente, habíamos obtenido que la función de costes a corto plazo sería:

\eqblock{
\begin{aligned}
    C^*(\vec w, \bar Q,\bar K)=w_L\cdot L^*(\vec w, \bar Q,\bar K)+w_K\cdot \bar K
\end{aligned}
}{}

Los $CV(Q)$ son curvados en forma de S: (i) en un primer momento, crecen cada vez más lentamente a medida que aumenta la producción debido a los rendimientos marginales crecientes; (ii) llegado cierto nivel, alcanzan un punto de inflexión; y, (iii) comienzan a crecer aceleradamente debido a los rendimientos marginales decrecientes.

Podemos hallar los costes marginales como la pendiente de los costes totales en cada punto (o como la pendiente de la tangente en cada punto) y los CMe como los $CT(Q)$ divididos entre la cantidad producida (o como la pendiente de la secante que parte del origen de coordenadas).\footnote{%
    Para representar el coste marginal hay que fijarse en la curvatura de la función de CT(Q). Los CMg decrecerán cuando la función CT sea cóncava (la pendiente decrece a medida que aumenta la producción) y crecerá cuando la función CT sea convexa. El mínimo del coste marginal se da en el punto de inflexión de la curva de costes totales.
}\textsuperscript{,}\footnote{%
    Para representar el coste medio hay que fijarse en la pendiente de la secante que parte desde el origen de coordenadas. Los CMe decrecerán hasta que la recta secante a la curva de CT sea tangente a la curva CT y a partir de ese punto comienza a crecer.
} La forma en que costes fijos y variables afectan a la producción está relacionada con las esconomías a escala. Los rendimientos a escala se estudian analíticamente mediante la elasticidad-producción-coste (se trata de la escala económica):

\end{aligned}

\paragraph{Representación gráfica: Costes y Economías a escala}
Dividiremos el eje de abcisas en tres Fases, tal y como hicimos para el estudio de la $PMg$ y la $PMe$, con el fin de averiguar que rendimientos existen en cada Fase.

Sobre el plano cartesiano trazamos las lineas de $CT(Q)$, $CV(Q)$ y $CF$. Obtenemos las derivadas de la función: $CMg(Q)$; y también $CTMe(Q)$. Además, podemos también desagregar los $CTMe(Q)$ en dos subtipos: $CVMe(Q)$ y $CFMe(Q)$.
\begin{la}
    \item \textbf{FASE I}. Donde la elasticidad-producto-coste es menor a uno ($CMg<CMe$), hay economías de escala. Esto implica que el coste total varía en menor proporción que la producción (i.e. coste medio decreciente). Esta Fase iría desde el origen hasta el final del punto de inflexión experimentado por la función de $CV(Q)$, es decir, cuando los $CMg$ empiezan a crecer sustancialmente rápdo, $CMg > CV$ (y $CMg < CVMe$);
    \item \textbf{FASE II}. En esta Fase (puntual) la elasticidad-producto-coste es igual a uno ($CMg=CMe$). Esto implica que el coste total varía en la misma proporción que la producción (no hay economía a escala); y,
    \item \textbf{FASE III}. Que se inicia cuando el $CMg > CVMe$, existen deseconomías de escala. Esto implica que el coste total varía en mayor proporción que la producción (i.e. coste medio creciente).
\end{la}

\subsubsection{Corto plazo (C/p) vs. Largo plazo (L/p): Principio de LE CHATELIER-SAMUELSON}
En el largo plazo, la curva de costes totales es la envolvente de todas las curvas de costes totales a corto plazo. Del mismo modo, la curva de costes medios es la envolvente de todas las curvas de costes medios a corto plazo.
En relación a las curvas de $CMg$ a C/p, éstas cortan a las curvas de $CMe$ a C/p por su mínimo. Además, las curvas de $CMg$ a C/p cortan a la curva de $CMg$’s a L/p en el nivel de producción en el que la curva de $CMe$ a C/p es tangente a la $CMe$ a L/p.

Nótese que para el nivel de producción en que el CT a C/p es tangente con el CT a L/p, la curva de CMe a C/p es tangente a la curva de CMe a L/p y además, la curva de CMg a C/p corta a la curva de CMg a L/p.

Otra cuestión que observamos a partir del gráfico es que el tramo creciente de las curvas de CMg es más rígido en el caso C/p que el caso de L/p. Esto es relevante porque en competencia perfecta, el tramo creciente de la curva de CMg a partir del umbral de producción es precisamente la curva de oferta. El fenómeno de la mayor rigidez se explica por el Teorema de LE CHÂTELIER-SAMUELSON, de acuerdo con el cual las funciones de oferta de la empresa precio-aceptante en el C/p son más rígidas que en el L/p (y tanto más rígidas cuanto mayor sea el número de factores considerados fijos) debido a que la capacidad de adaptación de la empresa en el corto plazo es menor al tener algunas dotaciones fijas.\footnote{%

La teoría neoclásica, tal y como ha sido planteada a lo largo de esta exposición hasta este punto, ha sido duramente cuestionada tanto respecto a sus fundamentos teóricos como a sus implicaciones empíricas. Las críticas han partido esencialmente de los autores de la Teoría de la Organización Industrial.

\subsubsection{La Teoría de Costes de STIGLER: la reserva de capacidad}
GEORGE STIGLER (1939) cuestiona la forma en U de los costes medios de la teoría neoclásica, tanto a nivel empírico como teórico. Considera que como existen factores productivos que no son perfectamente divisibles y existe incertidumbre sobre cómo va a ser la demanda, la función de costes va a tener un tramo en el que los costes medios y los costes marginales son constantes. Como consecuencia, no habría un tamaño óptimo, sino que habría una horquilla de tamaños óptimos que el denomina: reserva de capacidad.

\imagenfit{STIGLER_ReservaCapacidad.png}{Reserva de capacidad: STIGLER}{Apuntes Víctor Gutiérrez. Tema 3.A.12}
 
STIGLER argumenta que esto se debe, principalmente a dos razones:
\begin{lnum}
    \item \textbf{Razones económicas}. Por ejemplo, posibles cambios en la demanda del bien debido a razones cíclicas, cambios favorables en los gustos de los consumidores, o como un método preventivo para evitar una parada de la producción en caso de que haya una avería o una reparación.
    \item \textbf{Razones técnicas}. Por ejemplo, el hecho de algunas instalaciones básicas son indivisibles, o que gerentes y trabajadores son contratados en exceso para permitir futuras ampliaciones.
\end{lnum}
	
Hay que señalar que esta teoría permite que la máxima eficiencia exista durante cierto tiempo, lo que difiere completamente de la visión neoclásica, que ve la máxima eficacia como un punto de inflexión, en lugar de una línea recta.

\imagenfit{CMe_LP_STIGLER_ReservaCapacidad.png}{Curva de costes medios a largo plazo. Teoría de Reserva de capacidad. STIGLER}{Apuntes Víctor Gutiérrez. Tema 3.A.12}

Por otro lado, a largo plazo tendríamos una curva de $CMe$ tal que, los costes de producción disminuyen debido a economías de escala hasta que se alcanza el tamaño óptimo, y luego permanecen constante. Una mayor reducción del coste medio de producción podría darse gracias a una mejora de la tecnología o de las habilidades de los trabajadores. En cuanto a los costes de gestión, STIGLER considera que empiezan a aumentar después de que se alcance un cierto tamaño. Sin embargo, las deseconomías de escala de gestión serían compensadas por las economías de escala técnicas, con un efecto neto de suma cero y por lo tanto se mantendrían constantes.

\subsubsection{LEIBENSTEIN: Ineficiencia X}
La teoría anterior cuestiona la forma de U de las curvas de coste medio. Sin embargo, no pone en duda cuestiones fundamentales como la minimización del coste por parte de la empresa, que es precisamente lo que hace HARVEY LEIBENSTEIN (1966).
Una ineficiencia X surge como resultado de que ciertos inputs no den lugar a la mayor cantidad de output correspondiente como consecuencia de cierto factor X. Esto se traduce en un fallo de minimización de costes y maximización de producción e implica una pérdida de eficiencia. Este autor pone en duda el óptimo del productor, partiendo de los siguientes supuestos:
\begin{lnum}
    \item El esfuerzo de uno de los factores de producción es endógeno y constituye un mal para éste. Como postula la ley psicológica de YERKES-DODSON, los individuos que están expuestos a baja presión no pondrán mucho esfuerzo en sus acciones. Según crece la presión, la situación cambia hasta alcanzar el punto en que demasiada presión resulta ineficiente;
    \item Contratos incompletos (i.e. no pueden prever todas las contingencias). Los contratos de empleo definen la compensación económica de los trabajadores pero no otras preocupaciones como el esfuerzo, la carga de trabajo o tareas específicas que no están definidas, lo que crea un vacío en los contratos;
    \item Inercia. Las fuerzas que son externas a la compañía pueden ejercer presión para hacer que la empresa sea más competitiva. Solo con estas fuerzas el esfuerzo aumentará; y,
    \item Prudencia (información asimétrica). Se asume que los trabajadores no mostrarán su pleno potencial y que al mismo tiempo, los empleadores no pagarán el máximo salario que pueden ofrecer.
\end{lnum}

Por todo lo anterior, los trabajadores (que son los que deciden el nivel de esfuerzo a llevar a cabo), escogerán por lo general un grado de esfuerzo inferior al que maximiza beneficios. Así, el comportamiento optimizador (i.e. producción que minimiza el coste medio) es una excepción, produciéndose generalmente una ineficiencia X.

Es muy interesante como esta ineficiencia de X planteada por LEIBENSTEIN se desarrollo más en profundidad en la Economía de la Información [\authorfont{Ver Tema 3.A.13}] Además, el grado de competencia juega un papel fundamental en la teoría de LEIBENSTEIN:
\begin{lnum}
    \item Con competencia alta, los trabajadores, en aras de asegurar la supervivencia de la empresa (y conservar, por tanto, su puesto de trabajo), se esforzarán y mejorarán su productividad marginal;
    \item Con competencia baja, los trabajadores tendrán incentivos a relajares. Así, un aumento de los precios del producto no tiene por qué traducirse en un mayor beneficio, ya que si los trabajadores perciben que ese aumento del precio de venta se debe a un mark-up posibilitado por una menor competencia, entonces el beneficio podría incluso disminuir al aumentar los costes por la ineficiencia X; y,
    \item Sin embargo, pruebas empíricas y otros modelos sobre la relevancia de las ineficiencias X tienen resultados mixtos por lo que no se puede llegar a ninguna conclusión. 
\end{lnum}

\subsection{Análisis envolvente de datos: DEA}
Desarrollado por CHARNES, COOPER y RHODES en 1978, el análisis envolvente de datos surgió como respuesta a las limitaciones de los métodos tradicionales de medición de la eficiencia. Se trata de un método no-paramétrico utilizado en investigación operativa para medir o evaluar la eficiencia de las unidades de toma de decisiones (DMU’s), como empresas, servicios públicos o cualquier entidad que convierta inputs en outputs. El DEA es particularmente útil en situaciones en las que intervienen múltiples inputs y/o outputs, lo que permite una evaluación integral de la eficiencia relativa. Al emplear métodos de programación lineal, el DEA construye una frontera de rendimiento óptimo, lo que permite la comparación de entidades similares.

Aunque aparentemente conduce a la creación de una Frontera de Posibilidades de Producción, en puridad, al ser un proceso de benchmarking la evaluación a través del DEA conduce a la creación de una “Frontera de Mejores Prácticas” sobre la que se evalúa el rendimiento de los DMU’s. 

\subsubsection{Modelo básico}
Se busca asignar un valor de eficiencia a cada DMU en base a los inputs y outputs observados en éste, sujeto a la restricción impuesta por el resto de DMU’s y por otros de consistencia. En concreto, se maximiza el siguiente DMU:

\eqblock{
\begin{aligned}
    &\boxed{\begin{aligned}
        &\max\qquad \theta_j=\frac{\sum_{m=1}^My_m^ju_m^j}{\sum_{n=1}^Nx_n^jv_n^j}\\[1em]
        &\text{s.a.}\qquad 
        \begin{cases}
            \frac{\sum_{m=1}^My_m^ju_m^j}{\sum_{n=1}^Nx_n^jv_n^j}\leq1\qquad \forall k=1,\dots,K\\[1em]
            \text{\scriptsize
            $\begin{aligned}
                &0\ll(y_1^j,y_2^j,\dots, y_M^j)\\
                &0\ll(u_1^j,u_2^j\dots, u_M^j)\\
                &0\ll(x_1^j.x_2^j,\dots,x_N^j)\\
                &0\ll(v_1^j,v_2^j,\dots,v_N^j)
            \end{aligned}$}
        \end{cases}
    \end{aligned}}\\[2em]
    &\left\{\begin{aligned}
        &(y_1^j,y_2^j,\dots, y_M^j)\equiv &&\text{Outputs del proceso productivo}\\
        &(u_1^j,u_2^j\dots, u_M^j)\equiv &&\text{Pesos relativos de cada output producido}\\
        &(x_1^j.x_2^j,\dots,x_N^j)\equiv &&\text{Inputs del proceso productivo}\\
        &(v_1^j,v_2^j,\dots,v_N^j)\equiv &&\text{Pesos relativos de cada input necesario}
    \end{aligned}\right.
\end{aligned}
}
 
Sujeto a la restricción de no-negatividad de inputs, outputs y pesos relativos inferiores o iguales a uno (y mayores o iguales a cero), y una adicional (primera): que la eficiencia de cada sistema tiene que ser menor o igual a uno, en todo caso. Al tratarse de un problema de programación lineal no podremos utilizar cocientes, entonces deberemos reformular las ecuaciones tal que así:

$$\begin{aligned}
    &\max\qquad \theta_j\sum_{m=1}^My_m^ju_m^j\\[0.5em]
    \text{s.a.}\qquad
    &\left\{\begin{aligned}
        &\begin{matrix}
            \sum_{m=1}^My_m^1u_m^1-\sum_{n=1}^Nx_n^1v_n^1\leq0\\[0.5em]
            \vdots\;(\forall j=1,\dots,J)\\[0.5em]
            \sum_{m=1}^My_m^Ju_m^J-\sum_{n=1}^Nx_n^Jv_n^J\leq0\\[0.5em]
        \end{matrix}\\[0.5em]
        &\sum_{m=1}^Mx_m^jv_m^j=1\\[0.5em]
        &0\leq v_n,u_n
    \end{aligned}\right.
\end{aligned}$$

Con estas restricciones obtendremos un ránking de eficiencia (comparativa) y una relación de inputs que defina la frontera eficiente. De forma que:

\imagenfit{Frontera_DEA.png}{Frontera eficiente (\textit{benchmarking}): Análisis Envolvente de Datos (DEA)}{Elaboración propia}
 
Para medir la Ratio de ineficiencia deberíamos hacer ($1 – \frac{\over{(0,A')}}{\over{(0,A)}}$). Además de este resultado, podemos obtener también lo siguiente:
\begin{lnum}
    \item Ponderaciones, o precios de los inputs y outputs (soluciones v y u): indica el peso asignado a cada variable output e input en la formación del ratio de eficiencia;
    \item Reducción proporcional de los inputs, para el caso de los modelos orientados a los inputs. Esto equivaldría a la minimización de la distancia radial desde la unidad ineficiente a la frontera. En estos se busca la eficiencia técnica de la producción. En el modelo orientado a los inputs nos estaríamos moviendo hacía dentro de la nube de datos (hacia el origen); y,
    \item Aumento proporcional de los outputs. El incremento proporcional que debe experimentar la unidad ineficiente en su eficiencia para llegar a la frontera. En el modelo orientado a los outputs nos estaríamos moviendo hacía afuera de la nube de datos (expansión del output).\footnote{%
        La elección del problema de minimización o maximización depende del criterio del analista y de su conocimiento del sector objeto del análisis. En determinados problemas, las empresas no tendrán libertad a la hora de fijar la cantidad de output por lo que será mejor un enfoque de minimización de inputs. Por el contrario, en otros contextos la cantidad de inputs puede venir establecida por una regulación y el operador se concentrará entonces en maximizar su output.
    }
\end{lnum}

\subsubsection{Limitaciones}
A pesar de sus ventajas, el análisis envolvente de datos tiene sus propias limitaciones. Una desventaja importante es su sensibilidad a outliers, lo que puede distorsionar los puntajes de eficiencia y llevar a conclusiones erróneas. Además, DEA no tiene en cuenta la naturaleza estocástica (no-paramétrico) de los datos, lo que significa que supone que todas las desviaciones de la frontera se deben a la ineficiencia y no a la variación aleatoria. Esta limitación puede afectar la confiabilidad de los resultados, en particular en conjuntos de datos con alta variabilidad. Algunas alternativas propuestas para mejorar alguno de sus inconvenientes o potenciar sus ventajas son: el DEA Estocástico  o el Stochastic Frontier Analysis.\footnote{%
    Ver: \href{https://www.sciencedirect.com/science/article/abs/pii/S0377221715007134}{\textit{Stochastic Data Envelopment Analysis—A review} (OLESSEN, 2016)}
}

\subsection{Stochastic Frontier Analysis}
Simultaneamente introducido por AIGNER, LOVELL y SCHMIDT (1977) and MEEUSEN and VAN DEN BROECK (1977), el análisis fontera estocástica es uno de los métodos más utilizados para analizar y medir la eficiencia. Se trata de una sofisticación de los modelos más tradicionales de frontera de producción. 

\subsubsection{Modelo}
Podemos definir el modelo frontera de producción estocástica como:

\eqblock{
\begin{aligned}
    \boxed{y_i=f(\vec{x_i},\vec\beta)\cdot e^{v_i}\cdot E_i}\\[2em]
    \begin{aligned}
        &y,x\equiv \text{Inputs y outputs del productor $i$}\\[0.5em]
        &\underset{\text{\scriptsize Debe venir explícitamente definida, por lo que es una limitación importante}}{f(\vec{x_i})\equiv\text{Función de Producción del productor $i$}}\\[0.5em]
        &\underset{\text{\scriptsize Se trata de los parámetros a estimar (parámetros del modelo)}}{\vec{\beta}\equiv\text{Coeficientes de producción}}\\[0.5em]
        &v_i\equiv\text{Componente aleatorio para productor $i$}\\[0.5em]
        &E_i\equiv \text{Término de eficiencia ténica para productor $i$}
    \end{aligned}
\end{aligned}
}{Análisis por frontera estocástica (verosimilitud): Formulación del modelo general. Aplicaciones empíricas: Métodos de estimación}

El componente aleatorio puede eliminarse del modelo, igual que el componente de eficiencia técnica puede expresarse en forma exponencial (negativa, ya que estará acotado entre 0 y 1). De hecho, si suponemos que la forma funcional es del tipo COBB-DOUGLAS, formularemos la expresión en su forma logarítmica tal que:

\eqblock{
\begin{aligned}
    &\ln{y_i}=\overbrace{\beta_0+\sum_{n=1}^N\beta_nx_{i,n}+v_i}^{\text{Frontera estocástica}}-u_i\qquad \text{y,}\qquad \varepsilon_i=v_i-u_i\\[0.5em]
    &\Longrightarrow\qquad \boxed{\ln{y_i}=\beta_0+\sum_{n=1}^N\beta_nx_{i,n}+\varepsilon_i}
\end{aligned}
}{Análisis por frontera estocástica (verosimilitud): Formulación para forma funcional COBB-DOUGLAs. Aplicaciones empíricas: Métodos de estimación}

Ahora bien, esto es solo el inicio del problema, es decir, no pretendemos estimar los parámetros mediante MCO’s sino que nuestra intención será encontrar la distribución del error compuesto, ajustando los parámetros, de tal forma que se ajuste lo máximo posible a los datos obtenidos, siempre asumiendo o teniendo:
\begin{lnum}
    \item $v$ es una variable de ruido aleatorio (captura errores de medición, clima, suerte, etc.) sobre la que se asume (generalmente) una distribución $v\sim N(0,\sigma_v)$; y,
    \item $u$ corresponde a la ineficiencia técnica (captura cuánto se desvía la empresa de la frontera estocástica), para esta se asumirá, por norma general, una distribución $u\sim N(0,\sigma_u)\;(+)$ (solo la parte positiva, pues la ineficiencia no puede ser negativa).
\end{lnum}
	
Para conseguirlo, el método más utilizado es el de Máxima Verosimiliud  que, usualmente, es el que proporciona resultados más eficientes si la suposición de independencia tanto de los efectos como de los regresores es plausible y tiene sentido económico. 

\begin{specialblock}{Método de Máxima Verosimilitud}
    La función de verosimilitud (o, simplemente, verosimilitud) es una función de los parámetros de un modelo estadístico (por MCO’s por ejemplo) que permite realizar inferencias acerca de su valor a partir de un conjunto de observaciones. En cierto sentido, la verosimilitud es una versión inversa de la probabilidad condicional.
    
    Vamos a seguir el método de Máxima Verosimilitud por lo siguiente: en el modelo planteado lo que nos interesa obtener es el término de ineficiencia, que es lo verdaderamente interesante. Ahora bien, la introducción de ese término altera la distribución “generalmente aceptada” del término de error (la Normal) puesto que, al ser ahora un error compuesto, el error ya no es simétrico por ser la diferencia entre un ruido normal y un término positivo (la ineficiencia): su distribución será ahora asimétrica a la izquierda, entonces, el MCO’s ya no servirá, necesitaremos un método que ajuste esa distribución asimétrica $\to$ La Máxima Verosimilitud.
\end{specialblock}

El Método de Máxima Verosimilitud no estima directamente una recta por MCO’s, sino que busca los parámetros (parámetros de la función y varianzas tanto del error ruido y del error ineficiencia, cada uno por separado) que maximizan la probabilidad de observar los datos de nuestra muestra:

$$\begin{aligned}
    L(\beta,\sigma_v,\sigma_u)=\Pi_{i=1}^nf_\varepsilon(\varepsilon_i;\;\sigma_v,\sigma_u)
\end{aligned}$$
 
Donde $f_\varepsilon$ es la función de densidad del error compuesto, derivada de las distribuciones asumidas de los dos componentes. Mediante integración, obtendremos una función de densidad no simétrica que será la función de verosimilitud individual para cada observación. 

Como las observaciones son independientes, la función de verosimilitud total se obtendrá de:

$$\begin{aligned}
    L(\beta,\sigma_v,\sigma_u)=\Pi_{i=1}^nf(\varepsilon_i)
\end{aligned}$$
 
Y, si aplicamos logarítmos, trabajaremos con el sumatorio de funciones individuales. A partir de aquí, el método de Máxima Verosimilitud ajusta los parámetros mediante ciclos iterativos hasta que la log-verosimilitud alcance su máximo (lo que implicaría que la probabilidad de obtener los resultados de nuestra muestra dados esos parámetros es la más alta, sería como una reversión del proceso de MCO’s). El modelo final nos entregará:
\begin{lnum}
    \item Las betas, elasticidades de los inputs;
    \item Las varianzas de ineficiencia y ruida; y,
    \item Y las medidas individuales de eficiencia, esto es:
    $$\mathbb E[u_i\;|\;\varepsilon_i]$$
    Esto es lo que realmente nos interesa, puesto que, para calcular la eficiencia técnica de cada empresa solo tendremos que hacer laexponencial a la medida anterior. Y, además, podremos calcular el de cualquier otra que queramos ya que esto consiste en: la ineficiencia esperada de la empresa dado el ruido observado en la regresión. 
\end{lnum}

\anexo{La elección de la empresa: costes, beneficio y dualidad}\label{anx:eleccion}

La teoría de la producción describe lo que la empresa puede hacer; la de los costes, lo que le cuesta hacerlo y, con ello, lo que elige. Este anexo estudia esa elección cuando la empresa toma como dados los precios de los factores: qué coste importa, qué objetivo persigue la empresa y por qué su decisión puede descomponerse en dos etapas, cómo se obtiene la combinación de factores de mínimo coste y qué propiedades tiene la función de costes que resulta, qué añade la maximización del beneficio y qué relaciones de dualidad unen la tecnología con las funciones de costes y de beneficios. Va fuera del cuerpo porque el primer bloque recoge sus resultados; aquí se demuestran, se discuten sus condiciones y se completan con las formulaciones alternativas del problema. Su tesis es que la función de costes contiene toda la información sobre la tecnología que importa a una empresa que toma como dados los precios de los factores, y que por eso la teoría de los costes es más general que la del beneficio: sirve con cualquier estructura del mercado del producto y con cualquier tipo de rendimientos a escala.\footnote{%
    Con dos factores, trabajo, $L$, y capital, $K$, sus precios son el salario, $w$, y el coste de uso del capital, $r$; con $n$ factores, $z=(z_1,\dots,z_n)$, con precios $w=(w_1,\dots,w_n)$. El producto es $q$, y su precio, $p$. La tecnología se representa con una función de producción $f(z)$ continua, no decreciente y cuasicóncava, con las propiedades que se derivan del conjunto de producción ([\authorfont{Ver Tema 3.A.11}]).
}

\subsection{El coste económico: ¿qué cuesta producir?}\label{anx:coste}
Para elegir, la empresa compara lo que obtiene con lo que sacrifica, y la teoría mide ese sacrificio con el coste de oportunidad, no con el gasto que registra la contabilidad. La distinción decide qué beneficio es el que se maximiza y qué costes deben pesar en cada decisión.

\subsubsection{Coste de oportunidad y beneficio económico}
El \textbf{coste de oportunidad} de emplear una unidad de un factor es el valor de la mejor alternativa a la que se renuncia por emplearla. Si la empresa no dispone de esa unidad, tiene que comprarla o alquilarla, y su coste de oportunidad es el precio de mercado. Si ya dispone de ella, no hace ningún desembolso, pero renuncia a lo que obtendría vendiéndola o alquilándola, de modo que su coste de oportunidad vuelve a ser el precio de mercado. Con mercados de factores competitivos, el coste de oportunidad de cada factor es, en los dos casos, su precio.\footnote{%
    La expresión ``coste de oportunidad'' aparece en GREEN (1894), en el debate con quienes medían el coste por el esfuerzo y el sacrificio de los que producen, la doctrina del coste real. BUCHANAN (1969) subrayó que el coste de oportunidad es subjetivo: lo soporta quien elige, en el momento de elegir, y se refiere a una alternativa que nunca llega a realizarse, de modo que ningún observador puede medirlo después.
}

El coste contable registra los pagos efectivos a terceros: salarios, compras, alquileres e intereses. El \textbf{coste económico} añade los costes implícitos de los factores que aporta el propio empresario: el interés que obtendría su capital en otro empleo, el salario al que renuncia por dirigir la empresa o el alquiler que cobraría por el local que usa. La diferencia entre los ingresos y el coste económico es el \textbf{beneficio económico}. Cuando es nulo, los ingresos remuneran todos los factores a su coste de oportunidad, y el beneficio contable que queda es el \textbf{beneficio normal}, el que basta para que los recursos sigan en la empresa. Un beneficio económico nulo no indica, por tanto, que la empresa vaya mal, sino que gana lo mismo que en su mejor alternativa, que es la situación del equilibrio competitivo a largo plazo ([\authorfont{Ver Tema 3.A.16}]).

\subsubsection{Costes fijos, variables y hundidos}
Los costes se clasifican también por su relación con la cantidad producida y por la posibilidad de evitarlos. Los \textbf{costes variables} dependen del nivel de producción, como los de las materias primas o el trabajo que se paga por hora o por pieza; los \textbf{costes fijos} no dependen de él, como el alquiler de la planta o los gastos generales de administración. Solo hay costes fijos cuando algún factor no puede ajustarse, es decir, a corto plazo: a largo plazo todos los costes son variables, como veremos al estudiar los plazos.\footnote{%
    Entre unos y otros están los costes \textbf{cuasifijos}, que no varían con la cantidad producida pero se evitan si la producción se interrumpe, como la calefacción mínima de una nave.
}

La distinción que importa para decidir es otra: la que separa los costes \textbf{evitables} de los \textbf{hundidos}. Un coste está hundido cuando ya se ha comprometido y no puede recuperarse, como una inversión específica que no tiene otro uso o una campaña de publicidad ya realizada; un coste fijo, en cambio, puede no estar hundido, si el factor que lo origina puede venderse o subarrendarse. Los costes hundidos no deben influir en las decisiones, porque nada de lo que se haga los cambia: solo cuentan los costes que todavía pueden evitarse. La regla explica por qué una empresa puede seguir produciendo con pérdidas a corto plazo, mientras cubra sus costes evitables, y por qué los costes hundidos son una barrera a la entrada que condiciona la competencia en el mercado ([\authorfont{Ver Tema 3.A.15, Tema 3.A.16}]).

\subsection{El objetivo de la empresa: ¿por qué minimizar costes?}\label{anx:objetivo}
La teoría supone que la empresa maximiza su beneficio. Ese objetivo puede descomponerse en dos etapas, y la primera, producir cada cantidad al menor coste posible, no depende del mercado en que la empresa vende ni de que su objetivo sea exactamente el beneficio. Esa es la razón de que la teoría de los costes tenga entidad propia.

\subsubsection{El beneficio como objetivo}
La maximización del beneficio se justifica por quién decide y por la competencia. El beneficio es la renta residual de los propietarios, que controlan la empresa y se quedan con lo que resta tras pagar a los demás factores, de modo que es lo que ellos quieren maximizar. Y la competencia selecciona: ALCHIAN (1950) sostuvo que, aunque las empresas no calculen ningún óptimo, el mercado elimina a las que obtienen pérdidas y expande a las que obtienen beneficios, de modo que sobreviven las que se comportan como si maximizaran. FRIEDMAN (1953) añadió que una hipótesis se juzga por sus predicciones y no por el realismo de sus supuestos, como el buen jugador de billar, que no conoce las leyes de la física pero juega como si las aplicara.

El supuesto se ha discutido. Cuando la propiedad y el control están separados, como en las grandes sociedades anónimas, los directivos pueden perseguir sus propios objetivos: BAUMOL (1959) propuso que maximizan las ventas sujetas a un beneficio mínimo, y otros autores, el crecimiento de la empresa o su propia utilidad ([\authorfont{Ver Tema 3.A.15}]). Para la teoría de los costes, sin embargo, la discusión importa poco. Cualquier objetivo que valore el producto y no valore el gasto por sí mismo exige producir cada cantidad al menor coste posible: la empresa que maximiza sus ventas también minimiza el coste de lo que vende, y el organismo público que maximiza su producción con un presupuesto dado también lo gasta en la combinación de factores más barata. \textbf{La minimización del coste es una condición más general que la maximización del beneficio}, y solo deja de cumplirse cuando la empresa no tiene incentivos para buscarla, como en la ineficiencia X que veremos en el tercer anexo.

\subsubsection{Las dos etapas de la decisión}
Una empresa que toma como dados el precio de su producto, $p$, y los de sus factores, $w$, elige a la vez cuánto producir y con qué factores, en el problema de \textbf{maximización del beneficio en una etapa}, que en la formulación de VARIAN (1992) y MAS-COLELL, WHINSTON y GREEN (1995) es:

$$\max_{q,\,z\ge0}\ \ p\,q-w\cdot z\qquad \text{s.a.}\qquad q\le f(z).$$

El problema puede resolverse en dos etapas. En la primera, la empresa calcula, para cada cantidad $q$, el menor coste con que puede obtenerla, que es la \textbf{función de costes}, en la misma formulación:

$$C(w,q)=\min_{z\ge0}\ \ w\cdot z\qquad \text{s.a.}\qquad f(z)\ge q.$$

En la segunda, elige la cantidad que maximiza el beneficio, $\max_q\ p\,q-C(w,q)$. Las dos formulaciones llevan al mismo resultado. Para cualquier plan factible, $w\cdot z\ge C(w,q)$, de modo que su beneficio no supera $p\,q-C(w,q)$; y el máximo de esta expresión se alcanza con la combinación de factores que minimiza el coste de producir $q$. De ahí se sigue el resultado que da sentido a la teoría de los costes: \textbf{toda combinación de factores que maximiza el beneficio minimiza el coste de producir la cantidad elegida}.

La descomposición separa lo que depende de cada mercado. La primera etapa solo exige que la empresa tome como dados los precios de los factores, y es la misma tanto si vende en competencia como si es un monopolio: la función de costes resume la tecnología y los precios de los factores, y nada más. La segunda etapa es la que depende del mercado del producto. La empresa precio-aceptante produce donde el precio iguala el coste marginal, $p=\partial C(w,q)/\partial q$, en un tramo en que este crece; el monopolista, donde el ingreso marginal iguala el coste marginal ([\authorfont{Ver Tema 3.A.16, Tema 3.A.17}]). Si la empresa tiene poder en el mercado de un factor, el precio de ese factor depende de la cantidad que contrata y la primera etapa cambia: el coste de una unidad más no es su precio, sino el coste marginal del factor, como en el monopsonio ([\authorfont{Ver Tema 3.A.17}]).

\subsubsection{Tres formulaciones del problema}
La teoría formula de tres maneras la elección de la empresa, y conviene distinguirlas porque no exigen lo mismo ni sirven para lo mismo. La primera es la maximización del beneficio en una etapa, que elige a la vez la cantidad y los factores y da la función de beneficios; exige que la empresa tome como dado también el precio del producto y que los rendimientos sean decrecientes, porque con rendimientos constantes o crecientes el problema puede no tener solución. La segunda es la descomposición en dos etapas, que coincide con la primera cuando esta tiene solución y sirve también cuando no la tiene, porque la función de costes existe sean cuales sean los rendimientos. La tercera es la maximización de la producción con un coste dado, $\max_z f(z)$ s.a. $w\cdot z\le\bar C$, que SEGURA (1986) presenta junto a la minimización del coste. Es su dual matemático exacto, como la maximización de la utilidad lo es de la minimización del gasto ([\authorfont{Ver Tema 3.A.9}]), pero describe a un productor con un presupuesto fijo, como un organismo público, más que a una empresa que elige cuánto gastar.

Las tres llevan a la misma combinación de factores para una misma cantidad producida, porque todas exigen la tangencia entre la isocuanta y la recta de coste que veremos a continuación. Difieren en la información que necesitan y en las condiciones para que exista solución: la minimización del coste solo necesita la tecnología y los precios de los factores, y por eso es el núcleo de la teoría. La maximización del beneficio en una etapa se estudia más adelante, con la función de beneficios, y la de la producción con un coste dado, con el dual estricto.

\subsection{La minimización del coste: ¿cómo se elige la combinación de factores?}\label{anx:minimizacion}
La primera etapa de la decisión es un problema de elección con restricción, como el del consumidor que minimiza su gasto: entre todas las combinaciones de factores que permiten obtener una cantidad dada, la empresa elige la más barata. Su solución da la regla de la tangencia, las demandas condicionadas de factores y la medida de la eficiencia económica, y su condición de segundo orden explica por qué la teoría necesita isocuantas convexas.

\subsubsection{El problema y la condición de tangencia}
El problema, que planteó SAMUELSON (1947) y estudió a fondo SHEPHARD (1953), es el de la primera etapa:

$$\min_{z\ge0}\ \ w\cdot z\qquad\text{s.a.}\qquad f(z)\ge q.$$

Con dos factores, las combinaciones con un mismo coste $C$, $wL+rK=C$, forman una \textbf{recta isocoste} de pendiente $-w/r$, y las de coste mayor quedan más lejos del origen. La empresa busca la isocoste más baja que toca la isocuanta de nivel $q$. Si la solución es interior, el lagrangiano $\mathcal L=w\cdot z-\mu\,[f(z)-q]$ da las condiciones de primer orden, en la formulación de MAS-COLELL, WHINSTON y GREEN (1995):

$$w_i=\mu\,f_i(z^*)\quad (i=1,\dots,n),\qquad\qquad f(z^*)=q,$$

de las que se obtiene, dividiendo dos a dos, la condición de tangencia:

$$\frac{f_i(z^*)}{f_j(z^*)}=\frac{w_i}{w_j},\qquad\text{y, con dos factores,}\qquad RMST_{LK}=\frac{PMg_L}{PMg_K}=\frac{w}{r}.$$

En el óptimo, la pendiente de la isocuanta, que mide a qué tasa la tecnología permite sustituir un factor por otro, iguala la de la isocoste, que mide a qué tasa lo permite el mercado. Si no fuera así, por ejemplo si $PMg_L/w>PMg_K/r$, el último euro gastado en trabajo daría más producto que el último gastado en capital, y trasladar gasto del capital al trabajo permitiría obtener lo mismo con menos coste. La condición equivale a la \textbf{regla del último euro}, en la formulación habitual (VARIAN, 1992):

$$\frac{PMg_1}{w_1}=\frac{PMg_2}{w_2}=\dots=\frac{PMg_n}{w_n}=\frac{1}{\mu}.$$

El multiplicador, $\mu$, tiene un significado preciso: es lo que aumenta el coste mínimo cuando se exige una unidad más de producto, es decir, el \textbf{coste marginal}, como demostraremos al estudiar la función de costes. La figura muestra la solución. En $E$, la isocoste $C_1$ es tangente a la isocuanta; las combinaciones $A$ y $B$ obtienen el mismo producto con un coste mayor, $C_2$, y ninguna combinación con un coste menor, como $C_0$, permite obtenerlo.

\subsubsection{Soluciones de esquina}
La tangencia caracteriza las soluciones interiores, en las que se emplean todos los factores. En general, con las restricciones de no negatividad, las condiciones de KUHN y TUCKER son, para cada factor y en la misma formulación:

$$w_i-\mu\,f_i(z^*)\ge0,\qquad\quad z_i^*\ge0,\qquad\quad z_i^*\,\big[w_i-\mu\,f_i(z^*)\big]=0.$$

Un factor no se emplea cuando su precio supera lo que vale, a coste marginal, su productividad marginal, $\mu f_i$: emplearlo encarecería la producción. Los dos casos extremos de la tecnología lo ilustran. Con \textbf{sustitutos perfectos}, $q=aL+bK$, la isocuanta es una recta y, salvo que su pendiente coincida con la de la isocoste, la solución es una esquina: la empresa emplea solo el factor más barato por unidad de producto, el trabajo si $w/a<r/b$ y el capital en caso contrario, como en el panel (a) de la figura, en el que la isocoste que pasa por la esquina $E$ es más baja que la que pasa por $B$. Con \textbf{proporciones fijas}, $q=\min\{L/a,K/b\}$, la isocuanta forma un ángulo recto y la solución está siempre en el vértice, $L=aq$ y $K=bq$, sean cuales sean los precios (panel b): la empresa no sustituye un factor por otro, y su coste es $C(w,r,q)=(aw+br)\,q$. Con varios procesos de proporciones fijas, la isocuanta es quebrada, y los cambios en los precios relativos llevan de un vértice a otro de golpe, como vimos con el análisis de actividades ([\authorfont{Ver Tema 3.A.11}]).

\subsubsection{La condición de segundo orden: el hesiano orlado}
La tangencia es necesaria, pero no suficiente, porque también una isocuanta cóncava puede ser tangente a una isocoste. Lo decide la condición de segundo orden: para un mínimo con una restricción, el determinante del hesiano orlado del lagrangiano debe ser negativo, en la formulación de las condiciones de segundo orden de SAMUELSON (1947). Con dos factores:

$$|\bar H|=\begin{vmatrix}0&f_L&f_K\\ f_L&-\mu f_{LL}&-\mu f_{LK}\\ f_K&-\mu f_{LK}&-\mu f_{KK}\end{vmatrix}=\mu\,\big(f_{LL}\,f_K^2-2f_{LK}\,f_L\,f_K+f_{KK}\,f_L^2\big)<0.$$

Como $\mu>0$, la condición exige que el paréntesis sea negativo, y ese paréntesis, cambiado de signo y dividido por $f_K^3$, es la curvatura de la isocuanta, $d^2K/dL^2$ ([\authorfont{Ver Tema 3.A.11}]). La condición de segundo orden dice, por tanto, que \textbf{la isocuanta debe ser estrictamente convexa en el punto de tangencia}: el hesiano orlado es la expresión algebraica de cómo se curva la isocuanta respecto de la isocoste. Con más factores, la condición se extiende a los menores principales orlados de orden superior, que deben ser todos negativos, y equivale a que la función de producción sea estrictamente cuasicóncava en el entorno del óptimo.

La figura lo muestra. Con una isocuanta convexa (panel a), la isocoste que pasa por el punto de tangencia, $E$, deja toda la isocuanta por encima: cualquier otra combinación que obtenga el mismo producto está en una isocoste mayor, y $E$ es el mínimo. Con una isocuanta cóncava (panel b), la isocoste tangente deja la isocuanta por debajo: alejarse de $E$ a lo largo de la isocuanta abarata la producción, de modo que $E$ es el máximo del coste y el mínimo se alcanza en una esquina, $A$ o $B$. El panel (c) dibuja el coste de obtener la misma cantidad a lo largo de cada isocuanta, $C(L)=wL+rK(L)$: con la isocuanta convexa tiene forma de U y su mínimo está en la tangencia; con la cóncava, forma de U invertida, y su máximo está en ella. Sin isocuantas convexas, la condición de tangencia señala el peor plan, y la combinación de mínimo coste salta de una esquina a otra cuando cambian los precios.

\subsubsection{Las demandas condicionadas de factores}
La solución del problema, para cada vector de precios y cada cantidad, son las \textbf{demandas condicionadas de factores}, $z_i^*=h_i(w,q)$, llamadas así porque dependen de la cantidad que se quiere producir, que la empresa toma como dada en esta etapa; las $n$ funciones forman el sistema de demandas condicionadas. Son el análogo de las demandas compensadas del consumidor ([\authorfont{Ver Tema 3.A.9}]) y tienen tres propiedades que se siguen directamente del problema. Son \textbf{homogéneas de grado cero en los precios de los factores}: si todos se multiplican por la misma constante, la pendiente de la isocoste no cambia y tampoco la combinación elegida, de modo que solo importan los precios relativos. Son \textbf{únicas} si la isocuanta es estrictamente convexa; con isocuantas convexas pero no estrictamente, como la recta de los sustitutos perfectos cuando su pendiente coincide con la de la isocoste, la solución es un conjunto de combinaciones igual de baratas. Y son \textbf{continuas} en los precios y en la cantidad cuando la solución es única, por el teorema del máximo. Una cuarta propiedad, que la demanda condicionada de un factor no aumenta cuando sube su precio, se demuestra con la función de costes, y la forma en que cambian cuando aumenta la cantidad producida, la senda de expansión, se estudia con los costes a largo plazo.

\subsubsection{Eficiencia técnica y eficiencia asignativa}
La minimización del coste define el patrón con que se mide la eficiencia de una empresa observada. FARRELL (1957), a partir de DEBREU (1951), separó dos fuentes de ineficiencia, como muestra la figura con una empresa que obtiene la cantidad $q$ con la combinación $P$. La primera es la \textbf{ineficiencia técnica}: $P$ está por encima de la isocuanta, y podría obtenerse lo mismo reduciendo en la misma proporción todos los factores hasta $Q$; su eficiencia técnica es $ET=OQ/OP$, la medida radial que vimos con la frontera de producción ([\authorfont{Ver Tema 3.A.11}]). La segunda es la \textbf{ineficiencia asignativa}: aunque fuera técnicamente eficiente, en $Q$ la empresa emplearía una proporción de factores que no es la de mínimo coste a esos precios, la de $E$, y podría reducir su coste hasta el de la isocoste que pasa por $E$, que corta el radio en $R$; su eficiencia asignativa es $EA=OR/OQ$. Su producto es la \textbf{eficiencia económica}, o eficiencia en costes, que en la descomposición de FARRELL (1957) es el cociente entre el coste mínimo y el coste efectivo:

$$EE=\frac{OR}{OP}=ET\times EA=\frac{C(w,q)}{w\cdot z_P}.$$

\includegraphics{anx_farrell_economica.pdf}\par\vspace{0.3em}

La eficiencia técnica depende de la tecnología y de cómo se usa; la asignativa, de cómo responde la empresa a los precios. Una empresa puede ser técnicamente eficiente y asignativamente ineficiente si los precios que tiene en cuenta no son los de mercado, como la empresa regulada a la que se garantiza una rentabilidad sobre su capital y que, por eso, emplea más capital del que minimiza el coste (AVERCH y JOHNSON, 1962; [\authorfont{Ver Tema 3.A.20}]). La medición de una y otra con datos de empresas es el objeto de los métodos de frontera que veremos en el tercer anexo.

\subsection{La función de costes: ¿qué información contiene?}\label{anx:funcion_costes}
Sustituyendo las demandas condicionadas en el gasto se obtiene la \textbf{función de costes}, el menor coste con que pueden obtenerse $q$ unidades a los precios $w$, en la definición de SHEPHARD (1953):

$$C(w,q)=w\cdot h(w,q)=\sum_i w_i\,h_i(w,q).$$

Sus propiedades no dependen de la forma de la tecnología, sino solo de que la empresa minimiza el coste, y de ellas se deducen, sin resolver ningún problema de optimización, las demandas condicionadas y la respuesta de la empresa a los precios. Por eso la función de costes es la pieza central de la teoría: resume lo que la empresa necesita saber de su tecnología para elegir.

\subsubsection{Propiedades respecto de los precios y lema de SHEPHARD}
La función de costes tiene tres propiedades respecto de los precios de los factores, que se demuestran comparando el plan óptimo con los que la empresa podría elegir. Es \textbf{no decreciente}: si sube el precio de algún factor, cualquier plan cuesta al menos lo mismo que antes, y también el más barato. Es \textbf{homogénea de grado uno}: si todos los precios se multiplican por $t$, la combinación elegida no cambia, porque las demandas condicionadas son homogéneas de grado cero, y el coste se multiplica por $t$; los precios relativos deciden la combinación, y su nivel, solo el coste. Y es \textbf{cóncava}: para unos precios intermedios, $w''=\lambda w+(1-\lambda)w'$, la combinación óptima $z''$ también es factible con los otros dos vectores de precios, aunque no sea la más barata con ellos, de modo que, en la demostración de MAS-COLELL, WHINSTON y GREEN (1995):

$$C(w'',q)=\lambda\,w\cdot z''+(1-\lambda)\,w'\cdot z''\ \ge\ \lambda\,C(w,q)+(1-\lambda)\,C(w',q).$$

La concavidad tiene una lectura económica, que muestra la figura con el precio del factor 1. Si la empresa no cambiara su combinación de factores cuando sube $w_1$, su coste crecería en línea recta, con una pendiente igual a la cantidad que emplea de ese factor: es el coste de una empresa pasiva, la recta discontinua. Pero la empresa sustituye el factor encarecido por los demás, y su coste crece menos; cuando $w_1$ baja, emplea más del factor abaratado, y su coste baja más que el de la empresa pasiva. Por eso la función de costes queda siempre por debajo de esa recta y la toca en $w_1^0$: es cóncava, y tanto más curvada cuanto más fácil es la sustitución entre factores ([\authorfont{Ver Tema 3.A.11}]).

\end{minipage}\end{center}

La figura contiene también el resultado más útil de la teoría, el \textbf{lema de SHEPHARD} (1953): la pendiente de la función de costes respecto del precio de un factor es la demanda condicionada de ese factor,

$$\frac{\partial C(w,q)}{\partial w_i}=h_i(w,q).$$

Se demuestra derivando $C=w\cdot h(w,q)$: $\partial C/\partial w_i=h_i+\sum_j w_j\,\partial h_j/\partial w_i$, y el segundo término se anula, porque por las condiciones de primer orden $w_j=\mu f_j$, y la producción no cambia al reajustar los factores, de modo que $\sum_j f_j\,\partial h_j/\partial w_i=0$. Es una aplicación del \textbf{teorema de la envolvente}: ante un cambio pequeño de un precio, el efecto de reajustar los factores sobre el coste es de segundo orden, porque la combinación ya era óptima, y solo cuenta el efecto directo, la cantidad empleada por la variación del precio. Su consecuencia práctica es grande: si se conoce o se estima la función de costes, las demandas de factores se obtienen derivándola, sin necesidad de conocer la función de producción, y en eso se basa la estimación de las demandas de factores que veremos en el tercer anexo.

\subsubsection{El coste marginal y la cantidad producida}
Respecto de la cantidad, la función de costes es \textbf{no decreciente}, porque exigir más producto reduce el conjunto de combinaciones de factores que lo permiten, y creciente si la producción es estrictamente creciente en los factores. Su derivada es el \textbf{coste marginal}, y el teorema de la envolvente aplicado al lagrangiano demuestra lo que adelantamos al estudiar la minimización: el coste marginal es el multiplicador de la restricción de producción, en la formulación de MAS-COLELL, WHINSTON y GREEN (1995):

$$CMg(w,q)=\frac{\partial C(w,q)}{\partial q}=\mu=\frac{w_i}{PMg_i}\qquad (i=1,\dots,n).$$

El coste marginal es el precio de un factor dividido por su productividad marginal: lo que cuesta obtener una unidad más de producto con más de ese factor. En el óptimo es el mismo sea cual sea el factor que se aumente, porque si fuera más barato ampliar la producción con un factor que con otro, la combinación no sería la de mínimo coste. Cómo depende de $q$ la forma de la función de costes, a corto y a largo plazo, y su relación con los rendimientos a escala son el objeto del segundo anexo.

\subsubsection{La matriz de sustitución}
Las segundas derivadas de la función de costes respecto de los precios forman la \textbf{matriz de sustitución}, que, por el lema de SHEPHARD, recoge las respuestas de las demandas condicionadas a los precios:

$$S=\left(\frac{\partial^2C(w,q)}{\partial w_i\,\partial w_j}\right)=\left(\frac{\partial h_i(w,q)}{\partial w_j}\right).$$

Hereda tres propiedades de la función de costes (MAS-COLELL, WHINSTON y GREEN, 1995). Es \textbf{simétrica}, porque el orden de derivación no importa: el efecto del precio del factor $j$ sobre la demanda condicionada del factor $i$ es igual al del precio de $i$ sobre la demanda de $j$, un resultado nada evidente que se sigue de la optimización. Es \textbf{semidefinida negativa}, por la concavidad, de modo que sus elementos diagonales no son positivos: $\partial h_i/\partial w_i\le0$, la \textbf{ley de la demanda condicionada}. A diferencia del consumidor, cuya demanda ordinaria de un bien de GIFFEN aumenta con su precio ([\authorfont{Ver Tema 3.A.9}]), la demanda condicionada de un factor nunca aumenta cuando sube su precio, porque no hay efecto renta: la cantidad producida está fija. Y cumple $S\,w=0$, por la homogeneidad de grado cero de las demandas: $\sum_j w_j\,\partial h_i/\partial w_j=0$. Con dos factores, esa condición y la ley de la demanda obligan a que $\partial h_K/\partial w\ge0$: los dos factores son \textbf{sustitutos netos}, y en general cada factor tiene al menos un sustituto neto. Las elasticidades de sustitución de ALLEN-UZAWA y de MORISHIMA, que generalizan la de HICKS a más de dos factores, se definen precisamente con estas derivadas de la función de costes ([\authorfont{Ver Tema 3.A.11}]) y se estiman con ellas, como veremos en el tercer anexo.

Por último, el lema de SHEPHARD da una propiedad sencilla de la respuesta del coste a los precios: la elasticidad del coste respecto del precio de un factor es la participación de ese factor en el coste,

$$\varepsilon_{C,w_i}=\frac{\partial C}{\partial w_i}\,\frac{w_i}{C}=\frac{w_i\,h_i(w,q)}{C(w,q)}=s_i.$$

Si sube un $10\%$ el precio de un factor que supone la quinta parte del coste, el coste sube, en primera aproximación, un $2\%$, y algo menos si la empresa sustituye. El coste medio tiene la misma elasticidad, porque la cantidad no cambia, de modo que la curva de coste medio se desplaza hacia arriba en esa proporción, que puede variar con el nivel de producción si lo hace la participación del factor.

\subsection{La maximización del beneficio: ¿qué añade el mercado del producto?}\label{anx:beneficio}
La segunda formulación elige a la vez la cantidad y los factores. Exige más que la minimización del coste, porque la empresa debe tomar como dado también el precio de su producto y su tecnología debe tener rendimientos decrecientes, al menos donde produce; a cambio, determina cuánto produce la empresa y da una segunda función dual, la de beneficios.

\subsubsection{Condiciones de primer y segundo orden}
El problema es elegir los factores que maximizan el beneficio, con la producción dada por la tecnología, en la formulación de VARIAN (1992):

$$\max_{z\ge0}\ \ p\,f(z)-w\cdot z.$$

Sus condiciones de KUHN y TUCKER son $p\,f_i(z^*)-w_i\le0$, $z_i^*\ge0$ y $z_i^*\,[p\,f_i(z^*)-w_i]=0$: si se emplea un factor, el \textbf{valor de su productividad marginal} iguala su precio, $p\,PMg_i=w_i$, porque si fuera mayor convendría contratar más y si fuera menor, menos. Dividiendo dos a dos se obtiene la misma tangencia que en la minimización del coste, $PMg_i/PMg_j=w_i/w_j$, como exige el resultado de las dos etapas, y combinando las condiciones con $CMg=w_i/PMg_i$ se obtiene la regla de la segunda etapa, $p=CMg$.

Con un solo factor, la figura lo muestra en el plano del factor y el producto. Las \textbf{rectas de isobeneficio}, $q=\pi/p+(w/p)\,z$, reúnen los planes con el mismo beneficio; tienen pendiente $w/p$ y cortan el eje vertical en $\pi/p$, de modo que las más altas dan más beneficio. La empresa elige el plan de la función de producción situado en la recta más alta que la toca, $E$, donde la pendiente de la función, la productividad marginal, iguala la de las rectas.

La condición de segundo orden exige que el hesiano del beneficio, que es $p$ por el hesiano de la función de producción, sea semidefinido negativo: la función de producción debe ser cóncava en el entorno del óptimo, con productividades marginales decrecientes y rendimientos decrecientes en ese entorno. Expresada en la segunda etapa, la condición es que el coste marginal sea creciente en el óptimo, $\partial CMg/\partial q>0$. Es más exigente que la de la minimización del coste, que solo necesita isocuantas convexas, compatibles con cualquier tipo de rendimientos a escala.

\subsubsection{Cuándo no hay solución}
Que la condición de segundo orden sea más exigente tiene una consecuencia importante: hay tecnologías para las que la función de costes existe y la maximización del beneficio no tiene solución finita. La figura lo muestra comparando el ingreso, $pq$, con el coste, $C(q)$, de una empresa precio-aceptante según sus rendimientos.

\begin{center}\begin{minipage}{\linewidth}\centering
{\small\textbf{El beneficio máximo según los rendimientos a escala}}\par\vspace{0.3em}
\includegraphics{anx_beneficio_rendimientos.pdf}\par\vspace{0.3em}
{\small\textit{Fuente: }elaboración propia.}
\end{minipage}\end{center}

Con \textbf{rendimientos decrecientes} (panel a), el coste es convexo y el beneficio, la distancia vertical entre las dos curvas, es máximo en $q^*$, donde la pendiente del coste, el coste marginal, iguala el precio. Con \textbf{rendimientos constantes} (panel b), el coste es proporcional a la cantidad, $C=c(w)\,q$, y todo depende de comparar el precio con el coste unitario: si $p>c$, el beneficio crece sin límite con la producción; si $p<c$, la empresa no produce; y si $p=c$, el beneficio es nulo con cualquier cantidad, y la tecnología no determina el tamaño de la empresa ([\authorfont{Ver Tema 3.A.15}]). Con \textbf{rendimientos crecientes} (panel c), el coste es cóncavo y el coste marginal, decreciente: la cantidad en que el precio iguala el coste marginal, $\hat q$, es el mínimo del beneficio, no el máximo, y a partir de cierto tamaño el beneficio crece sin límite.

De ahí se siguen tres conclusiones. La primera es que la regla $p=CMg$ solo describe la conducta de una empresa precio-aceptante en el tramo creciente del coste marginal. La segunda, que con rendimientos crecientes la empresa no puede ser precio-aceptante: su tendencia a crecer conduce al poder de mercado y, en el límite, al monopolio natural, en el que fijar el precio igual al coste marginal causa pérdidas, porque este queda por debajo del coste medio ([\authorfont{Ver Tema 3.A.17, Tema 3.A.20}]). La tercera, que con rendimientos constantes la competencia determina la producción de la industria, pero no cuántas empresas la reparten ni el tamaño de cada una ([\authorfont{Ver Tema 3.A.16}]). En la práctica se añade, además, que la empresa que puede cerrar no acepta un beneficio inferior al de no producir: solo produce si cubre sus costes evitables, la regla de cierre que veremos con los costes a corto plazo.

\subsubsection{Oferta, demandas no condicionadas y función de beneficios}
Cuando el problema tiene solución, da la \textbf{función de oferta}, $q(p,w)$, y las \textbf{demandas no condicionadas de factores}, $z_i(p,w)$, que dependen del precio del producto porque la cantidad producida se elige a la vez. Son homogéneas de grado cero en $(p,w)$: si todos los precios se multiplican por la misma constante, ni las rectas de isobeneficio ni la elección cambian. Sustituyéndolas en el objetivo se obtiene la \textbf{función de beneficios}, que es también el resultado de la segunda etapa, en la formulación de VARIAN (1992):

$$\pi(p,w)=p\,q(p,w)-w\cdot z(p,w)=\max_q\ \big[p\,q-C(w,q)\big].$$

Tiene propiedades simétricas a las de la función de costes. Es \textbf{no decreciente en $p$} y \textbf{no creciente en $w$}. Es \textbf{homogénea de grado uno} en $(p,w)$, porque la elección no cambia y el beneficio se multiplica por la constante. Y es \textbf{convexa} en $(p,w)$: con unos precios intermedios, el plan óptimo es factible con los otros dos vectores de precios y da con cada uno un beneficio no mayor que el máximo. La figura muestra la lectura económica. Si la empresa no cambiara su plan cuando sube el precio de su producto, su beneficio crecería en línea recta, con pendiente igual a la cantidad producida; como aumenta su producción, crece más, y la función de beneficios queda por encima de esa recta, a la que toca en $p^0$.

\begin{center}\begin{minipage}{\linewidth}\centering
{\small\textbf{La función de beneficios es convexa en los precios}}\par\vspace{0.3em}
\includegraphics{anx_convexidad_beneficio.pdf}\par\vspace{0.3em}
{\small\textit{Fuente: }elaboración propia.}
\end{minipage}\end{center}

La pendiente de esa recta es el \textbf{lema de HOTELLING} (1932), el análogo del de SHEPHARD:

$$\frac{\partial\pi(p,w)}{\partial p}=q(p,w),\qquad\qquad \frac{\partial\pi(p,w)}{\partial w_i}=-z_i(p,w).$$

La razón es la misma: ante un cambio pequeño de un precio, el reajuste del plan óptimo tiene un efecto de segundo orden, y solo cuenta el efecto directo. Por la convexidad, la matriz de segundas derivadas de la función de beneficios es simétrica y \textbf{semidefinida positiva}, y de ahí se siguen tres leyes de la empresa competitiva: la oferta no disminuye con el precio del producto, $\partial q/\partial p\ge0$; la demanda no condicionada de un factor no aumenta con su precio, $\partial z_i/\partial w_i\le0$; y los efectos cruzados son simétricos, $\partial z_i/\partial w_j=\partial z_j/\partial w_i$ y $\partial q/\partial w_i=-\partial z_i/\partial p$.

\subsection{El dual estricto: ¿cuánto se produce con un presupuesto dado?}\label{anx:dual}
La tercera formulación, la de SEGURA (1986), maximiza la producción con un coste dado. Es el problema dual exacto de la minimización del coste, como la maximización de la utilidad lo es de la minimización del gasto ([\authorfont{Ver Tema 3.A.9}]), y tiene un doble interés: completa el paralelismo con la teoría del consumidor y permite descomponer el efecto de un precio sobre la demanda de un factor, aunque describe a un productor con un presupuesto fijo más que a una empresa que elige cuánto gastar.

\subsubsection{La función de producción indirecta}
El problema es

$$\max_{z\ge0}\ \ f(z)\qquad\text{s.a.}\qquad w\cdot z\le\bar C,$$

y su solución son las demandas de factores con coste dado, $z_i(w,\bar C)$, análogas a las demandas ordinarias del consumidor. Su función valor es la \textbf{función de producción indirecta}, $\phi(w,\bar C)=f\big(z(w,\bar C)\big)$, la mayor producción que permite el presupuesto $\bar C$ a los precios $w$, que estudió SHEPHARD (1970)\textcolor{orange}{\textsuperscript{[Probable]}}. Las condiciones de primer orden, $f_i=\lambda\,w_i$, dan la misma tangencia que la minimización del coste, y el multiplicador es la producción adicional que permite un euro más de presupuesto, $\lambda=\partial\phi/\partial\bar C$, la inversa del coste marginal.

Si los dos problemas se plantean con datos coherentes, sus soluciones coinciden y se cumplen cuatro identidades, que en el problema de SEGURA (1986) son las análogas de las de la dualidad del consumidor:

$$C\big(w,\phi(w,\bar C)\big)=\bar C,\qquad \phi\big(w,C(w,q)\big)=q,\qquad h_i(w,q)=z_i\big(w,C(w,q)\big),\qquad z_i(w,\bar C)=h_i\big(w,\phi(w,\bar C)\big).$$

La primera dice que el coste mínimo de obtener la producción máxima que permite un presupuesto es ese presupuesto; la segunda, que la producción máxima con el coste mínimo de $q$ es $q$; las otras dos, que las demandas de los dos problemas coinciden cuando se evalúan en el mismo punto. Por el teorema de la envolvente, además, las demandas con coste dado se obtienen de la función de producción indirecta como las ordinarias de la utilidad indirecta, con un análogo de la identidad de ROY: $z_i(w,\bar C)=-\big(\partial\phi/\partial w_i\big)/\big(\partial\phi/\partial\bar C\big)$.

\subsubsection{Efecto sustitución y efecto escala}
El efecto de un precio sobre la demanda de un factor puede descomponerse, como en la ecuación de SLUTSKY del consumidor ([\authorfont{Ver Tema 3.A.9}]), en un \textbf{efecto sustitución}, el cambio de combinación con la producción constante, y un \textbf{efecto escala}, el cambio de la producción. La descomposición tiene dos versiones, según qué permanezca constante cuando cambia el precio.

Con el presupuesto fijo, derivando la tercera identidad respecto de $w_j$ y aplicando el lema de SHEPHARD, $\partial C/\partial w_j=h_j$, se obtiene la ecuación de SLUTSKY de la producción en el problema de SEGURA (1986):

$$\frac{\partial z_i(w,\bar C)}{\partial w_j}=\frac{\partial h_i(w,q)}{\partial w_j}-h_j\,\frac{\partial z_i(w,\bar C)}{\partial\bar C}.$$

El primer término es el efecto sustitución, a lo largo de la isocuanta; el segundo, el \textbf{efecto presupuesto}: la subida de $w_j$ encarece el plan, el presupuesto rinde menos y la producción cae, y con ella cae la demanda de $i$ si es un \textbf{factor normal}, cuya demanda aumenta con la escala, y sube si es un \textbf{factor inferior}. Como en el consumidor, el efecto escala podría así contrarrestar el de sustitución.

Para la empresa que maximiza el beneficio, que es la relevante, la producción no la fija un presupuesto, sino la regla $p=CMg$, y la conclusión cambia. La demanda no condicionada es la condicionada evaluada en la cantidad óptima, $z_i(p,w)=h_i\big(w,q(p,w)\big)$, y derivándola respecto del precio propio, en la formulación de MAS-COLELL, WHINSTON y GREEN (1995):

$$\frac{\partial z_i}{\partial w_i}=\frac{\partial h_i}{\partial w_i}+\frac{\partial h_i}{\partial q}\,\frac{\partial q}{\partial w_i}=\frac{\partial h_i}{\partial w_i}-\left(\frac{\partial h_i}{\partial q}\right)^2\frac{\partial q}{\partial p}\ \le\ \frac{\partial h_i}{\partial w_i}\ \le\ 0.$$

La segunda igualdad usa la simetría de HOTELLING, $\partial q/\partial w_i=-\partial z_i/\partial p=-(\partial h_i/\partial q)\,(\partial q/\partial p)$. El efecto escala tiene siempre el mismo signo que el de sustitución, sea el factor normal o inferior. Si el factor es normal, su encarecimiento eleva el coste marginal, la empresa produce menos y emplea menos del factor. Si es inferior, su encarecimiento reduce el coste marginal, porque $\partial CMg/\partial w_i=\partial h_i/\partial q<0$, la empresa produce más y, por ser inferior, emplea todavía menos de él. \textbf{La empresa competitiva no tiene factores de GIFFEN}: su demanda no condicionada de un factor responde a su precio más que la condicionada, porque el efecto escala refuerza siempre al de sustitución. Es la diferencia de fondo con el consumidor, cuyo efecto renta puede contrarrestar el de sustitución: la empresa no tiene un presupuesto que la limite, sino que ajusta su producción al coste marginal, y la convexidad de la función de beneficios disciplina esa respuesta. La representación gráfica de los dos efectos y la respuesta de los factores a la escala se estudian en el segundo anexo.

\subsubsection{Tres formulaciones: balance}
El cuadro resume las tres formulaciones. Coinciden en la combinación de factores para una cantidad dada, pero no en lo que exigen ni en lo que determinan. La minimización del coste es la más general: solo necesita que la empresa tome como dados los precios de los factores y vale con cualquier tipo de rendimientos y con cualquier estructura del mercado del producto. La maximización del beneficio determina además la cantidad, pero exige que la empresa tome como dado también el precio del producto y que los rendimientos sean decrecientes en el óptimo. La maximización de la producción con un coste dado completa el paralelismo con el consumidor, pero describe a un productor con un presupuesto fijo.

\subsection{La dualidad: ¿contiene la función de costes toda la tecnología?}\label{anx:dualidad}
Hasta aquí hemos ido de la tecnología a los costes: dada la función de producción, la minimización da el coste mínimo de cada cantidad. El teorema de dualidad recorre el camino inverso y dice que en ese camino no se pierde nada de lo que importa. Si la tecnología es convexa y admite la eliminación gratuita, la función de costes la describe por completo, y cualquier función con las propiedades de una función de costes procede de alguna tecnología. Su consecuencia práctica es que puede partirse de la función de costes, más fácil de estimar y de manipular que la de producción, sin renunciar a ninguna información sobre la tecnología.

\subsubsection{De la función de costes a la tecnología}
Para cada vector de precios, la función de costes da el gasto mínimo con que puede obtenerse $q$, de modo que toda combinación de factores que permita obtenerlo cuesta, a esos precios, al menos $C(w,q)$. Geométricamente, todo el conjunto de factores necesarios, $V(q)$, queda por encima de la isocoste $w\cdot z=C(w,q)$, que lo toca en la combinación de mínimo coste. Como eso ocurre con cualquier vector de precios, $V(q)$ está contenido en la intersección de todos esos semiespacios, que en la construcción de SHEPHARD (1953) y McFADDEN (1978) es:

$$V^*(q)=\big\{z\ge0:\ w\cdot z\ge C(w,q)\ \ \text{para todo}\ w\ge0\big\}.$$

La figura muestra qué se obtiene. Si $V(q)$ es convexo y admite la eliminación gratuita (panel a), cada isocoste toca la isocuanta en un punto, y la envolvente de todas ellas la reproduce: $V^*(q)=V(q)$, y la función de costes contiene toda la información de la tecnología. Si $V(q)$ no es convexo (panel b), la envolvente pasa por debajo de la protuberancia de la isocuanta, por el segmento $AB$, y $V^*(q)$ es la envolvente convexa de $V(q)$. Las combinaciones del tramo no convexo no son nunca de mínimo coste: a los precios relativos de la isocoste que pasa por $A$ y $B$, la empresa es indiferente entre esos dos puntos, y si los precios cambian salta de uno a otro sin pasar por las combinaciones intermedias. La función de costes no puede revelarlas, porque una empresa que minimiza costes nunca las emplea.

{\small\textbf{La tecnología como envolvente de las isocostes}}\par\vspace{0.3em}
\includegraphics{anx_dualidad_envolvente.pdf}\par\vspace{0.3em}

\end{minipage}\end{center}

El \textbf{teorema de dualidad} de SHEPHARD (1953), que generalizaron UZAWA (1964) y McFADDEN (1978), completa el resultado: toda función $C(w,q)$ no decreciente, homogénea de grado uno, cóncava y continua en $w$, y no decreciente en $q$, es la función de costes de la tecnología $V^*(q)$ que acabamos de construir. Las propiedades de la función de costes no son, por tanto, solo necesarias, sino también suficientes: cualquier función que las cumpla es una función de costes legítima, y existe una correspondencia uno a uno entre las tecnologías convexas con eliminación gratuita y esas funciones. Es el problema de la \textbf{integrabilidad}, el mismo que en la teoría del consumidor permite recuperar las preferencias a partir de la función de gasto ([\authorfont{Ver Tema 3.A.9}]).\footnote{%
    Las propiedades de la tecnología que garantizan la dualidad exacta, la convexidad y la eliminación gratuita, son las que hacen de la tecnología una tecnología de buen comportamiento ([\authorfont{Ver Tema 3.A.11}]). Sin ellas, la dualidad no falla, sino que se refiere a la envolvente convexa de la tecnología, que es la parte que puede llegar a elegir una empresa que toma como dados los precios. En el análisis de actividades, la función de costes es el valor de un programa lineal, y la dualidad de la programación lineal, con los precios sombra de los recursos como variables del programa dual, es un caso particular de la que estudiamos (KOOPMANS, 1951).
}

En la práctica, la tecnología puede recuperarse de dos maneras. La primera aplica la definición de $V^*(q)$. La tecnología $q=(z_1z_2)^{1/2}$ tiene la función de costes $C(w,q)=2q\,(w_1w_2)^{1/2}$, de modo que una combinación $z$ pertenece a $V^*(q)$ si $w_1z_1+w_2z_2\ge2q\,(w_1w_2)^{1/2}$ para todo $w$. Como la media aritmética de $w_1z_1$ y $w_2z_2$ nunca es menor que su media geométrica, el miembro izquierdo es al menos $2\,(w_1w_2z_1z_2)^{1/2}$, con igualdad cuando $w_1z_1=w_2z_2$, y la condición se cumple con cualquier vector de precios si, y solo si, $(z_1z_2)^{1/2}\ge q$: se recupera la función de producción. La segunda, más directa, usa el lema de SHEPHARD. Las demandas condicionadas, $z_1=q\,(w_2/w_1)^{1/2}$ y $z_2=q\,(w_1/w_2)^{1/2}$, dependen de los precios relativos, y eliminándolos, al multiplicar las dos, se obtiene $z_1z_2=q^2$, la ecuación de la isocuanta.

La misma idea se aplica a la función de producción indirecta del problema de SEGURA (1986). Con cualquier vector de precios, la combinación $z$ cuesta $w\cdot z$, y con ese presupuesto puede obtenerse al menos $f(z)$, de modo que $\phi(w,w\cdot z)\ge f(z)$; con los precios que hacen óptima a $z$, se cumple la igualdad. Por tanto, si la función de producción es cuasicóncava, en la formulación de ese problema:

$$f(z)=\min_w\ \phi\big(w,\,w\cdot z\big),$$

que es el análogo de la recuperación de la utilidad a partir de la función de utilidad indirecta ([\authorfont{Ver Tema 3.A.9}]).

\subsubsection{La función de beneficios y la tecnología}
La función de beneficios permite el mismo recorrido. A los precios $(p,w)$, ningún plan factible da más beneficio que $\pi(p,w)$, de modo que todo el conjunto de producción queda por debajo de la recta de isobeneficio $p\,q-w\cdot z=\pi(p,w)$, que lo toca en el plan óptimo: la función de beneficios es la \textbf{función soporte} del conjunto de producción (McFADDEN, 1978). La intersección de todos esos semiespacios,

$$Y^*=\big\{(q,z)\ge0:\ p\,q-w\cdot z\le\pi(p,w)\ \ \text{para todo}\ (p,w)\big\},$$

reproduce el conjunto de producción si es convexo, y su envolvente convexa si no lo es.

\includegraphics{anx_dualidad_beneficio.pdf}\par\vspace{0.3em}

La figura lo muestra con una tecnología que tiene un tramo inicial de rendimientos crecientes. La envolvente de las rectas de isobeneficio reproduce la función de producción a partir de $A$, donde la toca la recta que pasa por el origen, pero sustituye el tramo inicial por el segmento que une el origen con $A$. Ninguna combinación de ese tramo maximiza el beneficio: si el cociente $w/p$ es mayor que la pendiente de esa recta, a la empresa le conviene no producir, y si es menor, emplear al menos $z_A$. La función de beneficios no ve el tramo de rendimientos crecientes, por la misma razón que la de costes no ve las protuberancias de la isocuanta, y la oferta salta de cero a $f(z_A)$ cuando $w/p$ cruza ese umbral, un salto análogo al que veremos con la regla de cierre a corto plazo.

La función de producción se recupera de la de beneficios con la homogeneidad de grado uno y el lema de HOTELLING. Con los precios reales, $\omega=w/p$, el beneficio por unidad de precio del producto es $\pi(1,\omega)$, y el lema da las demandas, $z(\omega)=-\nabla_\omega\pi(1,\omega)$, y la oferta, $q(\omega)=\pi(1,\omega)+\omega\cdot z(\omega)$. Si las demandas pueden invertirse, de modo que para cada combinación $z$ se conocen los precios reales $\omega(z)$ que la harían óptima, la función de producción es:

$$f(z)=\pi\big(1,\omega(z)\big)+\omega(z)\cdot z.$$

La fórmula tiene una lectura sencilla. Los precios reales que hacen óptima a $z$ son sus productividades marginales, $\omega_i(z)=f_i(z)$, porque en el óptimo el valor de la productividad marginal de cada factor iguala su precio, y el producto es el beneficio real más el pago real a los factores. La reconstrucción solo recupera el tramo de la función de producción en que la empresa llegaría a producir, el cóncavo, como muestra la figura.

La dualidad con la función de beneficios exige más que con la de costes. Con rendimientos constantes, el beneficio máximo es nulo o infinito, y la función de beneficios degenera: toda su información está en la frontera de la región de precios en que es nula, $p=c(w)$, que es la función de coste unitario que veremos a continuación. Por eso, cuando la tecnología puede tener rendimientos constantes o crecientes, la dual natural es la función de costes.

\subsubsection{La función distancia}
Hay una tercera representación dual, en cantidades en lugar de precios. La función distancia de SHEPHARD (1953, 1970), $D(z,q)$, mide por cuánto pueden dividirse a la vez todos los factores sin dejar de obtener $q$ ([\authorfont{Ver Tema 3.A.11}]), y se relaciona con la de costes de forma simétrica, en la formulación de SHEPHARD (1970):

$$C(w,q)=\min_z\big\{w\cdot z:\ D(z,q)\ge1\big\},\qquad\qquad D(z,q)=\min_w\big\{w\cdot z:\ C(w,q)\ge1\big\}.$$

La función de costes es a los precios lo que la función distancia a las cantidades: la primera es cóncava y homogénea de grado uno en los precios; la segunda, en los factores. El lema de SHEPHARD tiene también su versión dual: la derivada de la función distancia respecto de un factor es el precio sombra de ese factor, normalizado por el coste, que haría óptima la combinación observada. Es la herramienta con que se miden la eficiencia técnica y los precios implícitos de los productos que no tienen mercado, como las emisiones, en los métodos de frontera del tercer anexo.

\subsubsection{El esquema de la dualidad}
La figura resume las relaciones de este anexo. Si la tecnología es convexa, la tecnología y sus dos funciones duales contienen la misma información. De la tecnología se pasa a la función de costes minimizando el gasto, y de esta a la tecnología con la envolvente de las isocostes; a la función de beneficios, maximizando el beneficio, y de vuelta con su función soporte; y de la función de costes a la de beneficios, con la segunda etapa de la decisión. Los lemas de SHEPHARD y HOTELLING dan, derivando, las demandas y la oferta, y la función de producción indirecta del problema de SEGURA (1986) es la inversa de la función de costes respecto de la cantidad y del presupuesto. La dirección más usada es la que parte de la función de costes, porque es la única que solo exige que la empresa tome como dados los precios de los factores.

\subsection{De la tecnología a los costes: ¿qué rasgos de la tecnología se ven en los costes?}\label{anx:correspondencias}
Si la función de costes contiene la tecnología, cada rasgo de esta debe reflejarse en aquella. Las correspondencias más útiles son las de la escala y las de las formas funcionales: las primeras permiten leer los rendimientos de la tecnología en la forma de los costes, y las segundas, elegir una forma funcional con la que estimarla.

\subsubsection{Escala, homogeneidad y homoteticidad}
Si la tecnología es \textbf{homotética}, $f(z)=F\big(g(z)\big)$, con $g$ homogénea de grado uno ([\authorfont{Ver Tema 3.A.11}]), obtener $q$ exige alcanzar el nivel $g(z)=F^{-1}(q)$, y por la homogeneidad de $g$ el coste de alcanzarlo es proporcional a ese nivel. La función de costes es entonces \textbf{multiplicativamente separable}, como mostró SHEPHARD (1953):

$$C(w,q)=c(w)\,F^{-1}(q),$$

donde $c(w)$ es el coste de obtener una unidad de $g$. Los precios y la cantidad entran por separado: las proporciones de mínimo coste dependen solo de los precios, de modo que, con precios constantes, la senda de expansión es un radio desde el origen, y la cantidad solo determina la escala. La relación es recíproca, por el teorema de dualidad. Si la función de costes es separable, $C(w,q)=c(w)\,\varphi(q)$, entonces $V^*(q)=\varphi(q)\,\tilde V$, con $\tilde V=\{z:\ w\cdot z\ge c(w)\ \text{para todo}\ w\}$, y todos los conjuntos de factores necesarios son ampliaciones proporcionales de uno solo, que es la definición de tecnología homotética.

Si la tecnología es \textbf{homogénea de grado $k$}, $F(g)=g^k$, $F^{-1}(q)=q^{1/k}$ y:

$$C(w,q)=q^{1/k}\,c(w).$$

Con rendimientos constantes, $k=1$, el coste es proporcional a la cantidad, $C(w,q)=q\,c(w)$, y el coste medio y el marginal coinciden y no dependen de ella: toda la tecnología está en la \textbf{función de coste unitario} $c(w)$, cuyas derivadas son, por el lema de SHEPHARD, las cantidades de cada factor por unidad de producto, $a_i(w)=\partial c(w)/\partial w_i$. Con rendimientos crecientes, $k>1$, el coste crece menos que la cantidad y el coste medio disminuye; con rendimientos decrecientes, crece más y el coste medio aumenta. La relación general entre la escala de la tecnología y la forma de las curvas de coste, que no exige la homoteticidad, se estudia en el segundo anexo.\footnote{%
    La función de coste unitario es la pieza de las condiciones de beneficio nulo de los modelos de equilibrio general y de comercio internacional con rendimientos constantes, en los que el precio de cada bien iguala su coste unitario, $p_j=c_j(w)$, y los coeficientes técnicos dependen de los precios de los factores ([\authorfont{Ver Tema 3.A.21, Tema 3.B.5}]).
}

\subsubsection{Las formas funcionales duales}
Cada forma funcional de la función de producción tiene su función de costes, y conocer la correspondencia permite trabajar con la que resulte más cómoda. Las de las formas clásicas se obtienen resolviendo la minimización del coste, y las formas flexibles se formulan directamente como funciones de costes, porque el teorema de dualidad garantiza que basta con que cumplan sus propiedades.

La \textbf{COBB-DOUGLAS}, $q=AK^\alpha L^\beta$, con $\nu=\alpha+\beta$, tiene la función de costes, en la derivación habitual (VARIAN, 1992):

$$C(r,w,q)=\nu\left(\frac{q}{A}\right)^{1/\nu}\left(\frac{r}{\alpha}\right)^{\alpha/\nu}\left(\frac{w}{\beta}\right)^{\beta/\nu}.$$

Es también una COBB-DOUGLAS, ahora en los precios y la cantidad: la forma es \textbf{autodual}. Su logaritmo es lineal en los logaritmos de la cantidad y de los precios, y el coeficiente de $\ln q$ es $1/\nu$, la inversa del grado de homogeneidad, lo que permite estimar los rendimientos a escala a partir de los costes, como veremos en el tercer anexo. Por el lema de SHEPHARD, las participaciones de los factores en el coste son constantes, $s_K=\alpha/\nu$ y $s_L=\beta/\nu$, el reflejo de su elasticidad de sustitución unitaria.

La \textbf{CES} con rendimientos constantes, $q=A\left[\delta K^{-\rho}+(1-\delta)L^{-\rho}\right]^{-1/\rho}$, con $\sigma=1/(1+\rho)$, tiene la función de costes, en la misma derivación:

$$C(r,w,q)=\frac{q}{A}\left[\delta^\sigma r^{1-\sigma}+(1-\delta)^\sigma w^{1-\sigma}\right]^{1/(1-\sigma)}.$$

Es también una CES, en los precios, y el cociente de las participaciones depende de los precios relativos, $s_K/s_L=\big(\delta/(1-\delta)\big)^\sigma\,(r/w)^{1-\sigma}$: con $\sigma<1$, el factor que se encarece gana participación, porque su cantidad disminuye menos de lo que sube su precio, y con $\sigma>1$, la pierde. Es la versión, en costes, de la relación entre la elasticidad de sustitución y el reparto de la renta ([\authorfont{Ver Tema 3.A.11}]). Con $\sigma=1$ se obtiene la COBB-DOUGLAS y con $\sigma=0$, en unidades adecuadas, la de coeficientes fijos, cuyo coste es lineal en los precios, como vimos al estudiar las soluciones de esquina.

Entre las formas flexibles, la de \textbf{LEONTIEF generalizada} de DIEWERT (1971) es, con rendimientos constantes:

$$C(w,q)=q\sum_i\sum_j b_{ij}\,(w_iw_j)^{1/2},\qquad b_{ij}=b_{ji}.$$

Por el lema de SHEPHARD, da las cantidades de factores por unidad de producto, $h_i/q=\sum_j b_{ij}\,(w_j/w_i)^{1/2}$: coeficientes técnicos que dependen de los precios relativos y que se reducen a los fijos de LEONTIEF si $b_{ij}=0$ para $i\neq j$. La \textbf{translogarítmica}, la forma de CHRISTENSEN, JORGENSON y LAU (1973) aplicada a la función de costes, es una aproximación de segundo orden en logaritmos:

$$\ln C=\alpha_0+\alpha_q\ln q+\sum_i\alpha_i\ln w_i+\tfrac12\sum_i\sum_j\gamma_{ij}\ln w_i\ln w_j+\sum_i\gamma_{qi}\ln q\,\ln w_i+\tfrac12\,\gamma_{qq}(\ln q)^2.$$

Su ventaja está en el lema de SHEPHARD escrito en logaritmos: las participaciones de los factores en el coste son lineales en los parámetros,

$$s_i=\frac{\partial\ln C}{\partial\ln w_i}=\alpha_i+\sum_j\gamma_{ij}\ln w_j+\gamma_{qi}\ln q,$$

y se estiman junto con la función de costes. Las propiedades de la función de costes se traducen en restricciones sobre los parámetros: la homogeneidad de grado uno en los precios exige $\sum_i\alpha_i=1$ y $\sum_i\gamma_{ij}=\sum_i\gamma_{qi}=0$; la simetría, $\gamma_{ij}=\gamma_{ji}$; la homoteticidad, $\gamma_{qi}=0$; y la homogeneidad respecto de la cantidad, además, $\gamma_{qq}=0$. La concavidad no se reduce a restricciones sencillas sobre los parámetros y debe comprobarse con los datos. Es la forma de la mayoría de los estudios empíricos que veremos en el tercer anexo. El cuadro resume las correspondencias.

\anexo{Los costes en el tiempo}\label{anx:tiempo_costes}

El anexo anterior estudió la función de costes como si la empresa pudiera ajustar todos sus factores. Este introduce el tiempo en el sentido de MARSHALL (1890): a corto plazo algún factor está fijo, y a largo plazo todos son variables ([\authorfont{Ver Tema 3.A.11}]). Estudia cómo dependen los costes de la cantidad en cada horizonte, cómo se relacionan las curvas de los dos plazos, cuánto pierde la empresa por no poder ajustar todos sus factores y cómo responden sus demandas de factores a los precios. Va fuera del cuerpo porque el segundo bloque recoge sus resultados; aquí se demuestran y se completan con los casos que el cuerpo omite. Su tesis es que las curvas de coste son la imagen, en el espacio de la cantidad, de las leyes de la producción: las de corto plazo reflejan la ley de las proporciones variables; las de largo plazo, los rendimientos a escala; y la envolvente que las une, el principio de que una empresa con menos restricciones nunca produce más cara.

\subsection{El corto plazo: ¿cómo dependen los costes de la cantidad con un factor fijo?}\label{anx:corto_costes}
Con un factor fijo, la empresa solo puede variar la cantidad de los demás, y su problema de minimización tiene una restricción más. Sus costes se dividen entre los que dependen de la cantidad y los que no, y las curvas que resultan son la imagen de las productividades del factor variable.

\subsubsection{El problema y los costes fijos y variables}
A corto plazo, con el capital fijo en $\bar K$, el problema de la empresa es el de la minimización del coste con esa restricción adicional, en la formulación de VARIAN (1992):

$$C_s(w,r,q;\bar K)=\min_{L\ge0}\ \ wL+r\bar K\qquad\text{s.a.}\qquad f(\bar K,L)\ge q.$$

Con un solo factor variable no hay nada que elegir entre factores: la cantidad de trabajo es la que exige la isocuanta para el capital dado, $L_s(q;\bar K)$, la inversa de la función de producción a corto plazo, y el coste es $C_s=w\,L_s(q;\bar K)+r\bar K$. Con varios factores variables, la empresa sigue igualando la relación marginal de sustitución técnica entre ellos con sus precios relativos, pero no entre los variables y los fijos.

El \textbf{coste total} a corto plazo se divide en el \textbf{coste fijo}, $CF=r\bar K$, que no depende de la cantidad, y el \textbf{coste variable}, $CV(q)=w\,L_s(q;\bar K)$, que sí. De ellos se derivan los costes por unidad: el \textbf{coste medio}, $CTMe=CT/q$, suma del \textbf{coste fijo medio}, $CFMe=CF/q$, que disminuye siempre, porque el coste fijo se reparte entre más unidades, y del \textbf{coste variable medio}, $CVMe=CV/q$; y el \textbf{coste marginal}, $CMg=dCT/dq=dCV/dq$, que no depende del coste fijo. Como el coste a corto plazo exige emplear $\bar K$ aunque no sea el capital que minimiza el coste de $q$, nunca es menor que el de largo plazo, una relación que estudiaremos con la envolvente.

\subsubsection{Las curvas de coste como reflejo de las productividades}
Con un solo factor variable, las curvas de coste se deducen directamente de las productividades del trabajo ([\authorfont{Ver Tema 3.A.11}]):

$$CVMe=\frac{wL}{q}=\frac{w}{PMe_L},\qquad\qquad CMg=w\,\frac{dL}{dq}=\frac{w}{PMg_L}.$$

El coste variable medio es el salario dividido por lo que obtiene, en promedio, cada trabajador, y el coste marginal, el salario dividido por lo que añade el último. Por eso las curvas de coste son la imagen invertida de las de productividad, como muestra la figura con la misma tecnología en los dos paneles.\footnote{%
    La figura usa la función de producción a corto plazo $q=L+L^2-L^3/3$, con el capital fijo, en la que la productividad marginal es máxima con $L_1=1$, la media es máxima con $L_2=3/2$ y la marginal se anula con $L_3=1+\sqrt2$.
} Mientras la productividad marginal crece, hasta $L_1$, el coste marginal disminuye, y su mínimo, en $q_1$, corresponde al máximo de la productividad marginal, el punto de inflexión de la función de producción. El mínimo del coste variable medio, en $q_2$, corresponde al máximo de la productividad media, el óptimo técnico, y en él se cortan las dos curvas, porque allí la productividad marginal iguala a la media. Y el coste marginal tiende a infinito cuando la productividad marginal se anula, en el máximo técnico, $q_3$, la mayor cantidad que permite el factor fijo. La fase III de la producción, en la que la productividad marginal es negativa, no aparece en los costes, porque emplear más trabajo para obtener menos nunca es de mínimo coste.

El coste marginal corta al coste medio en su mínimo por una razón general, que vale para cualquier curva de costes, a corto o a largo plazo. La derivada del coste medio es

$$\frac{d\,CMe}{dq}=\frac{1}{q}\,\big(CMg-CMe\big),$$

de modo que el coste medio disminuye mientras el marginal queda por debajo, aumenta cuando queda por encima y es mínimo donde se igualan: la unidad adicional abarata el promedio si cuesta menos que él. Es la misma relación que une la productividad marginal con la media, vista desde los costes.

\subsubsection{Mínimo de explotación, óptimo de explotación y regla de cierre}
La figura completa las curvas de corto plazo con el coste fijo. En el panel (a), el coste total es el variable desplazado hacia arriba en la cuantía del fijo, de modo que las dos curvas tienen la misma pendiente, el coste marginal, en cada cantidad. En el panel (b), el coste fijo medio disminuye sin cesar, y el coste medio total, suma del fijo medio y del variable medio, tiene su mínimo a la derecha del mínimo del variable medio: por encima de este, el coste variable medio crece, pero al principio menos de lo que disminuye el fijo medio. El coste marginal corta a las dos curvas de coste medio en sus mínimos. El del coste variable medio, $M$, es el \textbf{mínimo de explotación}, y el del coste medio total, $O$, el \textbf{óptimo de explotación}, la cantidad con la que la planta produce al menor coste por unidad.

Los dos puntos tienen una interpretación precisa para la empresa que toma el precio como dado ([\authorfont{Ver Tema 3.A.16}]). Si el precio supera el coste medio en el óptimo de explotación, la empresa obtiene beneficios. Si queda entre los dos mínimos, tiene pérdidas, pero le conviene producir, porque cubre el coste variable y una parte del fijo, que pagaría igual si cerrara. Y si queda por debajo del mínimo de explotación, le conviene cerrar: produciendo perdería el coste fijo y, además, una parte del variable. Es la \textbf{regla de cierre}, y por eso la oferta a corto plazo es el tramo del coste marginal situado por encima del mínimo del coste variable medio. La regla descansa en que el coste fijo esté hundido. Si una parte puede evitarse cerrando, como el alquiler de una nave que puede rescindirse, cerrar la ahorra, y el umbral de cierre pasa a ser el mínimo del coste medio evitable, la suma del variable y de esa parte del fijo, como vimos al distinguir los costes evitables de los hundidos.

El coste medio decreciente a corto plazo no es una economía de escala. El coste fijo medio disminuye porque se reparte entre más unidades, y el variable medio, porque la productividad media del trabajo crece, pero la escala de la empresa, su capital, no cambia. Las economías de escala son un concepto de largo plazo, en el que la empresa puede cambiar todos sus factores.

\subsection{El largo plazo: ¿cómo dependen los costes de la cantidad cuando todos los factores varían?}\label{anx:largo_costes}
A largo plazo la empresa elige también su capital, y su coste es el de la función de costes del primer anexo. Cómo cambia con la cantidad depende de cómo cambian los factores que emplea, lo que describe la senda de expansión, y de los rendimientos a escala de la tecnología, y su relación con los costes de corto plazo es la de una envolvente.

\subsubsection{La senda de expansión y los factores inferiores}
Con precios de los factores constantes, la combinación de mínimo coste cambia con la cantidad a lo largo de la \textbf{senda de expansión}, el lugar de los puntos de tangencia entre las isocuantas y las isocostes paralelas. A lo largo de ella, la relación marginal de sustitución técnica es constante e igual a los precios relativos, de modo que la senda es una isoclina ([\authorfont{Ver Tema 3.A.11}]); con una tecnología homotética es un radio desde el origen, como vimos al estudiar las correspondencias entre la tecnología y los costes, y la proporción entre los factores no cambia con la escala. Las demandas condicionadas, vistas como funciones de la cantidad, $h_i(w,q)$, son las \textbf{curvas de ENGEL} de los factores, por analogía con las de los bienes del consumidor ([\authorfont{Ver Tema 3.A.9}]), y su pendiente clasifica los factores: un factor es \textbf{normal} si su demanda aumenta con la cantidad, $\partial h_i/\partial q>0$, e \textbf{inferior} si disminuye. Lo mide la elasticidad $\varepsilon_{h_i,q}=(\partial h_i/\partial q)\,(q/h_i)$, que puede cambiar de signo con el nivel de producción.

La figura muestra un factor que es normal a pequeña escala e inferior a gran escala. Hasta $P_3$, la senda de expansión se aleja de los dos ejes, y los dos factores aumentan con la cantidad; a partir de ahí se vuelve hacia atrás, y la empresa emplea menos trabajo cuanto más produce, como cuando la producción a gran escala permite sustituir tareas manuales por procesos automatizados. El panel (b) muestra lo mismo con las curvas de ENGEL: la del trabajo crece hasta $q_3$ y después disminuye, y la del capital crece siempre.

No todos los factores pueden ser inferiores. Derivando el coste respecto de la cantidad, $\sum_i w_i\,\partial h_i/\partial q=CMg>0$, de modo que al menos un factor tiene que ser normal. Y la inferioridad tiene una consecuencia que vimos al descomponer la respuesta de las demandas a los precios: como $\partial CMg/\partial w_i=\partial h_i/\partial q$, el encarecimiento de un factor inferior reduce el coste marginal.

\subsubsection{Economías de escala y rendimientos a escala}
Hay \textbf{economías de escala} cuando el coste medio a largo plazo disminuye con la cantidad, y \textbf{deseconomías} cuando aumenta. Las mide la \textbf{elasticidad del coste} respecto de la cantidad, el cociente entre el coste marginal y el medio:

$$\varepsilon_C=\frac{\partial C(w,q)}{\partial q}\,\frac{q}{C(w,q)}=\frac{CMg}{CMe}.$$

Hay economías de escala si $\varepsilon_C<1$, porque entonces el coste marginal queda por debajo del medio y lo arrastra hacia abajo; ninguna, si $\varepsilon_C=1$; y deseconomías, si $\varepsilon_C>1$. Una misma empresa puede pasar por los tres casos, en tramos sucesivos, cuando su coste medio a largo plazo tiene forma de U, la idea con la que WICKSELL reconcilió el agotamiento del producto con un tamaño determinado de la empresa ([\authorfont{Ver Tema 3.A.11}]).

La relación con los rendimientos a escala es general, como mostró HANOCH (1975), y no exige que la tecnología sea homotética. En la combinación de mínimo coste, cada productividad marginal es el precio del factor dividido por el coste marginal, $f_i=w_i/CMg$, de modo que la elasticidad de escala, la suma de las elasticidades del producto respecto de cada factor ([\authorfont{Ver Tema 3.A.11}]), es, en la derivación de ese autor:

$$\varepsilon(z^*)=\sum_i\frac{f_i\,z_i^*}{q}=\sum_i\frac{w_i\,z_i^*}{CMg\cdot q}=\frac{C}{CMg\cdot q}=\frac{CMe}{CMg}=\frac{1}{\varepsilon_C}.$$

\textbf{En toda combinación de mínimo coste, la elasticidad de escala es la inversa de la elasticidad del coste}: hay economías de escala exactamente donde los rendimientos a escala, medidos en la combinación de factores que la empresa elige, son crecientes. La homoteticidad solo añade que la senda de expansión es un radio, de modo que, al crecer, la empresa recorre justamente el radio a lo largo del cual se miden los rendimientos; sin ella, la empresa cambia además la proporción entre los factores, pero la igualdad se sigue cumpliendo punto a punto.

La correspondencia se rompe cuando el coste no refleja solo la tecnología. Si la empresa obtiene descuentos al comprar más factores, o tiene poder en sus mercados, los precios de los factores dependen de la cantidad, y el coste medio puede disminuir con rendimientos constantes o decrecientes: son \textbf{economías pecuniarias}, que abaratan la producción de la empresa sin ahorrar recursos a la sociedad, a diferencia de las \textbf{tecnológicas}, que sí los ahorran, una distinción que introdujo VINER (1931). Y si los costes de cada empresa dependen de la producción de toda la industria, por la formación de la mano de obra o por los proveedores especializados que atrae, las \textbf{economías externas} de MARSHALL (1890) desplazan las curvas de coste de cada empresa, en lugar de moverla a lo largo de ellas ([\authorfont{Ver Tema 3.A.11, Tema 3.A.16}]).

\subsubsection{La envolvente: el corto y el largo plazo}
El coste a largo plazo es el menor de los costes a corto plazo cuando el capital puede elegirse, en la formulación habitual (VARIAN, 1992):

$$C(w,r,q)=\min_K\ C_s(w,r,q;K).$$

Para cada cantidad, el coste a largo plazo no supera el de ninguna planta, y lo iguala con la planta óptima para esa cantidad, $K^*(q)$. La curva de coste a largo plazo es, por tanto, la \textbf{envolvente inferior} de las de corto plazo: queda por debajo de todas y toca cada una en la cantidad para la que su planta es la óptima. Lo mismo ocurre con los costes medios, que se obtienen dividiendo por la misma cantidad.

La figura lo muestra con tres plantas.\footnote{%
    La figura usa la tecnología $q=(KL)^{1/3}$ con un coste de puesta en marcha que no depende de la cantidad, de modo que, con precios unitarios, $C_s(q;K)=1+K+q^3/K$ y $C(q)=1+2q^{3/2}$, y el coste medio a largo plazo, $1/q+2q^{1/2}$, es mínimo en $q=1$.
} En el panel (a), cada curva de coste medio a corto plazo toca la de largo plazo en la cantidad para la que su planta es la óptima, $q_1$, $q_2$ y $q_3$, pero solo la planta 2 la toca en su mínimo. La planta 1 la toca en su tramo decreciente, a la izquierda de su propio mínimo: para producir $q_1$ conviene una planta algo mayor que la que tendría su mínimo en $q_1$, utilizada por debajo de su capacidad óptima, porque una más pequeña aprovecharía peor las economías de escala. La planta 3 la toca en su tramo creciente, a la derecha de su mínimo: con deseconomías de escala, conviene una planta algo menor que la que tendría su mínimo en $q_3$, utilizada por encima de esa capacidad. La curva a largo plazo solo pasa por el mínimo de una curva a corto plazo en su propio mínimo, donde no hay economías ni deseconomías de escala. Esa planta es la \textbf{dimensión óptima} de la empresa, y la cantidad que produce, la escala mínima eficiente, cuando a partir de ella el coste medio deja de disminuir.\footnote{%

\subsubsection{La escala mínima eficiente y la forma de la curva a largo plazo}
La forma de la curva de coste medio a largo plazo resume lo que la tecnología implica para la estructura de la industria. Si las economías de escala se agotan pronto, con una escala mínima eficiente pequeña en relación con la demanda, el mercado admite muchas empresas; si se agotan tarde, pocas; y si el coste medio disminuye en todo el tramo relevante de la demanda, la producción más barata es la de una sola empresa, el monopolio natural ([\authorfont{Ver Tema 3.A.15, Tema 3.A.17}]). La evidencia de ingeniería y estadística encuentra más a menudo curvas en forma de L que en forma de U: el coste medio disminuye hasta la escala mínima eficiente y después se mantiene casi constante en un tramo amplio, lo que deja indeterminado el tamaño de la empresa por encima de ese umbral ([\authorfont{Ver Tema 3.A.15}]). Las razones de ese tramo plano, entre ellas la flexibilidad de las plantas que estudió STIGLER (1939), se analizan en el tercer anexo.

\subsection{Corto y largo plazo: ¿por qué responde más la empresa a largo plazo?}\label{anx:chatelier}
La envolvente tiene una consecuencia que va más allá de la forma de las curvas: una empresa con menos restricciones responde más a los cambios de los precios. Es el \textbf{principio de LE CHÂTELIER-SAMUELSON}, que SAMUELSON (1947) formuló para cualquier problema de maximización: cuantas más variables se mantienen fijas, menor es la respuesta de las demás a un cambio de los parámetros.\footnote{%
    SAMUELSON (1947) tomó el nombre del principio de LE CHÂTELIER de la química, según el cual un sistema en equilibrio responde a una perturbación de modo que la contrarresta en parte, y lo convirtió en un teorema sobre la estática comparativa de los problemas de maximización con restricciones. La demostración con la envolvente que seguimos se debe a SILBERBERG (1974). SAMUELSON recibió el premio Nobel de Economía en 1970, en buena parte por esa aplicación sistemática de la maximización al análisis económico.
}

\subsubsection{La oferta a corto y a largo plazo}
Sea $q^*$ la cantidad para la que la planta $\bar K$ es la óptima. Como el coste a corto plazo con esa planta nunca es menor que el de largo plazo, y lo iguala en $q^*$, la diferencia $C_s(q;\bar K)-C(q)$ es nula en $q^*$ y positiva en el resto: tiene un mínimo en $q^*$. En un mínimo, la primera derivada se anula, y es la igualdad de los costes marginales que vimos con la envolvente; y la segunda no es negativa, de modo que, en la formulación de SAMUELSON (1947):

$$\left.\frac{\partial\,CMg_s}{\partial q}\right|_{q^*}\ \ge\ \left.\frac{\partial\,CMg_L}{\partial q}\right|_{q^*}.$$

En $q^*$, el coste marginal a corto plazo crece más deprisa que el de largo plazo: ampliar la producción sin ampliar la planta encarece cada unidad adicional más que hacerlo con la planta adecuada. Para la empresa precio-aceptante, cuya oferta es su coste marginal, la consecuencia es que \textbf{la oferta a largo plazo es más elástica que la de corto plazo}. La figura lo muestra: cuando el precio sube de $p_0$ a $p_1$, la empresa amplía su producción hasta $q_s$ con la planta que tiene, y hasta $q_L$ cuando puede adaptarla.

El resultado se aplica a cualquier número de factores fijos: cada restricción adicional reduce la respuesta, de modo que la oferta de una empresa con varios factores fijos es todavía menos elástica que la de una con uno solo. Es la razón microeconómica de que la respuesta de la oferta de una industria a un aumento de la demanda sea pequeña al principio y crezca con el tiempo, a medida que las empresas ajustan sus plantas y entran otras nuevas ([\authorfont{Ver Tema 3.A.16}]).

\subsubsection{Las demandas de factores a corto y a largo plazo}
El mismo razonamiento se aplica a los precios de los factores. Si $\bar K$ es el capital óptimo con los precios $w^0$, el coste a corto plazo con esa planta, como función de los precios, nunca es menor que el de largo plazo y lo iguala en $w^0$, porque mantener $\bar K$ es una de las opciones de la empresa a largo plazo. Las dos funciones son cóncavas en los precios, pero la de largo plazo lo es más: en $w^0$, su segunda derivada respecto del precio de un factor variable es menor, y por el lema de SHEPHARD, en la formulación de SILBERBERG (1974):

$$\frac{\partial h_i^L}{\partial w_i}\ \le\ \frac{\partial h_i^s}{\partial w_i}\ \le\ 0.$$

\textbf{La demanda condicionada de un factor es más elástica a largo plazo}: cuando sube el salario, a corto plazo la empresa solo puede sustituir trabajo por los demás factores variables, y a largo plazo también por capital. En el caso extremo de un solo factor variable, la demanda condicionada a corto plazo no responde en absoluto a su precio, porque la isocuanta fija la cantidad de trabajo que exige cada cantidad de producto, y toda la respuesta llega a través de la cantidad producida. Lo mismo vale para las demandas no condicionadas de la empresa que maximiza el beneficio. Por eso los efectos sobre el empleo de un aumento de los costes laborales, o sobre la demanda de energía de su encarecimiento, son menores a corto plazo que a largo, cuando las empresas han podido adaptar su equipo ([\authorfont{Ver Tema 3.A.25}]).

\subsection{Los precios de los factores: ¿cómo responden las demandas?}\label{anx:precios_factores}
El primer anexo obtuvo con álgebra las propiedades de las demandas de factores: la demanda condicionada nunca aumenta con el precio propio y, en la empresa competitiva, el efecto escala refuerza siempre al de sustitución. Este apartado las representa gráficamente, mide la intensidad de la respuesta y la relaciona con la elasticidad de sustitución, y pasa de la empresa al mercado con la demanda derivada.

\subsubsection{La sustitución con la cantidad constante}
Cuando sube el salario y la cantidad producida no cambia, la empresa sustituye trabajo por capital a lo largo de la misma isocuanta. La figura lo muestra: con el salario $w_0$, la combinación de mínimo coste es $E_0$; con un salario mayor, $w_1$, la isocoste es más inclinada y la tangencia se desplaza a $E_1$, con menos trabajo y más capital. Con la cantidad constante, el lugar de las combinaciones de mínimo coste que corresponden a los distintos salarios es la propia isocuanta, y representar la cantidad de trabajo de cada una frente a su salario da la \textbf{demanda condicionada de trabajo} del panel (b), que es decreciente, como exige la concavidad de la función de costes.\footnote{%
    Representar, en cambio, las combinaciones elegidas con distintos salarios y un coste dado describe el problema de la producción con un coste dado, el dual estricto del primer anexo, en el que el encarecimiento de un factor reduce también la producción que permite el presupuesto.
}

La intensidad de la sustitución depende de la curvatura de la isocuanta, es decir, de la elasticidad de sustitución ([\authorfont{Ver Tema 3.A.11}]). Con dos factores, la relación es exacta. Con la cantidad constante, las variaciones de los factores cumplen $w\,dL+r\,dK=0$, porque la isocuanta y la isocoste son tangentes, y la proporción entre factores varía con el precio relativo según la elasticidad de sustitución; combinando las dos condiciones se obtienen las elasticidades de las demandas condicionadas, en la formulación de ALLEN (1938)\textcolor{orange}{\textsuperscript{[Probable]}}:

$$\varepsilon_{h_L,w}=-s_K\,\sigma,\qquad\qquad \varepsilon_{h_K,w}=s_L\,\sigma,$$

donde $s_L$ y $s_K$ son las participaciones del trabajo y del capital en el coste. La demanda condicionada de trabajo es tanto más elástica cuanto más fácil es sustituirlo, y cuanto mayor es la participación del otro factor, que es el que tiene que absorber la sustitución. Con más de dos factores, las elasticidades de ALLEN-UZAWA y de MORISHIMA generalizan la relación, como vimos con la matriz de sustitución.

\subsubsection{Efecto sustitución y efecto escala}
La empresa que maximiza el beneficio no mantiene constante la cantidad: cuando un factor se encarece, cambia también cuánto produce. La figura descompone la respuesta en dos pasos. El \textbf{efecto sustitución} lleva de $E_0$ a $E'$, a lo largo de la isocuanta inicial, $q_0$, hasta la combinación de mínimo coste con los nuevos precios: el trabajo disminuye de $L_0$ a $L'$, y el capital aumenta. El \textbf{efecto escala} lleva de $E'$ a $E_1$: como el trabajo es un factor normal, su encarecimiento eleva el coste marginal, la empresa produce menos, $q_1$, y emplea menos de los dos factores con la nueva proporción, de modo que el trabajo baja todavía más, hasta $L_1$.

La figura muestra también la diferencia entre sustitutos netos y brutos. Con la cantidad constante, el capital aumenta cuando sube el salario: los dos factores son \textbf{sustitutos netos}, como lo son siempre con dos factores. Pero, una vez incluido el efecto escala, el capital puede acabar disminuyendo, como en la figura, en la que es menor en $E_1$ que en $E_0$: los factores pueden ser \textbf{complementarios brutos} aunque sean sustitutos netos, porque la caída de la producción arrastra a todos los factores normales. Si el trabajo fuera un factor inferior, su encarecimiento reduciría el coste marginal, y la empresa produciría más, pero emplearía todavía menos trabajo, porque al ser inferior su demanda disminuye con la escala: en los dos casos, como demostramos en el primer anexo, el efecto escala refuerza al de sustitución.\footnote{%

\subsubsection{La demanda derivada}
Las demandas de factores de una industria son \textbf{demandas derivadas}: dependen de la demanda del producto que los factores contribuyen a obtener. MARSHALL (1890) enunció las condiciones que hacen más elástica la demanda de un factor, y HICKS (1932) las precisó. En una industria competitiva con rendimientos constantes, dos factores y una oferta del otro factor perfectamente elástica, la elasticidad de la demanda de trabajo es, en la formulación habitual (HAMERMESH, 1993):

$$\eta_{LL}=-\big[(1-s_L)\,\sigma+s_L\,\eta\big],$$

donde $\eta$ es la elasticidad de la demanda del producto, en valor absoluto. La expresión reúne los dos efectos que acabamos de ver, ahora a escala de la industria: el de sustitución, $(1-s_L)\,\sigma$, que es la elasticidad de la demanda condicionada, y el de escala, $s_L\,\eta$, que recoge que el encarecimiento del trabajo eleva el coste, y el precio del producto, en proporción a su participación, y reduce la cantidad demandada del producto. De ahí las \textbf{reglas de MARSHALL y HICKS}: la demanda de un factor es más elástica cuanto más fácil es sustituirlo, cuanto más elástica es la demanda del producto y cuanto más elástica es la oferta de los demás factores. La regla de la participación es la más sutil: la demanda de un factor es más elástica cuanto mayor es su participación en el coste solo si la demanda del producto es más elástica que la sustitución, $\eta>\sigma$, la corrección de HICKS (1932) a MARSHALL, conocida como la importancia de no ser importante. Es la base del análisis de la incidencia de los impuestos sobre el trabajo y del poder de negociación de los sindicatos ([\authorfont{Ver Tema 3.A.25, Tema 3.A.27}]).

\subsection{La empresa con varias plantas: ¿cómo se reparte la producción?}\label{anx:plantas}
Una empresa con varias plantas, o con varios procesos para obtener el mismo producto, tiene que decidir cuánto produce en cada una. Su coste es el menor coste con que puede repartirse la cantidad total entre ellas:

$$C(q)=\min_{q_1,\dots,q_m}\ \sum_j C_j(q_j)\qquad\text{s.a.}\qquad \sum_j q_j=q.$$

En la solución interior, \textbf{el coste marginal es el mismo en todas las plantas que producen}, e igual al de la empresa: si en una planta fuera menor que en otra, trasladar producción de la segunda a la primera abarataría el total. La curva de coste marginal de la empresa es, por tanto, la suma horizontal de las de sus plantas, como muestra la figura: a cada nivel de coste marginal, $c^*$, la empresa produce lo que producen todas las plantas que tienen ese coste marginal. Una planta cuyo coste marginal mínimo supera $c^*$ no produce, como la planta 2 cuando la producción es pequeña: la empresa pone en marcha sus plantas por orden de coste marginal, que es la lógica del \textbf{orden de mérito} con que se despachan las centrales eléctricas. El resultado se aplica igual al monopolista con varias plantas, que iguala el ingreso marginal con el coste marginal común ([\authorfont{Ver Tema 3.A.17}]).
