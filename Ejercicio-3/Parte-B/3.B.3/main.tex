    % --- Imagen adaptativa ---
    % \imagenfit[alto]{fichero}{título}{fuente}: la imagen queda fija entre los párrafos donde se escribe (no flota), con su
    % título en el índice de figuras y la fuente debajo. Sin [alto]: las figuras PDF (TikZ) salen al tamaño con que se
    % dibujaron y solo se reducen si no caben en el ancho o en el alto de la página; las PNG/JPG ocupan el ancho del texto
    % hasta \imgaltoras del alto de la página. Con [alto] (fracción del alto de la página; [1] = página entera) se fija el
    % tamaño de esa imagen. Si no cabe en lo que queda de página, pasa entera a la siguiente.
    % \imagenfithorizontal{fichero}{título}{fuente}: la imagen en una página girada (apaisada), en ese mismo punto del texto.
    % Reglas: main/GUIA_TEMAS.md, §4 (Figuras).
    \newdimen\imgcap
    \newdimen\imgh
    \newdimen\imgw
    \newdimen\imgneed
    \newsavebox\imgcaja
    \newcommand{\imgaltoras}{0.30}% alto por defecto de las PNG/JPG (fracción del alto de la página)
    \makeatletter
    \newcommand{\imagenfit}[4][]{%
      \addvspace{10pt}%
      \par\begingroup
        \imgcap=3.5\baselineskip% alto aproximado de título + fuente
        \imgw=\linewidth
        \filename@parse{#2}\def\img@pdf{pdf}%
        \if\relax\detokenize{#1}\relax
          \ifx\filename@ext\img@pdf
            \sbox\imgcaja{\includegraphics{\detokenize{#2}}}% tamaño natural
            \ifdim\wd\imgcaja<\imgw \imgw=\wd\imgcaja \fi
            \imgh=\ht\imgcaja
          \else
            \imgh=\imgaltoras\textheight
          \fi
        \else
          \imgh=#1\textheight
        \fi
        \dimen@=\textheight \advance\dimen@ by -\imgcap
        \ifdim\imgh>\dimen@ \imgh=\dimen@ \fi
        \sbox\imgcaja{\includegraphics[width=\imgw,height=\imgh,keepaspectratio]{\detokenize{#2}}}%
        \imgneed=\ht\imgcaja \advance\imgneed by \imgcap
        \Needspace{\imgneed}%
        \noindent\begin{minipage}{\textwidth}
          \centering
          \captionof{figure}{\textemdash\ #3}\par\vspace{0.3em}% título arriba
          \usebox\imgcaja\par
          \vspace{0.3em}\small\textit{Fuente: }#4% fuente abajo
        \end{minipage}%
      \endgroup
      \par\addvspace{10pt}\RequestRestore
    }
    \newcommand{\imagenfithorizontal}[3]{%
      \begin{landscape}
        \centering
        \captionof{figure}{\textemdash\ #2}\par\vspace{0.3em}%
        \includegraphics[width=\linewidth,height=\dimexpr\textheight-4\baselineskip\relax,keepaspectratio]{\detokenize{#1}}\par
        \vspace{0.3em}{\small\textit{Fuente: }#3}%
      \end{landscape}%
    }
    \makeatother

    % \anexo{Título}: un anexo equivale a un \section (su título sale del mismo tamaño). Dentro se usan \subsection,
    % \subsubsection y \paragraph, nunca \section: se numeran I.1, I.1.1, I.1.1.1 y las páginas del anexo, I-1, I-2…
    % En el índice de contenido inicial solo figura el nombre del anexo. Sus apartados forman el índice del anexo, que sale
    % solo si el anexo los tiene, justo después de la introducción del anexo (antes del primer \subsection).
    % \anexo*{Título}: el mismo anexo, sin ese índice. Reglas: main/GUIA_TEMAS.md, §6.

    \newcommand{\anx@empezar}[2]{%
      \clearpage
      \refstepcounter{anexo}%
      \global\EnAnexotrue
      \setcounter{page}{1}\gdef\thepage{\theanexo-\arabic{page}}%
      \setcounter{section}{\numexpr\value{anexo}-1\relax}\stepcounter{section}% el anexo hace de \section
      \gdef\thesection{\Roman{section}}\gdef\theHsection{anexo.\Roman{section}}%
      \phantomsection
      \protected@xdef\leq@current@section{Anexo \theanexo.- \unexpanded{#2}}\gdef\leq@current@subsection{}% índice de ecuaciones
      \addtocontents{stc}{\protect\contentsline{section}{Anexo \theanexo.- #2}{}{\@currentHref}}% índice inicial: solo el nombre, sin página
      \addcontentsline{toc}{section}{Anexo \theanexo.- #2}% marcadores del PDF
      {\noindent\normalfont\Large\bfseries Anexo \theanexo.- #2\par}\nobreak\vspace{1.5ex \@plus .2ex}%
      #1\@ifundefined{anx@lista@\theanexo}{\global\anx@pendientefalse}{}%
    }
    % Índice del anexo: cada apartado escribe una línea en el fichero .atc (anexo, nivel, texto, página, enlace); al
    % empezar el documento se leen las de la compilación anterior y se guardan por anexo (dos pasadas, como cualquier índice).
    % En ese índice los apartados suben un nivel, porque el anexo hace de \section.
    \def\anx@entrada#1#2#3#4#5{\@ifundefined{anx@lista@#1}{\global\@namedef{anx@lista@#1}{}}{}%
      \expandafter\g@addto@macro\csname anx@lista@#1\endcsname{\contentsline{#2}{#3}{#4}{#5}}}
    \tcbuselibrary{breakable}
    \def\anx@indice{\ifanx@pendiente\global\anx@pendientefalse

            before upper={\parindent=0pt\parskip=2pt}]
          \small
          % números más anchos que en el índice inicial (I.1, I.1.1, I.1.1.1; páginas I-1)
          \def\@pnumwidth{2.6em}\def\@tocrmarg{3.4em}%
          \def\l@section##1##2{\addpenalty\@secpenalty\addvspace{1.0em \@plus\p@}\setlength\@tempdima{2.6em}%
            \begingroup\parindent\z@\rightskip\@pnumwidth\parfillskip-\@pnumwidth\leavevmode\bfseries
            \advance\leftskip\@tempdima\hskip-\leftskip##1\nobreak\hfil\nobreak\hb@xt@\@pnumwidth{\hss##2\kern-\p@\kern\p@}\par\endgroup}%

          \csname anx@lista@\theanexo\endcsname

      \fi}
    \def\anx@escribir#1#2#3{\ifEnAnexo

    % margen superior, así que no cambia las páginas del tema. Se enciende con \pagenumbering{arabic} y se apaga con

    \cs_new_protected:Npn \cab_guardar:n #1
      {
        \tl_gset:Nn \g_cab_titulo_tl {#1}
        \regex_replace_once:nnN { \A \s* Tema \s+ \S+ \s* \-\-? \s* } { } \g_cab_titulo_tl % sin «Tema X.Y.N --»
        \regex_replace_once:nnN { \.? \s* \c{\\} (\[ [^\]]* \])? \s* (\c{large} \s*)? } { .\  } \g_cab_titulo_tl % título. subtítulo
        \regex_replace_once:nnN { [\.\s]* \Z } { } \g_cab_titulo_tl % sin subtítulo: nada detrás
      }

\section{Financiación propia frente a financiación ajena}
\puntosclave{
    La empresa no tiene riqueza propia: el valor de sus activos se reparte entre quienes la financian. Cada fuente de financiación es un contrato que reparte de forma distinta los flujos de caja, el riesgo y el control.

    La distinción entre financiación propia y ajena atiende a la naturaleza del derecho, residual o fijo; la distinción entre interna y externa, a su origen. La primera determina el reparto del riesgo, del control y de la carga fiscal; la segunda, los costes de información que hay que pagar para obtener los fondos.

    El beneficio retenido no es gratuito: cuesta lo que exigen los accionistas. Su ventaja es ahorrar los costes de acudir a inversores externos, y su riesgo, liberar a los directivos de la disciplina del mercado.

    La deuda es el contrato que menos información exige. El banco la concede cuando la empresa es opaca y necesita supervisión; el mercado basta cuando la empresa tiene reputación y capital propio suficiente. Las empresas de la zona del euro y de España dependen del banco mucho más que las de EE. UU.

    Ampliar capital cede beneficios futuros, control y, si se emite por debajo del valor de la acción, riqueza de los antiguos accionistas. Los instrumentos híbridos combinan rasgos de la deuda y de las acciones para resolver conflictos concretos.
}

Toda inversión exige recursos antes de producirlos: la empresa paga hoy la maquinaria, las existencias o el desarrollo de un producto cuyos ingresos llegarán, si llegan, más adelante. Las fuentes de las que obtiene esos recursos son muchas, y la innovación financiera no ha dejado de multiplicarlas, pero todas responden a las mismas preguntas: quién aporta los fondos, qué recibe a cambio, quién decide sobre su uso y qué ocurre si las cosas van mal. La tesis del bloque es que cada fuente es un contrato que reparte de forma distinta los flujos de caja, el riesgo y el control, y que la que conviene a cada empresa depende de las fricciones a las que se enfrenta (el coste de informar a los inversores, el de supervisar a los directivos o la falta de activos que sirvan de garantía), no de una ordenación general entre fuentes baratas y caras. Como cada empresa combina esas fricciones a su manera, ningún marco permite valorar de una vez todos los instrumentos; sí permite entender por qué existe cada uno y quién lo usa. Si esas diferencias alteran, además, el valor de la empresa es la pregunta del bloque siguiente.

\subsection{La empresa en términos financieros: ¿a quién pertenece su valor?}
En términos financieros, la empresa no tiene riqueza propia. El valor de sus activos ($V$), que es el valor actual de los flujos de caja que se espera que generen, pertenece por entero a quienes la han financiado, y su \textbf{estructura financiera} no es más que la forma en que ese valor se reparte entre ellos. En valores de mercado,
$$V=E+D,$$
donde $D$ es el valor de las deudas y $E$ el de los fondos propios. Es el balance económico de la empresa: la misma identidad que el balance contable, pero con cada partida valorada a precios de mercado y no por su valor en libros.\footnote{%
    Además de accionistas y acreedores financieros, tienen derechos sobre los flujos de la empresa los proveedores que le conceden crédito comercial, los trabajadores, por los salarios devengados, y la Hacienda Pública, cuya participación en el beneficio, a través del impuesto sobre sociedades, depende de cómo se financie la empresa, como veremos en el bloque siguiente. Al hablar de estructura del capital, $D$ se limita a la deuda financiera. La diferencia entre el valor de mercado de los fondos propios y su valor en libros recoge lo que el mercado espera que los activos rindan por encima de su coste histórico ([\authorfont{Ver Tema 3.B.1}]).
}

Los dos derechos son de naturaleza distinta, y de esa diferencia nace la distinción entre financiación propia y ajena. El acreedor tiene un \textbf{derecho fijo}: una promesa de pago de intereses y principal en fechas pactadas, exigible con independencia de los resultados, que se cobra antes que los accionistas si la empresa se liquida. El accionista tiene un \textbf{derecho residual}: cobra lo que queda después de pagar a los acreedores, sin vencimiento ni cuantía pactados, y a cambio decide con su voto cómo se gestiona la empresa. El reparto del control es, por tanto, contingente: los accionistas mandan mientras la empresa paga, y los acreedores adquieren el control cuando deja de hacerlo, mediante el concurso o la renegociación (AGHION y BOLTON, 1992).\footnote{%
    AGHION y BOLTON (1992) parten de contratos incompletos: como no pueden preverse todas las decisiones futuras, lo esencial de un contrato financiero es quién decide en cada estado. La deuda asigna el control al empresario mientras paga y lo transfiere al acreedor cuando incumple. HART y MOORE (1994) añaden que, como el empresario puede marcharse con su capital humano, que no es embargable, solo se le puede prestar contra la amenaza de quedarse con los activos.
} A ello se suman dos diferencias más: los intereses son un gasto deducible en el impuesto sobre sociedades y los dividendos no, y las acciones son perpetuas mientras que la deuda vence.

La figura lo muestra con lo que cobra cada uno cuando vence la deuda. Si el valor de los activos supera el nominal de la deuda ($V\geq F$), el acreedor cobra $F$ y el accionista se queda con la diferencia; si no lo supera, la empresa no puede pagar, el acreedor se queda con todo y el accionista con nada, protegido por su responsabilidad limitada. La deuda paga, así, $\min(V,F)$, y los fondos propios, $\max(V-F,0)$, que es exactamente lo que paga una opción de compra sobre los activos de la empresa con precio de ejercicio $F$ (BLACK y SCHOLES, 1973; MERTON, 1974).

\imagenfit{pagos_deuda_acciones.pdf}{Lo que cobran la deuda y los fondos propios al vencimiento}{Elaboración propia a partir de MERTON (1974). Curvas ilustrativas.}

Esta lectura tiene dos consecuencias que recorren el tema. La primera es que, como el pago del accionista es convexo, su opción vale más cuanto más arriesgados sean los activos, y parte de ese valor lo pierde el acreedor: de ahí los conflictos entre ambos que veremos al tratar la agencia. La segunda es que lo que cobra el acreedor apenas depende del valor exacto de la empresa mientras esta sea solvente, de modo que la deuda es menos sensible que las acciones a lo que los inversores ignoran sobre ese valor ([\authorfont{Ver Tema 3.B.25}]).

¿Por qué busca financiación una empresa que no tiene riqueza propia? Porque sus oportunidades de inversión y los fondos que genera no coinciden ni en el tiempo ni en la cuantía: la inversión se paga antes de que produzca ingresos y suele llegar en bloques, mientras que los fondos se generan de forma gradual. La financiación traslada recursos de quienes ahorran a quien puede invertirlos. El criterio para decidir qué inversiones acometer no es crecer, sino crear valor para los propietarios: invertir en los proyectos cuyo valor actual neto, descontado al coste del capital, sea positivo ([\authorfont{Ver Tema 3.B.2}]). Con mercados de capitales perfectos, ese criterio no depende de las preferencias de cada propietario, porque cada uno puede ajustar después su consumo prestando o pidiendo prestado (separación de FISHER, 1930). Crecer es un medio: crea valor solo si la nueva inversión rinde más de lo que cuesta financiarla, y crecer por crecer es justamente el objetivo de los directivos contra el que advierte la teoría de la agencia (JENSEN, 1986).

Aun así, el crecimiento es la forma más directa de ver cómo se encadenan las decisiones financieras. Si la empresa no emite acciones y mantiene constante su endeudamiento, el mayor ritmo al que puede crecer es la \textbf{tasa de crecimiento sostenible} de HIGGINS (1977), que, en su versión con el efecto del apalancamiento sobre la rentabilidad financiera, es:
$$g^{*}=b\cdot \text{ROE},\qquad \text{ROE}=\left[\text{ROA}+\left(\text{ROA}-i\right)\frac{D}{E}\right](1-t),$$
donde $b$ es la proporción del beneficio que se retiene, ROA la rentabilidad económica (beneficio antes de intereses e impuestos sobre activo), $i$ el coste de la deuda y $t$ el tipo del impuesto sobre sociedades.\footnote{%
    Si la empresa no emite acciones, sus fondos propios solo crecen con el beneficio retenido, $\Delta E=b\cdot\text{BN}$, y crecen a la tasa $\Delta E/E=b\cdot\text{ROE}$, con el ROE calculado sobre los fondos propios iniciales. Si además mantiene constante $D/E$, la deuda y el activo crecen a ese mismo ritmo, y con ellos las ventas si la rotación del activo no cambia. La descomposición del ROE se obtiene de $\text{BN}=(\text{BAII}-iD)(1-t)$ y $\text{BAII}=\text{ROA}\cdot(E+D)$ ([\authorfont{Ver Tema 3.B.1}]). El ROA puede sustituirse por el ROCE, el beneficio operativo sobre el capital empleado, con los ajustes debidos, sin que cambien las conclusiones; el ROCE se interpreta además como la rentabilidad de los fondos propios si la deuda fuera nula.
} La identidad recoge las tres decisiones de las que trata el tema. La proporción retenida, $b$, es la otra cara de la política de dividendos. La ratio $D/E$ es la estructura del capital: si la empresa obtiene sobre sus activos más de lo que paga por su deuda ($\text{ROA}>i$), endeudarse eleva la rentabilidad del accionista, y si obtiene menos, la reduce. Y crecer por encima de $g^{*}$ exige fondos externos, emitiendo acciones o endeudándose más, que es la elección entre financiación propia y ajena externa.

Pero es una identidad contable, en valores de libro, y no dice nada del valor. Un mayor endeudamiento eleva el ROE esperado, pero también su variabilidad y, con ella, la rentabilidad que exigen los accionistas; una mayor retención eleva $g^{*}$, pero solo crea valor si los fondos retenidos rinden más que esa rentabilidad exigida. Los dos bloques siguientes sustituyen las rentabilidades contables por las que exigen los inversores, en valores de mercado, y se preguntan si la estructura del capital y la política de dividendos alteran el valor de la empresa.

Las fuentes con que la empresa financia su crecimiento se ordenan con dos criterios. Según la \textbf{naturaleza del derecho}, la financiación es \textbf{propia} (fondos propios: derecho residual y no exigible) o \textbf{ajena} (deudas: derecho fijo y exigible). Según su \textbf{origen}, es \textbf{interna}, si procede de los fondos que genera la actividad y la empresa no reparte, o \textbf{externa}, si la aportan inversores de fuera. Los dos criterios no coinciden: el beneficio retenido es financiación propia e interna; una ampliación de capital, propia y externa; un préstamo bancario, ajena y externa. La figura los cruza.\footnote{%
    Los instrumentos concretos de cada fuente, sus variantes y los criterios prácticos para elegir entre ellos se describen en un anexo ([\authorfont{Ver Anexo \ref{anx:instrumentos}}]).
}

\imagenfit{fuentes_matriz.pdf}{Las fuentes de financiación según su origen y la naturaleza del derecho}{Elaboración propia.}

Cada criterio responde a una pregunta distinta. El primero dice cómo se reparten el riesgo, el control y la carga fiscal; el segundo, cuánto cuesta convencer a alguien de fuera para que aporte fondos, porque quien está fuera sabe menos que quien dirige la empresa. Por eso la literatura ha recurrido al segundo para explicar qué fuentes eligen las empresas y en qué orden, y al primero para explicar la estructura del capital.

\subsection{Financiación interna: ¿es gratis el beneficio retenido?}
El beneficio que la empresa no reparte es su principal fuente de financiación y la que parece más barata, porque no obliga a pagar intereses ni dividendos. No es gratuito: su coste es lo que exigen los accionistas por dejarlo dentro. Su verdadera ventaja es otra, ahorrar los costes de acudir a inversores externos, y su riesgo, liberar a los directivos de la disciplina que esos inversores imponen.

La financiación interna, o \textbf{autofinanciación}, comprende el beneficio retenido, que se acumula en las reservas y aumenta los fondos propios (\textbf{autofinanciación de enriquecimiento}), y las dotaciones a amortizaciones y provisiones, gastos que no suponen salida de caja y que, al reducir el beneficio repartible, retienen en la empresa los fondos necesarios para mantener su capacidad productiva (\textbf{autofinanciación de mantenimiento}).\footnote{%
    Las reservas pueden ser legales, estatutarias o voluntarias; la reserva legal obliga a destinar el $10\%$ del beneficio del ejercicio hasta que alcance al menos el $20\%$ del capital social (art. $274$ de la Ley de Sociedades de Capital). Las provisiones se registran en el pasivo y las amortizaciones minoran el activo, por lo que la autofinanciación de mantenimiento no encaja limpiamente en la distinción entre propia y ajena. Dos efectos amplían su alcance. Si la empresa mantiene constante su ratio de endeudamiento, $\alpha=D/E$, cada euro de reservas le permite endeudarse en $\alpha$ euros más, y sus recursos totales crecen en $(1+\alpha)$ euros: es el \textbf{efecto multiplicador} de la autofinanciación. Y si reinvierte las cuotas de amortización en nuevos activos, su capacidad productiva no solo se mantiene, sino que crece (\textbf{efecto LOHMANN-RUCHTI}). Los supuestos y un ejemplo están en un anexo ([\authorfont{Ver Anexo \ref{anx:instrumentos}}]).
}

Retener un euro equivale, para el accionista, a recibirlo como dividendo y volver a invertirlo en acciones nuevas de la empresa. Por eso el coste de la autofinanciación es el \textbf{coste de oportunidad} de los accionistas, la rentabilidad que exigen a los fondos propios ($r_E$), la misma que exigirían a una ampliación de capital salvo por los costes de emisión. Tratar el beneficio retenido como un recurso sin coste lleva a aceptar inversiones que rinden menos de lo que los accionistas obtendrían fuera; el efectivo que excede lo que la empresa puede invertir con rentabilidad debería devolverse a los accionistas, mediante dividendos o recompras de acciones, o destinarse a amortizar deuda.

La ventaja de la autofinanciación no está en su precio, sino en lo que evita. No diluye el control de los accionistas, no tiene costes de emisión, da a la empresa autonomía para decidir cuándo y en qué invierte y, sobre todo, no exige convencer a inversores que saben menos que los directivos sobre el valor de la empresa. Este último punto es decisivo: un inversor externo que no puede distinguir una empresa buena de una mala paga por sus títulos un precio medio, y la empresa buena prefiere entonces financiarse con lo que tiene (MYERS y MAJLUF, 1984), un argumento que desarrollaremos al tratar la estructura del capital. Para muchas pymes, además, es la única alternativa al crédito bancario para financiarse a largo plazo.

Sus límites son dos. El primero es cuantitativo: los fondos internos se generan poco a poco y las inversiones llegan en bloques, de modo que la empresa que crece deprisa agota pronto su autofinanciación y puede tener que renunciar a inversiones rentables. El segundo es de incentivos. Cuando la empresa genera más fondos de los que puede invertir con valor actual neto positivo, ese \textbf{flujo de caja libre} queda a disposición de unos directivos que, sin tener que rendir cuentas ante los inversores, tienden a invertirlo en crecer más allá de lo rentable (JENSEN, 1986). La deuda y los dividendos, que obligan a sacar ese efectivo de la empresa, son las respuestas que veremos en los bloques siguientes.

Si los fondos externos cuestan más que los internos, la autofinanciación deja de ser una fuente más y pasa a condicionar la inversión. Sin fricciones, la empresa invierte hasta que la rentabilidad marginal de la inversión iguala el coste de los fondos, $r$, tenga los fondos internos ($W$) que tenga. Con una prima por financiación externa, la oferta de fondos se quiebra en $W$, la inversión queda por debajo de su nivel eficiente y aumenta cuando aumentan los fondos internos, aunque no hayan cambiado las oportunidades de inversión.

FAZZARI, HUBBARD y PETERSEN (1988) encontraron que la inversión de las empresas que reparten pocos dividendos, presumiblemente las más restringidas, es más sensible a su flujo de caja que la de las demás. KAPLAN y ZINGALES (1997) mostraron que, dentro de ese grupo, la sensibilidad era mayor en las empresas que, según sus estados financieros, estaban menos restringidas, de modo que no sirve para medir las restricciones financieras. El debate sigue abierto sobre la forma de medirlas, no sobre la existencia de la prima.\textcolor{orange}{\textsuperscript{[Probable]}}\footnote{%
    En el plano macroeconómico, esa prima es el origen del acelerador financiero: como depende del patrimonio neto de los prestatarios, que cae en las recesiones, amplifica los \textit{shocks} y transmite la política monetaria a través del crédito ([\authorfont{Ver Tema 3.A.34, Tema 3.A.37}]).
}

\subsection{Financiación ajena: ¿por qué deuda y de quién?}
La deuda es el contrato de financiación que menos información exige al acreedor: mientras la empresa paga, no necesita saber cuánto gana. De quién la obtiene la empresa, de un banco o del mercado, depende de cuánta supervisión necesita para que alguien se la conceda, y eso depende de su tamaño, de su reputación y de su capital propio.

Si los inversores pudieran observar sin coste los resultados de la empresa, podrían firmar con ella cualquier contrato que hiciera depender sus cobros de esos resultados. Cuando verificarlos es costoso, el contrato óptimo es la deuda: un pago fijo que el acreedor no necesita comprobar mientras se cumple, y una verificación, con la liquidación de los activos, solo cuando la empresa declara que no puede pagar (TOWNSEND, 1979; GALE y HELLWIG, 1985). La deuda ahorra así costes de verificación, y su fuerza está en la amenaza de liquidación, que es la que obliga a pagar.

Para la empresa, la deuda tiene tres ventajas: no diluye el control de los accionistas, porque el acreedor no vota mientras cobra; sus intereses son deducibles en el impuesto sobre sociedades; y, si la rentabilidad de los activos supera su coste, eleva la rentabilidad del accionista. Su inconveniente es la otra cara de esa misma rigidez: es una obligación incondicional, cuyo impago puede llevar a la empresa al concurso, y su contrato suele limitar lo que la empresa puede hacer mientras la deuda está viva.\footnote{%
    Las cláusulas restrictivas (\textit{covenants}) limitan, entre otras cosas, el reparto de dividendos, la venta de activos, el endeudamiento adicional o el deterioro de determinadas ratios financieras, y su incumplimiento permite al acreedor exigir la devolución anticipada. Protegen al acreedor frente a decisiones que trasladen riesgo o valor de él a los accionistas, a costa de restar flexibilidad a la empresa.
}

Puede obtenerse de un banco, que la negocia bilateralmente y la mantiene en su balance, o en el mercado, emitiendo bonos que compran inversores dispersos. Las diferencias prácticas son conocidas: el préstamo bancario se obtiene con rapidez, por importes pequeños y sin hacer pública información sobre la empresa, mientras que la emisión exige volúmenes mínimos para que el título tenga liquidez, información pública y, a menudo, una calificación crediticia, pero da acceso a fondos más abundantes y, para las empresas grandes, más baratos.\footnote{%
    Los cinco criterios con que suelen compararse (coste, volumen, desarrollo financiero, rapidez e información), la colocación privada, los préstamos sindicados y otras formas de deuda se desarrollan en un anexo ([\authorfont{Ver Anexo \ref{anx:instrumentos}}]). El crédito comercial de los proveedores es otra forma de financiación ajena: el proveedor conoce al cliente por la relación comercial y puede recuperar la mercancía si no le paga, por lo que presta a empresas a las que un banco no prestaría, que lo usan más (PETERSEN y RAJAN, 1997).
}

La teoría explica por qué. Supervisar a un prestatario es costoso, y si cada pequeño inversor lo hiciera por su cuenta el coste se multiplicaría o, lo que es más probable, nadie lo haría. El banco supervisa en nombre de todos sus depositantes y, como presta a muchos prestatarios distintos, diversifica el riesgo de su cartera y puede garantizar a sus depositantes que hace bien ese trabajo (DIAMOND, 1984). La supervisión tiene, sin embargo, un precio: la información que el banco acumula le da poder para quedarse con parte de las rentas de la empresa, que por eso preferirá el mercado en cuanto pueda (RAJAN, 1992). Y puede hacerlo cuando ha construido una reputación pagando puntualmente bajo la vigilancia del banco (DIAMOND, 1991).

HOLMSTRÖM y TIROLE (1997) unen estas piezas en función del capital propio de la empresa. Como el empresario puede desviar parte de los resultados en su provecho, los inversores solo financian un proyecto si él conserva una participación suficiente en su éxito. Quien tiene mucho capital propio ($A\geq\overline{A}$) ofrece esa garantía por sí mismo y se financia directamente en el mercado. Quien tiene menos necesita además que un banco lo supervise, y el banco, a su vez, tiene que arriesgar capital propio en el préstamo para tener incentivos a supervisar. Quien tiene muy poco ($A<\underline{A}$) no obtiene financiación, aunque su proyecto sea rentable: el crédito se raciona.

El modelo explica el patrón observado: las empresas pequeñas y jóvenes dependen del banco, y las grandes, con historial y activos, acuden también al mercado. Entre 2009 y 2022, solo en torno al $9\%$ de las empresas de la encuesta del BCE sobre el acceso a la financiación emitía bonos, frente a algo más del $50\%$ que usaba productos bancarios (DURANTE et al., 2023). Las diferencias entre países reflejan, además de su tejido empresarial, su desarrollo financiero y su marco institucional (DE JONG, KABIR y NGUYEN, 2008). A finales de 2025, los préstamos de los bancos a las sociedades no financieras de la zona del euro ($5,3$ billones de euros) casi triplicaban los valores de deuda que estas tenían emitidos ($2,0$ billones), y en España los más que triplicaban ($463$ mil millones frente a $144$ mil millones). En EE. UU., en cambio, los valores de deuda de las empresas no financieras ($8,9$ billones de dólares) superaban en dos tercios el total de sus préstamos ($5,4$ billones).\footnote{%

}

\subsection{Financiación propia externa: ¿qué se cede al ampliar capital?}
Quien amplía capital cede tres cosas: una parte de los beneficios futuros, una parte del control y, si emite por debajo de lo que valen sus acciones, una parte de la riqueza de los antiguos accionistas. La primera es el precio de los fondos; las otras dos explican por qué las empresas se resisten a ampliar capital.

Las acciones dan un derecho residual, sin vencimiento y con voto. Por eso son la financiación natural de las empresas cuyos flujos de caja son demasiado inciertos, o cuyos activos demasiado intangibles, para sostener una deuda: las jóvenes, las de crecimiento rápido y las intensivas en investigación. En las primeras etapas, quien aporta ese capital no se limita a comprar acciones: los fondos de capital riesgo entregan el dinero por fases y se reservan derechos sobre los flujos, el consejo, el voto y la liquidación que dependen de los resultados, separando lo que en una acción ordinaria va unido (KAPLAN y STRÖMBERG, 2003). Su coste es la rentabilidad que exigen los accionistas dado el riesgo de la empresa ($r_E$), más los costes de emisión, y es mayor que el de la deuda porque el accionista cobra el último y soporta la variabilidad de los resultados.\footnote{%
    La salida a bolsa, mediante una oferta pública inicial o la fusión con una sociedad cotizada creada para ello (SPAC), da acceso a más inversores y liquidez a los accionistas, a cambio de dispersar la propiedad y de asumir obligaciones de información. Los tipos de acciones y la salida a bolsa se describen en un anexo ([\authorfont{Ver Anexo \ref{anx:instrumentos}}]); la valoración de las acciones, en otro tema ([\authorfont{Ver Tema 3.B.23}]).
}

La \textbf{dilución del control} depende de quién compre las nuevas acciones. Si los dos accionistas de una empresa, A y B, con participaciones $\alpha_A$ y $\alpha_B$, no acuden a una ampliación por una fracción $s$ del capital previo que suscribe un tercero, C, sus participaciones pasan a ser $\alpha_A/(1+s)$ y $\alpha_B/(1+s)$, y C obtiene $s/(1+s)$. Si suscriben en proporción a lo que ya tienen, las conservan; para eso la ley les reconoce un derecho de preferencia en las ampliaciones con aportaciones dinerarias (art. $304$ de la Ley de Sociedades de Capital).\footnote{%
    Por ejemplo, si A tiene el $80\%$ de un capital de $100$ y B el $20\%$, y C quiere el $20\%$ de la empresa mediante una ampliación, tiene que aportar $25$, no $20$, porque el capital pasa a ser $125$; A queda con el $64\%$ y B con el $16\%$. Si, en cambio, B vendiera su participación a C, el capital de la empresa y la posición de A no cambiarían: la dilución es propia de la ampliación, no de la entrada de un nuevo accionista.
} Que la dilución importe depende de quién entre y de qué quiera: las empresas familiares suelen anteponer la conservación del control; los inversores de capital riesgo, los inversores informales (\textit{business angels}) y los fondos de capital privado aportan, además de fondos, conocimiento y supervisión; los inversores institucionales suelen conformarse con participaciones minoritarias; y los empleados, que pueden entrar mediante acciones u opciones sobre acciones, forman un accionariado estable.

La \textbf{dilución del valor} depende del precio. Si la empresa emite $m$ acciones a un precio $p_e$ cuando tiene $n$ acciones que valen $v$ cada una, cada acción pasa a valer $v'=(nv+mp_e)/(n+m)$. Si $p_e=v$, nadie gana ni pierde; si $p_e<v$, los antiguos accionistas transfieren riqueza a los nuevos, y si $p_e>v$, al revés. Lo relevante es, por tanto, el precio de emisión frente al valor de la acción, no la prima de emisión sobre el nominal, que es un concepto contable.

\imagenfit{dilucion_paneles.pdf}{Las dos diluciones de una ampliación de capital}{Elaboración propia. Esquema ilustrativo.}

Como los directivos conocen ese valor mejor que el mercado, la decisión de ampliar capital transmite información: el mercado sospecha que la empresa emite acciones cuando cree que están sobrevaloradas, y el anuncio de una ampliación hace caer la cotización, en torno a un $3\%$ de media (ASQUITH y MULLINS, 1986). Por eso la empresa de crecimiento rápido, cuyos directivos suelen creer que sus acciones están infravaloradas, es reacia a ampliar capital. Además, en una empresa endeudada, el capital nuevo reduce el riesgo de impago y revaloriza la deuda existente, de modo que parte del valor que crea la inversión financiada se lo quedan los acreedores, y los accionistas pueden preferir renunciar a inversiones rentables (MYERS, 1977). Ambos problemas, que desarrollaremos en el bloque siguiente, abren el espacio de los instrumentos híbridos.

\subsection{Financiación híbrida: ¿por qué combinar deuda y acciones?}
Entre la deuda, que no da derechos políticos y paga con independencia de los resultados, y las acciones, que dan voto y cobran según los resultados, hay instrumentos que combinan rasgos de ambas. No son un término medio: cada uno resuelve un conflicto concreto entre las partes.

El más importante es el \textbf{bono convertible}, un bono que su titular puede canjear por acciones en unas condiciones fijadas al emitirlo. Si al convertir recibe una fracción $\gamma$ de la empresa, cobra $\max\{\min(V,F),\gamma V\}$: el pago de la deuda simple si la empresa va mal y su parte de los fondos propios si va bien. Tiene tres usos. Permite pagar un interés menor que la deuda simple, porque el inversor recibe a cambio la opción de convertir. Corrige el incentivo que crea la convexidad de los fondos propios: si los accionistas eligen proyectos más arriesgados a costa de los acreedores, el titular del convertible comparte las ganancias convirtiendo, lo que les quita buena parte del incentivo (GREEN, 1984); y, como su valor es poco sensible al riesgo de la empresa, emisor e inversor no necesitan ponerse de acuerdo sobre cuánto es ese riesgo (BRENNAN y SCHWARTZ, 1988). Y permite a la empresa a la que le cuesta emitir acciones, por la penalización informativa que acabamos de ver, emitirlas ``por la puerta de atrás'', con una deuda que espera convertir cuando su cotización suba (STEIN, 1992). Por eso lo usan las empresas de crecimiento rápido, aunque también las consolidadas, para diversificar su base de inversores: hay fondos especializados en bonos convertibles.

Los demás instrumentos responden a necesidades parecidas. Los \textbf{warrants} dan derecho a suscribir acciones nuevas a un precio y en un plazo fijados; suelen acompañar a una emisión de deuda, a la que hacen atractiva para inversores dispuestos a asumir más riesgo, y permiten captar a quien no entraría en el capital de una empresa en dificultades pero sí compraría la opción de hacerlo.\footnote{%
    En 2011, Berkshire Hathaway invirtió en acciones preferentes de Bank of America con \textit{warrants} para comprar $700$ millones de acciones a $7,14$ dólares; los ejerció en 2017, cuando la acción cotizaba en torno a $24$ dólares.
} Las \textbf{acciones preferentes} pagan un dividendo fijo, preferente al de las ordinarias, a cambio de restringir o eliminar el voto, lo que permite a una empresa familiar captar fondos propios sin ceder el control; los bancos las han usado por motivos regulatorios. Los \textbf{préstamos participativos} pagan un interés que depende de la actividad del prestatario, se cobran después de los acreedores comunes y, en el Derecho español, se consideran patrimonio neto a efectos de la reducción de capital y la liquidación de sociedades (art. $20$ del Real Decreto-ley 7/1996), lo que refuerza la solvencia de la empresa sin diluir su capital. Y los \textbf{bonos convertibles contingentes} (CoCos) que emiten los bancos se convierten en acciones o pierden valor cuando la ratio de capital ordinario del banco cae por debajo de un umbral, que en la Unión es como mínimo del $5,125\%$ (art. $54$ del Reglamento (UE) n.º 575/2013), de modo que absorben pérdidas cuando más se necesita ([\authorfont{Ver Tema 3.B.27}]).\footnote{%
    La deuda subordinada, que se cobra después de la deuda ordinaria, y la deuda perpetua, sin vencimiento, son otras formas intermedias; se describen, con los demás híbridos, en un anexo ([\authorfont{Ver Anexo \ref{anx:instrumentos}}]).
}

Hemos podido crear una división clara entre dos alternativas de financiación que, a primera vista, parecen tener implicaciones profundamente diferentes tanto en la composición del capital como en el futuro de la empresa: la financiación interna y la financiación externa. Esta pregunta puede resultar muy fácil de contestar para una persona corriente: evidentemente parece mejor auto financiar nuestros propios empréstitos que compartir nuestros beneficios con agentes externos, más teniendo en cuenta que nos veremos obligados a redimir nuestras deudas a riesgo de perder el colateral de garantía. Sin embargo, no parece riguroso aceptar este análisis superficial (y sesgado). Y esto es precisamente lo que pasaremos a contrastar ahora.

\textbf{¿Qué queremos decir?} De hecho, veremos como esta afirmación no solo es (aparentemente/analíticamente) falsa, sino que la composición del capital resulta irrelevante, siempre y cuando se cumplan una serie de supuestos, atendiendo al Teorema/Hipótesis de la irrelevancia en la composición/estructura del capital proppuesto por MODIGLIANI-MILLER. No obstante, presenta ciertas limitaciones que nos conducirán a pensar que (muy al pesar de los autores), el conocimieto vulgar es a menudo díficil de refutar.

Ahora bien, presentando todas estas fuentes financieras la pregunta es ¿por cuál debería la empresa optar? La literatura ha prestado atención a la deseabilidad de las 2 fuentes principales, es decir, las acciones y la deuda, y hay ciertos aspectos a tomar en consideración:
1º) Usar deuda para controlar la dirección: existe un importante problema informativo entre accionistas y directivos, pues los primeros no observan al 100\% el comportamiento de los segundos, y sus intereses pueden ser diferentes o incluso divergentes. La literatura señala que el uso de deuda puede ser una buena herramienta para controlar la actuación de los directivos. 
La idea es simple: los directivos tienen una preferencia por el uso de recursos internos, de modo que si la empresa atraviesa dificultades siempre estarán disponibles esos recursos para que el directivo siga manteniendo su remuneración. Los accionistas, en cambio, quieren maximizar su rendimiento, y eso puede suponer la necesidad de acudir a la financiación ajena para obtener recursos que se inviertan en proyectos rentables. La financiación mediante deuda da la tranquilidad al accionista de que los fondos se están invirtiendo y que, además, se están invirtiendo bien, porque el directivo sabe que si la empresa entra en dificultades económicas y no ha elegido correctamente los proyectos de inversión su remuneración estará en peligro. Por tanto, el endeudamiento facilita el control de la actuación del directivo.
¿Por qué esto no pasa con la financiación con acciones? Porque la empresa no tiene la obligación de pagar dividendos, por lo que si la empresa entra en dificultades y los proyectos de inversión no son adecuados lo que hará el directivo será, simplemente, no repartir dividendos y mantener su remuneración. Pero el pago de la deuda sí es obligatorio, por lo que no le quedará más remedio que vigilar bien dónde pone el dinero. 
2º) Signalling: los problemas informativos no solo existen entre accionistas y directivas, sino también entre la empresa y los inversores. La razón es que la empresa sabe más de si misma que los inversores que tienen que dar los fondos, por lo que ante la selección adversa existente se puede producir una reducción de los fondos aportados por parte de los inversores. Una forma de solucionar este problema, tal y como demostró Spence, es mediante el signalling o señalización, que consiste en que la parte informada envía una señala inequívoca a la parte carente de información sobre esa información que le gustaría tener. En este caso concreto, los directivos lanzarían al mercado señales que demuestren la solidez de la empresa en cuestión. La deuda es una herramienta que permite lanzar esta señal al mercado pues si una empresa está pidiendo deuda para financiar un proyecto que aparentemente parece muy arriesgado, está lanzando una señal de que los directivos sabe que dicho proyecto va a salir bien o de que, en caso contrario, la empresa tiene la solidez suficiente para pagar dicha deuda. De lo contrario no estarían dispuestos a pedir deuda sabiendo que pueden acabar perdiendo su trabajo y sus paquetes de compensación al tener que retribuir a los acreedores por un proyecto que ha sido un fracaso. 
3º) Pecking order theory: debido a que el problema de información asimétrica lleva a que los inversores actúen de una forma u otra, y que eso tenga costes para la empresa, la teoría del pecking order trata de analizar cómo deberían usarse las fuentes para hacer frente a los menores costes posibles derivados de ese problema de información asimétrica. La financiación que menores costes genera en este sentido es, obviamente, la financiación interna pues no precisa de la intervención de inversores externos y, por tanto, no está sometida a dichos problemas informativos. 
A continuación, si la empresa necesita más financiación de la que podría obtener con sus recursos internos podrá acceder a la financiación externa. Pero dado que la ampliación de capitales puede resultar costosa para la empresa ya que los inversores venderán las acciones, lo siguiente que deberían hacer las empresas es emitir deuda. Por último estaría la ampliación de capitales, porque es interpretada automáticamente como una señal negativa, pues la empresa solo estará dispuesta a ampliar capitales si la acción está sobrevalorada, lanzando esa señal a los inversores.  
Como vemos, entonces, las alternativas de financiación de las empresas son cuantiosas. Optar por una u otra puede tener un importante impacto en el mercado, especialmente por los problemas informativos existentes. Sin embargo, hay que tener en cuenta una cuestión fundamental al elegir entre las 2 principales fuentes de financiación externa y es que cada una tiene un rendimiento exigido diferente. Hasta ahora hemos comparado la deseabilidad en términos de problemas informativos, pero aún más importante es valorar la deseabilidad en términos de costes de financiación. Y es que hay que tener en cuenta que la financiación externa supone la necesidad de proporcionar un rendimiento a los inversores. Si asumimos que la empresa se financia fundamentalmente vía acciones y vía deuda debemos preguntarnos ¿hay alguna forma de minimizar el coste de financiación? 3.3 Estructura del capital de la empresa
Más allá de la relevancia que puedan tener las fuentes de financiación para solventar los problemas informativos señalados, la literatura se ha preguntado si existe una estructura óptima de capital. 
Al hacer referencia a este concepto estamos hablando de aquella distribución entre financiación a través de deuda y de acciones (se suele simplificar al máximo con estas dos alternativas que son las mayoritarias) tales que se minimice el coste medio del capital de la empresa o su WACC. ¿Por qué tendríamos este objetivo? Básicamente porque el objetivo del directivo debe ser maximizar el valor de la empresa, que se define como:
 
Por lo que si minimizamos el denominador, diseñando una estructura de capital que permita minimizar el WACC, estaríamos maximizando el valor de la empresa. Recordemos que el WACC o coste medio del capital nos indica el rendimiento exigido tanto por los accionistas como por los acreedores de la empresa (evidentemente la financiación interna no exige rendimiento, por eso no se tiene en cuenta). Por tanto, es el coste de financiación total de la empresa. Éste puede expresarse como: 
 
•	Postura tradicional
La perspectiva tradicional supone que hay una estructura financiera óptima para la cual el coste de capital es mínimo y, por tanto, el valor de la empresa es máximo. De esta forma, la estructura del pasivo va a ser relevante a la hora de determinar ese coste de capital y, como consecuencia, las decisiones de inversión y financiación están interrelacionadas.
Esta postura tradicional supone que:
+ Capital propio: el coste del capital propio es creciente con el endeudamiento. Esto se debe a que cuanto mayor sea la deuda contraída por la empresa, más variables serán los rendimientos que obtendrán los accionistas. Piensa que la deuda siempre hay que pagarla, pero los dividendos solo se dan si la empresa va bien. Por tanto, a mayor endeudamiento mayor riesgo financiero y, por ende, mayor rendimiento exigido. 
Además, la relación será no lineal, pues el riesgo no crece de forma lineal con el endeudamiento, y por tanto tampoco lo hará el coste del capital propio. Más concretamente, será convexa, pues crecerá a una tasa cada vez mayor. 
Adicionalmente, el coste de capital propio será superior al coste del capital ajeno, ya que los accionistas asumen un mayor riesgo que los acreedores: sus rendimientos no están garantizados de antemano y van por detrás en el orden de prelación en caso de bancarrota. 
+ Capital ajeno: el coste del capital ajeno es constante hasta un cierto nivel de endeudamiento, pero creciente para niveles superiores cuando el riesgo de insolvencia es más importante. 
+ Capital medio ponderado: el WACC será primero decreciente y después creciente. Primero es decreciente porque le vas dando más peso al capital ajeno y éste tiene un coste constante, por lo que le quitas peso al capital propio que tiene coste creciente. Sin embargo, luego comienza a crecer, porque el coste de ambas fuentes de financiación crece. 
 
Idea: la deuda es más barata que las acciones. Por tanto, al incrementar el endeudamiento, el WACC debe reducirse al estar dándole un mayor peso a la fuente de financiación más barata. Sin embargo, el aumento del endeudamiento también supone un mayor riesgo para los accionistas. Por tanto, conforme se aumenta el endeudamiento, se incrementa el rendimiento exigido por los accionistas.
De igual forma, al inicio el rendimiento de la deuda permanece inalterado; pero cuando el endeudamiento es ya muy elevado los acreedores perciben un mayor riesgo (es más posible ir a la bancarrota y aunque sean los primeros en el orden de prelación es posible que todos los activos no sean suficientes para compensar a todos los acreedores) por lo que el rendimiento exigido por los acreedores comienza a ascender también.
Por tanto, el WACC tiene un punto de inflexión evidente: inicialmente decrece gracias a que le damos mayor peso a la deuda, la cuál es más “barata” que las acciones y no se está encareciendo; pero llega un punto en el que comienza a ascender con el endeudamiento porque el rendimiento exigido por los accionistas ha aumentado mucho y porque los acreedores comienzan a exigir también un mayor rendimiento.  
Por ello, esta visión tradicional nos dice que efectivamente hay una estructura óptima de capital en el que podemos minimizar el WACC de la empresa.

Modigliani y Miller en la década de los 50 contradirán esta postura, desarrollando uno de los trabajos más importantes en el campo de las finanzas. Apuntan a que si los mercados de capitales son perfectos, el coste del capital medio ponderado y, por tanto, el valor de la empresa, son totalmente independientes de la composición de la estructura del capital de la compañía. 
Para ello se deben cumplir una serie de supuestos [recalca porque son muy exigentes]:
1. Los dividendos de la compañía van a ser constantes, a fin de abstraernos por completo de la política de dividendos. De igual forma, todos los beneficios disponibles son distribuidos en forma de dividendos, sin que exista reservas, a fin de evitar que su variación pudiera interferir en los resultados.
2. La estructura del capital solo está compuesta por acciones y deuda, y si se modifican sus pesos la variación será tal que el valor del pasivo sea siempre el mismo.
3. Los mercados de financieros son eficientes, lo que significa que no hay costes de transacción, todos tienen la misma información y todos los agentes pueden prestar y pedir prestado a la misma tasa. 
	4. No existen impuestos a las empresas ni a los individuos
	5. Las empresas no pueden ir a la bancarrota
En su primer artículo de 1958 desarrollan 3 proposiciones:
+ Proposición I: 
El valor de la empresa solo dependerá de la capacidad generadora de renta de sus activos sin importar en absoluto de dónde han procedido los recursos financieros que los han financiado.
La idea defendida por esta proposición, por tanto, se basa en que en un mercado de capitales perfecto el valor de una empresa solo depende del tamaño de los flujos de caja que se espera genera en el futuro y del coste de oportunidad del capital, pero no de cómo se reparten esos flujos de caja entre accionistas y obligacionistas. 
La clave estará en la condición de no arbitraje. Se apoyan en él para demostrar que si la única diferencia entre dos empresas es la composición de su estructura de capital, su valor deberá ser el mismo.
Supongamos 2 empresas que son capaces de generar los mismos FC y son idénticas en todo excepto en que una está apalancada (tiene deuda) y la otra no:
 
Vemos que un inversor que adquiera un X% de la empresa apalancada, y otro que adquiera un X% de la empresa no apalancada están percibiendo exactamente el mismo flujo de caja. Por ello estarán dispuestos a pagar el mismo precio por esa proporción X%, y por tanto el valor total de sus activos será el mismo, y por ende el valor de la empresa. 
Si el valor no fuese igual, el inversor podría realizar operaciones de arbitraje (homemade leverage) endeudándose al tipo de interés y comprando acciones de la empresa infravalorada, de manera que el hecho de que todos los inversores se lanzasen a explotar dichas actividades de arbitraje haría que el valor de ambas empresas coincidiese. Para que exista este arbitraje es crucial que el coste de financiación de individuos y empresas sea el mismo.

+ Proposición III:
Como consecuencia de las dos anteriores proposiciones, MM desarrollan la tercera y última de este trabajo de 1958, que establece que teniendo en cuenta todo lo anterior, las decisiones de inversión pueden realizarse de forma independiente de las decisiones de financiación. Es decir, si el proyecto de inversión es malo, o bueno, lo será independientemente de la estructura de capital de la empresa. 
Para decidir si invertir o no simplemente compara la tasa de retorno de la inversión con el coste de oportunidad del capital de la financiación obtenida, pero no te preocupes por cómo configuras las fuentes de financiación porque dicho coste de oportunidad de capital será siempre el mismo. 
De esta forma, este primer trabajo de MM proporcionó unos corolarios rompedores: da igual cómo se financie la empresa, el coste del capital va a ser siempre el mismo, por lo que no se debe perder el tiempo en buscar una estructura de capital óptima porque esta no existe. El arbitraje en todo este proceso va a ser una herramienta fundamental que lo va a sostener todo. 
El trabajo de MM formalizó una nueva forma de pensar acerca de los mercados financieros que ya había enunciado Williams, la cual se basa en el principio de conservación de valor: en un mercado de capitales perfecto el valor de una empresa solo depende del tamaño de los flujos de caja que se espera que genere en el futuro, y del coste de su capital, pero no de cómo se reparten esos flujos de caja entre accionistas y obligacionistas. El valor se mantiene independientemente de la estructura del capital. Parafraseando a Miller al recoger su premio Nobel “lo importante es el tamaño del pastel, no en cuántas porciones se parta éste”.
Ahora bien, tal y como Ross afirma, los directivos de empresas en la vida real se comportan de una manera muy diferente a la que esperaríamos con MM, de manera que la estructura de capital no se determina según el libre albedrío. Por ello, trabajos posteriores, incluso de los propios M-M (eran totalmente conscientes de esto) tratan de incorporar ciertos aspectos de la realidad que este trabajo no incorpora al basarse en unos supuestos excesivamente “idealizados”.

•	Los costes de insolvencia. Kim (1974)
La probabilidad de que la empresa no pueda hacer frente a las obligaciones contraídas por su endeudamiento aumenta cuanto mayor sea su tamaño. Al fin y al cabo el endeudamiento lleva a que haya unos costes a los que hay que hacer frente sí o sí, y si esas obligaciones no se ven satisfechas la empresa puede hacer frente a algún tipo de shock o limitación financiera cuyo máximo exponente sería la bancarrota.
Hasta ahora hemos supuesto que la empresa no caería en bancarrota pero ¿qué ocurre si levantamos este supuesto? Los procesos de bancarrota suponen unos costes elevados para todos aquellos involucrados:
+ Costes directos: costes administrativos, esfuerzos de accionistas para recibir la liquidación del dividendo, tasas legales, peritaje, notarios, obsolescencia de los activos por el tiempo del procedimiento legal hasta que se liquidan...
+ Costes indirectos: quedarse sin acceso a financiación, cancelación de pedidos por temor a que la empresa desaparezca, menor productividad por huelgas o infrautilización de recursos, estrategias “egoístas” de los accionistas (asumir inversiones excesivamente arriesgadas, pagar dividendos excesivos...) que afectan negativamente al valor de la empresa...
Ahora bien, lo cierto es que estos costes están presentes aunque la empresa no llegue finalmente a la bancarrota. Es decir, más que costes de bancarrota o de quiebra deberían llamarse costes de tensiones financieras. Por ejemplo, si están en una situación financiera delicada podrás reducir gastos en I+D, entrenamiento de personal... para hacer frente al pago de tus deudas, y encontrarás más difícil poder realizar nuevos proyectos que sean rentables y beneficiosos. Pero lo que hay que tener claro es que estas tensiones financieras son más propensas a ocurrir cuanto mayor sea el nivel de endeudamiento. Por tanto, la empresa tendrá que valorar un trade-off, pues el endeudamiento tendrá una ventaja fiscal, pero tiene importantes desventajas por los costes de tensiones financieros. Para niveles reducidos de endeudamiento la posibilidad de insolvencia es pequeña y, por tanto, las ventajas fiscales suelen dominar. Sin embargo, conforme el endeudamiento aumenta, los costes esperados de insolvencia van aumentando. Por tanto, si tenemos en cuenta el valor presente de los costes de quiebra y malfuncionamiento tenemos que el valor de una empresa endeudada será [gráficos Mascareñas]:
 
Los costes de quiebra los pagan los accionistas, pero los acreedores saben que los acabarán pagando ellos (menos activos se llevarán cuando liquiden) si dicha eventualidad se produce, por lo que piden una compensación por adelantado en forma de mayor rendimiento conforme la empresa se va endeudando más y más. Esto incrementa el WACC y reduce el valor de la empresa. 
Por ello, Kim argumenta que es de esperar que exista una estructura financiera óptima que no suponga un 100\% de endeudamiento. Se llega así a la “teoría del equilibrio de la estructura de capital” o “trade-off theory” que defiende que SÍ existirá una estructura de capital óptima en cada empresa.
Por tanto, el endeudamiento no debe superar un determinado nivel: hay una estructura de capital óptima. ¿Cuál será? El punto en el que el valor actual del ahorro fiscal marginal debido al endeudamiento adicional se viese exactamente compensado por el aumento marginal del valor de los costes de insolvencia. Como los costes de tensión financieros no se expresan de una manera precisa, no podemos saber exactamente cuál será ese nivel óptimo de endeudamiento. Paradójicamente este tour por tantos trabajos nos ha llevado a la situación de partida: algo de endeudamiento está bien, pero no demasiado. Resultado ambiguo que en realidad no dice nada.
Por tanto, como vemos, hemos observado cómo bajo unos supuestos simplificadores podríamos decir heroicos no existía una estructura de capital óptima. Sin embargo, los mismos Modligiani y Miller se esforzaron posteriormente en demostrar que las conclusiones obtenidas no deben convertirse en dogma, sino que al levantar ciertos supuestos de partida las conclusiones cambiaban de manera considerable. Por tanto, para determinar la estructura de capital óptima hay que tener en cuenta consideraciones como los impuestos empresariales, los impuestos a los inversores o los costes de bancarrota. El CFO deberá encargarse de tomar en consideración todas estas cuestiones para tratar de determinar si existe o no una estructura de capital óptima y, en caso afirmativo, cuál es ésta.
No obstante, hasta ahora nos hemos centrado en responder ¿cómo podemos minimizar el WACC para maximizar el valor de la empresa? Sin embargo podríamos prestar atención a otra cuestión fundamental que es maximizar el valor del equity. Para ello la política de dividendos cobra una importancia fundamental.

El punto crucial para el desarrollo del Teorema de Irrelevancia es la comprensión de sus supuestos. Entenderlos supuestos es la única manera de comprender bajo qué condiciones se mantiene el principal resultado: el valor de la empresa (definida corno la suma del valor de su deuda y su equity) no depende de la estructura del capital. MODIGLIANI y MILLER demuestran que, siempre y cuando:
\begin{lnum}
    \item \textbf{Mercados eficientes}. Los mercados financieros son perfectos y no tienen ningún tipo de fricción, por lo que el precio es el correcto y no hay oportunidades de arbitraje;
    \item \textbf{Fungibilidad perfecta del dinero}. El dinero es un bien fungible, lo que significa que es intercambiable, sustituible y no tiene identidad propia. Una unidad es una unidad venga de donde venga;
    \item \textbf{No hay distorsiones inducidas}. En especial, no hay impuestos de sociedades ni personales; y,
    \item \textbf{Deuda y Equity}. Consideremos que las empresas sólo pueden recurrir a dos tipos de financiación, deuda sin riesgo y acciones.
\end{lnum}

\subsubsection{Información asimétrica: Señalización (ROSS, 1977)}
Existen varias formas de incorporar un contexto de información incompleta al escenerio de las finanzas corporativas. No obstante, las lineas más destacadas en este ámbito (dentro de la ortodoxia) son: los problemas Principal-Agente y la introducción de costes de señalización o informacionales. El primero de ellos fue desarrollado por inicialmente por ROSS (1997). 

El autor considera que los directivos tienen mejor información que cualquier outsider y son los únicos que conocen las características de su empresa, ROSS plantea un modelo en el que el objetivo del directivo es mostrar, mediante su política de endeudamiento, la capacidad de las empresas de generar rendimientos futuros. Por tanto, la \textbf{emisión de deuda serviría de señal} a inversores \textbf{neutrales al riesgo}. 

Supongamos que existen dos tipos de empresas y que el directivo planifica en un horizonte de dos periodos. En el primer periodo, la empresa $A$ tendrá un valor de $V_A$ mientras que la empresa $B$ tendrá un valor de $V_B$, siendo $V_A>V_B$, con una probabilidad $p_A$ de que la empresa sea tipo $A$ y una probabilidad $(1-p_A)$ de que sea tipo $B$:

\eqblock{
\begin{aligned}
    &\underset{\text{\scriptsize esquema de compensación del directivo}}{\boxed{M=(1+r)\gamma_0V_0+\gamma_1\tilde V_1}}\qquad :\,\,\,\tilde V_1=
    \left\{\begin{aligned}
        &V_1 &&\text{si}\,\,B_0\leq V_1\\
        &V_1-L &&\text{si}\,\,B_0> V_1
    \end{aligned}\right.\\[1em]
    &\text{donde,}
    \left\{\begin{aligned}
        &B_0&&\equiv \text{Valor emitido de deuda en $0$}\\
        &L&&\equiv \text{Penalización por bancarrota}\\
        &V_i&&\equiv \underset{\text{\scriptsize(conocido por le directivo)}}{\text{Valor de la empresa en cada periodo}}\\
        &\gamma_i&&\equiv \underset{\text{\scriptsize (a cada periodo, teniendo en cuenta que depende su remuneración)}}{\text{Peso asignado por el directivo}}\\
        &r&&\equiv \underset{\text{\scriptsize (neutrales al riesgo)}}{\text{Tasa de descuento inversores}}
    \end{aligned}\right.\\[1em]
    &\Longrightarrow\qquad \boxed{\text{Mercados}:\mathbb E[V_0=\{A,B\}\,|\,B_0]}
\end{aligned}
}{}

\paragraph{Pecking order Theory: MYERS (1984)}
Una forma alternativa de incorporar el supuesto de información incompleta es a través de los costes de señalización (información). En el anterior ejemplo hemos incorporado este supuesto considerando la presencia de información asimétrica, es decir, la compatibilidad entre los intereses de los directivos y los agentes y como esto condiciona la situación financiera de la empresa. 

Sin embargo, también podemos considerar que son las diferentes fuentes de financiación externa las que están sujetas a problemas de información. Este es el planteamiento de la \textit{pecking order theory}, desarrollada inicialmente por MYERS y MAJLUF (1984) y MYERS (1984).

Brevemente, el punto central de este enfoque supone que en un contexto de información asimétrica donde los inversores no conocen el valor real de las acciones de la compañía ni el valor de las oportunidades de inversión la financiación vía ampliaciones de capital puede realizarse únicamente con un descuento, ya que emite dos señales a los mercados: 
\begin{la}
    \item Una positiva, que indica la existencia de oportunidades de inversión; y,
    \item Una negativa, ya que la propia emisión de acciones emite la señal de que las acciones están sobrevaloradas (al emitirse con descuento). 
\end{la}

El recurso a este \textbf{descuento} es simplemente un recurso analítico para replicar precisamente la evidencia empírica, puesto que la emisión de acciones parece tener un efecto negativo en el valor de la compañía (ASQUITH y MULLINS, 1986), que caen en torno a un 3\% después de la emisión de acciones.\footnote{%
    También puede pensarse como que la deuda minimiza la ventaja informativa de los directivos: ya que los directivos optimistas (piensan que las acciones de la empresa están infravaloradas) emitirán deuda en lugar de equity, mientras que los directivos pesimistas emitirán equity.
} De esta forma, el resultado teórico establece un orden en cuanto a las fuentes de financiación de la empresa: (i) financiación interna, ya que este tipo de financiación no está sujeto a costes informativos; (ii) emisión deuda; y, (iii) vía equity, cuando la deuda alcance ratios elevados. 

Además, permitiría explicar por qué el grueso de la financiación de las empresas proviene de la financiación vía deuda y por qué las empresas más rentables tienen menos deuda: no es porque tengan objetivos de deuda más conservadores, sino porque tiene a su disposición más financiación interna.

No obstante, estas teorías adolecen de algunas críticas: la primera es que, aunque la asimetría informativa entre directivos y otros agentes es crucial para el desarrollo de esta teoría, no se desarrollan las preferencias de los directivos, como si hacen otras teorías anteriormente presentadas (Ross, 1977). La segunda crítica es que el escenario que se presenta es demasiado simplista y estos problemas pueden mitigarse con otro tipo de instrumentos, como la deuda convertible.

Cabe entonces preguntarse ¿por qué alguien compraría equity? O, ¿quién comprará equity? Teniendo en cuenta lo anterior, la emisión de equity revelará una visión pesimista de la compañía, por lo que en el equilibrio de le dará prioridad a la emisión de deuda siempre y cuando la empresa no se acerque a puntos donde se anticipen costes derivados de un alto endeudamiento.


\paragraph{Otros modelos: LEELAND y PYLE (1977) y STULTZ (1990)}
Otros autores han desarrollado teorías diferentes con otras implicaciones apoyándose también en asimetrías informativas. 

Por ejemplo, LEELAND y PYLE (1977) presentan una versión alternativa de señalización donde un emprendedor averso al riesgo necesita financiar un proyecto con un mix de deuda y equity, siendo la calidad del proyecto solamente conocida por el propio emprendedor y no por agentes externos. Financiarse principalmente vía deuda y menos con equity es costoso, puesto que el emprendedor tendrá que mantener una mayor proporción de equity y tomar más riesgos en el proyecto, siendo este coste menor para los emprendedores con proyectos con alta calidad. El principal resultado de este modelo es que la emisión de deuda puede ser una señal de alta calidad del proyecto, ya que en los proyectos de mayor calidad el emprendedor estará menos dispuesto a deshacerse de su participación en la empresa.

Otros autores han señalado el rol de la deuda como mecanismo de control. Stulz ( 1990) presenta un modelo donde el propio directivo recibe beneficios privados de la inversión de la empresa ( este conocido como el fenómeno del empire building). Por tanto, el directivo puede verse tentado a llevar a cabo proyectos e inversiones incluso si tienen un valor presente neto negativo. En este caso, la financiación vía deuda tiene un tradeotf simple: 

Por una parte, la financiación vía deuda crea una empresa más apalancada y, de cierto modo, disciplina al directivo, puesto que el repago de la deuda conlleva una serie de obligaciones, deadlines y compromisos. 

Por otra parte, la financiación vía deuda, puesto que restringe al directivo, limita la flexibilidad empresarial frente a una compañía con menor deuda. De este modo, el uso de la deuda o equity dependerá tanto de las posibles perspectivas de crecimiento de la empresa como de la capacidad del directivo de llevar a cabo proyectos de valor presente negativo para aumentar su utilidad a expensas de los accionistas.

Estos autores ampliaron su modelo básico introduciendo impuestos de sociedades, demostrando que la estructura de capital de la empresa sí afecta al valor de ésta. Sobre la Hipótesis y las conclusiones originales, al introducir impuesto de sociedades el endeudamiento genera un aumento del valor de la empresa debido a la deducibilidad de los intereses de la deuda del impuesto de sociedades. 

\eqblock{
\begin{aligned}
    r_{WACC}=r_E\frac{E}{V}+r_D\frac{D}{V}(1-T)\quad\Longrightarrow\quad \boxed{r_E=r_{WACC}+[r_{WACC}-r_D(1-T)]\frac{D}{E}}
\end{aligned}
}{Proposición II: Coste del capital accionario. Teorema de la irrelevancia (I): Composición}

La idea principal es que las empresas más endeudadas pueden deducirse los gastos de intereses del impuesto de sociedades, reduciendo así la base imponible. De esta manera, la deductibilidad de los gastos por intereses entra en el modelo como una especie de subsidio que incentiva el endeudamiento. 

Además, si relajamos el supuesto de bancarrota llegamos a conclusiones más interesantes. Suponer que los costes de bancarrota son nulos para los inversores con una buena cartera diversificada (donde las minusvalías se compensan con las plusvalías) no aporta gran información. Pero si consideramos que estos costes no son nulos, entonces el efecto de estos se proyecta directamente en una disminución del valor de la empresa a medida que ésta aumenta su endeudamiento.\footnote{%
    ¿Cómo de importantes son estos costes? GRAHAM (2000) calcula que los beneficios derivados de la deducción de los intereses de la deuda del impuesto de sociedades ascenderían al 9,7\% del valor de la empresa (o al 4,3\% si descontamos los impuestos personales). Para GRAHAM, las empresas grandes estarían actuando de manera conservadora en cuanto a la emisión de deuda. Por otro lado, ALMEIDA y PHILLIPON (2007) calculan los costes de bancarrota en aproximadamente el 4.5\% del valor de la empresa, lo que cancelaría los beneficios del endeudamiento derivados del sistema impositivo y ayudaría a explicar el comportamiento conservador de las empresas subrayado por Graham.
} Es decir, opera en la dirección contraria que los impuestos corporativos y se genera un \textit{trade-off} entre estos.

Como vemos, el endeudamiento comenzaría a aumentar el valor de la empresa, pero solamente hasta cierto punto. Pasado dicho punto, el valor de la empresa comienza a disminuir.

\begin{specialblock}{MILLER (1977) -- Impuestos personales: sobre dividendos}
    Más adelante, Miller (1977) incluyó impuestos personales afirmando que los impuestos pagados por los inversores pueden anular los pagados por las empresas, lo que nos devuelve al punto de partida de MODIGLIANI-MILLER (1958): no debería existir una estructura óptima de capital.

    Para esto, MILLER introduce: (i) un impuesto de sociedades ($T_c$); (ii) un impuesto a los dividendos ($T_e$); y, (iii) un impuesto a la renta de los bonos ($T_d$). En este caso, el valor del endeudamiento (Go) sería:

    $$G_D=\left[1-\frac{(1-T_c)(1-T_e)}{1-T_d}\right]D$$
    
    Podríamos contemplar varios escenarios:
    \begin{la}
        \item Cuando el impuesto a los dividendos y la renta (bonos) son equivalentes $( 1 - T_e)/(1-T_d) = 1$. Entonces, la introducción de impuestos personales no altera el resultado previo y continuaríamos de cierto modo en el escenario de MILLER (1963), donde el valor delendeudamiento sería igual a $T_cD$;
        \item Cuando el impuesto de dividendos y de la renta son $(1-T_e)/(1-T_d)=(1-T_c)$, entonces la introducción de impuestos personales cancela las ganancias de la imposición de sociedades y la estructura del capital vuelve a ser irrelevante; y,
        \item Para otros valores, la introducción de los impuestos personales podría amplificar las ganancias del endeudamiento $(T_e>T_d)$ o mitigarlo $(T_d>T_e)$ 
    \end{la}    
    
    Para Miller, además, existen dos conceptos que son clave para entender las implicaciones de sus aportaciones: 
    \begin{lnum}
        \item \textbf{El inversor marginal}. Es necesario tener en cuenta que los impuestos personales son, generalmente, progresivos; por eso, en un universo con inversores de diferentes rentas, los individuos se enfrentarán a diferentes tipos impositivos y, por tanto, a medida que varíe su $T_p$, también se encontrarán en diferentes puntos de la fórmula. El inversor marginal es aquel cuyo beneficio del endeudamiento de la empresa es de 0; es decir, aquel en el que se cumple $(1-T_e)/(1-T_d)=(1-T_c)$. 
        \item \textbf{Equilibrio de MILLER}. Establece que el sector corporativo en su conjunto se situará en un punto donde $G_D$ sea $0$ para el inversor marginal, estableciendo un nivel de deuda/equity determinado para el sector en su conjunto. No obstante, a nivel individual de empresa, la decisión de deuda/equity es irrelevante. 
    \end{lnum}

    Imaginemos un sector empresarial con un determinado nivel de deuda que se encuentra inicialmente en equilibrio. Al estar en equilibrio, todas las empresas (independientemente de su nivel de endeudamiento) también se encuentran en un punto donde su valor no aumentaría si su endeudamiento aumenta o disminuye. Supongamos también una demanda de bonos que depende positivamente del tipo de interés. En este contexto, si el nivel de deuda corporativa agregada aumentase, para vaciar los mercados, la demanda de bonos también debería aumentar, correspondiendo con un tipo de interés mayor. Al aumentar el tipo de interés, aumenta el coste de la deuda y las empresas encontrarán rentable sustituir deuda por equity hasta el punto inicial de endeudamiento que correspondía al equilibrio.

    Sin embargo ¿Cómo puede ser compatible un nivel de deuda agregada única y de equilibrio con la irrelevancia de la estructura de capital de cada empresa de manera individual? La clave está en los diferentes tipos de inversores: los inversores con menores (mayores) tipos en el impuesto sobre la renta necesitarán rendimientos menores (mayores) de la deuda para ser indiferentes entre deuda y equity; la lógica es que el rendimiento que le importa a los inversores es el rendimiento neto de impuestos, por lo que, suponiendo un impuesto sobre dividendos constante, para igualar el rendimiento de deuda y equity los inversores en los tramos más elevados del impuesto sobre la renta exigirán rendimientos de la deuda mayores. En este contexto, las empresas menos endeudadas colocarán su deuda en los agentes que se enfrentan a menores tipos impositivos de renta. Si la empresa aumenta su endeudamiento, tendrá que acceder a in versores cada vez que se enfrentan a tipos impositivos mayores (y que, por tanto, exigirán un rendimiento mayor de su deuda). Por lo tanto, las ventajas del endeudamiento de la empresa derivadas del escudo fiscal quedarían canceladas por el mayor rendimiento de la deuda que exigirán los nuevos inversores.
    
    De este modo, MILLER demuestra, en un nuevo contexto y con impuestos, la irrelevan cia de la estructura del capital. Posteriormente, DEANGELO y MASULIS (1980) extendieron el modelo de MILLER (1977) para incorporar escudos fiscales diferentes al de la deuda (por ejemplo, la incorporación de depreciaciones, deducciones fiscales de algunas inversiones, etc.). El resultado de su modelo es que sí existe un nivel óptimo de deuda distinto de $0$. Aunque ambos marcos conceptuales tienen sustentos teóricos sólidos, la evidencia empírica es mixta y es complejo obtener los tipos impositivos marginales. 
    
    Una excepción es GRAHAM (2000), donde el valor del endeudamiento considerando todos los impuestos era positivo.
\end{specialblock}

Cuando las empresas disponen de FCL positivos para el capital propio debe decidir cómo usarlos. Si tienen oportunidades de inversión con VAN positivo pueden reinvertir ese dinero para incrementar el valor de la compañía. Sin embargo, las empresas ven que en ocasiones tienen más efectivo del que necesitan para financiar sus oportunidades de inversión. Ese excedente de efectivo lo pueden conservar como reservas, o lo pueden pagar a sus accionistas.
La política de dividendos hará referencia entonces a cómo la empresa decide redistribuir sus FCL entre accionistas pagando dividendos. La pregunta es ¿afecta esta política de dividendos al precio de la acción?
Al fin y al cabo, si el precio de una acción no es más que la suma descontada de los dividendos futuros, podría parecer a simple vista que si incrementamos la cuantía de los dividendos que pagamos los accionistas verán maximizado el precio de la acción. ¿Pero es esto realmente así? 
Para responder a esta pregunta, resulta fundamental aislar la política de dividendos de cualquier otra decisión que tome la empresa. En particular:
+ De decisiones de inversión: por ejemplo, supongamos que una empresa ha retenido beneficios para invertirlos en proyectos de inversión pero, llegado el momento, descubre que dichos proyectos que parecían seguros y muy rentables se han evaporado. La empresa decide entonces repartir esos beneficios retenidos entre sus accionistas en forma de dividendos. ¿Cómo podemos separar el efecto del aumento de los dividendos del impacto de la insatisfacción de las pérdidas de oportunidades de crecimiento? Ambas cuestiones podrían impactar sobre el precio de la acción. Por ello las decisiones de inversión tienen que permanecer inalteradas ante cambios en la política de dividendos.
+ De decisiones de financiación: la empresa podría pedir prestado a acreedores para repartir ese dinero en forma de dividendos a sus accionistas. En este caso la política de dividendos estará íntimamente relacionada con las decisiones de endeudamiento, siendo difícil separar el efecto sobre el precio de la acción del cambio en los dividendos y del cambio en el endeudamiento. Las decisiones de financiación tienen que permanecer inalteradas. 
Por tanto debemos preguntarnos ¿cuál es el efecto del cambio en la política de dividendos dada las decisiones de inversión y de endeudamiento de la empresa? Démonos cuenta que suponer esto nos lleva a que las dos formas de pagar al accionista (dividendos o recompra de acciones) estén íntimamente relacionadas. Más concretamente:
- Supongamos que la empresa decide incrementar el dividendo a pagar. No obstante, el gasto de inversión y el endeudamiento están fijos, por lo que solo hay una posible forma de obtener ese dinero para pagar dividendos: emitir acciones.
- Supongamos que la empresa decide reducir el dividendo a pagar. Si los gastos de inversión y el endeudamiento están fijos solo hay una forma de usar ese dinero: recomprar acciones.
Por tanto, la política de dividendos requiere encontrar el equilibrio entre mayores o menores pagos de dividendos y la emisión o recompra de acciones. En este sentido, la literatura no ha mostrado unanimidad sobre cómo puede afectar esta política de dividendos al precio de la acción. 
Si partimos de un punto de vista “intuitivo”, si el precio de la acción no es más que el valor descontado de los dividendos que se esperan obtener, entonces una política de dividendos más generosa en el pago de dividendos debería aumentar el precio de la acción. De aquí subyace la idea clásica de “un euro de dividendos vale lo mismo que tres en reservas”. Por tanto, si lo que los directivos quieren es maximizar la riqueza de los accionistas simplemente habrá que maximizar los dividendos pagados a los accionistas.
Sin embargo, MM en un relevante trabajo demostraron cómo esta visión puede estar altamente equivocada.

Gordon y Lintner señalan no obstante que el rendimiento exigido cae al repartirse el pago de dividendos porque los pagos de dividendos son más seguros que los que provienen de ganancias de capital. Esto es, desde el punto de vista del accionista 1 euro en dividendos vale más que 1 de ganancias de capital. La idea es que la reducción del pago de dividendos manteniendo el valor de la empresa constante hace que esos dividendos mayores se perciban en el futuro, volviendo a dichos pagos más arriesgados que los realizados actualmente y que, por tanto, aumente el rendimiento exigido. 
2º) Efecto de los impuestos: cuando el tratamiento fiscal a los dividendos y a las ganancias de capital es diferente la política de dividendos sí que podrá tener un impacto sobre el precio de la acción. Por ejemplo, supongamos 2 empresas iguales (y que por tanto tienen que dar el mismo rendimiento después de impuestos) solo que una paga dividendos y la otra no. En la que no se pagan dividendos el rendimiento vendrá únicamente de la ganancia de capital. En la que pagan dividendos, el rendimiento vendrá de la ganancia del capital y del dividendo, ambos con su correspondiente gravamen. Pues bien, se puede demostrar (por detrás, Mascareñas) que si el rendimiento sobre los dividendos son menores, el precio actual de la empresa que sí paga dividendos será mayor para que haya una menor ganancia de capital que compense el hecho de que percibes un dividendo que se grava más beneficiosamente. 
3º) Efecto clientela: hemos dicho que los inversores podían lograr su propia política de dividendos en un mercado de capitales eficiente. Sin embargo, si hay costes de transacción, el hecho de comprar o vender títulos le resulta caro al inversor. Esto puede llevar a que el inversor no quiera diseñar su propia política de dividendos, de tal forma que buscarán una empresa que sí tenga una política de dividendos como la que el inversor desea. Así, si por ejemplo un inversor quiere grandes dividendos líquidos ahora buscará empresas que se los proporcionen y viceversa. Por ello, esta teoría recibe el nombre de efecto clientela, pues cada empresa tendrá sus propios inversores-clientes. Si el % de inversores que desean dividendos presentes grandes es igual al % que prefiere dividendos presentes pequeños la política de dividendos no tendría efecto sobre el precio, simplemente se turnarían unos inversores por otro. Pero si el % está desequilibrado usar una política u otra sí que puede tener un impacto en el precio de la acción. 
Esto puede ser extrapolado al comportamiento del mercado. En una situación en la que el mercado es alcista es de esperar que los inversores prefieran pocos dividendos pero grandes ganancias de capital. Sin embargo, si el mercado es bajista los inversores podrían tener preferencia por dividendos elevados y constantes y no por ganancias de capital (lo que les da más tranquilidad al ser una liquidez “segura”).

\subsubsection{Impuestos} 
Gran parte de la literatura sobre la política de dividendos se centra en la importancia de los impuestos e intenta conciliar varias de las observaciones empíricas ya comentadas: las empresas reparten una gran parte de sus beneficios en forma de dividendos y muchos de los beneficiarios se encuentran en tramos impositivos elevados. Para ello, vamos a ver cómo los impuestos pueden explicar estos hechos estilizados tanto en modelos estáticos como en modelos dinámicos: En los modelos estáticos, vamos a explorar el impacto de los impuestos en el caso particular donde todos los inversores se enfrentan a los mismos impuestos y los impuestos sobre dividendos (renta personal) son superiores a los impuestos sobre el capital, como suele ser el caso. En este contexto, la política óptima de dividendos es la de no repartir dividendos, ya que todos los inversores preferirían recibir los ingresos vía recompra de acciones (ganancias de capital). No obstante, esta no es la política habitual de muchas empresas, tal y como se ha visto en los hechos estilizados. Por eso, hay que considerar también:
• Que las ganancias de capital pueden posponerse y el inversor puede elegir el timing de realización de las ganancias, puesto que puede elegir el momento de vender sus acciones. No es el caso de los dividendos, donde el inversor individual no tiene poder de decisión. Como algunos autores han mostrado (Constantinides ( 1984)), los inversores estarían dispuestos a pagar por esta opción de posponer los impuestos sobre las ganancias de capital.
• Algunas instituciones financieras (fondos de pensiones, por ej emplo) no pagan impuestos, por lo que no tienen, a priori, motivos para preferir los ingresos derivados de las ganancias de capital frente a dividendos. Además, en algunas jurisdicciones (Estados Unidos o la Unión Europea, por ej emplo), el reparto de dividendos entre empresas del mismo grupo está exentos de impuestos. No obstante, también es necesario tener en cuenta que, en un mundo más complejo, algunos inversores institucionales tienen restricciones a la hora de adquirir acciones de empresas que no reparten dividendos.

En este contexto (inversores que se enfrentan a diferentes tipos impositivos de dividendos y ganancias del capital y diferentes empresas con diferentes políticas de reparto de dividendos) se desarrollan los cliente/e mode/s estáticos: según esta familia de modelos, los inversores que sean personas físicas se enfrentan a altos tipos impositivos de dividendos y preferirán empresas con un bajo reparto de dividendos; las empresas, al estar exentas de la tributación de dividendos de otras empresas pero tienen que tributar por sus ganancias de capital invertirán en empresas con un mayor reparto de dividendos; y los inversores institucionales invertirán en todo tipo de empresas, puesto que no pagan impuestos. De nuevo, se puede desarrollar un concepto de equilibrio semejante al equilibrio de Miller de tal manera que el valor de la compañía sea independiente de su política de dividendos: imaginemos que, en empresas idénticas, las empresas que no reparten dividendos tienen un valor de 1050€ y las empresas que reparten dividendos un valor de 1000€: en este contexto, algunas empresas qu reparten dividendos tendrían el incentivo a dejar de repartir dividendos, incrementando la oferta de las acciones de empresas que no reparten dividendos y disminuyendo su precio hasta que éstos se igualen. Al respecto, algunos autores (Blume, Crockett y Friend (1974)) han encontrado cierta relación entre los tipos impositivos a los que se enfrentan los inversores y las políticas de reparto de las acciones que compran (los individuos que se enfrentan a mayores tipos impositivos de renta personal tienen acciones de empresas que reparten menos dividendos). No obstante, los modelos dinámicos dan una perspectiva diferente. Si se permite la compraventa de acciones en diferentes puntos del tiempo, entonces las predicciones de los modelos estáticos dejan de ser válidas. Desde una perspectiva puramente teórica, Miller y Scholes ( 1978) demostraron que, siguiendo diferentes estrategias dinámicas basadas en vender y comprar a instituciones libres de impuestos, en ausencia de costes de transacción se podrían evitar todo tipo de impuestos. Esto haría que los diferentes tipos impositivos de dividendos y ganancias de capital sean irrelevantes. No obstante, la evidencia muestra que no se hace un uso extensivo de este tipo de estrategias. Por otro lado, las estrategias dinámicas post día de dividendos sí cuentan con apoyo empírico: la idea básica de estos modelos (Kalai (1 982)) es que el arbitraje dinámico puede eliminar los efectos de los impuestos en los precios, ya que los inversores que se enfrentan al mismo tipo impositivo de dividendos y ganancias del capital pueden comprar las acciones justo antes del reparto de dividendos, cobrar el dividendo y venderlas justo después. En mercados perfectos, estos inversores venderán con un descuento igual al dividendo cobrado. La evidencia empírica apoya la existencia de este tipo de estrategias dinámicas: Michaely y Vita ( 1996) muestran un volumen de transacciones mucho mayor el día de reparto de dividendos y al día siguiente. Este hecho contradice los modelos estáticos descritos anteriormente, donde este fenómeno no debería suceder. Estos autores también demuestran que las acciones con menos costes de transacción registraron mayores incrementos en el volumen de transacciones, lo cual también apoya las predicciones de los modelos dinámicos (mientras mayores sean los costes de transacción de las acciones, menos se incentiva el uso de este tipo de estrategias). Además, la evidencia muestra que inversores situados en tramos elevados de los impuestos a los dividendos reciben una gran cantidad de dividendos, lo cual también contradice a los modelos estáticos.

La validación de este tipo de modelos dinámicos nos devuelve, de cierto modo, al punto de partida donde la política de dividendos es irrelevante.


\subsubsection{Información asimétrica}
Los mercados de capitales son imperfectos no solamente porque los agentes tengan que pagar impuestos o porque existan costes de transacción, sino también porque, si los insiders tienen más información sobre el futuro de la empresa, la política de reparto de dividendos puede contener información relevante para los inversores y afectar a la valoración de la empresa. En este sentido, el modelo de Bhattacharya ( 1979) supone un gran paso adelante, ya que permite formalizar un modelo con información asimétric a donde el reparto de dividendos se usa para señalizar los resultados esperados de la empresa. Este es un modelo de dos periodos donde el directivo de la empresa busca maximizar el valor de la empresa para los accionistas. En el primer periodo, el directivo invierte en un proyecto del cual solamente éste conoce su rentabilidad esperada (los inversores no la conocen). En este momento, el directivo también se compromete con una política de dividendos. En el segundo periodo, el proyecto genera un beneficio que se utiliza para pagar los dividendos comprometidos en el primer periodo. Un supuesto crucial del modelo es que, si el beneficio es insuficiente para cubrir los dividendos, la empresa debe recurrir a financiación externa costosa (puesto que está sujeta a costes de transacción) para pagar los dividendos. En el primer periodo, los directivos pueden señalizar que el proyecto de la empresa es bueno comprometiéndose a repartir un dividendo alto en el siguiente periodo. Si la empresa tiene efectivamente un buen proyecto, los dividendos se reparten. Si una empresa tiene un buen proyecto, normalmente podrá pagar el dividendo sin recurrir a financiación externa y, por lo tanto, no tendrá que soportar los costes de transacción asociados. En el equilibrio separador, una empresa con un mal proyecto no reparte tantos dividendos porque tendrá que recurrir más a menudo a la financiación externa y, por tanto, tendrá que soportar unos costes de transacción más elevados. En este caso, bajo la existencia de un equilibrio separador, el proyecto más rentable recibe una mayor valoración debido a que su política de reparto de dividendos actúa como elemento señalizador. Además, dado que el tradeoff crítico en el modelo es entre los costes de transacción y el valor de la empresa, estos resultados se mantienen incluso si existen impuestos a los dividendos. Esta familia de modelos tiene un gran punto a favor y otro punto en contra: el principal punto a favor es que permite explicar la reacción positiva de los mercados ante un aumento del reparto dedividendos o la recompra de acciones. No obstante, para estos modelos, él reparto de dividendosy la recompra de acciones son perfectamente sustitutivas, por lo que no son capaces de explicarpor qué, si los impuestos a los dividendos son mayores que a las ganancias de capital, las empresas continúan utilizando los repartos de dividendos como señalización cuando la recompra de acciones podría tener el mismo efecto. No obstante, existen otras versiones de esta familia de modelos donde reparto de dividendos y recompra de acciones no son sustitutivos. Por ejemplo, el modelo de Allen, Bernardo y Welch (2000) tiene en común con el resto de los modelos de esta familia que el reparto de dividendos señaliza buenos resultados futuros. Sin embargo, la principal diferencia es que aquí reparto de dividendos y recompra de acciones no son sustitutivos, por lo que acomoda el hecho de que los dividendos se enfrenten a una imposición más alta que las ganancias de capital y que aun así las empresas utilicen el reparto de dividendos y no la recompra de acciones como señalización. La principal diferencia en este modelo es que las empresas estarán interesadas en atraer a los inversores mejores infomiados para revelar su valor: como generalmente las instituciones que no pagan impuestos (fondos de pensiones, mutual funds, ... ) están mejor infomiadas, la manera de atraer a este tipo de inversores es mediante el pago de dividendos y no mediante la recompra de acciones, ya que estas instituciones no sufren la desventaja fiscal de los dividendos frente a las ganancias de capital y, en muchas ocasiones, tienen reglas de inversión que les impiden invertir en empresas que no pagan dividendos. Según este modelo, las empresas pagarían dividendos y no recomprarían sus acciones para atraer a estos inversores con mayor capacidad de detectar la calidad de la empresa, mientras que las empresas con inversiones menos rentables no tienen incentivos a replicar dicha estrategia para que no se revele su tipo de empresa. Además, este modelo pennite justificar teóricamente la suavización de la política de dividendos: las empresas que pagan dividendos serán reacios a disminuirlos, ya que podrían expulsar a este tipo de instituciones y verse penalizados por ello, por lo que tienen una preferencia por mantener el reparto de dividendos en el tiempo. De este modo, las teorías de señalización contienen 3 hipótesis que pued��n ser validadas empíricamente: 

1. Los cambios en las políticas de dividendos deben ir acompañados de cambios en los ingresos en la misma dirección en los siguientes periodos (si no fuese así, la política de dividendos no se estaría utilizando para señalizar). 

2. Cambi9s no anticipados en los dividendos deben ir acompañados por variaciones en el precio de las acciones en la misma dirección. 

3. Cambios no anticipados en los dividendos deben ir acompañados por revisiones en la misma dirección de los ingresos esperados de la empresa. 

Es necesario tener en cuenta que estas hipótesis son necesarias, pero no suficientes para garantizar que los modelos de señalización representan los mecanismos por los cuales se reparten dividendos. Por regla general, la literatura se ha concentrado en testear la segunda hipótesis, generalmente refrendándola: por ejemplo, Grullon, · Michaely y Swaminathan (2002) han encontrado un retomo extraordinario de las acciones de un 1.34\% cuando se aumentan significativamente las políticas de reparto de dividendos y de un -3.71\% cuando se disminuyen. Esta asimetría entre aumentos y reducciones de dividendos implica, según estas teorías, que los recortes de dividendos contienen más infonnación sobre las perspectivas de la empresa que los incrementos. Por otra parte, otra familia de modelos son los modelos de agencia, que inciden en otro problema no señalado anterionnente, y es que los intereses de los directivos y accionistas no tienen necesariamente que estar alineados. Este hecho es algo que ya se ha visto anterionnente, y su justificación nace en que los directivos pueden querer llevar a cabo proyectos con rentabilidad negativa porque les deriva utilidad (por ejemplo, comprar un jet privado para el directivo de la empresa) o por motivos de empire building. En este caso, autores como Grossman y Hart (1980) han sugerido que los accionistas pueden minimizar estos problemas repartiendo dividendos y disminuyendo la cantidad de recursos bajo el control total del directivo. En este caso, el reparto de dividendos disminuye la discrecionalidad del directivo en el uso de los fondos. No obstante, estos modelos no resuelven por qué el mecanismo para disciplinar al directivo es el reparto de dividendos y no la recompra de acciones o, como se vio en el apartado anterior, la emisión de deuda. Una potencial explicación es que, como hemos visto anteriormente, los precios de las acciones suelen disminuir al disminuir la política de dividendos, así como hemos visto algunas justificaciones teóricas que explican la decisión de suavizar la política de dividendos: enestos casos, el reparto de dividendos disciplina más al directivo que otras alternativas. Los modelos de agencia tienen más dificultades para derivar implicaciones claras. Por ejemplo ¿Qué debería suceder después de un aumento de los dividendos en estos modelos? Depende, por ejemplo, si esta decisión se toma después de que el directivo invierta en proyectos con rentabilidad negativa, la rentabilidad debería aumentar. No obstante, si le decisión se toma antes de que el directivo invierta en proyectos de rentabilidad negativa, los resultados son ambiguos. Una clara implicación de estos modelos es que estos problemas son más pronunciados en empresas estables, con más tesorería y sin muchas oportunidades de crecimiento, ya que son empresas donde las oportunidades para llevar a cabo inversiones con rentabilidad negativa son mayores. Aunque existe evidencia a favor de esta hipótesis, tampoco es unánime.

\anexo{Instrumentos de financiación}\label{anx:instrumentos}
Este anexo reúne los instrumentos con que se materializan las fuentes de financiación del primer bloque: sus rasgos jurídicos y operativos, sus variantes y los criterios prácticos que orientan la elección entre ellos. \subsection{Autofinanciación}
\subsubsection{Reservas}
Las reservas son los beneficios de ejercicios anteriores que no se han repartido como dividendos. Pueden ser legales, estatutarias o voluntarias. La reserva legal obliga a destinar el $10\%$ del beneficio del ejercicio hasta que alcance al menos el $20\%$ del capital social (art. $274$ de la Ley de Sociedades de Capital); las estatutarias dependen de lo que dispongan los estatutos, y las voluntarias, de la decisión de la junta. Todas aumentan los fondos propios y constituyen la autofinanciación de enriquecimiento, el componente principal de la autofinanciación.

\subsubsection{Amortizaciones y provisiones}
Las amortizaciones imputan cada año a los gastos la depreciación de los activos. Pueden calcularse con cuotas fijas, por el método lineal, $A=(V_0-V_R)/n$, donde $V_0$ es el valor de adquisición, $V_R$ el valor residual y $n$ la vida útil, o con métodos acelerados, cuyas cuotas son mayores al principio, por ejemplo aplicando un coeficiente constante a una base que decrece. Sin ellas, la reposición de un activo al final de su vida recaería de golpe sobre el resultado y la tesorería de un solo ejercicio. Las provisiones son retenciones de beneficio vinculadas a riesgos concretos de la actividad, que pueden materializarse o no; a diferencia de las reservas, se registran en el pasivo. Unas y otras evitan que el patrimonio neto disminuya, por lo que forman la autofinanciación de mantenimiento.

\subsubsection{Efecto multiplicador}
Si la empresa quiere mantener constante su ratio de endeudamiento, $\alpha=D/E$, todo aumento de los fondos propios le permite, y le exige, aumentar su deuda en $\alpha$ veces esa cuantía. Los recursos totales varían entonces en
$$\Delta(E+D)=(1+\alpha)\,\Delta E,$$
de modo que la autofinanciación tiene un efecto expansivo sobre la capacidad de inversión, mayor cuanto mayor sea el endeudamiento que la empresa quiere mantener.

\subsubsection{Efecto LOHMANN-RUCHTI}
Si se cumplen cuatro supuestos (los activos no quedan obsoletos, de modo que su productividad no cambia durante su vida; son perfectamente divisibles; no hay inflación; y la amortización libera recursos financieros, porque los ingresos cubren los gastos), las cuotas de amortización reinvertidas no solo mantienen la capacidad productiva, sino que la aumentan. Una empresa que compra $k$ activos de vida útil $n$ y valor residual nulo, y los amortiza linealmente, recupera cada año $k/n$ veces el precio de un activo; si reinvierte esa cuota, compra activos nuevos que a su vez se amortizan y se reinvierten. La capacidad crece hasta estabilizarse, con amortización lineal, en torno a $2n/(n+1)$ veces la inicial.\textcolor{orange}{\textsuperscript{[Probable]}}\footnote{%
    Con una flota de diez barcos de vida útil de diez años, la cuota del primer año equivale al precio de un barco, y la flota pasa a once; con el tiempo, se estabilizaría en torno a $18$ barcos ($2\cdot 10/11\approx 1,8$ veces la inicial). El efecto descansa en que la capacidad de un activo no disminuye con su valor contable: si los activos pierden productividad con la edad o hay inflación, se reduce o desaparece.
}

\subsubsection{Ventajas e inconvenientes}
La autofinanciación da a la empresa autonomía y libertad de acción, porque no tiene que acudir a los mercados de capitales ni negociar con inversores; es especialmente valiosa cuando hay problemas de información, que encarecen la financiación externa; y, para las pymes, es la principal alternativa al crédito bancario para financiarse a largo plazo. Sus inconvenientes son que puede ser insuficiente, porque las necesidades de inversión se presentan de forma discontinua y la autofinanciación se genera de forma gradual; que, aun siendo suficiente, no es siempre la fuente menos costosa; que, al no tener un coste explícito, puede llevar a acometer inversiones poco rentables; y que retener beneficios reduce los dividendos, lo que, según la teoría que se adopte, puede afectar a la cotización, como veremos al tratar la política de dividendos.

\subsection{Acciones}
\subsubsection{Características}
Las acciones son partes alícuotas del capital social. Son títulos de renta variable: su remuneración depende de la evolución económica y financiera del emisor y no de una relación contractual de pagos. No tienen vencimiento, por lo que el accionista solo recupera su aportación vendiéndolas en el mercado secundario o en la liquidación de la sociedad. Dan derechos económicos (participación en los beneficios, derecho de preferencia en las ampliaciones) y políticos (voto, asistencia a la junta e información), y en caso de liquidación se cobran después de todos los acreedores. Contablemente, forman parte del patrimonio neto del emisor y del activo del inversor ([\authorfont{Ver Tema 3.B.23}]).

\subsubsection{Clases}
Según la forma de designar al titular, las acciones son nominativas o al portador. Según los derechos que incorporan, son ordinarias, preferentes (con algún derecho adicional, como un mayor dividendo o preferencia en el cobro) o sin voto. Según el precio de emisión, se emiten a la par, por su valor nominal, o sobre la par, con prima de emisión; en España no pueden emitirse por debajo de su valor nominal (art. $59$ de la Ley de Sociedades de Capital). Según su origen, pueden ser nuevas, procedentes de una ampliación con aportaciones, o liberadas, total o parcialmente, cuando la ampliación se hace con cargo a reservas y el accionista no paga, o paga solo en parte, las acciones que recibe.

\subsubsection{Mercados}
Las empresas jóvenes captan fondos propios en los mercados privados, de capital riesgo y capital privado, más concentrados y sin acceso de los inversores minoristas. La salida a bolsa amplía el número de inversores potenciales y da liquidez a los accionistas, a cambio de dispersar la propiedad y de asumir obligaciones de información. Sus dos modalidades más comunes son la oferta pública inicial y la fusión con una sociedad cotizada creada para ello, sin actividad propia (\textit{special purpose acquisition company}, SPAC).

\subsection{Deuda}
\subsubsection{Obligaciones y préstamos}
Las obligaciones son partes alícuotas de un empréstito, como las acciones lo son del capital. El obligacionista se diferencia del accionista en que recibe una renta periódica pactada y no una parte de los resultados; en que recupera su aportación al vencimiento; en que es acreedor y no socio, con derechos económicos pero no políticos; en que asume menos riesgo, porque cobra antes en caso de liquidación; y en que, para el emisor, su aportación es financiación ajena ([\authorfont{Ver Tema 3.B.24}]). Los préstamos, como las obligaciones, son recursos de terceros que se devuelven con intereses en plazos y cuantías fijados, pero los concede, por lo general, un solo acreedor. Los de corto plazo suelen financiar la tesorería, y los de largo plazo, el activo no corriente.

\subsubsection{Banca frente a mercado: criterios prácticos}
Los préstamos bancarios siguen una lógica de negociación e intermediación, y la emisión de deuda, una lógica de mercado. Suelen compararse con cinco criterios:
\begin{lnum}
    \item \textbf{Coste}. Puede ser mayor o menor en cualquiera de las dos vías, porque el préstamo incorpora un margen de intermediación, la emisión tiene costes que el préstamo no tiene, y el tipo de interés de un préstamo puede no corresponder con el coste real de financiar a la empresa;
    \item \textbf{Volumen}. La emisión exige volúmenes mínimos, por razones institucionales (las plazas financieras suelen exigirlos) y económicas: para que su deuda sea atractiva, la empresa necesita que tenga liquidez suficiente y emitir a distintos plazos para construir una curva de tipos. En la Unión, las empresas medianas y grandes, de más de $50$ empleados, emiten bonos unos $4$ puntos porcentuales más a menudo que las pequeñas (DURANTE et al., 2023);
    \item \textbf{Desarrollo financiero}. Además de las características de cada empresa, el desarrollo del sistema financiero de un país y su crecimiento económico están positivamente relacionados con el uso de la financiación de mercado (DE JONG, KABIR y NGUYEN, 2008). La financiación bancaria predomina en la zona del euro, donde los préstamos bancarios a las sociedades no financieras casi triplican los valores de deuda que estas emiten, y la de mercado en EE. UU. y otros países anglosajones;
    \item \textbf{Rapidez}. Un préstamo bancario suele obtenerse con rapidez, mientras que preparar la primera emisión puede llevar semanas y no garantiza captar los fondos; en cambio, las grandes empresas con buena calidad crediticia pueden captar mucho dinero en pocas horas en el mercado; y,
    \item \textbf{Información}. La relación más estrecha entre prestamista y prestatario hace más apropiado el préstamo bancario cuando la empresa es poco transparente (RAJAN y ZINGALES, 2003).
\end{lnum}

\subsubsection{Colocación privada}
Como la emisión requiere un volumen mínimo elevado para generar liquidez, las empresas pequeñas tienen dificultades para acceder a ella. La colocación privada (\textit{private placement}) consiste en emitir entre inversores preseleccionados con los que se negocia directamente, en lugar de en el mercado. Como esos inversores suelen mantener los bonos hasta el vencimiento, no necesitan liquidez y pueden emitirse cantidades pequeñas; pero la falta de documentación normalizada encarece la emisión y limita su uso por las pymes. Es atractiva para las empresas que quieren diversificar sus fuentes y acceder a financiación a largo plazo sin necesidad de calificación crediticia, aunque se aleja de la lógica de mercado y se acerca a la negociación bilateral.

\subsubsection{Préstamos sindicados}
Cuando el importe es tan elevado que un solo banco no estaría dispuesto a prestarlo, el banco que lidera la operación (\textit{mandated lead arranger}) invita a otros a formar un sindicato que aporte la cantidad total. La sindicación puede ser voluntaria, si el banco director se limita a conseguir lo que el mercado permita, o asegurada, si se compromete a cubrir lo que falte. Desde la teoría de la información y de la agencia, la lógica del banco como supervisor delegado de sus depositantes (DIAMOND, 1984) se aplica también aquí: los bancos del sindicato delegan la supervisión en el banco director, lo que ahorra costes y permite especializarse; y, frente a la emisión de bonos, la sindicación puede evitar costes de supervisión innecesarios y problemas de polizón entre acreedores dispersos (ALTUNBAŞ, KARA y MARQUÉS-IBÁÑEZ, 2010).

\subsubsection{Fondos de deuda}
Los bancos no son los únicos intermediarios que prestan a las empresas. Los fondos de deuda o de préstamo directo (\textit{direct lending}) ofrecen financiación más flexible que la bancaria: pueden devolver todo el principal al vencimiento (\textit{bullet}), a plazos más largos, por importes a veces mayores y a tipos distintos. Suelen estar disponibles solo para empresas establecidas, con activos e historial que respalden la decisión del fondo.

\subsubsection{Financiación con garantía de activos}
Una de las formas más seguras de garantizar el reembolso es usar un activo de la empresa como garantía (\textit{asset-backed lending}). Restringe a la empresa, que no puede vender ese activo, pero abarata su financiación, y ofrece al prestamista un perfil de riesgo y rentabilidad más moderado que la deuda ordinaria. Las empresas cuyo negocio descansa en intangibles muy específicos, como muchas empresas emergentes, apenas pueden usarla (OCDE, 2015). Son de esta naturaleza el arrendamiento financiero, el \textit{renting} y el \textit{factoring}:
\begin{la}
    \item El \textbf{arrendamiento financiero} (\textit{leasing}) permite a la empresa usar un activo no corriente a cambio de un canon, durante un plazo al término del cual puede comprarlo, devolverlo o renovar el contrato. Intervienen la empresa, la entidad arrendadora y el fabricante del bien; en el arrendamiento operativo, el fabricante y el arrendador coinciden. Evita desembolsar de una vez grandes inversiones, su canon es fiscalmente deducible y, en el operativo, la empresa se libra del riesgo de obsolescencia, porque puede rescindir el contrato. A cambio, puede resultar caro a largo plazo, porque la empresa pierde el valor residual del bien, y no todos los activos pueden arrendarse;
    \item El \textbf{factoring} es la cesión a un tercero de los derechos de cobro sobre clientes a cambio de liquidez inmediata, con un coste elevado. Su reverso es el \textbf{confirming}, un servicio de pago a proveedores: la empresa encarga a una entidad financiera el pago de sus facturas, y el proveedor puede cobrarlas por anticipado a cambio de una comisión y de los intereses correspondientes.
\end{la}

\subsubsection{Titulización}
Las empresas también pueden refinanciar parte de sus activos transformando créditos u otros derechos en valores negociables. La utilizan con frecuencia los grupos automovilísticos, como el grupo Mercedes: sus filiales financieras venden a un vehículo especial los préstamos y arrendamientos concedidos a sus clientes, y el vehículo los financia emitiendo bonos de titulización (\textit{asset-backed securities}), a menudo en tramos con distinta prelación (sénior, subordinados, \textit{mezzanine}) para ofrecer distintos perfiles de riesgo y rentabilidad. Aislados del resto del balance, esos activos pueden ser de mayor calidad que la empresa en su conjunto y financiarse a tipos más bajos; pero la operación es más costosa que la deuda directa para una empresa de buena calidad crediticia, y difícil de montar por menos de $50$ millones de euros salvo que varias empresas compartan la plataforma. No debe confundirse con la financiación con garantía de activos: esta respalda préstamos individuales con un activo, mientras que la titulización agrupa muchos préstamos parecidos y los vende, de modo que el inversor solo cuenta con ese conjunto de activos.

\subsubsection{Project finance}
\textcolor{red}{\textbf{OJO! China} - Project finance: una opción para grandes proyectos.} El \textit{project finance} financia un proyecto según su propia capacidad de generar fondos y no según la del promotor, y usa como garantía los activos del propio proyecto. Se ha utilizado, por ejemplo, para prospecciones petrolíferas, refinerías o plantas de energías renovables que no podían ofrecer las garantías de un préstamo ordinario: parte del petróleo, incluso antes de extraerse, servía de garantía, y parte de las ventas futuras devolvía los préstamos. Como está pensado para grandes proyectos de energía o infraestructuras, lo usan a menudo los bancos regionales de desarrollo. El banco asume más riesgo que en un préstamo convencional (si el proyecto no devuelve el préstamo, puede acabar siendo propietario de una central eléctrica o de un yacimiento), por lo que es difícil aplicarlo a proyectos tecnológicos de rendimiento incierto. Permite plazos de hasta $30$ años, un apalancamiento muy elevado y sacar la deuda del balance del promotor, sin afectar a sus ratios de endeudamiento.

\subsubsection{Otras fuentes}
Completan el panorama el crédito comercial de los proveedores, el descuento comercial, los pagarés de empresa, las líneas de crédito, la financiación participativa (\textit{crowdfunding}), en la que un proyecto se publica en una plataforma, por lo general en internet, para que muchos pequeños aportantes lo financien a cambio de un rendimiento, y los mercados internacionales de eurodivisas y eurobonos.

\subsection{Instrumentos híbridos}
\subsubsection{Acciones preferentes}
Comparten rasgos con la deuda y con las acciones. Como la deuda, pagan un dividendo fijo, que, si la empresa no tiene efectivo para pagarlo, suele acumularse hasta que pueda, y sus titulares no participan en la gestión: su voto se limita a lo que afecte a sus derechos de cobro. Como las acciones, su remuneración no es deducible y no tienen vencimiento. En la liquidación, cobran después de los obligacionistas y antes que los accionistas ordinarios.

\subsubsection{Deuda subordinada y deuda perpetua}
La deuda subordinada solo se cobra después de la deuda con mayor prelación. La deuda perpetua no tiene vencimiento ni obligación de devolución: el emisor la amortiza cuando lo considera oportuno, y el inversor solo recupera su aportación si el emisor decide amortizarla, si la sociedad se liquida o si vende el título en el mercado secundario.

\subsubsection{Bonos convertibles y warrants}
Un bono convertible puede canjearse por un número predeterminado de acciones a elección de su titular. Al emitirse, la conversión no suele interesar, pero se vuelve atractiva a medida que sube la cotización; las empresas añaden la opción de conversión para pagar un interés menor. Los \textit{warrants} dan derecho a suscribir, durante un período, otro título de nueva emisión (una acción, un bono o incluso otro \textit{warrant}) en una proporción y a un precio fijados de antemano.

\subsubsection{Bonos convertibles contingentes}
Los CoCos son bonos que se convierten en acciones, o pierden valor, cuando se cumple una condición fijada de antemano, por lo general que la ratio de capital ordinario de nivel 1 del banco emisor caiga por debajo de un umbral. Mientras están vivos pagan cupones elevados, y el inversor asume el riesgo de la conversión. Los bancos los han emitido en abundancia para reforzar sus balances ante las exigencias regulatorias: la conversión reduce su deuda y mejora su capital cuando más dudas hay sobre su viabilidad ([\authorfont{Ver Tema 3.B.27}]).

\subsubsection{Préstamos participativos}
El prestamista recibe un interés variable, ligado a la actividad de la empresa, que puede combinarse con uno fijo. En España, se cobran después de los acreedores comunes, solo pueden amortizarse anticipadamente si la empresa amplía en la misma cuantía sus fondos propios y se consideran patrimonio neto a efectos de la reducción de capital y la liquidación de sociedades (art. $20$ del Real Decreto-ley 7/1996).
