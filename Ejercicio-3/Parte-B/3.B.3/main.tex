    % --- Imagen adaptativa ---
    % \imagenfit[alto]{fichero}{título}{fuente}: la imagen queda fija entre los párrafos donde se escribe (no flota), con su
    % título en el índice de figuras y la fuente debajo. Sin [alto]: las figuras PDF (TikZ) salen al tamaño con que se
    % dibujaron y solo se reducen si no caben en el ancho o en el alto de la página; las PNG/JPG ocupan el ancho del texto
    % hasta \imgaltoras del alto de la página. Con [alto] (fracción del alto de la página; [1] = página entera) se fija el
    % tamaño de esa imagen. Si no cabe en lo que queda de página, pasa entera a la siguiente.
    % \imagenfithorizontal{fichero}{título}{fuente}: la imagen en una página girada (apaisada), en ese mismo punto del texto.
    % Reglas: main/GUIA_TEMAS.md, §4 (Figuras).
    \newdimen\imgcap
    \newdimen\imgh
    \newdimen\imgw
    \newdimen\imgneed
    \newsavebox\imgcaja
    \newcommand{\imgaltoras}{0.30}% alto por defecto de las PNG/JPG (fracción del alto de la página)
    \makeatletter
    \newcommand{\imagenfit}[4][]{%
      \addvspace{10pt}%
      \par\begingroup
        \imgcap=3.5\baselineskip% alto aproximado de título + fuente
        \imgw=\linewidth
        \filename@parse{#2}\def\img@pdf{pdf}%
        \if\relax\detokenize{#1}\relax
          \ifx\filename@ext\img@pdf
            \sbox\imgcaja{\includegraphics{\detokenize{#2}}}% tamaño natural
            \ifdim\wd\imgcaja<\imgw \imgw=\wd\imgcaja \fi
            \imgh=\ht\imgcaja
          \else
            \imgh=\imgaltoras\textheight
          \fi
        \else
          \imgh=#1\textheight
        \fi
        \dimen@=\textheight \advance\dimen@ by -\imgcap
        \ifdim\imgh>\dimen@ \imgh=\dimen@ \fi
        \sbox\imgcaja{\includegraphics[width=\imgw,height=\imgh,keepaspectratio]{\detokenize{#2}}}%
        \imgneed=\ht\imgcaja \advance\imgneed by \imgcap
        \Needspace{\imgneed}%
        \noindent\begin{minipage}{\textwidth}
          \centering
          \captionof{figure}{\textemdash\ #3}\par\vspace{0.3em}% título arriba
          \usebox\imgcaja\par
          \vspace{0.3em}\small\textit{Fuente: }#4% fuente abajo
        \end{minipage}%
      \endgroup
      \par\addvspace{10pt}\RequestRestore
    }
    \newcommand{\imagenfithorizontal}[3]{%
      \begin{landscape}
        \centering
        \captionof{figure}{\textemdash\ #2}\par\vspace{0.3em}%
        \includegraphics[width=\linewidth,height=\dimexpr\textheight-4\baselineskip\relax,keepaspectratio]{\detokenize{#1}}\par
        \vspace{0.3em}{\small\textit{Fuente: }#3}%
      \end{landscape}%
    }
    \makeatother

    % \anexo{Título}: un anexo equivale a un \section (su título sale del mismo tamaño). Dentro se usan \subsection,
    % \subsubsection y \paragraph, nunca \section: se numeran I.1, I.1.1, I.1.1.1 y las páginas del anexo, I-1, I-2…
    % En el índice de contenido inicial solo figura el nombre del anexo. Sus apartados forman el índice del anexo, que sale
    % solo si el anexo los tiene, justo después de la introducción del anexo (antes del primer \subsection).
    % \anexo*{Título}: el mismo anexo, sin ese índice. Reglas: main/GUIA_TEMAS.md, §6.

    \newcommand{\anx@empezar}[2]{%
      \clearpage
      \refstepcounter{anexo}%
      \global\EnAnexotrue
      \setcounter{page}{1}\gdef\thepage{\theanexo-\arabic{page}}%
      \setcounter{section}{\numexpr\value{anexo}-1\relax}\stepcounter{section}% el anexo hace de \section
      \gdef\thesection{\Roman{section}}\gdef\theHsection{anexo.\Roman{section}}%
      \phantomsection
      \protected@xdef\leq@current@section{Anexo \theanexo.- \unexpanded{#2}}\gdef\leq@current@subsection{}% índice de ecuaciones
      \addtocontents{stc}{\protect\contentsline{section}{Anexo \theanexo.- #2}{}{\@currentHref}}% índice inicial: solo el nombre, sin página
      \addcontentsline{toc}{section}{Anexo \theanexo.- #2}% marcadores del PDF
      {\noindent\normalfont\Large\bfseries Anexo \theanexo.- #2\par}\nobreak\vspace{1.5ex \@plus .2ex}%
      #1\@ifundefined{anx@lista@\theanexo}{\global\anx@pendientefalse}{}%
    }
    % Índice del anexo: cada apartado escribe una línea en el fichero .atc (anexo, nivel, texto, página, enlace); al
    % empezar el documento se leen las de la compilación anterior y se guardan por anexo (dos pasadas, como cualquier índice).
    % En ese índice los apartados suben un nivel, porque el anexo hace de \section.
    \def\anx@entrada#1#2#3#4#5{\@ifundefined{anx@lista@#1}{\global\@namedef{anx@lista@#1}{}}{}%
      \expandafter\g@addto@macro\csname anx@lista@#1\endcsname{\contentsline{#2}{#3}{#4}{#5}}}
    \tcbuselibrary{breakable}
    \def\anx@indice{\ifanx@pendiente\global\anx@pendientefalse

            before upper={\parindent=0pt\parskip=2pt}]
          \small
          % números más anchos que en el índice inicial (I.1, I.1.1, I.1.1.1; páginas I-1)
          \def\@pnumwidth{2.6em}\def\@tocrmarg{3.4em}%
          \def\l@section##1##2{\addpenalty\@secpenalty\addvspace{1.0em \@plus\p@}\setlength\@tempdima{2.6em}%
            \begingroup\parindent\z@\rightskip\@pnumwidth\parfillskip-\@pnumwidth\leavevmode\bfseries
            \advance\leftskip\@tempdima\hskip-\leftskip##1\nobreak\hfil\nobreak\hb@xt@\@pnumwidth{\hss##2\kern-\p@\kern\p@}\par\endgroup}%

          \csname anx@lista@\theanexo\endcsname

      \fi}
    \def\anx@escribir#1#2#3{\ifEnAnexo

    % margen superior, así que no cambia las páginas del tema. Se enciende con \pagenumbering{arabic} y se apaga con

    \cs_new_protected:Npn \cab_guardar:n #1
      {
        \tl_gset:Nn \g_cab_titulo_tl {#1}
        \regex_replace_once:nnN { \A \s* Tema \s+ \S+ \s* \-\-? \s* } { } \g_cab_titulo_tl % sin «Tema X.Y.N --»
        \regex_replace_once:nnN { \.? \s* \c{\\} (\[ [^\]]* \])? \s* (\c{large} \s*)? } { .\  } \g_cab_titulo_tl % título. subtítulo
        \regex_replace_once:nnN { [\.\s]* \Z } { } \g_cab_titulo_tl % sin subtítulo: nada detrás
      }

\subsection{Estructura de la exposición}
Uno puede preguntarse por qué unas empresas se apoyan en la adquisición de capital en rondas de financiación, por qué unas empresas están más endeudadas que otras, o si existe un endeudamiento o estructura de capital óptima. El objetivo de este tema es abordar estas cuestiones desde la perspectiva de las finanzas corporativas:

En un primer apartado, se repasarán las diferentes fuentes de financiación empresarial, tanto internas como externas (equity y deuda), con el objetivo de entender los fundamentos de cada una de estas fuentes de financiación y los motivos por los que detenninados tipos de empresas hacen (o no) uso de ellas. 

El segundo apartado abordará la estructura del capital: una vez que ya se ha explicado cada una de las fuentes de financiación por separado, se partirá de las proposiciones de Modigliani-Miller para justificar que, cuando los mercados son perfectos, en realidad la composición de equity y deuda de la empresa es irrelevante. \section{Financiación propia frente a financiación ajena}
\puntosclave{
    La empresa no tiene riqueza propia: el valor de sus activos se reparte entre quienes la financian. Cada fuente de financiación es un contrato que reparte de forma distinta los flujos de caja, el riesgo y el control.

    La distinción entre financiación propia y ajena atiende a la naturaleza del derecho, residual o fijo; la distinción entre interna y externa, a su origen. La primera determina el reparto del riesgo, del control y de la carga fiscal; la segunda, los costes de información que hay que pagar para obtener los fondos.

    El beneficio retenido no es gratuito: cuesta lo que exigen los accionistas. Su ventaja es ahorrar los costes de acudir a inversores externos, y su riesgo, liberar a los directivos de la disciplina del mercado.

    La deuda es el contrato que menos información exige. El banco la concede cuando la empresa es opaca y necesita supervisión; el mercado basta cuando la empresa tiene reputación y capital propio suficiente.

    Ampliar capital cede beneficios futuros, control y, si se emite por debajo del valor de la acción, riqueza de los antiguos accionistas. Los instrumentos híbridos combinan rasgos de la deuda y de las acciones para resolver conflictos concretos.
}

Toda inversión exige recursos: la empresa paga hoy la maquinaria, las existencias o el desarrollo de un producto cuyos ingresos llegarán, si llegan, más adelante. 

Las fuentes de las que obtiene esos recursos son muchas, y la innovación financiera no ha dejado de multiplicarlas, pero todas responden a las mismas preguntas: quién aporta los fondos, qué recibe a cambio, quién decide sobre su uso y qué ocurre si las cosas van mal. 

La tesis del bloque es que cada fuente es un contrato que reparte de forma distinta los flujos de caja, el riesgo y el control, y que la que conviene a cada empresa depende de las fricciones a las que se enfrenta (el coste de informar a los inversores, el de supervisar a los directivos o la falta de activos que sirvan de garantía), no de una ordenación general entre fuentes baratas y caras. Como cada empresa combina esas fricciones a su manera, ningún marco permite valorar de una vez todos los instrumentos; sí permite entender por qué existe cada uno y quién lo usa. Si esas diferencias alteran, además, el valor de la empresa es la pregunta del bloque siguiente.

\subsection{Estructura financiera: ¿a quién pertenece el valor de la empresa?}
En términos financieros, la empresa no tiene riqueza propia. El valor de sus activos ($V$), que es el valor actual de los flujos de caja que se espera que generen, pertenece por entero a quienes la han financiado, y su \textbf{estructura financiera} no es más que la forma en que ese valor se reparte entre ellos. En valores de mercado:

\eqblock{
\begin{aligned}
  \overset{\text{\scriptsize Activo}}{\text{Valor empresa}}=\overset{\text{\scriptsize Pasivo $+$ Patrimonio Neto}}{\underset{\text{\scriptsize Fondos propios}}{E}+\underset{\text{\scriptsize Valor pasivo}}{D}}
\end{aligned}
}{Estructura financiera: ¿Qué es la empresa?. Financiación:}

Como vemos, se trata del balance económico de la empresa: la misma identidad que el balance contable, pero con cada partida valorada a precios de mercado y no por su valor en libros.\footnote{%
    Además de accionistas y acreedores financieros, tienen derechos sobre los flujos de la empresa los proveedores que le conceden crédito comercial, los trabajadores, por los salarios devengados, y la Hacienda Pública, cuya participación en el beneficio, a través del impuesto sobre sociedades, depende de cómo se financie la empresa, como veremos en el bloque siguiente. Al hablar de estructura del capital, $D$ se limita a la deuda financiera. La diferencia entre el valor de mercado de los fondos propios y su valor en libros recoge lo que el mercado espera que los activos rindan por encima de su coste histórico ([\authorfont{Ver Tema 3.B.1}]).
}

\subsubsection{Derechos: ¿qué diferencia cada derecho sobre el valor de la empresa?}
Los dos derechos son de naturaleza distinta, y de esa diferencia nace la distinción entre financiación propia y ajena. 

El acreedor tiene un \textbf{derecho fijo}: una promesa de pago de intereses y principal en fechas pactadas, exigible con independencia de los resultados, que se cobra antes que los accionistas si la empresa se liquida. 

El accionista tiene un \textbf{derecho residual}: cobra lo que queda después de pagar a los acreedores, sin vencimiento ni cuantía pactados, y a cambio decide con su voto cómo se gestiona la empresa. 

El reparto del control es, por tanto, contingente: los accionistas mandan mientras la empresa paga, y los acreedores adquieren el control cuando deja de hacerlo, mediante el concurso o la renegociación (AGHION y BOLTON, 1992).\footnote{%
    AGHION y BOLTON (1992) parten de contratos incompletos ([\authorfont{Ver Tema 3.A.15}]): como no pueden preverse todas las decisiones futuras, lo esencial de un contrato financiero es quién decide en cada estado. La deuda asigna el control al empresario mientras paga y lo transfiere al acreedor cuando incumple. HART y MOORE (1994) añaden que, como el empresario puede marcharse con su capital humano, que no es embargable, solo se le puede prestar contra la amenaza de quedarse con los activos.

    A ello se suman dos diferencias más: los intereses son un gasto deducible en el impuesto sobre sociedades y los dividendos no, y las acciones son perpetuas mientras que la deuda vence.
} \textcolor{magenta}{La figura lo muestra con lo que cobra cada uno cuando vence la deuda.

\subsubsection{Financiación: ¿por qué busca financiarse la empresa?}
Ahora bien, habida cuenta de lo expuesto, resultan confusas las razones por las que la empresa decidiría financiarse: ¿por qué busca financiación una empresa que no tiene riqueza propia? 

\textcolor{magenta}{La razón, evidentemente, es porque sus oportunidades de inversión y los fondos que genera no coinciden ni en el tiempo ni en la cuantía: la inversión se paga antes de que produzca ingresos y suele llegar en bloques, mientras que los fondos se generan de forma gradual. 

El criterio para decidir qué inversiones acometer no es crecer, sino crear valor para los propietarios: invertir en los proyectos cuyo valor actual neto, descontado al coste del capital, sea positivo ([\authorfont{Ver Tema 3.B.2}]). \eqblock{
  \begin{aligned}
    g^{*}=b\cdot \text{ROE},\qquad \text{ROE}=\left[\text{ROA}+\left(\text{ROA}-i\right)\frac{D}{E}\right](1-t)
  \end{aligned}
}{Crecimiento de la empresa: Tasa de crecimiento sostenible (HIGGINS, 1977). Estructura financiera: ¿Por qué financiarse?}

Donde $b$ es la proporción del beneficio que se retiene, ROA la rentabilidad económica (beneficio antes de intereses e impuestos sobre activo), $i$ el coste de la deuda y $t$ el tipo del impuesto sobre sociedades.\footnote{%
    Si la empresa no emite acciones, sus fondos propios solo crecen con el beneficio retenido, $\Delta E=b\cdot\text{BN}$, y crecen a la tasa $\Delta E/E=b\cdot\text{ROE}$, con el ROE calculado sobre los fondos propios iniciales. 
    
    Si además mantiene constante $D/E$, la deuda y el activo crecen a ese mismo ritmo, y con ellos las ventas si la rotación del activo no cambia. La descomposición del ROE se obtiene de $\text{BN}=(\text{BAII}-iD)(1-t)$ y $\text{BAII}=\text{ROA}\cdot(E+D)$ ([\authorfont{Ver Tema 3.B.1}]). 
    
    El ROA puede sustituirse por el ROCE, el beneficio operativo sobre el capital empleado, con los ajustes debidos, sin que cambien las conclusiones; el ROCE se interpreta además como la rentabilidad de los fondos propios si la deuda fuera nula.
} 

\paragraph{Valor: ¿cómo pasar de la identidad contable a las preferencias de los agentes?}
La identidad recoge las tres decisiones de las que trata el tema. La proporción retenida, $b$, es la otra cara de la política de dividendos. La ratio $D/E$ es la estructura del capital: si la empresa obtiene sobre sus activos más de lo que paga por su deuda ($\text{ROA}>i$), endeudarse eleva la rentabilidad del accionista, y si obtiene menos, la reduce. Y crecer por encima de $g^{*}$ exige fondos externos, emitiendo acciones o endeudándose más, que es la elección entre financiación propia y ajena externa.

Pero es una identidad contable, en valores de libro, y no dice nada del valor. Un mayor endeudamiento eleva el ROE esperado, pero también su variabilidad y, con ella, la rentabilidad que exigen los accionistas. Una mayor retención eleva $g^{*}$, pero solo crea valor si los fondos retenidos rinden más que esa rentabilidad exigida. 

Por eso, los dos bloques siguientes sustituyen las rentabilidades contables por las que exigen los inversores, en valores de mercado, y se preguntan si la estructura del capital y la política de dividendos alteran el valor de la empresa.

\paragraph{Fuentes: ¿cómo puede financiarse la empresa?}
Las fuentes con que la empresa financia su crecimiento se ordenan con dos criterios. Según la \textbf{naturaleza del derecho}, la financiación es propia (fondos propios: derecho residual y no exigible) o ajena (deudas: derecho fijo y exigible). Según su \textbf{origen}, es interna, si procede de los fondos que genera la actividad y la empresa no reparte, o externa, si la aportan inversores de fuera. 

Los dos criterios no coinciden siempre: el beneficio retenido es financiación propia e interna; una ampliación de capital, propia y externa; un préstamo bancario, ajena y externa. La figura los cruza.\footnote{%
    Los instrumentos concretos de cada fuente, sus variantes y los criterios prácticos para elegir entre ellos se describen en [\authorfont{Ver Anexo \ref{anx:instrumentos}}].
}

\imagenfit{fuentes_matriz.pdf}{Las fuentes de financiación según su origen y la naturaleza del derecho}{Elaboración propia.}

Cada criterio responde a una pregunta distinta. El primero dice cómo se reparten el riesgo, el control y la carga fiscal; el segundo, cuánto cuesta convencer a alguien de que aporte fondos, porque quien está fuera sabe menos que quien dirige la empresa. 

Por eso la literatura ha recurrido al segundo para explicar qué fuentes eligen las empresas y en qué orden, y al primero para explicar la estructura del capital.

\subsection{Financiación interna: ¿es gratis el beneficio retenido?}
El beneficio que la empresa no reparte es su principal fuente de financiación y la que parece más barata, porque no obliga a pagar intereses ni dividendos. Pero no nos equivoquemos, \textbf{no es gratuito}: su coste es lo que exigen los accionistas por dejarlo dentro.

La financiación interna, o \textbf{autofinanciación}, comprende el beneficio retenido, que se acumula en las reservas y aumenta los fondos propios (\textbf{autofinanciación de enriquecimiento}), y las dotaciones a amortizaciones y provisiones, gastos que no suponen salida de caja y que, al reducir el beneficio repartible, retienen en la empresa los fondos necesarios para mantener su capacidad productiva (\textbf{autofinanciación de mantenimiento}).\footnote{%
    Las reservas pueden ser legales, estatutarias o voluntarias; la reserva legal obliga a destinar el $10\%$ del beneficio del ejercicio hasta que alcance al menos el $20\%$ del capital social (art. $274$ de la Ley de Sociedades de Capital). Las provisiones se registran en el pasivo y las amortizaciones minoran el activo, por lo que la autofinanciación de mantenimiento no encaja limpiamente en la distinción entre propia y ajena. Dos efectos amplían su alcance. Si la empresa mantiene constante su ratio de endeudamiento, $\alpha=D/E$, cada euro de reservas le permite endeudarse en $\alpha$ euros más, y sus recursos totales crecen en $(1+\alpha)$ euros: es el \textbf{efecto multiplicador} de la autofinanciación. Y si reinvierte las cuotas de amortización en nuevos activos, su capacidad productiva no solo se mantiene, sino que crece (\textbf{efecto LOHMANN-RUCHTI}). Los supuestos y un ejemplo están en un anexo ([\authorfont{Ver Anexo \ref{anx:instrumentos}}]).
}

Retener un euro equivale, para el accionista, a recibirlo como dividendo y volver a invertirlo en acciones nuevas de la empresa. Por eso el coste de la autofinanciación es el \textbf{coste de oportunidad} de los accionistas y la \textbf{rentabilidad exigida} ($r_E$) es la misma que exigirían a una ampliación de capital (salvo por los costes de emisión). Tratar el beneficio retenido como un recurso sin coste lleva a aceptar inversiones que rinden menos de lo que los accionistas obtendrían fuera; el efectivo que excede lo que la empresa puede invertir con rentabilidad debería devolverse a los accionistas, mediante dividendos o recompras de acciones, o destinarse a amortizar deuda.

\subsubsection{Ventajas: ¿por qué financiarse internamente?}
La ventaja de la autofinanciación no está en su precio, sino en lo que evita. No diluye el control de los accionistas, no tiene costes de emisión, da a la empresa autonomía para decidir cuándo y en qué invierte y, sobre todo, no exige convencer a inversores que saben menos que los directivos sobre el valor de la empresa. 

Este último punto es decisivo: un inversor externo que no puede distinguir una empresa buena de una mala paga por sus títulos un precio medio, y la empresa buena prefiere entonces financiarse con lo que tiene (MYERS y MAJLUF, 1984), un argumento que desarrollaremos al tratar la estructura del capital. Para muchas pymes, además, es la única alternativa al crédito bancario para financiarse a largo plazo.

Su verdadera ventaja, por tanto, es \textbf{ahorrar los costes de acudir a inversores externos}. 

\subsubsection{Límites: ¿por qué no financiarse internamente?}
Sus límites son dos. El primero es cuantitativo: los fondos internos se generan poco a poco y las inversiones llegan en bloques, de modo que la empresa que crece deprisa agota pronto su autofinanciación y puede tener que renunciar a inversiones rentables. 

El segundo es de incentivos. Cuando la empresa genera más fondos de los que puede invertir con valor actual neto positivo, ese \textbf{flujo de caja libre} queda a disposición de unos directivos que, sin tener que rendir cuentas ante los inversores, tienden a invertirlo en crecer más allá de lo rentable (JENSEN, 1986). 

La deuda y los dividendos, que obligan a sacar ese efectivo de la empresa, son las respuestas que veremos en los bloques siguientes.

\subsubsection{Financiación ajena: ¿por qué deuda y de quién?}
La deuda es el contrato de financiación que menos información exige al acreedor: mientras la empresa paga, no necesita saber cuánto gana. De quién la obtiene la empresa, de un banco o del mercado, depende de cuánta supervisión necesita para que alguien se la conceda, y eso depende de su tamaño, de su reputación y de su capital propio.

Si los inversores pudieran observar sin coste los resultados de la empresa, podrían firmar con ella cualquier contrato que hiciera depender sus cobros de esos resultados. Cuando verificarlos es costoso, el contrato óptimo es la deuda: un pago fijo que el acreedor no necesita comprobar mientras se cumple, y una verificación, con la liquidación de los activos, solo cuando la empresa declara que no puede pagar (TOWNSEND, 1979; GALE y HELLWIG, 1985). 

La deuda ahorra así costes de verificación, y su fuerza está en la amenaza de liquidación, que es la que obliga a pagar.

\paragraph{Taxonomía: ¿es la fuente de financiación una elección?}
Como sabemos, la financiación ajena puede obtenerse de un banco, que la negocia bilateralmente y la mantiene en su balance, o en el mercado, emitiendo bonos que compran inversores dispersos. 

Las diferencias prácticas son conocidas: el préstamo bancario se obtiene con rapidez, por importes pequeños y sin hacer pública información sobre la empresa, mientras que la emisión exige volúmenes mínimos para que el título tenga liquidez, información pública y, a menudo, una calificación crediticia, pero da acceso a fondos más abundantes y, para las empresas grandes, más baratos.\footnote{%
    Los cinco criterios con que suelen compararse (coste, volumen, desarrollo financiero, rapidez e información), la colocación privada, los préstamos sindicados y otras formas de deuda se desarrollan en un anexo ([\authorfont{Ver Anexo \ref{anx:instrumentos}}]). 
    
    El crédito comercial de los proveedores es otra forma de financiación ajena: el proveedor conoce al cliente por la relación comercial y puede recuperar la mercancía si no le paga, por lo que presta a empresas a las que un banco no prestaría, que lo usan más (PETERSEN y RAJAN, 1997).
}

La teoría explica por qué: \textbf{supervisar a un prestatario es costoso, y si cada pequeño inversor lo hiciera por su cuenta el coste se multiplicaría o, lo que es más probable, nadie lo haría}. 

El banco supervisa en nombre de todos sus depositantes y, como presta a muchos prestatarios distintos, diversifica el riesgo de su cartera y puede garantizar a sus depositantes que hace bien ese trabajo (DIAMOND, 1984). La supervisión tiene, sin embargo, un precio: la información que el banco acumula le da poder para quedarse con parte de las rentas de la empresa, que por eso preferirá el mercado en cuanto pueda (RAJAN, 1992). Y puede hacerlo cuando ha construido una reputación pagando puntualmente bajo la vigilancia del banco (DIAMOND, 1991).

HOLMSTRÖM y TIROLE (1997) unen estas piezas en función del capital propio de la empresa. Como el empresario puede desviar parte de los resultados en su provecho, los inversores solo financian un proyecto si él conserva una participación suficiente en su éxito.

Quien tiene menos necesita además que un banco lo supervise, y el banco, a su vez, tiene que arriesgar capital propio en el préstamo para tener incentivos a supervisar. Quien tiene muy poco ($A<\underline{A}$) no obtiene financiación, aunque su proyecto sea rentable: el crédito se raciona.

El modelo explica el patrón observado: las empresas pequeñas y jóvenes dependen del banco, y las grandes, con historial y activos, acuden también al mercado.\footnote{%

\paragraph{Balance: ¿por qué sí o por qué no financiarse externamente?}
Para la empresa, la deuda tiene tres ventajas: no diluye el control de los accionistas, porque el acreedor no vota mientras cobra; sus intereses son deducibles en el impuesto sobre sociedades; y, si la rentabilidad de los activos supera su coste, eleva la rentabilidad del accionista. 

Los inconvenientes de cualquier forma de financiación externa son la otra cara de la misma rigidez informativa que genera las ventajas: \textbf{es una obligación incondicional}, cuyo impago puede llevar a la empresa al concurso, y su contrato suele limitar lo que la empresa puede hacer mientras la deuda está viva.

Las cláusulas restrictivas (\textit{covenants}) limitan, entre otras cosas, el reparto de dividendos, la venta de activos, el endeudamiento adicional o el deterioro de determinadas ratios financieras, y su incumplimiento permite al acreedor exigir la devolución anticipada. 

Por tanto, protegen al acreedor frente a decisiones que trasladen riesgo o valor de él a los accionistas, a costa de restar flexibilidad a la empresa.

\subsubsection{Financiación propia: ¿qué se cede al ampliar capital?}
Veamos la otra fuente relevante: la financiación propia externa. En primer lugar, debe quedar claro que quien amplía capital cede tres cosas: una parte de los beneficios futuros, una parte del control y, si emite por debajo de lo que valen sus acciones, una parte de la riqueza de los antiguos accionistas. 

La primera es el precio de los fondos; las otras dos explican por qué las empresas se resisten a ampliar capital.

Las acciones dan un derecho residual, sin vencimiento y con voto. Por eso son la financiación natural de las empresas cuyos flujos de caja son demasiado inciertos, o cuyos activos demasiado intangibles, para sostener una deuda: las jóvenes, las de crecimiento rápido y las intensivas en investigación. 

En las primeras etapas, quien aporta ese capital no se limita a comprar acciones: los fondos de capital riesgo entregan el dinero por fases y se reservan derechos sobre los flujos, el consejo, el voto y la liquidación que dependen de los resultados, separando lo que en una acción ordinaria va unido (KAPLAN y STRÖMBERG, 2003). Su coste es la rentabilidad que exigen los accionistas dado el riesgo de la empresa ($r_E$), más los costes de emisión, y es mayor que el de la deuda porque el accionista cobra el último y soporta la variabilidad de los resultados.\footnote{%
    La salida a bolsa, mediante una oferta pública inicial o la fusión con una sociedad cotizada creada para ello (SPAC), da acceso a más inversores y liquidez a los accionistas, a cambio de dispersar la propiedad y de asumir obligaciones de información. Los tipos de acciones y la salida a bolsa se describen en un anexo ([\authorfont{Ver Anexo \ref{anx:instrumentos}}]); la valoración de las acciones, en otro tema ([\authorfont{Ver Tema 3.B.23}]).
}

\paragraph{Dilución del control: ¿quién entra y qué quiere?}
La \textbf{dilución del control} depende de quién compre las nuevas acciones. Si los dos accionistas de una empresa, A y B, con participaciones $\alpha_A$ y $\alpha_B$, no acuden a una ampliación por una fracción $s$ del capital previo que suscribe un tercero, C, sus participaciones pasan a ser $\alpha_A/(1+s)$ y $\alpha_B/(1+s)$, y C obtiene $s/(1+s)$. En cambio, si suscriben en proporción a lo que ya tienen, las conservan.\footnote{%
    Para eso la ley les reconoce un derecho de preferencia en las ampliaciones con aportaciones dinerarias (art. $304$ de la Ley de Sociedades de Capital).

    Por ejemplo, si A tiene el $80\%$ de un capital de $100$ y B el $20\%$, y C quiere el $20\%$ de la empresa mediante una ampliación, tiene que aportar $25$, no $20$, porque el capital pasa a ser $125$; A queda con el $64\%$ y B con el $16\%$. Si, en cambio, B vendiera su participación a C, el capital de la empresa y la posición de A no cambiarían: la dilución es propia de la ampliación, no de la entrada de un nuevo accionista.
} 

\textcolor{magenta}{(Aquí)}

Que la dilución importe depende de quién entre y de qué quiera. Por ejemplo, las empresas familiares suelen anteponer la conservación del control. Por el contrario, los inversores de capital riesgo, los inversores informales (\textit{business angels}) y los fondos de capital privado aportan, además de fondos, conocimiento y supervisión. En términos medios: los inversores institucionales suelen conformarse con participaciones minoritarias; y los empleados, que pueden entrar mediante acciones u opciones sobre acciones, forman un accionariado estable.

\paragraph{Dilución del valor: ¿a qué precio se vende la acción?}
La \textbf{dilución del valor} depende del precio. Si la empresa emite $m$ acciones a un precio $p_e$ cuando tiene $n$ acciones que valen $v$ cada una, cada acción pasa a valer $v'=(nv+mp_e)/(n+m)$. 

Si $p_e=v$, nadie gana ni pierde; si $p_e<v$, los antiguos accionistas transfieren riqueza a los nuevos, y si $p_e>v$, al revés. Lo relevante es, por tanto, el \textbf{precio de emisión frente al valor de la acción}, no la prima de emisión sobre el nominal, que es un concepto contable.

\section{Estructura del capital: ¿existe una estructura óptima?}
\puntosclave{
    Sin fricciones, la estructura del capital es irrelevante (MODIGLIANI y MILLER, 1958): el inversor puede replicar o deshacer cualquier endeudamiento, el valor depende solo de los activos y la rentabilidad exigida a las acciones crece con la deuda lo justo para dejar constante el coste del capital.

    El valor del teorema es metodológico: cada supuesto que se relaja es una razón para que la estructura del capital importe, y cada razón predice una estructura distinta que puede contrastarse.

    El impuesto sobre sociedades premia la deuda; los impuestos personales pueden anular esa ventaja; los costes de insolvencia la frenan. Del equilibrio entre ahorro fiscal y costes de insolvencia sale una estructura óptima, pero las empresas se endeudan menos de lo que la teoría predice.

    Con información asimétrica, endeudarse puede ser una señal de calidad y emitir acciones, de lo contrario: las empresas prefieren los fondos internos, después la deuda y, en último lugar, las acciones.

    La deuda disciplina a los directivos de las empresas que generan más fondos de los que pueden invertir, pero enfrenta a accionistas y acreedores: sustitución de activos y sobreendeudamiento.

    Los determinantes del endeudamiento son estables (tangibilidad, tamaño, rentabilidad, oportunidades de crecimiento), pero ninguna teoría los explica todos; el endeudamiento es persistente y refleja el aprovechamiento de las condiciones del mercado.
}


La intuición corriente dice que unas fuentes de financiación son mejores que otras: que es preferible financiarse con lo propio antes que compartir los beneficios con extraños o, al revés, que la deuda es más barata que las acciones y conviene usarla cuanto se pueda. 

El bloque anterior mostró que cada fuente reparte de forma distinta los flujos de caja, el riesgo y el control. Ahora toca preguntarse si ese reparto altera, además, el valor de la empresa. 

La respuesta de MODIGLIANI y MILLER (1958) es que, sin fricciones, no: la estructura del capital es irrelevante, porque los inversores pueden deshacer por su cuenta cualquier decisión de financiación de la empresa. La tesis del bloque es que el teorema no describe la realidad, sino que indica dónde buscar: si la estructura del capital importa, es por alguno de los supuestos que el teorema excluye (los impuestos, los costes de insolvencia, la información asimétrica o los conflictos de agencia), y cada uno predice una estructura distinta que la evidencia permite contrastar. La intuición corriente queda así en parte rehabilitada, pero por razones que la teoría puede precisar.

\subsection{Estructura y coste del capital: ¿qué se quiere minimizar?}
La \textbf{estructura del capital} es la proporción en que la empresa se financia con deuda y con fondos propios, medidos a valor de mercado, y su \textbf{coste del capital}, la rentabilidad que exigen en conjunto quienes la financian, dado el riesgo que asumen. 

La pregunta del bloque es si existe una proporción que minimice ese coste y, con ello, maximice el valor de la empresa. Para planteársela hay que pasar de la contabilidad a las finanzas: \textbf{pasar de rentabilidades pasadas medidas en libros a las rentabilidades que exigen los inversores sobre valores de mercado}. 

La rentabilidad económica (ROA) da paso a la rentabilidad que se exige al conjunto de los activos; la rentabilidad financiera (ROE), a la que se exige a los fondos propios ($r_E$); y el coste contable de la deuda, a la rentabilidad que se exige a la deuda ($r_D$).\footnote{%
    Cómo se estima el coste del capital se trata en otro tema ([\authorfont{Ver Tema 3.B.2}]). Puede hacerse de forma directa, aplicando un modelo de valoración de activos al riesgo de los activos, con CAPM o APT ([\authorfont{Ver Tema 3.B.23}]); de forma implícita, calculando la tasa que iguala el valor de mercado de la empresa con los flujos de caja que se esperan de ella, lo que exige conocer esos flujos; o de forma indirecta, estimando por separado $r_E$ y $r_D$ y ponderándolos, que es el método más usado.
}

El coste del capital de la empresa es la media de esas rentabilidades, ponderada por el peso de cada fuente en su valor. Es el \textbf{coste medio ponderado del capital} (WACC), que, por definición y sin impuestos, es:

  \end{aligned}

Es la tasa a la que se descuentan los flujos de caja libres de la empresa para obtener su valor; en el caso más sencillo, con flujos constantes y perpetuos, $V=\text{FCL}/r_{\text{WACC}}$.\footnote{%
  \textbf{NOTA.-} Por eso minimizar el coste del capital equivale a maximizar el valor, con una condición que suele olvidarse: que los flujos de caja no dependan de cómo se financia la empresa. Si la estructura del capital altera esos flujos, porque cambia los impuestos que se pagan, los costes de una posible insolvencia o las decisiones de inversión, el problema deja de ser solo de tasas de descuento.
}

Hasta mediados del siglo XX, la respuesta habitual era que sí existe una estructura óptima. La deuda es más barata que los fondos propios, porque el acreedor cobra antes y asume menos riesgo, de modo que sustituir fondos propios por deuda reduce al principio el coste medio. Pero, por otro lado, endeudarse aumenta el riesgo del accionista y le lleva a exigir más.

De esta forma se obtiene la relación no lineal de la figura: la convexidad se debe a que el riesgo crece cada vez más deprisa; y, a partir de cierto nivel, también los acreedores exigen más ante el riesgo de impago. 

Como vemos, el coste medio cae primero y sube después, y la estructura óptima ($L^{*}$) es la que lo minimiza.\footnote{%
  DURAND (1952) situó esta visión entre dos enfoques extremos: el del beneficio neto, en el que el valor de la empresa crece siempre con la deuda, y el del beneficio de explotación, en el que no depende de ella.
}

La visión tradicional carecía de un fundamento riguroso, porque suponía, sin explicarlo, que los accionistas reaccionan tarde al riesgo que añade la deuda; e implicaba que las decisiones de inversión y de financiación están ligadas, porque la tasa de descuento de un proyecto depende de cómo se financie.

\subsection{MODIGLIANI y MILLER (1958): ¿por qué la estructura sería irrelevante?}
MODIGLIANI y MILLER (1958) demostraron que, en un mercado de capitales sin fricciones, el valor de una empresa no depende de cómo se financie: el tamaño de la tarta no cambia por la forma de cortarla. 

La demostración no exige suponer nada sobre las preferencias de los inversores, solo que puedan replicar por su cuenta cualquier estructura financiera, de modo que nadie esté dispuesto a pagar a la empresa por lo que puede hacer gratis.

Sus supuestos originales eran cinco: (i) mercados de capitales perfectos, sin impuestos ni costes de transacción y con inversores que no influyen en los precios; (ii) empresas agrupables en clases de riesgo, cuyos rendimientos son proporcionales entre sí; (iii) deuda sin riesgo, al mismo tipo para empresas y particulares; (iv) flujos de caja esperados constantes y perpetuos; y, (v) una política de inversión dada. 

Varios no son esenciales:\footnote{%
  STIGLITZ (1969, 1974) y FAMA (1978) mostraron que el resultado se mantiene sin clases de riesgo, con deuda arriesgada si la insolvencia no tiene costes y con cualquier número de títulos distintos, no solo con acciones y bonos.
} lo esencial es que ninguna fricción haga distinto financiarse dentro o fuera de la empresa, que los inversores puedan endeudarse en las mismas condiciones que la empresa y que la inversión no dependa de la financiación. Además, nótese que no hace falta suponer neutralidad ante el riesgo ni ausencia de incertidumbre: el teorema se formula precisamente en un mundo incierto.\footnote{%
    El arbitraje exige, además, que los inversores solo atiendan a su riqueza total, sin distinguir entre el endeudamiento de la empresa y el suyo propio ni entre cobrar intereses, dividendos o plusvalías. 
    
    MILLER y MODIGLIANI (1961) lo harán explícito, al tratar los dividendos, con su supuesto de comportamiento racional, que la contabilidad mental de la economía conductual pone en cuestión.
}

\subsubsection{Imposición: ¿puede conducir a una estructura financiera óptima?}
Con un impuesto sobre sociedades al tipo $T_c$ (el $t$ del bloque anterior), la empresa U paga $T_cX$ y reparte $(1-T_c)X$, mientras que L, que deduce los intereses, paga $T_c(X-r_DD)$ y reparte entre accionistas y acreedores $(1-T_c)X+T_cr_DD$. 

La diferencia ($T_cr_DD$) es el \textbf{ahorro fiscal} anual de la deuda, tan seguro como los propios intereses. Si la deuda es perpetua y ese ahorro se descuenta a $r_D$, su valor actual es $T_cD$ (proposición I, con deuda perpetua):

$$V_L=V_U+T_cD.$$

Por tanto, el coste medio del capital cae con el endeudamiento y, llevado al extremo, la estructura óptima sería toda deuda.\footnote{%
    Con impuestos, la proposición II pasa a ser $r_E=r_U+(r_U-r_D)(1-T_c)\,D/E$, y el coste medio del capital, $r_{\text{WACC}}=r_E\,E/V+r_D(1-T_c)\,D/V=r_U\left(1-T_c\,D/V\right)$, decrece con el endeudamiento; el desarrollo está en un anexo ([\authorfont{Ver Anexo \ref{anx:desarrollos}}]). La imagen de la pizza se extiende así: cada trozo que se vende a los accionistas obliga a dar otro al Estado, y el que se vende a los acreedores no, de modo que vender más trozos a los acreedores deja más pizza para los inversores. La ventaja fiscal de la deuda es, en todo caso, un sesgo real de los sistemas tributarios, que los legisladores han intentado acotar.\footnote{%

\paragraph{Refinamiento: ¿qué es el equilibrio de MILLER?}
MILLER (1977) señaló que la ventaja debe medirse incluyendo los impuestos personales. Un euro que la empresa paga como interés llega al acreedor como $(1-T_d)$, donde $T_d$ es el tipo personal sobre los intereses; un euro de beneficio que se paga a los accionistas llega como $(1-T_c)(1-T_e)$, donde $T_e$ es el tipo personal sobre las rentas de las acciones. La ganancia del endeudamiento con deuda perpetua es entonces (MILLER, 1977):

$$G_L=\left[1-\frac{(1-T_c)(1-T_e)}{1-T_d}\right]D.$$

Si los intereses y las rentas de las acciones tributan igual ($T_d=T_e$), la ganancia vuelve a ser $T_cD$; si los intereses tributan tanto más que se cumple $1-T_d=(1-T_c)(1-T_e)$, desaparece; y, en los casos intermedios, se reduce. 

MILLER sostuvo que el mercado conduce al segundo caso: como los impuestos personales son progresivos, las empresas emiten deuda hasta que el inversor marginal que la compra tiene un tipo personal que anula la ventaja, y en ese \textbf{equilibrio de MILLER} el endeudamiento agregado queda determinado, pero la estructura de cada empresa vuelve a ser irrelevante.\footnote{%
    El razonamiento completo del equilibrio, con el inversor marginal y la oferta y la demanda de bonos, está en un anexo ([\authorfont{Ver Anexo \ref{anx:desarrollos}}]). 
    
    La evidencia es mixta, en parte porque es difícil medir los tipos marginales de los inversores; GRAHAM (2000) encuentra que, incluso descontando los impuestos personales, la ventaja neta de la deuda sigue siendo positiva.
} 

\textcolor{magenta}{DEANGELO y MASULIS (1980) rompieron de nuevo la irrelevancia: si la empresa dispone de otras deducciones, como las amortizaciones o las deducciones por inversión, puede no tener beneficios suficientes para aprovechar todo el ahorro fiscal de la deuda, cuyo valor marginal cae al endeudarse más, y aparece un óptimo propio de cada empresa.

\subsubsection{Insolvencia: ¿tiene costes la dificultad financiera?}
Sin embargo, el freno real debía estar en otra parte. Precisamente por lo que la literatura introduce: los \textbf{costes de insolvencia}. 

Los directos (abogados, administradores concursales, tribunales) son pequeños en relación con el valor de la empresa: entre el $1\%$ y el $5,3\%$ en los ferrocarriles estadounidenses que estudió WARNER (1977), según el momento en que se midan. Sin embargo, los indirectos son mucho mayores y aparecen antes de cualquier concurso, en cuanto la empresa entra en dificultades: clientes y proveedores que se retiran por miedo a que desaparezca, empleados valiosos que se marchan, ventas precipitadas de activos, recortes en investigación y una dirección absorbida por la supervivencia. Por eso conviene hablar de costes de las \textbf{dificultades financieras}, que ANDRADE y KAPLAN (1998) estiman en torno al $10\%$ del valor de empresas muy endeudadas. No son una transferencia entre acreedores y accionistas que un inversor diversificado pueda compensar, sino recursos que se pierden; y aunque en una liquidación los soporten los acreedores, estos los anticipan y exigen más rentabilidad a medida que la empresa se endeuda, de modo que los acaban pagando los accionistas.

La empresa se endeuda, así, hasta que el valor actual del ahorro fiscal de un euro más de deuda iguala el aumento del valor actual de los costes de insolvencia esperados (KRAUS y LITZENBERGER, 1973; KIM, 1978). Es la teoría del equilibrio estático, en la formulación de MYERS (1984):

$$V_L=V_U+\text{VA(ahorro fiscal)}-\text{VA(costes de insolvencia)}.$$

La conclusión, algo de deuda pero no demasiada, puede parecer una vuelta a la visión tradicional, pero ahora tiene fundamento y hace predicciones contrastables: se endeudarán más las empresas con activos tangibles y beneficios estables y sujetos a impuesto, cuya insolvencia es menos probable y menos costosa, y menos las que dependen de intangibles y de oportunidades de crecimiento, que se pierden en una crisis.

\textcolor{magenta}{Su punto débil es la magnitud. GRAHAM (2000) estimó que el ahorro fiscal de la deuda vale el $9,7\%$ del valor de la empresa (el $4,3\%$ descontando los impuestos personales) y que la empresa típica podría duplicarlo endeudándose más antes de que su beneficio marginal empezara a caer: las empresas parecen endeudarse menos de lo que les conviene. ALMEIDA y PHILIPPON (2007) muestran que el enigma se atenúa si los costes de insolvencia se valoran ajustados por riesgo, porque las crisis llegan cuando más duelen: para una empresa con calificación BBB, su valor actual sube del $1,4\%$ al $4,5\%$ del valor previo a las dificultades, del mismo orden que el ahorro fiscal marginal. Aun así, el equilibrio estático deja sin explicar regularidades importantes, como que las empresas más rentables sean las menos endeudadas, que obligan a mirar a la información y a la agencia.\footnote{%
    En la versión dinámica, ajustar la estructura cuesta, de modo que la empresa deja que su endeudamiento fluctúe dentro de una banda y solo recapitaliza cuando sale de ella (FISCHER, HEINKEL y ZECHNER, 1989); LELAND (1994) valoró la deuda con riesgo de impago y obtuvo el endeudamiento óptimo de forma explícita. Con costes de ajuste, el endeudamiento observado puede alejarse del óptimo durante largos períodos.

\subsection{Información asimétrica: ¿qué revela la forma de financiarse?}
MODIGLIANI y MILLER suponían que todos saben lo mismo, y eso no es cierto. No toda falta de información rompe la irrelevancia. La incertidumbre que comparten todos, directivos e inversores, está ya dentro del teorema, que se formula en un mundo incierto. 

Ahora bien, si los directivos saben más que los inversores sobre el valor de la empresa, la forma de financiarse transmite información, y la empresa elige la fuente que menos se presta a ser malinterpretada: endeudarse puede ser una señal de calidad, y emitir acciones, una señal de lo contrario.

Por tanto, lo que la rompe es la \textbf{información asimétrica}: que una parte sepa algo que la otra ignora ([\authorfont{Ver Tema 3.A.13}]). Cabe en dos formas. 

Si lo que los inversores ignoran es una característica de la empresa, como el valor de sus activos o de sus proyectos (información oculta), la estructura del capital sirve para revelarla o para no tener que revelarla; es lo que tratan los modelos de señalización y el orden jerárquico de las fuentes. Si lo que ignoran es lo que hará el directivo una vez financiado (acción oculta), la estructura del capital sirve para condicionar su conducta.


\subsubsection{Información oculta: ¿...?}
ROSS (1977) aplicó a la estructura del capital la señalización de SPENCE (1973). El directivo conoce el valor de su empresa, cobra más si el mercado la valora bien hoy y sufre una penalización si la empresa quiebra mañana. 

Endeudarse, por tanto, puede ser una señal creíble porque solo la empresa buena puede permitírselo: si la mala la imitara, quebraría, y la penalización de su directivo superaría lo que ganaría con una valoración inflada. Así, en equilibrio el mercado lee el endeudamiento como un indicador de calidad, y el valor de la empresa y su endeudamiento crecen juntos.\footnote{%
    El modelo, con la remuneración del directivo y la condición para que la empresa mala no imite a la buena, está en un anexo ([\authorfont{Ver Anexo \ref{anx:desarrollos}}]). La señal funciona mejor cuanto mayor es la penalización por quiebra, cuanto más pesa el futuro en la remuneración del directivo y cuanto menor es la diferencia de valor entre los dos tipos de empresa. La idea general la había adelantado el propio ROSS: cualquier cambio en la política de financiación altera la percepción que tienen los inversores de la empresa y es, por tanto, una señal.
} 

\textcolor{magenta}{Incluir ilustración que evidencia el insight (parecido a los gráficos de selección adversa que utilizamos en otros temas si puede ser)}

LELAND y PYLE (1977) llegan a una conclusión parecida por otra vía: el empresario que conserva una parte grande de su proyecto, en lugar de venderla, demuestra que confía en él, porque mantenerla le cuesta más si el proyecto es malo, y financia el resto con deuda.

\paragraph{Jerarquía financiera: ¿diferentes fuentes emiten diferentes señales?}
MYERS y MAJLUF (1984) mostraron, además, por qué la información asimétrica ordena las fuentes. 

Supongamos que la empresa U del ejemplo tiene unos activos en funcionamiento que valen $a_A$ si es buena y $a_B$ si es mala, y que solo sus directivos saben cuál es el caso; tiene también una oportunidad de inversión que exige $I$ y cuyo valor actual neto es $b$, pero no tiene fondos para acometerla. Si emite acciones, el mercado, que no distingue, las paga a un precio que corresponde al valor medio de las dos. Para la empresa buena eso supone vender barato: los nuevos accionistas reciben una parte de la empresa que vale más de lo que pagan, y esa transferencia, $T$, sale de los antiguos accionistas, en cuyo interés actúan los directivos. La empresa buena solo emite si el proyecto compensa la transferencia ($b\geq T$); si no, renuncia a una inversión rentable. 

\imagenfit{mm84_paneles.pdf}{Información asimétrica: la decisión de emitir y el orden jerárquico}{Elaboración propia a partir de MYERS y MAJLUF (1984) y MYERS (1984). Esquema ilustrativo.}

El mercado lo anticipa: cuando ve emitir, deduce que la empresa probablemente es mala, y la cotización cae al anunciarse la ampliación, como mostraba la evidencia del bloque anterior. El descuento no es, por tanto, un supuesto del modelo, sino el resultado de equilibrio de la información asimétrica.\footnote{%
    El desarrollo, con el valor de la transferencia, $T=I\,(a_A-\overline{a})/(\overline{a}+I+b)$, donde $\overline{a}$ es el valor medio de los activos, está en un anexo ([\authorfont{Ver Anexo \ref{anx:desarrollos}}]). Visto desde los directivos, los optimistas, que creen que su acción está infravalorada, emitirán deuda, y los pesimistas, acciones: la deuda reduce la ventaja informativa de los directivos frente al mercado.
} La salida es financiarse con fondos cuyo valor no dependa de lo que el mercado ignora. Los fondos internos no exigen convencer a nadie; la deuda sin riesgo vale lo mismo sea cual sea el tipo de empresa, de modo que no hay transferencia; la deuda con riesgo provoca una transferencia pequeña, y las acciones, la mayor. 

De ahí el \textbf{orden jerárquico} (\textit{pecking order}): fondos internos primero, después deuda y, como último recurso, acciones (MYERS, 1984), un patrón que DONALDSON (1961) ya había observado al estudiar cómo se financiaban las grandes empresas estadounidenses, y que atribuía a la preferencia de los directivos por no someterse al escrutinio de los mercados.\textcolor{orange}{\textsuperscript{[Probable]}} 

La teoría no tiene un endeudamiento objetivo: la ratio de deuda de cada empresa es el resultado acumulado de sus necesidades de financiación externa. Explica así lo que el equilibrio estático no explica que las empresas más rentables se endeuden menos, no porque busquen una estructura más conservadora, sino porque tienen más fondos internos; y explica que la mayor parte de la financiación externa sea deuda.\footnote{%
    Se le han hecho dos críticas más: que, a diferencia de ROSS, no modeliza las preferencias de los directivos, y que el problema puede atenuarse con instrumentos menos sensibles a la información, como la deuda convertible. Y deja una pregunta abierta: si emitir acciones revela pesimismo, quién las compra. En equilibrio, la empresa prefiere la deuda mientras no se acerque a niveles en que el endeudamiento tenga costes, y emite acciones cuando ya no puede endeudarse más.
}

Los contrastes, sin embargo, le son desfavorables fuera de las grandes empresas maduras. SHYAM-SUNDER y MYERS (1999) encontraron que estas cubren casi todo su déficit de financiación con deuda, como predice la teoría; FRANK y GOYAL (2003), con una muestra más amplia, que las empresas pequeñas y de crecimiento rápido, que son las que más deberían sufrir la información asimétrica, emiten acciones en lugar de deuda. LEARY y ROBERTS (2010) concluyen que el orden jerárquico no clasifica bien ni la mitad de las decisiones de emisión, que añadiendo los factores de otras teorías se superan las cuatro quintas partes, y que el orden que se observa obedece más a conflictos de incentivos que a información asimétrica.

\subsubsection{Acción oculta: ¿la deuda disciplina o crea conflictos?}
Cuando lo que los inversores no pueden observar es la conducta de quien gestiona la empresa, la estructura del capital importa porque cambia sus incentivos. La deuda disciplina a los directivos, pero enfrenta a accionistas y acreedores, y la estructura óptima equilibra ambos costes de agencia.

\paragraph{Conflictos: ¿cómo condiciona la interacción de los agentes?}
JENSEN y MECKLING (1976) identificaron los dos conflictos. El primero enfrenta a los \textbf{accionistas externos contra el directivo}: cuanto menor es su participación en la empresa, menos le cuesta cada euro que se gasta en su propio beneficio (despachos, plantillas o adquisiciones que aumentan su poder) y más consume. Financiarse con deuda en lugar de con acciones externas reduce ese coste, porque concentra la propiedad y obliga a pagar. El segundo enfrenta a los \textbf{accionistas contra los acreedores} y es consecuencia de la convexidad de los fondos propios que vimos en el bloque anterior. La empresa L del ejemplo, con una deuda de nominal $F$, puede elegir entre un proyecto seguro, que vale $m$ con certeza, y otro arriesgado, que vale $m-s$ o $m+s$ con la misma probabilidad. Si $m-s<F$, el proyecto arriesgado no cambia el valor esperado de la empresa, pero traslada valor de los acreedores, que pierden en el estado malo, a los accionistas, que se quedan con lo que sobra en el bueno; y los accionistas lo preferirán incluso si vale algo menos, mientras la transferencia lo compense. Es la \textbf{sustitución de activos}.

\imagenfit{sustitucion_activos.pdf}{La sustitución de activos: el riesgo traslada valor de los acreedores a los accionistas}{Elaboración propia a partir de JENSEN y MECKLING (1976). Esquema ilustrativo.}

Los acreedores lo anticipan, exigen más interés y cláusulas restrictivas, y el coste acaba recayendo en los accionistas. El mismo conflicto produce el \textbf{sobreendeudamiento} (\textit{debt overhang}) de MYERS (1977): una empresa muy endeudada puede renunciar a inversiones rentables porque buena parte del valor que crean iría a los acreedores. Por eso las empresas cuyo valor depende de oportunidades de crecimiento futuras deberían endeudarse menos.

\paragraph{Disciplina: ¿cómo condiciona la conducta del agente?}
En sentido contrario, la deuda disciplina. JENSEN (1986) sostuvo que en las empresas maduras, que generan más fondos de los que pueden invertir con rentabilidad, endeudarse obliga a los directivos a entregar ese flujo de caja libre a los acreedores en lugar de invertirlo en crecer por crecer; las compras apalancadas de los años ochenta, en las que los directivos se convertían en dueños de la empresa tras endeudarla fuertemente, llevaron esa lógica al extremo ([\authorfont{Ver Tema 3.B.4}]). 

A diferencia del dividendo, el pago de la deuda es obligatorio, y su impago le puede costar el puesto. STULZ (1990) formalizó el equilibrio: la deuda limita la sobreinversión cuando sobran fondos, pero provoca infrainversión cuando faltan, y el endeudamiento óptimo depende de cuál de los dos riesgos pese más, es decir, de las oportunidades de crecimiento de la empresa y de la tendencia del directivo a invertir en exceso. \textcolor{magenta}{HARRIS y RAVIV (1990) añadieron que la deuda da a los inversores información y la opción de liquidar la empresa cuando el directivo, que preferiría continuar siempre, no lo haría. (tipo de cuestión que debería ir en footnote o desarrollo pertinente en anexo)}

\paragraph{Fondos propios: ¿por qué las empresas acumulan liquidez?}
La agencia explica también por qué las empresas acumulan liquidez, algo innecesario en el mundo de MODIGLIANI y MILLER, en el que siempre podrían obtener fondos en el mercado. HOLMSTRÖM y TIROLE (1998, 2011) parten de que el empresario puede apropiarse de parte de los resultados, de modo que solo una parte de ellos puede comprometerse ante los inversores. 

\textcolor{magenta}{(Ilustrar)}

Si más adelante surge una necesidad de fondos o una oportunidad de inversión, la empresa quizá no consiga financiarla en ese momento, y por eso se asegura de antemano: acumula efectivo o contrata líneas de crédito, aunque le cueste. 

En 2016, el efectivo de las empresas del índice S\&P~500 rondaba los $1,45$ billones de dólares (FERNÁNDEZ-VILLAVERDE, 2016), y BATES, KAHLE y STULZ (2009) documentaron que la ratio de efectivo sobre activos de las empresas industriales estadounidenses se había más que duplicado entre 1980 y 2006, sobre todo por motivos de precaución.\footnote{%
    FERNÁNDEZ-VILLAVERDE (2016) añade un motivo informativo: verificar el valor de la tecnología de una empresa emergente que se quiere comprar es muy costoso para un banco, de modo que la empresa que crece mediante adquisiciones, como las grandes tecnológicas, prefiere tener el efectivo a mano en lugar de pedir financiación para cada compra. Del modelo de HOLMSTRÖM y TIROLE se deriva también por qué los fondos propios rinden más que la deuda: retribuyen las rentas que el empresario necesita para no desviar resultados.
} El modelo explica asimismo diferencias entre sectores: donde los proyectos son fáciles de verificar, como en los sectores maduros, las empresas pueden financiarse con mucha deuda; donde son difíciles, como en el tecnológico, necesitan muchos fondos propios.

\subsection{La evidencia: ¿qué teoría explica los datos?}
\textcolor{magenta}{Ninguna teoría explica por sí sola cómo se financian las empresas. Cada una acierta con algunos determinantes del endeudamiento y falla con otros, y la estructura del capital observada se entiende mejor como el resultado acumulado de decisiones condicionadas por varias fricciones a la vez.

Los determinantes del endeudamiento son notablemente estables entre países y períodos. RAJAN y ZINGALES (1995) encontraron en los países del G-7 que se endeudan más las empresas con más activos tangibles y las más grandes, y menos las más rentables y las que tienen más oportunidades de crecimiento, medidas por la relación entre su valor de mercado y su valor contable. FRANK y GOYAL (2009), con empresas estadounidenses entre 1950 y 2003, identifican seis factores fiables del endeudamiento a valor de mercado: el endeudamiento medio del sector, la tangibilidad de los activos, el tamaño y la inflación esperada, que lo aumentan, y la relación entre valor de mercado y contable y la rentabilidad, que lo reducen. La figura compara esos signos con los que predice cada teoría.\textcolor{orange}{\textsuperscript{[Probable]}}

El equilibrio estático acierta con la tangibilidad, el tamaño, el sector y las oportunidades de crecimiento, pero no con la rentabilidad; el orden jerárquico acierta con la rentabilidad y casi con nada más; la agencia explica el efecto de las oportunidades de crecimiento. Dos regularidades complican además el cuadro. El endeudamiento es muy persistente: las empresas que salen a bolsa muy o poco endeudadas siguen estándolo veinte años después, y los determinantes habituales explican poco de esas diferencias (LEMMON, ROBERTS y ZENDER, 2008). Y las empresas no reequilibran su estructura: emiten acciones cuando su cotización es alta, de modo que su endeudamiento es el resultado acumulado de sus intentos de aprovechar el mercado (BAKER y WURGLER, 2002), y dejan que su endeudamiento a valor de mercado se mueva con la cotización sin corregirlo (WELCH, 2004). Los directores financieros confirman esa mezcla: al decidir sobre la deuda les preocupan sobre todo la flexibilidad financiera y la calificación crediticia, y al decidir sobre las acciones, la dilución del beneficio por acción y la evolución reciente de la cotización; sus respuestas dan algún apoyo al equilibrio estático y al orden jerárquico, pero apenas muestran preocupación por la sustitución de activos, la información asimétrica o el flujo de caja libre (GRAHAM y HARVEY, 2001).

En la práctica, la estructura del capital depende también de la fase de la vida de la empresa: en sus comienzos, con riesgos operativos muy altos, apenas tiene acceso a la deuda y se financia con fondos propios; madura y con flujos estables, prefiere la deuda. Y depende de su sector, del sistema fiscal, de la coyuntura (la inflación favorece el endeudamiento), del tamaño, de su grado de internacionalización y de la aversión al riesgo de directivos e inversores en cada momento. Como resumió MYERS (2001), no hay una teoría universal de la elección entre deuda y fondos propios, ni razón para esperarla, sino varias teorías condicionales útiles, cada una válida para las empresas en que domina la fricción que destaca.\textcolor{orange}{\textsuperscript{[Probable]}} Esa es la tesis del bloque: el teorema de MODIGLIANI y MILLER no dice que la estructura del capital no importe, sino que importa por razones concretas, distintas en cada empresa.

\section{Política de dividendos: ¿importa cómo se reparte el beneficio?}
\puntosclave{
    La política de dividendos decide qué parte del beneficio se reparte y cómo: dividendos o recompras de acciones. En la fórmula del crecimiento sostenible, repartir es la otra cara de retener.

    Sin fricciones, es irrelevante (MILLER y MODIGLIANI, 1961): dada la inversión, lo que el accionista cobra en efectivo lo pierde en el valor de sus acciones, y puede fabricarse por su cuenta el dividendo que quiera. El argumento del pájaro en mano es una falacia.

    Las empresas suavizan sus dividendos y los recortan a regañadientes; cada vez reparten más mediante recompras, y cada vez menos empresas pagan dividendos.

    Si los dividendos tributan más que las plusvalías, no deberían pagarse: es el enigma de los dividendos. Los impuestos explican cómo se reparten los inversores entre empresas, pero no por qué se paga.

    Las regularidades encajan mejor con la información (el dividendo como señal), la agencia (sacar el efectivo de las manos de los directivos) y la forma en que los inversores perciben las ganancias que con los impuestos.
}

Los dos bloques anteriores se preguntaban cómo se financia la inversión y si la mezcla de deuda y fondos propios altera el valor de la empresa. 

Este se ocupa de la otra cara de la financiación interna: qué hacer con el beneficio que no se reinvierte. En la tasa de crecimiento sostenible del primer bloque, la proporción que se retiene, $b$, y la que se reparte, $1-b$, son la misma decisión vista desde dos lados; hasta ahora la pregunta era cómo minimizar el coste del capital, y ahora es si el reparto altera el valor de las acciones. 

La respuesta sigue la misma pauta que en la estructura del capital: sin fricciones, la política de dividendos es irrelevante (MILLER y MODIGLIANI, 1961), porque el accionista puede fabricarse el dividendo que quiera vendiendo acciones. Como veremos, las fricciones que la hacen relevante apuntan en direcciones distintas, y las regularidades observadas se explican mejor por la información y la agencia que por los impuestos.

\subsection{Qué se decide y qué se observa: ¿cómo reparten las empresas?}
La política de dividendos decide qué parte del beneficio se reparte y de qué forma. Las empresas lo hacen con una estabilidad notable y cada vez más mediante recompras, contrariamente a lo que sucedió durante décadas, cuando se creía que repartir más elevaba el valor. Como vemos, ninguna de las dos cosas encaja a primera vista con una decisión irrelevante.

La empresa puede repartir su beneficio de varias formas. El \textbf{dividendo} ordinario se paga periódicamente en proporción a las acciones; el extraordinario reparte ganancias que no se espera que se repitan; el dividendo a cuenta adelanta el del ejercicio, y el dividendo en acciones permite al accionista elegir entre cobrar en efectivo o recibir acciones nuevas.\footnote{%
    Existen también dividendos preferentes, que se pagan a las acciones preferentes o, en algunas jurisdicciones, a los accionistas que han mantenido sus acciones un tiempo mínimo. El dividendo a cuenta permite, además, suavizar el perfil de tesorería de la empresa. Las empresas de sectores muy cíclicos tienen dividendos más volátiles.
} La \textbf{recompra de acciones} devuelve el efectivo comprando acciones propias, normalmente en el mercado, y amortizándolas. Si la empresa recompra a todos los accionistas en proporción a sus participaciones, el efecto es el mismo que el de un dividendo; en la práctica, difieren en tres cosas: la recompra puede alterar el reparto del control, porque solo venden quienes quieren; es más flexible, porque no crea la expectativa de repetirse; y transmite menos información sobre los resultados futuros.\footnote{%
    En la redención, la empresa obliga a los accionistas a venderle acciones que se emitieron como rescatables, a un precio fijado al emitirlas. La recompra altera también algunas ratios, como la de endeudamiento, y, en algunas jurisdicciones, obliga a ajustar el precio de ejercicio de las opciones sobre acciones.
}

Las formas de reparto presentan regularidades muy estables (ALLEN y MICHAELY, 2003).\textcolor{orange}{\textsuperscript{[Probable]}} Las grandes empresas consolidadas reparten una parte importante de sus beneficios. Los dividendos fueron la forma dominante hasta mediados de los años ochenta, y desde entonces las recompras no han dejado de ganar peso: en 2024, las empresas del S\&P~500 dedicaron $942,5$ mil millones de dólares a recompras y $629,6$ mil millones a dividendos (CAREY, 2025). La proporción de empresas cotizadas estadounidenses que pagan dividendos cayó del $66,5\%$ en 1978 al $20,8\%$ en 1999, en parte porque las nuevas cotizadas eran pequeñas, poco rentables y de crecimiento rápido, pero sobre todo porque se redujo la propensión a pagar (FAMA y FRENCH, 2001). Las recompras son más volátiles que los dividendos; los inversores con tipos impositivos altos reciben, aun así, muchos dividendos; y el mercado reacciona positivamente a los anuncios de aumentos de dividendos y de recompras.\footnote{%
    GRULLON y MICHAELY (2002) muestran que las recompras han sustituido en parte a los dividendos: las empresas que antes habrían aumentado su dividendo financian ahora recompras. La encuesta de BRAV, GRAHAM, HARVEY y MICHAELY (2005) confirma que los directivos siguen considerando casi intocable el nivel del dividendo, mientras que usan las recompras con flexibilidad, para repartir el efectivo residual.
}

LINTNER (1956), tras entrevistar a los directivos de grandes empresas estadounidenses, describió cómo se fija el dividendo. Cada empresa tiene una proporción objetivo de reparto, $p$, y, como sus directivos solo suben el dividendo cuando creen que podrán mantenerlo y son muy reacios a recortarlo, se acerca a ella solo en una fracción $c$ cada año (modelo de ajuste parcial de LINTNER, 1956):
$$D_t-D_{t-1}=c\left(p\,E_t-D_{t-1}\right).$$
El dividendo sigue así al beneficio con retraso y suavizado, y los aumentos de beneficio que se consideran transitorios apenas se trasladan a él.\footnote{%
    FAMA y BABIAK (1968) confirmaron el modelo con datos de cientos de empresas, y sigue describiendo bien la conducta de las grandes empresas que pagan dividendos. El desarrollo, en un anexo ([\authorfont{Ver Anexo \ref{anx:desarrollos}}]).
}

\imagenfit{lintner_ajuste.pdf}{El ajuste parcial de LINTNER: el dividendo sigue al beneficio con retraso y suavizado}{Elaboración propia a partir de LINTNER (1956). Valores ilustrativos.}

Frente a la irrelevancia, la visión tradicional sostenía que repartir más eleva el valor de la acción, porque un euro de dividendo hoy es más seguro que la promesa de ganancias futuras: más vale pájaro en mano (GORDON, 1959, 1962; LINTNER, 1962). WALTER (1956) y GORDON (1959) lo plantearon con un modelo de crecimiento. Si la empresa retiene la proporción $b$ de su beneficio y la reinvierte a la rentabilidad $y$, su beneficio crece a la tasa $g=b\,y$, la tasa de crecimiento sostenible del primer bloque, y el valor de la acción, descontado a la rentabilidad exigida $r_E$, es (modelo de crecimiento de WALTER, 1956, y GORDON, 1959, con rentabilidad de la reinversión constante):
$$P_0=\frac{(1-b)\,B_1}{r_E-b\,y}.$$
Retener eleva el valor si $y>r_E$, lo reduce si $y<r_E$ y no lo altera si $y=r_E$. La política de dividendos importa aquí solo porque encubre una decisión de inversión: la de reinvertir a una rentabilidad distinta de la que exigen los accionistas.\footnote{%
    El desarrollo, con la derivada del valor respecto de la proporción retenida, está en un anexo ([\authorfont{Ver Anexo \ref{anx:desarrollos}}]). El modelo supone que la rentabilidad de la reinversión es constante, lo que difícilmente se sostiene, y que la empresa no recurre a la financiación externa. GORDON y LINTNER añadieron que $r_E$ crece con la proporción retenida, porque los beneficios retenidos se perciben como más arriesgados que los dividendos: de ahí el pájaro en mano. Una versión popular de esta visión sostenía que un euro de dividendos vale tanto como tres retenidos.
}

\subsection{MILLER y MODIGLIANI (1961): ¿por qué el reparto sería irrelevante?}
Dadas las decisiones de inversión y de endeudamiento, el dividendo es solo la diferencia entre los fondos que genera la empresa y los que invierte: si reparte más, tiene que emitir más acciones o recomprar menos, y lo que el accionista gana en efectivo lo pierde en el valor de sus acciones.

MILLER y MODIGLIANI (1961) parten de los mismos mercados perfectos que en 1958 (sin impuestos ni costes de transacción, con información igual y gratuita para todos), de una política de inversión dada y de un supuesto de \textbf{comportamiento racional}: los inversores prefieren más riqueza a menos y les da igual recibirla como dividendo o como revalorización de sus acciones. Es la fungibilidad del dinero, que el trabajo de 1958 daba por supuesta sin decirlo: un euro es un euro, venga del dividendo o de la venta de acciones.

El argumento se ve con la empresa U del ejemplo. Ha decidido ya sus inversiones y tiene, además, un efectivo excedente $C$; sus activos en funcionamiento valen $V$ y tiene $n$ acciones, que valen $P_0=(V+C)/n$. Puede usar el efectivo de tres formas. Si lo reparte como dividendo, cada accionista cobra $C/n$ y su acción pasa a valer $V/n$: su riqueza sigue siendo $P_0$, y el precio cae exactamente en la cuantía del dividendo. Si recompra acciones al precio $P_0$, quedan $n-C/P_0$ acciones que se reparten un valor $V$, y cada una sigue valiendo $P_0$: quien vende cobra lo que vale su acción y quien no vende conserva una que vale lo mismo. Y si quiere repartir más de lo que tiene, un dividendo $d$ mayor que $C/n$, debe emitir acciones nuevas por la diferencia; los nuevos accionistas pagan lo que valen, y los antiguos cobran $d$, pero su acción pasa a valer $P_0-d$. En los tres casos la riqueza del accionista es la misma: el valor de la empresa no ha cambiado, y la política de dividendos solo reparte de otra forma en el tiempo el valor actual de lo que recibe.\footnote{%
    La demostración general, en la que el dividendo desaparece de la ecuación del valor una vez fijadas la inversión y la financiación, está en un anexo ([\authorfont{Ver Anexo \ref{anx:desarrollos}}]). Como escribieron los propios autores, la irrelevancia de la política de dividendos, dada la política de inversión, es evidente una vez que se piensa en ella. Su aportación fue mostrar que valorar la empresa por sus dividendos futuros o por sus beneficios futuros, descontada la inversión, da el mismo resultado.
}

\imagenfit{mm61_paneles.pdf}{MILLER y MODIGLIANI: la riqueza del accionista no depende de la forma de reparto}{Elaboración propia a partir de MILLER y MODIGLIANI (1961). Esquema ilustrativo.}

Si el accionista prefiere efectivo y la empresa no reparte, puede vender una parte de sus acciones; si prefiere reinvertir y la empresa reparte, puede comprar más acciones con el dividendo. Puede fabricarse su propio \textbf{dividendo casero}, como en 1958 se fabricaba el endeudamiento, y por eso no pagará más por una empresa que reparta de una forma que por otra. El arbitraje garantiza, además, que el día en que la acción deja de dar derecho al dividendo su precio caiga exactamente en la cuantía de este: si cayera menos, bastaría comprar justo antes, cobrar el dividendo y vender justo después para ganar sin riesgo.

De ahí que el argumento del pájaro en mano sea una falacia: el dividendo de hoy no reduce el riesgo de la empresa, solo cambia la forma en que el accionista lo soporta, y la caída de la cotización compensa exactamente el efectivo recibido; BHATTACHARYA (1979) lo llamó, precisamente, la falacia del pájaro en mano. DEANGELO y DEANGELO (2006) han objetado que el teorema es menos general de lo que parece, porque descansa en que la empresa reparta siempre todo el flujo de caja libre que no invierte. Si puede retenerlo e invertirlo sin rentabilidad, la política de reparto sí afecta al valor: la irrelevancia solo se cumple entre políticas que reparten todo lo que la empresa no puede invertir con valor actual neto positivo.

\subsection{Impuestos y clientelas: ¿por qué pagar dividendos si tributan más?}
Si los dividendos tributan más que las plusvalías, lo racional sería no repartirlos y devolver el efectivo mediante recompras; que las empresas sigan pagándolos es el \textbf{enigma de los dividendos} (BLACK, 1976). Los impuestos explican cómo se reparten los inversores entre empresas, pero no por qué se paga.

Durante buena parte del siglo XX, en EE. UU. los dividendos tributaban en el impuesto sobre la renta al tipo marginal del inversor, y las plusvalías a un tipo menor y solo al venderse la acción. BRENNAN (1970) mostró que, en equilibrio, las acciones con mayor rentabilidad por dividendo deberían ofrecer una rentabilidad antes de impuestos mayor para compensarlo, y LITZENBERGER y RAMASWAMY (1979) encontraron ese efecto en los datos, aunque su magnitud se ha discutido. Las plusvalías tienen, además, la ventaja de que el inversor elige cuándo tributa: puede aplazar la venta o realizar antes las pérdidas que las ganancias (CONSTANTINIDES, 1984). Ese mismo motivo fiscal explica que las empresas acumulen efectivo en lugar de repartirlo.\footnote{%
    FERNÁNDEZ-VILLAVERDE (2016) recuerda que las empresas estadounidenses repartían pocos dividendos mientras estos tributaban más que las plusvalías, porque retener elevaba la cotización y el accionista podía vender cuando quisiera, a un tipo menor. La reforma de 2003, que gravó los dividendos al $15\%$, cambió esa conducta: Microsoft, que nunca había pagado dividendos, empezó a hacerlo ese año (no en 2006, como dice la fuente) y en 2004 pagó un dividendo extraordinario.\textcolor{orange}{\textsuperscript{[Probable]}} Un motivo parecido llevaba a las multinacionales estadounidenses a retener en el exterior los beneficios de sus filiales para no tributar al repatriarlos. En España, los dividendos y las plusvalías tributan en la base del ahorro del IRPF con la misma escala, del $19\%$ al $30\%$ desde 2025 (arts. $46$ y $66$ de la Ley 35/2006, modificados por la Ley 7/2024), de modo que la diferencia se reduce al aplazamiento de la plusvalía hasta la venta. EE. UU. grava desde 2023 las recompras de las sociedades cotizadas con un impuesto especial del $1\%$ (\textit{Inflation Reduction Act} de 2022).
}

Como los inversores soportan tipos distintos, cada uno debería preferir las empresas cuya política le conviene: los de tipos altos, las que reparten poco; las instituciones exentas, como los fondos de pensiones, cualquiera; y las sociedades, que en muchas jurisdicciones apenas tributan por los dividendos que reciben de otras, las que reparten mucho. Es el \textbf{efecto clientela}, que ya anticiparon MILLER y MODIGLIANI (1961) y que los datos confirman en parte (BLUME, CROCKETT y FRIEND, 1974).\textcolor{orange}{\textsuperscript{[Probable]}} Pero, como en el equilibrio de MILLER, una vez que cada clientela encuentra su empresa, ninguna puede ganar valor cambiando de política: si las que no reparten valieran más, otras dejarían de repartir hasta igualar los precios. Los costes de transacción refuerzan el efecto, porque fabricarse el dividendo propio comprando y vendiendo acciones cuesta. El precio el día en que la acción deja de dar derecho al dividendo revela el tipo del inversor marginal: si los dividendos tributan al tipo $T_d$ y las plusvalías al tipo $T_g$, el precio no cae en la cuantía del dividendo, sino en esa cuantía multiplicada por $(1-T_d)/(1-T_g)$, menos si los dividendos tributan más (ELTON y GRUBER, 1970).

\imagenfit{exdividendo.pdf}{El precio el día en que la acción deja de dar derecho al dividendo}{Elaboración propia a partir de ELTON y GRUBER (1970). Esquema ilustrativo.}

Los modelos estáticos, sin embargo, chocan con la evidencia, porque los inversores pueden operar en el tiempo. MILLER y SCHOLES (1978) mostraron que, combinando endeudamiento e inversiones exentas, un inversor podría evitar el impuesto sobre los dividendos, aunque pocos lo hacen. Los inversores con el mismo tipo para dividendos y plusvalías pueden comprar la acción justo antes del dividendo y venderla justo después, lo que acerca la caída del precio a la cuantía del dividendo (KALAY, 1982); el volumen negociado en torno a esa fecha se dispara, más en las acciones con menores costes de transacción (MICHAELY y VILA, 1996). Y los inversores con tipos altos reciben muchos dividendos, contra lo que predice el modelo estático. Los impuestos sí importan en el margen: tras la reforma estadounidense de 2003, que rebajó el tipo de los dividendos del máximo del $35\%$ al $15\%$, los dividendos de las empresas no financieras aumentaron un $20\%$ y se multiplicaron las que empezaban a pagarlos (CHETTY y SAEZ, 2005). Explican cuándo se reparte más, pero no por qué se reparte mediante dividendos.

\subsection{Información, agencia y comportamiento: ¿qué explica las regularidades?}
Las regularidades que los impuestos no explican (la estabilidad de los dividendos, la reacción de las cotizaciones a sus cambios y su sustitución por recompras) encajan mejor con el valor informativo de los dividendos, con su papel frente a los conflictos de agencia y con la forma en que los inversores perciben las ganancias.

Si los directivos saben más que el mercado sobre los flujos futuros, el dividendo los revela. MILLER y MODIGLIANI (1961) ya advirtieron que lo que mueve el precio tras un aumento del dividendo no es el dividendo, sino la información que contiene. Como repartir es costoso para una empresa que no genera fondos, que tendría que acudir a una financiación externa cara o renunciar a inversiones, solo la que tiene buenas perspectivas puede comprometerse con un dividendo alto, y el dividendo es una señal creíble: a diferencia del beneficio contable, que admite cierto maquillaje, un dividendo en efectivo no puede fingirse (BHATTACHARYA, 1979; MILLER y ROCK, 1985; JOHN y WILLIAMS, 1985). La teoría explica la rigidez que observó LINTNER, porque subir el dividendo es comprometerse, y la reacción del mercado: GRULLON, MICHAELY y SWAMINATHAN (2002) estiman rentabilidades anormales del $1,34\%$ ante aumentos significativos y del $-3,71\%$ ante reducciones, una asimetría que indica que los recortes contienen más información.\textcolor{orange}{\textsuperscript{[Probable]}} Pero la evidencia no confirma lo esencial: los cambios del dividendo reflejan los beneficios pasados y presentes más que los futuros (BENARTZI, MICHAELY y THALER, 1997), y las empresas que lo aumentan no ven crecer después sus beneficios, sino reducirse su riesgo, como corresponde a empresas que maduran (GRULLON, MICHAELY y SWAMINATHAN, 2002). Tampoco se entiende, si dividendo y recompra son sustitutos, que se señalice con el que más tributa. ALLEN, BERNARDO y WELCH (2000) lo explican porque el dividendo atrae a las instituciones, mejor informadas, sin la desventaja fiscal y a menudo obligadas por sus normas a invertir en empresas que lo pagan, cuya presencia en el capital revela la calidad de la empresa y explica también por qué las empresas son reacias a recortarlo.\footnote{%
    Las teorías de señalización implican tres hipótesis contrastables: que los cambios del dividendo vayan seguidos de cambios del beneficio en el mismo sentido; que los cambios no anticipados del dividendo vayan acompañados de variaciones de la cotización en el mismo sentido; y que lleven a revisar en el mismo sentido los beneficios esperados. Son condiciones necesarias, pero no suficientes, y la literatura ha contrastado sobre todo la segunda, que suele confirmarse; la primera, en cambio, falla.
}

Para la teoría de la agencia, el dividendo saca el efectivo de las manos de los directivos. EASTERBROOK (1984) añadió que, al obligar a la empresa a volver al mercado para financiarse, el dividendo la somete a la supervisión de bancos e inversores; JENSEN (1986), que reduce el flujo de caja libre que los directivos podrían gastar en crecer por crecer, como vimos con la deuda. El dividendo disciplina más que la recompra precisamente por su rigidez: recortarlo se castiga. LA PORTA et al. (2000) encuentran que se reparte más donde la ley protege mejor a los accionistas minoritarios: el dividendo es el resultado de esa protección, que permite a los minoritarios forzar el reparto, más que un sustituto de ella. Y DEANGELO, DEANGELO y STULZ (2006) muestran que reparten las empresas maduras, cuyos fondos propios proceden sobre todo de beneficios acumulados, y no las jóvenes, que dependen de las aportaciones de capital: la política de dividendos sigue el ciclo de vida de la empresa.\footnote{%
    Los modelos de agencia derivan predicciones menos nítidas que los de señalización (el efecto de un aumento del dividendo depende, por ejemplo, de si llega antes o después de que el directivo invierta mal), pero coinciden en una: el problema es más grave en las empresas estables, con mucha tesorería y pocas oportunidades de crecimiento, que es donde más se reparte. En marzo de 2020, el Banco Central Europeo recomendó a los bancos que supervisa no pagar dividendos ni recomprar acciones al menos hasta octubre; prorrogó la recomendación hasta enero de 2021, la sustituyó después por una de prudencia extrema y la retiró en julio de 2021. Era la primera vez que lo pedía a bancos que cumplían sus requisitos de capital ([\authorfont{Ver Tema 3.B.27}]).
}

Por último, los inversores no siempre tratan el dinero como fungible, contra el supuesto de MILLER y MODIGLIANI. SHEFRIN y STATMAN (1984) explican la preferencia por el dividendo con la contabilidad mental y el autocontrol: el inversor que se impone gastar solo los dividendos y no tocar el capital necesita que la empresa los pague. Y, como según la teoría prospectiva las ganancias separadas se disfrutan más que una sola de igual cuantía y las pérdidas pesan menos si se compensan con una ganancia (KAHNEMAN y TVERSKY, 1979), el accionista prefiere cobrar un pequeño dividendo y una plusvalía antes que una plusvalía mayor, y un pequeño dividendo junto a una pérdida antes que una pérdida menor sin dividendo.\footnote{%
    La economía conductual ofrece también una explicación del enigma de la volatilidad, el hecho de que las cotizaciones, que en teoría son el valor actual de los dividendos futuros, oscilen mucho más que ellos. Tras acumular ganancias, los inversores pierden aversión al riesgo, porque las pérdidas se ven como menores ganancias (el efecto de jugar con el dinero de la casa; THALER y JOHNSON, 1990), lo que eleva las cotizaciones en las fases alcistas y las hunde en las bajistas.
} BAKER y WURGLER (2004) añaden que las empresas complacen a los inversores (\textit{catering}): empiezan a pagar dividendos cuando el mercado valora con prima a las que los pagan, como ocurre en las fases bajistas, en que se aprecia la seguridad del efectivo, y dejan de hacerlo cuando la prima desaparece, lo que explica que la proporción de empresas que reparten suba y baje con el tiempo.\footnote{%
    El auge de las recompras ha abierto un debate de política. Sus defensores las ven como la forma más flexible y fiscalmente eficiente de devolver el efectivo que la empresa no puede invertir, y las rentabilidades anormales que siguen a su anuncio sugieren que las empresas recompran cuando sus acciones están infravaloradas (IKENBERRY, LAKONISHOK y VERMAELEN, 1995). Sus críticos sostienen que sirven para sostener el beneficio por acción y la remuneración de los directivos a costa de la inversión (LAZONICK, 2014).
}

La política de dividendos resulta así, como la estructura del capital, irrelevante en un mundo sin fricciones y relevante por razones concretas. A diferencia de ella, las razones que mejor explican los datos no son fiscales, sino informativas, de agencia y de comportamiento.

\anexo{Instrumentos de financiación}\label{anx:instrumentos}
Este anexo reúne los instrumentos con que se materializan las fuentes de financiación del primer bloque: sus rasgos jurídicos y operativos, sus variantes y los criterios prácticos que orientan la elección entre ellos. Va fuera del cuerpo porque el bloque se ocupa de por qué existe cada fuente y qué empresas la usan, y aquí se describe cómo funciona cada instrumento.

\subsection{Autofinanciación}
\subsubsection{Reservas}
Las reservas son los beneficios de ejercicios anteriores que no se han repartido como dividendos. Pueden ser legales, estatutarias o voluntarias. La reserva legal obliga a destinar el $10\%$ del beneficio del ejercicio hasta que alcance al menos el $20\%$ del capital social (art. $274$ de la Ley de Sociedades de Capital); las estatutarias dependen de lo que dispongan los estatutos, y las voluntarias, de la decisión de la junta. Todas aumentan los fondos propios y constituyen la autofinanciación de enriquecimiento, el componente principal de la autofinanciación.

\subsubsection{Amortizaciones y provisiones}
Las amortizaciones imputan cada año a los gastos la depreciación de los activos. Pueden calcularse con cuotas fijas, por el método lineal, $A=(V_0-V_R)/n$, donde $V_0$ es el valor de adquisición, $V_R$ el valor residual y $n$ la vida útil, o con métodos acelerados, cuyas cuotas son mayores al principio, por ejemplo aplicando un coeficiente constante a una base que decrece. Sin ellas, la reposición de un activo al final de su vida recaería de golpe sobre el resultado y la tesorería de un solo ejercicio. Las provisiones son retenciones de beneficio vinculadas a riesgos concretos de la actividad, que pueden materializarse o no; a diferencia de las reservas, se registran en el pasivo. Unas y otras evitan que el patrimonio neto disminuya, por lo que forman la autofinanciación de mantenimiento.

\subsubsection{Efecto multiplicador}
Si la empresa quiere mantener constante su ratio de endeudamiento, $\alpha=D/E$, todo aumento de los fondos propios le permite, y le exige, aumentar su deuda en $\alpha$ veces esa cuantía. Los recursos totales varían entonces en
$$\Delta(E+D)=(1+\alpha)\,\Delta E,$$
de modo que la autofinanciación tiene un efecto expansivo sobre la capacidad de inversión, mayor cuanto mayor sea el endeudamiento que la empresa quiere mantener.

\subsubsection{Efecto LOHMANN-RUCHTI}
Si se cumplen cuatro supuestos (los activos no quedan obsoletos, de modo que su productividad no cambia durante su vida; son perfectamente divisibles; no hay inflación; y la amortización libera recursos financieros, porque los ingresos cubren los gastos), las cuotas de amortización reinvertidas no solo mantienen la capacidad productiva, sino que la aumentan. Una empresa que compra $k$ activos de vida útil $n$ y valor residual nulo, y los amortiza linealmente, recupera cada año $k/n$ veces el precio de un activo; si reinvierte esa cuota, compra activos nuevos que a su vez se amortizan y se reinvierten. La capacidad crece hasta estabilizarse, con amortización lineal, en torno a $2n/(n+1)$ veces la inicial.\textcolor{orange}{\textsuperscript{[Probable]}}\footnote{%
    Con una flota de diez barcos de vida útil de diez años, la cuota del primer año equivale al precio de un barco, y la flota pasa a once; con el tiempo, se estabilizaría en torno a $18$ barcos ($2\cdot 10/11\approx 1,8$ veces la inicial). El efecto descansa en que la capacidad de un activo no disminuye con su valor contable: si los activos pierden productividad con la edad o hay inflación, se reduce o desaparece.
}

\subsubsection{Ventajas e inconvenientes}
La autofinanciación da a la empresa autonomía y libertad de acción, porque no tiene que acudir a los mercados de capitales ni negociar con inversores; es especialmente valiosa cuando hay problemas de información, que encarecen la financiación externa; y, para las pymes, es la principal alternativa al crédito bancario para financiarse a largo plazo. Sus inconvenientes son que puede ser insuficiente, porque las necesidades de inversión se presentan de forma discontinua y la autofinanciación se genera de forma gradual; que, aun siendo suficiente, no es siempre la fuente menos costosa; que, al no tener un coste explícito, puede llevar a acometer inversiones poco rentables; y que retener beneficios reduce los dividendos, lo que, según la teoría que se adopte, puede afectar a la cotización, como veremos al tratar la política de dividendos.

\subsection{Acciones}
\subsubsection{Características}
Las acciones son partes alícuotas del capital social. Son títulos de renta variable: su remuneración depende de la evolución económica y financiera del emisor y no de una relación contractual de pagos. No tienen vencimiento, por lo que el accionista solo recupera su aportación vendiéndolas en el mercado secundario o en la liquidación de la sociedad. Dan derechos económicos (participación en los beneficios, derecho de preferencia en las ampliaciones) y políticos (voto, asistencia a la junta e información), y en caso de liquidación se cobran después de todos los acreedores. Contablemente, forman parte del patrimonio neto del emisor y del activo del inversor ([\authorfont{Ver Tema 3.B.23}]).

\subsubsection{Clases}
Según la forma de designar al titular, las acciones son nominativas o al portador. Según los derechos que incorporan, son ordinarias, preferentes (con algún derecho adicional, como un mayor dividendo o preferencia en el cobro) o sin voto. Según el precio de emisión, se emiten a la par, por su valor nominal, o sobre la par, con prima de emisión; en España no pueden emitirse por debajo de su valor nominal (art. $59$ de la Ley de Sociedades de Capital). Según su origen, pueden ser nuevas, procedentes de una ampliación con aportaciones, o liberadas, total o parcialmente, cuando la ampliación se hace con cargo a reservas y el accionista no paga, o paga solo en parte, las acciones que recibe.

\subsubsection{Mercados}
Las empresas jóvenes captan fondos propios en los mercados privados, de capital riesgo y capital privado, más concentrados y sin acceso de los inversores minoristas. La salida a bolsa amplía el número de inversores potenciales y da liquidez a los accionistas, a cambio de dispersar la propiedad y de asumir obligaciones de información. Sus dos modalidades más comunes son la oferta pública inicial y la fusión con una sociedad cotizada creada para ello, sin actividad propia (\textit{special purpose acquisition company}, SPAC).

\subsection{Deuda}
\subsubsection{Obligaciones y préstamos}
Las obligaciones son partes alícuotas de un empréstito, como las acciones lo son del capital. El obligacionista se diferencia del accionista en que recibe una renta periódica pactada y no una parte de los resultados; en que recupera su aportación al vencimiento; en que es acreedor y no socio, con derechos económicos pero no políticos; en que asume menos riesgo, porque cobra antes en caso de liquidación; y en que, para el emisor, su aportación es financiación ajena ([\authorfont{Ver Tema 3.B.24}]). Los préstamos, como las obligaciones, son recursos de terceros que se devuelven con intereses en plazos y cuantías fijados, pero los concede, por lo general, un solo acreedor. Los de corto plazo suelen financiar la tesorería, y los de largo plazo, el activo no corriente.

\subsubsection{Banca frente a mercado: criterios prácticos}
Los préstamos bancarios siguen una lógica de negociación e intermediación, y la emisión de deuda, una lógica de mercado. Suelen compararse con cinco criterios:
\begin{lnum}
    \item \textbf{Coste}. Puede ser mayor o menor en cualquiera de las dos vías, porque el préstamo incorpora un margen de intermediación, la emisión tiene costes que el préstamo no tiene, y el tipo de interés de un préstamo puede no corresponder con el coste real de financiar a la empresa;
    \item \textbf{Volumen}. La emisión exige volúmenes mínimos, por razones institucionales (las plazas financieras suelen exigirlos) y económicas: para que su deuda sea atractiva, la empresa necesita que tenga liquidez suficiente y emitir a distintos plazos para construir una curva de tipos. En la Unión, las empresas medianas y grandes, de más de $50$ empleados, emiten bonos unos $4$ puntos porcentuales más a menudo que las pequeñas (DURANTE et al., 2023);
    \item \textbf{Desarrollo financiero}. Además de las características de cada empresa, el desarrollo del sistema financiero de un país y su crecimiento económico están positivamente relacionados con el uso de la financiación de mercado (DE JONG, KABIR y NGUYEN, 2008). La financiación bancaria predomina en la zona del euro, donde los préstamos bancarios a las sociedades no financieras casi triplican los valores de deuda que estas emiten, y la de mercado en EE. UU. y otros países anglosajones;
    \item \textbf{Rapidez}. Un préstamo bancario suele obtenerse con rapidez, mientras que preparar la primera emisión puede llevar semanas y no garantiza captar los fondos; en cambio, las grandes empresas con buena calidad crediticia pueden captar mucho dinero en pocas horas en el mercado; y,
    \item \textbf{Información}. La relación más estrecha entre prestamista y prestatario hace más apropiado el préstamo bancario cuando la empresa es poco transparente (RAJAN y ZINGALES, 2003).
\end{lnum}

\subsubsection{Colocación privada}
Como la emisión requiere un volumen mínimo elevado para generar liquidez, las empresas pequeñas tienen dificultades para acceder a ella. La colocación privada (\textit{private placement}) consiste en emitir entre inversores preseleccionados con los que se negocia directamente, en lugar de en el mercado. Como esos inversores suelen mantener los bonos hasta el vencimiento, no necesitan liquidez y pueden emitirse cantidades pequeñas; pero la falta de documentación normalizada encarece la emisión y limita su uso por las pymes. Es atractiva para las empresas que quieren diversificar sus fuentes y acceder a financiación a largo plazo sin necesidad de calificación crediticia, aunque se aleja de la lógica de mercado y se acerca a la negociación bilateral.

\subsubsection{Préstamos sindicados}
Cuando el importe es tan elevado que un solo banco no estaría dispuesto a prestarlo, el banco que lidera la operación (\textit{mandated lead arranger}) invita a otros a formar un sindicato que aporte la cantidad total. La sindicación puede ser voluntaria, si el banco director se limita a conseguir lo que el mercado permita, o asegurada, si se compromete a cubrir lo que falte. Desde la teoría de la información y de la agencia, la lógica del banco como supervisor delegado de sus depositantes (DIAMOND, 1984) se aplica también aquí: los bancos del sindicato delegan la supervisión en el banco director, lo que ahorra costes y permite especializarse; y, frente a la emisión de bonos, la sindicación puede evitar costes de supervisión innecesarios y problemas de polizón entre acreedores dispersos (ALTUNBAŞ, KARA y MARQUÉS-IBÁÑEZ, 2010).

\subsubsection{Fondos de deuda}
Los bancos no son los únicos intermediarios que prestan a las empresas. Los fondos de deuda o de préstamo directo (\textit{direct lending}) ofrecen financiación más flexible que la bancaria: pueden devolver todo el principal al vencimiento (\textit{bullet}), a plazos más largos, por importes a veces mayores y a tipos distintos. Suelen estar disponibles solo para empresas establecidas, con activos e historial que respalden la decisión del fondo.

\subsubsection{Financiación con garantía de activos}
Una de las formas más seguras de garantizar el reembolso es usar un activo de la empresa como garantía (\textit{asset-backed lending}). Restringe a la empresa, que no puede vender ese activo, pero abarata su financiación, y ofrece al prestamista un perfil de riesgo y rentabilidad más moderado que la deuda ordinaria. Las empresas cuyo negocio descansa en intangibles muy específicos, como muchas empresas emergentes, apenas pueden usarla (OCDE, 2015). Son de esta naturaleza el arrendamiento financiero, el \textit{renting} y el \textit{factoring}:
\begin{la}
    \item El \textbf{arrendamiento financiero} (\textit{leasing}) permite a la empresa usar un activo no corriente a cambio de un canon, durante un plazo al término del cual puede comprarlo, devolverlo o renovar el contrato. Intervienen la empresa, la entidad arrendadora y el fabricante del bien; en el arrendamiento operativo, el fabricante y el arrendador coinciden. Evita desembolsar de una vez grandes inversiones, su canon es fiscalmente deducible y, en el operativo, la empresa se libra del riesgo de obsolescencia, porque puede rescindir el contrato. A cambio, puede resultar caro a largo plazo, porque la empresa pierde el valor residual del bien, y no todos los activos pueden arrendarse;
    \item El \textbf{factoring} es la cesión a un tercero de los derechos de cobro sobre clientes a cambio de liquidez inmediata, con un coste elevado. Su reverso es el \textbf{confirming}, un servicio de pago a proveedores: la empresa encarga a una entidad financiera el pago de sus facturas, y el proveedor puede cobrarlas por anticipado a cambio de una comisión y de los intereses correspondientes.
\end{la}

\subsubsection{Titulización}
Las empresas también pueden refinanciar parte de sus activos transformando créditos u otros derechos en valores negociables. La utilizan con frecuencia los grupos automovilísticos, como el grupo Mercedes: sus filiales financieras venden a un vehículo especial los préstamos y arrendamientos concedidos a sus clientes, y el vehículo los financia emitiendo bonos de titulización (\textit{asset-backed securities}), a menudo en tramos con distinta prelación (sénior, subordinados, \textit{mezzanine}) para ofrecer distintos perfiles de riesgo y rentabilidad. Aislados del resto del balance, esos activos pueden ser de mayor calidad que la empresa en su conjunto y financiarse a tipos más bajos; pero la operación es más costosa que la deuda directa para una empresa de buena calidad crediticia, y difícil de montar por menos de $50$ millones de euros salvo que varias empresas compartan la plataforma. No debe confundirse con la financiación con garantía de activos: esta respalda préstamos individuales con un activo, mientras que la titulización agrupa muchos préstamos parecidos y los vende, de modo que el inversor solo cuenta con ese conjunto de activos.

\subsubsection{Project finance}
\textcolor{red}{\textbf{OJO! China} - Project finance: una opción para grandes proyectos.} El \textit{project finance} financia un proyecto según su propia capacidad de generar fondos y no según la del promotor, y usa como garantía los activos del propio proyecto. Se ha utilizado, por ejemplo, para prospecciones petrolíferas, refinerías o plantas de energías renovables que no podían ofrecer las garantías de un préstamo ordinario: parte del petróleo, incluso antes de extraerse, servía de garantía, y parte de las ventas futuras devolvía los préstamos. Como está pensado para grandes proyectos de energía o infraestructuras, lo usan a menudo los bancos regionales de desarrollo. El banco asume más riesgo que en un préstamo convencional (si el proyecto no devuelve el préstamo, puede acabar siendo propietario de una central eléctrica o de un yacimiento), por lo que es difícil aplicarlo a proyectos tecnológicos de rendimiento incierto. Permite plazos de hasta $30$ años, un apalancamiento muy elevado y sacar la deuda del balance del promotor, sin afectar a sus ratios de endeudamiento.

\subsubsection{Otras fuentes}
Completan el panorama el crédito comercial de los proveedores, el descuento comercial, los pagarés de empresa, las líneas de crédito, la financiación participativa (\textit{crowdfunding}), en la que un proyecto se publica en una plataforma, por lo general en internet, para que muchos pequeños aportantes lo financien a cambio de un rendimiento, y los mercados internacionales de eurodivisas y eurobonos.

\subsection{Instrumentos híbridos}
\subsubsection{Acciones preferentes}
Comparten rasgos con la deuda y con las acciones. Como la deuda, pagan un dividendo fijo, que, si la empresa no tiene efectivo para pagarlo, suele acumularse hasta que pueda, y sus titulares no participan en la gestión: su voto se limita a lo que afecte a sus derechos de cobro. Como las acciones, su remuneración no es deducible y no tienen vencimiento. En la liquidación, cobran después de los obligacionistas y antes que los accionistas ordinarios.

\subsubsection{Deuda subordinada y deuda perpetua}
La deuda subordinada solo se cobra después de la deuda con mayor prelación. La deuda perpetua no tiene vencimiento ni obligación de devolución: el emisor la amortiza cuando lo considera oportuno, y el inversor solo recupera su aportación si el emisor decide amortizarla, si la sociedad se liquida o si vende el título en el mercado secundario.

\subsubsection{Bonos convertibles y warrants}
Un bono convertible puede canjearse por un número predeterminado de acciones a elección de su titular. Al emitirse, la conversión no suele interesar, pero se vuelve atractiva a medida que sube la cotización; las empresas añaden la opción de conversión para pagar un interés menor. Los \textit{warrants} dan derecho a suscribir, durante un período, otro título de nueva emisión (una acción, un bono o incluso otro \textit{warrant}) en una proporción y a un precio fijados de antemano.

\subsubsection{Bonos convertibles contingentes}
Los CoCos son bonos que se convierten en acciones, o pierden valor, cuando se cumple una condición fijada de antemano, por lo general que la ratio de capital ordinario de nivel 1 del banco emisor caiga por debajo de un umbral. Mientras están vivos pagan cupones elevados, y el inversor asume el riesgo de la conversión. Los bancos los han emitido en abundancia para reforzar sus balances ante las exigencias regulatorias: la conversión reduce su deuda y mejora su capital cuando más dudas hay sobre su viabilidad ([\authorfont{Ver Tema 3.B.27}]).

\subsubsection{Préstamos participativos}
El prestamista recibe un interés variable, ligado a la actividad de la empresa, que puede combinarse con uno fijo. En España, se cobran después de los acreedores comunes, solo pueden amortizarse anticipadamente si la empresa amplía en la misma cuantía sus fondos propios y se consideran patrimonio neto a efectos de la reducción de capital y la liquidación de sociedades (art. $20$ del Real Decreto-ley 7/1996).


\anexo{Desarrollos analíticos}\label{anx:desarrollos}
Este anexo recoge las demostraciones y los desarrollos formales de los modelos del cuerpo. Van fuera porque el cuerpo expone la intuición y el resultado de cada modelo; aquí se reproduce el razonamiento completo, para seguirlo paso a paso.

\subsection{MODIGLIANI y MILLER (1958): la demostración por arbitraje}
\subsubsection{Proposición I}
Sean dos empresas con los mismos activos, que generan cada año el mismo flujo de explotación incierto, de valor esperado $\overline{X}$. La empresa 1 no tiene deuda, de modo que su valor es el de sus acciones, $V_1=E_1$; la empresa 2 tiene una deuda sin riesgo $D_2$ al tipo $r$, y vale $V_2=E_2+D_2$. Los particulares pueden prestar y endeudarse al mismo tipo $r$.

Supongamos que $V_2>V_1$. Un inversor que posee una fracción $\alpha$ de las acciones de la empresa 2 ha invertido $\alpha E_2$ y cobra
$$y_2=\alpha\left(X-rD_2\right).$$
Si vende esas acciones, pide prestado $\alpha D_2$ por su cuenta y compra acciones de la empresa 1 por $\alpha(E_2+D_2)=\alpha V_2$, adquiere la fracción $\alpha V_2/V_1$ de la empresa 1 y cobra, después de pagar los intereses de su préstamo,
$$y_1=\alpha\,\frac{V_2}{V_1}\,X-\alpha rD_2.$$
Como $V_2/V_1>1$, $y_1>y_2$ para cualquier $X>0$: con la misma inversión y el mismo riesgo financiero, obtiene más. Todos los inversores harán lo mismo, venderán acciones de la empresa 2 y comprarán de la 1, hasta que los valores se igualen.

Supongamos ahora que $V_2<V_1$. Quien posee la fracción $\alpha$ de la empresa 1 ha invertido $\alpha V_1$ y cobra $\alpha X$. Si vende y compra la misma fracción $\beta$ de las acciones y de la deuda de la empresa 2, con $\beta(E_2+D_2)=\alpha V_1$, es decir, $\beta=\alpha V_1/V_2>\alpha$, cobra
$$\beta\left(X-rD_2\right)+\beta rD_2=\beta X>\alpha X.$$
De nuevo, el arbitraje elimina la diferencia. Solo $V_1=V_2$ es compatible con el equilibrio: el valor de mercado de la empresa no depende de su estructura del capital (MODIGLIANI y MILLER, 1958, proposición I).

\subsubsection{Proposición II}
La rentabilidad esperada de las acciones de la empresa endeudada es el flujo esperado que les queda tras pagar a los acreedores, sobre su valor:
$$r_E=\frac{\overline{X}-r_DD}{E}.$$
Por la proposición I, el flujo esperado se capitaliza a la misma tasa que si la empresa no tuviera deuda, $\overline{X}=r_U\,V=r_U\,(E+D)$. Sustituyendo:
$$r_E=\frac{r_U(E+D)-r_DD}{E}=r_U+\left(r_U-r_D\right)\frac{D}{E},$$
y el coste medio ponderado del capital, $r_{\text{WACC}}=r_E\,E/V+r_D\,D/V=r_U$, no depende del endeudamiento (MODIGLIANI y MILLER, 1958, proposición II).

\subsection{MODIGLIANI y MILLER (1963): el impuesto sobre sociedades}
Con un impuesto sobre sociedades al tipo $T_c$, la empresa sin deuda reparte $(1-T_c)X$, y la endeudada reparte entre accionistas y acreedores
$$\left(1-T_c\right)\left(X-r_DD\right)+r_DD=\left(1-T_c\right)X+T_c\,r_DD.$$
El primer sumando es el flujo de la empresa sin deuda, que vale $V_U$; el segundo, el ahorro fiscal, es tan seguro como los intereses y, si la deuda es perpetua, vale $T_cr_DD/r_D=T_cD$. Por tanto, $V_L=V_U+T_cD$ (MODIGLIANI y MILLER, 1963).

Para la rentabilidad de las acciones, el flujo esperado de los accionistas es $(1-T_c)(\overline{X}-r_DD)$, y $(1-T_c)\overline{X}=r_U\,V_U=r_U\,(E+D-T_cD)$. Entonces
$$r_E\,E=r_U\left[E+(1-T_c)D\right]-(1-T_c)\,r_DD\quad\Longrightarrow\quad r_E=r_U+\left(r_U-r_D\right)\left(1-T_c\right)\frac{D}{E},$$
y el coste medio ponderado del capital, con los intereses netos del impuesto, es
$$r_{\text{WACC}}=r_E\,\frac{E}{V}+r_D\left(1-T_c\right)\frac{D}{V}=r_U\left(1-T_c\,\frac{D}{V}\right),$$
que decrece con el endeudamiento.

\subsection{MILLER (1977): los impuestos personales y el equilibrio}
\subsubsection{La ganancia del endeudamiento}
Sean $T_d$ el tipo personal sobre los intereses y $T_e$ el tipo personal sobre las rentas de las acciones. Después de todos los impuestos, los accionistas de la empresa endeudada reciben $(1-T_c)(1-T_e)(X-r_DD)$ y sus acreedores $(1-T_d)\,r_DD$. En total:
$$\left(1-T_c\right)\left(1-T_e\right)X+r_DD\left[\left(1-T_d\right)-\left(1-T_c\right)\left(1-T_e\right)\right].$$
El primer sumando es lo que reciben los accionistas de la empresa sin deuda, que vale $V_U$. El segundo es una renta perpetua y segura después de impuestos personales, que los acreedores descuentan a su rentabilidad neta, $r_D(1-T_d)$. Su valor es la ganancia del endeudamiento:
$$G_L=\left[1-\frac{(1-T_c)(1-T_e)}{1-T_d}\right]D.$$
Caben tres casos. Si $T_e=T_d$, los impuestos personales no alteran el resultado de 1963 y $G_L=T_cD$. Si $1-T_d=(1-T_c)(1-T_e)$, la mayor tributación de los intereses anula la ventaja del impuesto sobre sociedades y la estructura del capital vuelve a ser irrelevante. Para otros valores, los impuestos personales amplifican la ventaja de la deuda (si $T_e>T_d$) o la reducen (si $T_d>T_e$).

\subsubsection{El inversor marginal y el equilibrio de MILLER}
Los impuestos personales son progresivos, de modo que inversores con rentas distintas soportan tipos $T_d$ distintos. A un inversor le interesa comprar deuda en lugar de acciones si la rentabilidad neta de la deuda, $r_D(1-T_d)$, supera la de las acciones. Los inversores con tipos bajos son los primeros dispuestos a comprar bonos; para colocar más deuda, las empresas tienen que atraer a inversores con tipos cada vez más altos, que exigen intereses mayores para quedar indiferentes. El \textbf{inversor marginal} es aquel para el que el endeudamiento de la empresa no reporta ninguna ganancia, es decir, para el que $1-T_d=(1-T_c)(1-T_e)$; con $T_e=0$, el caso extremo que planteó MILLER, aquel cuyo tipo sobre los intereses iguala al del impuesto sobre sociedades.

En el \textbf{equilibrio de MILLER}, el sector empresarial en su conjunto emite deuda hasta ese punto. Si la deuda agregada superara ese nivel, para colocarla haría falta un tipo de interés mayor, la deuda dejaría de compensar y las empresas la sustituirían por acciones; si fuera menor, ocurriría lo contrario. El endeudamiento agregado queda así determinado, pero cada empresa, individualmente, es indiferente: las ventajas fiscales de endeudarse quedan anuladas por el mayor interés que exigen los nuevos acreedores. MILLER demuestra así, con impuestos, la irrelevancia de la estructura del capital de cada empresa.

DEANGELO y MASULIS (1980) extendieron el modelo con otros escudos fiscales distintos de la deuda (amortizaciones, deducciones por inversión), y el resultado es un nivel óptimo de deuda distinto de cero para cada empresa. Aunque ambos marcos tienen fundamentos teóricos sólidos, la evidencia es mixta, porque es difícil observar los tipos marginales; GRAHAM (2000) encuentra que el valor del endeudamiento, considerando todos los impuestos, es positivo.

\subsection{ROSS (1977): la deuda como señal}
Hay dos tipos de empresa, $A$ y $B$, cuyos activos valdrán en el período siguiente $V_A$ y $V_B$, con $V_A>V_B$. El directivo conoce el tipo de su empresa; los inversores, neutrales ante el riesgo y que descuentan al tipo $r$, solo observan el nominal de la deuda que emite, $D$, y valoran la empresa hoy, $V_0$, según lo que esa deuda les dice. La remuneración del directivo pondera el valor de mercado de hoy y el valor de mañana, con una penalización $L$ si la empresa quiebra (modelo de ROSS, 1977, en la versión simplificada con valores ciertos):
$$M=(1+r)\,\gamma_0V_0+\gamma_1\begin{cases}V_1 & \text{si } D\leq V_1,\\ V_1-L & \text{si } D>V_1.\end{cases}$$

Supongamos que los inversores creen que una empresa con $D>D^{*}$ es de tipo $A$ y una con $D\leq D^{*}$ es de tipo $B$, con $V_B\leq D^{*}<V_A$. El directivo de la empresa $A$ emite una deuda entre $D^{*}$ y $V_A$: no quiebra y cobra $\gamma_0V_A+\gamma_1V_A$, más de lo que cobraría haciéndose pasar por $B$. El directivo de la empresa $B$ cobra $\gamma_0V_B+\gamma_1V_B$ si dice la verdad y $\gamma_0V_A+\gamma_1(V_B-L)$ si imita a $A$, porque entonces quiebra. No imita si
$$\gamma_1L>\gamma_0\left(V_A-V_B\right),$$
es decir, si la penalización por quiebra, ponderada por el peso del futuro en su remuneración, supera lo que ganaría con la sobrevaloración de hoy. Se cumplen entonces las creencias de los inversores: el endeudamiento separa a los dos tipos y el valor de la empresa crece con su deuda. La señal es más eficaz cuanto mayor es la penalización, cuanto más pesa el futuro ($\gamma_1$ frente a $\gamma_0$) y cuanto menor es la diferencia entre los tipos, que es lo que el directivo de la empresa mala ganaría fingiendo. En el modelo original, el valor futuro es incierto y el nivel crítico de deuda crece con la calidad de la empresa.\textcolor{orange}{\textsuperscript{[Probable]}}

\subsection{MYERS y MAJLUF (1984): la decisión de emitir}
La empresa tiene activos en funcionamiento que valen $a_A$ con probabilidad $p$ (empresa buena) y $a_B<a_A$ con probabilidad $1-p$ (empresa mala); su valor medio es $\overline{a}=p\,a_A+(1-p)\,a_B$. Tiene una oportunidad de inversión que exige $I$ y tiene un valor actual neto $b$, y no tiene fondos. Los directivos conocen el tipo y actúan en interés de los accionistas actuales.

Si los dos tipos emiten acciones, el mercado valora la empresa, tras la emisión, en $\overline{a}+I+b$, y los nuevos accionistas reciben la fracción $s$ que cubre su aportación: $s\,(\overline{a}+I+b)=I$. Los antiguos accionistas de la empresa buena se quedan con $(1-s)(a_A+I+b)$, frente a $a_A$ si no emiten. La diferencia es
$$(1-s)(a_A+I+b)-a_A=b-T,\qquad T=\frac{I\,(a_A-\overline{a})}{\overline{a}+I+b}>0,$$
donde $T$ es lo que la empresa buena transfiere a los nuevos accionistas por vender sus acciones al precio medio (MYERS y MAJLUF, 1984). La empresa buena solo emite si $b\geq T$. Si $b<T$, renuncia al proyecto aunque sea rentable; solo emite la mala, el mercado lo sabe y la emisión revela el tipo: la cotización cae al anunciarse.

Si en lugar de acciones la empresa emite deuda sin riesgo, los acreedores reciben exactamente lo que aportan, sea cual sea el tipo, no hay transferencia y la empresa buena invierte siempre. Con deuda arriesgada hay transferencia, pero menor que con acciones, porque el valor de la deuda depende menos del tipo de empresa. De ahí la jerarquía: fondos internos, deuda sin riesgo, deuda con riesgo y, por último, acciones.

\subsection{MILLER y MODIGLIANI (1961): la irrelevancia de los dividendos}
Una empresa financiada solo con acciones tiene $n$ acciones. Con mercados perfectos, el precio de hoy es el valor actual del dividendo del período, $D_1$, y del precio al final de este, $P_1$, descontados a la rentabilidad que exigen los accionistas, $r_E$:
$$P_0=\frac{D_1+P_1}{1+r_E}\qquad\Longrightarrow\qquad nP_0=\frac{nD_1+nP_1}{1+r_E}.$$
Si la empresa emite $m$ acciones nuevas al final del período, a precio $P_1$, puede escribirse, sumando y restando $mP_1$,
$$nP_0=\frac{nD_1+(n+m)P_1-mP_1}{1+r_E}.$$
Con la inversión $I$ y el flujo de la empresa $X$ dados, la emisión cubre lo que la inversión y el dividendo exceden de los fondos generados:
$$mP_1=I-\left(X-nD_1\right).$$
Sustituyendo, el dividendo desaparece:
$$nP_0=\frac{(n+m)P_1-I+X}{1+r_E}.$$
El valor de la empresa depende de sus flujos, de su inversión y de su valor futuro, que tampoco depende de la política de dividendos futura por el mismo razonamiento, pero no del dividendo de hoy: todo aumento del dividendo se paga con una emisión equivalente, cuyo valor se resta del de los antiguos accionistas (MILLER y MODIGLIANI, 1961). Los supuestos son los de 1958 más el de comportamiento racional.

\subsection{WALTER (1956) y GORDON (1959): retención y valor}
Si la empresa retiene la proporción $b$ de su beneficio y la reinvierte a la rentabilidad $y$, su beneficio crece a la tasa $g=b\,y$: $B_t=B_0(1+b\,y)^t\approx B_0\,e^{b\,y\,t}$. El dividendo es $(1-b)B_t$ y, descontado de forma continua a $r_E$, con $r_E>b\,y$ para que el valor esté acotado,
$$V=\int_0^{\infty}(1-b)\,B_0\,e^{-(r_E-b\,y)\,t}\,dt=\frac{(1-b)\,B_0}{r_E-b\,y}.$$
El efecto de la retención sobre el valor es
$$\frac{\partial V}{\partial b}=\frac{B_0\,(y-r_E)}{\left(r_E-b\,y\right)^{2}},$$
positivo si $y>r_E$ (conviene no repartir), nulo si $y=r_E$ (el caso de MILLER y MODIGLIANI) y negativo si $y<r_E$ (conviene repartir todo). Sus límites son que supone constante la rentabilidad de la reinversión y que no admite financiación externa. GORDON (1962) y LINTNER (1962) añadieron que $r_E$ crece con $b$, porque los beneficios retenidos se perciben como más arriesgados que los dividendos.

\subsection{LINTNER (1956): el ajuste parcial del dividendo}
Cada empresa tiene un dividendo objetivo, proporcional a su beneficio, $D_t^{*}=p\,E_t$, donde $p$ es la proporción de reparto a largo plazo. El dividendo se ajusta cada año solo en una fracción $c$, con $0<c<1$, de la diferencia entre el objetivo y el dividendo anterior:
$$D_t-D_{t-1}=a+c\left(p\,E_t-D_{t-1}\right),$$
donde $a$ es una constante, a menudo positiva, que refleja la resistencia a recortar el dividendo. Cuanto menor es $c$, más lento y suave es el ajuste: un aumento transitorio del beneficio apenas se traslada al dividendo, y uno permanente se traslada poco a poco. FAMA y BABIAK (1968) confirmaron el modelo con datos de cientos de empresas estadounidenses.
