    % --- Imagen adaptativa ---
    % \imagenfit[alto]{fichero}{título}{fuente}: la imagen queda fija entre los párrafos donde se escribe (no flota), con su
    % título en el índice de figuras y la fuente debajo. Sin [alto]: las figuras PDF (TikZ) salen al tamaño con que se
    % dibujaron y solo se reducen si no caben en el ancho o en el alto de la página; las PNG/JPG ocupan el ancho del texto
    % hasta \imgaltoras del alto de la página. Con [alto] (fracción del alto de la página; [1] = página entera) se fija el
    % tamaño de esa imagen. Si no cabe en lo que queda de página, pasa entera a la siguiente.
    % \imagenfithorizontal{fichero}{título}{fuente}: la imagen en una página girada (apaisada), en ese mismo punto del texto.
    % Reglas: main/GUIA_TEMAS.md, §4 (Figuras).
    \newdimen\imgcap
    \newdimen\imgh
    \newdimen\imgw
    \newdimen\imgneed
    \newsavebox\imgcaja
    \newcommand{\imgaltoras}{0.30}% alto por defecto de las PNG/JPG (fracción del alto de la página)
    \makeatletter
    \newcommand{\imagenfit}[4][]{%
      \addvspace{10pt}%
      \par\begingroup
        \imgcap=3.5\baselineskip% alto aproximado de título + fuente
        \imgw=\linewidth
        \filename@parse{#2}\def\img@pdf{pdf}%
        \if\relax\detokenize{#1}\relax
          \ifx\filename@ext\img@pdf
            \sbox\imgcaja{\includegraphics{\detokenize{#2}}}% tamaño natural
            \ifdim\wd\imgcaja<\imgw \imgw=\wd\imgcaja \fi
            \imgh=\ht\imgcaja
          \else
            \imgh=\imgaltoras\textheight
          \fi
        \else
          \imgh=#1\textheight
        \fi
        \dimen@=\textheight \advance\dimen@ by -\imgcap
        \ifdim\imgh>\dimen@ \imgh=\dimen@ \fi
        \sbox\imgcaja{\includegraphics[width=\imgw,height=\imgh,keepaspectratio]{\detokenize{#2}}}%
        \imgneed=\ht\imgcaja \advance\imgneed by \imgcap
        \Needspace{\imgneed}%
        \noindent\begin{minipage}{\textwidth}
          \centering
          \captionof{figure}{\textemdash\ #3}\par\vspace{0.3em}% título arriba
          \usebox\imgcaja\par
          \vspace{0.3em}\small\textit{Fuente: }#4% fuente abajo
        \end{minipage}%
      \endgroup
      \par\addvspace{10pt}\RequestRestore
    }
    \newcommand{\imagenfithorizontal}[3]{%
      \begin{landscape}
        \centering
        \captionof{figure}{\textemdash\ #2}\par\vspace{0.3em}%
        \includegraphics[width=\linewidth,height=\dimexpr\textheight-4\baselineskip\relax,keepaspectratio]{\detokenize{#1}}\par
        \vspace{0.3em}{\small\textit{Fuente: }#3}%
      \end{landscape}%
    }
    \makeatother

    % \anexo{Título}: un anexo equivale a un \section (su título sale del mismo tamaño). Dentro se usan \subsection,
    % \subsubsection y \paragraph, nunca \section: se numeran I.1, I.1.1, I.1.1.1 y las páginas del anexo, I-1, I-2…
    % En el índice de contenido inicial solo figura el nombre del anexo. Sus apartados forman el índice del anexo, que sale
    % solo si el anexo los tiene, justo después de la introducción del anexo (antes del primer \subsection).
    % \anexo*{Título}: el mismo anexo, sin ese índice. Reglas: main/GUIA_TEMAS.md, §6.

    \newcommand{\anx@empezar}[2]{%
      \clearpage
      \refstepcounter{anexo}%
      \global\EnAnexotrue
      \setcounter{page}{1}\gdef\thepage{\theanexo-\arabic{page}}%
      \setcounter{section}{\numexpr\value{anexo}-1\relax}\stepcounter{section}% el anexo hace de \section
      \gdef\thesection{\Roman{section}}\gdef\theHsection{anexo.\Roman{section}}%
      \phantomsection
      \protected@xdef\leq@current@section{Anexo \theanexo.- \unexpanded{#2}}\gdef\leq@current@subsection{}% índice de ecuaciones
      \addtocontents{stc}{\protect\contentsline{section}{Anexo \theanexo.- #2}{}{\@currentHref}}% índice inicial: solo el nombre, sin página
      \addcontentsline{toc}{section}{Anexo \theanexo.- #2}% marcadores del PDF
      {\noindent\normalfont\Large\bfseries Anexo \theanexo.- #2\par}\nobreak\vspace{1.5ex \@plus .2ex}%
      #1\@ifundefined{anx@lista@\theanexo}{\global\anx@pendientefalse}{}%
    }
    % Índice del anexo: cada apartado escribe una línea en el fichero .atc (anexo, nivel, texto, página, enlace); al
    % empezar el documento se leen las de la compilación anterior y se guardan por anexo (dos pasadas, como cualquier índice).
    % En ese índice los apartados suben un nivel, porque el anexo hace de \section.
    \def\anx@entrada#1#2#3#4#5{\@ifundefined{anx@lista@#1}{\global\@namedef{anx@lista@#1}{}}{}%
      \expandafter\g@addto@macro\csname anx@lista@#1\endcsname{\contentsline{#2}{#3}{#4}{#5}}}
    \tcbuselibrary{breakable}
    \def\anx@indice{\ifanx@pendiente\global\anx@pendientefalse

            before upper={\parindent=0pt\parskip=2pt}]
          \small
          % números más anchos que en el índice inicial (I.1, I.1.1, I.1.1.1; páginas I-1)
          \def\@pnumwidth{2.6em}\def\@tocrmarg{3.4em}%
          \def\l@section##1##2{\addpenalty\@secpenalty\addvspace{1.0em \@plus\p@}\setlength\@tempdima{2.6em}%
            \begingroup\parindent\z@\rightskip\@pnumwidth\parfillskip-\@pnumwidth\leavevmode\bfseries
            \advance\leftskip\@tempdima\hskip-\leftskip##1\nobreak\hfil\nobreak\hb@xt@\@pnumwidth{\hss##2\kern-\p@\kern\p@}\par\endgroup}%

          \csname anx@lista@\theanexo\endcsname

      \fi}
    \def\anx@escribir#1#2#3{\ifEnAnexo

    % margen superior, así que no cambia las páginas del tema. Se enciende con \pagenumbering{arabic} y se apaga con

    \cs_new_protected:Npn \cab_guardar:n #1
      {
        \tl_gset:Nn \g_cab_titulo_tl {#1}
        \regex_replace_once:nnN { \A \s* Tema \s+ \S+ \s* \-\-? \s* } { } \g_cab_titulo_tl % sin «Tema X.Y.N --»
        \regex_replace_once:nnN { \.? \s* \c{\\} (\[ [^\]]* \])? \s* (\c{large} \s*)? } { .\  } \g_cab_titulo_tl % título. subtítulo
        \regex_replace_once:nnN { [\.\s]* \Z } { } \g_cab_titulo_tl % sin subtítulo: nada detrás
      }

\section{Primera Generación: Inconsistencia del \textit{policy-mix}}
\puntosclave{

}


Los modelos de primera generación tratan de explicar el colapso de los sistemas de tipo de cambio fijo que tuvo lugar en los años 70 y 80 del siglo XX en varios países en desarrollo. Sus bases tienen su origen en SALANT y HENDERSON (1978), y posteriormente fueron desarrollados por KRUGMAN (1979) y FLOOD y GARBER (1984). 

Para estos modelos de primera generación, las causas de las crisis de balanza de pagos radican en un deterioro de los fundamentos económicos debido a la incompatibilidad entre los objetivos de política interna y externa: si las autoridades optan por instrumentar una política monetaria para alcanzar fines internos que son prioritarios sobre el objetivo externo, bajo un sistema de tipo de cambio fijo, la crisis cambiaria será inevitable. 

Este modelo puede entenderse tanto como una demostración de un policy-mix inconsistente o como una aplicación de la regla de TINBERGEN, según la cual es imposible que un solo instrumento (la política monetaria) logre dos objetivos de política económica (la defensa del ancla cambiaria y la financiación del déficit público). 

Como podemos intuir, bajo el marco analítico de estos modelos, las políticas internas prioritarias conducen a incurrir en déficit público que se financia vía monetización. Esta política monetaria expansiva genera tensiones depreciatorias que hacen necesario que las autoridades compren moneda nacional a cambio de reservas internacionales para mantener la paridad cambiaria. Los agentes privados anticipan que en último término las reservas internacionales, que son finitas, se agotarán y el abandono de la paridad cambiaria será inevitable. En este contexto, el ataque especulativo será la respuesta racional ante una evolución insostenible. 

Los agentes, al anticipar correctamente que el tipo de cambio finalmente tendrá que ser alterado, precipitan la crisis inexorable con su actuación.

\subsection{FLOOD y GARBER (1984): ¿Qué pasa cuando la Autoridad Monetaria persigue dos objetivos?}
La formulación original es de KRUGMAN (1979), que traslada al tipo de cambio el análisis de SALANT y HENDERSON (1978) sobre los ataques especulativos contra un programa público de fijación del precio del oro: cuando un Gobierno defiende un precio vendiendo un stock finito de un activo, llega un momento en que los especuladores compran de golpe todo el stock restante. El modelo de KRUGMAN (1979), con una demanda de dinero de forma general, caracteriza la crisis pero no ofrece una expresión explícita de su fecha\textcolor{orange}{\textsuperscript{[Probable]}}; FLOOD y GARBER (1984) lo reformulan en una versión lineal que sí la ofrece, y por eso es la presentación habitual de la primera generación.

FLOOD y GARBER (1984) desarrollan un modelo para una economía pequeña y abierta
sin imperfecciones.

Sobre esta base, los agentes económicos son capaces de hacer previsiones perfectas y pueden invertir en tres activos financieros: (i) el dinero nacional; (ii) los bonos nacionales; y, (iii) los bonos extranjeros. Los agentes no desean mantener dinero extranjero en su cartera porque es un activo dominado fuertemente por los otros tres, ya que no proporciona rendimiento nominal ni servicios monetarios a los agentes. Sin embargo, los bonos nacionales y extranjeros sí se consideran sustitutivos perfectos. 

El déficit público es exógeno y el banco central posee un stock de divisa extranjera para intervenir con el fin de mantener el tipo de cambio fijo, pero también se le exige monetizar parte del déficit público. 

A partir de estos supuestos, el modelo se construye sobre cuatro ecuaciones principales, dos estáticas y dos dinámicas (modelo de FLOOD y GARBER (1984), versión lineal en tiempo continuo con previsión perfecta; el desarrollo analítico completo se recoge en el Anexo~\ref{modelos_primera}):

\eqblock{
\begin{aligned}
    &\text{Estáticas}:
    &&\left\{\begin{aligned}
        &(1)\quad\text{Mercado Monetario}: && \frac{M(t)}{P(t)}=Y(t)-k\cdot i(t)\,\,>0\\[1em]
        &(2)\quad\text{Condición PPA}: &&P(t)=P^*(t)\cdot s(t)
    \end{aligned}\right.\quad\text{\scriptsize$\text{donde,}
    \quad
    \left\{\begin{aligned}
        &\underset{\text{\scriptsize reservas internacionales $+$ crédito interno}}{M(t)=R(t)+D(t)}\\[0.5em]
        &s(t)\equiv \text{Tipo de Cambio spot}
    \end{aligned}\right.$}\\[1em]
    &\text{Dinámicas}:
    &&\left\{\begin{aligned}
        &(3)\quad\underset{\text{\scriptsize señoreaje: crecimiento exógeno}}{\text{Crecimiento déficit}:}&& \dot D(t)=\mu\\[1em]
        &(4)\quad\text{Condición de PNCI}: &&i(t)=i^*(t)+\left[\frac{\dot s(t)}{s(t)}\right]
    \end{aligned}\right.
\end{aligned}
}{Ecuaciones funcionales: Modelo FLOOD y GARBER (1984). Primera generación: Inconsistencia del policy-mix}

La ecuación (1) es el equilibrio del mercado monetario: la oferta real de dinero se iguala a una demanda real que depende positivamente de la renta, que se toma como dada, y negativamente del tipo de interés nominal, con una sensibilidad $k>0$. La oferta nominal de dinero es la suma de los dos activos del banco central: las \textbf{reservas internacionales} ($R$) y el \textbf{crédito interno} ($D$), que es la vía por la que se monetiza el déficit. La ecuación (2) es la paridad de poder adquisitivo: el tipo de cambio $s$ se mide en unidades de moneda nacional por unidad de divisa (un aumento es una depreciación) y el arbitraje de bienes hace que una unidad monetaria tenga la misma capacidad adquisitiva con independencia de su moneda de denominación. Las ecuaciones dinámicas inciden en ambos lados de la oferta monetaria: (3) recoge el crecimiento del crédito interno a una tasa constante $\mu$, inducido por la necesidad de obtener ingresos por señoreaje para financiar el déficit; y (4) determina el tipo de interés nacional mediante la paridad no cubierta de intereses: con bonos nacionales y extranjeros sustitutivos perfectos y previsión perfecta, el mercado cambiario solo está en equilibrio cuando los depósitos en todas las divisas ofrecen idéntica tasa de rentabilidad, de modo que el tipo de interés interno es el externo más la depreciación prevista.

Si sustituimos la condición PPA (2) y la PNCI (4) en el equilibrio del mercado monetario (1), el modelo se reduce a una sola relación entre la oferta monetaria, el nivel del tipo de cambio y su variación (forma reducida del modelo de FLOOD y GARBER (1984)). Con el tipo de cambio fijo, esa relación determina la evolución de las reservas:

\eqblock{
\begin{aligned}
    &\boxed{M(t)=\beta\,s(t)-\alpha\,\dot s(t)}\qquad\text{donde,}\quad\beta\equiv P^*\,(Y-k\,i^*)>0,\quad \alpha\equiv k\,P^*>0\\[1em]
    &\underset{\substack{\text{\scriptsize TC fijo:}\\ s(t)=\bar s,\ \dot s(t)=0}}{\Longrightarrow}\quad M(t)=\beta\,\bar s\quad\Longrightarrow\quad R(t)=\beta\,\bar s-D(t)\quad\Longrightarrow\quad\boxed{\dot R(t)=-\dot D(t)=-\mu}
\end{aligned}
}{Forma reducida y pérdida de reservas: Modelo FLOOD y GARBER (1984). Primera generación: Inconsistencia del policy-mix}

Los dos parámetros tienen una lectura económica directa. $\beta$ es la demanda de saldos nominales por unidad de tipo de cambio cuando no se espera ninguna depreciación: un tipo de cambio más alto eleva, por la PPA, el nivel de precios interno y, con él, la cantidad de dinero que el público desea mantener. Es positivo porque la demanda real de dinero lo es (la restricción ``$>0$'' de la ecuación (1)). $\alpha$ mide cuánto se reduce la demanda de dinero cuando se espera una depreciación, ya que la PNCI la traslada íntegramente al tipo de interés interno: es la sensibilidad $k$ de la demanda de dinero, expresada en términos nominales. Como $P^*$ e $i^*$ pueden suponerse dados para un país pequeño y la renta es constante, $\beta$ y $\alpha$ son constantes.

Mientras el tipo de cambio esté fijo en $\bar s$ no se espera depreciación alguna ($\dot s(t)=0$) y la demanda de dinero queda anclada en $\beta\bar s$. El banco central, por tanto, no controla la cantidad total de dinero, sino solo su composición: cada unidad de crédito interno que crea para financiar el déficit es un exceso de oferta que el público le devuelve a cambio de divisas, de modo que las reservas caen en la misma cuantía.

Para mantener el tipo de cambio constante a la vez que se monetiza el déficit público, es necesario que los activos externos en el balance del Banco Central se reduzcan a la misma tasa a la que aumentan los activos públicos, de forma que la oferta monetaria no varíe. Con una tasa de crecimiento ($\mu$) exógena y constante y unas reservas internacionales finitas, el sistema de tipo de cambio fijo no podrá sobrevivir indefinidamente: cualquier stock de reservas internacionales mantenido para garantizar el sistema cambiario fijo acabará agotándose.\footnote{%
    Suponemos que el límite inferior de reservas internacionales es cero, pero los resultados serían similares si se supusiese que ese límite inferior viene dado por cualquier otro valor positivo o negativo.
}

\subsubsection{Crisis cambiaria: ¿cuándo y cómo se origina la crisis?}
Los agentes privados, que son capaces de prever de forma perfecta que las políticas del gobierno conducirán inexorablemente al agotamiento de las reservas internacionales, acometerán de forma racional un ataque especulativo mediante la compra masiva de bonos extranjeros a cambio de moneda nacional antes de que las reservas internacionales se agoten para evitar pérdidas por sus tenencias en moneda nacional, que precipita la crisis del sistema cambiario. 

Debemos tener en cuenta que el ataque se producirá antes de que se agoten las reservas internacionales, ya que en caso contrario habría un salto discreto y perfectamente anticipado en el tipo de cambio que supondría una tasa de ganancia instantáneamente infinita.

Pero, ¿cuándo cabe esperar este tipo de comportamiento por parte de los agentes? ¿podemos anticipar la crisis cambiaria? 

\paragraph{Tipo de cambio sombra: ¿cómo se anticipan los agentes?}
Para resolver esta cuestión se define el \textbf{tipo de cambio sombra} ($\tilde s$) como el tipo de cambio que prevalecería si se pasase a un régimen de tipo de cambio flexible. Podemos visualizarlo gráficamente de la siguiente forma:

\imagenfit{1era_Gen_dinamica.pdf}{Dinámica de las reservas y del tipo de cambio sombra en el modelo de FLOOD y GARBER (1984)}{Elaboración propia a partir de FLOOD y GARBER (1984). Valores ilustrativos.}

El modelo anticipa que el stock de reservas se agotará progresivamente en el tiempo ($t$) por la necesidad de financiar el déficit. El panel izquierdo lo muestra: mientras dura la paridad, la oferta monetaria permanece constante en $\beta\bar s$, porque el crédito interno crece a la tasa $\mu$ y las reservas caen a esa misma tasa ($\dot R(t)=-\mu$); si nada lo interrumpiese, se agotarían en $R(0)/\mu$. El panel derecho muestra el tipo de cambio sombra. Si el tipo de cambio fuese flexible, las reservas dejarían de moverse y toda la expansión del crédito interno se trasladaría a la oferta monetaria y, por tanto, al tipo de cambio, que se depreciaría (un aumento es una depreciación). Al suponer una tasa de monetización constante, el tipo de cambio sombra se representa como una recta creciente en el tiempo, de pendiente $\mu/\beta$.

¿Cuándo se produciría dicho ataque? Existen diversas formas de aproximarlo. Siguiendo a FLOOD y GARBER (1984), podemos considerar que dicho ataque se aproxima por la \textbf{condición de no arbitraje}. Con ayuda del gráfico, podemos identificar tres zonas.

En primer lugar, la \textbf{región estable: $\tilde s <\bar s$}. Si el régimen fijo colapsara en este momento, el mercado llevaría el tipo de cambio a un nivel menor que el tipo fijo actual: se apreciaría la moneda nacional y no existen incentivos económicos para el ataque especulativo. ¿Por qué? Supongamos que $\bar s=10$ y $\tilde s=8$. El Banco Central vende una unidad de divisa por diez unidades de moneda nacional. Si compramos divisa extranjera (reservas) a ese precio, tras el colapso la paridad se fijará en $\tilde s=8$. Esto implica que ahora obtendremos dos unidades monetarias nacionales menos por cada unidad de divisa que tengamos, ergo generaremos una pérdida por la transacción. Esta generación de pérdidas por parte de los inversores elimina los incentivos de iniciar un ataque especulativo y, en consecuencia, nos sitúa en la región estable del gráfico.

En segundo lugar, tendríamos la \textbf{región \textit{ficticia}: $\tilde s>\bar s$}. Si nos encontramos en esta región del gráfico y ocurre el colapso del sistema cambiario, el tipo fijo saltaría a un nivel más alto: se depreciaría la moneda nacional y existirían incentivos económicos para el ataque especulativo. ¿Por qué? Suponiendo que el tipo está en $\bar s=10$ y el tipo sombra en $\tilde s=15$, comprando divisa por moneda nacional generaríamos una ganancia tras el colapso de la paridad. Por cada unidad de divisa obtendríamos cinco unidades adicionales de moneda nacional de la que pagamos por obtener la divisa en primer lugar, ergo la moneda se ha depreciado. Esta oportunidad de arbitraje genera un incentivo constante para el ataque especulativo. Sin embargo, la imposibilidad de que existan dichas oportunidades hace que esta situación sea virtualmente imposible.

Por último, tenemos el punto de \textbf{equilibrio especulativo: $\tilde s=\bar s$}. Lo que nos conduce a la única situación en la que se genera un incentivo para el ataque compatible con los supuestos del modelo. ¿Qué pasaría si ambos tipos de cambio fuesen iguales? Este es el momento en el que se produce el ataque especulativo en equilibrio. La razón es de teoría de arbitraje bajo previsión perfecta: si los especuladores esperaran un poco más allá de $T$ existirían beneficios positivos, pero entonces cada especulador tendría incentivo a adelantarse ligeramente a los demás para capturar primero las reservas del banco central. Esa competencia hace imposible que el sistema sobreviva cuando ya existen beneficios esperados y, en consecuencia, el ataque ocurre exactamente cuando el beneficio esperado pasa de negativo (primera situación, $\tilde s<\bar s$) a cero ($\tilde s=\bar s$).

¿Cómo podemos conocer el momento exacto $T$? La clave está en lo que ocurre después del ataque: las reservas internacionales se han agotado y el tipo de cambio flota, de modo que la oferta monetaria pasa a coincidir con el crédito interno y es el tipo de cambio el que debe ajustarse para que el mercado de dinero siga en equilibrio. El tipo de cambio sombra es precisamente esa trayectoria de equilibrio en flotación. Como el crédito interno crece de forma lineal, el tipo sombra también es una función lineal del crédito interno; llevando esa solución a la forma reducida se identifican sus dos coeficientes y, al igualarlo con la paridad, se obtiene la fecha del ataque (modelo de FLOOD y GARBER (1984), solución por el método de coeficientes indeterminados; el desarrollo paso a paso, en el Anexo~\ref{modelos_primera}):

\eqblock{
\begin{aligned}
    &\text{Tipo sombra}\ \big(R=0,\ M(t)=D(t)\big):&&\tilde s(t)=\frac{\alpha\,\mu}{\beta^2}+\frac{D(t)}{\beta}=\frac{\alpha\,\mu}{\beta^2}+\frac{D(0)+\mu\,t}{\beta}\\[1em]
    &\text{Colapso}\ \big(\tilde s(T)=\bar s\big):&&\boxed{T=\frac{R(0)}{\mu}-\frac{\alpha}{\beta}}\qquad\text{donde,}\quad R(0)\equiv\beta\,\bar s-D(0)
\end{aligned}
}{Tipo de cambio sombra y momento del ataque especulativo: Modelo FLOOD y GARBER (1984). Primera generación: Inconsistencia del policy-mix}

Cada término del tipo sombra tiene una interpretación económica, y es la que da forma al panel derecho de la figura. El coeficiente $1/\beta$ indica cuánto se deprecia la moneda por cada unidad adicional de crédito interno una vez que el tipo de cambio flota: como $\beta$ es la demanda de dinero por unidad de tipo de cambio, para que el público absorba más dinero el nivel de precios (y, por la PPA, el tipo de cambio) tiene que subir en proporción. Por eso, si el crédito crece a la tasa $\mu$, el tipo sombra crece a la tasa $\mu/\beta$, que es la pendiente de la recta. El término constante, $\alpha\mu/\beta^2$, recoge el efecto de las expectativas: en flotación el público prevé una depreciación continua de $\mu/\beta$, que la PNCI traslada al tipo de interés interno; la demanda de dinero cae en $\alpha\mu/\beta$ y, para que el mercado monetario vuelva a equilibrarse, el tipo de cambio ha de situarse $\alpha\mu/\beta^2$ por encima del que correspondería solo al crédito interno, $D(t)/\beta$. Este término es el que adelanta el colapso: sin él, el tipo sombra alcanzaría la paridad justo cuando las reservas se agotan por sí solas.

El ataque se produce cuando el tipo sombra alcanza la paridad. La fecha resultante tiene dos componentes. $R(0)/\mu$ es el momento en que las reservas se agotarían sin ataque, simplemente al ritmo de la monetización; $\alpha/\beta$ es el plazo en que la actuación de los agentes racionales adelanta el colapso. Ese adelanto tiene una contrapartida en el panel izquierdo de la figura: en $T$ el régimen pasa de no esperar depreciación a esperar una depreciación de $\mu/\beta$, el tipo de interés interno salta de $i^*$ a $i^*+\mu/(\beta\bar s)$ y la demanda de dinero se reduce de golpe en $\alpha\mu/\beta$. El público se deshace de ese exceso de dinero comprando al banco central las reservas que le quedan, que son exactamente $\alpha\mu/\beta$ (las que habrían durado $\alpha/\beta$ más al ritmo $\mu$). Así se explica una propiedad característica del modelo: en el momento del ataque no salta el precio, el tipo de cambio, que pasa de forma continua de $\bar s$ a la senda de $\tilde s$; lo que salta es la cantidad, las reservas, que caen a cero de forma discreta. Si las reservas iniciales no alcanzan siquiera para absorber ese salto ($R(0)\leq\alpha\mu/\beta$), el ataque es inmediato.

En esencia, el ataque especulativo se dilatará más o menos en el tiempo en función de: (i) el nivel inicial de reservas internacionales, pues cuanto mayor sea, más tarde llega el ataque; (ii) el nivel de la paridad que se defiende, pues cuanto más depreciado esté el tipo de cambio fijo de la moneda doméstica, mayor es la demanda de dinero que lo sostiene ($\beta\bar s$) y, para un mismo crédito interno, mayores son las reservas iniciales; (iii) la tasa de crecimiento del crédito interno, pues cuanto menor sea, más tarde se agotan las reservas; y, (iv) la sensibilidad de la demanda de dinero al tipo de interés, pues cuanto mayor sea $\alpha$ en relación con $\beta$, mayor es la huida del dinero en el momento del ataque y más se adelanta el colapso. Ninguno de estos factores evita la crisis: solo la retrasan. La única salida es corregir su causa, la incompatibilidad entre la monetización del déficit y la paridad.

\subsection{Implicaciones de Política Económica}
En definitiva, los modelos de primera generación muestran cómo la mera fijación del tipo de cambio no basta para proporcionar disciplina monetaria y fiscal, sino que las políticas internas tienen que ser consistentes con el objetivo de mantenimiento de la paridad cambiaria: es imprescindible mantener una autoridad monetaria independiente y con dominancia sobre la Política Fiscal.

Como señala KRUGMAN (1999), a propósito de la \textit{trinidad imposible} de MUNDELL: \textbf{el punto es que no se puede tener todo, un país debe elegir dos de cada tres}: (i) puede fijar su tipo de cambio sin debilitar a su banco central, pero sólo mediante el mantenimiento de controles sobre los flujos de capital (como China en 1999); (ii) puede permitir la libre movilidad de capitales manteniendo la autonomía monetaria, pero sólo a expensas de dejar que el tipo de cambio fluctúe (Gran Bretaña o Canadá); o, (iii) puede optar por permitir el libre movimiento de capitales y estabilizar la moneda, pero sólo optando por el abandono de cualquier posibilidad de ajustar las tasas de interés para combatir la inflación o la recesión (la mayor parte de Europa, con el euro, o, como caso paradigmático, la Argentina de MENEM-CAVALLO).

La crisis de primera generación es, en el fondo, la forma en que el mercado hace cumplir este trilema ([\authorfont{Ver Tema 3.B.15}]): un país con libre movilidad de capitales que fija su tipo de cambio y, a la vez, usa la política monetaria para financiar al Tesoro está intentando tenerlo todo, y el ataque especulativo es el mecanismo que le obliga a renunciar a uno de los tres objetivos, normalmente la paridad.

\textbf{El problema de fondo es fiscal, no monetario.} El banco central del modelo no elige monetizar el déficit: se le exige. SARGENT y WALLACE (1981) mostraron que, cuando la autoridad fiscal fija su senda de déficit y la monetaria debe acomodarla (\textbf{dominancia fiscal}), la disciplina monetaria es solo aparente, porque tarde o temprano el banco central tendrá que financiar al Tesoro. En una economía cerrada esa financiación acaba en inflación; con tipo de cambio fijo y movilidad de capitales, acaba en pérdida de reservas y en crisis cambiaria. La independencia del banco central es, por tanto, necesaria pero no suficiente: el ancla cambiaria solo es creíble si la política fiscal es sostenible sin recurrir al señoreaje.

\textbf{El ataque no es la causa de la crisis.} El ataque especulativo es el mecanismo por el que se materializa una inconsistencia previa, no su origen. Los especuladores se limitan a adelantar en $\alpha/\beta$ un colapso que se produciría igualmente al agotarse las reservas, de modo que culparlos o restringir su actuación no evita la crisis. En el modelo, incluso, el ataque tiene un efecto estabilizador sobre el precio: si las reservas se agotasen sin ataque, en $R(0)/\mu$ el tipo de cambio saltaría de forma discreta de $\bar s$ al tipo sombra de ese momento, mientras que con el ataque pasa de un régimen a otro sin salto.

\textbf{Ganar tiempo no es resolver.} Todo lo que aumenta las reservas disponibles retrasa el ataque, pero no lo evita mientras la monetización continúe: como muestra la fecha del colapso, $T=R(0)/\mu-\alpha/\beta$, cada unidad adicional de reservas solo lo retrasa en $1/\mu$ unidades de tiempo. KRUGMAN (1979) ya señaló que, si el Gobierno puede endeudarse en divisas para reponer reservas, la crisis puede posponerse, a costa de una alternancia de ataques y recuperaciones de la confianza\textcolor{orange}{\textsuperscript{[Probable]}}. WYPLOSZ (1986) mostró que los controles de capital retrasan el ataque, pero no lo eliminan si persiste la inconsistencia de fondo\textcolor{orange}{\textsuperscript{[Probable]}}. De ahí la lógica de la condicionalidad de los programas del FMI de los años ochenta, que combinaban la financiación de las reservas con límites al crecimiento del crédito interno neto y con ajuste fiscal\textcolor{orange}{\textsuperscript{[Probable]}}: la ayuda financiera solo tiene sentido si se corrige la causa, porque de lo contrario se limita a financiar la salida de capitales.

\textbf{Las crisis se pueden anticipar.} Si la crisis es la consecuencia lógica de unos fundamentos insostenibles, debe poder anticiparse observándolos: la pérdida continuada de reservas, el crecimiento del crédito interno por encima de la demanda de dinero y el déficit monetizado. En las extensiones con precios rígidos, la monetización se traduce además en una apreciación real y en un deterioro de la cuenta corriente antes del ataque\textcolor{orange}{\textsuperscript{[Probable]}}. Esta lógica inspiró la literatura de indicadores adelantados: KAMINSKY, LIZONDO y REINHART (1998), con el método de señales, encuentran que entre los indicadores más fiables figuran la desviación del tipo de cambio real respecto de su tendencia, el cociente entre la oferta monetaria amplia y las reservas, las exportaciones y la producción\textcolor{orange}{\textsuperscript{[Probable]}}.

\textbf{Mejor una salida ordenada que el colapso.} Si la paridad es insostenible, abandonarla de forma ordenada (devaluando o dejando flotar el tipo de cambio mientras aún quedan reservas) evita perderlas de golpe en el ataque final. En la práctica, las autoridades tienden a retrasar esa decisión por su coste en credibilidad, y esa resistencia, ausente del modelo, es la que analizarán los modelos de segunda generación. La experiencia de las crisis de los años noventa llevó, además, a la llamada \textbf{visión bipolar}: según FISCHER (2001), con plena movilidad de capitales los regímenes intermedios (paridades fijas pero ajustables) son vulnerables, y los países tienden a desplazarse hacia los extremos, la flotación o la fijación institucionalmente muy firme ([\authorfont{Ver Tema 3.B.15}]).

Los modelos de primera generación explican las crisis monetarias a través del deterioro de los fundamentos económicos, lo que permitió explicar las sucesivas crisis cambiarias/soberanas de la década perdida sudamericana: no fueron señal de un mal funcionamiento de los mercados sino de la inconsistencia de objetivos en sus políticas, la consecución de sus planes internos y el mantenimiento del objetivo cambiario anunciado.

La evidencia econométrica respalda esta lectura en algunos episodios. BLANCO y GARBER (1986) estimaron, con una versión estocástica del modelo, la probabilidad trimestral de devaluación del peso mexicano, que aumentaba en los periodos previos a las devaluaciones de 1976 y 1982\textcolor{orange}{\textsuperscript{[Probable]}}. CUMBY y VAN WIJNBERGEN (1989) aplicaron un enfoque similar a la ``tablita'' argentina de 1979-1981 y atribuyeron su colapso a una expansión del crédito interno incompatible con la pauta cambiaria anunciada\textcolor{orange}{\textsuperscript{[Probable]}}.




\subsubsection{Limitaciones: del determinismo de los fundamentos a la autorrealización}

La principal virtud del modelo, su determinismo, es también su principal limitación: describe crisis inevitables, previsibles y causadas solo por los fundamentos. Ni la evidencia empírica ni los supuestos de partida favorecieron su consolidación posterior como explicación general de las crisis cambiarias.

\textbf{Un Gobierno que no decide.} Mientras que los agentes privados son capaces de utilizar toda la información, las autoridades se perfilan como un agente estático con un comportamiento lineal: monetizan el déficit a una tasa fija y defienden la paridad hasta agotar las reservas. El modelo no considera otras alternativas de financiación (como la deuda), ni el ajuste fiscal o las políticas de austeridad, ni la subida de los tipos de interés para defender la moneda, ni la devaluación anticipada. Tampoco pondera los costes y los beneficios de mantener la paridad. En la realidad, mantener o abandonar un tipo de cambio fijo es una decisión de política económica que depende de lo costoso que resulte defenderlo, en desempleo, en coste de la deuda o en la salud del sistema bancario. Es justo lo que introducirán los modelos de segunda generación al dotar al Gobierno de una función objetivo y de una \textit{cláusula de escape}.

\textbf{Un único equilibrio, sin papel para las expectativas.} Los fundamentos determinan por sí solos si hay crisis y cuándo: los agentes se limitan a anticipar correctamente un desenlace que no pueden alterar. Sin embargo, ya OBSTFELD (1986) mostró, con este mismo marco, que si los agentes anticipan que tras el ataque el Gobierno cambiará su política (por ejemplo, acelerando la creación de crédito interno una vez liberado del compromiso cambiario), pueden coexistir dos equilibrios para unos mismos fundamentos, uno con ataque y otro sin él\textcolor{orange}{\textsuperscript{[Probable]}}. La crisis puede entonces ser \textbf{autocumplida}: se produce porque se espera. Es el germen de la segunda generación.

\textbf{Previsión perfecta: crisis sin sorpresa.} En el modelo el colapso llega en una fecha conocida y sin salto del tipo de cambio. En la realidad, las crisis suelen sorprender a los mercados y vienen acompañadas de fuertes devaluaciones. Las versiones estocásticas del modelo, como la del propio FLOOD y GARBER (1984) o la de BLANCO y GARBER (1986), introducen incertidumbre sobre el crédito interno, de modo que el ataque se produce con una probabilidad y el tipo de cambio puede saltar en el momento del colapso\textcolor{orange}{\textsuperscript{[Probable]}}; pero no cambian el diagnóstico, porque la causa sigue siendo la monetización.

\textbf{Sin economía real ni sistema financiero.} El modelo carece de producción, de empleo, de cuenta corriente y de balances. Por ello no puede explicar por qué un Gobierno solvente abandonaría la paridad (el coste de defenderla en términos de desempleo, que será el eje de la segunda generación), ni cómo una crisis cambiaria se entrelaza con una crisis bancaria y con el deterioro de los balances privados (eje de la tercera). Las extensiones posteriores, que recogen AGÉNOR, BHANDARI y FLOOD (1992), incorporaron la esterilización de la pérdida de reservas, los efectos reales y los controles de capital, pero sin abandonar la lógica de unos fundamentos que se deterioran de forma progresiva\textcolor{orange}{\textsuperscript{[Probable]}}.

\textbf{Una evidencia que deja de encajar.} Las grandes crisis de los años noventa no responden al patrón. En la crisis del SME de 1992-1993, los países atacados no financiaban sus déficits con el banco central ni perdían reservas de forma gradual antes del ataque: lo que se cuestionaba era su disposición a mantener tipos de interés altos en plena recesión, como veremos. En Asia, en 1997, los países afectados tenían superávits o presupuestos equilibrados e inflación baja, y el problema estaba en el sistema financiero y en los balances privados.

En suma, el modelo explica bien las crisis de países que financian su déficit con el banco central bajo un tipo de cambio fijo, pero no las de países con fundamentos aparentemente sólidos. Ese hueco es el que tratan de llenar los modelos de segunda generación, al hacer que el Gobierno decida y que las expectativas importen.

\section{Segunda Generación: Expectativas}
\puntosclave{

}

Estos y otros factores, junto con la evidencia de crisis posteriores sin que se observasen deterioros progresivos de los fundamentos (SME, 1992), justifican la aparición de los modelos de segunda generación.\footnote{%
    El SME era un sistema de zonas objetivo en el que se establecían unas bandas de fluctuación de las divisas de los Estados miembros en torno a la paridad de referencia con el ECU, unidad de cuenta del sistema. Su predecesora, entre 1972 y 1979, fue la llamada \textbf{serpiente europea}. El origen de la crisis del SME (1992) puede remontarse al proceso de reunificación alemana (1990), que produjo una coyuntura interna difícil de sortear: una elevada disparidad territorial y un objetivo declarado de convergencia nacional. Para perseguir este objetivo, el país optó por combinar una política fiscal expansiva con una política monetaria contractiva (para controlar la inflación). 

Como el SME estaba dominado, en la práctica, por el marco alemán, dicho \textit{policy-mix} se proyectó más allá de las fronteras alemanas y generó desequilibrios en el sistema, obligando a los países a optar entre sus objetivos internos o sus objetivos externos. La estabilidad del SME pasaba por sortear las dificultades que presentaba la asimetría en los ciclos entre los integrantes del sistema. Especialmente, entre una Alemania unificada y el resto del SME, parte de estos en etapa recesiva (especialmente UK). El mantenimiento de la paridad cambiaria obligaba una política monetaria contractiva (simetría con DEU), con un alto coste en términos de desempleo. La pregunta era entonces sencilla: ¿cuán comprometidos estaban, en realidad, los países? Las expectativas de los agentes del abandono del sistema cambiario generaron fuertes ataques especulativos en el mercado que condujeron a la crisis del SME de 1992. Lo importante, para nuestro objeto del Tema, es que esta crisis no vino provocada por un paulatino agotamiento de las reservas que condujese inexorablemente al abandono del tipo de cambio fijo, sino por la progresiva pérdida de credibilidad de dicho compromiso cuando está generando altos costes internos.
}

\textbf{¿Qué queremos decir?} Esta rama pretende explicar cómo la imposibilidad de mantener un compromiso cuando está generando profundos costes internos puede conducir a una crisis del sistema cambiario. En este contexto, los ataques se producen incluso cuando las políticas macroeconómicas son consistentes con el mantenimiento del tipo de cambio fijo: basta que los especuladores perciban que el mantenimiento del tipo de cambio es difícil de compatibilizar con la consecución de otros objetivos alternativos para que se produzca el ataque. No obstante, vamos a ver cómo esta construcción sobre expectativas hace que la causalidad no sea unidireccional: las expectativas de los agentes sobre la credibilidad del gobierno pueden alterar a su vez los incentivos de las autoridades a la hora de evaluar la idoneidad del objetivo anunciado. De esta forma emergen equilibrios múltiples ya que en un mismo contexto macroeconómico podrá o no darse o ser un ataque en función de las diferentes percepciones sobre una misma situación.\footnote{%
    Al existir varios equilibrios hacia los que la economía puede encaminarse, cualquier nueva información o una variable superflua podrían alterar las expectativas de los agentes en un sentido u otro y, por tanto, predominan los conceptos de corte keynesiano como efectos rebaño, expectativas autocumplidas, etc.
} 

Las principales aportaciones de esta nueva rama de modelos corresponden a OBSTFELD (1994), CALVO (1995), COLE y KEHOE (1996), DRAZEN (1998) y JEANNE (2000).\footnote{%
    El modelo de zonas objetivo de KRUGMAN (1991), que plantea una ecuación de determinación del tipo de cambio que depende de los fundamentos y de la tasa de depreciación esperada. podría englobarse en esta generación de modelos. La credibilidad del compromiso de la autoridad monetaria con el mantenimiento del tipo de cambio resultará crucial para la supervivencia del sistema cambiario cuando el tipo de cambio se acerque a los límites de las bandas objetivo.
}

En esencia, estos modelos nos muestran cómo y por qué, cuando aumentan las expectativas de devaluación, su comportamiento conduce a que aumenten los costes de mantener un tipo de cambio fijo.

\subsection{JEANNE (2000): ¿Cómo afectan las expectativas?}
Supongamos una economía pequeña, abierta, con un objetivo cambiario fijo declarado y en la que se cumple la condición de Paridad de Poder Adquisitivo. Asimismo, consideraremos un escenario biperiodo en el que las autoridades mantienen inicialmente la paridad cambiaria y, en el segundo periodo, pueden decidir si devaluar la moneda o no. De esta forma, introducimos la principal diferencia con los modelos de primera generación: el compromiso cambiario es ahora estado-dependiente, pues las autoridades pueden ejercer una \textbf{cláusula de escape}, evaluando de forma continua la idoneidad de mantener la paridad por medio de un análisis coste-beneficio. Seguimos la versión simplificada del enfoque de cláusula de escape, en la línea de OBSTFELD (1994, 1996), que sintetiza JEANNE (2000)\textcolor{orange}{\textsuperscript{[Probable]}}.

Para simplificar la exposición, presentaremos únicamente las ecuaciones clave del modelo y recurriremos al análisis gráfico; el desarrollo completo se recoge en el Anexo~\ref{anx:modelo_segunda}.

La autoridad monetaria persigue dos objetivos que pueden entrar en conflicto: el objetivo externo de mantener la paridady el objetivo interno de estabilizar la economía. 

Consideraremos que esta decisión se tomará en función, únicamente, de sus efectos sobre el desempleo (el \textbf{objetivo interno contrapuesto}). Por tanto, este objetivo puede representarse mediante la minimización de una función de pérdidas con la siguiente forma (modelo de cláusula de escape, versión con pérdida cuadrática y coste fijo de la devaluación):\footnote{%
    Las variables que la componen son sencillas de interpretar: $u_{t-1}$ y $u_t$ son el desempleo de cada periodo (o sus desviaciones respecto a la tasa natural); $\pi_t$, la inflación; y, $\mathbb E_{t-1}[\pi_t]$, la inflación esperada al cerrar los contratos.

    Nótese que $u_t$ es la tasa de desempleo en el segundo periodo y viene determinada por una curva de Phillips aumentada por las expectativas. Esta relación puede justificarse por la existencia de cierta rigidez salarial que implique que los salarios nominales del segundo periodo se determinen en el primer periodo, de forma que aumentos no esperados de la inflación puedan permitir situar la tasa de desempleo por debajo de su nivel natural en el corto plazo por medio de una reducción de los salarios reales y por tanto un aumento del empleo.
}

\eqblock{
\begin{aligned}
    &\text{Objetivo}:&&\min\ L_t=(u_t)^2+\delta\,C\\[1em]
    &\text{donde},&&u_t=\underset{\text{\scriptsize (persistencia)}}{\rho\,u_{t-1}}-\underset{\text{\scriptsize (rigidez salarial)}}{\alpha\,(\pi_t-\mathbb E_{t-1}[\pi_t])}\\[1em]
    &\text{\scriptsize PPA}:&&\pi_t=\delta\,d,\qquad \delta=\{0,1\}
\end{aligned}
}{Función de pérdidas y curva de Phillips: Modelo de cláusula de escape (JEANNE, 2000). Segunda generación: Expectativas}

Nótese que: (i) $\delta=\{0,1\}$ es una variable ficticia que recoge la decisión de las autoridades de devaluar o no; (ii) $C$ es el coste de abandonar el tipo de cambio fijo; y, (iii) por la condición de PPA, la tasa de inflación será igual a la cuantía de la devaluación si ésta se produce ($\pi_t = d$), o nula ($\pi_t=0$) si las autoridades optan por mantener la paridad. Los dos parámetros de la curva de Phillips tienen un papel central: $\rho$ mide la \textbf{persistencia} del desempleo, cuánto del desempleo heredado se arrastra al periodo siguiente, y $\alpha$, la \textbf{eficacia de la sorpresa inflacionaria}, cuánto reduce el desempleo una inflación no anticipada. El desempleo heredado, $u_{t-1}$, es el \textbf{fundamento} del modelo.

No es necesario especificar el comportamiento de los agentes para establecer las condiciones de equilibrio (o regiones estables) del modelo. La razón es sencilla: es razonable suponer que los agentes actúen primero, tomando sus decisiones (los salarios nominales) en función del nivel de inflación esperado, es decir, atribuyendo una probabilidad $\mathrm P(d)=\{0,1\}$ al abandono del compromiso del Gobierno. En este escenario, las autoridades simplemente reaccionarán óptimamente a las decisiones de los agentes, es decir, a los niveles de inflación esperada. El equilibrio se alcanza entonces iterando hacia atrás: para cada expectativa posible calculamos la respuesta óptima del Gobierno, y una expectativa es de equilibrio cuando esa respuesta la confirma. Basta comparar, en cada caso, la pérdida de mantener la paridad ($L^f$) con la de devaluar ($L^d$):

\eqblock{
\begin{aligned}
    &\mathbb E_{t-1}[\pi_t]=0:&&\underset{\text{\scriptsize mantener}}{L^f=(\rho\,u_{t-1})^2}\ \ \text{vs.}\ \ \underset{\text{\scriptsize devaluar}}{L^d=(\rho\,u_{t-1}-\alpha d)^2+C}\quad\Longrightarrow\quad\text{devalúa si}\ \ 2\rho\,u_{t-1}>\frac{C}{\alpha d}+\alpha d\\[1.5em]
    &\mathbb E_{t-1}[\pi_t]=d:&&\underset{\text{\scriptsize mantener}}{L^f=(\rho\,u_{t-1}+\alpha d)^2}\ \ \text{vs.}\ \ \underset{\text{\scriptsize devaluar}}{L^d=(\rho\,u_{t-1})^2+C}\quad\Longrightarrow\quad\text{devalúa si}\ \ 2\rho\,u_{t-1}>\frac{C}{\alpha d}-\alpha d
\end{aligned}
}{Reacción óptima del Gobierno según las expectativas: Modelo de cláusula de escape (JEANNE, 2000). Segunda generación: Expectativas}

¿Cuál será entonces la reacción óptima de las autoridades? Como vemos, dependerá de las expectativas de los agentes, el coste de la devaluación y, crucialmente, de la sorpresa inflacionaria. Si los agentes creen firmemente en el compromiso cambiario, una devaluación por sorpresa reduce el desempleo; habrá incentivos para devaluar siempre que esa ganancia supere su coste, es decir, cuando $(\rho\,u_{t-1})^2-(\rho\,u_{t-1}-\alpha d)^2>C$, lo que exige un desempleo heredado elevado. Por el contrario, si los agentes no confían en el compromiso cambiario, la devaluación se vuelve mucho más atractiva: (i) si la autoridad devalúa, acarrea el coste de la devaluación ($C$), pero la inflación no sorprende a nadie y el desempleo se limita a su componente heredado; pero, (ii) si la autoridad mantiene el objetivo cambiario, se produce una sorpresa desinflacionaria ($\pi_t-\mathbb E_{t-1}[\pi_t]=-d$): los salarios, pactados contando con la devaluación, resultan demasiado altos en términos reales y el desempleo aumenta, por lo que mantener la paridad deviene muy costoso. La expectativa de devaluación encarece la defensa del tipo de cambio, y ese es el canal por el que las expectativas pueden autocumplirse. La figura recoge ambos casos: el desempleo heredado a partir del cual la pérdida de mantener supera a la de devaluar es mucho menor cuando se espera la devaluación ($u_M$) que cuando no se espera ($u_D$).

\imagenfit{2da_Gen_perdidas.pdf}{Pérdidas de mantener y de devaluar según las expectativas en el modelo de cláusula de escape}{Elaboración propia a partir de JEANNE (2000). Valores ilustrativos.}

En definitiva, vemos como tomar la decisión de abandonar o no el objetivo cambiario depende de si los costes de abandonarlo son inferiores o no a los de mantenerlo. Por un lado, algunos \textbf{costes de mantenerlo} son: (i) si el mantenimiento de la paridad requiere incrementar los tipos de interés, el coste de la deuda pública podría aumentar; (ii) si existe libre movilidad de capitales, mantener un tipo de cambio fijo implica renunciar a la política monetaria para objetivos internos; o, (iii) se renuncia a los efectos positivos de una depreciación sobre la balanza de pagos vía ganancias de competitividad precio. Por otro lado, algunos \textbf{costes de abandonarlo} incluyen: (i) pérdida de credibilidad antiinflacionista, ya que, generalmente, el tipo de cambio se fija frente a la moneda de un país con baja inflación, de manera que permite importar credibilidad antiinflacionista; o, (ii) posibles costes políticos del abandono del tipo de cambio fijo. En el modelo, los primeros se resumen en la pérdida por desempleo y los segundos en el coste fijo $C$.

La figura anterior permite leer directamente las regiones de equilibrio. En cada panel, el punto en que se cruzan las dos curvas marca el desempleo heredado a partir del cual al Gobierno le compensa devaluar: $u_D$ cuando nadie espera la devaluación (panel izquierdo) y $u_M$ cuando todos la esperan (panel derecho). Como $u_M$ es menor que $u_D$, los dos cortes dividen el eje del desempleo heredado en tres tramos. Por debajo de $u_M$, mantener es la mejor opción en los dos paneles: aunque los agentes esperasen la devaluación, el Gobierno no la validaría, de modo que el único equilibrio es el \textbf{mantenimiento}. Por encima de $u_D$, devaluar es la mejor opción en los dos paneles: aunque nadie la esperase, el Gobierno devaluaría, y el único equilibrio es la \textbf{devaluación}. Entre ambos umbrales, en cambio, la respuesta del Gobierno depende del panel en que nos encontremos, es decir, de lo que esperen los agentes: si esperan que mantenga, mantiene, y si esperan que devalúe, devalúa. Las dos expectativas se autocumplen y hay \textbf{dos equilibrios}: es la \textbf{región inestable}. Analíticamente, los umbrales son (el desarrollo, en el Anexo~\ref{anx:modelo_segunda}):

\eqblock{
\begin{aligned}
    &\underset{\text{\scriptsize siempre existe si: $\alpha, d>0$}}{\text{Región inestable}:}&&\boxed{u_M<u_{t-1}\leq u_D}\qquad\text{amplitud:}\quad u_D-u_M=\frac{\alpha d}{\rho}
\end{aligned}
}{Regiones de equilibrio: Modelo de cláusula de escape (JEANNE, 2000). Segunda generación: Expectativas}

Hay dos interpretaciones a este respecto. En primer lugar, tenemos la interpretación literal que obtenemos del estudio de las desigualdades: existen tres regiones bien diferenciadas que corresponden a dos conductas firmes y coherentes de la autoridad monetaria (devaluación/mantenimiento) y otra región inestable en la que la autoridad monetaria quedará sometida al vigor de cualquier ataque especulativo lo suficientemente grande (región inestable). Sin embargo, la otra interpretación, más sutil, permite apreciar la potencia del modelo.

\subsubsection{Crisis cambiaria: ¿cuándo y cómo se origina?}
Como vemos, todas las condiciones se han expresado en función de la tasa de desempleo (un fundamento importante de la economía). Esto tiene una razón de ser: los umbrales dependen de los parámetros del modelo y de la devaluación esperada, y es el desempleo heredado el que sitúa a la economía en una u otra región. La figura siguiente reúne en dos paneles todo lo que el modelo dice sobre esas regiones.

El panel A condensa en un solo gráfico la figura anterior. La \textbf{ganancia neta de devaluar}, $G=L^f-L^d$, es la distancia vertical entre la curva de pérdida de mantener y la de devaluar en cada uno de sus paneles: es positiva donde la curva de mantener está por encima, y se anula justo donde las dos curvas se cruzaban. La recta $G^0$ recoge esa ganancia cuando no se espera devaluación (el panel izquierdo de la figura anterior) y la recta $G^d$, cuando sí se espera (el panel derecho). Restando las pérdidas, ambas son rectas en el desempleo heredado: $G^0=2\rho\,\alpha d\,u_{t-1}-(\alpha d)^2-C$ y $G^d=2\rho\,\alpha d\,u_{t-1}+(\alpha d)^2-C$. Tienen la misma pendiente, $2\rho\,\alpha d$, porque cuanto mayor es el desempleo heredado más vale la sorpresa inflacionaria; y $G^d$ está siempre $2(\alpha d)^2$ por encima de $G^0$, porque la expectativa de devaluación encarece mantener la paridad. Sus cortes con el eje son los dos umbrales, $u_D$ y $u_M$, y la zona sombreada entre ellos es la región inestable.

\imagenfit{2da_Gen_regiones.pdf}{Regiones de equilibrio en el modelo de cláusula de escape}{Elaboración propia a partir de JEANNE (2000). Valores ilustrativos.}

El panel B muestra cómo se desplazan esos umbrales cuando cambia la devaluación esperada, y las rectas del panel A explican por qué. Al aumentar $\alpha d$ ocurren dos cosas. Por un lado, las dos rectas se inclinan más, porque su pendiente, $2\rho\,\alpha d$, crece: es el \textbf{deflactor de escape}, pues una devaluación mayor rinde más en términos de desempleo y el coste fijo $C$ pesa relativamente menos ($\frac{C}{\alpha d}$ cae), lo que acerca los dos cortes al origen y facilita el recurso a la cláusula de escape. Por otro lado, las dos rectas se separan más, porque la distancia entre ellas, $2(\alpha d)^2$, también crece: es la \textbf{perturbación nominal}, pues una devaluación mayor es un shock nominal más intenso, que perjudica al Gobierno si la hace por sorpresa ($G^0$ baja) y le perjudica todavía más si no valida la que se espera ($G^d$ sube). Las dos vías empujan en el mismo sentido el corte de $G^d$, de modo que el umbral de mantenimiento, $u_M$, cae siempre al aumentar la devaluación esperada (la curva discontinua del panel B). Sobre el corte de $G^0$, en cambio, actúan en sentido contrario: mientras la devaluación esperada es moderada domina el deflactor y $u_D$ cae; cuando es grande domina la perturbación nominal y $u_D$ sube. Por eso el umbral de devaluación tiene forma de U, con un mínimo en $\alpha d=\sqrt{C}$.\footnote{%
    Se puede comprobar derivando el umbral, $u_D(d)=\frac{1}{2\rho}\left(\frac{C}{\alpha d}+\alpha d\right)$, respecto a $d$: $\frac{\partial u_D}{\partial d}=\frac{\alpha}{2\rho}\left(1-\frac{C}{(\alpha d)^2}\right)$. Por tanto, el umbral se acortará ($\frac{\partial u_D}{\partial d}<0$) si y solo si $(\alpha d)^2<C$, y se ensanchará ($\frac{\partial u_D}{\partial d}>0$) si y solo si $(\alpha d)^2>C$.
} Por último, como las dos rectas son paralelas, la distancia horizontal entre sus cortes es la distancia vertical entre ellas dividida por su pendiente: $\frac{2(\alpha d)^2}{2\rho\,\alpha d}=\frac{\alpha d}{\rho}$. Es la amplitud de la región inestable, y crece de forma lineal con la devaluación esperada: la zona sombreada del panel B se ensancha hacia la derecha.

Con esta geometría, cada región tiene una lectura económica clara. Si los fundamentos de la economía están muy deteriorados ($u_{t-1}>u_D$), una devaluación será imperante: aunque nadie la espere, la sorpresa inflacionaria reduce los salarios reales, pactados de antemano, y con ellos el desempleo (en un modelo más amplio, la ganancia de competitividad, al aumentar la demanda externa, refuerza ese efecto). Esta primera región estable expresa, por tanto, la imposibilidad de mantener la paridad por la gravedad interna y la intensidad de los incentivos de la autoridad monetaria para recurrir a la devaluación para garantizar el ajuste interno. Además, cuánto mayor es la persistencia del desempleo ($\rho$), más inclinadas son las rectas del panel A y menor es el umbral, puesto que más tentador resultará usar la devaluación como mecanismo de escape cuando las fricciones del mercado de trabajo dificultan el ajuste.

Lo anterior es también aplicable, a la inversa, cuando nos encontramos en la otra región estable: el mantenimiento. Si los fundamentos económicos son muy robustos (desempleo heredado bajo, $u_{t-1}<u_M$), el compromiso es plenamente creíble: ni siquiera una expectativa generalizada de devaluación lograría que al Gobierno le compensase validarla, de modo que no pueden generarse incentivos a desviarse y el único equilibrio es el mantenimiento. Ahora bien, esta región solo existe si la devaluación esperada es moderada: el umbral de mantenimiento es positivo únicamente si $\alpha d<\sqrt{C}$ (en el panel B, la curva discontinua cruza el eje en $\sqrt{C}$). Con pleno empleo, el equilibrio creíble siempre existe, porque el Gobierno nunca devaluaría por sorpresa; pero, si se espera una devaluación lo bastante grande ($\alpha d>\sqrt{C}$), el coste de defraudar esa expectativa, $(\alpha d)^2$, supera al de devaluar, $C$, y el ataque también puede autocumplirse.

La situación interesante se da en la región inestable. Lo primero que debemos notar es que esta región expresa un deterioro \textbf{moderado} de los fundamentos económicos: no estamos tan mal. Ahora bien, un ataque especulativo podría forzar la devaluación de la moneda. Esto es así porque nos encontramos en la región de equilibrios múltiples: los dos equilibrios son posibles, y que la economía acabe en uno u otro no depende de los fundamentos, sino de lo que esperen los agentes. Si confían en la paridad, el Gobierno la mantiene; si esperan la devaluación, el Gobierno la valida. La figura siguiente lo resume: para cada tramo de desempleo heredado, recoge lo que hace el Gobierno según lo que esperan los agentes y marca las expectativas que se autocumplen.

\imagenfit{2da_Gen_mapa.pdf}{Mapa de equilibrios del modelo de cláusula de escape}{Elaboración propia a partir de JEANNE (2000).}

Por tanto, ¿de qué depende que sucedan o no dichos ataques? Depende crucialmente de la devaluación esperada, $d$, porque determina la amplitud de la zona de fragilidad, $\alpha d/\rho$: cuanto mayor es, más amplio es el tramo central del mapa. Como vemos, la intensidad esperada de la devaluación actúa como amplificador de la región de fragilidad.

Este es el insight importante: aunque una devaluación esperada muy alta no siempre reduce el umbral de devaluación (como hemos visto con la forma de U del panel B), sí aumenta de forma inequívoca el rango de fundamentos en el que la economía queda expuesta a equilibrios múltiples o ataques autocumplidos.

Precisamente es el mecanismo de los modelos de segunda generación o \textbf{escape clause}: \textbf{los fundamentos importan, pero las expectativas pueden desplazar a la economía hacia una crisis incluso sin que los fundamentos sean tan malos}.

Además, es interesante la información que nos proporciona la tasa de crecimiento de esta fragilidad, pues la amplitud $\alpha d/\rho$ aumenta con $\alpha$ y disminuye con $\rho$. En primer lugar, vemos como $\alpha$ mide la eficacia de la sorpresa inflacionaria/devaluatoria sobre el objetivo interno. Así, cuánto mayor es $\alpha$, más poderosa es la devaluación como instrumento de estabilización, pero también mayor es el daño de no validar una devaluación esperada pues amplía la zona de equilibrios múltiples. En segundo lugar, vemos que $\rho$ mide la severidad/persistencia del desempleo interno. Por tanto, cuánto mayor es $\rho$, más sensible será la autoridad monetaria al desempleo pasado y más se reducirán los umbrales de acción, por ello comprime la zona de fragilidad en términos de desempleo heredado, puesto que será una cuestión ya internalizada por la autoridad monetaria.

\subsection{Implicaciones de Política Económica}
Recapitulando, los modelos de segunda generación permiten explicar cómo las expectativas de los agentes pueden conducir a una crisis cambiaria incluso cuando los fundamentos económicos no son tan negativos y prevalece la disciplina macroeconómica.

Esto sucede porque los fundamentos económicos, no siendo irrelevantes, son suficientemente malos como para que la autoridad pueda tener incentivos a devaluar si los mercados coordinan expectativas contra la paridad, pero no tan malos como para que la devaluación sea inevitable. Ese es el núcleo del modelo: entre la zona de credibilidad plena y la zona de colapso inevitable existe una zona intermedia en la que las expectativas son capaces de seleccionar el equilibrio. La intensidad esperada de la devaluación, $d$, es crucial porque amplía inequívocamente esa zona de fragilidad: aunque $d$ tiene un efecto no monótono sobre el umbral de devaluación, sí tiene un efecto monótono sobre la amplitud del intervalo inestable. Cuanto mayor es la devaluación esperada, mayor es el rango de fundamentos para el cual la economía queda expuesta a ataques autocumplidos.

Las razones que originan estas expectativas son muy diversas. Los ataques especulativos pueden estimular su aparición, pero también otras variables pueden contribuir a esta fragilidad. Por ejemplo, el uso frecuente y oportunista de la devaluación para restablecer el equilibrio interno, o incluso el volumen de las reservas. También puede ser determinante la presión política que se ejerza por parte de otros gobiernos o los compromisos que se adquieran en virtud de acuerdos internacionales. De este diagnóstico se derivan varias implicaciones, algunas opuestas a las de la primera generación.

\textbf{Los fundamentos siguen importando, pero de otra forma.} En la primera generación bastaba con evitar la monetización del déficit; ahora no basta con tener unas cuentas saneadas, porque lo que se juzga es la disposición del Gobierno a soportar el coste interno de la paridad. La prevención consiste en mantener la economía lejos de la zona de fragilidad: un desempleo, una deuda pública o un sistema bancario frágiles hacen creíble que el Gobierno prefiera devaluar, aunque hoy no lo necesite. Por eso las crisis de este tipo son difíciles de anticipar: en la zona de equilibrios múltiples el mismo estado de los fundamentos es compatible con la calma y con el ataque, y el desencadenante puede ser cualquier noticia que coordine las expectativas.

\textbf{La defensa con tipos de interés puede ser contraproducente.} En la primera generación, subir los tipos de interés no tenía sentido porque la crisis era inevitable; en la segunda, la subida puede incluso validar el ataque, porque eleva el coste de mantener la paridad (más desempleo, más carga de la deuda pública e hipotecaria) y, con ello, la tentación de abandonarla. En septiembre de 1992 el Riksbank sueco llegó a elevar su tipo de interés marginal al $500\%$ y el Reino Unido anunció dos subidas en un mismo día antes de salir del Mecanismo de Tipos de Cambio del SME\textcolor{orange}{\textsuperscript{[Probable]}}: los mercados interpretaron que esa defensa no era sostenible en plena recesión.

\textbf{La credibilidad se gana elevando el coste de la salida.} Si el problema es que el Gobierno conserva una cláusula de escape, la solución es encarecer su uso, es decir, aumentar $C$. Los regímenes de fijación firme (cajas de conversión, dolarización o unión monetaria) reducen el margen para la devaluación y desplazan la economía hacia el equilibrio sin ataque ([\authorfont{Ver Tema 3.B.15, Tema 3.B.16}]). Esta lógica, junto con la experiencia de 1992-1993, alimentó la visión bipolar que vimos al tratar la primera generación. Pero el coste de salida nunca es infinito: si los fundamentos se deterioran lo suficiente, incluso un compromiso institucional puede abandonarse, como mostró la caja de conversión argentina en 2001-2002.

\textbf{La coordinación de expectativas es un objetivo de política.} Como la economía puede acabar en cualquiera de los dos equilibrios, todo lo que ayude a coordinar las expectativas en el bueno tiene valor: la comunicación y la transparencia de las autoridades, unas reservas y líneas de crédito suficientes para disuadir el ataque y el apoyo de los socios o de las instituciones internacionales. En la segunda generación las reservas no determinan la fecha del colapso, como en la primera, pero su volumen puede influir en lo que esperan los agentes.

\textbf{La especulación puede ser desestabilizadora.} A diferencia de la primera generación, en la que el ataque se limitaba a adelantar un colapso inevitable, aquí el ataque puede provocar una crisis que no habría ocurrido. Esto proporciona un argumento para limitar la especulación a corto plazo: EICHENGREEN, TOBIN y WYPLOSZ (1995) defendieron ``echar arena en las ruedas'' de las finanzas internacionales con impuestos sobre las transacciones cambiarias o depósitos obligatorios sobre las entradas de capital. España, Irlanda y Portugal recurrieron a controles de capital temporales durante la crisis del SME\textcolor{orange}{\textsuperscript{[Probable]}}.

\textbf{¿Fundamentos o profecías autocumplidas?} El debate empírico sigue abierto, porque es difícil distinguir un ataque autocumplido de uno motivado por fundamentos que el investigador no observa. JEANNE (1997), al estimar un modelo de cláusula de escape para el franco francés en 1992-1993, encuentra que los fundamentos explican parte de las expectativas de devaluación, pero que el modelo con equilibrios múltiples se ajusta mejor a los datos\textcolor{orange}{\textsuperscript{[Probable]}}.

En definitiva, los modelos de segunda generación pueden verse como una manifestación cambiaria de los problemas de inconsistencia dinámica de KYDLAND y PRESCOTT (1977) y BARRO y GORDON (1983) ([\authorfont{Ver Tema 3.A.6, Tema 3.A.36}]): para que una serie de objetivos e instrumentos de Política Económica pueda catalogarse como dinámicamente consistente, su adecuada valoración debe incluir la posible reacción de los agentes en su diseño.

\subsubsection{Limitaciones: equilibrios múltiples y un sector privado ausente}
La principal aportación de la segunda generación, los equilibrios múltiples, es también su principal debilidad, y sus supuestos dejan fuera precisamente lo que resultaría decisivo en las crisis de finales de los noventa.

\textbf{Un modelo que lo explica todo y no predice nada.} En la región inestable, el modelo admite tanto el ataque como su ausencia, pero no dice cuál de los dos se producirá ni cuándo: la selección del equilibrio queda en manos de factores ajenos al modelo (las llamadas \textit{manchas solares}). Un modelo compatible con cualquier resultado es difícil de contrastar y poco útil para la política económica. MORRIS y SHIN (1998) mostraron que esta indeterminación depende del supuesto de información común: si cada especulador observa los fundamentos con un pequeño error, el equilibrio pasa a ser único y el ataque se produce cuando los fundamentos caen por debajo de un umbral. Las expectativas siguen importando, pero vuelven a estar ancladas en los fundamentos.

\textbf{Un Gobierno estilizado.} La función de pérdidas es \textit{ad hoc}, el coste de la devaluación ($C$) es una caja negra y todo el análisis se reduce a dos periodos. La decisión depende de parámetros difíciles de medir, lo que dificulta convertir el modelo en una guía operativa.

\textbf{Sin balances ni sistema financiero.} El único agente relevante es el Gobierno, y el canal de transmisión es el desempleo a través de la curva de Phillips. No hay bancos, ni deuda privada, ni descalces de monedas o de plazos. Sin embargo, las crisis cambiarias de los años noventa vinieron acompañadas, a menudo, de crisis bancarias: KAMINSKY y REINHART (1999) documentan que las crisis bancarias suelen preceder a las cambiarias y que ambas se refuerzan mutuamente (\textbf{crisis gemelas}).

\textbf{Una devaluación siempre expansiva.} En el modelo, devaluar reduce el desempleo; por eso es la salida tentadora. En las economías con mucha deuda en moneda extranjera ocurre lo contrario: la devaluación encarece esa deuda, deteriora los balances de empresas y bancos y hunde la inversión, de modo que es contractiva. El modelo no puede explicar por qué un país querría evitar la devaluación a toda costa ni por qué, una vez producida, la recesión fue tan profunda.

\textbf{Asia, 1997.} Las economías asiáticas que sufrieron la crisis no tenían problemas de desempleo que hicieran tentadora la devaluación: crecían con rapidez, tenían superávits o presupuestos equilibrados e inflación moderada. Ni la primera generación (no había monetización del déficit) ni la segunda (no había un Gobierno con incentivos para devaluar) encajan con lo ocurrido.

No obstante, debido a la naturaleza misma de esta aproximación metodológica, antes o después algún acontecimiento se presentará imposible de explicar a partir de la paleta de modelos disponibles.

\section{Tercera Generación: Temor de un hombre sabio}
\puntosclave{

}

Es precisamente lo que sucedió con la crisis acontecida a finales de los años noventa, iniciada en las economías asiáticas (en particular, Tailandia y Corea del Sur) y que se acabaría trasladando a Rusia (1998), Brasil (1999) y, en último término, Argentina (2001-2002). ¿Qué sucedió en este caso? Si bien es cierto que había serias inconsistencias en los fundamentos y sendas de déficit en algunas de estas economías, algunos países, y en particular aquellos donde surgió el \textbf{brote}, presentaban unas cuentas públicas saneadas, políticas monetarias prudentes y niveles de reservas holgados, y no sufrían, sobre todo, problemas de desempeño macroeconómico (estancamiento del PIB, desempleo o inflación).

La verdad es que, aún hoy en día, no hay consenso al respecto. Muchos autores argumentan que esta aparente solidez escondía ciertas debilidades del sector financiero acumuladas tras un proceso de liberalización brusca y descontrolada de los movimientos internacionales de capitales no precedido del establecimiento de sistemas de supervisión y regulación financiera adecuados.\footnote{%
    Esto dio lugar a fuertes entradas de capital traducidas en un endeudamiento excesivo, generalmente denominado en moneda extranjera y a corto plazo. Cuando cambiaron las expectativas sobre estas economías, ese cambio se tradujo en ataques especulativos contra el baht tailandés, que se dejó flotar el 2 de julio de 1997, y en el posterior contagio a toda la región, que acabó desembocando en una grave crisis financiera. Japón era la principal fuente de Inversión Directa y de préstamos bancarios internacionales de los países emergentes asiáticos, y los problemas de su sistema bancario, arrastrados desde comienzos de los noventa y agudizados en 1997-1998, contribuyeron a la retirada de buena parte de esa financiación\textcolor{orange}{\textsuperscript{[Probable]}}.
} Otros, por el contrario, argumentan que este proceso de liberalización de los movimientos de capital fue el verdadero detonante de la crisis experimentada y lo justifican por la inestabilidad y el comportamiento irracional que caracteriza los mercados financieros.\footnote{%
    Un argumento a favor de la segunda lectura es que ni siquiera los inversores más sofisticados están a salvo de esa inestabilidad, como muestra el caso de Long-Term Capital Management (LTCM). Este fondo de inversión, fundado en 1994 por John MERIWETHER y en cuya dirección participaban Myron SCHOLES y Robert MERTON (premios Nobel de Economía en 1997), constituía la referencia de lo que podría considerarse el \textbf{inversor racional}. LTCM, aunque muy apalancado, mantenía una cartera muy diversificada, centrada en operaciones de arbitraje con bonos y otros activos de bajo riesgo. No obstante, tras la suspensión de pagos rusa de agosto de 1998 acumuló en pocas semanas pérdidas tan cuantiosas que, en septiembre de ese año, tuvo que ser rescatado por un consorcio de catorce bancos coordinado por la Reserva Federal de Nueva York. El caso recuerda una frase atribuida a KEYNES: \textit{los mercados pueden mantenerse irracionales mucho más tiempo del que uno puede mantenerse solvente}.
}

\textbf{¿Qué queremos decir?} Aquí nos centraremos en presentar brevemente el primer enfoque, que dio origen a lo que se conoce como \textbf{modelos de tercera generación}. Estos, enfatizan las interrelaciones que existen entre el mercado financiero y el mercado cambiario, por lo que, en situaciones de crisis, estas deben de abordarse conjuntamente en modelos de \textbf{crisis gemelas}.\footnote{%
    Los vínculos de causalidad entre ambas crisis no son unidireccionales: 
\begin{lnum}
    \item \textbf{Crisis financiera} $\to$ \textbf{Crisis cambiaria}. La existencia de problemas en el sector financiero podría dar lugar al colapso de la moneda y por tanto a una crisis cambiaria. Por ejemplo, si el Banco Central, en cuanto prestamista de última instancia, se ve obligado a emitir dinero para financiar el rescate de las instituciones financieras domésticas con problemas;
    \item \textbf{Crisis cambiarias} $\to$ \textbf{Crisis financieras}. Por otro lado, una crisis cambiaria podría ser la causa de una crisis financiera cuando una devaluación o depreciación suponga un crecimiento explosivo de la deuda denominada en divisas. 
    \item \textbf{Fundamentos} $\to$ \textbf{Crisis cambiarias $+$ Crisis financieras}. Las crisis monetaria y financiera podrían tener origen en causas comunes, como un proceso de liberalización financiera desordenado combinado con la existencia de un sistema de garantía de depósitos implícito. La posibilidad para las entidades financieras de financiarse en el exterior podría provocar un aumento del crédito bancario que genere el colapso de los sistemas cambiario y bancario si por cualquier razón se produce una reversión de los flujos de capital.
\end{lnum}
} Vamos a ver, por tanto, la forma en la que podemos abordar este problema a través de un marco analítico completo. 


\subsection{Marco de análisis: ¿cómo abordar las crisis de tercera generación?}
Debido a la evidente heterogeneidad de las crisis que tratan de explicar, hay que tener en cuenta que esta literatura no cuenta con un único modelo representativo, como sí ocurría con FLOOD y GARBER (1984) en la primera generación o con el enfoque de cláusula de escape en la segunda. Por el contrario, se trata de un conjunto de aportaciones que ponen el énfasis en aspectos diferentes pero relacionados entre sí, difiriendo principalmente en los fenómenos explicativos.

Todas comparten, sin embargo, cuatro rasgos que las separan de las generaciones anteriores. En primer lugar, el protagonista deja de ser el Gobierno y pasa a ser el sector privado, en particular el sistema financiero y las empresas. En segundo lugar, la crisis cambiaria y la crisis bancaria o financiera dejan de analizarse por separado y se tratan como \textbf{crisis gemelas}. En tercer lugar, el detonante es una interrupción brusca de las entradas de capital, que habían financiado un endeudamiento a corto plazo y en moneda extranjera. Y, en cuarto lugar, la devaluación deja de ser una salida expansiva, como en la segunda generación, y se convierte en un factor que agrava la crisis.

Lo que distingue a las distintas ramas es dónde se origina la fragilidad: en los incentivos creados por las garantías públicas (riesgo moral), en el desajuste entre los plazos de los activos y los pasivos de los intermediarios financieros (crisis de liquidez) o en el desajuste entre las monedas en que están denominados los activos y los pasivos de las empresas (efectos balance). La primera rama conserva la lógica de la primera generación, porque la crisis es el desenlace previsible de un deterioro acumulado; las otras dos heredan los equilibrios múltiples de la segunda.

Se distinguen esencialmente tres.

\subsubsection{Riesgo moral: ¿por qué son predecibles?}
En primer lugar estarían los modelos de \textbf{riesgo moral}, o \textbf{modelos de crisis gemelas fundamentales o predecibles} (DOOLEY, 1997; publicado en 2000).

Estos modelos ponen el énfasis en el papel del Banco Central como prestamista de última instancia y, más en general, en las garantías públicas, explícitas o implícitas, sobre los pasivos del sistema financiero. Si los inversores creen que el Estado acudirá al rescate de las entidades en dificultades, prestarles es un negocio sin riesgo: la existencia de esta garantía conduce a las instituciones financieras a endeudarse a bajo coste, también en el exterior, y a llevar a cabo inversiones sin evaluar debidamente los riesgos, apareciendo problemas de apalancamiento y de riesgo excesivo. Las ganancias de las apuestas arriesgadas se las quedan los accionistas y las pérdidas recaen sobre el contribuyente, lo que alimenta un \textbf{sobreendeudamiento} y burbujas en los precios de los activos (KRUGMAN, 1998; CORSETTI, PESENTI y ROUBINI, 1999)\textcolor{orange}{\textsuperscript{[Probable]}}.

Esta frágil situación puede acabar abocando a la crisis cuando los agentes perciban que las reservas del Banco Central no pueden cubrir los pasivos garantizados. En el modelo de DOOLEY, el Gobierno dispone de un stock de reservas con el que respalda esos pasivos, y los inversores extranjeros prestan mientras el valor de lo garantizado no supere al de las reservas. Cuando ambos se igualan, cada inversor tiene incentivos para liquidar su posición antes de que la garantía se agote: es razonable suponer que se produzca una fuga masiva de capitales, colapsando las reservas y provocando la quiebra del sistema bancario. La ausencia de reservas impedirá defender el objetivo cambiario. En definitiva, se trata de crisis impulsadas por el agotamiento de las reservas, nexo entre el balance exterior y la estabilidad cambiaria: si los pasivos garantizados crecen a mayor ritmo que las reservas, estamos ante una situación de crisis predecible. La analogía con la primera generación es directa: el déficit público que se monetizaba se sustituye por un déficit oculto, el coste futuro del rescate, y el ataque se produce cuando el mercado lo anticipa\textcolor{orange}{\textsuperscript{[Probable]}}.

\paragraph{Implicaciones de Política Económica}

La primera implicación es que la liberalización financiera debe ir precedida de una regulación y una supervisión adecuadas: abrir la cuenta financiera en presencia de garantías implícitas equivale a subvencionar el endeudamiento exterior arriesgado. La segunda es que las garantías públicas deben ser explícitas, limitadas y con un precio acorde al riesgo (por ejemplo, un seguro de depósitos con primas ligadas al riesgo), y que deben ir acompañadas de una regulación que limite la toma de riesgos de las entidades aseguradas. La tercera, menos intuitiva, es que acumular más reservas no evita la crisis: en el modelo de DOOLEY, unas reservas mayores amplían el respaldo disponible y atraen más entradas de capital, de modo que solo retrasan el ataque y aumentan su tamaño\textcolor{orange}{\textsuperscript{[Probable]}}. Por último, el problema se reproduce a escala internacional: los grandes paquetes de rescate del FMI tras las crisis de México (1995) y Asia (1997) abrieron el debate sobre si las ayudas internacionales generan, a su vez, riesgo moral en los acreedores.

\paragraph{Limitaciones}

Sin embargo, también están sujetos a una serie de limitaciones, tanto analíticas (positivas y normativas) como empíricas. En el plano analítico, el modelo da por supuesto que las garantías existen y que todos los agentes conocen su alcance, y vuelve a describir crisis previsibles cuya fecha depende del agotamiento de las reservas. En el plano empírico, aunque en las economías asiáticas había abundantes indicios de riesgo moral (los conglomerados empresariales coreanos y las sociedades financieras tailandesas, por ejemplo), la crisis no fue anticipada por los mercados: los diferenciales de tipos de interés apenas se movieron antes del ataque\textcolor{orange}{\textsuperscript{[Probable]}}, y el contagio alcanzó a economías con sistemas de garantías muy diferentes. En el plano normativo, la lectura de la crisis como castigo a una inversión imprudente se usó para justificar la negativa a rescatar, cuando buena parte del daño se debió al propio pánico. Ese pánico es precisamente el objeto de la segunda rama.


\subsubsection{Asimetrías de balance: ¿por qué son impredecibles?}
En segundo lugar, tendríamos los modelos de \textbf{asimetrías en los plazos de vencimiento}, o \textbf{modelos de crisis gemelas autocumplidas o impredecibles} (CHANG y VELASCO, 1998; publicado en 2001).

Estos modelos se separan del papel de prestamista de última instancia del Banco Central como origen del problema y ponen el énfasis en cómo y por qué la asimetría entre el vencimiento de los activos y de los pasivos de las propias entidades financieras intermedias puede generar la crisis. Trasladan a una economía abierta el modelo de pánicos bancarios de DIAMOND y DYBVIG (1983): los bancos se financian a corto plazo, con depósitos y con préstamos exteriores que pueden retirarse en cualquier momento, e invierten en proyectos productivos a largo plazo que solo pueden liquidarse antes de su vencimiento con grandes pérdidas. Esa transformación de plazos es útil, porque permite financiar inversiones rentables con ahorro que los ahorradores quieren tener disponible, pero constituye un fenómeno de riesgo: el sistema es solvente si se le deja llegar al vencimiento, pero no puede atender a la vez a todos sus acreedores si estos reclaman su dinero al mismo tiempo.

Por tanto, es razonable suponer que este riesgo pueda conducir a pánicos bancarios: si los agentes dejan de confiar en el sistema, se producirán retiradas masivas de fondos. Cada acreedor sabe que, si los demás retiran sus fondos, los bancos tendrán que malvender sus activos y no podrán pagar a todos, de modo que lo racional es adelantarse. Se trata de modelos de equilibrios múltiples en los que el paso de una situación de equilibrio estable a otra de crisis se produce por retiradas aleatorias provocadas por manchas solares que hagan dudar de la solvencia de los bancos y por comportamientos de tipo rebaño, aunque los fundamentos económicos deben ser frágiles. En una economía abierta, la fragilidad relevante es la \textbf{iliquidez internacional}: que las obligaciones a corto plazo en moneda extranjera del sistema financiero superen los recursos en moneda extranjera que puede movilizar con rapidez, incluidas las reservas del Banco Central. Los acreedores que se niegan a renovar sus préstamos son tan peligrosos como los depositantes que retiran sus depósitos; por eso estos modelos hablan de crisis financiera y no solo de crisis bancaria: el pánico puede proceder de cualquiera de los acreedores a corto plazo del sistema.

¿Cómo se transforma ese pánico en una crisis cambiaria? Depende de cómo responda el Banco Central y del régimen cambiario (CHANG y VELASCO, 2000)\textcolor{orange}{\textsuperscript{[Probable]}}. Si el tipo de cambio es fijo y el Banco Central actúa como prestamista de última instancia, inyecta liquidez en moneda nacional para que los bancos puedan atender las retiradas; pero los depositantes y acreedores que reciben ese dinero lo cambian por divisas, de modo que el rescate se convierte en un ataque contra las reservas. Es el mecanismo de la primera generación, con la diferencia de que la creación de crédito no responde a un déficit público sino al rescate bancario, y de que puede desencadenarse de un día para otro. Si, por el contrario, el Banco Central no actúa como prestamista de última instancia, como ocurre con una caja de conversión, la paridad se salva, pero a costa de la quiebra del sistema bancario. Con tipo de cambio flexible, el Banco Central puede proporcionar liquidez en moneda nacional sin comprometer una paridad, y el pánico se traduce en una depreciación en lugar de en el agotamiento de las reservas. La crisis bancaria y la crisis cambiaria son, así, dos caras del mismo pánico: cuál de ellas domina depende de la política del Banco Central, y con tipo de cambio fijo y prestamista de última instancia se producen las dos.

\paragraph{Implicaciones de Política Económica}

Si el problema es la iliquidez y no la insolvencia, la política debe centrarse en la liquidez. En primer lugar, conviene vigilar la relación entre la deuda exterior a corto plazo y las reservas: la llamada regla de GREENSPAN y GUIDOTTI recomienda que las reservas cubran al menos la deuda exterior que vence en el plazo de un año\textcolor{orange}{\textsuperscript{[Probable]}}. En segundo lugar, alargar los plazos del endeudamiento exterior y desincentivar las entradas de capital a corto plazo, como hizo Chile con su encaje sobre las entradas de capital en la década de los noventa. En tercer lugar, disponer de un prestamista de última instancia internacional que pueda aportar liquidez en divisas, como defendió FISCHER (1999), o de acuerdos para que los acreedores renueven sus préstamos, como el que alcanzaron los bancos acreedores de Corea del Sur a finales de 1997\textcolor{orange}{\textsuperscript{[Probable]}}. Y, en cuarto lugar, el régimen cambiario importa: la flexibilidad cambiaria permite al Banco Central actuar como prestamista de última instancia sin poner en juego una paridad. En el debate sobre Asia, esta lectura de la crisis como un pánico de acreedores sostuvo la crítica de RADELET y SACHS (1998) a las recetas del FMI, que, al cerrar entidades y endurecer la política monetaria y fiscal en plena huida de capitales, habrían alimentado el pánico\textcolor{orange}{\textsuperscript{[Probable]}}.

\paragraph{Limitaciones}

Al igual que en la segunda generación, el modelo explica por qué el pánico es posible, pero no por qué se produce en un momento dado: la selección del equilibrio vuelve a depender de manchas solares, lo que dificulta la predicción y el contraste empírico. Además, el análisis se centra en los bancos y en el desajuste de plazos, y deja en segundo plano el desajuste de monedas en el balance de las empresas, el canal por el que la depreciación hundió la inversión y la producción en Asia. Por eso no explica la profundidad de la recesión posterior a la crisis, ni por qué una depreciación que debería haber estimulado las exportaciones resultó contractiva. Ese es el objeto de la tercera rama.


\subsection{Modelo de Balance: KRUGMAN (1999)}
Por último, tendríamos los modelos de \textbf{depreciación del tipo de cambio real}, o \textbf{modelos de balance} (KRUGMAN, 1999).

Por razones de tiempo, nos centraremos en desarrollar el modelo de balance propuesto por KRUGMAN (1999), por ser el más reciente; su desarrollo analítico completo se recoge en el Anexo~\ref{anx:modelos_tercera}.

A diferencia de los anteriores, en este modelo ni el Banco Central ni las instituciones intermedias de crédito tienen un papel central. Por el contrario, \textbf{la crisis se explica a través de los problemas de balance en las propias empresas}. El razonamiento es sencillo: una reversión de los flujos de capital extranjero, por cualquier motivo, que genere una pérdida de confianza por parte de los agentes provocará una depreciación del tipo de cambio real y por tanto un aumento del valor de las deudas denominadas en moneda extranjera y una reducción del valor de los activos nacionales, lo que acabará desequilibrando el balance de las empresas.

Todo el modelo puede resumirse en un \textbf{bucle de retroalimentación} de cuatro eslabones, que conviene presentar antes de desarrollar cada uno (modelo de balance de KRUGMAN (1999), versión con restricción de apalancamiento):

\eqblock{
\begin{aligned}
    &\text{Retroalimentación}:\quad \downarrow i_t^e\ \Rightarrow\ \uparrow p_t\ \Rightarrow\ \downarrow w_t\ \Rightarrow\ \downarrow i_t^f\ \Rightarrow\ \downarrow i_t^e\qquad \underset{\text{\scriptsize ¿cuánto?}}{\Longrightarrow}\qquad \boxed{\Delta i_t^f=\lambda\,\Delta i_t^e}
\end{aligned}
}{Bucle de retroalimentación y multiplicador financiero: Modelo de balance (KRUGMAN, 1999). Tercera generación}

Es decir, si los inversores esperan que la inversión ($i_t^e$) va a ser menor, prestan menos y sale capital del país; la menor demanda de bienes nacionales deprecia el tipo de cambio real ($p_t$); la depreciación encarece la deuda en divisas y reduce la riqueza de los empresarios ($w_t$); y, con menos riqueza, los empresarios pueden endeudarse menos, de modo que la inversión que pueden financiar ($i_t^f$) cae. Si cae más de lo que se esperaba, las expectativas se revisan de nuevo a la baja y el bucle se repite. Cuánto se amplifica el impulso inicial en cada vuelta lo mide el \textbf{multiplicador financiero}, $\lambda$.

\imagenfit{3era_Gen_bucle.pdf}{Bucle de retroalimentación del modelo de balance}{Elaboración propia a partir de KRUGMAN (1999).}

¿Cómo impacta esto en la economía real? Estos problemas de balance dificultan el acceso al crédito de empresas y particulares, lo que lastra la inversión (y el consumo) y genera disminuciones adicionales de las entradas de capital. De esta forma, las complicaciones financieras se transmiten a la economía real y este círculo de retroalimentación negativa puede conducir, sucesivamente, hasta la crisis. Por tanto, se trata de modelos de equilibrios múltiples en los que una pérdida de confianza puede generar un colapso financiero y por tanto fenómenos de expectativas autocumplidas.

Como vemos, se parte de una hipótesis muy básica: la capacidad de endeudamiento de los empresarios depende de su propia riqueza, que es función a su vez del nivel de endeudamiento de la economía, dado que el volumen de entradas de capital tiene efectos sobre el tipo de cambio real y por tanto sobre la valoración de la deuda en moneda extranjera. De esta forma, una disminución de las entradas de capital extranjero puede tener efectos negativos sobre el balance de las empresas, reduciendo su capacidad de endeudamiento y en consecuencia disminuyendo a su vez las entradas de capitales. Veamos ahora cada eslabón.

\textbf{Supuestos y mercado de bienes: el problema de transferencia}. Partimos de una economía pequeña y abierta que produce un único bien con capital y trabajo. El capital dura un único periodo, de modo que la producción de cada periodo está dada por la inversión del anterior. Los trabajadores consumen toda su renta, $(1-\alpha)y_t$, y los empresarios, que perciben la renta del capital, $\alpha y_t$, ahorran e invierten toda la suya.\footnote{%
    Por simplicidad, supondremos que la función de producción es COBB-DOUGLAS ($F(K,L)=K^\alpha L^{1-\alpha}$), de manera que la renta de los trabajadores sea igual a $(1-\alpha)y_t$ y la del capital, $\alpha y_t$.
} El gasto nacional en consumo e inversión se reparte en proporciones constantes: una fracción $\mu$ va a bienes importados y el resto, $1-\mu$, a bienes nacionales; las exportaciones, $X_t$, están dadas. Lo que no está dado es el \textbf{tipo de cambio real}, $p_t$, el precio de los bienes extranjeros en unidades de bien nacional (una subida es una depreciación real y un empeoramiento de la relación real de intercambio, $1/p_t$): es la variable que equilibra el mercado de bienes nacionales.\footnote{%
    \textbf{NOTA.-} El modelo no distingue entre tipo de cambio nominal y real: no hay inflación y todo se expresa en términos reales. Por eso, cuando se habla de depreciación, se trata siempre de una depreciación real.
} Esa es la pieza clave del modelo y la razón por la que se habla de un \textbf{problema de transferencia}: si el país recibe menos capital y cae la inversión, cae la demanda de bienes nacionales, y la producción, que está dada, solo puede seguir vendiéndose si se abarata respecto a los bienes extranjeros. Del equilibrio de ese mercado se obtiene cuánto se deprecia el tipo de cambio real por cada unidad de inversión perdida, $\frac{\partial p_t}{\partial i_t}=-\frac{1-\mu}{X_t}$: la depreciación es mayor cuanto más gasto recae sobre bienes nacionales y cuanto menores son las exportaciones.

\textbf{Balance y restricción de crédito: el efecto balance}. La riqueza de los empresarios es el valor de su capital menos sus deudas, denominadas en bienes nacionales ($b_t^d$) o en bienes extranjeros ($b_t^f$), estas últimas valoradas al tipo de cambio real: $w_t=\alpha y_t-b_t^d-p_t\,b_t^f$. Como no pueden endeudarse por más de $\theta$ veces su riqueza, su inversión no puede superar $i_t^f=(1+\theta)\,w_t$ (\textbf{restricción de apalancamiento}). Así, cada punto de depreciación real reduce la riqueza en $b_t^f$, la deuda en divisas, y cada unidad de riqueza perdida reduce la inversión financiable en $1+\theta$. Si la depreciación es suficientemente intensa, la riqueza se vuelve negativa y los empresarios quiebran; y, en el otro extremo, la inversión nunca supera el nivel $i^{\max}$ en el que su rentabilidad se iguala a la del exterior. Este es el efecto balance: \textbf{una depreciación real ($\uparrow p_t$) aumenta el valor de la deuda extranjera y reduce la riqueza neta, restringiendo la capacidad de apalancarse o endeudarse y afectando negativamente a la inversión}.

La figura resume los tres eslabones del bucle, uno por panel: (1) en el mercado de bienes, una menor inversión esperada exige un tipo de cambio real más depreciado; (2) en el balance, la depreciación reduce la riqueza en proporción a la deuda en divisas, hasta llevarla a la quiebra; y, (3) en el crédito, la riqueza limita la inversión financiable, que crece con ella a razón de $1+\theta$ entre la quiebra (inversión nula) y la inversión máxima rentable.

\subsubsection{Crisis cambiaria: ¿cuándo y cómo sucede?}
Multiplicando los efectos de los tres eslabones se obtiene cuánto varía la inversión financiable por cada unidad que varía la inversión esperada, tomando como dadas la producción y las exportaciones a corto plazo (modelo de balance de KRUGMAN (1999), multiplicador en el tramo en el que la restricción de apalancamiento es operativa; desarrollo en el Anexo~\ref{anx:modelos_tercera}):

\eqblock{
\begin{aligned}
    &\lambda\equiv\frac{\partial i_t^f}{\partial i_t^e}=\underset{\text{\scriptsize crédito}}{\frac{\partial i_t^f}{\partial w_t}}\cdot\underset{\text{\scriptsize balance}}{\frac{\partial w_t}{\partial p_t}}\cdot\underset{\text{\scriptsize bienes}}{\frac{\partial p_t}{\partial i_t^e}}=\boxed{\frac{(1+\theta)(1-\mu)\,b_t^f}{X_t}}>0
\end{aligned}
}{Multiplicador financiero: Modelo de balance (KRUGMAN, 1999). Tercera generación}

Esta ecuación, definida como el multiplicador financiero, nos permite conocer si acabaremos o no en una situación de crisis por el deterioro de las expectativas y su impacto sobre el balance de las empresas, y cómo.

Por un lado, podemos ver que el \textbf{multiplicador puede servir de amortiguador: $\lambda<1$}. Si el multiplicador es inferior a la unidad, el deterioro de las expectativas producirá una salida de capitales que empeorará las condiciones de financiación por el deterioro del balance. Sin embargo, cada vuelta del bucle reduce la inversión financiable menos de lo que se había reducido la esperada, de modo que el círculo de retroalimentación se irá amortiguando con el tiempo y no cabe esperar una crisis: el equilibrio es único.

Sin embargo, el \textbf{multiplicador puede también servir de acelerador: $\lambda>1$}, y abrir la región de equilibrios múltiples. Si el multiplicador es superior a la unidad, el deterioro de las expectativas producirá una salida de capitales que empeorará las condiciones de financiación por el deterioro del balance en mayor medida que la intensidad de la salida: cada vuelta del bucle reduce la inversión financiable más de lo que se esperaba. 

¿Qué sucede entonces? El proceso continúa agravándose, se potencia a sí mismo y puede adquirir una senda explosiva hasta un equilibrio de crisis financiera y cambiaria. No es inevitable: con $\lambda>1$ coexisten un equilibrio sin crisis y otro con crisis, y es la pérdida de confianza la que decide si la economía pasa de uno a otro.

\textbf{Los determinantes de la fragilidad}. ¿Cuáles son los factores que influyen en este multiplicador? Como cabría esperar: (i) el grado de apalancamiento ($\theta$), que determina cuánto se traslada a la inversión cada pérdida de riqueza; (ii) el peso de la deuda en moneda extranjera en relación con las exportaciones ($b_t^f/X_t$), que determina cuánta riqueza se pierde con cada depreciación y cuánto debe depreciarse el tipo de cambio para equilibrar el mercado de bienes; y, (iii) una baja propensión marginal a importar ($\mu$): cuanto más se concentra el gasto en bienes nacionales, más recae sobre ellos la caída de la inversión y mayor es la depreciación necesaria. Cada uno de estos factores refuerza un eslabón distinto del bucle.\footnote{%
    Todas las economías afectadas por la crisis asiática presentaban un alto nivel de apalancamiento durante las décadas anteriores a la crisis. Además, en los años noventa, con el proceso de apertura de la cuenta financiera, empezaron también a endeudarse en moneda extranjera.
} Esta situación puede ilustrarse por medio de dos gráficos complementarios.

\imagenfit{3era_Gen_equilibrios.pdf}{Equilibrios del modelo de balance}{Elaboración propia a partir de KRUGMAN (1999). Valores ilustrativos.}

El panel A reúne los tres eslabones en el plano formado por la inversión y el tipo de cambio real. La recta $GG$ es el mercado de bienes: cuanto menor es la inversión, más depreciado está el tipo de cambio real. La curva $FF$ es la restricción de crédito: indica cuánta inversión puede financiarse para cada tipo de cambio real y presenta tres tramos. Con un tipo de cambio real apreciado, los balances son sólidos y la restricción no opera: la inversión es la máxima rentable, $i^{\max}$ (tramo vertical inferior). Con un tipo de cambio muy depreciado, la riqueza de los empresarios es negativa: quiebran y la inversión es nula (tramo vertical superior). Entre ambos, la restricción es operante y la inversión cae al depreciarse el tipo de cambio. Si $\lambda>1$, la recta $GG$ es más inclinada que ese tramo central y corta a $FF$ tres veces.

El panel B traduce lo mismo al lenguaje del bucle: en el eje horizontal, la inversión que los prestamistas esperan; en el vertical, la que efectivamente puede financiarse con esas expectativas. Los equilibrios son los puntos en que ambas coinciden, es decir, los cortes con la bisectriz. En el tramo en que la restricción opera, la curva tiene pendiente $\lambda$; si es mayor que la unidad, corta la bisectriz desde abajo y aparecen tres equilibrios. Estas tres regiones generan tres puntos de equilibrio del modelo, en los que la inversión real se iguala a la esperada: dos puntos extremos ($L$ y $H$), equilibrios estables, y un punto intermedio ($U$), inestable.

En primer lugar, el \textbf{equilibrio estable superior, $H$}, muestra que con un nivel elevado de inversión se invierte hasta el punto en el que se igualan el retorno de la inversión doméstica y extranjera. Es la situación de una economía en auge: entran capitales, el tipo de cambio real está apreciado, los balances son sólidos y la restricción de crédito no limita la inversión.

En segundo lugar, el \textbf{equilibrio estable inferior, $L$}, indica que los prestamistas no ofrecen financiación porque creen que los empresarios no disponen de colateral. Esto provoca una depreciación del tipo de cambio real que a su vez genera la quiebra de los empresarios, y confirma así la desconfianza inicial. Es lo que se observa en una crisis de este tipo: frenazo de las entradas de capital, depreciación brusca, quiebras empresariales y caída de la inversión y de la producción.

Por último, en el \textbf{equilibrio inestable, $U$}, si la inversión esperada por cualquier motivo se desplazase ligeramente hacia la derecha o hacia la izquierda, la inversión real se modificaría hasta alcanzar el punto $H$ o $L$ respectivamente. Las escaleras del panel B recorren el bucle vuelta a vuelta: a la izquierda de $U$, cada revisión de expectativas reduce la inversión financiable más de lo esperado y la economía desciende hasta $L$; a la derecha, la recuperación se autoalimenta hasta $H$. El punto $U$ funciona, así, como un umbral de confianza.

Nótese que los puntos de equilibrio estables son únicamente locales. Desde el equilibrio superior, si la inversión esperada se desplazase hasta un punto a la izquierda de $U$, la dinámica del modelo conduciría a un nuevo equilibrio en $L$, y viceversa. De esta forma, si por cualquier motivo los prestamistas se vuelven pesimistas, podría producirse el colapso de la economía. La razón no es que la inversión anterior fuese poco sólida, sino la fragilidad financiera.

\subsubsection{Implicaciones de Política Económica}
En definitiva, hemos visto como las crisis enmarcadas en estos modelos se caracterizan por ser mucho menos predecibles que las anteriores. Además, como se puede intuir, los efectos derivados de crisis gemelas son de mayor magnitud que los de crisis segmentadas o separadas. Por tanto, ¿qué podemos hacer para evitar esta secuencia destructiva? La respuesta tiene dos tiempos: reducir la fragilidad antes de la crisis y saber reaccionar cuando estalla.

\paragraph{Anticipación: Prudencia macroeconómica}
Lo primero y, evidentemente, más importante, sería identificar el riesgo del sistema: ¿dónde se está generando?

Hemos visto que el multiplicador financiero recoge los factores que alimentan el bucle, por tanto puede ser un buen lugar por donde empezar. Para preservar, en la medida de lo posible, la estabilidad del sistema, la política económica debe tratar de mantener ese multiplicador por debajo de la unidad. Las palancas son dos de sus tres componentes. Por un lado, la deuda en moneda extranjera en relación con las exportaciones ($b^f_t/X_t$): conviene limitar el endeudamiento en divisas y, a más largo plazo, ampliar la base exportadora (abriendo otros mercados), que es la que permite hacer frente a esa deuda. Por otro lado, el apalancamiento ($\theta$): mejorar los canales de supervisión y control bancarios, de manera que el grado de apalancamiento se mantenga siempre en niveles prudentes. El tercer componente, la propensión a importar, no es un instrumento razonable: forzar más importaciones reduciría el multiplicador sobre el papel, pero a costa de debilitar la demanda de producción nacional sin atacar la raíz del problema, que son los balances.

\paragraph{Descalces de moneda y pecado original}
El origen último de la fragilidad es que las empresas y los bancos tienen ingresos en moneda nacional y deudas en moneda extranjera. ¿Por qué se endeudan así? EICHENGREEN y HAUSMANN (1999) acuñaron el término \textbf{pecado original} para describir la incapacidad de la mayoría de las economías emergentes para endeudarse en el exterior en su propia moneda, e incluso para endeudarse a largo plazo en su propio mercado. Si es así, el descalce no es fruto de la imprudencia de cada empresa, sino una restricción que afecta a todo el país, y la prudencia individual no basta. De ahí dos líneas de actuación: limitar y vigilar los descalces de moneda en los balances de bancos y empresas, y desarrollar mercados de bonos en moneda local que permitan financiarse sin asumir riesgo cambiario. Tras la crisis asiática, los países de la región lanzaron con ese fin la Iniciativa de Mercados de Bonos Asiáticos (2003)\textcolor{orange}{\textsuperscript{[Probable]}}.

\paragraph{Miedo a flotar y acumulación de reservas}
Los efectos balance explican un comportamiento que, a primera vista, contradice la visión bipolar de los regímenes cambiarios. CALVO y REINHART (2002) observaron que muchos países que declaraban un tipo de cambio flexible intervenían intensamente para evitar que se moviera: es el \textbf{miedo a flotar}. Con deudas en moneda extranjera, una depreciación fuerte no es una vía de ajuste, sino una amenaza para la solvencia del sector privado; la cautela ante el tipo de cambio es, en estas economías, el temor de un hombre sabio. La consecuencia práctica fue la acumulación masiva de reservas por parte de las economías emergentes, sobre todo asiáticas, tras 1997: un \textbf{autoseguro} que permite atender una salida de capitales sin depender del FMI ni de sus condiciones. Este seguro tiene, sin embargo, un coste de oportunidad (la diferencia entre el rendimiento de las reservas y el coste de la deuda) y efectos sobre el resto del mundo, pues contribuyó a los desequilibrios globales de la década de 2000\textcolor{orange}{\textsuperscript{[Probable]}}.

\paragraph{Gestión de los flujos de capital}
La crisis asiática erosionó el consenso de los años noventa en favor de la liberalización plena de la cuenta financiera. Si las entradas de capital a corto plazo alimentan los balances frágiles que luego se derrumban, limitar esas entradas es una forma de prevención. Esta idea, que se aplicó con la regulación de las entradas en Chile, acabó incorporándose a la doctrina oficial: en 2012 el FMI adoptó una ``visión institucional'' que acepta, en determinadas circunstancias, las medidas de gestión de los flujos de capital, como complemento y no como sustituto de la política macroeconómica y de la regulación macroprudencial\textcolor{orange}{\textsuperscript{[Probable]}}. La regulación macroprudencial (límites a la posición en divisas de los bancos, colchones de capital anticíclicos, límites al endeudamiento) actúa directamente sobre los dos componentes del multiplicador que pueden controlarse.

\paragraph{Reacción: ¿qué hacer?}
Ahora bien, imaginemos que nos damos cuenta de esta situación de fragilidad anticipando un colapso inminente. ¿Cómo deberíamos reaccionar ante un empeoramiento de las expectativas de inversión que produzca una leve (pero peligrosa) salida de capitales?

La solución tradicionalmente propuesta por el FMI consistía en defender el tipo de cambio a través de un aumento de los tipos de interés, evitando también el incremento del valor de los pasivos denominados en divisa como consecuencia de la depreciación. Sin embargo, tal y como expone KRUGMAN (1999), el aumento de los tipos de interés cerrará un canal de la propagación de la crisis pero abrirá uno nuevo vía disminución de la inversión, y por tanto de la producción y de la riqueza de los capitalistas. Por esta razón, otros autores como STIGLITZ (2000) también critican la estrategia del FMI. KRUGMAN propone en su lugar otras medidas como poner a disposición de los países afectados \textbf{grandes líneas de crédito de emergencia} o bien la imposición de \textbf{restricciones temporales a la salida de capitales}.

El problema de fondo es un dilema sin salida cómoda: con balances descalzados, dejar caer la moneda destruye la riqueza de las empresas, y defenderla con tipos de interés altos destruye su inversión y su producción. Malasia ofreció en septiembre de 1998 una vía alternativa: impuso controles a la salida de capitales y fijó el ringgit frente al dólar, lo que le permitió bajar los tipos de interés sin provocar una nueva depreciación. Su evaluación sigue siendo objeto de debate, aunque KAPLAN y RODRIK (2001) concluyen que la economía malaya se recuperó más rápido y con menos costes de lo que habría ocurrido con un programa del FMI\textcolor{orange}{\textsuperscript{[Probable]}}. En definitiva, una vez que la crisis ha estallado, todas las opciones son costosas, y por eso la prevención resulta decisiva.

\paragraph{Red de seguridad financiera global}
La lógica de la tercera generación es, en esencia, la de un pánico ante la falta de liquidez en moneda extranjera. Ninguna economía emergente puede crear dólares, de modo que la respuesta natural es un prestamista de última instancia internacional que los proporcione. Esa red se ha ido construyendo por capas: las reservas propias, los acuerdos regionales (la Iniciativa de Chiang Mai, lanzada en 2000 por los países de la ASEAN, China, Japón y Corea del Sur), las líneas de crédito preventivas del FMI (la Línea de Crédito Flexible, creada en 2009, para países con buenos fundamentos) y las líneas de intercambio de divisas (\textit{swaps}) entre bancos centrales, que la Reserva Federal extendió a varias economías avanzadas y emergentes durante la crisis de 2008\textcolor{orange}{\textsuperscript{[Probable]}}. Cuanto más creíble sea esta red, menos probable es que un pánico empuje a la economía al equilibrio malo.

\subsubsection{Limitaciones: un modelo para Asia}
Los modelos de tercera generación, y en particular el de balance, también tienen límites. En el plano analítico, el modelo de KRUGMAN es estático y muy estilizado: un único bien, exportaciones exógenas, ausencia de bancos y de Gobierno, y un tipo de cambio real que se mueve sin que intervenga ninguna política. Además, como en la segunda generación, la existencia de equilibrios múltiples deja sin explicar qué provoca el cambio de expectativas y cuándo: el modelo identifica la fragilidad, pero no el detonante. Las tres ramas, por otro lado, ofrecen explicaciones complementarias más que un marco unificado, de modo que su capacidad para orientar la política económica depende de identificar en cada caso cuál de los mecanismos domina.

En definitiva, desde una perspectiva empírica, los modelos de tercera generación son capaces de explicar (parcialmente) algunas de las últimas crisis financieras en los países emergentes, caracterizadas por grandes entradas de capital en el periodo inmediatamente anterior, complejas interacciones entre agentes, impredictibilidad y una importante ralentización del crecimiento económico en el periodo posterior.

No obstante, como todo, presentan grandes limitaciones para explicar las \textbf{crisis financieras} y las \textbf{crisis cambiarias}. Por ejemplo, aunque la crisis financiera de 2007-2008 podría presentar algunos rasgos que caracterizan este tipo de modelos (elevado endeudamiento, riesgo moral, deterioro de balance, salidas de capitales, etc.), las relaciones observadas, así como sus efectos, son mucho más complejas e imposibles de explicar a través de este marco. Aquella crisis nació en economías avanzadas, endeudadas en su propia moneda y con tipos de cambio flexibles; su epicentro no fue el balance de las empresas no financieras, sino un sistema financiero muy apalancado, con un sector bancario en la sombra y activos titulizados cuyo riesgo era difícil de valorar; y su dimensión internacional se manifestó a través de la escasez de financiación en dólares de los bancos europeos, que la Reserva Federal tuvo que cubrir con líneas de intercambio de divisas, más que a través de ataques contra un tipo de cambio fijo.

Por último, todos estos modelos son modelos de un solo país: explican por qué una economía concreta es frágil, pero no por qué la crisis tailandesa se extendió en pocos meses a Indonesia, Malasia, Filipinas y Corea del Sur, y después a Rusia y a Brasil, economías con fundamentos y vínculos muy distintos. Para entenderlo hay que estudiar cómo se transmiten las crisis de unos países a otros: el contagio.

\section{Contagio: ¿cómo se transmiten las crisis?}
\puntosclave{

}


Es por eso que la literatura reciente se ha centrado en el estudio de un fenómeno cada vez más recurrente, posiblemente más sencillo de modelar y con fuertes repercusiones desde un punto de vista internacional: muchos de los episodios de crisis monetarias y financieras recientes se han transmitido rápidamente entre países. ¿Por qué? Para responder a esta pregunta, la literatura académica se ha centrado en el estudio del fenómeno de contagio.

EICHENGREEN, ROSE y WYPLOSZ (1996) definen el contagio como la situación en la que \textbf{la crisis de un país incrementa la probabilidad de ocurrencia de crisis en otros países}. Por tanto, conforme a esta definición, hablaremos de contagio cuando la crisis de un país ponga en marcha una serie de mecanismos de transmisión que hagan que aumente por sí misma la probabilidad de crisis en otro país. Esto significa que no basta con la simultaneidad de las crisis, que podría venir dada por pura coincidencia o por la existencia de una causa común, sino que es necesario que exista un \textbf{vínculo de causalidad entre ellas}. Un ejemplo de causa común sería un cambio en la política monetaria de EEUU o en los precios internacionales de las materias primas que afecte a varios países de forma simultánea. Evidentemente, esta delimitación no es generalizada, ya que algunos autores como MASSON (1999) incluyen también la aparición de crisis simultáneas en varios países por una causa común en el concepto de contagio. Pero es un buen punto de partida. 

\textbf{¿Qué queremos decir?} A lo largo de esta sección vamos a intentar responder, brevemente, a la siguiente pregunta: ¿cómo se transmite esta adversidad y cómo podemos reducir nuestra exposición? Para hacerlo, recurriremos a las aportaciones de KAMINSKY y REINHART (2000), que distinguen dos tipos de contagio: (i) \textbf{basado en fundamentos}, cuando el país en crisis tiene vínculos comerciales o financieros con otros; y, (ii) \textbf{puro}, cuando no existen interconexiones entre los países sino que el contagio se produce por la aparición de efectos de tipo rebaño entre los inversores ante el estallido de la crisis en uno o varios países. Veamos qué sabemos de cada uno de ellos. 

\subsection{Fundamentos: ¿Qué lazos unen nuestras economías?}
Al hablar de contagio basado en fundamentos, podemos y debemos concretar el tipo de lazo que nos une lo suficiente con otra u otras economías como para que un shock adverso en alguna de estas acabe contagiándose vía ese lazo al nuestro. 

\subsubsection{Lazos comerciales: Competitividad}
El primer lazo que nos une con otros países es el comercial. A través de este canal, la forma en la que un shock se contagia presenta causas y efectos diferentes en función del origen del que traiga causa, pero el canal de transmisión será siempre el mismo: la \textbf{competitividad}. 

Este canal tiene, además, una particularidad: hace muy difícil distinguir entre una devaluación \textbf{forzosa} y una devaluación \textbf{deliberada}. Cuando un país devalúa porque ha sufrido un ataque especulativo, lo hace a la fuerza, porque no tiene reservas o porque no está dispuesto a soportar el coste de defender la paridad. Pero el efecto sobre sus socios es el mismo que si hubiera devaluado a propósito para abaratar sus exportaciones a costa de las de los demás, es decir, para aplicar una \textbf{política de empobrecimiento del vecino} (\textit{beggar-thy-neighbour}), como las devaluaciones competitivas de los años treinta. Desde fuera, ambas situaciones son indistinguibles, y el dilema de los socios también es el mismo: si no devalúan a su vez, pierden competitividad y, con ella, actividad y empleo. Por eso el contagio comercial funciona como una reacción en cadena: la devaluación de un país eleva el coste de mantener la paridad en los demás (en el lenguaje de la segunda generación, empeora sus fundamentos y amplía su zona de fragilidad), y los especuladores, que lo anticipan, atacan la siguiente moneda. GERLACH y SMETS (1995) formalizan este mecanismo: la devaluación de un país abarata sus productos frente a los de su socio comercial, empeora la balanza comercial de este y le hace perder reservas, lo que puede precipitar un ataque contra su moneda\textcolor{orange}{\textsuperscript{[Probable]}}. De ahí la importancia de la coordinación internacional para evitar carreras de devaluaciones, que fue uno de los motivos fundacionales del sistema de Bretton Woods y del FMI.

Imaginemos que una economía se ve forzada a devaluar su moneda, independientemente de la razón. ¿Cómo afectaría esto a nuestra economía? 

\paragraph{Lazos bilaterales: competitividad frente al socio}
Si este país mantiene una relación comercial muy importante con nosotros, la devaluación de su moneda afectará negativamente a nuestro sector exportador, porque nuestros productos se encarecen en relación con los suyos: nuestro tipo de cambio real frente a ese país se aprecia. Es decir, el encarecimiento relativo de sus importaciones procedentes de nuestro país disminuirá su demanda (y, en función de la duración, podría aumentar la intensidad de sus exportaciones hacia nosotros), por lo que podría forzar a su vez la devaluación de nuestra moneda. 

Esta situación debería sernos familiar, pues es la que se observó durante la crisis del SME. Los ataques especulativos a Reino Unido y la consiguiente devaluación de la libra esterlina precipitaron la devaluación de la libra irlandesa, en enero de 1993. De la misma forma, los ataques a la peseta española en 1992 y su posterior devaluación afectaron a Portugal por los estrechos lazos comerciales, provocando la devaluación del escudo. Una década después, la devaluación del peso argentino en 2002 seguida de una fuerte contracción económica, tuvo repercusiones económicas en el resto de países del bloque Mercosur (Uruguay, Brasil y Paraguay) ese mismo año. 

\paragraph{Lazos indirectos: Competencia}
Existen también otros lazos comerciales que nos unen a otros países. Por ejemplo, imaginemos que nuestro sector exportador compite con el sector exportador de ese país en un país tercero. La devaluación de su moneda podría tener efectos perjudiciales sobre nuestro comercio vía pérdida de competitividad. Lo que, en último término, podría forzarnos a devaluar nuestra moneda para no perder mercado. 

Sucesos similares se produjeron durante la crisis de la deuda latinoamericana de los años ochenta, compitiendo estos países por el mercado estadounidense, aunque en aquella crisis pesó más un shock común, la fuerte subida de los tipos de interés en EEUU. También durante la crisis asiática: la depreciación del yen entre 1995 y 1997 restó competitividad a las exportaciones del sudeste asiático y, ya en plena crisis, la depreciación del dólar taiwanés en octubre de 1997 aumentó la presión sobre el dólar de Hong Kong y el won coreano\textcolor{orange}{\textsuperscript{[Probable]}}. 

\subsubsection{Lazos financieros: inversores y acreedores comunes}
Además de los lazos comerciales, los lazos financieros pueden ser causa de fenómenos de contagio. De hecho, en la Unión Europea parece claro que el elevado grado de integración financiera ha amplificado enormemente estos efectos de contagio entre Estados miembros.

No obstante, aunque los efectos contagio generados por lazos financieros de carácter bilateral pueden estar presentes, son poco interesantes. Un ejemplo de contagio por vínculos bilaterales se produjo en la crisis argentina de 2002. Los argentinos retiraron precipitadamente sus depósitos en dólares mantenidos en Uruguay. Ante esta retirada, los bancos se vieron forzados a retirar reservas mantenidas en el Banco Central, disminuyendo a la mitad el valor de las reservas internacionales en su posesión. Además, tuvo un impacto muy importante en la economía real.\footnote{%
    Para más información: \href{https://www.lanacion.com.ar/politica/uruguay-fue-el-pais-mas-afectado-por-la-crisis-argentina-nid402468/}{\textit{Uruguay fue el país más afectado por la crisis argentina}. La Nación (2002)}
}

\paragraph{Prestamista común}
El efecto contagio más presente y que más problemas genera proviene del fenómeno \textbf{prestamista común}: vínculos financieros indirectos por inversores comunes. 

El riesgo de contagio emerge cuando existen inversores que, como diversifican internacionalmente sus carteras, cuando se encuentren expuestos a países en crisis, pueden cortar la financiación o reevaluar el riesgo del resto de sus activos. 

Por tanto, en situación límite, podría generarse la liquidación de posiciones y repatriación del capital en países que se perciban con características similares con el fin de garantizar la calidad de sus carteras de activos, pudiendo generar una crisis en esos países.

VAN RIJCKEGHEM y WEDER (2001) encuentran que el efecto prestamista común fue relevante en la transmisión de las crisis mexicana (en la que el prestamista común eran los bancos de EEUU), tailandesa (Japón) y rusa (Alemania). 

No obstante, otros autores, como SCHINASI y SMITH (2000), consideran este efecto como una decisión óptima de reajuste de cartera por parte de los agentes en respuesta a la crisis financiera en un país. Muestran cómo un shock adverso en la distribución del rendimiento de un activo podría conducir a la reducción óptima de la exposición en activos en otros países si la regla de gestión de la cartera así lo determina.

\subsection{Puro: ¿por qué se contagian economías sin vínculos?}  
La línea que defiende la optimalidad de esta recomposición de carteras tiene también una aproximación más general a los fenómenos de contagio. 

Algunos autores consideran que la propagación de shocks a través de cualquiera de los fundamentos es el resultado de una respuesta óptima ante un shock externo y, como tal, no debería considerarse siquiera el concepto de contagio por fundamentos particulares, sino solamente el denominado \textbf{contagio puro}. 

Ahora bien, ¿qué es el contagio puro? En vez de responder a esta cuestión de forma directa, vamos a comprenderla sensu contrario. 

Fijémonos que el contagio por fundamentos no permite explicar la transmisión de crisis entre países con vínculos económicos poco claros ni la magnitud de algunos fenómenos de contagio. Es decir, resulta razonable suponer que la intensidad del vínculo tiene que ser muy fuerte para que se produzca tal fenómeno. Sin embargo, muchas veces no es así. 

En esta línea, la investigación sobre el \textbf{contagio puro} busca demostrar cómo y por qué los cambios en las expectativas, incluso las no vinculadas a fundamentos, pueden generar comportamientos desestabilizadores de tipo rebaño, dando lugar a múltiples equilibrios y relaciones \textit{ad hoc} y esporádicas entre Estados, independientemente del origen e intensidad del lazo. 

Como vemos, este fenómeno es muy abstracto. Por ello, podemos analizar la diversidad de justificaciones teóricas mediante la distinción entre contagio generado por la \textbf{existencia de imperfecciones informativas} y contagio por \textbf{reacciones en cascada derivadas de conexiones financieras} entre países.

\subsubsection{Imperfecciones informativas: ¿qué ven los inversores?}
Las imperfecciones informativas dificultan la valoración de los fundamentos económicos, por lo que ante el estallido de la crisis en un país los agentes del mercado podrían reevaluar los fundamentos en otros y tomar decisiones de venta de activos o reversión de préstamos, aunque estos fundamentos se hayan mantenido inalterados. Esto puede producirse por toda una serie de causas no excluyentes entre sí.


\paragraph{Extracción de señales}
En primer lugar, puede haber problemas de \textbf{extracción de señales}. Cuando los agentes del mercado desconocen el impacto que la crisis en un país puede generar en otro, pueden formar expectativas sesgadas y asumir de forma errónea que existe una interdependencia fuerte entre ambos. También podrían considerar que países con rasgos similares se enfrentan a problemas similares y, por tanto, el shock sobre un país podría transmitirse al otro por una evaluación incorrecta de sus fundamentos e interrelaciones.


\paragraph{Llamada de atención (\textit{wake-up call})}
En segundo lugar, puede producirse lo que se conoce como \textbf{llamada de atención} (\textit{wake-up call}). Esto es, la crisis en un país podría conducir a los agentes a revisar la situación de países con rasgos similares y detectar problemas existentes que no habían identificado anteriormente. Por tanto, se trataría de una corrección eficiente de la evaluación de los riesgos.

Un ejemplo de llamada de atención podríamos encontrarlo en la crisis de deuda soberana que se experimentó dentro de la Unión Europea. La crisis de deuda griega que estalló en 2009 cuando el nuevo gobierno revisó al alza los datos de déficit público, disparó la preocupación de los mercados por la sostenibilidad de la deuda en otros países. De esta forma se experimentó un contagio a otros países periféricos, provocando elevados diferenciales de tipos por emisión.

\paragraph{Riesgo moral}
En tercer lugar, podría haber fenómenos de contagio como consecuencia de problemas de \textbf{riesgo moral}. En este sentido, la existencia de un sistema de garantías explícitas o implícitas conduce a los inversores a esperar que se produzca un rescate en caso de crisis, con lo que en sus decisiones ignorarán la debilidad de la economía del país mientras consideren que las reservas disponibles y la ayuda internacional potencial son capaces de cubrir las posibles pérdidas. Cuando la crisis en otro lugar conduzca a los inversores a revisar sus expectativas de asistencia, éstos podrían reajustar sus expectativas y contagiar (iniciar) la crisis.

\paragraph{Contagio político}
Por último, es posible que el contagio se inicie a conciencia y por voluntad de los Gobiernos, sin que medie ninguna imperfección informativa (\textbf{contagio político}; DRAZEN, 1999)\textcolor{orange}{\textsuperscript{[Probable]}}. Por ejemplo, entre países que mantengan un objetivo cambiario mutuo. Si un país rompe este acuerdo, el resto de países percibirán un menor coste político de abandonarlo, porque el compromiso colectivo ya está roto. Así, la devaluación por parte de uno de los miembros podría tener un efecto de contagio sobre el resto.

\subsubsection{Movimientos de capital: Efecto dominó/cascada}
Por otro lado, el contagio puro también puede justificarse recurriendo a la propagación potencial que ofrecen los mercados financieros. Este fenómeno puede deberse a diferentes causas.

\paragraph{Incumplimiento de contraparte}
En primer lugar, el \textbf{incumplimiento de contraparte: morosidad concentrada}. Una elevada exposición a deudores morosos podría provocar transmisiones transfronterizas si las pérdidas para los acreedores son suficientemente elevadas. Este efecto es especialmente fuerte en el caso de los acreedores bancarios debido a que operan con una ratio de capital sobre activos relativamente reducida. Durante la crisis de deuda de latinoamérica (1982), los préstamos morosos estaban ampliamente concentrados en un reducido número de entidades bancarias, factor que contribuyó a motivar la concesión de ayuda internacional con el fin de frenar el contagio por esta vía.

\paragraph{Reestructuración de carteras}
Por otro lado, también podría ser consecuencia de una \textbf{reestructuración de carteras: deshacer posiciones}. Una forma común pero más sutil, a través de la cual se puede observar un efecto dominó es observando el comportamiento de inversores multinacionales. Si estos se enfrentan al deterioro del valor de los activos que poseen en un mercado en crisis, pueden verse obligados a deshacer sus posiciones en otros mercados (muchas veces rentables) para compensar sus pérdidas o atender las peticiones de garantías de sus acreedores. El caso de LTCM es un buen ejemplo: sus pérdidas tras la suspensión de pagos rusa de 1998 le obligaron a deshacer posiciones en mercados que nada tenían que ver con Rusia. Esto puede producirse en masa si el deterioro de las expectativas se origina en un país considerado \textbf{modélico}, cuya solidez nadie cuestionaba, porque la sorpresa obliga a revisar todas las posiciones en mercados similares.

Un ejemplo de ambos canales se puede observar en la crisis de las hipotecas subprime. Los bancos europeos, a través de sus fondos de inversión, habían comprado deuda hipotecaria de EEUU, lo que trasladó dicha exposición a estas entidades. Esta exposición estuvo muy concentrada en un reducido número de entidades de gran importancia. La falta de diligencia y transparencia en su actuación y declaración de este deterioro, provocó una pérdida de confianza de los inversores, que no se fiaban de la exposición que declaraban.\footnote{%
    El desencadenante de esta pérdida de confianza fue que la entidad BNP Paribas, que había transmitido que su exposición era reducida\textcolor{orange}{\textsuperscript{[Probable]}}, congeló el 9 de agosto de 2007 tres fondos que invertían en títulos \textit{subprime}.
} Por tanto, tenemos dos efectos: (i) por un lado, el deterioro de los activos y la elevada concentración que restringen el crédito; y, (ii) la necesidad de reestructurar carteras, ejecutando activos \textbf{buenos} con el fin de sanear el balance. Sumado, esto provocó un colapso bursátil (deterioro patrimonial), un colapso del mercado interbancario (falta de credibilidad) y una consecuente crisis de liquidez. De esta forma, los problemas en el sector financiero estadounidense traspasaron las fronteras y tuvieron un impacto fuerte en la economía real. 

\subsection{Implicaciones de Política Económica}
Además de esta aproximación teórica, la literatura empírica ha tratado de contrastar la existencia de estos episodios debido a sus relevantes implicaciones de Política Económica. En particular, podría justificar el rescate por parte de instituciones multilaterales a países en apuros para prevenir el contagio a otros países con fundamentos sanos. De este modo se podría sortear las dificultades de ajuste que plantean los modelos de tercera generación. 

No obstante, resulta complicado distinguir entre contagio basado en vínculos comerciales o basado en vínculos financieros ya que los países con relaciones comerciales entre sí suelen tener también vínculos financieros. Uno de los primeros (EICHENGREEN, ROSE y WYPLOSZ, 1996), encontró que éste se propaga en mayor medida entre países con vínculos comerciales. Otros estudios posteriores otorgan un papel limitado al contagio, argumentando que los shocks simétricos y comunes parecen explicar con mejor ajuste la ocurrencia simultánea de crisis financieras. En particular, FORBES y RIGOBON (2002) muestran que las correlaciones entre mercados aumentan de forma mecánica cuando aumenta la volatilidad, de modo que, una vez corregido ese sesgo, buena parte de lo que parecía contagio es simple interdependencia: ``no hay contagio, solo interdependencia''. Esta distinción es también relevante por las recomendaciones/implicaciones de Política Económica, que podrían diferir en función del canal.\footnote{%
    Así, la provisión de liquidez podría ser recomendable únicamente para algunos casos de contagio puro. En caso contrario, la provisión de liquidez podría únicamente retrasar el necesario ajuste. Por otro lado, los shocks comunes y las interdependencias entre países deberían abordarse con políticas que busquen mejorar los fundamentos de la economía. 

De acuerdo con este razonamiento, las cuantiosas inyecciones de liquidez con las que se trató de contener la crisis financiera de 2008 estarían justificadas en la medida en que su transmisión respondiera al pánico y no solo al deterioro de los fundamentos\textcolor{orange}{\textsuperscript{[Probable]}}.
}

En general, sí se acepta que las imperfecciones informativas juegan un papel fundamental en los episodios de contagio. Por tanto, es muy recomendable mantener políticas transparentes que contribuyan a evitar estos fenómenos: estableciendo objetivos claros. Esto podría reducir la vulnerabilidad de los mercados. Por último, la cooperación y coordinación internacional, fijar normas y estándares de supervisión comunes o acometer intervenciones simultáneas en los mercados, podría reducir también el riesgo sistémico de crisis y fenómenos de contagio.

\anexo{Primera generación: Desarrollo analítico}\label{modelos_primera}

Este anexo recoge el desarrollo analítico completo del modelo de primera generación que se presenta en el cuerpo del tema. Lo que impulsó esta línea de investigación fue una regularidad empírica que los modelos de tipo de cambio de la época no explicaban: los regímenes de tipo de cambio fijo de varios países en desarrollo, sobre todo latinoamericanos, que financiaban sus déficits públicos con creación de dinero (México en 1976 y 1982, Argentina en 1981), no se desmoronaban de forma gradual a medida que perdían reservas, sino de golpe, mediante un ataque especulativo que agotaba las reservas restantes en muy poco tiempo, cuando todavía quedaba un colchón apreciable. ¿Por qué el colapso es brusco y no gradual? ¿Por qué se produce antes de que las reservas se agoten? ¿Y puede preverse su fecha?

KRUGMAN (1979) respondió a estas preguntas trasladando al tipo de cambio el análisis de SALANT y HENDERSON (1978) sobre los ataques especulativos contra un precio del oro fijado por el Gobierno, y FLOOD y GARBER (1984) lo reformularon en la versión lineal que aquí se desarrolla, que permite obtener la fecha del ataque de forma explícita. La conclusión es que el ataque no es un fenómeno irracional ni una manipulación del mercado, sino la consecuencia lógica, en un mundo con previsión perfecta, de una política económica incoherente. El desarrollo sigue cuatro pasos: (i) los supuestos y las ecuaciones funcionales del modelo; (ii) su reducción a una sola ecuación que liga la oferta monetaria con el tipo de cambio; (iii) la dinámica de las reservas bajo el tipo de cambio fijo; y, (iv) el tipo de cambio sombra y la fecha del ataque. A lo largo del anexo, $s$ denota el tipo de cambio (unidades de moneda nacional por unidad de divisa), $\bar s$ la paridad y $\tilde s$ el tipo de cambio sombra.


\textbf{Primera generación: FLOOD y GARBER (1984)}

FLOOD y GARBER (1984) desarrollan un modelo para una economía pequeña y abierta
sin imperfecciones.

Sobre esta base, el modelo descansa en los siguientes supuestos. En primer lugar, la economía es pequeña y abierta: toma como dados el nivel de precios exterior ($P^*$) y el tipo de interés exterior ($i^*$), que se suponen constantes. En segundo lugar, los precios son plenamente flexibles y la paridad de poder adquisitivo se cumple en todo momento, y la renta real ($Y$) se toma como constante, de modo que el modelo no tiene lado real: todo el ajuste recae en el tipo de cambio, en los precios y en el balance del banco central. En tercer lugar, los agentes económicos son capaces de hacer previsiones perfectas y pueden invertir en tres activos financieros: (i) el dinero nacional; (ii) los bonos nacionales; y, (iii) los bonos extranjeros. Los agentes no desean mantener dinero extranjero en su cartera porque es un activo dominado fuertemente por los otros tres, ya que no proporciona rendimiento nominal ni servicios monetarios a los agentes. Sin embargo, los bonos nacionales y extranjeros sí se consideran sustitutivos perfectos, y no hay costes de transacción ni controles de capital, de modo que se cumple la paridad no cubierta de intereses. En cuarto lugar, la política fiscal es exógena: el déficit público es exógeno y el banco central posee un stock de divisa extranjera para intervenir con el fin de mantener el tipo de cambio fijo, pero también se le exige monetizar parte del déficit público, lo que hace crecer el crédito interno a una tasa constante y positiva. Por último, el banco central defiende la paridad mientras le queden reservas; cuando estas alcanzan su límite inferior, abandona la paridad, deja de intervenir y el tipo de cambio pasa a flotar libremente de forma permanente, mientras el crédito interno sigue creciendo al mismo ritmo. A partir de estos supuestos, el modelo se construye sobre cuatro ecuaciones principales que podemos dividir en dos bloques: ecuaciones dinámicas y ecuaciones estáticas.

En primer lugar, las ecuaciones estáticas serían


$$\begin{aligned}
  \text{Estáticas}:
    &&\left\{\begin{aligned}
        &(1)\quad\text{Mercado Monetario}: && \frac{M(t)}{P(t)}=Y(t)-k\cdot i(t)\,\,>0\\[1em]
        &(2)\quad\text{Condición PPA}: &&P(t)=P^*(t)\cdot s(t)
    \end{aligned}\right.
\end{aligned}$$

Por tanto, la ecuación (1) iguala la oferta real de dinero con su demanda, que es una función lineal de la renta y del tipo de interés nominal: el parámetro $k>0$ mide cuánto se reduce la demanda de saldos reales cuando sube el coste de oportunidad de mantener dinero, y la restricción ``$>0$'' exige que la demanda real de dinero sea positiva. La ecuación (2) es la paridad de poder adquisitivo en su versión absoluta: el arbitraje de bienes hace que el nivel de precios interno sea el exterior multiplicado por el tipo de cambio. Para cerrar el bloque estático falta definir los dos elementos que aparecen en él y que conectan con las ecuaciones dinámicas: la composición de la oferta monetaria y la variable cambiaria. Analíticamente:


$$\text{donde,}
    \quad\left\{
    \begin{aligned}
        &\underset{\text{\scriptsize reservas internacionales $+$ crédito interno}}{M(t)=R(t)+D(t)}\\[0.5em]
        &s(t)\equiv \text{Tipo de Cambio spot}
    \end{aligned}\right.$$

Es decir, la oferta monetaria es el pasivo del banco central, y su contrapartida en el activo son las reservas internacionales ($R$, valoradas en moneda nacional) y el crédito interno ($D$), que incluye la financiación concedida al sector público. El tipo de cambio $s$ se expresa en unidades de moneda nacional por unidad de moneda extranjera, de modo que un aumento de $s$ es una depreciación. Esta descomposición es la clave del modelo: el banco central puede controlar el crédito interno, pero, con el tipo de cambio fijo, no la cantidad total de dinero, de modo que lo que crea por la vía del crédito lo pierde por la vía de las reservas.

Por otro lado, las ecuaciones dinámicas describen cómo evolucionan en el tiempo los dos lados del modelo: la oferta de crédito interno y el rendimiento de los activos.

$$\begin{aligned}
  \text{Dinámicas}:
    &&\left\{\begin{aligned}
        &(3)\quad\underset{\text{\scriptsize señoreaje: crecimiento exógeno}}{\text{Crecimiento déficit}:}&& \dot D(t)=\mu\\[1em]
        &(4)\quad\text{Condición de PNCI}: &&i(t)=i^*(t)+\left[\frac{\dot s(t)}{s(t)}\right]
    \end{aligned}\right.
\end{aligned}$$

Es decir, la ecuación (3) recoge la monetización del déficit: el crédito interno crece cada periodo en una cantidad constante $\mu>0$, fijada al margen de la situación cambiaria, porque el banco central debe financiar al Tesoro. La ecuación (4) es la paridad no cubierta de intereses con previsión perfecta: como los bonos nacionales y extranjeros son sustitutivos perfectos, solo pueden coexistir en las carteras si ofrecen el mismo rendimiento expresado en una misma moneda, de modo que el tipo de interés interno es igual al exterior más la tasa de depreciación prevista, que, con previsión perfecta, coincide con la efectiva.

En esencia, manteniendo siempre el equilibrio en el mercado de dinero, se cumple la paridad de poder adquisitivo mediante la fijación del tipo de cambio spot, que fluctuará de manera que una unidad monetaria tenga la misma capacidad adquisitiva vía arbitraje de bienes, con independencia de su moneda de denominación. Las ecuaciones dinámicas inciden en ambos lados de la oferta monetaria pues expresan (3) la tasa de crecimiento de la oferta monetaria (inducida por la necesidad de generar ingresos por señoreaje) y (4) el tipo de interés nacional en base a la paridad no cubierta de intereses: el mercado cambiario estará en equilibrio cuando los depósitos de todas las divisas ofrezcan idéntica tasa de rentabilidad.

Con estas ecuaciones de partida, debemos ahora reducir el modelo a una sola ecuación. El motivo es que el nivel de precios y el tipo de interés interno son variables endógenas que, en una economía pequeña y abierta con plena movilidad de capitales, quedan determinadas desde el exterior por las dos condiciones de arbitraje: la PPA fija $P$ a partir de $P^*$ y de $s$, y la PNCI fija $i$ a partir de $i^*$ y de la depreciación esperada. Si sustituimos ambas condiciones en el equilibrio del mercado monetario, eliminamos $P$ e $i$ y obtenemos una relación entre las dos variables que interesan para analizar la crisis: la oferta monetaria (y, por tanto, las reservas) y la trayectoria del tipo de cambio. Sustituimos, pues, la PPA (2) y la PNCI (4) en el equilibrio del mercado monetario (1). Analíticamente, obtenemos:

$$\begin{aligned}
  &(2)+(4)\Rightarrow(1):&&\frac{M(t)}{P^*(t)\cdot s(t)}=Y(t)-k\cdot \left(i^*(t)+\frac{\dot s(t)}{s(t)}\right)
\end{aligned}$$

Si multiplicamos ambos lados por $P^*(t)\cdot s(t)$ y agrupamos los términos en $s(t)$ y en $\dot s(t)$, obtendremos:

$$\begin{aligned}
  &\underset{\text{\scriptsize (operando y despejando)}}{\Longrightarrow}&&M(t)=(\underbrace{Y(t)\, P^*(t)-k\, P^*(t)\,i^*(t)}_{\equiv\,\beta})\cdot s(t)-\underbrace{k\,P^*(t)}_{\equiv\,\alpha}\,\dot s(t)\\[1em]
\end{aligned}$$

Es decir, la oferta monetaria que el público está dispuesto a mantener depende positivamente del nivel del tipo de cambio, que eleva el nivel de precios, y negativamente de su variación, que eleva el tipo de interés interno. Si, además, sustituimos esas dos agrupaciones de parámetros por $\beta$ y $\alpha$, obtenemos la forma reducida del modelo de FLOOD y GARBER (1984):

$$\begin{aligned}
  &\underset{\text{\scriptsize (sustituyendo)}}{\Longrightarrow}&&\boxed{M(t)=\beta\,s(t)-\alpha\,\dot s(t)}
\end{aligned}$$

Es decir, una única ecuación diferencial que liga la oferta monetaria con el tipo de cambio y su variación, válida tanto en el régimen de tipo fijo como en el de flotación. Dado que $P^*(t)$ e $i^*(t)$ pueden suponerse dados para un país pequeño y la renta es constante, $\beta$ y $\alpha$ son constantes. $\beta=P^*(Y-k\,i^*)$ es positivo, porque $Y-k\,i^*$ es la demanda real de dinero cuando no se espera depreciación, que la restricción de la ecuación (1) exige positiva; $\alpha=k\,P^*$ es positivo porque $k$ lo es. Y, como en el régimen de tipo de cambio fijo $s(t)=\bar s$ en todo momento, la depreciación es nula ($\dot s(t)=0$), y obtenemos la ecuación:

$$\begin{aligned}
  \quad\underset{\substack{\text{\scriptsize como TC fijo,}\\ s(t)=\bar s,\ \dot s(t)=0}}{\Longrightarrow}\quad\boxed{M(t)=\beta\,\bar s}
\end{aligned}$$

Es decir, mientras se mantiene la paridad, la demanda de dinero es constante, y el equilibrio del mercado monetario exige que la oferta también lo sea. El banco central no puede alterar la cantidad total de dinero sin alterar el tipo de cambio: cualquier exceso de oferta lleva al público a cambiar moneda nacional por divisas, y el banco central, para defender la paridad, tiene que atender esa demanda con sus reservas.

Por tanto, como la oferta monetaria es constante e igual a $\beta\bar s$ y, por definición, es la suma de las reservas internacionales y del crédito interno, las reservas son lo que queda de esa oferta monetaria fija una vez descontado el crédito interno:

$$\begin{aligned}
  &M(t)=R(t)+D(t)=\beta\,\bar s&&\Longrightarrow\quad R(t)=\underset{\text{\scriptsize cte.}}{\beta\,\bar s}-D(t)
\end{aligned}$$

Luego, derivando respecto al tiempo y utilizando la ecuación (3), la variación de las reservas es exactamente la contraria a la del crédito interno, y las reservas caen de forma lineal desde su nivel inicial $R(0)=\beta\bar s-D(0)$:

$$\begin{aligned}
  \dot R(t)=-\dot D(t)&&\Longrightarrow\quad\boxed{\dot R(t)=-\mu}\quad\Longrightarrow\quad R(t)=R(0)-\mu\,t
\end{aligned}$$

Por tanto, vemos que, para mantener el tipo de cambio constante a la vez que se monetiza el déficit público, es necesario que los activos externos en el balance del Banco Central se reduzcan a la misma tasa a la que aumentan los activos públicos, de forma que la oferta monetaria no varíe.

Con una tasa de crecimiento ($\mu$) exógena y constante y unas reservas internacionales finitas, el sistema de tipo de cambio fijo no podrá sobrevivir indefinidamente: \textbf{cualquier stock de reservas internacionales mantenido para garantizar el sistema cambiario fijo acabará agotándose}. Si nada lo interrumpiese, las reservas se agotarían en el momento $t=R(0)/\mu$.

En este caso, suponemos que el límite inferior de reservas internacionales es cero, pero los resultados serían similares si se supusiese que ese límite inferior viene dado por cualquier otro valor positivo o negativo.


Los agentes privados, que son capaces de prever de forma perfecta que las políticas del gobierno conducirán inexorablemente al agotamiento de las reservas internacionales, acometerán de forma racional un ataque especulativo mediante la compra masiva de bonos extranjeros a cambio de moneda nacional antes de que las reservas internacionales se agoten para evitar pérdidas por sus tenencias en moneda nacional, que precipita la crisis del sistema cambiario.

El ataque se producirá antes de que se agoten las reservas internacionales, ya que en caso contrario habría un salto discreto y perfectamente anticipado en el tipo de cambio que supondría una tasa de ganancia instantáneamente infinita.

Pero, ¿cuándo cabe esperar este tipo de comportamiento por parte de los agentes? ¿podemos anticipar la crisis cambiaria? Para esto se define el \textbf{tipo de cambio sombra} ($\tilde s$) como el tipo de cambio que prevalecería si se pasase a un régimen de tipo de cambio flexible.

Podemos visualizarlo gráficamente de la siguiente forma (en la figura, $\bar e$ y $\tilde e$ corresponden a $\bar s$ y $\tilde s$):

\imagenfit{1era_Gen.png}{}{Tema 3.B.17. Apuntes ICEX-CECO.}

El modelo anticipa que el stock de reservas se agotará progresivamente en el tiempo ($t$) por la necesidad de financiar el déficit. Si el tipo de cambio fuese flexible, las reservas dejarían de variar y la expansión del crédito interno se trasladaría íntegramente a la oferta monetaria y, por tanto, al tipo de cambio, que se depreciaría (un aumento es una depreciación). Al suponer una tasa de monetización constante, el tipo de cambio sombra se representa como una recta de pendiente igual a $\mu/\beta$ en función del tiempo, como demostraremos a continuación.

¿Cuándo se produciría dicho ataque? Existen diversas formas de aproximarlo. Siguiendo a FLOOD y GARBER (1984), podemos considerar que dicho ataque se puede aproximar por la \textbf{condición de no arbitraje}. Con ayuda del gráfico, podemos ver como existen diferentes regiones.

En primer lugar, una \textbf{región estable: $\tilde s <\bar s$}. Si el régimen fijo colapsara en este momento, el mercado llevaría el tipo de cambio a un nivel menor que el tipo fijo actual: se apreciaría la moneda nacional y no existen incentivos económicos para el ataque especulativo. ¿Por qué? Supongamos que $\bar s=10$ y $\tilde s=8$. El Banco Central vende una unidad de divisa por diez unidades de moneda nacional. Si compramos divisa extranjera (reservas) a ese precio, tras el colapso la paridad se fijará en $\tilde s=8$. Esto implica que ahora obtendremos dos unidades monetarias nacionales menos por cada unidad de divisa que tengamos, ergo generaremos una pérdida por la transacción. Esta generación de pérdidas por parte de los inversores elimina los incentivos de iniciar un ataque especulativo y, en consecuencia, nos sitúa en la región estable del gráfico.

Por otro lado, existe una \textbf{región ficticia: $\tilde s>\bar s$}. Si nos encontramos en esta región del gráfico y ocurre el colapso del sistema cambiario, el tipo fijo saltaría a un nivel más alto: se depreciaría la moneda nacional y existirían incentivos económicos para el ataque especulativo. ¿Por qué? Suponiendo que el tipo está en $\bar s=10$ y el tipo sombra en $\tilde s=15$, comprando divisa por moneda nacional generaríamos una ganancia tras el colapso de la paridad. Por cada unidad de divisa obtendríamos cinco unidades adicionales de moneda nacional de la que pagamos por obtener la divisa en primer lugar, ergo la moneda se ha depreciado. Esta oportunidad de arbitraje genera un incentivo constante para el ataque especulativo. Sin embargo, la imposibilidad de que existan dichas oportunidades hace que esta situación sea virtualmente imposible.

Por último, existe un \textbf{equilibrio especulativo: $\tilde s=\bar s$}. Lo que nos conduce a la única situación en la que se genera un incentivo para el ataque compatible con los supuestos del modelo. ¿Qué pasaría si ambos tipos de cambio fuesen iguales? Este es el momento en el que se produce el ataque especulativo en equilibrio. La razón es de teoría de arbitraje bajo previsión perfecta: si los especuladores esperaran un poco más allá de $T$ existirían beneficios positivos, pero entonces cada especulador tendría incentivo a adelantarse ligeramente a los demás para capturar primero las reservas del banco central. Esa competencia hace imposible que el sistema sobreviva cuando ya existen beneficios esperados y, en consecuencia, el ataque ocurre exactamente cuando el beneficio esperado pasa de negativo (primera situación, $\tilde s<\bar s$) a cero ($\tilde s=\bar s$).

Ahora bien, ¿cómo podemos conocer el momento exacto $T$? Dado que tras el ataque se agotan las reservas internacionales, podemos conocer el momento del ataque en tres pasos: (i) calcular la trayectoria que seguiría el tipo de cambio en flotación, una vez agotadas las reservas, es decir, el tipo sombra; (ii) igualarla a la paridad, aplicando la condición de no arbitraje; y, (iii) despejar la fecha. Para el primer paso, tras el ataque las reservas son nulas, la oferta monetaria coincide con el crédito interno ($M(t)=D(t)$) y el tipo de cambio flota, de modo que la forma reducida se convierte en una ecuación diferencial en el tipo sombra. Como el crédito interno crece de forma lineal, conjeturamos que el tipo sombra es una función lineal del crédito interno (método de coeficientes indeterminados). Analíticamente, sea:

$$\begin{aligned}
  &D(t)=\beta\,\tilde s(t)-\alpha \,\dot{\tilde s}(t)\qquad \text{donde,}\,\,\,\tilde s(t)=\lambda_0+\lambda_1\,D(t)\quad\Longrightarrow\quad\dot{\tilde s}(t)=\lambda_1\,\dot D(t)=\lambda_1\,\mu
\end{aligned}$$

Es decir, si el tipo sombra es lineal en el crédito interno, su variación es constante e igual a $\lambda_1\mu$: en flotación, la moneda se deprecia a un ritmo constante. Entonces, tenemos que, sustituyendo la conjetura en la forma reducida, $D(t)=\beta\lambda_0+\beta\lambda_1 D(t)-\alpha\lambda_1\mu$, que debe cumplirse para cualquier valor de $D(t)$. Igualando los coeficientes de $D(t)$ y los términos constantes a ambos lados ($\beta\lambda_1=1$ y $\beta\lambda_0-\alpha\lambda_1\mu=0$), se identifican los dos coeficientes. Analíticamente:

$$\begin{aligned}
  &\left\{\begin{aligned}
        &\lambda_0=\frac{\alpha\,\mu}{\beta^2}\\[0.5em]
        &\lambda_1=\frac{1}{\beta}
    \end{aligned}\right\}\Longrightarrow\quad \tilde s(t)=\frac{\alpha\,\mu}{\beta^2}+\frac{D(t)}{\beta}
\end{aligned}$$

Es decir, cada unidad de crédito interno deprecia el tipo sombra en $1/\beta$, porque $\beta$ es la demanda de dinero por unidad de tipo de cambio; y el término constante $\alpha\mu/\beta^2$ recoge el efecto de la depreciación esperada de $\mu/\beta$ que, a través de la PNCI, eleva el tipo de interés, reduce la demanda de dinero en $\alpha\mu/\beta$ y exige un tipo de cambio más alto para equilibrar el mercado monetario.\footnote{%
    La solución general de la ecuación diferencial incluye, además, un término de la forma $C\,e^{(\beta/\alpha)\,t}$, que representa una burbuja especulativa: el tipo de cambio se depreciaría de forma explosiva sin relación con los fundamentos. Siguiendo la práctica habitual, se supone $C=0$ y se toma solo la solución ligada a los fundamentos.
} Teniendo en cuenta que el crédito interno crece de forma lineal desde su nivel inicial, $D(t)=D(0)+\mu\,t$ (ecuación (3)), entonces tendríamos:

$$\begin{aligned}
  \left(\substack{\text{\scriptsize sustituyendo}\\[0.5em]\\ D(t)=D(0)+\mu\,t}\right)\Longrightarrow \tilde s(t)=\frac{\alpha\,\mu}{\beta^2}+\frac{D(0)+\mu\,t}{\beta}\\[1em]
\end{aligned}$$

En definitiva, tenemos que el tipo sombra es una recta creciente en el tiempo, con ordenada en el origen $\tilde s(0)=\alpha\mu/\beta^2+D(0)/\beta$ y pendiente $\mu/\beta$, que es la que aparece en la figura. Por tanto, podemos describir analíticamente el momento del colapso como aquel en que el tipo sombra alcanza la paridad, $\tilde s(T)=\bar s$. Es decir:
$$\begin{aligned}
    {\bar s=\frac{\alpha\,\mu}{\beta^2}+\frac{D(0)+\mu\,T}{\beta}}\quad\Longrightarrow\quad{T=\frac{\bar s\,\beta-D(0)}{\mu}-\frac{\alpha}{\beta}}
\end{aligned}$$

Dicho de otra forma, multiplicando la condición por $\beta$ se obtiene $\beta\bar s-D(T)=\alpha\mu/\beta$: el ataque se produce cuando las reservas que quedan, $\beta\bar s-D(T)$, son justo las necesarias para absorber la caída de la demanda de dinero que provoca la depreciación esperada ($\alpha\mu/\beta$). Por último, como antes del ataque la oferta monetaria es $\beta\bar s=R(0)+D(0)$, el numerador del primer término son las reservas iniciales, por tanto:

$$\begin{aligned}
  &\quad\underset{R(0)\equiv\beta\,\bar s-D(0)}{\Longrightarrow}\quad\boxed{T=\frac{R(0)}{\mu}-\frac{\alpha}{\beta}}
\end{aligned}$$

Como vemos, el colapso se producirá en el momento en el que el tipo de cambio fijo se iguale con el tipo de cambio sombra ya que, en ausencia de ataque especulativo las reservas se agotarían en el momento $R(0)/\mu$ y $\alpha/\beta$ es el plazo de tiempo en el que se adelanta el colapso por la actuación de los agentes racionales. En esencia, el ataque especulativo se dilatará más o menos en el tiempo en función de: (i) el nivel inicial de reservas internacionales; (ii) cuanto más depreciado se encuentre el tipo de cambio fijo de la moneda doméstica que se esté defendiendo; y, (iii) cuanto menor sea la tasa de crecimiento del crédito doméstico.

Estos efectos, junto con el de la sensibilidad de la demanda de dinero al tipo de interés, se obtienen directamente derivando la fecha del colapso:

$$\frac{\partial T}{\partial R(0)}=\frac{1}{\mu}>0,\qquad \frac{\partial T}{\partial \bar s}=\frac{\beta}{\mu}>0,\qquad \frac{\partial T}{\partial \mu}=-\frac{R(0)}{\mu^2}<0,\qquad \frac{\partial T}{\partial \alpha}=-\frac{1}{\beta}<0$$

Por último, el volumen del ataque se obtiene evaluando las reservas justo antes del colapso, $R(T^-)=R(0)-\mu\,T=\alpha\mu/\beta$. Ese mismo resultado se obtiene desde la demanda de dinero. Justo antes del ataque no se espera depreciación y $M=\beta\bar s$; justo después, el tipo de cambio sigue siendo $\bar s$ (el ataque no provoca salto en el precio), pero ya se espera una depreciación de $\mu/\beta$, el tipo de interés interno sube de $i^*$ a $i^*+\mu/(\beta\bar s)$ y la demanda de dinero cae a $\beta\bar s-\alpha\mu/\beta$. La diferencia, $\alpha\mu/\beta$, es la cantidad de dinero que el público devuelve al banco central a cambio de sus últimas reservas. Si las reservas iniciales no alcanzan esa cuantía ($R(0)\leq\alpha\mu/\beta$), $T\leq 0$ y el ataque es inmediato.

\anexo{Segunda generación: Desarrollo analítico}\label{anx:modelo_segunda}

Este anexo recoge el desarrollo analítico completo del modelo de segunda generación que se presenta en el cuerpo del tema. Desde una perspectiva empírica, lo que impulsó esta línea de investigación fueron las crisis de los años noventa en países que no encajaban en la primera generación. En la crisis del SME de 1992-1993, la libra esterlina y la lira abandonaron el Mecanismo de Tipos de Cambio en septiembre de 1992, la peseta y el escudo se devaluaron en varias ocasiones y, en agosto de 1993, las bandas de fluctuación se ampliaron al $\pm 15\%$. Los países atacados no monetizaban sus déficits ni perdían reservas de forma gradual, y algunos, como el Reino Unido, tenían medios para defender la paridad si hubieran querido hacerlo. La crisis mexicana de 1994 planteó un problema parecido. La pregunta dejó de ser cuándo se agotan las reservas y pasó a ser por qué un Gobierno que puede defender su tipo de cambio decide no hacerlo, y por qué los mercados atacan a monedas cuyos fundamentos no parecen insostenibles.

Desde una perspectiva analítica, la respuesta exigía abandonar el supuesto de un Gobierno mecánico, que sigue una regla fija hasta agotar sus reservas, y sustituirlo por un Gobierno que optimiza: compara el coste de mantener la paridad, en términos de desempleo, con el de abandonarla, y conserva una cláusula de escape. En cuanto el Gobierno decide en función de lo que esperan los agentes, y los agentes forman sus expectativas en función de lo que creen que decidirá el Gobierno, aparece la posibilidad de equilibrios múltiples y de crisis autocumplidas (OBSTFELD, 1994, 1996). Esta posibilidad abrió además un debate empírico, aún no cerrado, sobre si las crisis responden a los fundamentos o a cambios de expectativas sin relación con ellos (JEANNE, 1997)\textcolor{orange}{\textsuperscript{[Probable]}}. El desarrollo sigue la versión simplificada de JEANNE (2000) y consta de tres pasos: (i) la función de pérdidas del Gobierno y la curva de Phillips; (ii) la reacción óptima del Gobierno ante cada expectativa posible de los agentes; y, (iii) las regiones de equilibrio y la amplitud de la región de fragilidad.

Supongamos una economía pequeña, abierta, con un objetivo cambiario fijo declarado y que se cumple la condición de Paridad de Poder Adquisitivo. Asimismo, consideraremos un escenario biperiodo en el que las autoridades mantienen inicialmente la paridad cambiaria y pueden decidir si devaluar la moneda o no.

De esta forma, introducimos la principal diferencia con los modelos de primera generación: su compromiso cambiario es ahora estado-dependiente, pueden ejercer una \textbf{cláusula de escape}. En este contexto, la autoridad evaluará de forma continua la idoneidad de mantenerlo, por medio de un análisis coste-beneficio.

No obstante, para simplificar el análisis, consideraremos que esta decisión se toma únicamente en función de sus efectos sobre el desempleo: el \textbf{objetivo interno contrapuesto}. Por tanto, este objetivo puede representarse mediante la minimización de una función de pérdidas con la siguiente forma (modelo de cláusula de escape, versión con pérdida cuadrática y coste fijo de la devaluación):

$$\begin{aligned}
  \text{Objetivo}:&&\min \,\,L_t=(u_t)^2+\delta C\\[1em]
  \text{donde},&&u_t=\underset{\text{\scriptsize (persistencia)}}{\rho\,u_{t-1}}-\underset{\text{\scriptsize (rigidez salarial)}}{\alpha(\pi_t-\mathbb E_{t-1}[\pi_t])}\\[2em]
\end{aligned}$$

Como vemos, las variables que componen la función son sencillas de interpretar: $u_{t-1}$ y $u_t$ son el desempleo de cada periodo (o sus desviaciones respecto a la tasa natural); $\pi_t$, la inflación; y, $\mathbb E_{t-1}[\pi_t]$, la inflación esperada. Nótese que $u_t$ es la tasa de desempleo en el segundo periodo y viene determinada por una Curva de PHILLIPS aumentada por las expectativas. Esta relación puede justificarse por la existencia de cierta rigidez salarial que implique que los salarios nominales del segundo periodo se determinen en el primer periodo, de forma que aumentos no esperados de la inflación puedan permitir situar la tasa de desempleo por debajo de su nivel natural en el corto plazo por medio de una reducción de los salarios reales y por tanto un aumento del empleo. Los parámetros son positivos: $\rho$ mide la persistencia del desempleo y $\alpha$, la eficacia de la sorpresa inflacionaria.

Nótese que: (i) $\delta=\{0,1\}$ es una variable ficticia que recoge la decisión de las autoridades de devaluar o no; (ii) $C$ es el coste de abandonar el tipo de cambio fijo; y, (iii) por la condición de PPA, la tasa de inflación será igual a la cuantía de la devaluación si ésta se produce ($\pi_t = d$), o nula ($\pi_t=0$) si las autoridades optan por mantener la paridad. Es decir, analíticamente:

$$\begin{aligned}
  &\text{Por condición PPA}&& \text{$\underset{\text{\scriptsize si se produce devaluación, la inflación es igual a su cuantía, y viceversa}}{\{\delta=1\iff \pi_t=d\,\,|\,\,\delta =0\iff \pi_t=0\}}$}
\end{aligned}$$

Con esto tendríamos definida la función objetivo de la Autoridad. Además, no es necesario especificar el comportamiento de los agentes para establecer las condiciones de equilibrio (o regiones estables) del modelo. ¿Por qué?

La razón es sencilla: es razonable suponer que los agentes actúen primero tomando sus decisiones en función del nivel de inflación esperado, atribuyendo una $\mathrm P(d)=\{0,1\}$ al abandono del compromiso del Gobierno. En este escenario, las autoridades simplemente reaccionarán óptimamente a las decisiones de los agentes, es decir, a los niveles de inflación esperada. El equilibrio se alcanza entonces iterando hacia atrás. La secuencia es la siguiente: (i) los agentes forman sus expectativas y fijan los salarios nominales; (ii) el Gobierno, que observa el desempleo heredado y las expectativas, decide si devalúa; y, (iii) se determina el desempleo del periodo. Con expectativas racionales, una expectativa es de equilibrio solo si la decisión del Gobierno la confirma.

Ampliamos la función objetivo sustituyendo en ella la curva de Phillips: así la pérdida queda expresada en función de lo que el Gobierno hereda ($u_{t-1}$), de lo que elige ($\pi_t$, a través de $\delta$) y de lo que esperan los agentes ($\mathbb E_{t-1}[\pi_t]$). Analíticamente:

$$\begin{aligned}
  &L_t=(u_t)^2+\delta C\quad\Longrightarrow\quad \boxed{\min_{\pi_t} L_t=[\rho\,u_{t-1}-\alpha(\pi_t-\mathbb E_{t-1}[\pi_t])]^2+\delta C}
\end{aligned}$$

¿Cuál será entonces la reacción óptima de las autoridades? Dependerá de las expectativas de los agentes, el coste de la devaluación y, crucialmente, de la sorpresa inflacionaria. Esto es, si los agentes creen firmemente en el compromiso cambiario, devaluar permite una sorpresa inflacionaria que reduce el desempleo, y habrá incentivos para la devaluación siempre y cuando esa ganancia supere su coste, $(\rho\,u_{t-1})^2-(\rho\,u_{t-1}-\alpha d)^2>C$. Por el contrario, si los agentes no confían en el compromiso cambiario la devaluación se vuelve mucho más atractiva: (i) si la autoridad devalúa, acarrea el coste de la devaluación ($C$); pero, (ii) si la autoridad mantiene el objetivo cambiario, se produce una sorpresa desinflacionaria ($\pi_t-\mathbb E_{t-1}[\pi_t]=-d$) que eleva el desempleo/pérdida, por tanto deviene muy costoso mantener la paridad.

Analíticamente, tendríamos:

$$\left\{\begin{aligned}
  &\text{E}1:\mathrm P(d)=0\,\Longrightarrow \mathbb E_{t-1}[\pi_t]=0\quad\Longrightarrow\quad&&
  \begin{aligned}
    &\underset{\text{\scriptsize Devaluación: $[\pi_t=d\iff\delta=1]$}}{L^d=(\rho\,u_{t-1}-\alpha\,d)^2+C}&&\text{¿}\gtrless\text{?}&& \underset{\text{\scriptsize Mantenimiento: $[\pi_t=0\iff \delta=0]$}}{(\rho\,u_{t-1})^2=L^f}
  \end{aligned}\\[1em]
  &\text{E}2:\mathrm P(d)=1\,\Longrightarrow \mathbb E_{t-1}[\pi_t]=d\quad\Longrightarrow\quad&&
  \begin{aligned}
    &\underset{\text{\scriptsize Devaluación: $[\pi_t=d\iff\delta=1]$}}{L^d=(\rho\,u_{t-1})^2+C}&&\text{¿}\gtrless\text{?}&&\underset{\text{\scriptsize Mantenimiento: $[\pi_t=0\iff \delta=0]$}}{(\rho\,u_{t-1}+\alpha\,d)^2=L^f}
    \end{aligned}
\end{aligned}\right.$$

Es decir, en el escenario E1 los agentes confían en la paridad y no esperan inflación. Si el Gobierno la mantiene, no hay sorpresa y el desempleo es solo el heredado, $\rho u_{t-1}$; si devalúa, la sorpresa inflacionaria $d$ reduce el desempleo en $\alpha d$, pero incurre en el coste $C$. En el escenario E2 los agentes esperan la devaluación y han fijado sus salarios en consecuencia. Si el Gobierno devalúa, valida esa expectativa: no hay sorpresa, el desempleo es el heredado y se paga $C$; si mantiene la paridad, la inflación resulta $d$ puntos inferior a la esperada y el desempleo aumenta en $\alpha d$. En ambos casos, devaluar mejora el desempleo en $\alpha d$ respecto a mantener; lo que cambia es el punto de partida, y por eso la expectativa de devaluación hace más atractiva la devaluación.

En definitiva, tomar la decisión de abandonar o no el objetivo cambiario depende de si los costes de abandonarlo son inferiores o no a los de mantenerlo.

Por un lado, sabemos que los \textbf{costes de mantenerlo} son varios, por ejemplo: (i) si el mantenimiento de la paridad requiere incrementar los tipos de interés, el coste de la deuda pública podría aumentar; (ii) si existe libre movilidad de capitales, mantener un tipo de cambio fijo implica renunciar a la política monetaria para objetivos internos; o, (iii) se renuncia a los efectos positivos de una depreciación sobre la balanza de pagos vía ganancias de competitividad precio.

Por otro lado, sabemos que los \textbf{costes de abandonarlo} incluyen: (i) pérdida de credibilidad antiinflacionista, ya que, generalmente, el tipo de cambio se fija frente a la moneda de un país con baja inflación, de manera que permite importar credibilidad antiinflacionista; o, (ii) posibles costes políticos del abandono del tipo de cambio fijo.

Además, si desarrollamos y simplificamos las desigualdades, podemos encontrar las regiones estables e inestable del objetivo cambiario. Es decir, para cada escenario de expectativas obtendremos el nivel de desempleo heredado a partir del cual al Gobierno le compensa devaluar; comparando ambos umbrales sabremos en qué casos solo una expectativa es coherente con la conducta del Gobierno (regiones estables) y en qué casos lo son las dos (región inestable).

Para alcanzarlo analíticamente, el desarrollo es simple. Partiendo de las desigualdades formuladas, para evitar líos debemos mantener siempre la misma estructura, bien sea una desigualdad positiva a la izquierda o a la derecha.

Utilizando la desigualdad positiva hacia la izquierda (el Gobierno devalúa si la pérdida de mantener supera a la de devaluar), desarrollando los cuadrados y dividiendo por $\alpha d>0$, el sistema de desigualdades queda como:

$$\begin{aligned}
    &(1)\ \ \mathbb E_{t-1}[\pi_t]=0:\quad \overset{\text{\scriptsize (pérdida mantenimiento)}}{(\rho u_{t-1})^2}>\overset{\text{\scriptsize (pérdida devaluación)}}{(\rho u_{t-1}-\alpha d)^2+C}&&\Longrightarrow\quad 2\rho u_{t-1}\,\alpha d-(\alpha d)^2>C&&\Longrightarrow\quad 2\rho u_{t-1}>\frac{C}{\alpha d}+\alpha d\\[1em]
    &(2)\ \ \mathbb E_{t-1}[\pi_t]=d:\quad \overset{\text{\scriptsize (pérdida mantenimiento)}}{(\rho u_{t-1}+\alpha d)^2}>\overset{\text{\scriptsize (pérdida devaluación)}}{(\rho u_{t-1})^2+C}&&\Longrightarrow\quad 2\rho u_{t-1}\,\alpha d+(\alpha d)^2>C&&\Longrightarrow\quad 2\rho u_{t-1}>\frac{C}{\alpha d}-\alpha d
\end{aligned}$$

Es decir, en ambos casos el Gobierno devalúa cuando el desempleo heredado supera un umbral, pero el umbral es más alto cuando nadie espera la devaluación: en (1) la ganancia de devaluar, $2\rho u_{t-1}\alpha d$, debe compensar el coste fijo y, además, el término $(\alpha d)^2$; en (2) ese término juega a favor de la devaluación, porque mantener la paridad provocaría una sorpresa desinflacionaria. A partir de aquí, definimos $x\equiv 2\rho u_{t-1}$ y $p\equiv \alpha d$. Con esto, podemos ver como la primera desigualdad domina a la segunda, ya que:

$$\begin{aligned}
    (1)\quad x>\frac{C}{p}+p\qquad \Longrightarrow\quad (2)\quad x>\frac{C}{p}+p>\frac{C}{p}-p
\end{aligned}$$

Por tanto, \textbf{si se cumple la primera siempre se cumple la segunda}: si el Gobierno devalúa incluso cuando nadie lo espera, con mayor razón devaluará cuando todos lo esperan. La región en la que se cumplen ambas desigualdades se define simplemente como la región en la que la primera desigualdad se cumple estrictamente.

Con esto, podemos saber qué expectativas son coherentes con la conducta del Gobierno en cada caso. Si se cumple (1), el Gobierno devalúa con independencia de lo que se espere: la expectativa de mantenimiento no puede ser de equilibrio, y la única coherente es la devaluación. Si no se cumple (2), el Gobierno mantiene la paridad aunque todos esperen la devaluación: el único equilibrio es el mantenimiento. Y si se cumple (2) pero no (1), el Gobierno mantiene si se espera que mantenga y devalúa si se espera que devalúe, de modo que ambas expectativas se autocumplen. Despejando $u_{t-1}$, obtenemos las tres regiones. Analíticamente:

$$\begin{aligned}
  &\text{Devaluación}:&&u_{t-1}>u_D\equiv\frac{1}{2\rho}\left(\frac{C}{\alpha d}+\alpha d\right)\\[1em]
  &\text{Mantenimiento}: && u_{t-1}<u_M\equiv\frac{1}{2\rho}\left(\frac{C}{\alpha d}-\alpha d\right)\\[1em]
  &\underset{\text{\scriptsize siempre existe si: $\alpha, d>0$}}{\text{Región inestable}:}&&\boxed{\underset{\text{\scriptsize mantenimiento}}{\frac{1}{2\rho}\left(\frac{C}{\alpha d}-\alpha d\right)}<u_{t-1}\leq \underset{\text{\scriptsize devaluación}}{\frac{1}{2\rho}\left(\frac{C}{\alpha d}+\alpha d\right)}}
\end{aligned}$$

Hay dos interpretaciones a este respecto. En primer lugar, tenemos la interpretación literal que obtenemos del estudio de las desigualdades: existen tres regiones bien diferenciadas que corresponden a dos conductas firmes y coherentes de la autoridad monetaria (devaluación/mantenimiento) y otra región inestable en la que la autoridad monetaria quedará sometida al vigor de cualquier ataque especulativo lo suficientemente grande (región inestable).

Sin embargo, la otra interpretación, más sutil, permite apreciar la potencia del modelo. Como vemos, todas las condiciones se han expresado en función de la tasa de desempleo (un fundamento importante de la economía). Esto tiene una razón de ser.

Si los fundamentos de la economía están muy deteriorados, una devaluación será imperante: la sorpresa inflacionaria reduce los salarios reales, pactados de antemano, y con ellos el desempleo (en un modelo más amplio, también ganaremos en competitividad y aumentará la demanda externa). Además, cuánto mayor es la persistencia del desempleo menor es el umbral, puesto que más tentador resultará usar la devaluación como mecanismo de escape.

La primera región estable, por tanto, expresa la imposibilidad de mantener la paridad por la gravedad interna y la intensidad de los incentivos de la autoridad monetaria para recurrir a la devaluación para garantizar el ajuste interno. Esto se puede apreciar perfectamente analizando como afecta la intensidad esperada de la devaluación ($d$) sobre el umbral de la devaluación.

Por un lado, afecta a través del \textbf{deflactor de escape}. Es decir, $\frac{C}{\alpha d}$ refleja que, cuánto mayor es la devaluación esperada ($\uparrow d$), menor es el coste fijo relativo de validar la devaluación: el umbral se reducirá (aumentará) haciendo más fácil (difícil) el recurso a la cláusula de escape.

Y, por otro lado, a través de la \textbf{perturbación nominal}. Una devaluación de mayor magnitud también implica una perturbación nominal más elevada, de modo que el umbral de devaluación no cae, necesariamente, de forma monótona ($+\alpha d$). La condición relevante depende de si $\alpha d$ es mayor o menor que $\sqrt{C}$. Es decir, para devaluaciones esperadas moderadas (en intensidad) un aumento de $d$ hace más probable (acorta el umbral) la devaluación, y viceversa. Se puede apreciar pensando:

$$u_{t-1}>\frac{1}{2\rho}\left(\frac{C}{\alpha d}+\alpha d\right)\Longrightarrow u_{D}(d)=\frac{1}{2\rho}\left(\frac{C}{\alpha d}+\alpha d\right)$$

Si obtenemos la derivada, vemos como:

$$\frac{\partial u_{D}}{\partial d}=\frac{1}{2\rho}\left(-\frac{C}{\alpha d^2}+\alpha\right)=\frac{\alpha}{2\rho}\left(1-\frac{C}{(\alpha d)^2}\right)$$

Por tanto, el umbral se acortará $(\frac{\partial u_{D}}{\partial d}<0)$ si y solo si $(\alpha d)^2<C$, y se ensanchará $(\frac{\partial u_{D}}{\partial d}>0)$ si y solo si $(\alpha d)^2>C$. El umbral de devaluación tiene, pues, forma de U, con un mínimo en $\alpha d=\sqrt{C}$, donde vale $u_D=\sqrt{C}/\rho$.

Ahora bien, todo lo expuesto es también válido a la inversa, cuando nos encontramos en región estable de mantenimiento. No obstante, si la llevamos al extremo es más fácil apreciar sus límites. Si los fundamentos económicos son muy robustos (cercanos al pleno empleo, $u_{t-1}=0$), el equilibrio creíble siempre existe: la condición (1) no puede cumplirse, porque su lado derecho es siempre positivo, de modo que el Gobierno nunca devaluaría por sorpresa. Pero eso no basta para descartar el ataque. El umbral de mantenimiento, $u_M$, es positivo solo si $\alpha d<\sqrt{C}$: si se espera una devaluación lo bastante grande ($\alpha d>\sqrt{C}$), la condición (2) se cumple incluso con pleno empleo, porque el coste de defraudar esa expectativa, $(\alpha d)^2$, supera al de devaluar, $C$. \textbf{Con pleno empleo, el compromiso es inmune a los ataques solo si la devaluación que podrían esperar los agentes es moderada.}

Con todo, nos damos cuenta que la situación interesante se da en la región inestable. Lo primero que notamos es que esta región expresa un deterioro \textbf{moderado} de los fundamentos económicos: no estamos tan mal. Sin embargo, un ataque especulativo podría forzar la devaluación de la moneda. Esto es así porque nos encontramos en la región de equilibrios múltiples: podríamos acabar en cualquiera de los dos equilibrios en función, únicamente, de lo que esperen los agentes.

Por tanto, ¿de qué depende que sucedan o no dichos ataques? Depende crucialmente de la devaluación esperada, $d$. ¿Cómo podemos ver esta relación de manera más intuitiva? La manera más sencilla es analizar la amplitud de dicho intervalo, es decir, la distancia entre el umbral de devaluación y el de mantenimiento, que mide el rango de fundamentos en el que la economía está expuesta a un ataque autocumplido. Analíticamente, empezamos con:

$$\begin{aligned}
  &\underset{\text{\scriptsize (límite superior menos límite inferior)}}{\text{Operando}:}&&u_D-u_M=\frac{1}{2\rho}\left[\left(\frac{C}{\alpha d}+\alpha d\right)-\left(\frac{C}{\alpha d}-\alpha d\right)\right]=\frac{\alpha d}{\rho}\\[1em]
\end{aligned}$$

Es decir, el coste fijo $C$ desaparece: desplaza los dos umbrales en la misma cuantía y, por tanto, determina dónde está la región inestable, pero no su tamaño. La amplitud depende solo de la eficacia de la sorpresa inflacionaria, de la devaluación esperada y de la persistencia del desempleo. Con esto, tenemos:

$$\begin{aligned}
  \frac{\partial (u_D-u_M)}{\partial d}=\frac{\alpha}{\rho}>0,\qquad\frac{\partial (u_D-u_M)}{\partial \alpha}=\frac{d}{\rho}>0,\qquad\frac{\partial (u_D-u_M)}{\partial \rho}=-\frac{\alpha d}{\rho^2}<0
\end{aligned}$$

Es decir, a diferencia de lo que ocurría con el umbral de devaluación, el efecto de la devaluación esperada sobre la amplitud de la región inestable es lineal y siempre positivo: cada punto adicional de devaluación esperada ensancha la zona de fragilidad en $\alpha/\rho$.

Como vemos, la intensidad esperada de la devaluación actúa como amplificador de la región de fragilidad. Este es el insight importante: aunque una devaluación esperada muy alta no siempre reduce el umbral de devaluación (como hemos visto en el razonamiento de la primera región estable), sí aumenta de forma inequívoca el rango de fundamentos en el que la economía queda expuesta a equilibrios múltiples o ataques autocumplidos.

Este es precisamente el mecanismo de escape de los modelos de segunda generación o \textbf{escape clause}: \textbf{los fundamentos importan, pero las expectativas pueden desplazar a la economía hacia una crisis incluso sin que los fundamentos sean tan malos.}

Además, es interesante la información que nos proporciona la tasa de crecimiento de esta fragilidad: (i) $\alpha$ mide la eficacia de la sorpresa inflacionaria/devaluatoria sobre el objetivo interno, cuánto mayor es $\alpha$, más poderosa es la devaluación como instrumento de estabilización, pero también mayor es el daño de no validar una devaluación esperada (amplía la zona de equilibrios múltiples); y, por otro lado, (ii) $\rho$ mide la severidad/persistencia del desempleo interno, cuánto mayor es $\rho$, más sensible será la autoridad monetaria al desempleo pasado y más se reducirán los umbrales de acción, por ello comprime la zona de fragilidad en términos de desempleo heredado, puesto que será una cuestión ya internalizada por la autoridad monetaria.

\anexo{Tercera generación: Desarrollo analítico}\label{anx:modelos_tercera}

Este anexo recoge el desarrollo analítico de las tres ramas de la tercera generación presentadas en el cuerpo del tema. A diferencia de la primera y la segunda generación, no existe un modelo canónico, de modo que cada rama se formaliza por separado. El modelo de riesgo moral de DOOLEY (2000) reproduce la lógica de la primera generación: una crisis previsible cuya fecha depende de unas reservas que se agotan, aunque ahora lo que las agota no es la monetización del déficit, sino las garantías públicas al sistema financiero. El modelo de iliquidez de CHANG y VELASCO (2001) traslada a una economía abierta los pánicos bancarios de DIAMOND y DYBVIG (1983) y hereda los equilibrios múltiples de la segunda generación. Y el modelo de balance de KRUGMAN (1999), el más desarrollado, combina los equilibrios múltiples con los efectos de la depreciación sobre el balance de las empresas. Para facilitar su memorización, las dos primeras se presentan en versiones simplificadas, que conservan el mecanismo esencial de los trabajos originales a costa de prescindir de su tratamiento formal completo\textcolor{orange}{\textsuperscript{[Probable]}}.


\textbf{Modelos de riesgo moral: DOOLEY (1997)}

El punto de partida de DOOLEY es que, en muchas economías emergentes, el Gobierno respalda de forma implícita los pasivos del sistema financiero: si las entidades tienen problemas, los acreedores esperan ser rescatados con los recursos públicos, y en particular con las reservas internacionales. Esa garantía funciona como un seguro gratuito, y el modelo describe qué ocurre cuando los inversores internacionales lo aprovechan (versión simplificada del modelo de seguro de DOOLEY (2000))\textcolor{orange}{\textsuperscript{[Probable]}}.

Supongamos que el Gobierno dispone de un stock de reservas, $R$, que está dispuesto a utilizar para cubrir las pérdidas de los acreedores, y que el sistema financiero ofrece a los inversores extranjeros una rentabilidad superior a la internacional, $i^*+\sigma$, porque invierte en proyectos arriesgados o simplemente ineficientes. Para un inversor, prestar al sistema financiero nacional es un negocio sin riesgo mientras la garantía cubra todo lo prestado: si las inversiones salen mal, cobrará igualmente con cargo a las reservas. Por tanto, los pasivos garantizados, $L_t$, crecen mientras su valor no supere al de las reservas. Supondremos, por simplicidad, que lo hacen a un ritmo constante, $g$. Analíticamente:

$$\begin{aligned}
  &\text{Pasivos garantizados}:&&L_t=L_0+g\,t\\[1em]
  &\text{Rentabilidad}:&&\text{si}\ L_t<R:\ \ i^*+\sigma\ \text{sin riesgo};\qquad \text{si}\ L_t>R:\ \text{cobertura parcial}\ \tfrac{R}{L_t}<1
\end{aligned}$$

Es decir, mientras los pasivos garantizados sean inferiores a las reservas, el seguro cubre a todos los acreedores y la prima $\sigma$ es una ganancia segura, lo que atrae más entradas de capital. En cuanto los pasivos superasen a las reservas, el seguro solo cubriría una parte de lo prestado, y los acreedores que esperasen cobrarían menos que los que retirasen antes sus fondos. La lógica es la misma que en FLOOD y GARBER (1984): ningún inversor esperará a que el seguro sea insuficiente, porque perdería frente a los que se adelanten. El ataque se produce, por tanto, en el momento exacto en que los pasivos garantizados igualan a las reservas. Analíticamente:

$$\begin{aligned}
  &\text{Ataque}:\ \ L_T=R\quad\Longrightarrow\quad \boxed{T=\frac{R-L_0}{g}}\qquad\text{y}\qquad \text{tamaño del ataque}=R
\end{aligned}$$

Es decir, en $T$ los acreedores liquidan todos sus pasivos garantizados, el Gobierno usa todas sus reservas para atenderlos y el sistema financiero, cuyas pérdidas se hacen visibles, quiebra. Si el tipo de cambio era fijo, la paridad se abandona, porque ya no quedan reservas para defenderla: crisis bancaria y cambiaria a la vez. La fecha de la crisis es perfectamente previsible, como en la primera generación, y depende de tres factores: las reservas disponibles, el nivel inicial de los pasivos garantizados y el ritmo al que crecen, que es mayor cuanto más intensa es la liberalización financiera y más débil la supervisión.

\imagenfit{3era_Gen_dooley.pdf}{Modelo de seguro de DOOLEY: pasivos garantizados, reservas y crisis recurrentes}{Elaboración propia a partir de DOOLEY (2000). Valores ilustrativos.}

La figura muestra, además, la implicación más llamativa del modelo. Tras la crisis, el Gobierno reconstruye sus reservas ($R'$), y, en cuanto lo hace, el seguro vuelve a estar disponible: los pasivos garantizados empiezan a crecer de nuevo y el ciclo se repite en $T'$. Las crisis son \textbf{recurrentes}. Por la misma razón, acumular más reservas no evita la crisis: como muestra la fecha del ataque, unas reservas mayores la retrasan ($\partial T/\partial R=1/g>0$), pero también aumentan su tamaño, porque el ataque se lleva todas las reservas. La única forma de romper el ciclo es eliminar el seguro, es decir, suprimir las garantías implícitas o regular los riesgos que asumen las entidades garantizadas.

CORSETTI, PESENTI y ROUBINI (1999) proponen una lectura complementaria: las garantías implícitas equivalen a un \textbf{déficit público oculto}, el coste futuro de los rescates. Cuando los agentes anticipan que ese coste se financiará, en parte, con creación de dinero, la crisis sigue la lógica de la primera generación, con la diferencia de que el déficit que la provoca no aparece en las cuentas públicas hasta que estalla\textcolor{orange}{\textsuperscript{[Probable]}}.


\textbf{Modelos de asimetrías de balance: CHANG y VELASCO (1998)}

CHANG y VELASCO (2001) trasladan a una economía abierta el modelo de pánicos bancarios de DIAMOND y DYBVIG (1983). Su objetivo es mostrar que una economía con buenos fundamentos puede sufrir una crisis financiera y cambiaria si su sistema financiero es ilíquido en moneda extranjera. Presentamos una versión simplificada con tres periodos ($t=0,1,2$)\textcolor{orange}{\textsuperscript{[Probable]}}.

En $t=0$, los bancos captan depósitos de los residentes y préstamos exteriores a corto plazo, $f$, y lo invierten todo, $k$, en un activo productivo a largo plazo. Ese activo rinde $q>1$ por unidad si se mantiene hasta $t=2$, pero, si se liquida antes, en $t=1$, solo se recupera $r<1$ por unidad. Los depositantes no saben en $t=0$ cuándo necesitarán su dinero: una fracción $\eta$ resultará ser \textbf{impaciente} y querrá consumir en $t=1$, y el resto, \textbf{paciente}, puede esperar a $t=2$. El banco ofrece un contrato de depósito que permite retirar $c_1$ en $t=1$ o $c_2$ en $t=2$, con $c_2>c_1$, de modo que a los pacientes les compense esperar. Además, el país dispone de una cantidad limitada de liquidez en moneda extranjera, $m$ (reservas y líneas de crédito), para atender pagos inesperados en $t=1$.

Este contrato es útil: permite financiar una inversión rentable a largo plazo con un ahorro que los depositantes quieren tener disponible. Pero crea una \textbf{asimetría de vencimientos}: el banco tiene pasivos exigibles a corto plazo y activos que solo rinden plenamente a largo plazo. Un día normal, solo retiran sus fondos los impacientes y los acreedores exteriores renuevan sus préstamos, de modo que el banco puede atenderlos liquidando una pequeña parte de la inversión, y los pacientes, que esperan, reciben $c_2$. Analíticamente:

$$\begin{aligned}
  &\text{Día normal}:&&\eta\,c_1\leq r\,k\qquad\text{y}\qquad c_2=\frac{q\,(k-\eta c_1/r)}{1-\eta}\geq c_1
\end{aligned}$$

Es decir, los recursos que el banco puede obtener liquidando su activo bastan para pagar a los impacientes, y lo que queda, al rendir $q$ en $t=2$, permite pagar a los pacientes al menos lo mismo que si hubieran retirado sus fondos antes. Nadie tiene incentivos para retirar anticipadamente: es un equilibrio.

Supongamos ahora que todos los depositantes, también los pacientes, acuden al banco en $t=1$, y que los acreedores exteriores se niegan a renovar sus préstamos. Las obligaciones a corto plazo pasan a ser $c_1+f$ (normalizando a la unidad la masa de depositantes), y los recursos que pueden movilizarse son el valor de liquidación de toda la inversión más la liquidez en moneda extranjera disponible. Si los primeros superan a los segundos, el sistema no puede pagar a todos: los que lleguen tarde no cobrarán nada. Analíticamente:

$$\begin{aligned}
  &\text{Iliquidez internacional}:&&\boxed{c_1+f>r\,k+m}
\end{aligned}$$

Es decir, cuando se cumple esta condición, a cada depositante paciente le compensa retirar su dinero si cree que los demás lo harán, porque, si espera, el banco habrá liquidado sus activos y no le quedará nada. El pánico se autocumple. Con los mismos fundamentos, coexisten dos equilibrios: el normal, en el que nadie se adelanta y los bancos son solventes, y el de pánico, en el que todos se adelantan y los bancos quiebran. Nótese que el problema no es de solvencia: si se dejara llegar la inversión a su vencimiento, rendiría lo suficiente para pagar a todos. Es un problema de liquidez, que nace de la asimetría de vencimientos y que es tanto más grave cuanto mayor es la deuda exterior a corto plazo ($f$), menor el valor de liquidación de los activos ($r$) y menor la liquidez internacional disponible ($m$). Por eso la relación entre la deuda exterior a corto plazo y las reservas es el indicador clave de esta rama.

\imagenfit{3era_Gen_liquidez.pdf}{Iliquidez internacional en una versión simplificada de CHANG y VELASCO}{Elaboración propia a partir de CHANG y VELASCO (2001). Valores ilustrativos.}

Falta ver cómo se transforma el pánico en una crisis cambiaria. Para ello hay que introducir el Banco Central y el régimen cambiario. Supongamos que los depósitos están denominados en moneda nacional y que el Banco Central puede actuar como prestamista de última instancia, creando moneda nacional para los bancos (CHANG y VELASCO, 2000)\textcolor{orange}{\textsuperscript{[Probable]}}. Caben tres situaciones. Si el país tiene una \textbf{caja de conversión}, el Banco Central no puede crear moneda sin respaldo en divisas, de modo que no actúa como prestamista: la paridad se mantiene, pero los bancos quiebran. Si el tipo de cambio es \textbf{fijo y el Banco Central actúa como prestamista}, crea moneda nacional para pagar a los depositantes, pero estos, que temen la crisis, la cambian inmediatamente por divisas al tipo fijo: el rescate bancario se convierte en un ataque contra las reservas, que se agotan si $c_1+f>r\,k+m$, y se producen a la vez la crisis bancaria y la cambiaria (\textbf{crisis gemelas}). Por último, si el tipo de cambio es \textbf{flexible}, el Banco Central puede crear la liquidez necesaria sin comprometer ninguna paridad: el pánico se traduce en una depreciación, que reduce el valor en divisas de lo que se retira, y los bancos sobreviven. En la versión de CHANG y VELASCO, esta combinación puede llegar a eliminar el equilibrio de pánico\textcolor{orange}{\textsuperscript{[Probable]}}.

\imagenfit{3era_Gen_regimen.pdf}{El pánico según el régimen cambiario}{Elaboración propia a partir de CHANG y VELASCO (2000, 2001).}

El modelo pone de manifiesto, en definitiva, un límite fundamental: el Banco Central puede crear moneda nacional, pero no divisas. Cuando las obligaciones a corto plazo están denominadas en moneda extranjera, como $f$, ni siquiera la flexibilidad cambiaria permite atenderlas, y la única respuesta posible es disponer de reservas suficientes o de un prestamista de última instancia internacional.


\textbf{Modelos de balance: KRUGMAN (1999)}


En este modelo ni el Banco Central ni las instituciones intermedias de crédito tienen un papel central.

Por el contrario, \textbf{la crisis se explica a través de los problemas de balance en las propias empresas}. El razonamiento es sencillo: una reversión de los flujos de capital extranjero, por cualquier motivo, que genere una pérdida de confianza por parte de los agentes provocará una depreciación del tipo de cambio real y por tanto un aumento del valor de las deudas denominadas en moneda extranjera y una reducción del valor de los activos nacionales, lo que acabará desequilibrando el balance de las empresas. ¿Cómo impacta esto en la economía real? Estos problemas de balance dificultan el acceso al crédito de empresas y particulares, lo que lastra la inversión (y el consumo) y genera disminuciones adicionales de las entradas de capital. De esta forma, las complicaciones financieras se transmiten a la economía real y este círculo de retroalimentación negativa puede conducir, sucesivamente, hasta la crisis. Por tanto, se trata de modelos de equilibrios múltiples en los que una pérdida de confianza puede generar un colapso financiero y por tanto fenómenos de expectativas autocumplidas.

Partimos de una hipótesis básica: la capacidad de endeudamiento de los empresarios depende de su propia riqueza, que es función a su vez del nivel de endeudamiento de la economía, dado que el volumen de entradas de capital tiene efectos sobre el tipo de cambio real y por tanto sobre la valoración de la deuda en moneda extranjera.

De esta forma, una disminución de las entradas de capital extranjero puede tener efectos negativos sobre el balance de las empresas, reduciendo su capacidad de endeudamiento y en consecuencia disminuyendo a su vez las entradas de capitales.

Para ver todo el mecanismo, partamos de una economía pequeña y abierta que produce un único bien homogéneo en cada periodo a partir de dos factores de producción: capital y trabajo.

Supongamos que el capital dura únicamente un periodo, de forma que el stock de capital en el periodo $t$ es igual a la inversión acometida en el periodo $t-1$. En consecuencia, la producción de cada periodo, $y_t$, está predeterminada por la inversión del periodo anterior: a corto plazo, la tomaremos como dada.

Los agentes se dividen entre: (i) trabajadores, sin acceso al mercado de capitales, que consumen toda su renta; y, (ii) empresarios, que ahorran e invierten toda su renta.

Por simplicidad, supondremos que la función de producción es COBB-DOUGLAS ($F(K,L)=K^\alpha L^{1-\alpha}$), de manera que la renta de los trabajadores sea igual a $(1-\alpha)y_t$ y la renta del capital, que perciben los empresarios, a $\alpha y_t$.

Por otro lado, los bienes domésticos y extranjeros no son sustitutivos perfectos, y supondremos una elasticidad de sustitución entre ambos unitaria, $\sigma=1$.

En consecuencia, el gasto nacional se reparte en proporciones constantes entre bienes nacionales y extranjeros. Por un lado, se destinará a \textbf{bienes importados} una fracción constante, $\mu$, del consumo y la inversión. Por otro lado, se destinará a \textbf{bienes nacionales} la fracción restante, $(1-\mu)$. La demanda extranjera de bienes nacionales son las exportaciones, $X_t$, medidas en unidades de bien extranjero.

Dado que se trata de un país pequeño, no tendrá influencia sobre la fijación del precio mundial: su cantidad de exportaciones, en cualquier mercado, no es lo suficientemente grande como para contribuir a la formación de ese precio. De esta manera, el volumen de exportaciones viene dado de forma exógena. Lo que no viene dado es el \textbf{tipo de cambio real}, $p_t$, el precio de los bienes extranjeros en unidades de bien nacional: es la variable que se ajusta para equilibrar el mercado de bienes nacionales. Una subida de $p_t$ es una depreciación real y, al mismo tiempo, un empeoramiento de la relación real de intercambio, que es su inversa, $1/p_t$.

\textbf{NOTA:} Se menciona a menudo el término \textbf{real}. Si se considera confuso, puede omitirse sin problemas. Como vemos, el modelo no especifica un tipo de cambio nominal y un tipo de cambio real. Esto es, no existe inflación y todo se expresa en términos reales.

Con esta especificación, podemos representar el equilibrio como la igualación entre la oferta del bien doméstico y su demanda. Es decir, la producción nacional debe ser igual a la parte del consumo y de la inversión nacionales que recae sobre bienes nacionales más las exportaciones, valoradas en unidades de bien nacional. Analíticamente:

$$\begin{aligned}
    &\text{Equilibrio}:&& y_t=(1-\mu)c_t+(1-\mu)i_t+p_tX_t&& \text{donde,}\quad p_t\equiv\text{\scriptsize Tipo de cambio real (precio relativo del bien extranjero)}
\end{aligned}$$

Por tanto, el primer término es la demanda de consumo dirigida a bienes nacionales, el segundo la demanda de inversión dirigida a bienes nacionales y el tercero la demanda extranjera. Si reescribimos la ecuación anterior teniendo en cuenta que solo consumen los trabajadores, y que consumen toda su renta ($c_t=(1-\alpha)y_t$), el consumo deja de ser una variable independiente. Analíticamente:

$$\begin{aligned}
  &\underset{\text{\scriptsize reescribiendo}}{\Longrightarrow}&&\boxed{y_t=(1-\mu)(1-\alpha)y_t+(1-\mu)i_t+p_tX_t}
\end{aligned}$$

Es decir, con la producción dada, el mercado de bienes solo puede equilibrarse a través de la inversión o del tipo de cambio real: si cae la inversión, la demanda de bienes nacionales disminuye y, para que la producción siga vendiéndose, los bienes nacionales deben abaratarse en relación con los extranjeros, lo que aumenta las exportaciones medidas en bienes nacionales. Por último, despejando el tipo de cambio real y derivando respecto a la inversión, obtenemos el primer eslabón del bucle. Analíticamente:

$$\begin{aligned}
  &\underset{\text{\scriptsize despejando}}{\Longrightarrow}&& \boxed{p_t=\frac{\left[1-(1-\mu)(1-\alpha)\right]y_t-(1-\mu)i_t}{X_t}}&&\Longrightarrow\quad \frac{\partial p_t}{\partial i_t}=-\frac{1-\mu}{X_t}<0:\quad  \downarrow i_t\,\,\Longrightarrow\,\,\underset{\text{\scriptsize (depreciación)}}{p_t\uparrow}
\end{aligned}$$

Como vemos, expresar el equilibrio en términos de igualación entre la demanda y la oferta domésticas nos permite despejar el tipo de cambio real. De la misma forma, esta reescritura nos permite ver que, si se produce una revisión de las expectativas a la baja por cualquier motivo (una disminución de la inversión), el tipo de cambio real se depreciará y empeorará la relación real de intercambio. La depreciación es mayor cuanto mayor es la parte del gasto que recae sobre bienes nacionales ($1-\mu$) y cuanto menores son las exportaciones.

Con el equilibrio exterior definido y las interacciones establecidas, debemos presentar la última pieza del modelo: ¿de dónde surgen los problemas de balance?

Para ello, empezamos por definir la riqueza de los empresarios. Como el capital solamente dura un periodo, el valor del capital es la porción de la renta que se destina al capital en el periodo en curso. Por tanto, la riqueza de los empresarios es el valor de su capital menos sus deudas, que pueden estar denominadas en bienes nacionales o en bienes extranjeros; estas últimas, para poder restarlas, deben expresarse en bienes nacionales multiplicándolas por el tipo de cambio real. Analíticamente:

$$\begin{aligned}
  &\text{Riqueza productores}:&&\boxed{w_t=\alpha y_t-b^d_t-p_t\,b^f_t}\\[2em]
    &\text{Donde,}&&
    \left\{\begin{aligned}
        &\alpha y_t\equiv\hspace{2em}\text{Capital (Fondos propios)}\\[0.5em]
        &\left.\begin{aligned}
            &b^d_t\equiv&&\text{Deuda neta (moneda nacional)}\\[0.5em]
            &p_t\,b_t^f\equiv&&\text{Deuda neta (divisa)}
        \end{aligned}\right\}\quad \left(\substack{\text{\scriptsize expresadas en}\\\text{\scriptsize bienes nacionales}}\right)
    \end{aligned}\right.
\end{aligned}$$

Es decir, la riqueza depende del tipo de cambio real solo a través de la deuda en moneda extranjera: si $b^f_t>0$, cada punto de depreciación real aumenta el valor de esa deuda en bienes nacionales y reduce la riqueza en $b^f_t$. Obsérvese que, de acuerdo con esta expresión, la depreciación de la moneda nacional provocará una disminución de la riqueza de los empresarios, y que, si es suficientemente intensa, puede llevarla a ser negativa: los empresarios quiebran.

Por tanto, ¿se enfrenta a algún límite la capacidad de endeudamiento? A priori, resulta razonable suponer que sí.

De hecho, podemos considerar que los empresarios no podrán endeudarse por un valor superior a $\theta$ veces su riqueza, $w_t$, porque los prestamistas exigen que una parte de la inversión se financie con fondos propios. Como la inversión es la suma de la riqueza propia y de la nueva deuda, la inversión no podrá superar $(1+\theta)$ veces la riqueza: $i_t\leq w_t+\theta\,w_t=(1+\theta)\,w_t$.

Utilizando esta expresión, podemos definir el nivel de inversión financiable con deuda como aquel que tendría lugar si se \textbf{saturase la restricción de apalancamiento}. Es decir, el máximo que los empresarios pueden invertir dada su riqueza, que será la inversión efectiva siempre que les compense invertir todo lo que pueden. Analíticamente, podemos expresarlo como:

$$\begin{aligned}
  &\text{Inversión financiable}:&&i_t^f=(1+\theta)w_t\quad\Longrightarrow\quad i_t^f=(1+\theta)(\alpha y_t-b_t^d-p_t\,b_t^f)
\end{aligned}$$

Es decir, la inversión financiable crece con la producción, que aumenta el valor del capital, y disminuye con las deudas y, a través de la deuda en divisas, con el tipo de cambio real. Además, teniendo en cuenta que la producción y las deudas están dadas a corto plazo, el efecto de una depreciación sobre la inversión financiable es directo. Analíticamente:

$$\begin{aligned}
  &\text{Si existe deuda externa}: &&\boxed{\frac{\partial i_t^f}{\partial p_t}=-(1+\theta)b_t^f<0}
\end{aligned}$$

Esta ecuación es precisamente la que describe el efecto balance: \textbf{una depreciación real ($\uparrow p_t$) aumenta el valor de la deuda extranjera y reduce la riqueza neta, restringiendo la capacidad de apalancarse/endeudarse y afectando negativamente a la inversión}.

La inversión financiable tiene, además, dos límites. Por abajo, no puede ser negativa: si la riqueza es negativa, los empresarios han quebrado y la inversión es nula. Por arriba, la inversión no supera el nivel $i^{\max}$ en el que su rentabilidad se iguala a la del exterior: aunque los empresarios pudieran endeudarse más, no les compensaría invertir. La inversión efectiva es, por tanto, $i_t=\min\{i^{\max},\max\{0,(1+\theta)w_t\}\}$, y la restricción de apalancamiento solo es operativa entre ambos límites.

Ahora solo nos falta un último paso para poder cuantificar el impacto de la depreciación sobre la inversión. Debemos obtener el \textbf{multiplicador financiero}. Para ello, distinguimos entre la inversión que esperan los prestamistas ($i_t^e$), que es la que determina el tipo de cambio real a través del mercado de bienes, y la inversión que efectivamente puede financiarse con ese tipo de cambio ($i_t^f$). Es decir, hemos definido ya:

$$\begin{aligned}
  &(1)\,\text{Equilibrio}:&&p_t=\frac{\left[1-(1-\mu)(1-\alpha)\right]y_t-(1-\mu)i_t^e}{X_t}\quad\underset{\text{\scriptsize c/p: $\bar y_t$ y $\bar X_t$}}{\Longrightarrow}&&\boxed{\frac{\partial p_t}{\partial i_t^e}=-\frac{1-\mu}{\bar X_t}<0}\\[1em]
  &(2)\,\text{Riqueza}:&&w_t=\alpha y_t-b_t^d-p_t\,b_t^f\quad \underset{\text{\scriptsize si $b_t^f>0$}}{\Longrightarrow}&&\boxed{\frac{\partial w_t}{\partial p_t}=-b_t^f<0}\\[1em]
  &(3)\,\text{Inversión financiable}:&&i_t^f=(1+\theta)w_t\quad\underset{\text{\scriptsize si $\theta>0$}}{\Longrightarrow}&&\boxed{\frac{\partial i_t^f}{\partial w_t}=1+\theta}
\end{aligned}$$

Por tanto, sabemos que la inversión esperada determina el tipo de cambio real (1), que el tipo de cambio real determina la riqueza (2) y que la riqueza determina la inversión financiable (3). Cada ecuación toma como dato la variable que determina la anterior, de modo que forman una cadena, y el efecto de la inversión esperada sobre la financiable es el producto de los tres efectos parciales (regla de la cadena). Analíticamente, tendríamos:

$$\begin{aligned}
  &(1)+(2)+(3)\quad \Longrightarrow\hspace{4em}\frac{\partial i_t^f}{\partial i_t^e}=\frac{\partial i_t^f}{\partial w_t}\frac{\partial w_t}{\partial p_t}\frac{\partial p_t}{\partial i_t^e}=\boxed{\frac{(1+\theta)(1-\mu)b_t^f}{\bar X_t}\equiv\lambda>0}
\end{aligned}$$

Es decir, los dos signos negativos se compensan y el resultado es positivo: una menor inversión esperada reduce la inversión financiable. Su magnitud depende de tres factores, uno por eslabón: el apalancamiento ($1+\theta$), la deuda en divisas ($b^f_t$) y la sensibilidad del tipo de cambio real a la inversión ($(1-\mu)/X_t$). Por tanto, llamaremos \textbf{multiplicador financiero} a $\lambda$: mide cuánto varía la inversión financiable por cada unidad que varía la inversión esperada. Como las expectativas son racionales, en equilibrio la inversión esperada debe coincidir con la que efectivamente puede financiarse ($i_t^e=i_t^f$); si no coinciden, los prestamistas revisan sus expectativas y el bucle se repite. Analíticamente, podríamos describir la relación del multiplicador financiero como:

$$\begin{aligned}
  &\text{Retroalimentación}: \quad \downarrow i_t^e \Rightarrow \uparrow p_t \Rightarrow \downarrow w_t \Rightarrow \downarrow i_t^f \Rightarrow \downarrow i_t^e\quad \underset{\text{\scriptsize ¿cómo?}}{\Longrightarrow}\quad \boxed{\Delta i_t^f=\lambda\Delta i_t^e}
\end{aligned}$$

Esta ecuación, definida como el \textbf{multiplicador financiero}, nos permite conocer si acabaremos o no en una situación de crisis por razones del deterioro de las expectativas y su impacto sobre el balance de las empresas, y cómo.

\begin{la}
    \item \textbf{Multiplicador amortigua: $\lambda<1$}. Si el multiplicador es inferior a la unidad, el deterioro de las expectativas producirá una salida de capitales que empeorará las condiciones de financiación por el deterioro del balance. Sin embargo, cada vuelta del bucle reduce la inversión financiable menos que la anterior, el círculo de retroalimentación se irá amortiguando con el tiempo y no cabe esperar una crisis;
    \item \textbf{Multiplicador acelerador: $\lambda>1$} (Equilibrios múltiples). Por el contrario, si el multiplicador es superior a la unidad, el deterioro de las expectativas producirá una salida de capitales que empeorará las condiciones de financiación por el deterioro del balance en mayor medida que la intensidad de la salida. ¿Qué sucede entonces? El proceso continúa agravándose, se potencia a sí mismo y puede adquirir una senda explosiva que conduce a la crisis financiera y cambiaria. No es inevitable: con $\lambda>1$ coexisten el equilibrio sin crisis y el de crisis, y es la pérdida de confianza la que decide el paso de uno a otro.
\end{la}

¿Cuáles son los factores que influyen en este multiplicador? Como cabría esperar: (i) el grado de apalancamiento ($\theta$); (ii) el peso de la deuda en moneda extranjera en relación con las exportaciones ($b_t^f/X_t$); y, (iii) una baja propensión marginal a importar ($\mu$), porque cuanto más se concentra el gasto en bienes nacionales, más recae sobre ellos la caída de la inversión y mayor es la depreciación necesaria para equilibrar el mercado de bienes.\footnote{%
    Todas las economías afectadas por la crisis asiática presentaban un alto nivel de apalancamiento durante las décadas anteriores a la crisis. Además, en los años noventa, con el proceso de apertura de la cuenta financiera, empezaron también a endeudarse en moneda extranjera.

Como vemos, la curva que representa la relación entre nivel de inversión esperado y real presenta tres tramos: (i) para niveles reducidos de inversión esperada, las empresas quebrarán y no podrán acometer decisiones de inversión; (ii) para niveles elevados de inversión esperada, la restricción crediticia no operará y el nivel de inversión vendrá determinado por su rendimiento; y, (iii) entre ambos, la restricción crediticia es operante y la inversión efectiva se ve limitada, siendo la pendiente ($\lambda$) mayor que la de la bisectriz.

Estas tres regiones generan tres puntos de equilibrio del modelo, en los que la inversión real se iguala a la esperada. Dos puntos extremos ($L$ y $H$) que son equilibrios estables, mientras que el punto intermedio ($U$) es inestable:
\begin{la}
    \item \textbf{Equilibrio estable superior, $H$}. En el equilibrio con nivel elevado de inversión se invierte hasta el punto en el que se igualen el retorno de la inversión doméstica y extranjera ($i_t=i^{\max}$). Existe si, con el tipo de cambio real que corresponde a esa inversión, la restricción de apalancamiento no impide alcanzarla: $(1+\theta)\left(\alpha y_t-b_t^d-p_t^Hb_t^f\right)\geq i^{\max}$;
    \item \textbf{Equilibrio estable inferior, $L$}. En el equilibrio de bajo nivel los prestamistas no ofrecen financiación porque creen que los empresarios no disponen de colateral. Esto provoca una depreciación del tipo de cambio real que a su vez genera la quiebra de los empresarios ($i_t=0$). Existe si, con el tipo de cambio real que corresponde a una inversión nula, la riqueza de los empresarios es efectivamente negativa: $\alpha y_t-b_t^d-p_t^Lb_t^f\leq 0$; y,
    \item \textbf{Equilibrio inestable, $U$}. En el punto intermedio, si la inversión esperada por cualquier motivo se desplazase ligeramente hacia la derecha o hacia la izquierda, la inversión real se modificaría hasta alcanzar el punto $H$ o $L$ respectivamente.
\end{la}

donde $p_t^H$ y $p_t^L$ son los tipos de cambio reales que resultan de la ecuación de equilibrio con $i_t=i^{\max}$ y con $i_t=0$, respectivamente. Cuanto mayor es la deuda en moneda extranjera, más fácil es que se cumpla la condición de $L$ y más exigente se vuelve la de $H$: la economía es más vulnerable.

Nótese que los puntos de equilibrio estables son únicamente locales. Desde el equilibrio superior, si la inversión esperada se desplazase hasta un punto inferior a $U$, la dinámica del modelo conduciría a un nuevo equilibrio en $L$, y viceversa. De esta forma, si cualquier motivo conduce al pesimismo a los prestamistas podría producirse el colapso de la economía. La razón no es que la inversión anterior fuese poco sólida, sino la fragilidad financiera.
