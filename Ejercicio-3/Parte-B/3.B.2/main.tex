    % --- Imagen adaptativa ---
    % \imagenfit[alto]{fichero}{título}{fuente}: la imagen queda fija entre los párrafos donde se escribe (no flota), con su
    % título en el índice de figuras y la fuente debajo. Sin [alto]: las figuras PDF (TikZ) salen al tamaño con que se
    % dibujaron y solo se reducen si no caben en el ancho o en el alto de la página; las PNG/JPG ocupan el ancho del texto
    % hasta \imgaltoras del alto de la página. Con [alto] (fracción del alto de la página; [1] = página entera) se fija el
    % tamaño de esa imagen. Si no cabe en lo que queda de página, pasa entera a la siguiente.
    % \imagenfithorizontal{fichero}{título}{fuente}: la imagen en una página girada (apaisada), en ese mismo punto del texto.
    % Reglas: main/GUIA_TEMAS.md, §4 (Figuras).
    \newdimen\imgcap
    \newdimen\imgh
    \newdimen\imgw
    \newdimen\imgneed
    \newsavebox\imgcaja
    \newcommand{\imgaltoras}{0.30}% alto por defecto de las PNG/JPG (fracción del alto de la página)
    \makeatletter
    \newcommand{\imagenfit}[4][]{%
      \addvspace{10pt}%
      \par\begingroup
        \imgcap=3.5\baselineskip% alto aproximado de título + fuente
        \imgw=\linewidth
        \filename@parse{#2}\def\img@pdf{pdf}%
        \if\relax\detokenize{#1}\relax
          \ifx\filename@ext\img@pdf
            \sbox\imgcaja{\includegraphics{\detokenize{#2}}}% tamaño natural
            \ifdim\wd\imgcaja<\imgw \imgw=\wd\imgcaja \fi
            \imgh=\ht\imgcaja
          \else
            \imgh=\imgaltoras\textheight
          \fi
        \else
          \imgh=#1\textheight
        \fi
        \dimen@=\textheight \advance\dimen@ by -\imgcap
        \ifdim\imgh>\dimen@ \imgh=\dimen@ \fi
        \sbox\imgcaja{\includegraphics[width=\imgw,height=\imgh,keepaspectratio]{\detokenize{#2}}}%
        \imgneed=\ht\imgcaja \advance\imgneed by \imgcap
        \Needspace{\imgneed}%
        \noindent\begin{minipage}{\textwidth}
          \centering
          \captionof{figure}{\textemdash\ #3}\par\vspace{0.3em}% título arriba
          \usebox\imgcaja\par
          \vspace{0.3em}\small\textit{Fuente: }#4% fuente abajo
        \end{minipage}%
      \endgroup
      \par\addvspace{10pt}\RequestRestore
    }
    \newcommand{\imagenfithorizontal}[3]{%
      \begin{landscape}
        \centering
        \captionof{figure}{\textemdash\ #2}\par\vspace{0.3em}%
        \includegraphics[width=\linewidth,height=\dimexpr\textheight-4\baselineskip\relax,keepaspectratio]{\detokenize{#1}}\par
        \vspace{0.3em}{\small\textit{Fuente: }#3}%
      \end{landscape}%
    }
    \makeatother

    % \anexo{Título}: un anexo equivale a un \section (su título sale del mismo tamaño). Dentro se usan \subsection,
    % \subsubsection y \paragraph, nunca \section: se numeran I.1, I.1.1, I.1.1.1 y las páginas del anexo, I-1, I-2…
    % En el índice de contenido inicial solo figura el nombre del anexo. Sus apartados forman el índice del anexo, que sale
    % solo si el anexo los tiene, justo después de la introducción del anexo (antes del primer \subsection).
    % \anexo*{Título}: el mismo anexo, sin ese índice. Reglas: main/GUIA_TEMAS.md, §6.

    \newcommand{\anx@empezar}[2]{%
      \clearpage
      \refstepcounter{anexo}%
      \global\EnAnexotrue
      \setcounter{page}{1}\gdef\thepage{\theanexo-\arabic{page}}%
      \setcounter{section}{\numexpr\value{anexo}-1\relax}\stepcounter{section}% el anexo hace de \section
      \gdef\thesection{\Roman{section}}\gdef\theHsection{anexo.\Roman{section}}%
      \phantomsection
      \protected@xdef\leq@current@section{Anexo \theanexo.- \unexpanded{#2}}\gdef\leq@current@subsection{}% índice de ecuaciones
      \addtocontents{stc}{\protect\contentsline{section}{Anexo \theanexo.- #2}{}{\@currentHref}}% índice inicial: solo el nombre, sin página
      \addcontentsline{toc}{section}{Anexo \theanexo.- #2}% marcadores del PDF
      {\noindent\normalfont\Large\bfseries Anexo \theanexo.- #2\par}\nobreak\vspace{1.5ex \@plus .2ex}%
      #1\@ifundefined{anx@lista@\theanexo}{\global\anx@pendientefalse}{}%
    }
    % Índice del anexo: cada apartado escribe una línea en el fichero .atc (anexo, nivel, texto, página, enlace); al
    % empezar el documento se leen las de la compilación anterior y se guardan por anexo (dos pasadas, como cualquier índice).
    % En ese índice los apartados suben un nivel, porque el anexo hace de \section.
    \def\anx@entrada#1#2#3#4#5{\@ifundefined{anx@lista@#1}{\global\@namedef{anx@lista@#1}{}}{}%
      \expandafter\g@addto@macro\csname anx@lista@#1\endcsname{\contentsline{#2}{#3}{#4}{#5}}}
    \tcbuselibrary{breakable}
    \def\anx@indice{\ifanx@pendiente\global\anx@pendientefalse

            before upper={\parindent=0pt\parskip=2pt}]
          \small
          % números más anchos que en el índice inicial (I.1, I.1.1, I.1.1.1; páginas I-1)
          \def\@pnumwidth{2.6em}\def\@tocrmarg{3.4em}%
          \def\l@section##1##2{\addpenalty\@secpenalty\addvspace{1.0em \@plus\p@}\setlength\@tempdima{2.6em}%
            \begingroup\parindent\z@\rightskip\@pnumwidth\parfillskip-\@pnumwidth\leavevmode\bfseries
            \advance\leftskip\@tempdima\hskip-\leftskip##1\nobreak\hfil\nobreak\hb@xt@\@pnumwidth{\hss##2\kern-\p@\kern\p@}\par\endgroup}%

          \csname anx@lista@\theanexo\endcsname

      \fi}
    \def\anx@escribir#1#2#3{\ifEnAnexo

    % margen superior, así que no cambia las páginas del tema. Se enciende con \pagenumbering{arabic} y se apaga con

    \cs_new_protected:Npn \cab_guardar:n #1
      {
        \tl_gset:Nn \g_cab_titulo_tl {#1}
        \regex_replace_once:nnN { \A \s* Tema \s+ \S+ \s* \-\-? \s* } { } \g_cab_titulo_tl % sin «Tema X.Y.N --»
        \regex_replace_once:nnN { \.? \s* \c{\\} (\[ [^\]]* \])? \s* (\c{large} \s*)? } { .\  } \g_cab_titulo_tl % título. subtítulo
        \regex_replace_once:nnN { [\.\s]* \Z } { } \g_cab_titulo_tl % sin subtítulo: nada detrás
      }

\section{Decisiones de inversión: ¿cuándo crea valor un proyecto?}
\puntosclave{
    El valor actual neto (VAN) es una herramienta ampliamente empleada para determinar la viabilidad de una inversión, si bien asume perfecta certidumbre acerca del futuro, por lo que se complementa típicamente con otras herramientas como las simulaciones de Montecarlo. 
    
    La tasa interna de retorno (TIR) mide la rentabilidad implícita de un proyecto de inversión y, de manera similar al VAN, permite la toma de decisiones en forma de reglas. Sin embargo, su correcta aplicación depende fuertemente de las condiciones del proyecto, por ejemplo, los fondos que genere el mismo se deben reinvertir siempre a la tasa TIR. 
    
    Al momento de determinar la viabilidad de un proyecto, dada la existencia deincertidumbre acerca del valor que tomarán ciertas variables que afectan a la viabilidad del proyecto, se ha de complementar el análisis estudiando el riego de dicha inversión mediante un análisis de sensibilidad o stress test. 
    
    Todas las metodologías para la evaluación de proyectos han de emplearse de manera complementaria, ya que cada indicador recogerá cuestiones diferentes y no siempre llevarán al mismo resultado. Como en Teoría Microeconómica, las preferencias de los agentes (en este caso, ante cuestiones como el riesgo) serán determinantes para la elección del proyecto de inversión.
}


Invertir es renunciar hoy a unos recursos a cambio de unos flujos futuros. Por eso toda decisión de inversión se somete a fuerzas que operan en sentido contrario. De un lado está la \textbf{rentabilidad}, que permite evaluar si el capital comprometido en un proyecto generará un rendimiento capaz de justificar la inversión. De otro, dos fuerzas que la contrapesan. La \textbf{intertemporalidad} exige tener en cuenta el coste de oportunidad de comprometer un determinado volumen de capital en un proyecto concreto durante un periodo determinado, es decir, el rendimiento al que se renuncia en la mejor alternativa disponible.\footnote{%
    Que un euro futuro valga menos que uno presente puede parecer trivial, pero para un economista no lo es: es el \textbf{valor temporal del dinero}, que depende tanto de la inflación como de la existencia de inversiones alternativas. Unas oportunidades de inversión con alto rendimiento o una inflación elevada llevan a descontar los flujos de caja lejanos con más intensidad que los que el inversor recibe en fechas cercanas. Para que el descuento sea coherente, los flujos nominales se descuentan con tipos nominales y los reales con tipos reales, que se relacionan por la ecuación de FISHER (1930): $(1+i)=(1+i_r)(1+\pi)$, con $i$ el tipo nominal, $i_r$ el real y $\pi$ la inflación.
} La \textbf{información} exige considerar la forma en que las contingencias futuras pueden alterar el curso del proyecto y, sobre todo, su rentabilidad.

Ambas fuerzas se condensan en la tasa con que se descuentan los flujos del proyecto, que mide el coste de oportunidad del capital, $r=\gamma+\lambda$: $\gamma$ remunera el tiempo (es el tipo de interés sin riesgo) y $\lambda$ remunera el riesgo (es una prima). Pero la información no actúa solo contra la rentabilidad. La incertidumbre resta valor al proyecto cuando es un riesgo que los inversores no pueden eliminar diversificando y exigen que se les remunere; y se lo añade cuando la empresa puede esperar a ver cómo se resuelve y adaptar sus decisiones, porque esa flexibilidad vale más cuanto mayor es la incertidumbre. De ahí la tesis de este bloque: con información perfecta y mercados de capitales perfectos, el valor actual neto es el criterio correcto para decidir, y los demás criterios lo son solo en la medida en que coinciden con él; con información imperfecta, sigue siéndolo únicamente si incorpora el valor de la información que se obtendrá a lo largo de la vida del proyecto, y un valor actual neto que lo ignora infravalora los proyectos flexibles.

Por eso distinguimos dos contextos. Con información perfecta, la única fuerza que se opone a la rentabilidad es el tiempo, y los criterios clásicos de valoración, con sus virtudes y sus problemas, se ven con nitidez. Con información imperfecta aparece la dimensión informativa, que extiende esos criterios en dos direcciones: cuánto vale la información que falta y cómo medir el efecto de la incertidumbre sobre el resultado del proyecto. Cuál es la tasa de descuento adecuada, el coste del capital, se aborda en el bloque siguiente.


\subsection{Información perfecta: la rentabilidad frente al coste del tiempo}
Con información perfecta, quien decide conoce con certeza los flujos de caja del proyecto y los tipos de interés a los que puede prestar y endeudarse en cada periodo. El único coste de oportunidad es el del tiempo, y la tasa de descuento es el tipo de interés sin riesgo, $r_t=\gamma_t$. En este marco se presentan los tres criterios básicos de evaluación de proyectos: el valor actual neto (VAN), la tasa interna de retorno (TIR) y el periodo de recuperación o \textit{payback}. Los tres se aplican en las primeras etapas de la evaluación y persiguen el mismo fin, estudiar la viabilidad de una inversión, pero responden a preguntas distintas: el VAN mide cuánto valor crea el proyecto, en unidades monetarias; la TIR, qué rentabilidad implícita ofrece, en tanto por ciento; y el payback, cuánto tarda en devolver lo invertido, en unidades de tiempo. Por eso no siempre llegan al mismo resultado y se emplean de forma complementaria. No tienen, sin embargo, el mismo rango: como veremos, solo el VAN se deriva directamente del objetivo de maximizar el valor de la empresa para sus propietarios, y la TIR y el payback son útiles en la medida en que coinciden con él o aportan información que el VAN no recoge.\footnote{%
    En la práctica se usan los tres. En la encuesta de GRAHAM y HARVEY (2001) a $392$ directores financieros estadounidenses, el $74,9\%$ declaraba emplear siempre o casi siempre el VAN, el $75,7\%$ la TIR y el $56,7\%$ el payback\textcolor{orange}{\textsuperscript{[Probable]}}. Que la TIR se use tanto como el VAN, pese a sus problemas, se explica en parte porque una rentabilidad en tanto por ciento es más fácil de comunicar y de comparar con el coste de los fondos\textcolor{orange}{\textsuperscript{[Probable]}}.
}

Los tres criterios comparten su materia prima, los \textbf{flujos de caja} del proyecto, cuya obtención no es evidente en la práctica. Lo relevante son los flujos incrementales: la diferencia entre los cobros y pagos de la empresa con el proyecto y sin él.\footnote{%
    De ahí cuatro reglas prácticas: (i) los costes hundidos, ya incurridos e irrecuperables, no se cuentan, porque no dependen de la decisión; (ii) los recursos propios que el proyecto ocupa (un terreno, una nave) se valoran por su coste de oportunidad, aunque no supongan un pago; (iii) se incluyen los efectos sobre el resto de la empresa, como la pérdida de ventas de otros productos propios; y, (iv) no se incluyen los flujos de la financiación (intereses, devolución de la deuda, dividendos), porque su coste ya está en la tasa de descuento y contarlos dos veces sesgaría el resultado (BREALEY, MYERS y ALLEN, 2020).
} Al evaluar una inversión deben considerarse tres tipos: (i) flujos de explotación;\footnote{%
    La contribución de la inversión al resultado bruto de explotación (EBITDA, por sus siglas en inglés: beneficio antes de intereses, impuestos, depreciaciones y amortizaciones) es la diferencia entre los ingresos y los gastos adicionales que genera, sin contar depreciaciones ni amortizaciones. Del EBITDA se deduce el impuesto teórico sobre el resultado de explotación adicional, que se calcula multiplicando el tipo impositivo que soporta el proyecto por la variación del resultado de explotación (después de amortizaciones, que son fiscalmente deducibles) y teniendo en cuenta los créditos fiscales por pérdidas.
} (ii) flujos de inversión;\footnote{%
    La inversión se entiende en sentido amplio: incluye desde el fondo de maniobra (\textit{working capital}) hasta los activos fijos. El fondo de maniobra es la diferencia entre los activos corrientes de la empresa (efectivo, existencias y cuentas por cobrar) y sus pasivos corrientes (cuentas por pagar y deudas a corto plazo); representa los fondos disponibles para las operaciones diarias y es una medida importante de su liquidez y de su salud financiera ([\authorfont{Ver Tema 3.B.1}]). Un fondo de maniobra positivo indica que la empresa tiene recursos a corto plazo suficientes para cumplir sus obligaciones; uno negativo puede anticipar tensiones de liquidez. Las variaciones del fondo de maniobra que exige el proyecto se deducen del EBITDA. La inversión en activos fijos comprende la inversión en capacidad de producción y crecimiento, ya sea en activos tangibles (maquinaria, terrenos, edificios), intangibles (investigación y desarrollo, patentes y licencias, capital empresarial) o financieros (acciones de filiales) para el crecimiento externo.
} y, (iii) flujos extraordinarios.

\subsubsection{Valor actual neto (VAN): el valor que crea el proyecto}
El primer criterio, y el que sirve de referencia a los demás, es el \textbf{valor actual neto}. Se fundamenta en el valor temporal del dinero: para determinar si un proyecto es viable, trae al presente todos sus flujos de caja netos, descontándolos al coste de oportunidad del tiempo, y les resta la inversión inicial. Con un tipo de descuento constante, el criterio del valor actual de FISHER (1930) se escribe:\footnote{%
    Calibrar sus componentes puede resultar difícil en la práctica. Si los tipos difieren entre periodos, cada flujo se descuenta con el producto de los factores de los periodos que median hasta su cobro, $VAN=-I_0+\sum_{t=1}^{T}CF_t\big/\prod_{s=1}^{t}(1+r_s)$, o, de forma equivalente, con el tipo de contado a plazo $t$ de la curva de tipos de interés. Suponer un tipo constante equivale a suponer que esa curva es totalmente plana, algo poco realista que aquí se adopta por simplicidad ([\authorfont{Ver Tema 3.B.24}]). Un método básico para fijar la tasa de descuento es tomar la curva de tipos del activo sin riesgo y sumarle una prima que depende de las características del proyecto, como veremos al introducir la incertidumbre y en el bloque siguiente.

    En tiempo continuo, el valor actual en $t$ de un flujo $CF_s$ que se cobra en $s$ es $CF_s\,e^{-\gamma_{t,s}(s-t)}=CF_s\,B(t,s)$, donde $\gamma_{t,s}$ es el tipo sin riesgo para el periodo $(s-t)$ y $B(t,s)$, el precio en $t$ de un bono que paga $1$ en $s$. Con riesgo, el flujo se sustituye por el previsto, $\mathbb E_t[CF_s]$, y el descuento añade la prima de riesgo del proyecto: $CF_s\,e^{-(\gamma_{t,s}+\lambda)(s-t)}=CF_s\,B(t,s)\,e^{-\lambda(s-t)}\approx CF_s/(1+\gamma_{t,s}+\lambda)^{s-t}$.
}
$$VAN=-I_0+\sum_{t=1}^{T}\frac{CF_t}{(1+r)^t},$$
donde $I_0$ es la inversión inicial, $CF_t$ el flujo de caja neto del periodo $t$, $T$ la vida del proyecto y $r$ la tasa de descuento, que con información perfecta es el tipo de interés sin riesgo. El VAN es, por tanto, una \textbf{medida de rentabilidad absoluta}: en unidades monetarias, mide cuánto aumenta (o disminuye) el valor de la empresa si se acomete la inversión.\footnote{%
    Si además los flujos son constantes, $CF_t=CF$, la expresión se reduce a $VAN=-I_0+CF\,\frac{1-(1+r)^{-T}}{r}$. De este caso se derivan otros de interés: la renta perpetua, cuyo valor actual es $CF/r$, y los flujos que crecen a una tasa constante $g<r$, cuyo valor actual es $CF_1/(r-g)$. Para estas extensiones, véase el capítulo $16$ de VERNIMMEN, QUIRY y LE FUR (2022)\textcolor{orange}{\textsuperscript{[Probable]}}.
}

La regla de decisión se sigue de esa interpretación. Si el VAN es positivo, el proyecto rinde más que su coste de oportunidad: obtener sus flujos en el mercado de capitales exigiría invertir más de lo que cuesta el proyecto, y acometerlo aumenta la riqueza de los propietarios en esa diferencia. Si es negativo, el proyecto destruye valor, porque los mismos fondos rendirían más en el mercado. Si es nulo, rinde exactamente su coste de oportunidad: no crea ni destruye valor, y la empresa es indiferente.\footnote{%
    \textbf{NOTA.-} Un VAN nulo equivale al beneficio normal de la teoría microeconómica: el capital obtiene justo su remuneración de mercado. Un VAN positivo es, por tanto, una renta económica, y en un mercado competitivo no debería abundar: los proyectos con VAN positivo suelen apoyarse en alguna ventaja que los competidores no pueden replicar de inmediato, como una patente, una tecnología, una posición de mercado o un coste menor. Un VAN muy elevado sin una ventaja que lo explique es, más a menudo, señal de un error en la estimación de los flujos que de una oportunidad excepcional.
} Por la lógica del arbitraje, sería deseable acometer todo proyecto con VAN positivo.

¿Por qué maximizar el VAN, y no la satisfacción de unos propietarios que pueden tener preferencias muy distintas sobre cuándo consumir? La respuesta es el \textbf{teorema de separación} de FISHER (1930), que formalizó HIRSHLEIFER (1958). Supongamos un propietario que dispone hoy de unos recursos $Y_0$, que puede invertir en proyectos de rendimiento decreciente (la frontera de inversión) o prestar y pedir prestado sin límite al tipo $r$ (la recta de mercado). La decisión se separa en dos etapas. Primero, la empresa invierte hasta que el rendimiento del último euro invertido iguala el tipo de interés: es el punto $P^*$, en que la frontera es tangente a una recta de pendiente $-(1+r)$. Esa inversión, $I^*$, es la que maximiza el valor actual de la riqueza, $W_0=Y_0+VAN$. Después, cada propietario se desplaza a lo largo de la recta de mercado hasta el consumo que prefiere, prestando si es paciente ($a$) o endeudándose si es impaciente ($b$). Como la recta de mercado más alejada del origen es la mejor para todos, cualesquiera que sean sus preferencias, todos los propietarios aprueban por unanimidad la misma decisión: maximizar el VAN. La empresa puede delegarla en sus directivos sin consultar a los propietarios, con una regla que solo exige conocer los flujos y el tipo de interés.

\imagenfit{fisher_separacion.pdf}{Separación de FISHER: la inversión óptima no depende de las preferencias}{Elaboración propia a partir de FISHER (1930) y HIRSHLEIFER (1958).}

El VAN tiene, además, una propiedad que ningún otro criterio comparte: es aditivo. El VAN de dos proyectos independientes es la suma de sus VAN, de modo que cada proyecto puede evaluarse por separado y el valor de la empresa es la suma del de sus proyectos. En definitiva, la principal virtud del VAN es que reduce un proyecto de inversión a una suma de flujos de caja descontados, lo que facilita el análisis objetivo de las alternativas e introduce, de forma inequívoca, el criterio más relevante: la creación de valor para la empresa.

El teorema descansa, sin embargo, en un mercado de capitales perfecto,\footnote{%
    HIRSHLEIFER (1958) mostró que, si el tipo al que la empresa se endeuda es mayor que el tipo al que presta, la separación se rompe: ya no hay una única recta de mercado, y la inversión óptima pasa a depender de las preferencias de los propietarios.
} y sus fricciones limitan la validez del resultado: la empresa puede no tener acceso ilimitado al mercado de capitales, pueden existir incompatibilidades entre proyectos, o costes de gestión y restricciones de capacidad que impidan acometer todos los proyectos con VAN positivo. Para fijar ideas, seguiremos a lo largo del bloque un mismo ejemplo. Una empresa estudia dos proyectos. El proyecto A exige hoy una inversión $I_A$ y devuelve un único flujo $CF_A$ al cabo de un periodo; el proyecto B, mayor y más lento, exige $I_B>I_A$ y devuelve un único flujo $CF_B$ al cabo de dos periodos. Aplicando la expresión del VAN, sus valores son
$$VAN_A=-I_A+\frac{CF_A}{1+r},\qquad VAN_B=-I_B+\frac{CF_B}{(1+r)^2}.$$
Supongamos que, al tipo de interés vigente, los dos son positivos y $VAN_B>VAN_A$. Si los proyectos son independientes y la empresa puede financiarse sin límite, acomete los dos, porque cada uno suma su VAN al valor de la empresa. Si son \textbf{excluyentes} (dos usos alternativos del mismo solar, dos tecnologías para la misma planta), elige B aunque exija más inversión: si eligiese A, la diferencia $I_B-I_A$ que deja de invertir solo podría colocarse en el mercado, con VAN nulo, y el valor total sería $VAN_A<VAN_B$. Cuando el límite es el capital disponible, la pregunta cambia.

\paragraph{Índice de rentabilidad: el VAN por euro invertido}
El VAN mide cuánto valor crea un proyecto, pero no cuánto capital necesita para crearlo. Cuando el capital es escaso, esa segunda dimensión importa, y la recoge el \textbf{índice de rentabilidad}, una extensión del VAN que lo divide por la inversión inicial del proyecto $p$:
$$R_p=\frac{VAN_p}{I_0^p}.$$
Es positivo si y solo si lo es el VAN, y mide la riqueza creada por cada euro invertido: una suerte de multiplicador del desembolso inicial sobre el VAN.\footnote{%
    Algunos textos lo definen como el cociente entre el valor actual de los flujos y la inversión, $VA_p/I_0^p=1+R_p$; el umbral de aceptación es entonces $1$, y no $0$. En el anexo de ejemplos numéricos se calcula con cifras.
} Mientras el VAN mide la rentabilidad en términos absolutos, el índice la mide en términos relativos, por lo que es complementario del VAN y no su sustituto. En el ejemplo, si B exige mucha más inversión que A, puede ocurrir que $R_A>R_B$ aunque $VAN_B>VAN_A$: A crea más valor por euro, pero B crea más valor en total. Si los proyectos son excluyentes y no hay restricción de capital, se elige B: un índice más alto no implica una inversión mejor.

La utilidad propia del índice aparece con el \textbf{racionamiento de capital}. Si la empresa dispone de un presupuesto $K$ que no le permite acometer todos los proyectos con VAN positivo, su problema es maximizar el VAN total sin que la inversión supere $K$. Con una sola restricción y proyectos divisibles, basta ordenar los proyectos de mayor a menor índice y aceptarlos hasta agotar el presupuesto. El índice es el VAN por unidad del recurso escaso, como el margen por unidad de un factor escaso en la teoría de la empresa, y el índice del último proyecto aceptado es el precio sombra del capital: el VAN que añadiría un euro más de presupuesto. En la figura, el índice de cada proyecto es la pendiente del radio que lo une con el origen; con un presupuesto $K=I_B$, A más la fracción $\theta=(K-I_A)/I_B$ de B que cabe en el resto crean más valor que B solo, porque cada euro invertido en A rinde más que en B.

\imagenfit{indice_rentabilidad.pdf}{Índice de rentabilidad y racionamiento de capital}{Elaboración propia a partir de LORIE y SAVAGE (1955).}

Esa ordenación falla en dos casos, que estudiaron LORIE y SAVAGE (1955) y WEINGARTNER (1963)\textcolor{orange}{\textsuperscript{[Probable]}}. El primero son los proyectos \textbf{indivisibles}. En el ejemplo, si A y B no pueden fraccionarse y el presupuesto es $K=I_B$, el índice recomienda A, que deja sin invertir $K-I_A$; ese resto solo rinde el tipo de interés, con VAN nulo, y la empresa obtiene $VAN_A<VAN_B$. Salvo que exista otro proyecto que quepa en el resto y compense la diferencia, B es mejor pese a su menor índice: la elección deja de ser una ordenación y se convierte en un problema de programación entera, el problema de la mochila. El segundo son las restricciones en \textbf{varios periodos} o sobre varios recursos, en que un único cociente no puede ordenar los proyectos. WEINGARTNER (1963) formuló entonces la selección como un programa matemático cuyos precios duales miden el valor de relajar el presupuesto de cada periodo.

Que haya racionamiento es, en sí mismo, una desviación del mundo de FISHER: con un mercado de capitales perfecto, todo proyecto con VAN positivo encuentra financiación. El racionamiento puede ser externo, cuando la empresa no obtiene fondos al tipo de mercado por imperfecciones como la información asimétrica entre la empresa y quienes la financian ([\authorfont{Ver Tema 3.B.3}]); o interno, cuando la dirección fija presupuestos para disciplinar a los responsables de cada división o porque su capacidad de gestión es limitada\textcolor{orange}{\textsuperscript{[Probable]}}. En el primer caso, el índice es un criterio de segundo óptimo; en el segundo, el presupuesto es una restricción que la empresa se impone, y la pregunta relevante es si el coste de relajarla es menor que el VAN al que renuncia. El índice relativiza el VAN por la escala del proyecto; la TIR, el criterio siguiente, va más allá y prescinde también de la tasa de descuento.


\subsubsection{Tasa interna de retorno (TIR): la rentabilidad implícita del proyecto}
El segundo criterio, que en la práctica se usa tanto como el VAN, es la \textbf{tasa interna de retorno}, que mide la \textbf{rentabilidad implícita de una inversión}. Es la tasa de descuento que iguala el valor actual de los flujos de caja del proyecto con su inversión inicial, es decir, la que anula su VAN; es lo que KEYNES (1936) llamó eficiencia marginal del capital ([\authorfont{Ver Tema 3.A.34}]):\footnote{%
    La ecuación es un polinomio de grado $T$ en $x=1/(1+TIR)$, de modo que, salvo en proyectos de uno o dos periodos, no tiene solución explícita y se resuelve por aproximaciones sucesivas. En el ejemplo, $TIR_A=CF_A/I_A-1$ y $TIR_B=\sqrt{CF_B/I_B}-1$. El anexo de ejemplos numéricos la calcula para un proyecto de dos periodos.
}\textsuperscript{,}\footnote{%
    \textbf{OJO.-} KEYNES (1936) definió la eficiencia marginal del capital como la tasa que iguala el valor actual de los rendimientos esperados de un bien de capital con su precio de oferta, es decir, como la TIR de un proyecto. El tipo de rendimiento sobre el coste de FISHER (1930), con el que se la suele identificar, se define en cambio al comparar dos alternativas: es la tasa que iguala sus valores actuales. ALCHIAN (1955) mostró que coincide con la TIR del proyecto diferencial, que veremos al tratar el punto de FISHER.
}
$$-I_0+\sum_{t=1}^{T}\frac{CF_t}{(1+TIR)^t}=0.$$
Para un proyecto, la TIR es lo que la rentabilidad al vencimiento es para un bono ([\authorfont{Ver Tema 3.B.24}]): el rendimiento que obtiene quien paga hoy $I_0$ a cambio de la serie de flujos.

La regla de decisión compara esa rentabilidad con el coste de oportunidad: el proyecto se acepta si su TIR supera la tasa de descuento, $TIR>r$, y se rechaza si es inferior. La lógica es la del arbitraje: si el proyecto rinde más de lo que los mismos fondos obtendrían en el mercado, crea valor. La TIR es, así, el mayor coste del capital que el proyecto puede soportar sin destruir valor, y la diferencia $TIR-r$ mide su margen de seguridad frente a una subida de ese coste. Cuando el VAN decrece con la tasa de descuento, como en un proyecto en que primero se invierte y después se cobra, la regla coincide con la del VAN: $VAN(r)>0$ si y solo si $r<TIR$.

De ahí sus dos ventajas como complemento del VAN, con el que comparte el objetivo de comparar rentabilidades con el coste del capital. La primera es que expresa la creación de valor en unidades de tipo de interés y no en unidades monetarias, y que puede calcularse sin conocer la tasa de descuento, que solo hace falta para compararla: sortea así, en parte, la dificultad de medir el coste del capital, porque basta saber si es mayor o menor que la TIR. La segunda es que, como el índice de rentabilidad, permite comparar la rentabilidad de proyectos de distinta escala, entre los que el VAN, por construcción, favorece a los grandes aunque rindan menos por euro invertido. Esa segunda ventaja es también su principal riesgo, porque, cuando los proyectos son excluyentes, el de mayor TIR no es necesariamente el que más valor crea. Los dos resultados que siguen se obtienen al relacionar el VAN y la TIR a través del \textbf{perfil del VAN}, el VAN como función de la tasa de descuento.


\paragraph{El perfil del VAN: cuándo la TIR no existe o no es única}
La TIR es el punto en que el perfil del VAN, la función $VAN(r)$, corta el eje. Que ese punto exista, sea único y separe los tipos que crean valor de los que lo destruyen depende de la forma del perfil, y esta, del calendario de signos de los flujos. En un proyecto convencional, en que los flujos negativos se acumulan en los primeros periodos y los positivos en los últimos, el perfil decrece con el tipo de descuento, porque al subir $r$ pierden más valor los cobros, que son lejanos, que los pagos, que son próximos. Hay entonces una sola TIR, el VAN es positivo a su izquierda y negativo a su derecha, y la regla de la TIR coincide con la del VAN (panel a). Es el caso de A y de B en el ejemplo.

Si los flujos cambian de signo más de una vez, esa coincidencia se pierde. La regla de los signos de DESCARTES acota el número de TIR por el de cambios de signo en la serie de flujos: con dos cambios puede haber dos. Supongamos que A obliga a restaurar el terreno al cabo de dos periodos, con un coste $C$. Sus flujos pasan a ser $-I_A$, $+CF_A$ y $-C$, y, si la restauración es lo bastante costosa, el perfil crece primero y decrece después: el VAN es negativo con tipos bajos, que dan mucho peso al coste final, y con tipos altos, que restan valor al cobro, y es positivo entre sus dos TIR (panel b).\footnote{%
    Con $x=1/(1+r)$, $VAN_A=-I_A+CF_A\,x-C\,x^2$ es una parábola en $x$. Tiene dos TIR positivas si es negativa en $r=0$, $CF_A<I_A+C$, y positiva en su máximo, $CF_A^2>4\,I_A\,C$. LORIE y SAVAGE (1955) ilustraron el caso con una bomba que acelera la extracción de un pozo de petróleo: adelanta al primer año un ingreso que, sin ella, se obtendría en el segundo (el anexo de ejemplos numéricos da sus cifras).
} ¿Cuál es entonces la verdadera TIR? Ninguna de las dos mide la rentabilidad del proyecto, porque el proyecto no es una inversión pura: después del cobro, la empresa debe un pago futuro, como quien ha recibido un préstamo, y para esa parte un tipo de interés más alto es favorable. La TIR mezcla en un solo número una inversión y una financiación, y por eso deja de ordenar. La decisión correcta la sigue dando el VAN: el proyecto crea valor si el coste del capital está entre las dos TIR.

En el caso extremo, la TIR no existe. Un proyecto en que el cliente paga un anticipo, la empresa ejecuta la obra y cobra el resto al final tiene flujos $+$, $-$, $+$, y puede tener un VAN positivo a cualquier tipo de descuento sin cortar nunca el eje (panel c): la TIR no dice nada, mientras que el VAN dice que se acometa.\footnote{%
    La inexistencia de la TIR, y otras cuestiones de su cálculo, se tratan en el capítulo $17$ de VERNIMMEN, QUIRY y LE FUR (2022)\textcolor{orange}{\textsuperscript{[Probable]}}.
}

\imagenfit{perfil_van.pdf}{El perfil del VAN y la TIR según el calendario de signos de los flujos}{Elaboración propia a partir de LORIE y SAVAGE (1955).}

La enseñanza es que la TIR es una propiedad de la serie de flujos, no de la decisión: solo es un criterio fiable cuando el perfil del VAN es decreciente. Incluso entonces, como veremos a continuación, puede ordenar mal proyectos excluyentes.

\paragraph{El punto de FISHER: cuándo VAN y TIR ordenan distinto}
El segundo resultado aparece al comparar proyectos excluyentes, aunque ambos sean convencionales y tengan el mismo riesgo. En el ejemplo, A es pequeño y rápido y B, grande y lento. Supongamos que B crea más valor sin descontar, $CF_B-I_B>CF_A-I_A$, pero que A tiene mayor TIR. Sus perfiles se cortan: con tipos bajos, B tiene mayor VAN; con tipos altos, A. El tipo en que se cortan, en que los dos proyectos tienen el mismo VAN, es la \textbf{tasa de FISHER}, $r_F$, y la decisión correcta depende del lado en que quede el coste del capital: si $r<r_F$, se elige B, aunque su TIR sea menor; si $r>r_F$, se elige A. La TIR, en cambio, elige A siempre, y se equivoca en todo el tramo $r<r_F$. Por eso el punto de FISHER es útil cuando hay que elegir entre proyectos que no pueden acometerse a la vez: dice para qué tipos de descuento conviene cada uno.

\imagenfit{punto_fisher.pdf}{El punto de FISHER: VAN y TIR ordenan distinto}{Elaboración propia a partir de FISHER (1930).}

La tasa de FISHER es la TIR del \textbf{proyecto diferencial}, $B-A$, cuyos flujos son la diferencia entre los de B y los de A: es la rentabilidad de los fondos adicionales que exige elegir B en lugar de A.\footnote{%
    De ahí que coincida con el tipo de rendimiento sobre el coste de FISHER (1930), definido precisamente al comparar dos alternativas (ALCHIAN, 1955). En el ejemplo, los flujos del proyecto diferencial son $-(I_B-I_A)$, $-CF_A$ y $+CF_B$; como solo cambian de signo una vez, su TIR es única. La tasa de FISHER no existe cuando un proyecto tiene mayor VAN que el otro a cualquier tipo de descuento, y entonces VAN y TIR coinciden en la elección.
} Ese es el criterio que reconcilia ambos instrumentos. Elegir B equivale a acometer A más el proyecto diferencial, y este conviene si su TIR supera el coste del capital, es decir, si $r<r_F$. Aplicada a los incrementos, la TIR lleva a la misma decisión que el VAN.

¿Por qué se cortan los perfiles? Por dos diferencias entre los proyectos. La primera es la \textbf{escala}: B exige más inversión y, con tipos bajos, genera más valor en euros, aunque menos por euro. La segunda es el \textbf{calendario}: los flujos de B llegan más tarde, y el valor de los flujos lejanos es más sensible al tipo de descuento. La pendiente del perfil es proporcional al valor actual de los flujos y a su duración, el plazo medio ponderado en que se cobran, de modo que un proyecto lento se comporta como un bono largo: su valor cae más cuando sube el tipo de interés.\footnote{%
    Derivando, $dVAN/dr=-\sum_{t}t\,CF_t/(1+r)^{t+1}=-D\cdot VA/(1+r)$, donde $VA$ es el valor actual de los flujos y $D$, su duración de MACAULAY ([\authorfont{Ver Tema 3.B.24}]).
}

El punto de FISHER ilustra, así, la sensibilidad del VAN al tipo de descuento y la necesidad de emplear el VAN y la TIR como indicadores complementarios, pero la regla de selección debe aplicarse con cautela. Hereda los supuestos de los criterios que la componen, como la certeza y el reduccionismo con que resumen las características de los proyectos. Si los proyectos difieren en riesgo, compararlos con un mismo tipo de descuento no es correcto, porque cada uno tiene su coste del capital; y si difieren en duración, la comparación debe tener en cuenta qué ocurre cuando termina el más corto.\footnote{%
    Si A dura un periodo y B dos, elegir A deja abierta la posibilidad de invertir de nuevo al cabo de un periodo. Cuando los proyectos se repiten, se comparan cadenas de reemplazo de igual duración o el VAN expresado como anualidad equivalente: la renta constante que, durante la vida de cada proyecto, tiene su mismo valor actual (BREALEY, MYERS y ALLEN, 2020).
}

\paragraph{Limitaciones de la TIR}
Los supuestos de fondo de la TIR son los del VAN (certeza, flujos independientes y maximización del valor), y sus limitaciones comunes se tratan más adelante. Pero la TIR tiene, además, limitaciones propias, que nacen de resumir un proyecto en una tasa y no en un valor.

La primera, como vimos, es que puede no existir o no ser única cuando los flujos cambian de signo más de una vez, y entonces no ordena los proyectos. La segunda es que prescinde de la escala y del calendario: ordenar proyectos excluyentes por su TIR favorece a los pequeños y a los rápidos, y la decisión correcta exige calcular la TIR del proyecto diferencial, lo que en la práctica equivale a hacer el análisis del VAN. La tercera es que no es aditiva: la TIR de dos proyectos acometidos juntos no es la suma ni la media de sus TIR, de modo que no puede evaluarse cada proyecto por separado y agregar, y un proyecto malo puede esconderse dentro de uno bueno. La cuarta es que se compara con una sola tasa: si el coste del capital varía con el plazo, porque la curva de tipos no es plana, no hay un único umbral con el que comparar la TIR, mientras que el VAN descuenta cada flujo con su tipo.

La quinta es objeto de debate: la \textbf{tasa de reinversión}. SOLOMON (1956) atribuyó los conflictos entre VAN y TIR a que cada criterio supone implícitamente que los flujos intermedios se reinvierten a una tasa distinta: el VAN, al tipo de descuento, y la TIR, a la propia TIR. Si esa tasa es irreal, como en un proyecto con una TIR muy alta, la TIR exagera la rentabilidad. De ahí la \textbf{TIR modificada}, que capitaliza los flujos intermedios al coste del capital hasta el final del proyecto y calcula la tasa que iguala ese valor final con la inversión inicial, y que tiene siempre una única solución. DUDLEY (1972)\textcolor{orange}{\textsuperscript{[Probable]}} y KEANE (1979) objetaron que ninguno de los dos criterios supone nada sobre la reinversión: ambos se calculan solo con los flujos del proyecto, y la TIR es una propiedad de esa serie, como la rentabilidad al vencimiento de un bono, que no depende de lo que su tenedor haga con los cupones. La reinversión solo importa al comparar proyectos con calendarios distintos, y entonces la pregunta relevante es qué rinden los fondos que uno libera antes que el otro: en un mercado de capitales perfecto, el coste del capital, que es lo que supone el VAN. El debate no cambia, por tanto, la conclusión: los conflictos se resuelven a favor del VAN, y la TIR es útil como información complementaria sobre el margen de rentabilidad del proyecto. El criterio siguiente, el periodo de recuperación, renuncia a ambos: no pregunta cuánto rinde el proyecto, sino cuándo devuelve lo invertido.


\subsubsection{Regla del periodo de recuperación: Payback}
\textcolor{magenta}{La última técnica que presentaremos para evaluar la rentabilidad de los proyectos de inversión (en certidumbre) es el payback. Este indicador determina el \textbf{tiempo necesario para recuperar la inversión} inicial ($I_0$) y, como tal, se expresa en unidades de tiempo (años o meses) dependiendo de periodicidad de los flujos de caja. En otras palabras: estima el tiempo requerido para que los flujos de caja del proyecto generen, al menos, la inversión inicial. La obtención del payback resulta de despejar $t$ de la siguiente fórmula:

\eqblock{
\begin{aligned}
    VAN=0=-I_0+\sum_{t=1}^T\frac{CF_t}{(1+r)^t}\quad\underset{\text{\scriptsize si $CF_i=CF_j,$ $\forall i,j\in T$}}{\Longrightarrow}\quad \boxed{Payback=\frac{I_0}{CF}}
\end{aligned}
}{Payback: Priodo de recuperación. Valoración de los proyectos de inversión: Rentabilidad}

Se trata por tanto de un método de evaluación de proyectos adecuado para aquellos casos en que la empresa cuenta con restricciones en su acceso al mercado de capitales (restricciones de liquidez, alto coste del endeudamiento, etc.).\footnote{%
    La manera más sencilla de de aplicar esta herramientas es el establecimiento de un número de años (o meses), que actúe como punto de corte para aceptar o rechazar un proyecto. Luego, si el payback es menor o igual que este valor preestablecido, el proyecto se acepta. Caso contrario, la inversión es descartada.
}

No obstante, el payback no tiene en cuenta los flujos de caja que se generan después de que la inversión inicial haya sido recuperada y, por tanto, se convierte en una medida limitada para el estudio de la rentabilidad y el riesgo a largo plazo de una inversión; debiendo ser utilizado con los demás instrumentos de valoración de proyectos. Es decir, el payback no tiene por fin determinar la elección del proyecto más rentable (como harían VAN o TIR) sino complementar ese estudio para empresas con dificil acceso a la liquidez, o por grandes empresas que enfrentan proyectos menores.

Además, idealmente debería ser aplicada descontando al presente los flujos de caja futuros. No hacerlo, aunque sea más sencillo y comúnmente aceptado por los analistas, implica un reduccionismo excesivo y únicamente valdría como aproximación en contextos con baja inflación y coste de oportunidad reducido, ya que se está omitiendo el problema intertemporal del valor del dinero.

\begin{specialblock}{Ejemplo -- Payback rule}
    Supongamos que una empresa está contemplando la instalación de una nueva maquinaria que requiere una inversión inicial de $100$ mil unidades monetarias (UM) y, durante los próximos 5 años, se espera que genere flujos de caja anuales de $30$ mil UM. Para calcular el payback de esta inversión:

    \begin{center}
    \begin{tabular}{|p{0.30\linewidth}|p{0.30\linewidth}|p{0.30\linewidth}|}
    \hline
    \textbf{Año} & \textbf{Flujo de caja} & \textbf{Flujo de caja acumulado} \\
    \hline
    0 & $-100{.}000$ &  \\
    \hline
    1 & $30{.}000$ & $30{.}000$ \\
    \hline
    2 & $30{.}000$ & $60{.}000$ \\
    \hline
    3 & $30{.}000$ & $90{.}000$ \\
    \hline
    4 & $30{.}000$ & $120{.}000$ \\
    \hline
    5 & $30{.}000$ & $150{.}000$ \\
    \hline
    \end{tabular}
    \end{center}

    Como vemos, el flujo de caja acumulado alcanza el valor de $100$ mil UM a partir del cuarto periodo. Por lo tanto, el payback de este proyecto es 4. Como se ha mencionado, el payback por sí solo no parece ser suficiente para tomar una decisión acerca de la idoneidad de una inversión sobre otra. Así, por ejemplo, si el valor de referencia para el analista es 5 años, la empresa debería materializar el proyecto (pues la inversión inicial se recupera antes del plazo límite tolerado), ello siempre condicionado a que los valores del VAN y la TIR sean satisfactorios.
    
\end{specialblock}

\subsection{Información incompleta: }
\textcolor{magenta}{Hemos comentado reiteradamente algunas de las limitaciones que enfrentan estos métodos de valoración basados en la rentabilidad del proyecto de inversión. 

Sin embargo, una de sus limitaciones fundamentales es su \textbf{carácter estático}. Estos métodos asumen implícitamente que la decisión de inversión es irreversible y que la información relevante está disponible ex-ante. 

Sin embargo, en muchos proyectos reales, la información se revela de forma progresiva y las decisiones óptimas pueden variar a lo largo del tiempo. Esta es una derivada lógica de uno de los fenómenos que perturban el mundo real: la \textbf{información imperfecta}. 

Presentamos alternativas de valoración en un contexto de información imperfecta, pudiéndose modelizar de dos formas.

\subsubsection{Contingencias futuras: inconsistencias dinámica}
En primer lugar, mediante problemas de inconsistencia dinámica, en los que una inversión que resulta rentable en un momento temprano puede dejar de serlo en un momento futuro debido a las interrelaciones que existen con el contexto.

En este contexto, resulta necesario introducir herramientas que permitan modelizar explícitamente la secuencia de decisiones y la llegada de información, como los árboles de decisión o las opciones reales, que capturan el valor de la flexibilidad y la adaptación.

\paragraph{Riesgo: Árboles de deicisión} 
El árbol de decisión es un instrumento de gran importancia en el ámbito financiero al posibilitar la evaluación de múltiples alternativas de inversión bajo diversos escenarios, permitiendo ahorrar tiempo y recursos en la selección de proyectos.\footnote{%
    En estas notas limitamos la referencia a los árboles a un instrumento gráfico, Sin embargo, puede emplearse para construir modelos mucho más sofisticados. Como árbol es por definición un objeto matemático/computacional recursivo, y por tanto serviría también como herramienta para otras cuestiones como hacer simulaciones, resolver problemas de optimización dinámica o predecir resultados utilizando machine learning.
} Un árbol de decisiones se compone de diferentes tipos de nodos, los cuales pueden representar decisiones, riesgo y resultados en una serie de etapas del problema analizado. De esta forma, podemos representar de forma compacta, atendiendo a las normas de la probabilidad, el valor esperado de un proyecto expresado mediante una estructura de árbol como:\footnote{%
    Formalmente, tenemos tres tipos de nodos:
\begin{la}
    \item \textbf{Nodos de decisión}. Que simbolizan las opciones de acción disponibles para el inversor;
    \item \textbf{Nodos de resultado}. Que denotan los resultados finales del proceso de toma de decisiones; y,
    \item \textbf{Conjuntos de información: nodos de riesgo}. Que ilustran situaciones en las que el resultado no está completamente determinado.
\end{la}
}

\eqblock{
\begin{aligned}
    \mathbb E[VAN_p]\equiv\max u(n_0):
    \left\{\begin{aligned}
        &u(n_i)\equiv\text{Nodo terminal}\\[0.5em]
        &\sum_ip_iu(n_i)\equiv\text{Conjunto de información}\\[0.5em]
        &\max_iu(n_0)\equiv\text{Nodos de decisión}
    \end{aligned}\right.
\end{aligned}
}{Árbol de decisión: Valos actual neto del proyecto. Valoración de proyectos de inversión: Riesgo}

Este tipo de análisis brindan una evaluación sistemática y completa de múltiples factores, lo que facilita la selección de la mejor alternativa de inversión considerando tanto la rentabilidad como el riesgo asociado a cada una de ellas, descartando las opciones que no son atractivas y simplificando considerablemente el proceso de decisión; ya que, una vez desestimadas las opciones irrelevantes, el inversor solo tendrá que escoger entre las alternativas del nodo inicial.

No obstante, también presenta cierta limitaciones. En particular, está basado en dos aspectos claves: (i) la elección anticipada del sendero que seguirá el proyecto; y, (ii) la idea de que las decisiones son irreversibles. Sin embargo, en la realidad esto no suele suceder así. En general, a medida que el proyecto avanza, aparece nueva información y, junto con ello, nuevas opciones de inversión.\footnote{%
    Por ejemplo, se podría incorporar nuevas tecnologías, cancelar el proyecto o, simplemente, aplazarlo.
} Es decir, el proceso de inversión conlleva un grado de flexibilidad.

\begin{specialblock}{Ejemplo -- Árboles de decisión}
    Consideremos un inversor que debe decidir entre ampliar su planta actual o crear una nueva planta en una locación diferente. Para ambas opciones, el inversor puede escoger entre un emprendimiento \textit{Grande} o \textit{Pequeño}. Los posibles escenarios para cada alternativa se representan en el siguiente árbol de decisión. 
    
    \imagenfit{ArbolDecision.png}{Árbol de decisión}{Apuntes ICEX-CECO}
    
    Para detenninar qué opción es la más conveniente se debe estudiar una a una cada rama del diagrama. Para ello, primero el inversor debe situarse en la rama izquierda del árbol y analizar los pagos que conlleva realizar una ampliación. Si escoge una ampliación grande, la cual está situada en un nodo de incertidumbre, la inversión tendrá un VAN de $150$ UM con probabilidad $0.5$ y un VAN de $-50$ UM con igual probabilidad. 
    
    Por lo tanto, puesto a elegir entre una ampliación grande o pequeña, el inversor debería llevar a cabo una inversión pequeña, pues es la opción con un mayor VAN esperado. Procediendo de igual manera sobre la rama derecha, el inversor encuentra optimo crear una nueva planta pequeña en lugar de una grande, ya que la primera opción tiene un VAN esperado de $271.8$ UM, mientras que la segunda tiene un VAN esperado igual a cero.
\end{specialblock}

\paragraph{Incertidumbre: Opciones reales}
Como incrementar u obtener cierta versatilidad en el transcurso de una inversión tiene un valor y éste no se ve reflejado en el análisis convencional, surge la necesidad de incorporar dicho valor en los instrumentos de apoyo a las deicisones de inversión. Las opciones reales son, probablemente, la opción más popular para incorporar estas características temporales de los proectos de inversión. Representan el derecho, pero no la obligación, de modificar un proyecto de inversión ante la disponibilidad de nueva información.

Lamentablemente, si bien muchos proyectos presentan la posibilidad de incluir opciones reales, son especificas al tipo de inversión, lo que hace que presentar una teoría general sobre opciones reales este fuera del alcance. Sin embargo, en la práctica aparecen básicamente tres tipos de opciones reales:
\begin{la}
    \item \textbf{Opción de posponer}. En muchas industrias las empresas pueden considerar óptimo retrasar el comienzo de un proyecto hasta que las condiciones de mercado sean propicias (muy característico de nuestro ejemplo del petroleo). En este tipo de negocios los proyectos suelen requerir enormes inversiones y, por consiguiente, son muy sensibles a ciertas variables económicas como los precios o tasa de interés;
    \item \textbf{Opción de crecer}. A menudo los gestores deben decidir entre diferentes proyectos destinados a ajustar su producción ante posibles incrementos de demanda. Optar por un proyecto previsto para un crecimiento potencial muy alto nos expone al riesgo de que la inversión sea excesiva. Mientras que optar por un ajuste pequeño habida cuenta de un posible incremento alto de la demanda puede exponernos a altos costes si queremos ampliar la capacidad productiva. Así, las opciones reales aparecen como una forma de brindar flexibilidad al proyecto, permitiendo al inversor comenzar con una planta pequeña, pero con la opción de ampliar las instalaciones si el mercado así lo demanda. 
    \item \textbf{Opción de renunciar a una inversión}. Por último, es importante tener en cuenta la opción de abandonar la inversión en cualquier etapa del proyecto.
\end{la}

Además, independientemente del tipo de opción y proyecto de inversión que enfrentemos, sí que existen algunos métodos orientativos que podemos utilizar para establecer el \textit{valor} de una opción, uno de los cuales es el modelo de BLACK-SCHOLES. 

\eqblock{
\begin{aligned}
    &[\text{Opción de posponer}\,|\,\text{Opción de crecer}\,|\,\text{Opción de renunciar}]\\[1em]
    &\left.\begin{aligned}
        &(1)&&S^*=S-V(CF_t)\\[0.5em]
        &(2)&&V(I_0)=\frac{I_t}{(1+r)^t}\\[0.5em]
        &(3)&&d_1=\frac{\ln[S^*/V(I_0)]}{\sigma\sqrt{t}}\\[0.5em]
        &(4)&&d_2=d_1-\sigma\sqrt{t}
    \end{aligned}\right.\quad \text{donde,}
    \left\{\begin{aligned}
        &S^*&&\equiv\text{Valor actual del activo}\\[0.5em]
        &V(I_0)&&\equiv\text{Valor actual de la inversión pospuesta}\\[0.5em]
        &d_1&&\equiv\text{\textit{Cúan probable es ganar dinero}}\\[0.5em]
        &d_2&&\equiv\text{\textit{Cuánto influye la opción en el subyacente}}\\[0.5em]
        &N(d_j),\,j=1,2&&\equiv\text{Distribución acumulada para $d_j$}\\[0.5em]
        &\sigma&&\equiv\text{Volatilidad sector}\\[0.5em]
        &t&&\equiv\text{Decisión final}
    \end{aligned}\right.\\[2em]
    &\text{Valor de la opción}:\boxed{C=\overset{\scriptsize{\text{(Valor de posponer)}}}{\underbrace{S^*\cdot N(d_1)}_{\scriptsize{\text{$\mathbb E_t[VAN_{t+1}]$ ponderado por exposición}}}}-\overset{\scriptsize{\text{(Coste de posponer)}}}{\underbrace{V(I_0)\cdot N(d_2)}_{\scriptsize{\text{ $\mathbb E_t[I_t]$, ponderado por la probabilidad de ejercer la opción}}}}}\\[0.5em]
    &\hspace{8em}\Longrightarrow\text{si,}
    \left\{\begin{aligned}
        &C<\mathbb E_0[VAN]\Rightarrow\text{No posponer}\\[0.5em]
        &C\geq\mathbb E_0[VAN]\Rightarrow\text{Posponer}
    \end{aligned}\right.
\end{aligned}
}{Valoración opciones reales: Modelo BLACK-SCHOLES. Valoración de proyectos de inversión: Incertidumbre}

En resumen, las opciones reales otorgan flexibilidad a los proyectos al permitir posponer, abandonar o ampliar una inversión y están implícitamente consideradas en cualquier proyecto de inversión. 

\begin{specialblock}{Ejemplo - Opciones reales}
    Considere un proyecto que consiste en sembrar 500 hectáreas de trigo. Antes de comenzar, el inversor tiene la opción de no invertir, lo que le dará un valor de O UM. Sin embargo, si el proyecto se materializa, se deben desembolsar 1000 UM. Si esto sucede, el inversor puede obtener 2500 UM si el clima es bueno o -200 si el clima es desfavorable. Ambas alternativas pueden ocurrir con una probabilidad de 0.8 y 0.2, respectivamente. Mediante un esquema de árbol, el proceso de inversión puede representarse de la siguiente manera:

    Asumiendo que la tasa de interés es igual a cero, o lo que es lo mismo, que el valor intertemporal del dinero no cambia, el VAN esperado de esta inversión esta dado por E(VAN) = -1000 + 0.8 (2500) - 0.2(200) = 96 0 UM y, por lo tanto, de no existir restricciones a la financiación o de otro tipo, el proy ecto debería llevarse a cabo. Ahora, suponga que el inversor tiene la posibilidad de no ir al campo a controlar su cultivo una vez que conoce cómo será el clima. Así, por ejemplo, después de haber invertido 1000 UM y observar que el clima es desfavorable, el inversor puede quedarse en su casa y sortear la pérdida extra de 200 UM. Observe que el inversor también tiene la opción de quedarse en casa si el clima es bueno, pero claramente esta opción no es atractiva para él. Por lo tanto, el inversor estaría dispuesto a pagar hasta 40 UM por tener la opción de no controlar el cultivo cuando el clima es desfavorable

\subsubsection{Contingencias futuras: errores de estimación y sucesos adversos}
Otra de las formas de incorporar información incompleta en la valoración de los proyectos es mediante la imposibilidad de anticipar correctamente todas las contingencias relevantes. En la práctica, los flujos de caja, las tasas de descuento o incluso la propia estructura del proyecto están sujetos a errores de estimación y a la ocurrencia de sucesos adversos no previstos. 

En este sentido, el problema no radica tanto en la secuencia de decisiones como en la robustez del análisis ante desviaciones respecto al escenario central. Para abordar esta cuestión, se recurre a herramientas como el análisis de sensibilidad o las simulaciones de Montecarlo, que permiten evaluar cómo varían los resultados del proyecto ante perturbaciones en los supuestos iniciales, proporcionando así una medida más realista del riesgo asociado a la inversión.

\paragraph{Riesgo: análisis de sensibilidad}
El análisis de sensibilidad permite al inversor identificar aquellas variables críticas que influyen en el desempeño financiero de un proyecto, mejorando la gestión de riesgo y la toma de decisiones durante el proceso de inversión. En otras palabras, el propósito del análisis de sensibilidad es reconocer los elementos que contribuyen al éxito o fracaso de una inversión y medir cuánto lo hacen de manera marginal. 

De esta forma, el tipo de pregunta que buscamos responder con esta metodología son: ¿cuál es el escenario de costes que hace que el VAN de una inversión sea cero? ¿Cómo varía la TIR ante una caída de la demanda de un 10\%? ¿Los flujos de caja son sensibles a variaciones en los precios? 

Por tanto, el análisis de sensibilidad se realiza modificando una (algunas) variable(s) a la vez, lo que nos permite valorar con claridad el impacto de los cambios sobre la viabilidad del proyecto.\footnote{%
    Es posible realizar un análisis de sensibilidad de dos o más variables simultáneamente a fin de examinar cómo la interacción de estas variables afecta la viabilidad del proyecto. Sin embargo, es importante tener en cuenta que el análisis de sensibilidad de dos variables es considerablemente más complejo que estudiar una variable a la vez y se requiere un cuidadoso análisis para obtener resultados precisos y confiables.
} No obstante, también dificulta su formulación general puesto que estas \textit{modificaciones} pueden influir sobre el resultado de muy diversa manera y forma, no solo por incidir sobre diferentes variables sino también en función del instrumento sobre el que llevemos a cabo el análisis.

Con el fin de simplificar la exposición, podemos decir que el análisis de sensibilidad se podría llevar a cabo sobre el VAN, teniendo en cuenta las diferentes variables de las que pueden depender cada uno de los componentes de la fórmula:

\eqblock{
\begin{aligned}
    VAN_p=-I_0+\sum_{t=1}^T\frac{CF_t}{(1+r)^t}\quad\text{donde,}
    \left\{\begin{aligned}
        &CF(\succsim,\,CT(q),\,\text{Mkting},\,\text{Regulación},\,\cdot)\\[0.5em]
        &r=(\gamma+\lambda)\\[0.5em]
        &I_0=\mathbb E_t[I_0]+\underset{\text{\scriptsize componente no esperado}}{\Delta \mathbb E_t[I_0]}
    \end{aligned}\right.
\end{aligned}
}{Análisis de sensibilidad: Alteraciones sobre el cálculo del VAN. Valoración de proyectos de inversión: Riesgo}

Por ejemplo, si una pequeña caída en la demanda de los productos ($\succsim$) vuelve el VAN negativo, se dice que el proyecto es muy sensible o riesgoso a variaciones en la demanda. En síntesis, el análisis de sensibilidad es de vital importancia para una gestión adecuada del riesgo al identificar aquellos factores que podrían influir en el éxito del proyecto. De hecho, una vez tomada la decisión, el conocimiento de estos factores de riesgo resulta de gran utilidad en la elaboración de estrategias de cobertura ante cualquier eventualidad.\footnote{%
    Aunque en muchas ocasiones el peor escenario contemplado puede no ser el más adverso en la práctica. 

Un caso emblemático es el caso del Eurotúnel que conecta el Reino Unido y Francia a través del Canal de la Mancha. Este túnel fue inaugurado en 1994 y es considerado corno una de las obras de ingeniería más impresionantes del mundo. Sin embargo, el análisis de sensibilidad fue ejecutado de manera incorrecta. No solo la estimación de los costes fue casi dos veces menor a los costes soportados en la finalización del proyecto. sino que el pronóstico sobre la demanda del servicio durante el primer afio fue de 17 millones de pasajeros. aunque solo fue utilizado por 4 millones de personas generando enormes pérdidas que no fueron contempladas.
}

Sin embargo, también enfrenta sus limitaciones. En concreto, cabe destacar que: (i) omite correlaciones, ya que al realizar un análisis individual del impacto que tienen las variables relevantes sobre la rentabilidad de la inversión omite las correlaciones que puedan existir entre diferentes variables (que suelen ser comunes en la práctica);\footnote{%
    Por ejemplo, cambios en los costes de los insumos, podrían estar asociados a cambios en el coste de financiamiento o a variaciones en la demanda.
} y, (ii) no proporciona probabilidades de manera directa, ya que las alteraciones son ajustadas por el analista (discrecionales)

\begin{specialblock}{Ejemplo -- Análisis de sensibilidad}
    Consideremos que un analista está evaluando una inversión que requiere un desembolso inicial de 1000€ y, además, se espera que genere los siguientes flujos de efectivo netos:

    \begin{center}
    \begin{tabular}{|p{0.45\linewidth}|p{0.45\linewidth}|}
    \hline
    \textbf{Año} & \textbf{Flujo de caja}\\
    \hline
    0 & $-10{.}000$ \\
    \hline
    1 & $300$ \\
    \hline
    2 & $500$ \\
    \hline
    3 & $600$ \\
    \hline
    \end{tabular}
    \end{center}

    Asumiendo una tasa de descuento del 10\%, el proyecto tendrá un VAN igual a:

    $$VAN=-1.000+\frac{300}{(1+0,1)}+\frac{500}{(1+0,1)^2}+\frac{600}{(1+0,1)^3}$$

    Lo que sugiere que el proyecto es rentable al ser el valor presente de los flujos de caja mayor a la inversión inicial. Ahora, suponga que el analista recurre al análisis de sensibilidad para determinar cuán sensible es este proyecto ante cambios en la tasa descuento. Para ello, se asume la tasa de descuento puede asumir el siguiente rango de valores,

    \begin{center}
    \begin{tabular}{|p{0.45\linewidth}|p{0.45\linewidth}|}
    \hline
    \textbf{Tasa de descuento} & \textbf{VAN}\\
    \hline
    10\% & $136,74$ \\
    \hline
    13\% & $72,89$ \\
    \hline
    14\% & $52,87$ \\
    \hline
    15\% & $33,45$ \\
    \hline
    16\% & $14,59$ \\
    \hline
    17\% & $-3,71$ \\
    \hline
    18\% & $-21,49$ \\
    \hline
    19\% & $-38,77$ \\
    \hline
    \end{tabular}
    \end{center}

    En la tabla se puede apreciar que el VAN decrece con la tasa de descuento. En particular, si la tasa de descuento es menor al 16\% el proyecto tiene un VAN positivo, mientras que si es mayor el proyecto deja de ser rentable.\footnote{Para ser exactos, la TIR de este proyecto es igual a $16,85$
    } Por consiguiente, dado que hoy el coste de oportunidad es del 10\%, se podría decir que, bajo condiciones económicas normales, el riesgo ante cambios en la tasa de interés es bajo, pues los intereses deberían incrementarse al menos un 60\% para que el proyecto deje de ser rentable. 
    
    Además, el analista podría evaluar, por ejemplo, qué tan sensible es el proyecto ante cambios en el flujo de caja recibido en el periodo 3. Aplicando la definición del VAN, el mínimo valor que puede asumir $CF_3$ para que el proyecto sea viable es, $CF_3\geq\left[1000-\frac{300}{1,1}-\frac{500}{(1,1)^2}\right](1,1)^3=418$.
    
    Entonces, el proyecto tendrá un VAN positivo solo si $CF_3 \geq 418$ y, por lo tanto, este valor puede tomarse como un valor de corte o referencia para $CF_3$ .
    
\end{specialblock}


\paragraph{Incertidumbre: Método de MONTECARLO}
Para abordar estos problemas de manera más efectiva, sorteando las limitaciones del análisis de sensibilidades podemos recurrir al método de MONTECARLO. 

El método de MONTECARLO es, probablemente, el instrumento más utilizado en la modelización y estimación en el ámbito de las ciencias sociales (y, a menudo, también en ciencias puras). Se trata de un método no determinista usado para aproximar problemas complejos y costosos de evaluar con exactitud.\footnote{%
    Para ver toda la información al respecto: [\authorfont{Ver Tema 3.A.9}], donde se recopila toda la información al respecto para la oposición. 

También: \href{https://ocw.mit.edu/courses/6-0002-introduction-to-computational-thinking-and-data-science-fall-2016/resources/lecture-6-monte-carlo-simulation/}{Lecture 6: Monte Carlo Simulation. MIT (opencouse)} y, \href{https://www.youtube.com/watch?v=L5uAPOE8lMM}{Monte Carlo Simulation to Assess Net Present Value (Youtube.com)}
}

En nuestro contexto, el método Montecarlo nos permitiría obtener una distribución de probabilidad para el VAN que facilite la evaluación del riesgo asociado a la inversión, a partir de diferentes sendas de comportamiento para las variables relevantes, como los ingresos, costes y tasa de descuento. El procedimiento para aplicar el método de Montecarlo se puede resumir de la siguiente manera: 
\begin{lnum}
    \item Identificar las funciones de probabilidad que mejor se ajusten al comportamiento de las variables bajo estudio;\footnote{%
        Por ejemplo, precio de venta, costes, tasa de descuento, etc. Para este proceso, es muy importante reunir toda la información disponible y utilizar el análisis estadístico
    }
    \item Una vez que se ha identificado la distribución de probabilidad de las variables relevantes, se deben producir las sendas esperadas para dichos parámetros;
    \item Proceder al cómputo del VAN para cada uno de los escenarios obtenidos mediante simulación; y,
    \item Analizar los resultados y evaluar en función de los objetivos. 
\end{lnum}

Por tanto, el método de Montecarlo brinda la posibilidad de analizar el impacto que un conjunto de variables tiene sobre la viabilidad del proyecto. En consecuencia, un analista es capaz de identificar las correlaciones existentes entre las variables de interés y a enfocar su atención en aquellos aspectos claves que impulsan el valor del proyecto. 

No obstante, la aplicación del método de Montecarlo al cálculo del VAN puede llegar a ser un proceso complejo y presentar algunas limitaciones.\footnote{%
    Por ejemplo, la precisión del análisis depende de manera crucial del proceso de identificación de las distribuciones de probabilidad de las variables relevantes. Si estas distribuciones no se ajustan adecuadamente a la realidad, los resultados pueden ser inexactos y/o sesgados. Otro factor importante es la magnitud de la muestra utilizada en el proceso de simulación. Si el tamaño de la muestra utilizado es demasiado pequeño, el comportamiento de la población podría no verse reflejado en el proceso de simulación, lo que podría inducir al analista a tomar decisiones equivocadas. Finalmente, aunque es posible recurrir a la implementación de software especializado que facilite la generación de los datos de entrada, la realización de grandes simulaciones (o complejas) puede limitar la velocidad y eficiencia del análisis.
} En resumen, el método de Montecarlo es una herramienta útil para modelar diferentes escenarios y resultados financieros que un proyecto puede soportar con el fin de precisar los riesgos asociados a la inversión. Asimismo, como ocurre con el análisis de sensibilidad, la aplicación del método de Montecarlo posibilita el reconocimiento de factores claves que impulsan el valor del proyecto.

}


\textcolor{magenta}{Ahora que conocemos las diversas herramientas que tiene al alcance un inversor para evaluar la factibilidad de un proyecto, debemos abordar un tema crucial: ¿qué tasa de descuento es la apropiada para el proyecto? Recordemos que la tasa de descuento o coste del capital representa el coste de oportunidad que conlleva comprometer una cantidad de dinero a un proyecto así como también el riesgo que supone el empréstito, puesto que si no lo tuviera en cuenta, cualquier alternativa de inversión libre de riesgo sería una opción mejor que se vería encubierta por su omisión. De esta forma, recuperamos la formulación inicial de la tasa de descuento, de manera que:

\eqblock{
\begin{aligned}
    \boxed{r_t=(\gamma_t+\lambda)}\quad \text{donde,}
    \left\{\begin{aligned}
        &\gamma_t&&\equiv \underset{\text{\scriptsize o, alternativamente, se puede entender como el coste de oportunidad de la inversión}}{\text{Rentabilidad del activo libre de riesgo}}\\[0.5em]
        &\lambda&&\equiv\underset{\text{\scriptsize incorpora implícitamente el riesgo}}{\text{Prima de riesgo}}
    \end{aligned}\right.
\end{aligned}
}{Tipo de descuento: Coste de oportunidad y prima de riesgo. Coste del capital: WACC}

Por tanto, la valoración entre diferentes alternativas dependerá, esencialmente, del \textbf{riesgo}. 

Para ilustrar esto, pensemos en dos alternativas que puede tener una empresa: (i) mejorar su precario sistema de almacenamiento en la Planta A, que evitaría el deterioro de sus existencias; o, (ii) construir una nueva planta, en vistas a un aumento de la demanda de su producto, la Planta B. Es normal pensar, que la mejora del almacenamiento no acarrea ningún riesgo, mientras que la abertura de una nueva planta anticipándose a un posible aumento de la demanda de su producto sí es un proyecto que acarrea riesgo implícitamente. Por tanto, mientras que el primero requiere que el $VAN(\gamma_t)$ se superior al $VAN(\gamma_t)$ esperado de un activo libre de riesgo; el segundo necesita que el $VAN(\gamma_t+\lambda)$ esperado sea superior al obtenido por un activo alternativo libre de riesgo $VAN(\gamma_t)$.

\textbf{¿Qué queremos decir?} El objetivo de esta parte es abordar las metodologías a través de las cuáles medir el coste del capital ($r$), discutiendo diferentes problemas teóricos y prácticos que pueden surgir en este proceso. Anticipamos que, como puede haberse deducido, los principales problemas surgen de la falta de datos precisos, la elección de las tasa libre de riesgo, la selección de la estructura de capital adecuada y la estimación de la prima de riesgo. Por ello, se utilizará una variable estrechamente relacionados con el coste del capital o tasa de descuento, pero que no presenten tantos problemas de estimación: el \textbf{coste promedio del capital} (WACC, por sus siglas en inglés). Este concepto, crucial en el ámbito de las finanzas, se refiere al retorno mínimo que una empresa debe lograr para compensar a sus inversores, incluyendo a los tenedores de acciones (coste del equity) y bonos (coste de la deuda) y puede estimarse mediante un método directo y uno indirecto.\footnote{%
    Es decir, estamos entendiendo capital de manera similar al factor $K$ en términos microeconómicos: fuentes de financiación de la actividad de la empresa. Distinguimos únicamente entre \textit{equity} (o capital accionarial) y \textit{deuda}, aunque existen otras fuentes de financiación como la deuda bancaria, beneficios no distribuidos o los accionistas preferentes; pero por simplicidad y sin pérdida de generalidad empleamos las dos anteriores sensu lato.
}\textsuperscript{,}\footnote{%
    Como apunta VEMIMMEN, ha de entenderse la diferencia entre el coste del capital (medido por el WACC) y la rentabilidad del capital (medido por otros indicadores como el ROE o el PIE).
\begin{la}
    \item \textbf{Coste promedio del capital}. Recoge la rentabilidad exigida y/o esperada de los inversores;
    \item \textbf{Rentabilidad del capital}. Hace lo propio con rentabilidad realizada. 
\end{la}
En caso de mercados sin ninguna fricción, ambos conceptos se igualarán gracias al arbitraje, pero conceptualmente son diferentes: mientras que el primero es exante, el segundo es expost. La importancia de esta advertencia radica en la frecuencia con la que analistas emplean los segundos indicadores como medidas del coste de capital, lo que sólo puede tomarse como una aproximación, con riesgo de introducir ruido y sesgo en la estimación.
}

\subsection{Estimación directa (marco general): WACC} 
El método directo consiste en estimar el coste del capital mediante modelos de valoración de activos como el APT (\textit{Arbitrage Pricing Theory}) o el modelo CAPM (\textit{Capital Asset Pricing Model}). 

Empleando el segundo, que pretende explicar la relación entre riesgo y rentabilidad existente en el mercado (MARKOWITZ), obtenemos la siguiente ecuación:

\eqblock{
\begin{aligned}
    &\left\{\begin{aligned}
        &(1)\,\mathbb E_t[r_{WACC}]\\[0.5em]
        &(2)\,\forall i:\sum_{t=1}^T\frac{1}{(1+r_{WACC})^t}\quad \wedge\quad \mathrm P_i[r_{WACC}\leq r]=p\\[0.5em]
        &(3)\,\text{Mcdos. sin fricciones}
    \end{aligned}\right\}\quad\underset{\text{CAPM}}{\Longrightarrow}\quad \boxed{\mathbb E_t[r_{WACC}]=\gamma+\underbrace{\beta_A(\underbrace{\mathbb E_t[r_M]-\gamma}_{=\lambda_M})}_{=\lambda}}\\[2em]
    &\text{Donde}:
    \left\{\begin{aligned}
        &\beta_A&&\equiv \underset{\text{\scriptsize porque sino sería posible reducirlo por arbitraje diversificando la cartera}}{\text{Riesgo sistemático o no diversificable de la empresa}}\\
        &\mathbb E_t[r_{WACC}]&&\equiv\text{Tasa de retorno esperada}\\
        &\mathbb E_t[r_{M}]&&\equiv\underset{\text{\scriptsize (parámetro no observable: estilo portfolio del mercado)}}{\text{Rentabilidad \textit{media} del mercado}}
    \end{aligned}\right.
\end{aligned}
}{Modelo CAPM (MARKOWITZ): Supuestos y ecuación del coste de capital. Coste de capital: Método directo}

Es decir, si todos los inversores toman sus decisiones en base a la rentabilidad esperada, comparten el mismo horizonte de inversión y las mismas distribuciones de probabilidad y no hay fricciones en los mercados de capitales (posibilidad de arbitraje), el modelo CAPM (que podemos aplicar por analogía) establece que: existe una relación lineal entre el riesgo sistemático de un activo y su retorno esperado.\footnote{%
    Se remite al opositor al Tema 3.B.23 para la demostración de cómo se obtiene esta fórmula.

\textbf{OJO.-} Bajo el CAPM, el riesgo relevante es el riesgo sistemático o no diversificable medido. Por tanto, el riesgo sistemático (no diversificable) es el único retribuido por el mercado, ya que el riesgo no sistemático puede ser eliminado mediante una correcta diversificación del portafolio.
} 

De esta forma, en equilibrio el retorno esperado de cualquier activo es igual a la tasa libre de riesgo más una prima, la cual es igual a la beta del activo en cuestión multiplicada por prima de riesgo observada en el mercado ($\lambda=(\mathbb E_t[r_{M}]-\gamma_t)$).\footnote{%
    Para aplicar el método directo, suele emplearse la tasa de las letras del Tesoro de EEUU como representativa de la tasa de libre de riesgo de la economía.

Además, en base a datos históricos, es posible estimar los retornos de la acción en cuestión, la prima de mercado y, en consecuencia, la beta del capital accionario. Actualmente, este proceso se simplifica considerablemente para las betas de empresas que cotizan en bolsa, pues las betas de estas compañías se pueden recoger de manera directa a través de diferentes fuentes de información de libre acceso.
} 

\subsubsection{Riesgo no diversificable: Obtención}
Ahora bien, si conocemos cómo calcular el coste promedio del capital, quedaría una pregunta evidente por responder: ¿cómo se obtiene el riesgo no diversificable de la empresa ($\beta_A$)? Denotemos $E$ y $D$ el valor del equity y de la deuda de la empresa, respectivamente. 

El valor del capital ($E$) se puede obtener multiplicando el precio de la acción en el mercado por el número en circulación, mientras que el valor de la deuda ($D$) se refiere al monto original de préstamos o emisión de bonos. Por tanto, el valor del capital accionario más el valor de la deuda estará dado por la suma de ambos componentes ($V=E+D$) y podemos definir la estructura del cpaital como $\frac{E}{V}+\frac{D}{V}$. Aplicando la fórmula que conocemos para el riesgo no diversificable en un portfolio de activos, tenemos que:

\eqblock{
\begin{aligned}
    &\text{Estructura capital}:\frac{E}{V}+\frac{D}{V}\quad \underset{\text{\scriptsize$\beta_p=\sum_ix_i\beta_i$}}{\sim}\quad \beta_A=\beta_E\frac{E}{V}+\beta_D\frac{D}{V}\\[1em]
    &\beta_A=\frac{\beta_E+\beta_D\frac{D}{E}}{1+\frac{D}{E}}\quad \underset{\text{\scriptsize si $D/E<1,$ $\beta_D\to 0$}}{\Longrightarrow}\quad\boxed{\beta_A=\frac{\beta_E}{1+\frac{D}{E}}}\sim\text{\scriptsize$\beta_A=\frac{\beta_E}{1+(1-T)\frac{D}{E}}$}
\end{aligned}
}{Modelo CAPM (MARKOWITZ): Derivación de la $\beta_A$ del modelo. Coste del capital: Método directo}

Por tanto, sabiendo cómo calcular la beta de un portafolio en base a la ponderación de activos que la componen, y aplicando este método al cálculo del riesgo sistemático de la empresa obtenemos la ecuación de la $\beta_A$. Además, siempre y cuando las empresas tengan un bajo nivel de endeudamiento, podemos asumir $\beta_D$ (puesto que no hay riesgo en ese contexto), se simplifica la ecuación. Por último, también cabría la posibilidad de estimar el coste promedio del capital en un contexto impositivo, donde los intereses pagados por una empresa pueden ser deducibles. 

En definitiva, la principal fortaleza que tiene la estimación del coste del capital mediante este método es que se puede aplicar independientemente del patrón de dividendos de la firma y, además, determina de manera explícita el nivel de riesgo. Como punto negativo, es evidente el papel que ostentan la calidad y cantidad de datos obtenidos, además de que el empleo de datos históricos para situaciones futuros supone una cierta continuidad o ergodicidad (hipotética).

\subsection{Estimación indirecta: Composición del capital}
Como dijimos, también existe un método alternativo que exige únicamente disponibilidad de datos y no asume, en tanta medida, hipótesis adicionales que puedan generar \textit{ceguere inducida por la teoría}.

Este método se conoce como indirecto y se basa en determinar los costes de las principales fuentes de financiación utilizadas por la compañía, que una vez estimados nos proporcionarán el coste promedio del capital ($r_{WACC}$):

\eqblock{
\begin{aligned}
    \boxed{r_{WACC}=r_E\frac{E}{V}+r_D\frac{D}{V}}\quad \text{o,}\quad \underset{\text{\scriptsize teniendo en cuenta los impuestos}}{r_{WACC}=r_E\frac{E}{V}+r_D(1-T)\frac{D}{V}}
\end{aligned}
}{WACC: Coste \textit{equity} y coste \textit{leverage}. Coste de capital: Estimación indirecta}

Por tanto, el WACC se obtiene como promedio de costes, cuyo peso viene determinado por la composición del capital. Ahora bien, ¿cómo calculamos el coste de cada fuene de financiación?

\subsubsection{Coste de capital accionario: \textit{Equity}}
Por lo que respecta al coste del capital accionario, esta es la parte más compleja. Normalmente se recurre a dos métodos: el CAPM y/o el modelo fundamental de valoración de acciones. Como el primero lo hemos visto en el apartado anterior, veamos brevemente el modelo de GORDON-SHAPIRO.  

Este establece que el valor teórico de una acción está determinado por los flujos de caja que generará la acción en el futuro. Por lo tanto, es necesario especificar las variables que permiten definir tales flujos y descontar esa corriente, en principio ilimitada, hasta el presente:\footnote{%
    Observe que se asume que no hay inflación, por lo tanto, es el valor de una acción en términos reales. En el apéndice se analiza el valor de compañía cuando existe inflación.
} 

\eqblock{
\begin{aligned}
    &\boxed{p_A=\sum_{t=1}^\infty\frac{d_t}{(1+r_E)^t}}\qquad \text{donde,}
    \left\{\begin{aligned}
        &d_t\equiv\text{Dividendo de la compañía}\\
        &p_A\equiv\text{Precio actual de la acción}
    \end{aligned}\right.\\[1em]
    &\underset{\substack{\text{\scriptsize asumimos tasa}\\\text{\scriptsize crecimiento: $g$}}}{\Longrightarrow}\quad \boxed{p_A=\frac{d_1}{r_E-g}}\quad \Longrightarrow\quad \underset{\text{\scriptsize coste del capital accionario}}{\boxed{r_E=\frac{d_1}{p_A}+g}}
\end{aligned}
}{Modelo GORDON-SHAPIRO: Estimación coste del capital accionario. Coste del capital: Estimación indirecta}

Por tanto, el origen del valor de la acción es el conjunto de pagos futuros.\footnote{
\begin{cita}[W. BUFFET][1994][Annual report to shareholders of Berkshire Hathaway]
    We define intrinsic value as the disco unted value of the cash that can be taken out of a business during its remaining life. Anyone calculating intrinsic value necessarily comes up with a highly subjective figure that will change both as estimates of future cashflows are revised and as interest rates move. Despite its fuzziness, however, intrinsic value is all-important and is the only logical way to evaluate the relative attractiveness of investments and business.
\end{cita} 
} Para facilitar el análisis, la expresión se simplifica si asumimos que los dividendos crecen a una tasa constante $g$, lo que es de gran utilidad a la hora de aislar el coste del capital.

Sobra decir que esta metodología es aplicable a empresas que cotizan en bolsa donde tanto los dividendos como el precio de mercado se pueden observar de manera directa, aunque el parámetro $g$ no. Su estimación se puede hacer mediante la utilización de datos históricos de la compañía o a través de la información publicada por los analistas.

\subsubsection{Coste de la deuda: \textit{Leverage}}
Por otro lado, el coste de la deuda es más sencillo, puesto que es el coste en que una empresa debe incurrir para financiarse.

Este puede obtenerse por dos vías: el modelo CAPM o con la observación directa la tasa de interés que la empresa debe pagar por el acceso a nuevos préstamos en el mercado (o de manera indirecta si una compañía ha emitido bonos en el pasado, en cuyo caso el coste será el rendimiento al vencimiento si estos siguen en circulación). 

Una vez conocemos como calcular el coste promedio de la deuda por el método indirecto, cabe hacer una pregunta fundamental: ¿dependen el coste del equity y el coste de la deuda de la estructura financiera de una empresa (o un proyecto de inversión)? Esto será tratado con más pronfundidad en el Tema 3.B.3, pero podemos decir anticipadamente que es uno de los puntos de mayor divirgencia entre el análisis empírico y la teoría económica. 

Brevemente, si sí, el cálculo del WACC por el método indirecto incorpora dos variables endógenas (los dos costes en cuestión). Por el contrario, si estos costes no dependen de la composición deuda-equity, el WACC no será sensible a cambios en la estructura financiera.

\textbf{INTRODUCIR GRAFICO -- ICEX}

La primera visión considera que un mayor apalancamiento para financiar un proyecto de inversión implica un mayor riesgo, por lo que pese a que incrementos en la ratio de apalancamiento generan primero caídas en el WACC (dado que el coste de la deuda es menor que el coste del equity), a partir de cierto punto, un mayor apalancamiento llevará consigo un mayor WACC, reflejando una mayor prima de riesgo. Por el contrario, en línea con la literatura de MODIGLIANI y MILLER (1958), el coste de capital vendrá determinado exclusivamente por la capacidad de generar valor del proyecto de inversión y por los riesgos asociados al mismo, independientemente de cuál sea la estructura de financiación elegida para llevarlo a cabo. En otras palabras, el WACC permanecería constante sin importar la composición de la estructura de capital escogida o, lo que es lo mismo, el nivel de apalancamiento.\footnote{%
    Bajo supuestos restrictivos. En concreto, se requiere de un mercado de capitales perfecto y de perfecta fungibilidad del dinero. 

Así, como se verá con detalle en el [\authorfont{Ver Tema 3.B.3}], se tiene que el valor total de una empresa es igual al valor de mercado del total de flujos de efectivos generados por sus activos, y, por lo tanto, no lo afecta la selección de la estructura del capital.
}

\anexo{Ejemplos numéricos}\label{anx:ejemplos}
Este anexo reúne los ejemplos numéricos del primer bloque. Van fuera del cuerpo porque el bloque razona con un ejemplo en letras, que permite cambiar el orden de las magnitudes y ver cuándo cambia la decisión; aquí se comprueban esas conclusiones con cifras concretas.

\subsection{Valor actual neto e índice de rentabilidad}
Dos proyectos, A y B, son mutuamente excluyentes, y la empresa dispone de un presupuesto de $80$ mil euros. Ambos duran cuatro periodos (por ejemplo, cuatro años) y se descuentan al $8\%$, que incluye un tipo sin riesgo del $2\%$ y una prima de $6$ puntos (el riesgo se introduce en el segundo apartado del bloque). Sus flujos de caja netos, en miles de euros, son:

\begin{center}
\begin{tabular}{|p{0.25\linewidth}|p{0.25\linewidth}|p{0.25\linewidth}|}
\hline
\textbf{Año} & \textbf{Proyecto A} & \textbf{Proyecto B} \\
\hline
0 & $-50$ & $-80$ \\
\hline
1 & $20$ & $30$ \\
\hline
2 & $22$ & $32$ \\
\hline
3 & $25$ & $36$ \\
\hline
4 & $30$ & $40$ \\
\hline
\end{tabular}
\end{center}

Sus valores actuales netos, en miles de euros, son
$$VAN_A=-50+\frac{20}{1,08}+\frac{22}{1,08^2}+\frac{25}{1,08^3}+\frac{30}{1,08^4}\approx 29,28,\qquad VAN_B=-80+\frac{30}{1,08}+\frac{32}{1,08^2}+\frac{36}{1,08^3}+\frac{40}{1,08^4}\approx 33,19.$$
Al $8\%$, los dos tienen un VAN positivo, de modo que cualquiera de ellos aumentaría el valor de la empresa. Sus índices de rentabilidad son
$$R_A=\frac{29,28}{50}\approx 0,586,\qquad R_B=\frac{33,19}{80}\approx 0,415:$$
cada euro invertido en A crea unos $59$ céntimos de valor, y cada euro invertido en B, unos $41$. A tiene mejor índice, pero, como los proyectos son excluyentes e indivisibles, elegir A deja sin invertir $30$ mil euros del presupuesto, que solo rendirían el tipo de interés. La empresa debe elegir B, que crea $3,92$ mil euros más de valor, salvo que pueda invertir esos $30$ mil euros en otro proyecto con un VAN superior a esa diferencia, es decir, con un índice de rentabilidad superior a $3,92/30\approx 0,13$.

Dos casos breves ilustran lo mismo. Un proyecto con un VAN de $10$ euros y una inversión de $50$ tiene un índice de $0,2$: crea $20$ céntimos de valor por cada euro invertido. Y un proyecto con una inversión y un VAN de $2$ euros (índice $1$) es peor, si los proyectos son excluyentes, que otro que exige $100$ euros y tiene un VAN de $60$ (índice $0,6$).

\subsection{Tasa interna de retorno}
Un proyecto exige una inversión de $1$ mil euros y genera $600$ y $700$ euros al final del primer y del segundo año. Su TIR, en miles de euros, resuelve
$$0=-1+\frac{0,6}{1+TIR}+\frac{0,7}{(1+TIR)^2},$$
una ecuación de segundo grado en $x=1/(1+TIR)$ cuya raíz positiva es $x\approx 0,841$, de modo que $TIR\approx 18,9\%$. El proyecto se acepta si el coste del capital es inferior a esa tasa.

La bomba de LORIE y SAVAGE (1955) exige $1,6$ mil dólares y adelanta al primer año $10$ mil dólares de petróleo que, sin ella, se extraerían en el segundo: sus flujos son $-1,6$, $+10$ y $-10$ mil dólares. Su VAN se anula con tipos del $25\%$ y del $400\%$, es negativo fuera de ese intervalo y positivo dentro de él. Un proyecto con flujos $+1$, $-3$ y $+2,5$ no tiene TIR: su VAN es positivo a cualquier tipo de descuento, con un mínimo de $0,1$ en torno al $67\%$.

En el ejemplo del índice de rentabilidad, $TIR_A\approx 30,3\%$ y $TIR_B\approx 24,5\%$, de modo que la TIR prefiere A. El proyecto diferencial, B menos A, tiene flujos $-30$, $10$, $10$, $11$ y $10$ mil euros, y su TIR, la tasa de FISHER, es de un $13,7\%$. Al $8\%$, por debajo de esa tasa, B tiene mayor VAN ($33,19$ frente a $29,28$ mil euros), y la TIR ordena mal; al $15\%$, por encima, A tiene mayor VAN ($17,62$ frente a $16,82$), y ambos criterios coinciden.
