    % --- Imagen adaptativa ---
    % \imagenfit[alto]{fichero}{título}{fuente}: la imagen queda fija entre los párrafos donde se escribe (no flota), con su
    % título en el índice de figuras y la fuente debajo. Sin [alto]: las figuras PDF (TikZ) salen al tamaño con que se
    % dibujaron y solo se reducen si no caben en el ancho o en el alto de la página; las PNG/JPG ocupan el ancho del texto
    % hasta \imgaltoras del alto de la página. Con [alto] (fracción del alto de la página; [1] = página entera) se fija el
    % tamaño de esa imagen. Si no cabe en lo que queda de página, pasa entera a la siguiente.
    % \imagenfithorizontal{fichero}{título}{fuente}: la imagen en una página girada (apaisada), en ese mismo punto del texto.
    % Reglas: main/GUIA_TEMAS.md, §4 (Figuras).
    \newdimen\imgcap
    \newdimen\imgh
    \newdimen\imgw
    \newdimen\imgneed
    \newsavebox\imgcaja
    \newcommand{\imgaltoras}{0.30}% alto por defecto de las PNG/JPG (fracción del alto de la página)
    \makeatletter
    \newcommand{\imagenfit}[4][]{%
      \addvspace{10pt}%
      \par\begingroup
        \imgcap=3.5\baselineskip% alto aproximado de título + fuente
        \imgw=\linewidth
        \filename@parse{#2}\def\img@pdf{pdf}%
        \if\relax\detokenize{#1}\relax
          \ifx\filename@ext\img@pdf
            \sbox\imgcaja{\includegraphics{\detokenize{#2}}}% tamaño natural
            \ifdim\wd\imgcaja<\imgw \imgw=\wd\imgcaja \fi
            \imgh=\ht\imgcaja
          \else
            \imgh=\imgaltoras\textheight
          \fi
        \else
          \imgh=#1\textheight
        \fi
        \dimen@=\textheight \advance\dimen@ by -\imgcap
        \ifdim\imgh>\dimen@ \imgh=\dimen@ \fi
        \sbox\imgcaja{\includegraphics[width=\imgw,height=\imgh,keepaspectratio]{\detokenize{#2}}}%
        \imgneed=\ht\imgcaja \advance\imgneed by \imgcap
        \Needspace{\imgneed}%
        \noindent\begin{minipage}{\textwidth}
          \centering
          \captionof{figure}{\textemdash\ #3}\par\vspace{0.3em}% título arriba
          \usebox\imgcaja\par
          \vspace{0.3em}\small\textit{Fuente: }#4% fuente abajo
        \end{minipage}%
      \endgroup
      \par\addvspace{10pt}\RequestRestore
    }
    \newcommand{\imagenfithorizontal}[3]{%
      \begin{landscape}
        \centering
        \captionof{figure}{\textemdash\ #2}\par\vspace{0.3em}%
        \includegraphics[width=\linewidth,height=\dimexpr\textheight-4\baselineskip\relax,keepaspectratio]{\detokenize{#1}}\par
        \vspace{0.3em}{\small\textit{Fuente: }#3}%
      \end{landscape}%
    }
    \makeatother

    % \anexo{Título}: un anexo equivale a un \section (su título sale del mismo tamaño). Dentro se usan \subsection,
    % \subsubsection y \paragraph, nunca \section: se numeran I.1, I.1.1, I.1.1.1 y las páginas del anexo, I-1, I-2…
    % En el índice de contenido inicial solo figura el nombre del anexo. Sus apartados forman el índice del anexo, que sale
    % solo si el anexo los tiene, justo después de la introducción del anexo (antes del primer \subsection).
    % \anexo*{Título}: el mismo anexo, sin ese índice. Reglas: main/GUIA_TEMAS.md, §6.

    \newcommand{\anx@empezar}[2]{%
      \clearpage
      \refstepcounter{anexo}%
      \global\EnAnexotrue
      \setcounter{page}{1}\gdef\thepage{\theanexo-\arabic{page}}%
      \setcounter{section}{\numexpr\value{anexo}-1\relax}\stepcounter{section}% el anexo hace de \section
      \gdef\thesection{\Roman{section}}\gdef\theHsection{anexo.\Roman{section}}%
      \phantomsection
      \protected@xdef\leq@current@section{Anexo \theanexo.- \unexpanded{#2}}\gdef\leq@current@subsection{}% índice de ecuaciones
      \addtocontents{stc}{\protect\contentsline{section}{Anexo \theanexo.- #2}{}{\@currentHref}}% índice inicial: solo el nombre, sin página
      \addcontentsline{toc}{section}{Anexo \theanexo.- #2}% marcadores del PDF
      {\noindent\normalfont\Large\bfseries Anexo \theanexo.- #2\par}\nobreak\vspace{1.5ex \@plus .2ex}%
      #1\@ifundefined{anx@lista@\theanexo}{\global\anx@pendientefalse}{}%
    }
    % Índice del anexo: cada apartado escribe una línea en el fichero .atc (anexo, nivel, texto, página, enlace); al
    % empezar el documento se leen las de la compilación anterior y se guardan por anexo (dos pasadas, como cualquier índice).
    % En ese índice los apartados suben un nivel, porque el anexo hace de \section.
    \def\anx@entrada#1#2#3#4#5{\@ifundefined{anx@lista@#1}{\global\@namedef{anx@lista@#1}{}}{}%
      \expandafter\g@addto@macro\csname anx@lista@#1\endcsname{\contentsline{#2}{#3}{#4}{#5}}}
    \tcbuselibrary{breakable}
    \def\anx@indice{\ifanx@pendiente\global\anx@pendientefalse

            before upper={\parindent=0pt\parskip=2pt}]
          \small
          % números más anchos que en el índice inicial (I.1, I.1.1, I.1.1.1; páginas I-1)
          \def\@pnumwidth{2.6em}\def\@tocrmarg{3.4em}%
          \def\l@section##1##2{\addpenalty\@secpenalty\addvspace{1.0em \@plus\p@}\setlength\@tempdima{2.6em}%
            \begingroup\parindent\z@\rightskip\@pnumwidth\parfillskip-\@pnumwidth\leavevmode\bfseries
            \advance\leftskip\@tempdima\hskip-\leftskip##1\nobreak\hfil\nobreak\hb@xt@\@pnumwidth{\hss##2\kern-\p@\kern\p@}\par\endgroup}%

          \csname anx@lista@\theanexo\endcsname

      \fi}
    \def\anx@escribir#1#2#3{\ifEnAnexo

    % margen superior, así que no cambia las páginas del tema. Se enciende con \pagenumbering{arabic} y se apaga con

    \cs_new_protected:Npn \cab_guardar:n #1
      {
        \tl_gset:Nn \g_cab_titulo_tl {#1}
        \regex_replace_once:nnN { \A \s* Tema \s+ \S+ \s* \-\-? \s* } { } \g_cab_titulo_tl % sin «Tema X.Y.N --»
        \regex_replace_once:nnN { \.? \s* \c{\\} (\[ [^\]]* \])? \s* (\c{large} \s*)? } { .\  } \g_cab_titulo_tl % título. subtítulo
        \regex_replace_once:nnN { [\.\s]* \Z } { } \g_cab_titulo_tl % sin subtítulo: nada detrás
      }

\section{Decisiones de inversión: ¿cuándo crea valor un proyecto?}
\puntosclave{
    Toda inversión compara su rentabilidad con un coste de oportunidad que tiene dos componentes, el tiempo ($\gamma$) y el riesgo ($\lambda$). Con información perfecta y mercados de capitales perfectos, el VAN es el criterio correcto: mide el valor que crea el proyecto y todos los propietarios aprueban maximizarlo, sean cuales sean sus preferencias (separación de FISHER).

    El índice de rentabilidad tiene sentido con racionamiento de capital: ordena los proyectos por el VAN que crea cada euro, pero falla con proyectos indivisibles o con restricciones en varios periodos.

    La TIR mide la rentabilidad implícita del proyecto y el mayor coste del capital que soporta. Coincide con el VAN si el perfil del VAN es decreciente, pero puede no existir, no ser única u ordenar mal proyectos excluyentes (punto de FISHER); aplicada al proyecto diferencial, lleva a la misma decisión que el VAN. Que suponga reinvertir los flujos a la propia TIR es objeto de debate.

    El payback mide la liquidez y la exposición temporal del proyecto, no su valor: ignora lo que ocurre después del plazo, y el plazo máximo es arbitrario.

    Los tres criterios son estáticos: suponen que se decide ahora o nunca. Con información imperfecta, la flexibilidad tiene valor y el VAN estático infravalora los proyectos flexibles: con flexibilidad, más incertidumbre aumenta el valor del proyecto.

    Los árboles de decisión miden el valor de esperar a conocer la información; las opciones reales lo valoran con el descuento correcto (VAN ampliado). Su uso formal es limitado, y la competencia erosiona el valor de esperar.

    La sensibilidad, los escenarios y la simulación de Montecarlo miden, con profundidad creciente, la dispersión del resultado, con un plan fijo o con flexibilidad. Pero la dispersión total no es el riesgo que el mercado remunera, salvo que haya fricciones financieras: solo entonces importan las preferencias ante el riesgo.
}


Invertir es renunciar hoy a unos recursos a cambio de unos flujos futuros. Por eso toda decisión de inversión se somete a fuerzas que operan en sentido contrario. De un lado está la \textbf{rentabilidad}, que permite evaluar si el capital comprometido en un proyecto generará un rendimiento capaz de justificar la inversión. De otro, dos fuerzas que la contrapesan. La \textbf{intertemporalidad} exige tener en cuenta el coste de oportunidad de comprometer un determinado volumen de capital en un proyecto concreto durante un periodo determinado, es decir, el rendimiento al que se renuncia en la mejor alternativa disponible.\footnote{%
    Que un euro futuro valga menos que uno presente puede parecer trivial, pero para un economista no lo es: es el \textbf{valor temporal del dinero}, que depende tanto de la inflación como de la existencia de inversiones alternativas. Unas oportunidades de inversión con alto rendimiento o una inflación elevada llevan a descontar los flujos de caja lejanos con más intensidad que los que el inversor recibe en fechas cercanas. Para que el descuento sea coherente, los flujos nominales se descuentan con tipos nominales y los reales con tipos reales, que se relacionan por la ecuación de FISHER (1930): $(1+i)=(1+i_r)(1+\pi)$, con $i$ el tipo nominal, $i_r$ el real y $\pi$ la inflación esperada.
} La \textbf{información} exige considerar la forma en que las contingencias futuras pueden alterar el curso del proyecto y, sobre todo, su rentabilidad. Ambas fuerzas entran en la valoración por dos vías: la tasa con que se descuentan los flujos del proyecto y los propios flujos esperados:

\eqblock{
  \begin{aligned}
    \underset{\text{\scriptsize tiempo y riesgo no diversificable}}{r=\gamma+\lambda}\qquad \underset{\text{\scriptsize riesgo específico}}{CF_t}
  \end{aligned}
}{Tasa de descuento: ¿qué rentabilidad exige el mercado? Decisión de inversión: ¿cuándo crea valor un proyecto?}

La tasa refleja la intertemporalidad mediante $\gamma$, el tipo de interés sin riesgo, que es la parte del coste de oportunidad del capital que remunera el tiempo; y refleja la información mediante $\lambda$, una prima que remunera solo el riesgo que los inversores no pueden eliminar diversificando. El riesgo específico del proyecto, que la diversificación elimina y nadie remunera, no entra en la tasa, sino en los flujos esperados, $CF_t$.
 
Pero la información no actúa solo contra la rentabilidad: la incertidumbre resta valor al proyecto cuando es un riesgo que los inversores no pueden eliminar diversificando y exigen que se les remunere; y se lo añade cuando la empresa puede esperar a ver cómo se resuelve y adaptar sus decisiones, porque esa flexibilidad vale más cuanto mayor es la incertidumbre. Con información perfecta y mercados de capitales perfectos, el valor actual neto es el criterio correcto para decidir, y los demás criterios lo son solo en la medida en que coinciden con él; con información imperfecta, sigue siéndolo únicamente si incorpora el valor de la información que se obtendrá a lo largo de la vida del proyecto, y un valor actual neto que lo ignora infravalora los proyectos flexibles.

Por eso distinguimos dos dimensiones y, dentro de la primera, dos contextos. La primera dimensión se centra en la rentabilidad del proyecto: los flujos de caja futuros que genera, el riesgo específico y la flexibilidad que se recogen en ellos y los instrumentos sobre los que apoyar la valoración. La segunda se centra en el riesgo no diversificable de cualquier proyecto dado, y nos permite determinar el coste correcto al que descontar sus flujos. Los dos contextos distinguen cómo se evalúa la primera. Con información perfecta, la única fuerza que se opone a la rentabilidad es el tiempo, y los criterios clásicos de valoración, con sus virtudes y sus problemas, se ven con nitidez. Con información imperfecta entra en juego la información, que extiende esos criterios en dos direcciones: cuánto vale la información que falta y cómo medir el efecto de la incertidumbre sobre el resultado del proyecto. En ningún caso deben considerarse dimensiones o contextos excluyentes; todo lo contrario: se interrelacionan e interactúan con el objetivo de proporcionar más información a la hora de tomar las decisiones de inversión.

Cómo se valoran los flujos del proyecto y cómo inciden en su rentabilidad el riesgo específico y la información son objeto de este bloque; cuál es la tasa de descuento adecuada, el coste del capital, se aborda en el siguiente.


\subsection{Información perfecta: la rentabilidad frente al coste del tiempo}
\textcolor{magenta}{Con información perfecta, quien decide conoce con certeza los flujos de caja del proyecto y los tipos de interés a los que puede prestar y endeudarse en cada periodo. El único coste de oportunidad corresponde al riesgo no diversificable, y la tasa de descuento es el tipo de interés $r_t=\gamma_t+\gamma_{t}$.

)(¿coreción correcta?)
} 

En este marco se presentan los tres criterios básicos de evaluación de proyectos: el valor actual neto (VAN), la tasa interna de retorno (TIR) y el periodo de recuperación o \textit{payback}. Los tres se aplican en las primeras etapas de la evaluación y persiguen el mismo fin, estudiar la viabilidad de una inversión, pero responden a preguntas distintas: el VAN mide cuánto valor crea el proyecto, en unidades monetarias; la TIR, qué rentabilidad implícita ofrece, en tanto por ciento; y el payback, cuánto tarda en devolver lo invertido, en unidades de tiempo. 

Por eso no siempre llegan al mismo resultado y se emplean de forma complementaria. No tienen, sin embargo, el mismo rango: como veremos, solo el VAN se deriva directamente del objetivo de maximizar el valor de la empresa para sus propietarios, y la TIR y el payback son útiles en la medida en que coinciden con él o aportan información que el VAN no recoge.\footnote{%
    En la práctica se usan los tres. En la encuesta de GRAHAM y HARVEY (2001) a $392$ directores financieros estadounidenses, el $74,9\%$ declaraba emplear siempre o casi siempre el VAN, el $75,6\%$ la TIR y el $56,7\%$ el payback. Que la TIR se use tanto como el VAN, pese a sus problemas, se explica en parte porque una rentabilidad en tanto por ciento es más fácil de comunicar y de comparar con el coste de los fondos.
}

Los tres criterios comparten su materia prima, los \textbf{flujos de caja} del proyecto, cuya obtención no es evidente en la práctica. Lo relevante son los flujos incrementales: la diferencia entre los cobros y pagos de la empresa con el proyecto y sin él.\footnote{%
    De ahí cuatro reglas prácticas: (i) los costes hundidos, ya incurridos e irrecuperables, no se cuentan, porque no dependen de la decisión; (ii) los recursos propios que el proyecto ocupa (un terreno, una nave) se valoran por su coste de oportunidad, aunque no supongan un pago; (iii) se incluyen los efectos sobre el resto de la empresa, como la pérdida de ventas de otros productos propios; y, (iv) no se incluyen los flujos de la financiación (intereses, devolución de la deuda, dividendos), porque su coste ya está en la tasa de descuento y contarlos dos veces sesgaría el resultado (BREALEY, MYERS y ALLEN, 2020).
} Al evaluar una inversión deben considerarse tres tipos: (i) flujos de explotación;\footnote{%
    La contribución de la inversión al resultado bruto de explotación (EBITDA, por sus siglas en inglés: beneficio antes de intereses, impuestos, depreciaciones y amortizaciones) es la diferencia entre los ingresos y los gastos adicionales que genera, sin contar depreciaciones ni amortizaciones. Del EBITDA se deduce el impuesto teórico sobre el resultado de explotación adicional, que se calcula multiplicando el tipo impositivo que soporta el proyecto por la variación del resultado de explotación (después de amortizaciones, que son fiscalmente deducibles) y teniendo en cuenta los créditos fiscales por pérdidas.
} (ii) flujos de inversión;\footnote{%
    La inversión se entiende en sentido amplio: incluye desde el fondo de maniobra (\textit{working capital}) hasta los activos fijos. El fondo de maniobra es la diferencia entre los activos corrientes de la empresa (efectivo, existencias y cuentas por cobrar) y sus pasivos corrientes (cuentas por pagar y deudas a corto plazo); representa los fondos disponibles para las operaciones diarias y es una medida importante de su liquidez y de su salud financiera ([\authorfont{Ver Tema 3.B.1}]). Un fondo de maniobra positivo indica que la empresa tiene recursos a corto plazo suficientes para cumplir sus obligaciones; uno negativo puede anticipar tensiones de liquidez. Las variaciones del fondo de maniobra que exige el proyecto se deducen del EBITDA. La inversión en activos fijos comprende la inversión en capacidad de producción y crecimiento, ya sea en activos tangibles (maquinaria, terrenos, edificios), intangibles (investigación y desarrollo, patentes y licencias, capital empresarial) o financieros (acciones de filiales) para el crecimiento externo.
} y, (iii) flujos extraordinarios.

\subsubsection{Valor actual neto (VAN): el valor que crea el proyecto}
El primer criterio, y el que sirve de referencia a los demás, es el \textbf{valor actual neto}. Se fundamenta en el valor temporal del dinero: para determinar si un proyecto es viable, trae al presente todos sus flujos de caja netos, descontándolos al coste de oportunidad, y les resta la inversión inicial. Con un tipo de descuento constante, el criterio del valor actual de FISHER (1930) se escribe:\footnote{%
    Donde $I_0$ es la inversión inicial, $CF_t$ el flujo de caja neto del periodo $t$, $T$ la vida del proyecto y $r$ la tasa de descuento, que con información perfecta es el tipo de interés sin riesgo.

    Calibrar sus componentes puede resultar difícil en la práctica. Si los tipos difieren entre periodos, cada flujo se descuenta con el producto de los factores de los periodos que median hasta su cobro, $VAN=-I_0+\sum_{t=1}^{T}CF_t\big/\prod_{s=1}^{t}(1+r_s)$, o, de forma equivalente, con el tipo de contado a plazo $t$ de la curva de tipos de interés. Suponer un tipo constante equivale a suponer que esa curva es totalmente plana, algo poco realista que aquí se adopta por simplicidad ([\authorfont{Ver Tema 3.B.24}]). Un método básico para fijar la tasa de descuento es tomar la curva de tipos del activo sin riesgo y sumarle una prima que depende de las características del proyecto, como veremos al introducir la incertidumbre y en el bloque siguiente.

    En tiempo continuo, el valor actual en $t$ de un flujo $CF_s$ que se cobra en $s$ es $CF_s\,e^{-\gamma_{t,s}(s-t)}=CF_s\,B(t,s)$, donde $\gamma_{t,s}$ es el tipo sin riesgo para el periodo $(s-t)$ y $B(t,s)$, el precio en $t$ de un bono que paga $1$ en $s$. Con riesgo, el flujo se sustituye por el previsto, $\mathbb E_t[CF_s]$, y el descuento añade la prima de riesgo del proyecto: $CF_s\,e^{-(\gamma_{t,s}+\lambda)(s-t)}=CF_s\,B(t,s)\,e^{-\lambda(s-t)}\approx CF_s/(1+\gamma_{t,s}+\lambda)^{s-t}$.
}

\eqblock{
  \begin{aligned}
    VAN=-I_0+\sum_{t=1}^{T}\frac{CF_t}{(1+r)^t}
  \end{aligned}
}{Valor Actual Neto (VAN): ¿cuánto valor crea el proyecto? Decisiones de inversión: Información perfecta}

El VAN es, por tanto, una \textbf{medida de rentabilidad absoluta}: en unidades monetarias, mide cuánto aumenta (o disminuye) el valor de la empresa si se acomete la inversión.\footnote{%
    Si además los flujos son constantes, $CF_t=CF$, la expresión se reduce a $VAN=-I_0+CF\,\frac{1-(1+r)^{-T}}{r}$. De este caso se derivan otros de interés: la renta perpetua, cuyo valor actual es $CF/r$, y los flujos que crecen a una tasa constante $g<r$, cuyo valor actual es $CF_1/(r-g)$. Para estas extensiones, véase el capítulo $16$ de VERNIMMEN, QUIRY y LE FUR (2022).
}

La regla de decisión se sigue de esa interpretación. Si el VAN es positivo, el proyecto rinde más que su coste de oportunidad: obtener sus flujos en el mercado de capitales exigiría invertir más de lo que cuesta el proyecto, y acometerlo aumenta la riqueza de los propietarios en esa diferencia. Si es negativo, el proyecto destruye valor, porque los mismos fondos rendirían más en el mercado. Si es nulo, rinde exactamente su coste de oportunidad: no crea ni destruye valor, y la empresa es indiferente.\footnote{%
    \textbf{NOTA.-} Un VAN nulo equivale al beneficio normal de la teoría microeconómica: el capital obtiene justo su remuneración de mercado. Un VAN positivo es, por tanto, una renta económica, y en un mercado competitivo no debería abundar: los proyectos con VAN positivo suelen apoyarse en alguna ventaja que los competidores no pueden replicar de inmediato, como una patente, una tecnología, una posición de mercado o un coste menor. Un VAN muy elevado sin una ventaja que lo explique es, más a menudo, señal de un error en la estimación de los flujos que de una oportunidad excepcional.
} Por la lógica del arbitraje, sería deseable acometer todo proyecto con VAN positivo.

¿Por qué maximizar el VAN, y no la satisfacción de unos propietarios que pueden tener preferencias muy distintas sobre cuándo consumir? La respuesta es el \textbf{teorema de separación} de FISHER (1930), que formalizó HIRSHLEIFER (1958). Supongamos un propietario que dispone hoy de unos recursos $Y_0$, que puede invertir en proyectos de rendimiento decreciente (la frontera de inversión) o prestar y pedir prestado sin límite al tipo $r$ (la recta de mercado). La decisión se separa en dos etapas. Primero, la empresa invierte hasta que el rendimiento del último euro invertido iguala el tipo de interés: es el punto $P^*$, en que la frontera es tangente a una recta de pendiente $-(1+r)$. Esa inversión, $I^*$, es la que maximiza el valor actual de la riqueza, $W_0=Y_0+VAN$. Después, cada propietario se desplaza a lo largo de la recta de mercado hasta el consumo que prefiere, prestando si es paciente ($a$) o endeudándose si es impaciente ($b$). Como la recta de mercado más alejada del origen es la mejor para todos, cualesquiera que sean sus preferencias, todos los propietarios aprueban por unanimidad la misma decisión: maximizar el VAN. La empresa puede delegarla en sus directivos sin consultar a los propietarios, con una regla que solo exige conocer los flujos y el tipo de interés.

\imagenfit{fisher_separacion.pdf}{Separación de FISHER: la inversión óptima no depende de las preferencias}{Elaboración propia a partir de FISHER (1930) y HIRSHLEIFER (1958).}

El VAN tiene, además, una propiedad que ningún otro criterio comparte: es aditivo. El VAN de dos proyectos independientes es la suma de sus VAN, de modo que cada proyecto puede evaluarse por separado y el valor de la empresa es la suma del de sus proyectos. En definitiva, la principal virtud del VAN es que reduce un proyecto de inversión a una suma de flujos de caja descontados, lo que facilita el análisis objetivo de las alternativas e introduce, de forma inequívoca, el criterio más relevante: la creación de valor para la empresa.

El teorema descansa, sin embargo, en un mercado de capitales perfecto,\footnote{%
    HIRSHLEIFER (1958) mostró que, si el tipo al que la empresa se endeuda es mayor que el tipo al que presta, la separación se rompe: ya no hay una única recta de mercado, y la inversión óptima pasa a depender de las preferencias de los propietarios.
} y sus fricciones limitan la validez del resultado: la empresa puede no tener acceso ilimitado al mercado de capitales, pueden existir incompatibilidades entre proyectos, o costes de gestión y restricciones de capacidad que impidan acometer todos los proyectos con VAN positivo. Para fijar ideas, seguiremos a lo largo del bloque un mismo ejemplo. Una empresa estudia dos proyectos. El proyecto A exige hoy una inversión $I_A$ y devuelve un único flujo $CF_A$ al cabo de un periodo; el proyecto B, mayor y más lento, exige $I_B>I_A$ y devuelve un único flujo $CF_B$ al cabo de dos periodos. Aplicando la expresión del VAN, sus valores son
$$VAN_A=-I_A+\frac{CF_A}{1+r},\qquad VAN_B=-I_B+\frac{CF_B}{(1+r)^2}.$$
Supongamos que, al tipo de interés vigente, los dos son positivos y $VAN_B>VAN_A$. Si los proyectos son independientes y la empresa puede financiarse sin límite, acomete los dos, porque cada uno suma su VAN al valor de la empresa. Si son \textbf{excluyentes} (dos usos alternativos del mismo solar, dos tecnologías para la misma planta), elige B aunque exija más inversión: si eligiese A, la diferencia $I_B-I_A$ que deja de invertir solo podría colocarse en el mercado, con VAN nulo, y el valor total sería $VAN_A<VAN_B$. Cuando el límite es el capital disponible, la pregunta cambia.

\paragraph{Índice de rentabilidad: el VAN por euro invertido}
El VAN mide cuánto valor crea un proyecto, pero no cuánto capital necesita para crearlo. Cuando el capital es escaso, esa segunda dimensión importa, y la recoge el \textbf{índice de rentabilidad}, una extensión del VAN que lo divide por la inversión inicial del proyecto $p$:
$$R_p=\frac{VAN_p}{I_0^p}.$$
Es positivo si y solo si lo es el VAN, y mide la riqueza creada por cada euro invertido: una suerte de multiplicador del desembolso inicial sobre el VAN.\footnote{%
    Algunos textos lo definen como el cociente entre el valor actual de los flujos y la inversión, $VA_p/I_0^p=1+R_p$; el umbral de aceptación es entonces $1$, y no $0$.
} Mientras el VAN mide la rentabilidad en términos absolutos, el índice la mide en términos relativos, por lo que es complementario del VAN y no su sustituto. En el ejemplo, si B exige mucha más inversión que A, puede ocurrir que $R_A>R_B$ aunque $VAN_B>VAN_A$: A crea más valor por euro, pero B crea más valor en total. Si los proyectos son excluyentes y no hay restricción de capital, se elige B: un índice más alto no implica una inversión mejor.

La utilidad propia del índice aparece con el \textbf{racionamiento de capital}. Si la empresa dispone de un presupuesto $K$ que no le permite acometer todos los proyectos con VAN positivo, su problema es maximizar el VAN total sin que la inversión supere $K$. Con una sola restricción y proyectos divisibles, basta ordenar los proyectos de mayor a menor índice y aceptarlos hasta agotar el presupuesto. El índice es el VAN por unidad del recurso escaso, como el margen por unidad de un factor escaso en la teoría de la empresa, y el índice del último proyecto aceptado es el precio sombra del capital: el VAN que añadiría un euro más de presupuesto. En la figura, el índice de cada proyecto es la pendiente del radio que lo une con el origen; con un presupuesto $K=I_B$, A más la fracción $\theta=(K-I_A)/I_B$ de B que cabe en el resto crean más valor que B solo, porque cada euro invertido en A rinde más que en B.

\imagenfit{indice_rentabilidad.pdf}{Índice de rentabilidad y racionamiento de capital}{Elaboración propia a partir de LORIE y SAVAGE (1955).}

Esa ordenación falla en dos casos, que estudiaron LORIE y SAVAGE (1955) y WEINGARTNER (1963). El primero son los proyectos \textbf{indivisibles}. En el ejemplo, si A y B no pueden fraccionarse y el presupuesto es $K=I_B$, el índice recomienda A, que deja sin invertir $K-I_A$; ese resto solo rinde el tipo de interés, con VAN nulo, y la empresa obtiene $VAN_A<VAN_B$. Salvo que exista otro proyecto que quepa en el resto y compense la diferencia, B es mejor pese a su menor índice: la elección deja de ser una ordenación y se convierte en un problema de programación entera, el problema de la mochila. El segundo son las restricciones en \textbf{varios periodos} o sobre varios recursos, en que un único cociente no puede ordenar los proyectos. WEINGARTNER (1963) formuló entonces la selección como un programa matemático cuyos precios duales miden el valor de relajar el presupuesto de cada periodo.

Que haya racionamiento es, en sí mismo, una desviación del mundo de FISHER: con un mercado de capitales perfecto, todo proyecto con VAN positivo encuentra financiación. El racionamiento puede ser externo, cuando la empresa no obtiene fondos al tipo de mercado por imperfecciones como la información asimétrica entre la empresa y quienes la financian ([\authorfont{Ver Tema 3.B.3}]); o interno, cuando la dirección fija presupuestos para disciplinar a los responsables de cada división o porque su capacidad de gestión es limitada. En el primer caso, el índice es un criterio de segundo óptimo; en el segundo, el presupuesto es una restricción que la empresa se impone, y la pregunta relevante es si el coste de relajarla es menor que el VAN al que renuncia. El índice relativiza el VAN por la escala del proyecto; la TIR, el criterio siguiente, va más allá y prescinde también de la tasa de descuento.


\subsubsection{Tasa interna de retorno (TIR): la rentabilidad implícita del proyecto}
El segundo criterio, que en la práctica se usa tanto como el VAN, es la \textbf{tasa interna de retorno}, que mide la \textbf{rentabilidad implícita de una inversión}. Es la tasa de descuento que iguala el valor actual de los flujos de caja del proyecto con su inversión inicial, es decir, la que anula su VAN; es lo que KEYNES (1936) llamó eficiencia marginal del capital ([\authorfont{Ver Tema 3.A.34}]):\footnote{%
    La ecuación es un polinomio de grado $T$ en $x=1/(1+TIR)$, de modo que, salvo en proyectos de uno o dos periodos, no tiene solución explícita y se resuelve por aproximaciones sucesivas. En el ejemplo, $TIR_A=CF_A/I_A-1$ y $TIR_B=\sqrt{CF_B/I_B}-1$.
}\textsuperscript{,}\footnote{%
    \textbf{OJO.-} KEYNES (1936) definió la eficiencia marginal del capital como la tasa que iguala el valor actual de los rendimientos esperados de un bien de capital con su precio de oferta, es decir, como la TIR de un proyecto. El tipo de rendimiento sobre el coste de FISHER (1930), con el que se la suele identificar, se define en cambio al comparar dos alternativas: es la tasa que iguala sus valores actuales. ALCHIAN (1955) mostró que coincide con la TIR del proyecto diferencial, que veremos al tratar el punto de FISHER.
}
$$-I_0+\sum_{t=1}^{T}\frac{CF_t}{(1+TIR)^t}=0.$$
Para un proyecto, la TIR es lo que la rentabilidad al vencimiento es para un bono ([\authorfont{Ver Tema 3.B.24}]): el rendimiento que obtiene quien paga hoy $I_0$ a cambio de la serie de flujos.

La regla de decisión compara esa rentabilidad con el coste de oportunidad: el proyecto se acepta si su TIR supera la tasa de descuento, $TIR>r$, y se rechaza si es inferior. La lógica es la del arbitraje: si el proyecto rinde más de lo que los mismos fondos obtendrían en el mercado, crea valor. La TIR es, así, el mayor coste del capital que el proyecto puede soportar sin destruir valor, y la diferencia $TIR-r$ mide su margen de seguridad frente a una subida de ese coste. Cuando el VAN decrece con la tasa de descuento, como en un proyecto en que primero se invierte y después se cobra, la regla coincide con la del VAN: $VAN(r)>0$ si y solo si $r<TIR$.

De ahí sus dos ventajas como complemento del VAN, con el que comparte el objetivo de comparar rentabilidades con el coste del capital. La primera es que expresa la creación de valor en unidades de tipo de interés y no en unidades monetarias, y que puede calcularse sin conocer la tasa de descuento, que solo hace falta para compararla: sortea así, en parte, la dificultad de medir el coste del capital, porque basta saber si es mayor o menor que la TIR. La segunda es que, como el índice de rentabilidad, permite comparar la rentabilidad de proyectos de distinta escala, entre los que el VAN, por construcción, favorece a los grandes aunque rindan menos por euro invertido. Esa segunda ventaja es también su principal riesgo, porque, cuando los proyectos son excluyentes, el de mayor TIR no es necesariamente el que más valor crea. Los dos resultados que siguen se obtienen al relacionar el VAN y la TIR a través del \textbf{perfil del VAN}, el VAN como función de la tasa de descuento.


\paragraph{El perfil del VAN: cuándo la TIR no existe o no es única}
La TIR es el punto en que el perfil del VAN, la función $VAN(r)$, corta el eje. Que ese punto exista, sea único y separe los tipos que crean valor de los que lo destruyen depende de la forma del perfil, y esta, del calendario de signos de los flujos. En un proyecto convencional, en que los flujos negativos se acumulan en los primeros periodos y los positivos en los últimos, el perfil decrece con el tipo de descuento, porque al subir $r$ pierden más valor los cobros, que son lejanos, que los pagos, que son próximos. Hay entonces una sola TIR, el VAN es positivo a su izquierda y negativo a su derecha, y la regla de la TIR coincide con la del VAN (panel a). Es el caso de A y de B en el ejemplo.

Si los flujos cambian de signo más de una vez, esa coincidencia se pierde. La regla de los signos de DESCARTES acota el número de TIR por el de cambios de signo en la serie de flujos: con dos cambios puede haber dos. Supongamos que A obliga a restaurar el terreno al cabo de dos periodos, con un coste $C$. Sus flujos pasan a ser $-I_A$, $+CF_A$ y $-C$, y, si la restauración es lo bastante costosa, el perfil crece primero y decrece después: el VAN es negativo con tipos bajos, que dan mucho peso al coste final, y con tipos altos, que restan valor al cobro, y es positivo entre sus dos TIR (panel b).\footnote{%
    Con $x=1/(1+r)$, $VAN_A=-I_A+CF_A\,x-C\,x^2$ es una parábola en $x$. Tiene dos TIR positivas si es negativa en $r=0$, $CF_A<I_A+C$, y positiva en su máximo, $CF_A^2>4\,I_A\,C$. LORIE y SAVAGE (1955) ilustraron el caso con una bomba que acelera la extracción de un pozo de petróleo: adelanta al primer año un ingreso que, sin ella, se obtendría en el segundo. Con una inversión de $1,6$ mil dólares y $10$ mil dólares de petróleo adelantados, sus flujos son $-1,6$, $+10$ y $-10$ mil dólares, y su VAN se anula con tipos del $25\%$ y del $400\%$.
} ¿Cuál es entonces la verdadera TIR? Ninguna de las dos mide la rentabilidad del proyecto, porque el proyecto no es una inversión pura: después del cobro, la empresa debe un pago futuro, como quien ha recibido un préstamo, y para esa parte un tipo de interés más alto es favorable. La TIR mezcla en un solo número una inversión y una financiación, y por eso deja de ordenar. La decisión correcta la sigue dando el VAN: el proyecto crea valor si el coste del capital está entre las dos TIR.

En el caso extremo, la TIR no existe. Un proyecto en que el cliente paga un anticipo, la empresa ejecuta la obra y cobra el resto al final tiene flujos $+$, $-$, $+$, y puede tener un VAN positivo a cualquier tipo de descuento sin cortar nunca el eje (panel c): la TIR no dice nada, mientras que el VAN dice que se acometa.\footnote{%
    La inexistencia de la TIR, y otras cuestiones de su cálculo, se tratan en el capítulo $17$ de VERNIMMEN, QUIRY y LE FUR (2022).
}

\imagenfit{perfil_van.pdf}{El perfil del VAN y la TIR según el calendario de signos de los flujos}{Elaboración propia a partir de LORIE y SAVAGE (1955).}

La enseñanza es que la TIR es una propiedad de la serie de flujos, no de la decisión: solo es un criterio fiable cuando el perfil del VAN es decreciente. Incluso entonces, como veremos a continuación, puede ordenar mal proyectos excluyentes.

\paragraph{El punto de FISHER: cuándo VAN y TIR ordenan distinto}
El segundo resultado aparece al comparar proyectos excluyentes, aunque ambos sean convencionales y tengan el mismo riesgo. En el ejemplo, A es pequeño y rápido y B, grande y lento. Supongamos que B crea más valor sin descontar, $CF_B-I_B>CF_A-I_A$, pero que A tiene mayor TIR. Sus perfiles se cortan: con tipos bajos, B tiene mayor VAN; con tipos altos, A. El tipo en que se cortan, en que los dos proyectos tienen el mismo VAN, es la \textbf{tasa de FISHER}, $r_F$, y la decisión correcta depende del lado en que quede el coste del capital: si $r<r_F$, se elige B, aunque su TIR sea menor; si $r>r_F$, se elige A. La TIR, en cambio, elige A siempre, y se equivoca en todo el tramo $r<r_F$. Por eso el punto de FISHER es útil cuando hay que elegir entre proyectos que no pueden acometerse a la vez: dice para qué tipos de descuento conviene cada uno.

\imagenfit{punto_fisher.pdf}{El punto de FISHER: VAN y TIR ordenan distinto}{Elaboración propia a partir de FISHER (1930).}

La tasa de FISHER es la TIR del \textbf{proyecto diferencial}, $B-A$, cuyos flujos son la diferencia entre los de B y los de A: es la rentabilidad de los fondos adicionales que exige elegir B en lugar de A.\footnote{%
    De ahí que coincida con el tipo de rendimiento sobre el coste de FISHER (1930), definido precisamente al comparar dos alternativas (ALCHIAN, 1955). En el ejemplo, los flujos del proyecto diferencial son $-(I_B-I_A)$, $-CF_A$ y $+CF_B$; como solo cambian de signo una vez, su TIR es única. La tasa de FISHER no existe cuando un proyecto tiene mayor VAN que el otro a cualquier tipo de descuento, y entonces VAN y TIR coinciden en la elección.
} Ese es el criterio que reconcilia ambos instrumentos. Elegir B equivale a acometer A más el proyecto diferencial, y este conviene si su TIR supera el coste del capital, es decir, si $r<r_F$. Aplicada a los incrementos, la TIR lleva a la misma decisión que el VAN.

¿Por qué se cortan los perfiles? Por dos diferencias entre los proyectos. La primera es la \textbf{escala}: B exige más inversión y, con tipos bajos, genera más valor en euros, aunque menos por euro. La segunda es el \textbf{calendario}: los flujos de B llegan más tarde, y el valor de los flujos lejanos es más sensible al tipo de descuento. La pendiente del perfil es proporcional al valor actual de los flujos y a su duración, el plazo medio ponderado en que se cobran, de modo que un proyecto lento se comporta como un bono largo: su valor cae más cuando sube el tipo de interés.\footnote{%
    Derivando, $dVAN/dr=-\sum_{t}t\,CF_t/(1+r)^{t+1}=-D\cdot VA/(1+r)$, donde $VA$ es el valor actual de los flujos y $D$, su duración de MACAULAY ([\authorfont{Ver Tema 3.B.24}]).
}

El punto de FISHER ilustra, así, la sensibilidad del VAN al tipo de descuento y la necesidad de emplear el VAN y la TIR como indicadores complementarios, pero la regla de selección debe aplicarse con cautela. Hereda los supuestos de los criterios que la componen, como la certeza y el reduccionismo con que resumen las características de los proyectos. Si los proyectos difieren en riesgo, compararlos con un mismo tipo de descuento no es correcto, porque cada uno tiene su coste del capital; y si difieren en duración, la comparación debe tener en cuenta qué ocurre cuando termina el más corto.\footnote{%
    Si A dura un periodo y B dos, elegir A deja abierta la posibilidad de invertir de nuevo al cabo de un periodo. Cuando los proyectos se repiten, se comparan cadenas de reemplazo de igual duración o el VAN expresado como anualidad equivalente: la renta constante que, durante la vida de cada proyecto, tiene su mismo valor actual (BREALEY, MYERS y ALLEN, 2020).
}

\paragraph{Limitaciones de la TIR}
Los supuestos de fondo de la TIR son los del VAN (certeza, flujos independientes y maximización del valor), y sus limitaciones comunes se tratan más adelante. Pero la TIR tiene, además, limitaciones propias, que nacen de resumir un proyecto en una tasa y no en un valor.

La primera, como vimos, es que puede no existir o no ser única cuando los flujos cambian de signo más de una vez, y entonces no ordena los proyectos. La segunda es que prescinde de la escala y del calendario: ordenar proyectos excluyentes por su TIR favorece a los pequeños y a los rápidos, y la decisión correcta exige calcular la TIR del proyecto diferencial, lo que en la práctica equivale a hacer el análisis del VAN. La tercera es que no es aditiva: la TIR de dos proyectos acometidos juntos no es la suma ni la media de sus TIR, de modo que no puede evaluarse cada proyecto por separado y agregar, y un proyecto malo puede esconderse dentro de uno bueno. La cuarta es que se compara con una sola tasa: si el coste del capital varía con el plazo, porque la curva de tipos no es plana, no hay un único umbral con el que comparar la TIR, mientras que el VAN descuenta cada flujo con su tipo.

La quinta es objeto de debate: la \textbf{tasa de reinversión}. SOLOMON (1956) atribuyó los conflictos entre VAN y TIR a que cada criterio supone implícitamente que los flujos intermedios se reinvierten a una tasa distinta: el VAN, al tipo de descuento, y la TIR, a la propia TIR. Si esa tasa es irreal, como en un proyecto con una TIR muy alta, la TIR exagera la rentabilidad. De ahí la \textbf{TIR modificada}, que capitaliza los flujos intermedios al coste del capital hasta el final del proyecto y calcula la tasa que iguala ese valor final con la inversión inicial, y que tiene siempre una única solución. DUDLEY (1972) y KEANE (1979) objetaron que ninguno de los dos criterios supone nada sobre la reinversión: ambos se calculan solo con los flujos del proyecto, y la TIR es una propiedad de esa serie, como la rentabilidad al vencimiento de un bono, que no depende de lo que su tenedor haga con los cupones. La reinversión solo importa al comparar proyectos con calendarios distintos, y entonces la pregunta relevante es qué rinden los fondos que uno libera antes que el otro: en un mercado de capitales perfecto, el coste del capital, que es lo que supone el VAN. El debate no cambia, por tanto, la conclusión: los conflictos se resuelven a favor del VAN, y la TIR es útil como información complementaria sobre el margen de rentabilidad del proyecto. El criterio siguiente, el periodo de recuperación, renuncia a ambos: no pregunta cuánto rinde el proyecto, sino cuándo devuelve lo invertido.


\subsubsection{Periodo de recuperación (\textit{payback}): cuándo se devuelve la inversión}
El último criterio para evaluar los proyectos con información perfecta es el \textbf{periodo de recuperación} o payback, que mide el \textbf{tiempo necesario para recuperar la inversión} inicial. Se expresa en unidades de tiempo (años o meses, según la periodicidad de los flujos de caja) y, en su versión simple, es el primer periodo $P$ en que los flujos acumulados del proyecto igualan, al menos, la inversión inicial:\footnote{%
    Si los flujos son constantes, $CF_t=CF$, el payback es simplemente $P=I_0/CF$. Su versión descontada, $P_d$, acumula los flujos descontados, $\sum_{t=1}^{P_d}CF_t/(1+r)^t\geq I_0$: es el plazo a partir del cual el VAN del proyecto, si terminase en ese momento, deja de ser negativo. Con flujos constantes, $P_d=-\ln(1-r\,I_0/CF)/\ln(1+r)$, que solo existe si $CF/I_0>r$, es decir, si la renta perpetua que genera el proyecto cubre la inversión.
}
$$\sum_{t=1}^{P}CF_t\geq I_0.$$
La regla de decisión fija un plazo máximo, $P^*$, que actúa como punto de corte: el proyecto se acepta si se recupera antes, $P\leq P^*$, y se descarta en caso contrario.

A diferencia del VAN y de la TIR, el payback no mide la rentabilidad del proyecto, sino la rapidez con que devuelve los fondos, y eso explica que su uso esté extendido pese a sus defectos: lo empleaba siempre o casi siempre más de la mitad de los directores financieros encuestados por GRAHAM y HARVEY (2001), sobre todo en las empresas pequeñas. Es un método adecuado, en primer lugar, cuando la empresa tiene restricciones en su acceso al mercado de capitales (de liquidez, o un alto coste del endeudamiento): si los fondos que devuelve el proyecto son los que financiarán los siguientes, recuperarlos pronto tiene un valor que el VAN, que supone un mercado de capitales perfecto, no recoge. En segundo lugar, es una aproximación al riesgo, aunque tosca: los flujos lejanos suelen ser los más inciertos, y un proyecto que se recupera pronto mantiene menos tiempo expuesto el capital comprometido. En tercer lugar, es sencillo de calcular y de comunicar, lo que permite a una gran empresa filtrar muchos proyectos menores sin un análisis completo de cada uno. Por último, la preferencia por los proyectos que devuelven pronto responde también a los incentivos de los directivos: quien es evaluado por sus resultados a corto plazo tiene motivos para preferirlos aunque creen menos valor (NARAYANAN, 1985).\footnote{%
    McDONALD (2000) sostiene, además, que reglas prácticas como un plazo de recuperación exigente o una tasa mínima de rentabilidad superior al coste del capital pueden aproximar la decisión óptima cuando invertir hoy supone renunciar a la opción de esperar, que veremos con la información imperfecta.
} Por eso el payback no tiene por fin determinar el proyecto más rentable, como el VAN o la TIR, sino complementar ese estudio en empresas con difícil acceso a la liquidez o en grandes empresas que afrontan proyectos menores.

Idealmente, además, debería calcularse descontando los flujos futuros. No hacerlo, aunque es más sencillo y habitual entre los analistas, supone omitir el valor temporal del dinero, y el payback simple solo vale como aproximación con inflación baja y coste de oportunidad reducido. El \textbf{payback descontado} corrige ese defecto y es siempre mayor que el simple. En el ejemplo, A devuelve su inversión en un periodo y B, en dos. Con un plazo máximo de un periodo, la regla rechaza B aunque cree más valor que A: como la TIR, el payback favorece a los proyectos rápidos, pero por una razón más simple, porque no mira más allá del plazo.

\paragraph{Limitaciones del payback}
Sus limitaciones nacen de que no es un criterio de valor. La primera, y principal, es que no tiene en cuenta los flujos de caja que se generan después de recuperar la inversión, por lo que es una medida limitada de la rentabilidad y del riesgo a largo plazo y debe usarse junto con los demás instrumentos de valoración. Dos proyectos con el mismo payback pueden crear valores muy distintos, y la regla sesga la inversión contra los proyectos de maduración larga, como la investigación o las infraestructuras, que son a menudo los que más valor crean. Puede incluso aceptar proyectos con VAN negativo, si tienen pagos importantes después del plazo, como un coste de desmantelamiento o la restauración del terreno en el proyecto A.

La segunda es que el plazo máximo es arbitrario: el VAN y la TIR toman su umbral del coste de oportunidad del capital, mientras que $P^*$ no se deriva de ningún criterio económico. GORDON (1955) mostró que, en un proyecto de vida larga con flujos constantes, la inversa del payback aproxima la TIR, de modo que exigir un plazo de tres años equivale a exigir una rentabilidad cercana al $33\%$, muy superior al coste del capital de la mayoría de las empresas.\footnote{%
    Con flujos constantes durante $T$ periodos, la TIR resuelve $I_0/CF=[1-(1+TIR)^{-T}]/TIR$; si $T$ es grande, el término $(1+TIR)^{-T}$ tiende a cero y $TIR\approx CF/I_0=1/P$. La aproximación empeora cuanto más corta es la vida del proyecto.
}

La tercera es que el payback simple da el mismo peso a todos los flujos anteriores al plazo, sin descontarlos; el descontado corrige este defecto, pero no el primero. La cuarta es que, como aproximación al riesgo, solo recoge el que depende del tiempo, y no la dispersión de los resultados posibles, que es lo que miden los instrumentos que veremos con la información imperfecta. El payback es, en suma, un indicador de liquidez y de exposición temporal, útil como complemento, pero no un criterio para elegir entre proyectos.

\subsubsection{Limitaciones comunes}
Los tres criterios comparten unos supuestos y, con ellos, unas limitaciones que no se resuelven eligiendo uno u otro. 

\paragraph{Tasa de descuento: ...}
La \textbf{tasa de descuento} determina el resultado del análisis y, sin embargo, no es observable directamente, sino que debe estimarla el analista. 

Como sabemos, debe reflejar el coste de oportunidad de la inversión ajustado por su riesgo, porque los proyectos con un elevado nivel de incertidumbre aumentan el riesgo global de la empresa. 

A esta cuestión dedicamos el último bloque del tema. Su efecto sobre el resultado es mayor cuanto más lejanos son los flujos: como vimos con el punto de FISHER, el VAN de un proyecto lento es muy sensible al tipo de descuento, de modo que un error en la tasa puede invertir la decisión. Por eso conviene acompañar el VAN de un análisis de su sensibilidad a la tasa, que veremos con la información imperfecta.

\paragraph{Regularidad: ¿calendario y reinversión?}
La segunda es el \textbf{calendario y la reinversión de los flujos}. Los criterios suponen que los flujos ocurren a intervalos temporales regulares y al final de cada periodo, lo que permite comparar flujos de momentos distintos. Pero también, tácitamente, supone que los fondos que genera el proyecto pueden reinvertirse con una rentabilidad igual a la tasa de descuento. Como vimos con la TIR, este segundo supuesto es objeto de debate, pero importa cuando no se cumple: si la empresa sufre racionamiento de capital, los fondos que libera un proyecto valen más que el tipo de interés, y el VAN calculado con el coste del capital los infravalora.\footnote{%
    Si los flujos no son regulares, se descuentan por sus fechas exactas (en las hojas de cálculo, con funciones como VNA.NO.PER). 
    
    Si la tasa de reinversión difiere del coste del capital, puede hacerse explícita: se capitalizan los flujos intermedios a esa tasa hasta el final del horizonte y se descuenta el valor final, que es la lógica de la TIR modificada; y, si hay racionamiento en varios periodos, se resuelve el programa matemático de WEINGARTNER (1963), cuyos precios duales valoran los fondos de cada periodo.
}

\paragraph{Independencia: ¿solo admite casos polares?}
La tercera es la \textbf{independencia de los flujos}. Los criterios evalúan cada proyecto por separado, como si sus flujos no dependieran de los de otros proyectos ni de las decisiones futuras de la empresa. Pero los proyectos interactúan, y sus flujos son endógenos a esas decisiones: uno puede restar ventas a otro, o abrir el camino a inversiones que sin él no serían posibles. Con interacciones, la estructura de los flujos no es lineal, y el valor de un conjunto de proyectos deja de ser la suma de sus valores.\footnote{%
    Las interacciones entre proyectos simultáneos se resuelven evaluando combinaciones: el VAN de cada alternativa es el del conjunto de proyectos que implica o, lo que es lo mismo, el del proyecto incremental. Las interacciones en el tiempo, en que un proyecto crea la posibilidad de otros futuros, son las opciones reales que veremos con la información imperfecta.
}

\paragraph{Objetivo: }
La cuarta es el \textbf{objetivo}. Los criterios suponen que la decisión persigue estrictamente maximizar el valor actual neto, y no consideran factores difíciles de monetizar, como las consideraciones estratégicas, la responsabilidad social corporativa o los problemas de información y de gestión que acompañan a un proyecto concreto, salvo que se incorporen explícitamente en los flujos o en la tasa de descuento. Detrás hay dos problemas distintos. 

Uno es de \textbf{alcance}: el valor para los propietarios no recoge los efectos del proyecto sobre terceros, salvo que la regulación o los precios los internalicen, y su evaluación desde el punto de vista social exige un análisis coste-beneficio. 

El otro es de \textbf{incentivos y de juicio}, porque quien estima los flujos no es neutral. Los directivos pueden perseguir objetivos propios, como el tamaño de la empresa (JENSEN y MECKLING, 1976; [\authorfont{Ver Tema 3.B.3}]), y quienes proponen un proyecto tienden a sobrestimar sus beneficios y a subestimar sus costes: es el sesgo optimista que KAHNEMAN y LOVALLO (1993) atribuyen a la visión desde dentro, la de quien razona a partir de los detalles de su proyecto y no de lo que ocurrió en proyectos parecidos.\footnote{%
    El Eurotúnel, inaugurado en 1994, es un caso emblemático. Su construcción costó un $80\%$ más de lo presupuestado, y su financiación, un $140\%$ más; las previsiones anunciaban $15,9$ millones de pasajeros en los trenes Eurostar en el primer año de explotación, y en 1995, el primer año completo, fueron $2,9$ millones, un $18\%$ de lo previsto (FLYVBJERG, BRUZELIUS y ROTHENGATTER, 2003). 
    
    El caso no es aislado: en los proyectos ferroviarios estudiados por estos autores, el sobrecoste medio fue del $44,7\%$ y el déficit medio de demanda, del $51,4\%$, y lo atribuyen tanto al sesgo optimista como a la presentación interesada de las cifras para lograr la aprobación del proyecto. La corrección que proponen KAHNEMAN y LOVALLO (1993) es adoptar la «visión desde fuera»: ajustar la previsión con la distribución de los resultados de proyectos parecidos.
}

\paragraph{Información: ...}
La quinta, que abre el siguiente apartado, es la \textbf{información}. Los tres criterios suponen que los flujos se conocen, o que su incertidumbre puede resumirse en un valor esperado y en una prima en la tasa de descuento, y que la decisión se toma ahora o nunca: invertir hoy o renunciar para siempre. 

Ninguno recoge que la información se revela de forma progresiva y que la empresa puede esperar, ampliar, reducir o abandonar el proyecto según lo que aprenda. Esa flexibilidad tiene valor, y medirlo exige abandonar el supuesto de información perfecta.

\subsection{Información imperfecta: el valor de la información y la medida del riesgo}
Hasta aquí, la información perfecta reducía la decisión a comparar la rentabilidad del proyecto con el coste del tiempo. A partir de ahora, la pregunta es cómo afectan al proyecto, a su rentabilidad y a su calendario, unas contingencias futuras que no se conocen de antemano. Como vimos al cerrar el apartado anterior, los criterios clásicos tienen un \textbf{carácter estático}: suponen implícitamente que la decisión de inversión se toma una vez y es irreversible, y que la información relevante está disponible \textit{ex ante}. En muchos proyectos reales, en cambio, la información se revela de forma progresiva y las decisiones óptimas pueden variar a lo largo del tiempo, porque una inversión rentable en un momento puede dejar de serlo después, al cambiar el contexto con el que se relaciona.\footnote{%
    \textbf{OJO.-} Este fenómeno no es la inconsistencia dinámica de KYDLAND y PRESCOTT (1977), en que un plan óptimo hoy deja de serlo mañana sin que llegue información nueva, porque cambian los incentivos de quien decide ([\authorfont{Ver Tema 3.A.36}]). Aquí el plan cambia porque llega información, y prever ese cambio forma parte de la decisión óptima.
} Es la consecuencia lógica de uno de los fenómenos que perturban el mundo real: la \textbf{información imperfecta}.

La información es imperfecta en dos dimensiones ([\authorfont{Ver Tema 3.A.13}]). Es \textbf{incompleta} cuando nadie conoce de antemano el estado de la naturaleza que se dará, aunque todos compartan la misma información; y es \textbf{asimétrica} cuando unos agentes saben más que otros. La información asimétrica distorsiona las decisiones de inversión de formas propias: los directivos, que saben más que los inversores, pueden renunciar a proyectos rentables para no emitir acciones infravaloradas (MYERS y MAJLUF, 1984) o invertir los fondos sobrantes en proyectos que no crean valor (JENSEN, 1986). Esos problemas se estudian en el Tema 3.B.3 ([\authorfont{Ver Tema 3.B.3}]); aquí nos ocupamos de la información incompleta, la que afecta al propio proyecto.

La información incompleta entra en la valoración por tres vías, y por eso no hay un único método. La primera es el \textbf{precio del riesgo}: si los flujos son inciertos, se sustituyen por su valor esperado y el descuento añade una prima, $r=\gamma+\lambda$, que remunera el riesgo que los inversores no pueden diversificar. Es la forma habitual de incorporar la incertidumbre, y su medida es el objeto del bloque siguiente.\footnote{%
    La alternativa es descontar al tipo sin riesgo los \textbf{equivalentes ciertos} de los flujos, la cantidad segura que se valora igual que cada flujo incierto. ROBICHEK y MYERS (1966) mostraron que descontar con una tasa ajustada constante supone que el riesgo crece con el plazo a un ritmo determinado, lo que no es adecuado cuando la incertidumbre se resuelve en una fecha concreta del proyecto; los equivalentes ciertos permiten ajustar por separado el riesgo de cada flujo.
} La segunda es el \textbf{valor de la información}: si la incertidumbre se resuelve con el tiempo y la empresa puede reaccionar, poder esperar, ampliar o abandonar el proyecto tiene un valor que el VAN estático ignora. La tercera es la \textbf{dispersión de los resultados}: cuánto pueden desviarse la rentabilidad y el valor del proyecto de lo esperado, y qué variables lo explican. A medir la segunda dedicamos el primer apartado, y a medir la tercera, en cualquier contexto, el segundo.

\subsubsection{Valor: qué precio tiene la información?}
Para entender por qué hacen falta métodos que incorporen los resultados contingentes, hay que saber primero cuánto vale la información que falta. Volvamos al proyecto A y supongamos que su flujo es incierto: será $CF^+$ si la demanda es alta, con probabilidad $p$, y $CF^-$ si es baja, con probabilidad $1-p$, de modo que, al tipo de descuento $r$, el proyecto crea valor con demanda alta y lo destruye con demanda baja. La demanda se conocerá al cabo de un periodo, y la empresa puede invertir hoy o esperar a conocerla.

Un simple \textbf{árbol de decisión} nos permite representar este problema de forma compacta y permite evaluar múltiples alternativas de inversión bajo diversos escenarios, con el consiguiente ahorro de tiempo y recursos en la selección de proyectos.\footnote{%
    \textbf{NOTA.-} Es un problema de un solo decisor frente a la naturaleza, no un juego entre agentes (RAIFFA, 1968): sus nodos son de decisión, en que el inversor elige entre las acciones disponibles; de azar, en que la naturaleza elige un estado con una probabilidad conocida; y terminales, con el resultado de cada trayectoria.
    
    Por ello, los nodos de azar no deben confundirse con los conjuntos de información de los juegos en forma extensiva, que agrupan los nodos que un jugador no puede distinguir ([\authorfont{Ver Tema 3.A.14}]). 
    
    En el árbol de decisión, el decisor conoce en cada nodo de decisión todo lo ocurrido hasta entonces. Aunque aquí se emplea como instrumento gráfico, el árbol es por definición un objeto matemático recursivo, y sirve también para hacer simulaciones, resolver problemas de optimización dinámica o predecir resultados con técnicas de aprendizaje automático.
} Se resuelve hacia atrás, con la regla de inducción del análisis de decisiones (RAIFFA, 1968) que vimos ene la programación dinámica ([\authorfont{Ver Tema 3.A.29}]).\footnote{%
  Se asigna a cada nodo $n$ un valor $u(n)$:
$$u(n)=\begin{cases}\text{su resultado,} & \text{si $n$ es terminal,}\\[0.3em] \sum_i p_i\,u(n_i), & \text{si es de azar,}\\[0.3em] \max_i u(n_i), & \text{si es de decisión,}\end{cases}$$
donde $n_i$ son los nodos que le siguen. El valor esperado del proyecto es el del nodo inicial.
}

\imagenfit{arbol_esperar.pdf}{Árbol de decisión: invertir hoy o esperar a conocer la demanda}{Elaboración propia a partir de RAIFFA (1968) y DIXIT y PINDYCK (1994).}

En el ejemplo, si la empresa invierte hoy, su VAN es el esperado, descontando los flujos al tipo $r$. Si espera (panel b), conoce la demanda antes de decidir, invierte solo si es alta y obtiene el flujo un periodo más tarde.\footnote{%
  Aplicando la regla anterior, los dos valores son
$$VAN_{\text{hoy}}=-I_A+\frac{p\,CF^++(1-p)\,CF^-}{1+r},\qquad VAN_{\text{esperar}}=\frac{p}{1+r}\left(-I_A+\frac{CF^+}{1+r}\right).$$
}

¿Qué decidiremos? Es sencillo, conviene esperar si $VAN_{\text{esperar}}>VAN_{\text{hoy}}$, y la diferencia es el \textbf{valor de esperar}, el de la información que se obtiene al hacerlo. Su origen está en la asimetría de la decisión: con la información, la empresa evita la pérdida del estado malo, porque no invierte, y conserva la ganancia del bueno. Su coste es el retraso de esa ganancia.\footnote{%
    Llamando $a=-I_A+CF^+/(1+r)>0$ y $b=-I_A+CF^-/(1+r)<0$ a los VAN de invertir hoy con demanda alta y baja, $VAN_{\text{hoy}}=p\,a+(1-p)\,b$ y $VAN_{\text{esperar}}=p\,a/(1+r)$, de modo que
    $$VAN_{\text{esperar}}-VAN_{\text{hoy}}=\underbrace{(1-p)\,|b|}_{\text{pérdida evitada}}-\underbrace{p\,a\,\frac{r}{1+r}}_{\text{coste del retraso}}.$$
    Si la información pudiera obtenerse sin esperar, su valor sería el primer término, el valor esperado de la información perfecta (RAIFFA y SCHLAIFER, 1961).
} De ahí un resultado que contradice la intuición del VAN con una prima de riesgo: si los flujos se vuelven más dispersos sin que cambie su media, el VAN de invertir hoy no varía, pero el de esperar aumenta, porque solo cuenta el estado bueno. Con flexibilidad, más incertidumbre aumenta el valor del proyecto.

Este tipo de análisis ofrece una evaluación sistemática de múltiples factores y facilita la elección de la mejor alternativa teniendo en cuenta tanto su rentabilidad como su riesgo: las opciones que no son atractivas se descartan en cada nodo, y, una vez desestimadas, el inversor solo tiene que escoger entre las alternativas del nodo inicial. 

Pero el árbol tiene también limitaciones. Un árbol en que todas las decisiones se toman al principio elige de antemano el sendero que seguirá el proyecto y trata las decisiones como irreversibles, cuando en la realidad, a medida que el proyecto avanza, aparece nueva información y, con ella, nuevas opciones, como incorporar nuevas tecnologías, cancelar el proyecto o aplazarlo. Representar esa flexibilidad exige nodos de decisión posteriores y el número de ramas crece entonces rápidamente con los periodos y los estados posibles. Además, y sobretodo, el árbol no dice con qué tasa descontar: la flexibilidad cambia el riesgo del proyecto, porque el valor de la oportunidad de esperar depende de forma no lineal del valor del proyecto, y una tasa ajustada única deja de ser correcta. Resolver ese problema es lo que aportan las opciones reales.

\paragraph{Cuantificación: ¿cuál es el valor de la flexibilidad?}
Incrementar u obtener cierta versatilidad en el transcurso de una inversión tiene un valor que el análisis convencional no refleja, y las \textbf{opciones reales} son la forma más extendida de incorporarlo a la decisión.\footnote{%
  MYERS (1977) acuñó el término para subrayar que muchas oportunidades de inversión son como opciones de compra sobre activos reales: representan el derecho, pero no la obligación, de modificar un proyecto de inversión cuando se dispone de nueva información.
}

La oportunidad de acometer A es una opción de compra cuyo subyacente es el valor actual de los flujos del proyecto, $V$; cuyo precio de ejercicio es la inversión, $I_A$; y cuyo vencimiento es el plazo durante el que puede posponerse la decisión (LUEHRMAN, 1998). Al vencimiento, la opción vale $\max\{V-I_A,\,0\}$, porque solo se ejerce si el proyecto crea valor; antes, vale más que el VAN de invertir hoy, porque conserva la posibilidad de no invertir si las cosas van mal. Por eso el valor de un proyecto flexible es su \textbf{VAN ampliado}, que según TRIGEORGIS (1996) es:
$VAN_{\text{ampliado}}=VAN_{\text{estático}}+\text{valor de las opciones}.$

La valoración de opciones es, en el fondo, el árbol de decisión con el descuento correcto. COX, ROSS y RUBINSTEIN (1979) mostraron que una opción puede valorarse con un árbol binomial en que, en cada nodo, su valor se replica con una cartera del subyacente y del activo sin riesgo, lo que equivale a calcular el valor esperado con unas probabilidades ajustadas por el riesgo y descontarlo al tipo sin riesgo, resolviendo el problema de la tasa. 

Cuando los periodos se acortan, el árbol converge al modelo de BLACK y SCHOLES (1973) ([\authorfont{Ver Tema 3.B.25}]), y podemos adaptarlo a una opción real de posponer, que toma su valor como:

\eqblock{
  \begin{aligned}
    C=S^*\,N(d_1)-V(I_0)\,N(d_2),\qquad d_1=\frac{\ln[S^*/V(I_0)]}{\sigma\sqrt{t}}+\frac{\sigma\sqrt{t}}{2},\qquad d_2=d_1-\sigma\sqrt{t}
  \end{aligned}
}{Información: ¿cuánto vale la opción de posponer? Información imperfecta: convergencia al modelo de BLACK-SCHOLES (1973)}

El valor actual de los flujos del proyecto menos el de los que se pierden por esperar es $S^*=S-V(CF_t)$. Por otro lado, $V(I_0)=I_t/(1+r)^t$ es el valor actual de la inversión pospuesta. En el campo informativo, $\sigma$ es la volatilidad del valor del proyecto, que suele aproximarse por la del sector, $t$ es el plazo hasta la decisión final y $N(\cdot)$ la función de distribución normal estándar. 

Entonces, el primer término es el valor de posponer: los flujos que se obtendrán si se ejerce la opción, ponderados por $N(d_1)$, la sensibilidad del valor de la opción al del proyecto (su delta). El segundo es el coste de posponer: la inversión, ponderada por $N(d_2)$. Esto último siendo la probabilidad, en un mundo neutral al riesgo, de que llegue a acometerse. 

La regla de decisión compara ese valor con el de invertir hoy: si el VAN de invertir hoy es mayor que el valor de la opción, $\mathbb E_0[VAN]>C$, no se pospone; si $C\geq\mathbb E_0[VAN]$, se pospone, porque invertir hoy destruye la opción.

De la analogía se siguen los determinantes del valor de una opción real, que son los de una opción financiera. Crece con la \textbf{volatilidad} del valor del proyecto: como la empresa solo invierte en los estados buenos, una mayor dispersión aumenta las ganancias posibles sin aumentar las pérdidas, que quedan limitadas a no invertir. Crece también con el \textbf{plazo} durante el que puede esperarse y con el tipo de interés, que reduce el valor actual del precio de ejercicio; y disminuye con los flujos que se pierden mientras se espera, que actúan como los dividendos de una acción. 

\imagenfit{opcion_valor.pdf}{Valor de la opción de invertir y VAN estático}{Elaboración propia a partir de DIXIT y PINDYCK (1994).}

Por eso invertir solo compensa si el valor del proyecto supera la inversión en un margen que crece con la incertidumbre: el umbral de inversión ya no es $V=I$, sino un valor $V^*>I$ (McDONALD y SIEGEL, 1986; DIXIT y PINDYCK, 1994), que la figura muestra para dos niveles de volatilidad.\footnote{%
  Las implicaciones para la inversión agregada, el retraso óptimo y la histéresis, se estudian en [\authorfont{Ver Tema 3.A.34}].
}

Si bien muchos proyectos encierran opciones reales, estas son específicas de cada tipo de inversión, y presentar una teoría general queda fuera del alcance de este tema. En la práctica aparecen, sobre todo, tres tipos. La \textbf{opción de posponer} es característica de las industrias en que los proyectos exigen enormes inversiones y son muy sensibles a variables como los precios o los tipos de interés, como la explotación de petróleo: la empresa puede considerar óptimo retrasar el comienzo del proyecto hasta que las condiciones del mercado sean propicias. La \textbf{opción de crecer} aparece cuando los gestores deben elegir la capacidad con que responder a un posible aumento de la demanda: un proyecto previsto para un crecimiento muy alto expone al riesgo de una inversión excesiva, y uno pequeño, a altos costes si después hay que ampliar la capacidad; la opción permite comenzar con una planta pequeña y ampliarla si el mercado lo demanda. La \textbf{opción de abandonar} permite renunciar a la inversión en cualquier etapa del proyecto y recuperar parte de lo invertido.\footnote{%
    TRIGEORGIS (1996) añade la opción de reducir la escala, la de cambiar de factores o de productos (flexibilidad operativa) y las opciones compuestas de los proyectos por etapas, como la investigación y el desarrollo, en que cada fase da derecho a acometer la siguiente.
} En suma, las opciones reales otorgan flexibilidad a los proyectos al permitir posponer, ampliar o abandonar una inversión, y están implícitas en casi cualquier proyecto.

Tienen, sin embargo, límites. La analogía con las opciones financieras descansa en que el proyecto podría replicarse con activos negociados, lo que rara vez ocurre: en la práctica se supone que el valor actual del propio proyecto, sin flexibilidad, es el mejor estimador de su valor de mercado (COPELAND y ANTIKAROV, 2001), y la volatilidad que se emplea es la de empresas del sector, que es solo una aproximación. Además, el valor de esperar supone que la oportunidad es exclusiva. Si un competidor puede adelantarse, esperar cuesta la cuota de mercado o la oportunidad misma, y la decisión pasa a ser un juego entre empresas en el que la competencia erosiona el valor de la opción y puede precipitar la inversión (GRENADIER, 2002; SMIT y TRIGEORGIS, 2004). Es aquí, y no en el árbol de decisión, donde entra la teoría de juegos. Por último, su uso formal es limitado: en la encuesta de GRAHAM y HARVEY (2001), en torno a una cuarta parte de los directores financieros declaraba incorporar opciones reales a la evaluación de proyectos.

\subsubsection{Dispersión: ¿cómo medir la incertidumbre en cualquier contexto}
La otra forma de incorporar la información incompleta parte de la imposibilidad de anticipar correctamente todas las contingencias relevantes. 

En la práctica, los flujos de caja, las tasas de descuento e incluso la propia estructura del proyecto están sujetos a errores de estimación y a sucesos adversos no previstos. El problema no radica ya en la secuencia de decisiones, sino en la robustez del análisis ante desviaciones respecto del escenario central: cuánto pueden alejarse la rentabilidad y el valor del proyecto de lo previsto, y qué variables lo explican. Para abordarlo se recurre al análisis de sensibilidad, al de escenarios y a la simulación de Montecarlo, que evalúan cómo varían los resultados del proyecto ante perturbaciones en los supuestos iniciales y proporcionan así una medida más realista del riesgo de la inversión.

A diferencia de los del apartado anterior, que miden cuánto vale reaccionar, estos responden a una misma pregunta, la dispersión del resultado, con una profundidad creciente: el análisis de sensibilidad varía una variable cada vez y no asigna probabilidades; el de escenarios combina varias de forma coherente; y la simulación de Montecarlo asigna a todas una distribución de probabilidad y obtiene la del resultado. Y se aplican en cualquier contexto: tanto al proyecto tal como se ha planteado, con un plan fijo, como al proyecto con flexibilidad, cuyo valor también depende de los supuestos.

\paragraph{Análisis de sensibilidad: ¿cómo varían ante diferentes escenarios?}
El \textbf{análisis de sensibilidad} permite identificar las variables críticas que influyen en el desempeño financiero de un proyecto, lo que mejora la gestión del riesgo y la toma de decisiones durante el proceso de inversión. 

Su propósito es reconocer los elementos que contribuyen al éxito o al fracaso de la inversión y medir cuánto lo hacen de manera marginal. Responde a preguntas como cuál es el escenario de costes que anula el VAN, cómo varía la TIR ante una caída de la demanda del $10\%$ o si los flujos de caja son sensibles a los precios. 

Modificando una variable, o unas pocas, y manteniendo las demás en su valor central, nos permite valorar con claridad el efecto de cada cambio sobre la viabilidad del proyecto.\footnote{%
    Es posible analizar la sensibilidad a dos o más variables a la vez, para examinar cómo su interacción afecta a la viabilidad del proyecto, pero el análisis es considerablemente más complejo que el de una variable y exige cuidado para obtener resultados fiables.
} Sin embargo, esa misma flexibilidad dificulta su formulación general, porque las modificaciones pueden influir sobre el resultado de formas muy diversas, según la variable y según el criterio sobre el que se haga el análisis.\footnote{%
  Para simplificar, puede hacerse sobre el VAN, teniendo en cuenta las variables de las que depende cada uno de sus componentes: los flujos de caja dependen de la demanda, de los costes, que varían con la cantidad producida, del gasto comercial y de la regulación; la tasa de descuento, del tipo sin riesgo y de la prima, $r=\gamma+\lambda$; y la inversión inicial, de su valor esperado y de un componente no esperado, los sobrecostes. 
  
  Si, por ejemplo, una pequeña caída de la demanda hace negativo el VAN, el proyecto es muy sensible, y por tanto arriesgado, ante variaciones de la demanda. 
}

Volvamos al proyecto A con el flujo incierto del árbol de decisión. Su VAN de invertir hoy, $VAN_A=-I_A+[p\,CF^++(1-p)\,CF^-]/(1+r)$, depende de cinco variables: la probabilidad de demanda alta, $p$, los flujos con demanda alta y baja, $CF^+$ y $CF^-$, el tipo de descuento, $r$, y la inversión, $I_A$. El análisis parte de un valor central de cada una, que da el VAN central, $VAN_0$, y las mueve de una en una.

\imagenfit{sensibilidad_paneles.pdf}{Análisis de sensibilidad del proyecto A: punto muerto y diagrama de tornado}{Elaboración propia.}

Una forma útil de expresarlo es el \textbf{punto muerto} de cada variable: el valor que anula el VAN, que dice cuánto margen tiene el proyecto frente a cada desviación. En el panel (a), el VAN de A crece con la probabilidad de demanda alta y se anula en $p^*=[(1+r)\,I_A-CF^-]/(CF^+-CF^-)$; la distancia entre el valor central, $p_0$, y $p^*$ es el margen frente a un error en esa probabilidad: si la demanda alta resulta menos probable que $p^*$, el proyecto destruye valor. Con las demás variables ocurre lo mismo: el punto muerto del tipo de descuento es la TIR, y el de la inversión, el valor actual de los flujos esperados. El panel (b), el \textbf{diagrama de tornado}, reúne el ejercicio para todas: cada barra recoge el intervalo del VAN al mover una variable entre sus valores pesimista y optimista, con las demás en su valor central, y las barras se ordenan de mayor a menor. Las que cruzan la línea de VAN nulo, en el ejemplo $p$ y $CF^+$, son las que pueden cambiar la decisión, y en ellas debe concentrarse el esfuerzo de estimación; las demás, como la inversión, apenas la afectan. El análisis de sensibilidad es, así, de vital importancia para una gestión adecuada del riesgo y, una vez tomada la decisión, el conocimiento de esos factores de riesgo resulta de gran utilidad para diseñar estrategias de cobertura ante cualquier eventualidad.\footnote{%
    Aunque, en muchas ocasiones, el peor escenario contemplado no es el más adverso que se produce en la práctica, como mostró el caso del Eurotúnel que vimos al tratar las limitaciones comunes de los criterios.
}

Tiene, sin embargo, dos limitaciones. La primera es que \textbf{omite las correlaciones}: al analizar por separado el efecto de cada variable, ignora las relaciones que suelen existir entre ellas.\footnote{%
    Por ejemplo, los cambios en el coste de los factores podrían estar asociados a cambios en el coste de la financiación o en la demanda.
} La segunda es que \textbf{no proporciona probabilidades}: las variaciones las fija el analista de forma discrecional, de modo que no dice cuán probable es que el proyecto destruya valor.\footnote{%
    \textbf{NOTA.-} El \textbf{análisis de escenarios} corrige la primera, porque mueve a la vez y de forma coherente las variables que van juntas (una recesión reduce la demanda y los precios y encarece la financiación) y calcula el VAN en unos pocos escenarios, típicamente pesimista, central y optimista; llevado al extremo, con escenarios adversos pero plausibles, es una prueba de resistencia. Pero sigue sin asignar probabilidades.
}

\paragraph{Simulación de Montecarlo: ¿una herramienta para todo?}
Para sortear las limitaciones del análisis de sensibilidad puede recurrirse a la \textbf{simulación de Montecarlo}, uno de los instrumentos más utilizados para modelizar y estimar en las ciencias sociales, y a menudo también en las ciencias puras. Es un método no determinista que aproxima problemas complejos y costosos de resolver con exactitud generando un gran número de realizaciones aleatorias.\footnote{%
    Para una introducción al método, \href{https://ocw.mit.edu/courses/6-0002-introduction-to-computational-thinking-and-data-science-fall-2016/resources/lecture-6-monte-carlo-simulation/}{Lecture 6: Monte Carlo Simulation (MIT OpenCourseWare)}; y, para su aplicación al VAN, \href{https://www.youtube.com/watch?v=L5uAPOE8lMM}{Monte Carlo Simulation to Assess Net Present Value (YouTube)}.
} 

HERTZ (1964) lo introdujo en la evaluación de proyectos: en lugar de un único valor para cada variable, el analista especifica su distribución de probabilidad y sus correlaciones, y obtiene la distribución del resultado. El procedimiento tiene cuatro pasos: (i) identificar las distribuciones de probabilidad que mejor se ajustan al comportamiento de las variables relevantes, como el precio de venta, los costes o la tasa de descuento, reuniendo toda la información disponible y con ayuda del análisis estadístico; (ii) generar, a partir de ellas, un gran número de sendas posibles de esas variables; (iii) calcular el VAN en cada uno de los escenarios simulados; y, (iv) analizar la distribución resultante y evaluarla en función de los objetivos de la empresa. 

Apliquémosla a los dos proyectos del ejemplo. Si los flujos de A y de B dependen de la demanda, de los precios y de los costes, cada uno con su distribución y con sus correlaciones, la simulación da la distribución del VAN de cada proyecto (panel a). B tiene un VAN esperado mayor, como vimos con información perfecta; pero, si, como es habitual en un proyecto mayor y más lento, sus flujos son más inciertos, su distribución es también más dispersa. El panel (b) presenta el mismo resultado como \textbf{perfil de riesgo}, la probabilidad acumulada de cada valor del VAN, que es como HERTZ (1964) proponía presentar el resultado para comparar inversiones: en el VAN nulo se lee la probabilidad de que cada proyecto destruya valor, mayor en B que en A.

\imagenfit{montecarlo_paneles.pdf}{Simulación de Montecarlo de los proyectos A y B: distribución y perfil de riesgo del VAN}{Elaboración propia a partir de HERTZ (1964).}

Como vemos, a diferencia del análisis de sensibilidad, recoge el efecto conjunto de todas las variables, con sus correlaciones, y dice, por ejemplo, cuál es la probabilidad de que el proyecto no recupere la inversión, lo que permite al analista centrar su atención en las variables que más impulsan el valor del proyecto. Lo que no dice es cuál de los dos proyectos elegir: que un VAN esperado mayor con más dispersión sea mejor o peor depende de qué riesgo importe, como veremos al tratar sus límites.

La simulación no se limita al plan fijo. Si en cada fecha la empresa puede decidir, por ejemplo, abandonar el proyecto, la simulación puede incorporar esa regla: LONGSTAFF y SCHWARTZ (2001) mostraron cómo estimar en cada fecha, a partir de las sendas simuladas, el valor de continuar y decidir de forma óptima, lo que permite valorar opciones reales con varias fuentes de incertidumbre, para las que no hay fórmulas como la de BLACK y SCHOLES: es la herramienta que sirve en cualquier contexto.

Su aplicación tiene, no obstante, tres límites. El primero es práctico: la precisión del análisis depende de manera crucial de que las distribuciones de probabilidad y las correlaciones se ajusten a la realidad, y si no lo hacen los resultados pueden ser inexactos o sesgados; además, si el número de simulaciones es demasiado pequeño, la distribución obtenida puede no reflejar la verdadera, y las simulaciones grandes o complejas pueden ser costosas en tiempo y en cálculo, aunque existan programas especializados. 

El segundo es conceptual: \textbf{qué tasa de descuento emplear}. Si cada senda se descuenta con una tasa ajustada por el riesgo, el riesgo se cuenta dos veces, en la tasa y en la dispersión; si se descuenta al tipo sin riesgo, la distribución obtenida no es la de valores de mercado del proyecto. Por eso se recomienda usar la simulación para obtener los flujos esperados y descontarlos después con la tasa ajustada (BREALEY, MYERS y ALLEN, 2020), o, para valorar opciones, simular con probabilidades neutrales al riesgo y descontar al tipo sin riesgo.

El tercero es de fondo: \textbf{qué riesgo importa}. La dispersión del VAN mide el \textbf{riesgo total} del proyecto, pero los propietarios con carteras diversificadas solo exigen una prima por el riesgo que no pueden eliminar diversificando, el \textbf{riesgo sistemático}, que es el que recoge la tasa de descuento ([\authorfont{Ver Tema 3.B.23}]). El riesgo total importa, en cambio, cuando hay fricciones: si la financiación externa es cara, una caída de los fondos internos obliga a renunciar a inversiones rentables, y la empresa tiene motivos para comportarse como si fuera aversa al riesgo y para cubrirlo (FROOT, SCHARFSTEIN y STEIN, 1993); también importan los costes de insolvencia ([\authorfont{Ver Tema 3.B.3}]) y la aversión al riesgo de unos directivos que no pueden diversificar su empleo. Las preferencias ante el riesgo, que con mercados perfectos no afectan a la decisión, vuelven a importar cuando los mercados son imperfectos. En resumen, la simulación de Montecarlo es una herramienta útil para modelizar los escenarios y resultados que puede atravesar un proyecto y precisar sus riesgos, y, como el análisis de sensibilidad, permite reconocer los factores clave que impulsan su valor; pero la medida de ese riesgo que el mercado remunera, la tasa de descuento, es el objeto del bloque siguiente.





\clearpage
\section{Coste del capital: ¿a qué tasa se descuenta un proyecto?}
\puntosclave{
    El coste del capital es el coste de oportunidad del proyecto, la rentabilidad esperada de inversiones de riesgo comparable, $r=\gamma+\lambda$: el precio del tiempo más el del riesgo. Solo se remunera el riesgo sistemático, cuya cantidad mide la beta; en el CAPM, la prima es la beta por su precio, $\lambda=\beta\,\lambda_M$. De la precisión con que se estime depende la fiabilidad del VAN como criterio de selección de proyectos.

    Es una propiedad del proyecto, no de la empresa: el WACC de la empresa solo debe emplearse si el proyecto tiene un riesgo similar al de la compañía. Una tasa única lleva a rechazar proyectos seguros que crean valor y a aceptar proyectos arriesgados que lo destruyen (falacia del WACC).

    El coste del proyecto no se observa y se infiere del de una empresa comparable, estimado por fuentes. El de sus recursos propios tampoco se observa: el CAPM lo deduce de la beta de las acciones, del tipo sin riesgo del plazo de los flujos y de la prima de mercado. Sus limitaciones, la dependencia de datos históricos y una relación entre beta y rentabilidad más plana que la teórica, han llevado a los modelos de varios factores y a la rentabilidad implícita en el precio (GORDON y SHAPIRO). Los métodos no tienen por qué coincidir, y ninguno evita errores de varios puntos.

    El coste de la deuda es el esperado por los acreedores en la deuda nueva, a valor de mercado y después de impuestos. Las tres vías para estimarlo dan la rentabilidad prometida, que lo sobrestima en la deuda arriesgada.

    Betas y costes son aditivos: los de los activos son la media ponderada de los de las acciones y la deuda. El endeudamiento eleva el coste y la beta de las acciones sin cambiar los del negocio; por eso el WACC del proyecto se construye en tres pasos: desapalancar el coste de la comparable para obtener el del negocio, $r_A$, reapalancarlo con la estructura objetivo del proyecto y recomponer su WACC, que resulta ser el coste del negocio menos el ahorro fiscal de la deuda.

    El método supone, con MODIGLIANI y MILLER, que la financiación no altera el coste del negocio: si lo alterara, los costes serían endógenos y el cálculo, circular, por lo que el teorema sirve de referencia. Si el endeudamiento cambia con el tiempo, el APV separa el valor de la inversión del de su financiación.

    En la práctica, las empresas usan tasas únicas y tasas mínimas muy superiores al WACC, que reflejan más sus restricciones de capital y de gestión que el riesgo.
}

El primer bloque daba por conocida la tasa de descuento. Este responde a la pregunta que dejó abierta: ¿qué tasa es la apropiada para un proyecto? 

La tasa de descuento de un proyecto es su \textbf{coste del capital}, $r$: el coste de oportunidad de comprometer fondos en él, es decir, la rentabilidad esperada que sus financiadores obtendrían en la mejor inversión alternativa de riesgo comparable. 

Es una sola magnitud que recibe tres nombres según quién la mire: para el proyecto, es la tasa con que se descuentan sus flujos; para la empresa, lo que le cuesta financiarlo; y para el inversor, la \textbf{rentabilidad exigida}, la mínima que acepta por comprometer sus fondos con ese riesgo. Por eso tiene los dos componentes que ya conocemos:

\eqblock{
  \begin{aligned}
    r=\gamma+\lambda
  \end{aligned}
}{Coste del capital: ¿cuál es el coste de oportunidad del capital?}

Donde $\gamma$ es el \textbf{tipo sin riesgo}, la rentabilidad de un título sin riesgo de impago al mismo plazo, que remunera el tiempo, y $\lambda$ es la \textbf{prima de riesgo}, la rentabilidad adicional que se exige por soportar el riesgo de la inversión.\footnote{%
  Esta descomposición se sigue de cualquier modelo de valoración de activos con un activo sin riesgo; lo que cambia de un modelo a otro es cómo se mide $\lambda$. 
  
  Sin la prima, el proyecto se valoraría como un título sin riesgo con su mismo rendimiento esperado; pero ese título es una alternativa mejor, porque rinde lo mismo sin riesgo, y la prima mide precisamente esa diferencia.
}

El objeto de este bloque es medir esa tasa y discutir los problemas teóricos y prácticos que plantea.\footnote{%
  Proceden, sobre todo, de la falta de \textbf{datos precisos}, de la \textbf{elección del tipo sin riesgo}, de la \textbf{estimación de la prima} y de la \textbf{estructura de capital} que debe suponerse. 
}

\subsection{Coste de oportunidad: ¿cómo se compone el coste del capital de un proyecto?}
El coste del capital de un proyecto suma dos precios: el del tiempo, que es el mismo para todas las inversiones, y el de su riesgo, que no lo es. Que la prima difiera de un proyecto a otro depende de qué riesgo remunera, de cómo se mide y de cuál es su precio.

¿Qué riesgo remunera la prima? Como vimos al tratar la simulación de Montecarlo, no mide la dispersión total de los resultados, sino solo el \textbf{riesgo sistemático}: el que procede de la marcha de la economía en su conjunto y afecta a la vez a casi todas las inversiones, por lo que no puede eliminarse diversificando.\footnote{%
  En cambio, el \textbf{riesgo específico}, propio de cada inversión, se compensa entre los títulos de una cartera bien diversificada, y nadie paga una prima por soportarlo.
} De esta forma, fijamos un concepto que nos acompañará el resto de la exposición: el precio del riesgo es igual para todos los proyectos, pues es el precio del riesgo sistemático, lo que variará de un proyecto a otro será la intensidad, exposición o cantidad de riesgo sistemático a la que se verá sometido.

La cantidad de riesgo sistemático se mide con la \textbf{beta}, $\beta$: cuánto se mueve, en promedio, la rentabilidad de una inversión cuando la del conjunto del mercado se mueve un punto.\footnote{%
    \textbf{NOTA.-} Una beta igual a $1$ es la del mercado en su conjunto, el riesgo sistemático medio. Una beta mayor que $1$ amplifica los movimientos del mercado, como ocurre en los sectores más cíclicos. Una menor que $1$, los amortigua, como en los servicios básicos. Y, por último, una beta nula significa que la inversión no tiene riesgo sistemático, de modo que no lleva prima y su coste del capital es $\gamma$.

    Formalmente, la beta de una inversión $i$ es $\beta_i=\text{Cov}(r_i,r_M)/\text{Var}(r_M)$, la covarianza de su rentabilidad con la del mercado dividida por la varianza de esta: la pendiente de la regresión de la rentabilidad de $i$ sobre la del mercado. El mercado se representa por la \textbf{cartera de mercado}, que contiene todos los títulos con riesgo en proporción a su valor. Su beta es $1$ por definición, porque su covarianza consigo misma es su varianza, y por eso la media de las betas de todos los títulos, ponderada por su valor, es también $1$.
} Que el riesgo sistemático sea común no significa que todas las inversiones lo soporten en la misma medida: común es su \textbf{fuente}, la marcha de la economía, lo que difiere es la \textbf{exposición} de cada inversión a ella, y eso es lo que mide la beta. Por tanto, la beta no tiene unidades, pues no es una rentabilidad: sino una \textbf{cantidad de riesgo} ([\authorfont{Ver Tema 3.B.23}]).\footnote{%
    Se puede ver con la siguiente analogía. La lluvia cae sobre todos, pero no todos se mojan igual: quien va a la intemperie se moja más que quien va bajo techo, y lo que nadie paga es el cubo de agua que cae solo sobre uno, porque en una cartera se compensa con los de los demás. 
    
    Por eso la beta depende de las características de cada proyecto, como lo cíclica que sea su demanda, el peso de sus costes fijos o su endeudamiento, sin medir por ello su riesgo específico: esas características deciden cuánto le afecta la marcha de la economía, no cuánto varía por su cuenta. El concepto y su desarrollo no forman parte de este tema, [\authorfont{Ver Tema 3.B.23}], pero es importante conocerlo en términos conceptuales por sus implicaciones sobre el coste del capital promedio.
}

Para pasar de la cantidad de riesgo a la prima hace falta su \textbf{precio}, la \textbf{prima de riesgo del mercado}, $\lambda_M=\mathbb E[r_M]-\gamma$: lo que el mercado rinde, en promedio, por encima del tipo sin riesgo, que es la remuneración de una unidad de beta. 

En el modelo más usado, el CAPM, la prima de una inversión cualquiera es la cantidad de riesgo por su precio, $\lambda=\beta\,\lambda_M$, y su coste del capital, $r=\gamma+\beta\,\lambda_M$. Así se relacionan las tres magnitudes que recorren el bloque: la beta mide el riesgo, la prima es lo que ese riesgo cuesta y el coste del capital suma a la prima el precio del tiempo. Conocida la beta, el coste del capital se sigue de ella, pero no son lo mismo: una beta de $2$ no es una rentabilidad del $2\%$, sino un riesgo por el que el mercado exige el doble de prima que por el riesgo medio ($\lambda_M$).

Tasas y betas llevan un subíndice que indica a qué inversión se refieren: $r_P$ y $\beta_P$, las de un proyecto; $r_M$ y $\beta_M=1$, las del mercado; y las de las acciones, la deuda y los activos de la empresa, con los subíndices $E$, $D$ y $A$, que definiremos al aparecer.\footnote{%
    \textbf{OJO.-} \textit{Activo} tiene dos sentidos que conviene no confundir. En \textit{activo sin riesgo} o \textit{modelo de valoración de activos} designa cualquier título financiero, como una acción o un bono. En \textit{beta de los activos} o \textit{coste del capital de los activos} designa el negocio de la empresa, sus inversiones reales, cuyo riesgo se reparten accionistas y acreedores. El subíndice $A$ se reserva para este segundo sentido.
} Todas las tasas del bloque son rentabilidades esperadas: $r_E$ y $\mathbb E[r_E]$ designan lo mismo, y la esperanza se escribe solo cuando conviene distinguirla de la rentabilidad que finalmente se obtiene.

El corolario de todo lo expuesto es directo: el coste del capital es una propiedad del proyecto que depende de su beta ($\beta_P$), y que la comparación entre alternativas depende, esencialmente, de su \textbf{riesgo}.\footnote{%
  Pensemos en dos alternativas de una empresa: mejorar el precario sistema de almacenamiento de su planta A, lo que evitaría el deterioro de sus existencias, o construir una nueva planta B, anticipándose a un posible aumento de la demanda. La mejora del almacenamiento apenas tiene riesgo sistemático, porque las pérdidas que evita no dependen de la marcha de la economía, por lo que basta con que su VAN, descontado al tipo sin riesgo, $\gamma$, sea positivo; la nueva planta depende de una demanda incierta, y su VAN debe ser positivo descontado a $\gamma+\beta_P\lambda_M= \gamma+\lambda_P$. Si la empresa usara la misma tasa para ambas, infravaloraría la primera y sobrevaloraría la segunda.
} Es imperativo tener esto siempre presente pues, muy a menudo, es un asunto que genera grandes confusiones en el ámbito financiero, empresarial e incluso académico. Veamos por qué.

\subsubsection{Falacia del WACC: ¿por qué no sirve un valor general?}
Si solo se remunera el riesgo sistemático, y su fuente es común a todas las inversiones, podría pensarse que basta una única tasa para todos los proyectos de la empresa. No es así, porque, como hemos expuesto, lo único común es el \textbf{precio} del riesgo: una misma relación entre la beta y la rentabilidad exigida, con el mismo tipo sin riesgo ($\gamma$) y la misma prima de mercado ($\lambda_M$), vale para todas las inversiones. 

Pero la \textbf{cantidad} de riesgo, la beta ($\beta$), es propia de cada una. De ahí la regla: cada proyecto se descuenta con el coste del capital que corresponde a la intensidad de riesgo.

De lo anterior se desprenden dos consecuencias. El WACC de la empresa es la tasa adecuada si y solo si el proyecto tiene el mismo riesgo que la empresa en su conjunto, como una ampliación de su actividad habitual. Pero si la empresa aplica su WACC a todos los proyectos, se equivoca de forma sistemática. 

Para verlo, representemos la rentabilidad que el mercado exige a cada nivel de riesgo sistemático mediante la \textbf{línea del mercado de valores} (SML, por sus siglas en inglés), $r=\gamma+\beta\,\lambda_M$:

\imagenfit{falacia_wacc.pdf}{La línea del mercado de valores y la falacia del WACC}{Elaboración propia a partir de BREALEY, MYERS y ALLEN (2020).}

La SML corta el eje vertical en el tipo sin riesgo, la rentabilidad de una beta nula, y su pendiente es la prima de riesgo del mercado, el precio de una unidad de beta. El mercado, con beta $1$, rinde $\gamma+\lambda_M$; un proyecto P con beta $\beta_P$ debe rendir $\gamma+\beta_P\,\lambda_M$, y la distancia vertical entre ese punto y $\gamma$ es su prima. Un proyecto crea valor si su rentabilidad esperada queda por encima de la SML, porque rinde más de lo que el mercado exige a su riesgo, y lo destruye si queda por debajo. Sin impuestos, el WACC de la empresa es el punto de la SML que corresponde a la beta de la empresa en su conjunto, la \textbf{beta de sus activos}, $\beta_A$: el riesgo sistemático de todo su negocio, que soportan entre accionistas y acreedores.

Como se puede ver en la figura, un proyecto seguro, como S, que rinde más de lo que el mercado exige a su riesgo pero menos que el WACC de la empresa, se rechaza aunque cree valor. Uno arriesgado, como R, que rinde más que el WACC pero menos de lo que exige su riesgo, se acepta aunque pueda destruirlo. De esta forma, resulta evidente que la empresa que usa una tasa única invierte de más en sus actividades arriesgadas y de menos en las seguras: acaba siendo más arriesgada de lo que sus propietarios eligieron.\footnote{%
  No es un error teórico sin consecuencias. KRÜGER, LANDIER y THESMAR (2015) mostraron que las empresas diversificadas invierten más en las divisiones cuyo riesgo sistemático supera al de su actividad principal y menos en las que lo tienen inferior, como si descontaran todas con la misma tasa; el sesgo es mayor cuando el coste de equivocarse es menor, como en las divisiones pequeñas, lo que sugiere que se trata de una regla práctica más que de una decisión meditada.
} 

Ahora que sabemos la importancia de adecuar el cálculo al objeto de evaluación, dos precisiones completan la regla. La primera, recordar que \textbf{el riesgo relevante es el sistemático}: un proyecto muy incierto puede tener una beta baja si su riesgo es muy específico, como una exploración petrolífera cuyo éxito no depende de la marcha de la economía, y su coste del capital es entonces bajo aunque su dispersión sea grande, porque la cartera diversificada anula el riesgo específico. La segunda, y más importante para el tema que nos ocupa, es que los \textbf{riesgos específicos de un proyecto} (un retraso, un sobrecoste) \textbf{se recogen mejor en los flujos esperados que en la tasa de descuento}: añadir puntos a la tasa para cubrirlos penaliza más a los flujos lejanos, sin que haya razón para ello (BREALEY, MYERS y ALLEN, 2020).

Pero de lo expuesto surge también una pregunta muy incómoda: ¿cómo podemos estimar la beta de un proyecto concreto o, más en general, su coste de capital? Como veremos a continuación, no es una cuestión sencilla.

\subsection{Coste del capital: ¿cuánto vale el capital en función de su fuente?}

El coste del capital de un proyecto no se observa: el proyecto no cotiza, y nadie revela qué rentabilidad exigiría por financiarlo. Por tanto, ¿qué podemos hacer?

Démonos cuenta, primeramente, de lo que sí se observa: los precios de las acciones y de la deuda de las empresas que ya existen. Sobre esta base, la solución que ofrece la teoría es un proceso de varios pasos que, de forma resumida, nos permitirá conocer el coste del capital de un proyecto a partir del coste del capital de una estructura económica completa (la empresa) sometida a un \textbf{riesgo comparable} (o la propia empresa, si el proyecto se parece a su actividad habitual).

Sabemos que el coste del capital es el rendimiento exigido por aquellos agentes que financian su actividad. Y también sabemos que en el caso de la empresa éstos son de dos clases: los \textbf{accionistas}, propietarios de los \textbf{recursos propios} ($E$), que cobran lo que queda después de pagar a los demás y soportan por eso más riesgo; y los \textbf{acreedores}, titulares de la \textbf{deuda} ($D$), que tienen derecho a unos pagos fijos.\footnote{%
    Entendemos el capital de forma similar al factor $K$ de la teoría microeconómica: las fuentes de financiación de la actividad de la empresa. Distinguimos únicamente entre recursos propios (\textit{equity}) y deuda, aunque existen otras fuentes, como la deuda bancaria, los beneficios no distribuidos o las acciones preferentes; por simplicidad, y sin pérdida de generalidad, las agrupamos en esas dos en sentido amplio ([\authorfont{Ver Tema 3.B.3}]).
} Por tanto, el coste del capital de una empresa deberá obtenerse \textbf{por fuentes}, porque cada una soporta un riesgo distinto, se negocia en su propio mercado y se estima con un método propio.\footnote{%
    \textbf{OJO.-} Como advierten VERNIMMEN, QUIRY y LE FUR (2022), el coste del capital, que mide el WACC, no es la rentabilidad del capital, que miden indicadores como la rentabilidad financiera (ROE) o la relación entre el precio y el beneficio por acción (PER). El primero recoge la rentabilidad exigida o esperada por los inversores, \textit{ex ante}; el segundo, la rentabilidad obtenida, \textit{ex post}. En mercados sin fricciones, el arbitraje tiende a igualarlos, pero son conceptos distintos, y usar los segundos como medida del coste del capital, como hacen a menudo los analistas, solo es una aproximación que introduce ruido y sesgo en la estimación.
}

Con este apartado aprenderemos a estimar, con datos de mercado, el coste de los recursos propios, $r_E$, y el de la deuda, $r_D$, de una empresa. Serán la materia prima del siguiente paso: obtener el \textbf{coste del capital promedio} (WACC) de la empresa y trasladarlo al proyecto corrigiendo las diferencias de endeudamiento.

\subsubsection{Recursos propios: ¿cuánto cuestan?}
El \textbf{coste de los recursos propios}, $r_E$, es la rentabilidad que esperan los accionistas por invertir en la empresa. 

Como se puede intuir, es la parte más compleja de la estimación, porque no se observa y hay que inferirla. Por eso, el método de referencia es un modelo de valoración de activos, el CAPM, que deduce la rentabilidad exigida del riesgo sistemático de las acciones.\footnote{%
  Sus limitaciones han llevado a dos alternativas, que veremos al hacer su balance: los modelos con varios factores de riesgo, como la teoría de valoración por arbitraje (APT), y la rentabilidad implícita en el precio de la acción.
}

\paragraph{Modelo de valoración de activos de capital (CAPM)}
El modelo de valoración de activos de capital de SHARPE (1964), LINTNER (1965) y MOSSIN (1966) explica la relación entre riesgo y rentabilidad que existe en el mercado ([\authorfont{Ver Tema 3.B.23}]).

Establece que, si todos los inversores toman sus decisiones según la media y la varianza de la rentabilidad, comparten el mismo horizonte de inversión y las mismas distribuciones de probabilidad y no hay fricciones en los mercados de capitales, en equilibrio: \textbf{existe una relación lineal entre el riesgo sistemático de un título y su rentabilidad esperada}. 

Es exactamente la SML que empleamos al tratar la falacia del WACC, que para cualquier título $i$ se escribe $\mathbb E[r_i]=\gamma+\beta_i\,\lambda_M$. Aplicado a las acciones de la empresa, el CAPM da el coste de los recursos propios:

\eqblock{
  \begin{aligned}
    \mathbb E[r_E]=\gamma+\beta_E\,(\mathbb E[r_M]-\gamma)=\gamma+\beta_E\,\lambda_M\quad\Longrightarrow\quad\boxed{\mathbb E[r_E]=\gamma+\lambda_E}
  \end{aligned}
}{Recursos propios: ¿cuánto cuestan? Coste del capital: Modelo CAPM}

Donde $\beta_E$ es la \textbf{beta de las acciones}, la intensidad de riesgo sistemático que soportan los accionistas de la empresa: la sensibilidad de la rentabilidad de sus acciones a la del mercado. Y, por otro lado, $\mathbb E[r_M]$ es la rentabilidad esperada de la cartera de mercado, que no es observable.\footnote{%
    La derivación del modelo, sus supuestos y sus limitaciones se desarrollan en [\authorfont{Ver Anexo \ref{anx:coste}}] y en [\authorfont{Ver Tema 3.B.23}].
} Es decir, la rentabilidad que se exige a las acciones es el tipo sin riesgo más una prima, que es su cantidad de riesgo sistemático, $\beta_E$, por el precio de cada unidad, $\lambda_M=\mathbb E[r_M]-\gamma$: $\lambda_E=\beta_E\,\lambda_M$. Aplicarlo exige estimar tres magnitudes. 

El \textbf{tipo sin riesgo} debe corresponder al plazo de los flujos que se descuentan: para un proyecto a largo plazo, el rendimiento de la deuda pública a un plazo parecido en la moneda de los flujos.\footnote{%
    En la zona del euro, suele tomarse el rendimiento de la deuda pública alemana, por ser la de menor riesgo de crédito; usar la de un país con prima de riesgo soberano incorpora ya parte de esa prima en $\gamma$.
} 

La \textbf{beta de las acciones} se estima con datos históricos, como la pendiente de la regresión de la rentabilidad de la acción sobre la del mercado, y para las empresas cotizadas puede obtenerse hoy directamente de fuentes de libre acceso; pero la estimación de una empresa aislada es muy imprecisa, por lo que conviene usar la media de un sector, y las betas extremas tienden a acercarse a uno con el tiempo, lo que llevó a BLUME (1971) a proponer ajustarlas hacia la unidad.\footnote{%
    La corrección más extendida, que aplican algunos proveedores de datos, pondera la beta estimada por dos tercios y la unidad por un tercio.
} 

La \textbf{prima de riesgo del mercado} es la magnitud más discutida. Puede estimarse con la media histórica del exceso de rentabilidad de las acciones sobre la deuda pública, que depende del periodo y de si se usa la media aritmética o la geométrica, o como prima implícita en los precios actuales, despejándola de un modelo de descuento de dividendos aplicado al conjunto del mercado, como el de GORDON y SHAPIRO que veremos en el balance.\footnote{%
  Su nivel histórico es, además, difícil de explicar con los modelos habituales de comportamiento del inversor: es el enigma de la prima de las acciones de MEHRA y PRESCOTT (1985) ([\authorfont{Ver Tema 3.B.23}]).
}

\paragraph{Balance: ¿por qué es o no adecuado el CAPM?}
La principal fortaleza del método es que puede aplicarse con independencia de la política de dividendos de la empresa y que determina de forma explícita el nivel de riesgo. Además, se deduce de una teoría del equilibrio del mercado, de modo que la prima no es un ajuste arbitrario, sino el precio del riesgo que los inversores no pueden diversificar. 

Como punto débil, el papel que tienen la calidad y la cantidad de los datos es evidente, y el empleo de datos históricos para situaciones futuras supone una cierta continuidad, o ergodicidad, hipotética.\footnote{%
    Una forma de evitarlo es estimar lo mismo en sentido inverso: en lugar de deducir la rentabilidad exigida del riesgo medido con datos pasados, despejar la que descuenta hoy el precio de la acción, la \textbf{rentabilidad implícita}. Si el precio es el valor actual de los dividendos esperados y estos crecen a una tasa constante $g$, el modelo de GORDON y SHAPIRO (1956) da $r_E=d_1/p+g$, donde $d_1$ es el dividendo del próximo año y $p$ el precio de la acción: el coste de los recursos propios es la rentabilidad por dividendo más su tasa de crecimiento. No necesita betas ni primas y mira al futuro, pero traslada el problema a $g$, que no se observa y de la que el resultado depende mucho, y solo sirve para empresas cotizadas que reparten dividendos con un crecimiento estable.

    \textbf{NOTA.-} Las dos estimaciones no tienen por qué coincidir, porque responden a preguntas distintas: el CAPM, qué rentabilidad debería exigirse a un riesgo; la rentabilidad implícita, qué rentabilidad descuenta el precio actual. Si difieren mucho, o el modelo de valoración está mal especificado, o las previsiones de crecimiento están sesgadas, o el precio se aleja de su valor fundamental.
} 

Además, la evidencia empírica añade una objeción de fondo: la relación entre la beta y la rentabilidad media de las acciones es más plana de lo que predice el modelo (FAMA y FRENCH, 1992), lo que ha llevado a modelos con varios factores de riesgo, como la APT de ROSS (1976) ([\authorfont{Ver Tema 3.B.23}]) y el modelo de tres factores de FAMA y FRENCH (1993).\footnote{%
  En ellos, la prima de una acción es la suma de varias primas, una por cada factor de riesgo sistemático, cada una igual a la exposición de la acción a ese factor, su beta respecto a él, por el precio del factor: $\lambda_E=\sum_k\beta_{E,k}\,\lambda_k$. El CAPM es el caso de un único factor, el mercado.

  Estos modelos explican mejor las rentabilidades medias, pero la teoría no dice qué factores incluir y sus primas son inestables, de modo que lo que ganan en ajuste lo pierden en fundamento. Tampoco resuelven el problema de la precisión: con el CAPM y con su modelo de tres factores, FAMA y FRENCH (1997) obtuvieron errores típicos de más de $3$ puntos al año en el coste de los recursos propios de $48$ sectores estadounidenses, sobre todo porque las primas de los factores se estiman mal ([\authorfont{Ver Anexo \ref{anx:coste}}]). 
} 

Aun así, el CAPM sigue siendo el método más usado: en la encuesta de GRAHAM y HARVEY (2001), lo empleaban siempre o casi siempre tres de cada cuatro directores financieros que estimaban el coste de los recursos propios ($73,5\%$), frente al $39,4\%$ que usaba la rentabilidad media histórica de la acción, el $34,3\%$ que añadía al CAPM otros factores de riesgo y el $15,7\%$ que lo despejaba de un modelo de descuento de dividendos.


\subsubsection{Deuda: ¿cuál es el coste de contraer la obligación?}
El \textbf{coste de la deuda}, $r_D$, es la rentabilidad que esperan los acreedores por prestar a la empresa. Es más sencillo de estimar que el de los recursos propios, porque los acreedores tienen un derecho fijo y su rentabilidad se observa en el mercado.

Es el coste en que la empresa incurre para financiarse con deuda, y lo que importa es el de la deuda \textbf{nueva}, no el tipo que figura en los préstamos antiguos. 

Puede obtenerse por tres vías. En primer lugar, de manera directa, con el \textbf{tipo de interés} que la empresa debe pagar por nuevos préstamos. En segundo lugar, de manera indirecta, si la empresa ha emitido bonos que siguen en circulación, por su \textbf{rendimiento al vencimiento}, la tasa que iguala el precio de mercado del bono con el valor actual de sus cupones y de su principal ([\authorfont{Ver Tema 3.B.24}]). Por último, si no los tiene, también de forma indirecta, por el \textbf{diferencial por calificación}: el tipo sin riesgo más el diferencial que el mercado exige a las empresas de su misma calificación crediticia.\footnote{%
  \textbf{NOTA.-} Podría estimarse también con el CAPM, con la beta de la deuda, pero rara vez se hace, porque esa beta es pequeña y difícil de medir.
}

\paragraph{Balance: ¿cómo se complementan las tres vías?}
Las tres vías no compiten, sino que se contrastan, y de su comparación se siguen dos conclusiones. La primera es que, para una empresa solvente, \textbf{convergen}: el banco, el mercado de bonos y las agencias de calificación valoran el mismo riesgo de impago, y la diferencia entre ellas es pequeña frente al error con que se estima el coste de los recursos propios, que además pesa menos en el WACC, porque se multiplica por $(1-T)$ y por el peso de la deuda. La segunda es que, cuando la deuda es arriesgada o poco líquida, divergen, y las tres sobrestiman el coste, por la razón que vemos a continuación.

Dos precisiones. La primera, que afecta a las tres vías, es que miden la rentabilidad \textbf{prometida}, no la esperada: si la deuda puede impagarse, la rentabilidad que esperan los acreedores es menor, porque descuenta la probabilidad de impago y la pérdida en ese caso. Si $y$ es la rentabilidad prometida, $p$ la probabilidad anual de impago y $L$ la proporción de la deuda que se pierde si se produce, la rentabilidad esperada es, aproximadamente, $r_D\approx y-p\,L$.\footnote{%
    Un bono a un periodo promete pagar $1+y$ por euro prestado y, si hay impago, paga $(1+y)(1-L)$. Su pago esperado es $(1+y)(1-p\,L)$, y su rentabilidad esperada, $r_D=(1+y)(1-p\,L)-1\approx y-p\,L$.
} Para la deuda de buena calidad la diferencia es pequeña, pero para la de alto riesgo es importante, y usar el rendimiento prometido sobrestima el coste de la deuda. Lo que queda por encima del tipo sin riesgo una vez descontado el impago esperado es la prima de riesgo de la deuda, $\lambda_D=\beta_D\,\lambda_M$ en el CAPM, que es pequeña en la deuda de buena calidad: por eso su beta suele tomarse como nula, como veremos al desapalancar la beta.\footnote{%
    El diferencial sobre el tipo sin riesgo depende de la solvencia del prestatario, y en particular de su patrimonio neto, que cae en las recesiones: es la prima de financiación externa que estudia el Tema 3.B.3 y que, en el plano macroeconómico, origina el acelerador financiero ([\authorfont{Ver Tema 3.B.3, Tema 3.A.34, Tema 3.A.37}]).
} 

La segunda es que los intereses son fiscalmente deducibles, de modo que el coste de la deuda para la empresa es el coste después de impuestos, $r_D(1-T)$, donde $T$ es el tipo marginal del impuesto sobre sociedades.\footnote{%
    \textbf{NOTA.-} En España, la deducción de los gastos financieros netos está limitada al $30\%$ del beneficio operativo del ejercicio, con un mínimo deducible de $1$ millón de euros (art. $16$ de la Ley 27/2014, del Impuesto sobre Sociedades). Por encima de ese límite, el ahorro fiscal de la deuda desaparece y el coste efectivo de la deuda adicional es $r_D$, no $r_D(1-T)$.
}

\subsection{Coste del capital: ¿cuánto vale el capital del proyecto?}
Con el coste de cada fuente, el \textbf{coste medio ponderado del capital}, $r_{\text{WACC}}$, es el promedio de esos costes, ponderados por el peso de cada fuente en la financiación (MODIGLIANI y MILLER, 1963):\footnote{%
  Donde $E$ es el valor de mercado de los recursos propios, $D$ el de la deuda y $V=E+D$ el valor total de la empresa, de modo que $E/V$ y $D/V$ son los pesos de cada fuente, y $T$ es el tipo marginal del impuesto sobre sociedades. Sin impuestos, desaparece el factor $(1-T)$.
}

\eqblock{
  \begin{aligned}
    r_{\text{WACC}}=r_E\,\frac{E}{V}+r_D\,(1-T)\,\frac{D}{V}
  \end{aligned}
}{}

Es la rentabilidad que deben generar los activos para pagar a todos sus financiadores lo que exigen: si rinden exactamente el WACC, los acreedores cobran $r_D$, los accionistas obtienen $r_E$ y el Estado soporta, a través de la deducción de los intereses, una parte del coste de la deuda. Por eso el WACC descuenta los flujos del proyecto antes de su financiación, sin restar intereses ni devoluciones de deuda: el coste de financiarlo ya está en la tasa, y contarlo también en los flujos sería contarlo dos veces, como vimos al definir los flujos de caja.

Además, las ponderaciones deben corresponder a la estructura de capital \textbf{objetivo} del proyecto, la que se mantendrá durante su vida, y medirse a \textbf{valor de mercado}: el valor de los recursos propios es el precio de la acción multiplicado por el número de acciones en circulación, y el de la deuda, el valor actual de sus pagos al tipo de mercado, no su importe original, que solo es una aproximación aceptable si los tipos de interés y la solvencia de la empresa apenas han cambiado desde su emisión.

\subsubsection{Problema: ¿cómo obtener el coste de los recursos propios?}
El WACC de un proyecto exige conocer el coste de sus recursos propios, que no se observa, porque el proyecto no cotiza. El apartado anterior da el de una comparable, pero ese coste mezcla dos riesgos: el de su negocio, que comparte con el proyecto, y el que añade su endeudamiento, que el proyecto no tiene por qué compartir. Por eso, como anticipamos, el coste del proyecto se obtiene con un procedimiento que separa el riesgo del negocio del que añade el endeudamiento. La beta que se introduce en el CAPM decide qué coste se obtiene. 

Con la \textbf{beta de las acciones} ($E$), da el coste de los recursos propios, $r_E$, que es el que entra en el WACC. 

Con la \textbf{beta de los activos}, $\beta_A$, que mide el riesgo del negocio sin el efecto del endeudamiento, da el \textbf{coste del capital de los activos}, $r_A$: la rentabilidad que exigirían los accionistas si la empresa no tuviera deuda.

Y con la \textbf{beta del proyecto}, $\beta_P$, da su coste del capital, $r_P$, que es el que importa para decidir.

Las tres betas son, en general, distintas: las acciones de una empresa endeudada son más arriesgadas que su negocio, $\beta_E>\beta_A$, y el proyecto solo tiene la beta de la empresa si se parece a su actividad habitual. Por eso la beta de las acciones de la comparable, la que se estima con sus cotizaciones en el apartado anterior, no puede aplicarse tal cual al proyecto.

Para ver cómo se pasa de una beta a otra, conviene mirar la empresa como un \textbf{balance a valor de mercado}. A un lado están sus activos, la estructura económica del negocio: con un valor ($V$), una intensidad de riesgo sistemático $\beta_A$ y un coste del capital $r_A$. Al otro la estructura financiera del negocio, los derechos sobre los flujos que generan esos activos, que se reparten entre los acreedores, titulares de una deuda de valor $D$, con beta $\beta_D$ y coste $r_D$, y los accionistas, propietarios de unos recursos propios de valor $E$, con beta $\beta_E$ y coste $r_E$. 

De modo que $V=E+D$. Lo que generan los activos es exactamente lo que cobran entre todos los financiadores: quien tuviera toda la deuda y todas las acciones tendría el negocio entero. 

Por eso la \textbf{beta de los activos}, que mide la intensidad de riesgo sistemático del negocio, es la de una cartera formada por las acciones y la deuda de la empresa, ponderadas por su peso en el valor, ya que la beta de una cartera es la media ponderada de las betas de los títulos que la componen:\footnote{%
    Despejando, $\beta_A=(\beta_E+\beta_D\,D/E)/(1+D/E)$. La beta de los activos es la que el Tema 3.B.3 denota $\beta_U$ (de \textit{unlevered}, sin apalancar), la de las acciones de la empresa si no tuviera deuda, y $r_A$, el coste del capital de los activos, es el coste del capital sin deuda, $r_U$.
}
\eqblock{
  \begin{aligned}
    \beta_A=\beta_E\,\frac{E}{V}+\beta_D\,\frac{D}{V}
  \end{aligned}
}{}

Como la SML es lineal, la misma relación se cumple con los costes: $r_A=r_E\,E/V+r_D\,D/V$. La beta y el coste del capital no son lo mismo, pero se corresponden uno a uno: la beta mide la intensidad de riesgo, y el coste, el precio que el mercado exige por él, de modo que cada relación entre betas tiene su gemela entre costes. El coste del capital de los activos es, así, el WACC sin impuestos: lo que el negocio debe rendir para pagar lo que exigen todos sus financiadores.

\paragraph{Deuda: ¿por qué hay que corregir por el apalancamiento financiero?}
Despejando la beta de las acciones obtenemos: $\beta_E=\beta_A+(\beta_A-\beta_D)\,D/E$. Si el negocio no cambia, su beta tampoco, pero cuanta más deuda tiene la empresa, mayor es la beta de sus acciones. La deuda, que cobra antes y apenas comparte el riesgo, deja el riesgo del negocio sobre una parte cada vez más pequeña del valor, la de los accionistas: es lo que se conoce como \textbf{apalancamiento financiero}. 

\imagenfit{balance_apalancamiento.pdf}{El mismo negocio con poca y con mucha deuda: balance a valor de mercado}{Elaboración propia.}

En la figura, la altura de cada caja es su valor y la intensidad del gris, su intensidad de riesgo sistemático por euro: el mismo negocio, con la misma beta de los activos, tiene una beta de las acciones mayor si se financia con mucha deuda que si lo hace con poca. 

Por eso la beta de las acciones de una empresa no sirve, tal cual, para otra con distinto endeudamiento, aunque se dedique a lo mismo; y por eso el coste de los recursos propios sube con la deuda, que es la proposición II de MODIGLIANI y MILLER (1958) ([\authorfont{Ver Tema 3.B.3}]).\footnote{%
  \textbf{NOTA.-} En un caso ideal, si la empresa es solvente y su endeudamiento moderado, la deuda es poco arriesgada y su beta será casi nula, como ya vimos al tratar el coste de la deuda. La corrección por niveles de endeudamiento diferentes sigue siendo necesaria, pero se simplifica: desapalancar es $\beta_A=\beta_E/(1+D/E)$, y reapalancar, la misma relación a la inversa, $\beta_E=\beta_A\,(1+D/E)$.

  Solo sería innecesaria si la empresa no tuviera deuda, porque entonces $\beta_E=\beta_A$.
}

Debemos, en consecuencia, separar el riesgo causado por el apalancamiento financiero de la intensidad de riesgo sistemático que soporta el negocio: debemos corregir por el apalancamiento que sufre la empresa.\footnote{%
  Hay que tener en cuenta, como acertadamente señaló HAMADA (1972), que, si los intereses son deducibles y la empresa mantiene constante el importe de su deuda, el ahorro fiscal de los intereses es un flujo tan seguro como la propia deuda. De esta forma, el efecto del ahorro fiscal compensa en parte el riesgo que el apalancamiento financiero traslada a los accionistas, y la relación pasa a ser la de HAMADA (1972): $\beta_A=\frac{\beta_E}{1+(1-T)\,D/E}$, es decir, en el apalancamiento la deuda pesa solo por su parte después de impuestos, $(1-T)\,D$. 
  
  Es la proposición II de MODIGLIANI y MILLER expresada en betas ([\authorfont{Ver Tema 3.B.3}]). Supone una deuda sin riesgo y de importe constante; si la empresa ajusta su deuda para mantener constante la proporción $D/V$, el ahorro fiscal tiene el mismo riesgo que los activos y la corrección por impuestos desaparece, de modo que vuelve a aplicarse $\beta_A=\beta_E/(1+D/E)$.

  Por simplicidad, omitiremos esta circunstancia, que solo complicaría el desarrollo del cuerpo; la variante del método con deuda de importe constante se recoge en una nota al describir sus pasos.
}

\subsubsection{Método: ¿qué procedimiento seguir?}
El procedimiento se sigue del balance. El proyecto se dedica a lo mismo que la comparable: comparte con ella los activos, el negocio, y con ellos su coste del capital de los activos, $r_A$; lo que no comparte es la financiación. Por eso \textbf{el método quita a los costes de la comparable el efecto de su financiación, se queda con el coste del negocio y le añade el de la financiación del proyecto}.

\imagenfit{wacc_proyecto_balances.pdf}{Del balance de la comparable al del proyecto}{Elaboración propia a partir de BREALEY, MYERS y ALLEN (2020).}

Como vemos, el negocio, con un coste de sus activos $r_A$, es el mismo en la comparable y en el proyecto; lo que cambia es cómo se reparte su riesgo, que, como en la figura anterior, se lee en la intensidad del gris. Por lo que tendremos que desapalancar (1), quitándole el riesgo derivado del apalancamiento financiero a la empresa comparable C; posteriormente, reapalancar (2) la estructura financiera del proyecto P conforme a su estructura financiera objetivo, $(D/E)_P$; y, por último, recomponer (3) con ella su WACC.

\paragraph{Desarrollo analítico: ¿cómo alcanzar la composición financiera del proyecto?}
Consideremos un proyecto P que la empresa quiere financiar con unos pesos objetivo, $(E/V)_P$ y $(D/V)_P$, y una empresa comparable C, que se dedica solo a la actividad del proyecto, con unos pesos $(E/V)_C$ y $(D/V)_C$ a valor de mercado y cuyos costes de los recursos propios, $r_E^C$, y de la deuda, $r_D^C$, se han estimado como en el apartado anterior. El método tiene tres pasos (BREALEY, MYERS y ALLEN, 2020) que son, asimismo, representables sobre el recorrido de la SML.\footnote{%
  Los mismos pasos tienen su versión gemela en betas, que es como suelen aplicarse cuando de la comparable solo se conoce la beta de sus acciones, $\beta_E^C$, estimada con sus cotizaciones. Con deuda casi sin riesgo, se desapalanca con su propia ratio de endeudamiento, $\beta_A=\beta_E^C/[1+(D/E)_C]$; se reapalanca con la del proyecto, $\beta_E^P=\beta_A\,[1+(D/E)_P]$; y el CAPM da el coste de sus recursos propios, $r_E^P=\gamma+\beta_E^P\,\lambda_M$, que entra en el tercer paso. O, más directamente, el CAPM aplicado a la beta de los activos da el coste del negocio, $r_A=\gamma+\beta_A\,\lambda_M$, y de él se sigue el WACC del proyecto como antes. Si hay varias comparables, se promedian sus betas de los activos, y con deuda de importe constante se usa la relación de HAMADA. 

  Las dos versiones llevan al mismo resultado si se aplican con los mismos supuestos, y cuál conviene se discute en el balance. La de betas muestra mejor qué se traslada de la comparable al proyecto: la cantidad de riesgo del negocio, porque su precio, $\lambda_M$, es común a todos.
}

\imagenfit{wacc_proyecto_sml.pdf}{El método sobre la línea del mercado de valores}{Elaboración propia a partir de BREALEY, MYERS y ALLEN (2020).}

Partimos de lo que da el apartado anterior. Sobre la SML están las acciones de la comparable, en $E_C$, con su beta y su coste, $\beta_E^C$ y $r_E^C$, y su deuda, en $D$, con una beta casi nula y un coste algo superior a $\gamma$: las dos están sobre la recta porque el mercado exige a cada título la rentabilidad que corresponde a su exposición al riesgo sistemático. Lo que no se observa es el punto del negocio, $A$, ni el de las acciones del proyecto, $E_P$, y el método consiste en encontrarlos.

\textbf{Primero, se desapalanca} el coste de la comparable: se elimina el efecto del endeudamiento sobre el coste de sus recursos propios para quedarse con el del negocio. Como los activos son una cartera de acciones y deuda, su coste es la media ponderada de los de ambas, es decir, el WACC sin impuestos de la comparable, que no depende de cómo se financie: es el coste del capital que corresponde a la intensidad de riesgo sistemático que soporta el negocio, $r_A=r_E^C\,\Big(\frac{E}{V}\Big)_C+r_D^C\,\Big(\frac{D}{V}\Big)_C$. Por tanto, como el negocio del proyecto es el mismo, ese es también su coste del capital de los activos.\footnote{%
    Si hay varias comparables, se promedian sus costes de los activos, lo que reduce el error de estimación.

    \textbf{NOTA.-} El procedimiento corrige las diferencias de apalancamiento financiero, no las de apalancamiento operativo: si el proyecto tiene más costes fijos que las comparables, sus resultados serán más sensibles a la marcha de la economía, y su beta de los activos, mayor que la de ellas (BREALEY, MYERS y ALLEN, 2020).
}

Sobre la gráfica, desapalancar (1) es buscar $A$ entre $D$ y $E_C$. La beta del negocio es la media ponderada de las de las acciones y la deuda, y su punto está sobre el segmento que las une, tanto más cerca de las acciones cuanto mayor es su peso. Como la SML es una recta, ese punto está también sobre ella, y por eso $A$ da a la vez la beta y el coste del negocio, $\beta_A$ y $r_A$: es la razón geométrica de que betas y costes sean gemelos. Además, $A$ funciona como el apoyo de una palanca en equilibrio, con el valor de la deuda en un extremo y el de los recursos propios en el otro, $E\,(\beta_E-\beta_A)=D\,(\beta_A-\beta_D)$, de modo que el brazo de las acciones es el de la deuda multiplicado por el endeudamiento, $\beta_E-\beta_A=(\beta_A-\beta_D)\,D/E$: cuanta más deuda, más largo. Desapalancar es recorrer ese brazo hacia el apoyo, de $E_C$ a $A$.

\textbf{Segundo, se reapalanca} con la estructura objetivo del proyecto: se añade al coste del negocio el efecto que el endeudamiento del proyecto tiene sobre el coste de sus recursos propios. La misma relación, con los pesos y el coste de la deuda del proyecto, da el coste de sus recursos propios, despejando: $r_E^P=r_A+(r_A-r_D^P)\,\Big(\frac{D}{E}\Big)_P$.\footnote{%
  Es la proposición II de MODIGLIANI y MILLER (1958): los accionistas del proyecto exigen el coste del negocio más una prima por el riesgo financiero, que crece con el endeudamiento y con la diferencia entre el coste del negocio y el de la deuda.
} El coste de la deuda que entra aquí, $r_D^P$, es el de la deuda que el proyecto puede sostener con su estructura objetivo, y no el de la deuda actual de la empresa o de la comparable: importa menos la vía con que se estima que su objeto. Si el proyecto se financia con más deuda, su calificación y su diferencial serán peores, y solo el diferencial por calificación permite estimarlo directamente.

Sobre la gráfica, reapalancar (2) es recorrer el brazo en sentido contrario con la financiación del proyecto. El apoyo no se mueve, porque el negocio es el mismo, y las acciones del proyecto se alejan sobre la recta hasta $E_P$, con un brazo $(\beta_A-\beta_D)\,(D/E)_P$. En la figura, el proyecto se financia con más deuda que la comparable, y por eso $E_P$ queda más arriba que $E_C$: sus accionistas soportan más riesgo por euro y exigen más, $r_E^P>r_E^C$.\footnote{%
  En la figura, la deuda del proyecto tiene el mismo riesgo que la de la comparable. Si fuera más arriesgada, su punto estaría más arriba sobre la recta, el brazo de la deuda sería más corto y, con el mismo endeudamiento, también el de las acciones: los acreedores soportarían una parte mayor del riesgo del negocio.
} Todo el recorrido se hace sobre la misma recta, porque lo único que cambia es la cantidad de riesgo que soportan las acciones, no su precio, $\lambda_M$, que es la pendiente común a todos.

\textbf{Tercero, se recompone} el WACC del proyecto con esos costes, sus pesos objetivo y el ahorro fiscal de los intereses: $r_{\text{WACC}}^P=r_E^P\,\Big(\frac{E}{V}\Big)_P+r_D^P\,(1-T)\,\Big(\frac{D}{V}\Big)_P$. Si sustituimos en él el segundo paso, se ve qué ha hecho realmente el método:\footnote{%
    Por construcción del segundo paso, $r_E^P\,(E/V)_P+r_D^P\,(D/V)_P=r_A$; el tercero resta a esa suma el ahorro fiscal, $T\,r_D^P\,(D/V)_P$.
} $r_{\text{WACC}}^P=r_A-T\,r_D^P\,\Big(\frac{D}{V}\Big)_P$: el WACC del proyecto es el \textbf{coste de su negocio menos el ahorro fiscal de su deuda}. El endeudamiento encarece los recursos propios exactamente lo que ahorra al sustituirlos por deuda más barata, de modo que, sin impuestos, el WACC del proyecto sería $r_A$ cualquiera que fuera su estructura financiera.\footnote{%
    \textbf{NOTA.-} Los dos primeros pasos suponen que el proyecto mantiene constante la proporción de deuda, $D/V$, que es lo que supone el propio WACC al ponderar con pesos fijos: el ahorro fiscal varía entonces con el valor del negocio y tiene su mismo riesgo. Si lo que se mantiene constante es el importe de la deuda, el ahorro fiscal es tan seguro como ella y la deuda pesa solo por su parte después de impuestos, como en la relación de HAMADA: el primer paso pasa a ser $r_A=[r_E^C\,E_C+r_D^C\,(1-T)\,D_C]/[E_C+(1-T)\,D_C]$; el segundo, $r_E^P=r_A+(r_A-r_D^P)(1-T)(D/E)_P$; y el resultado, $r_{\text{WACC}}^P=r_A\,[1-T\,(D/V)_P]$ (MODIGLIANI y MILLER, 1963).
}

Sobre la gráfica, recomponer (3) lleva el punto final fuera de la recta. Sin impuestos, la media ponderada de $E_P$ y $D$ con los pesos del proyecto volvería exactamente a $A$, por la misma razón que en el primer paso. Con impuestos, la deuda le cuesta a la empresa $r_D^P\,(1-T)$, y el WACC cae en vertical por debajo de $A$ en lo que vale el ahorro fiscal, $T\,r_D^P\,(D/V)_P$. Queda bajo la SML porque ya no es la rentabilidad que el mercado exige a un riesgo, sino lo que a la empresa le cuesta pagarla cuando el Estado soporta una parte de los intereses.

Con esto ya tenemos el WACC del proyecto, que es el punto de llegada: la tasa con la que se descuentan sus flujos antes de su financiación para obtener su VAN.\footnote{%
  Si el endeudamiento del proyecto va a cambiar a lo largo de su vida, unos pesos fijos no lo representan, y es preferible descontar los flujos a $r_A$ y sumar aparte el valor de la financiación, como veremos en el balance.
} Y se ve por qué puede trasladarse el coste de una empresa a un proyecto: lo que se traslada es el coste del negocio, que no depende de la financiación, y pasa de la comparable al proyecto por $A$, el único punto que comparten; la financiación de cada uno se añade después y solo decide dónde quedan sus acciones sobre la recta y cuánto baja la tasa final por el ahorro fiscal. Aplicar al proyecto el coste de las acciones de la comparable, $r_E^C$, sería tomar un punto que no corresponde ni a su negocio ni a sus accionistas.

\subsubsection{Balance: ¿qué supone el método y cuándo deja de valer?}
El método descansa en un supuesto: que la financiación no altera el coste del negocio. Es lo que permite desapalancar, porque solo si $r_A$ es independiente de la estructura de capital el coste del negocio de la comparable es también el del proyecto, y quitarle una deuda y ponerle otra lo deja intacto. Es la tesis de MODIGLIANI y MILLER (1958): con un mercado de capitales perfecto, el valor de una empresa es el de los flujos que generan sus activos, cualquiera que sea la forma en que se reparten entre accionistas y acreedores, y la rentabilidad que exigen los accionistas sube con el endeudamiento lo justo para que el coste medio no cambie, que es la relación del segundo paso. Los impuestos solo añaden el ahorro fiscal, que recoge el tercero.

Si la estructura de capital influye en el valor, los dos costes dejan de ser datos y pasan a ser variables endógenas: con impuestos, el WACC baja con el endeudamiento (MODIGLIANI y MILLER, 1963) hasta que los costes de insolvencia lo compensan, y los de agencia o la información asimétrica alteran también lo que exigen accionistas y acreedores.\footnote{%
  Es una cuestión que se estudia en otro tema y uno de los puntos de mayor divergencia entre la teoría y el análisis empírico ([\authorfont{Ver Tema 3.B.3}]).
} El cálculo se convierte entonces en un círculo: el coste del proyecto depende del de la empresa, este de su endeudamiento y de su valor, y el valor de la empresa, del de sus proyectos, que depende a su vez del coste con que se descuentan. Y el círculo no se cierra en un solo proyecto, porque una empresa reúne muchos proyectos y un proyecto se evalúa con los datos de muchas empresas comparables.

\imagenfit{circulo_endogeneidad.pdf}{El círculo de la endogeneidad}{Elaboración propia.}

El panel (a) recorre el círculo: el coste del proyecto determina su VAN; el VAN de los proyectos, el valor de la empresa; el valor, para una deuda dada, su endeudamiento, $D/V$; y el endeudamiento, el coste de cada fuente, del que sale de nuevo el coste del proyecto. MODIGLIANI y MILLER (1958) lo cortan en el último eslabón: el endeudamiento cambia el coste de cada fuente, pero no su media, de modo que el coste del proyecto deja de depender de él.

El panel (b) lo traslada a la SML. Los puntos llenos son los del método: el apoyo de la palanca, $A$, queda fijo, y el endeudamiento solo mueve las acciones sobre la recta. Con más deuda, sin embargo, la deuda deja de ser casi segura y su punto sube por la recta, de $D$ a $D'$. Además, los costes de insolvencia se producen sobre todo en las recesiones, cuando más pesan, de modo que tienen riesgo sistemático (ALMEIDA y PHILIPPON, 2007) y pueden elevar el del conjunto\textcolor{orange}{\textsuperscript{[Probable]}}: el apoyo se desplaza de $A$ a $A'$, las acciones del proyecto se alejan hasta $E_P'$ y el WACC cambia; en la figura, al alza, como ocurre más allá del endeudamiento que lo minimiza. Pero el nuevo WACC cambia el valor del proyecto y el de la empresa, y con ellos $D/V$, lo que vuelve a mover todos los puntos (la línea discontinua): el método ya no tiene un punto fijo del que partir.

Por eso se parte de MODIGLIANI y MILLER: no porque la financiación sea irrelevante, sino porque, al suponerlo, el círculo se rompe y el coste del negocio puede estimarse por separado. Ese es el \textbf{valor metodológico del teorema}: fija la referencia y permite ver qué ocurre cuando alguno de sus supuestos se incumple. La visión tradicional, según la cual endeudarse reduce primero el WACC y lo eleva después, de modo que existe una estructura que lo minimiza (DURAND, 1952), y la respuesta de MODIGLIANI y MILLER se desarrollan en ese tema ([\authorfont{Ver Tema 3.B.3}]).

Del mismo supuesto se sigue el límite práctico del método. El efecto de la financiación entra en el WACC a través del coste de la deuda después de impuestos y de unos pesos fijos, lo que solo es correcto si la estructura de capital se mantiene constante a lo largo del proyecto. Cuando no es así (una compra apalancada que reduce su deuda con los años, un proyecto con un préstamo subvencionado o con una financiación que se amortiza según un calendario), es preferible el \textbf{valor actual ajustado} (APV) de MYERS (1974):
$$APV=VAN_{\text{base}}+VA(\text{efectos de la financiación}),$$
donde el VAN base es el del proyecto financiado solo con recursos propios, descontado a $r_A$, y el segundo término suma el valor actual de los efectos de la financiación, como el ahorro fiscal de los intereses, las subvenciones o los costes de emisión. El APV separa lo que el WACC mezcla en la tasa: el valor que crea la inversión y el que crea, o destruye, su financiación, como hacía la separación de FISHER con la que empezamos.

Queda, por último, una elección práctica: cómo llegar al coste del negocio de la comparable, $r_A$. El método lo calcula \textbf{por fuentes}: estima por separado el coste de los recursos propios y el de la deuda, y los pondera por su valor de mercado. La alternativa es \textbf{estimar directamente el riesgo de los activos}, su beta, y deducir de ella el coste con el CAPM, $r_A=\gamma+\beta_A\,\lambda_M$, como en la versión del método en betas.\footnote{%
  Como el negocio no cotiza, su beta no puede estimarse con sus propias cotizaciones: se obtiene desapalancando la beta de las acciones de la comparable o, sin pasar por ellas, con \textbf{betas contables}, la pendiente de la regresión de la rentabilidad contable de la empresa sobre la del mercado. BEAVER, KETTLER y SCHOLES (1970) encontraron una asociación significativa entre la beta de mercado y medidas contables del riesgo como la variabilidad de los beneficios, el porcentaje de beneficios que se reparte como dividendo o la propia beta contable\textcolor{orange}{\textsuperscript{[Probable]}}.
}

Por fuentes hacen falta \textbf{más datos}, el coste y el valor de mercado de cada una, pero \textbf{menos hipótesis}: el coste de la deuda se aproxima con datos de mercado por cualquiera de las tres vías, la ponderación es una identidad que no depende de ningún modelo, porque quien tuviera todas las acciones y toda la deuda tendría el negocio entero, y el coste de los recursos propios admite cualquiera de los métodos del apartado anterior. Por la vía directa basta una beta, pero todo el cálculo descansa en el CAPM, que debe valer a la vez para las acciones, para la deuda y para el negocio, y en un supuesto sobre la beta de la deuda. Por eso el cálculo por fuentes reduce el riesgo de lo que KAHNEMAN (2011) llama \textbf{ceguera inducida por la teoría}: una vez aceptada una teoría y usada como herramienta para pensar, cuesta mucho ver sus fallos, y cada dato de mercado que entra en el cálculo es un contraste que la vía directa no tiene. La teoría, sin embargo, no desaparece: el coste de los recursos propios sigue necesitando uno de los modelos del apartado anterior, y con él sus limitaciones.

\subsection{Evidencia: ¿toman en consideración esto las empresas?}
La práctica se aleja de estas recomendaciones en dos direcciones. La primera ya la hemos visto: muchas empresas aplican una tasa única a todos sus proyectos; en la encuesta de GRAHAM y HARVEY (2001), más de la mitad de los directores financieros declaraba usar siempre o casi siempre el coste del capital de la empresa en su conjunto, más que el de cada proyecto. 

La segunda es que las tasas mínimas de rentabilidad que exigen a los proyectos superan con mucho su WACC: JAGANNATHAN, MATSA, MEIER y TARHAN (2016) encontraron que la diferencia es mayor en las empresas con más restricciones de capital y de capacidad de gestión, y no en las que afrontan más riesgo. La tasa mínima actúa, así, como un mecanismo de racionamiento: recoge el coste de oportunidad de unos fondos y de una capacidad de gestión escasos, como el índice de rentabilidad del primer bloque, y quizá el valor de la opción de esperar (McDONALD, 2000).

Otros dos problemas son frecuentes. Uno es la \textbf{prima de riesgo país} de los proyectos en economías emergentes: suele sumarse a la tasa el diferencial de la deuda soberana del país, pero, si el riesgo del país es diversificable para un inversor internacional, debería recogerse en los flujos esperados (la probabilidad de una expropiación, de una devaluación o de una crisis) y no en la tasa. El otro es la \textbf{coherencia}: los flujos nominales se descuentan con tasas nominales y los reales con tasas reales, como vimos al empezar, y los flujos en una moneda, con la tasa de esa moneda; mezclar unos y otros, por ejemplo flujos reales con un WACC nominal, sesga el VAN tanto más cuanto mayor es la inflación.

\anexo{Los métodos de valoración de activos aplicados al coste del capital}\label{anx:coste}
Este anexo desarrolla los métodos con que se estima el coste de los recursos propios, que el segundo bloque resume al tratar el CAPM y su balance. Va fuera del cuerpo porque cada método exige una derivación propia y porque, para valorar proyectos, basta con saber qué mide cada uno, en qué se diferencia de los demás y dónde falla; aquí se ve por qué. Se exponen de forma completa, sin dar por sabido lo que el cuerpo ya explica.

Son cuatro. El modelo de valoración de activos de capital (CAPM) deduce la rentabilidad exigida del equilibrio del mercado; la teoría de valoración por arbitraje (APT), de la ausencia de oportunidades de arbitraje; los modelos factoriales empíricos, de los factores que mejor explican las rentabilidades observadas; y el modelo de GORDON y SHAPIRO, del precio de la acción. Los tres primeros van del riesgo a la rentabilidad: dicen qué rentabilidad debería exigirse a un título dado su riesgo. El último recorre el camino inverso: dice qué rentabilidad descuenta el precio actual. Tras presentar los conceptos que comparten, cada método se expone en el mismo orden: objetivo, supuestos, desarrollo, resultados, implicaciones para el coste del capital y limitaciones. La derivación del CAPM y de la APT desde la teoría de carteras se estudia también en otro tema ([\authorfont{Ver Tema 3.B.23}]).

\subsection{Conceptos comunes}
Los cuatro métodos responden a la misma pregunta, qué rentabilidad esperan los accionistas, con los mismos materiales: la rentabilidad de un título, su riesgo, el efecto de la diversificación sobre ese riesgo y la relación entre la rentabilidad exigida y el precio.

\subsubsection{Rentabilidad y riesgo de un título}
La \textbf{rentabilidad} de una acción $i$ entre hoy y el final del periodo es lo que su tenedor obtiene por cada euro invertido: el dividendo cobrado más la variación del precio,
$$r_i=\frac{D_{i,1}+P_{i,1}-P_{i,0}}{P_{i,0}},$$
donde $P_{i,0}$ es el precio de hoy y $D_{i,1}$ y $P_{i,1}$ son el dividendo y el precio al final del periodo. Como estos son inciertos, $r_i$ es una variable aleatoria. Su media, la \textbf{rentabilidad esperada}, $\mathbb E[r_i]$, resume lo que se espera ganar; su varianza, $\sigma_i^2$, o su desviación típica, $\sigma_i$, la \textbf{volatilidad}, mide la dispersión en torno a esa media y es la medida más simple del riesgo. La relación entre dos títulos se mide con la covarianza de sus rentabilidades, $\sigma_{ij}$, o con su coeficiente de correlación, $\rho_{ij}=\sigma_{ij}/(\sigma_i\sigma_j)$. Para la empresa, la rentabilidad esperada de sus acciones es el coste de los recursos propios, $r_E$, y la de su deuda, el coste de la deuda, $r_D$. Ninguna rentabilidad esperada se observa: se observan las realizadas, y la esperada solo puede inferirse con un modelo o con la media de un periodo pasado.\footnote{%
    La media histórica es un estimador muy impreciso: su error típico es $\sigma_i/\sqrt{T}$, donde $T$ es el número de años, de modo que, con las volatilidades habituales de las acciones, incluso varias décadas de datos dejan un margen de error de varios puntos porcentuales al año. Es la razón última de la imprecisión de todos los métodos que siguen.
}

\subsubsection{Diversificación: riesgo sistemático y riesgo específico}
Una \textbf{cartera} reparte la riqueza entre $N$ títulos con pesos $w_i$ que suman uno. Su rentabilidad esperada es la media ponderada de las de sus títulos, $\mathbb E[r_p]=\sum_i w_i\,\mathbb E[r_i]$, pero su varianza no es la media de las varianzas, porque depende también de las covarianzas:
$$\sigma_p^2=\sum_{i=1}^{N}\sum_{j=1}^{N}w_iw_j\,\sigma_{ij}.$$
Con pesos iguales, $w_i=1/N$, la expresión se reduce a $\sigma_p^2=\frac{1}{N}\,\bar\sigma^2+\left(1-\frac{1}{N}\right)\bar\sigma_{ij}$, donde $\bar\sigma^2$ es la varianza media de los títulos y $\bar\sigma_{ij}$, su covarianza media. Al aumentar el número de títulos, el primer término desaparece y el riesgo de la cartera tiende a la covarianza media. La parte del riesgo de cada título que no se mueve con los demás, el \textbf{riesgo específico}, se elimina diversificando; la que se mueve con todos, el \textbf{riesgo sistemático}, no. De ahí la idea que comparten los tres primeros métodos: si los inversores pueden diversificar sin coste, nadie les paga por soportar riesgo específico, y la rentabilidad exigida a un título no depende de su varianza, sino de lo que aporta al riesgo de una cartera diversificada, que es su covarianza con ella. La forma de medir esa aportación distingue a unos métodos de otros: con una sola cartera de referencia, la de mercado, es la \textbf{beta}, $\beta_i=\sigma_{iM}/\sigma_M^2$; con varios factores de riesgo, un conjunto de betas, una por factor.

\subsubsection{Rentabilidad exigida, precio y coste del capital}
La rentabilidad exigida y el precio son dos caras de lo mismo. Si los inversores exigen a la acción una rentabilidad $r_E$, solo la comprarán a un precio que, con los pagos que esperan, les dé esa rentabilidad:
$$P_{i,0}=\frac{\mathbb E[D_{i,1}+P_{i,1}]}{1+r_E}.$$
Cuanto mayor es el riesgo, mayor es la rentabilidad exigida y menor el precio que se paga por los mismos pagos esperados. Para la empresa, esa rentabilidad es un coste: es lo que tiene que ofrecer, en promedio, para que los inversores le confíen sus fondos, y a lo que renuncia si los emplea en un proyecto que rinde menos. Todos los métodos la descomponen como en el cuerpo, $r=\gamma+\lambda$: el \textbf{tipo sin riesgo}, $\gamma$, la rentabilidad de un título sin riesgo de impago al mismo plazo, que remunera el tiempo; y la \textbf{prima de riesgo}, $\lambda$, que remunera el riesgo sistemático. Difieren en cómo miden $\lambda$.

\subsubsection{Tres formas de llegar a la rentabilidad exigida}
Hay tres razonamientos para determinar la prima, y cada método se apoya en uno. El \textbf{equilibrio} pregunta qué rentabilidades esperadas hacen que los inversores, eligiendo sus carteras de forma óptima, demanden exactamente los títulos que existen: es el del CAPM. La \textbf{ausencia de arbitraje} pregunta qué relación deben guardar las rentabilidades para que nadie pueda ganar sin invertir ni arriesgar, de modo que dos carteras con el mismo riesgo rindan lo mismo: es el de la APT. La \textbf{valoración por descuento} no deduce la prima de un modelo de comportamiento del inversor, sino que la lee en el precio, que ya la incorpora: es el de GORDON y SHAPIRO. Los modelos factoriales empíricos comparten la forma de la APT, pero eligen los factores por los datos y no por la teoría. Los dos primeros razonamientos dicen qué rentabilidad \textit{debería} exigirse; el tercero, cuál \textit{se está} exigiendo. Si el mercado valora bien los títulos, coinciden; si no, la diferencia es información.

\subsection{El modelo de valoración de activos de capital (CAPM)}
El CAPM, desarrollado de forma independiente por SHARPE (1964), LINTNER (1965) y MOSSIN (1966) a partir de la teoría de carteras de MARKOWITZ (1952), es un modelo de equilibrio del mercado de títulos con riesgo. Su resultado, la línea del mercado de valores, es el que el cuerpo emplea para medir el coste de los recursos propios y el de un proyecto.

\subsubsection{Objetivo}
El CAPM busca la relación entre el riesgo y la rentabilidad esperada de cada título que resulta del equilibrio, cuando todos los inversores eligen sus carteras de forma óptima y la demanda de cada título iguala su oferta. Aplicado al coste del capital, su objetivo es doble: dar una medida del riesgo que el mercado remunera, la beta, y un precio para ese riesgo, la prima de mercado, de modo que la rentabilidad exigida a las acciones de una empresa, a sus activos o a un proyecto se obtenga conociendo solo su beta.

\subsubsection{Supuestos}
El modelo descansa en seis supuestos, y cada uno cumple una función en la derivación. (i) Los inversores son aversos al riesgo y eligen su cartera atendiendo solo a la media y a la varianza de su rentabilidad, lo que exige que su utilidad sea cuadrática o que las rentabilidades sigan una distribución normal; es lo que permite resumir el riesgo en la varianza. (ii) Hay un único periodo, común a todos. (iii) Las expectativas son homogéneas: todos atribuyen las mismas medias, varianzas y covarianzas a los títulos, y por eso todos ven la misma frontera eficiente. (iv) Existe un título sin riesgo con el que se puede prestar y pedir prestado sin límite al tipo $\gamma$. (v) Los mercados son perfectos: no hay impuestos ni costes de transacción, los títulos son divisibles, se permiten las ventas en corto y nadie influye en los precios. Y, (vi) la oferta de títulos está dada.

Si falla alguno, el modelo cambia, aunque no siempre pierde su forma. Sin préstamos al tipo sin riesgo, BLACK (1972) mostró que la relación sigue siendo lineal en la beta, pero con una cartera de beta nula, $Z$, en lugar del título sin riesgo, $\mathbb E[r_i]=\mathbb E[r_Z]+\beta_i\,(\mathbb E[r_M]-\mathbb E[r_Z])$, con $\mathbb E[r_Z]>\gamma$, de modo que la recta es más plana. Con varios periodos y oportunidades de inversión cambiantes, MERTON (1973) mostró que los inversores demandan además títulos que los cubran frente a esos cambios, y aparecen otras betas con su propia prima: es el CAPM intertemporal, que da fundamento teórico a los modelos de varios factores. Con expectativas heterogéneas, impuestos o costes de transacción, no todos tienen la misma cartera con riesgo, y la cartera de mercado deja de ser necesariamente eficiente.

\subsubsection{Desarrollo}
La derivación tiene cuatro pasos.

El primero es la \textbf{frontera eficiente} de MARKOWITZ (1952). Con solo títulos con riesgo, cada inversor busca, para cada rentabilidad esperada $\mu$, la cartera de menor varianza:
$$\min_{w}\ \sigma_p^2=\sum_i\sum_j w_iw_j\,\sigma_{ij}\quad\text{s. a.}\quad\sum_i w_i\,\mathbb E[r_i]=\mu,\qquad\sum_i w_i=1.$$
Al variar $\mu$, las soluciones describen en el plano $(\sigma_p,\mathbb E[r_p])$ una hipérbola; su tramo superior, el de las carteras que nadie puede mejorar en media sin empeorar en varianza, es la frontera eficiente. Como la diversificación reduce el riesgo, la frontera queda a la izquierda de los títulos individuales.

El segundo es la introducción del \textbf{título sin riesgo}. Una cartera que invierte una proporción $a$ en una cartera con riesgo $T$ y el resto al tipo sin riesgo tiene $\mathbb E[r_p]=\gamma+a\,(\mathbb E[r_T]-\gamma)$ y $\sigma_p=a\,\sigma_T$: al variar $a$, recorre una recta que parte de $\gamma$ con pendiente $(\mathbb E[r_T]-\gamma)/\sigma_T$, la rentabilidad adicional por unidad de volatilidad, o ratio de SHARPE. La mejor cartera con riesgo es la que hace máxima esa pendiente, la que da la recta tangente a la frontera eficiente. De ahí el \textbf{teorema de separación} de TOBIN (1958): todos los inversores combinan la misma cartera con riesgo, la tangente, con el título sin riesgo, y sus preferencias solo deciden en qué proporción, prestando si son muy aversos al riesgo y endeudándose si lo son poco. La decisión de qué títulos tener se separa de la de cuánto riesgo asumir.

El tercero es el \textbf{equilibrio}. Si todos tienen las mismas expectativas, todos eligen la misma cartera tangente; como en conjunto deben tener todos los títulos que existen, en las proporciones en que existen, la cartera tangente es la \textbf{cartera de mercado}, $M$, que contiene cada título en proporción a su valor. La recta que une $\gamma$ con $M$, la \textbf{línea del mercado de capitales} (CML), describe las carteras eficientes:
$$\mathbb E[r_p]=\gamma+\frac{\mathbb E[r_M]-\gamma}{\sigma_M}\,\sigma_p.$$
Su pendiente es el precio de mercado de una unidad de volatilidad de una cartera eficiente.

El cuarto paso pasa de las carteras eficientes a cualquier título y da la \textbf{línea del mercado de valores} (SML). Consideremos una cartera que invierte una proporción $a$ en el título $i$ y $1-a$ en la de mercado. Su rentabilidad esperada y su volatilidad son
$$\mathbb E[r_a]=a\,\mathbb E[r_i]+(1-a)\,\mathbb E[r_M],\qquad \sigma_a=\left[a^2\sigma_i^2+(1-a)^2\sigma_M^2+2a(1-a)\,\sigma_{iM}\right]^{1/2}.$$
Con $a=0$, la cartera es la de mercado. Como el título ya forma parte de $M$, la curva que describen estas carteras al variar $a$ no puede cruzar la CML en $M$, porque entonces habría combinaciones mejores que la cartera de mercado: debe ser tangente a ella. Igualando sus pendientes en $a=0$,
$$\left.\frac{d\,\mathbb E[r_a]/da}{d\sigma_a/da}\right|_{a=0}=\frac{\mathbb E[r_i]-\mathbb E[r_M]}{(\sigma_{iM}-\sigma_M^2)/\sigma_M}=\frac{\mathbb E[r_M]-\gamma}{\sigma_M},$$
y despejando,
$$\mathbb E[r_i]=\gamma+\frac{\sigma_{iM}}{\sigma_M^2}\left(\mathbb E[r_M]-\gamma\right)=\gamma+\beta_i\,\lambda_M.$$

\subsubsection{Resultados}
El resultado central es la SML: la rentabilidad esperada de cualquier título, o de cualquier cartera, es el tipo sin riesgo más su beta por la prima de riesgo del mercado, $\lambda_M=\mathbb E[r_M]-\gamma$. De ella se siguen cuatro propiedades.

La primera es que \textbf{solo se remunera el riesgo sistemático}: la varianza de un título no aparece, solo su covarianza con el mercado, de modo que dos títulos con la misma beta exigen la misma rentabilidad aunque uno sea mucho más volátil. Los títulos individuales quedan por debajo de la CML, porque tienen un riesgo específico que no se remunera, pero todos están sobre la SML. La segunda es que la relación es \textbf{lineal}: un título con el doble de beta tiene el doble de prima. La tercera es que la beta es \textbf{aditiva}: la de una cartera es la media ponderada de las de sus componentes, $\beta_p=\sum_i w_i\beta_i$, y, por la linealidad, también lo es su rentabilidad esperada. La cuarta es que la SML puede escribirse como una regla de valoración. Si un título promete un pago incierto $X_i$ al final del periodo, su precio es
$$P_{i,0}=\frac{\mathbb E[X_i]-\lambda_M\,\text{Cov}(X_i,r_M)/\sigma_M^2}{1+\gamma}:$$
el pago esperado menos un descuento por su riesgo sistemático, el \textbf{equivalente cierto}, descontado al tipo sin riesgo. Descontar el pago esperado a la tasa ajustada por el riesgo, $\mathbb E[X_i]/(1+\mathbb E[r_i])$, y descontar el equivalente cierto al tipo sin riesgo dan el mismo precio.

\subsubsection{Implicaciones para el coste del capital}
Las propiedades anteriores sostienen casi todo lo que el cuerpo dice sobre la tasa de descuento.

La SML da el coste del capital de cualquier inversión a partir de su beta. Con la beta de las acciones, $\beta_E$, da el coste de los recursos propios, $r_E=\gamma+\beta_E\lambda_M$; con la de la deuda, $\beta_D$, el coste de la deuda, $r_D=\gamma+\beta_D\lambda_M$; con la de los activos de la empresa, $\beta_A$, el coste del capital de sus activos, $r_A$; y con la de un proyecto, $\beta_P$, el suyo, $r_P$. Como el coste depende solo de la beta de lo que se valora, \textbf{el coste del capital es una propiedad del proyecto}, no de la empresa que lo acomete ni de cómo lo financia: es el fundamento de la falacia del WACC.

La aditividad da la relación entre esas betas. Para los inversores, los activos de la empresa son la cartera formada por sus acciones y su deuda, con pesos $E/V$ y $D/V$, de modo que $\beta_A=\beta_E\,E/V+\beta_D\,D/V$ y, por la linealidad de la SML, $r_A=r_E\,E/V+r_D\,D/V$: el coste del capital de los activos es el WACC sin impuestos. Despejando, $\beta_E=\beta_A+(\beta_A-\beta_D)\,D/E$: la beta de las acciones crece con el endeudamiento aunque el negocio no cambie, que es la proposición II de MODIGLIANI y MILLER (1958) expresada en betas ([\authorfont{Ver Tema 3.B.3}]). Es la base del procedimiento de desapalancar y reapalancar la beta de empresas comparables. La aditividad implica también que el valor de una empresa es la suma del de sus proyectos: diversificar dentro de la empresa no crea valor, porque los accionistas pueden diversificar por su cuenta.

La forma del equivalente cierto conecta la tasa ajustada con la valoración de proyectos de varios periodos. Aplicar a todos los flujos de un proyecto una misma tasa, $r_P$, equivale a suponer que el riesgo crece a un ritmo constante con el plazo, porque el factor $1/(1+r_P)^t$ descuenta a la vez el tiempo y un riesgo que se acumula. FAMA (1977) mostró que una única tasa ajustada solo es correcta si la beta y la prima de mercado son constantes y la incertidumbre sobre los flujos se resuelve a un ritmo constante. MYERS y TURNBULL (1977) mostraron, además, que, incluso con el CAPM, la beta de una empresa depende de características de sus flujos como su crecimiento y su duración, de modo que no puede trasladarse sin más a un proyecto con un perfil temporal distinto, aunque su negocio sea el mismo. Cuando esas condiciones no se cumplen, como en un proyecto cuyo riesgo se concentra en una fase inicial, es preferible descontar cada flujo con su propia tasa o trabajar con equivalentes ciertos.

\subsubsection{Limitaciones}
Sus limitaciones son de tres tipos.

Las \textbf{teóricas} proceden de los supuestos. Un solo periodo, preferencias de media y varianza, expectativas homogéneas y préstamos ilimitados al tipo sin riesgo son simplificaciones fuertes, y, como hemos visto, relajarlas cambia el modelo (BLACK, 1972; MERTON, 1973). ROLL (1977) añadió una objeción de fondo: la cartera de mercado contiene todos los activos con riesgo, incluidos los inmuebles o el capital humano, y no es observable; los índices bursátiles son solo aproximaciones. Como cualquier cartera eficiente en media y varianza genera una relación lineal exacta entre la beta calculada con ella y la rentabilidad esperada, contrastar el CAPM equivale a contrastar si el índice elegido es eficiente, y no el modelo mismo.

Las \textbf{empíricas} proceden de los contrastes. Los primeros, de BLACK, JENSEN y SCHOLES (1972) y de FAMA y MacBETH (1973), encontraron una relación positiva y lineal entre la beta y la rentabilidad media, en la que el riesgo específico no se remuneraba, pero más plana que la teórica: la ordenada en el origen superaba el tipo sin riesgo y la pendiente era menor que la prima de mercado, como en el modelo de beta nula. Después se acumularon las anomalías, rentabilidades medias que la beta no explica, como la de las empresas pequeñas (BANZ, 1981) y la de las empresas con un valor contable alto en relación con el de mercado. FAMA y FRENCH (1992) mostraron que, una vez tenidos en cuenta el tamaño y esa ratio, la beta apenas explicaba las diferencias de rentabilidad media entre las acciones estadounidenses desde los años sesenta. Para el coste del capital, una SML más plana significa que el CAPM sobrestima la rentabilidad exigida a los títulos de beta baja y la subestima en los de beta alta, tanto más cuanto más extrema es la beta.

Las \textbf{de estimación} proceden de las tres magnitudes que requiere. La beta de una acción se estima con error, depende del índice, de la frecuencia de los datos y del periodo elegido, y cambia con el tiempo. La prima de mercado histórica depende del periodo y del tipo de media, y su nivel es difícil de explicar con preferencias razonables (MEHRA y PRESCOTT, 1985). Y el tipo sin riesgo debe ajustarse al plazo de los flujos, lo que no es coherente con un modelo de un solo periodo. FAMA y FRENCH (1997) estimaron que el error típico del coste de los recursos propios de un sector supera los $3$ puntos porcentuales al año, sobre todo por la incertidumbre sobre la prima de mercado. Aun así, el CAPM sigue siendo el método más usado, porque da una respuesta explícita y con fundamento teórico a la pregunta de qué riesgo debe remunerarse.

\subsection{La teoría de valoración por arbitraje (APT)}
La APT, propuesta por ROSS (1976), llega a una relación lineal entre riesgo y rentabilidad sin un modelo de equilibrio y sin la cartera de mercado. Parte de cómo se generan las rentabilidades y de una condición mínima, que no haya oportunidades de arbitraje, y admite varias fuentes de riesgo sistemático.

\subsubsection{Objetivo}
Busca explicar las rentabilidades esperadas con supuestos más débiles que los del CAPM y con varias fuentes de riesgo sistemático, no solo una. Aplicada al coste del capital, da la rentabilidad exigida a las acciones como el tipo sin riesgo más la suma de las primas de cada factor al que está expuesta la empresa, de modo que permite ver qué riesgos remunera la tasa: el ciclo económico, la inflación o los tipos de interés.

\subsubsection{Supuestos}
Son cuatro. (i) Las rentabilidades se generan por un \textbf{modelo factorial lineal},
$$r_i=\mathbb E[r_i]+\sum_{k=1}^{K}\beta_{ik}\,F_k+\varepsilon_i,$$
donde $F_k$ es la sorpresa del factor $k$, su desviación respecto de lo esperado, con media nula; $\beta_{ik}$, la sensibilidad del título $i$ a ese factor; y $\varepsilon_i$, el componente específico, con media nula e independiente de los factores y de los componentes específicos de los demás títulos. Los factores son las fuentes de riesgo sistemático, y el componente específico, el riesgo diversificable. (ii) Hay muchos títulos, de modo que pueden formarse carteras bien diversificadas, en las que el riesgo específico es despreciable. (iii) No hay oportunidades de arbitraje: una cartera que no exige inversión neta y no tiene riesgo no puede tener una rentabilidad positiva. Y, (iv) los mercados no tienen fricciones y se permiten las ventas en corto. No se supone nada sobre las preferencias, salvo que se prefiere más riqueza a menos, ni sobre la distribución de las rentabilidades, y no se necesita la cartera de mercado.

\subsubsection{Desarrollo}
La idea se ve con un solo factor. Dos carteras bien diversificadas, $a$ y $b$, con betas $\beta_a\neq\beta_b$ respecto del factor, solo tienen riesgo factorial. Si se invierte una proporción $\beta_b/(\beta_b-\beta_a)$ en $a$ y $-\beta_a/(\beta_b-\beta_a)$ en $b$, los pesos suman uno y la beta de la combinación es nula: no tiene riesgo, y debe rendir el tipo sin riesgo. Esa condición equivale a
$$\frac{\mathbb E[r_a]-\gamma}{\beta_a}=\frac{\mathbb E[r_b]-\gamma}{\beta_b}=\lambda_1:$$
la prima por unidad de beta tiene que ser la misma en todas las carteras. Si no lo fuera, se compraría la que paga más prima por unidad de riesgo, se vendería la otra y se obtendría un beneficio sin riesgo. Por tanto, $\mathbb E[r_i]=\gamma+\beta_i\lambda_1$.

Con $K$ factores, el razonamiento es el mismo. Consideremos una \textbf{cartera de arbitraje}, con pesos $w_i$ que suman cero, de modo que no exige inversión neta, bien diversificada, de modo que su riesgo específico es despreciable, y sin exposición a ningún factor, $\sum_i w_i\beta_{ik}=0$ para todo $k$. Su rentabilidad es cierta e igual a $\sum_i w_i\,\mathbb E[r_i]$, y, para que no haya arbitraje, debe ser nula. Así, todo vector de pesos ortogonal al vector de unos y a los $K$ vectores de betas es también ortogonal al vector de rentabilidades esperadas. Por un resultado de álgebra lineal, eso solo es posible si las rentabilidades esperadas son una combinación lineal de esos vectores:
$$\mathbb E[r_i]=\lambda_0+\sum_{k=1}^{K}\beta_{ik}\,\lambda_k.$$
Con un título sin riesgo, que tiene todas las betas nulas, $\lambda_0=\gamma$. Cada $\lambda_k$ es la \textbf{prima del factor} $k$: la rentabilidad adicional de una cartera con beta unitaria respecto de ese factor y nula respecto de los demás, una \textbf{cartera que replica el factor}, de modo que $\lambda_k=\mathbb E[r_{F_k}]-\gamma$. Con un número finito de títulos, la relación es aproximada: se cumple para la mayoría de ellos, y las desviaciones de unos pocos no pueden explotarse sin riesgo.

\subsubsection{Resultados}
La APT da una relación lineal con varias betas. Comparte tres propiedades con el CAPM: solo se remunera el riesgo sistemático, porque el componente específico desaparece en las carteras diversificadas; la relación es lineal; y las betas son aditivas, de modo que la de una cartera respecto de cada factor es la media ponderada de las de sus componentes. El CAPM es un caso particular, con un solo factor, el mercado, y con su prima deducida del equilibrio. La diferencia está en lo que se exige para llegar a ella: la APT no necesita preferencias de media y varianza ni una cartera de mercado observable, pero, a cambio, no dice cuántos factores hay, cuáles son ni cuánto vale su prima. Esas tres cosas hay que determinarlas con los datos.

\subsubsection{Implicaciones para el coste del capital}
Aplicada a las acciones de una empresa, la APT da su coste de los recursos propios como
$$r_E=\gamma+\sum_{k=1}^{K}\beta_{E,k}\,\lambda_k,$$
y su aplicación sigue cuatro pasos. El primero es \textbf{elegir los factores}. CHEN, ROLL y ROSS (1986) propusieron factores macroeconómicos: las variaciones de la producción industrial, las de la inflación esperada y la inflación no anticipada, las del diferencial entre los bonos de empresa de baja calidad y la deuda pública, que mide la prima por riesgo de impago, y las de la pendiente de la curva de tipos. Encontraron que esos factores tenían primas significativas en las acciones estadounidenses y que, una vez incluidos, ni la rentabilidad de un índice de mercado ni el consumo añadían una prima propia. El segundo paso es \textbf{estimar las betas} de la acción respecto de cada factor, con una regresión de su rentabilidad sobre las sorpresas de los factores. El tercero es \textbf{estimar las primas}, con la rentabilidad media de carteras que replican cada factor o con regresiones de corte transversal de las rentabilidades medias sobre las betas, el método de FAMA y MacBETH (1973). El cuarto es sumar el tipo sin riesgo y las betas por las primas.

La ventaja para valorar proyectos es que el coste del capital deja de ser una caja negra: un proyecto cuyo resultado depende mucho del ciclo, de los tipos de interés o de la inflación tiene betas altas respecto de esos factores, y la tasa recoge cada exposición por separado. Es, además, la forma en que razonan muchas empresas: en la encuesta de GRAHAM y HARVEY (2001), las grandes empresas ajustaban sobre todo por el riesgo de tipo de cambio, el del ciclo económico, el de los precios de las materias primas y el de los tipos de interés, aunque a menudo en los flujos y no en la tasa. Como las betas son aditivas factor a factor, el procedimiento de desapalancar y reapalancar puede aplicarse a cada una.

\subsubsection{Limitaciones}
La principal es la que la hace flexible: la teoría no identifica los factores. Ni su número ni su naturaleza están determinados, y los que se eligen dependen del periodo y de la muestra, de modo que especificaciones distintas dan costes del capital distintos para la misma empresa. Si los factores se extraen estadísticamente de las covarianzas de las rentabilidades, con un análisis de componentes principales, no tienen interpretación económica; si se eligen variables macroeconómicas, la elección es en parte arbitraria. SHANKEN (1982) señaló, además, que la APT es difícil de contrastar: como su relación es solo aproximada y la estructura factorial cambia según qué carteras se tomen como títulos, sus predicciones no son únicas y no pueden rechazarse con claridad. En la práctica, la estimación acumula errores, porque cada factor añade una beta y una prima que estimar, y las primas de los factores son inestables en el tiempo. Por eso, para el coste del capital, la APT se usa menos como fórmula que como guía de qué riesgos sistemáticos tiene un proyecto.

\subsection{Los modelos factoriales empíricos: FAMA y FRENCH}
Los modelos factoriales empíricos tienen la forma de la APT, pero eligen los factores por su capacidad para explicar las rentabilidades medias que el CAPM no explica. El más conocido es el de tres factores de FAMA y FRENCH (1993), ampliado después con otros.

\subsubsection{Objetivo}
Responden a las anomalías del CAPM, diferencias sistemáticas de rentabilidad media entre acciones con la misma beta. Su objetivo es describirlas con pocos factores construidos a partir de las propias acciones y, aplicados al coste del capital, dar una rentabilidad exigida que tenga en cuenta las características de la empresa que se asocian a ellas, como su tamaño o su ratio entre valor contable y de mercado.

\subsubsection{Supuestos}
Suponen que las rentabilidades esperadas dependen linealmente de las exposiciones a unos pocos factores, cada uno medido como la rentabilidad de una cartera que compra unas acciones y vende otras; que la prima de cada factor es estable y se estima con su rentabilidad media histórica; y que, si el modelo está completo, la regresión de la rentabilidad en exceso de cualquier acción sobre los factores tiene una constante nula. No suponen un modelo de comportamiento del inversor: su justificación es empírica, y queda abierto si los factores miden riesgos o errores de valoración.

\subsubsection{Desarrollo}
FAMA y FRENCH (1993) partieron de las dos anomalías más robustas: las empresas pequeñas rendían más de lo que predecía su beta (BANZ, 1981), y también las de ratio alta entre valor contable y de mercado, llamadas \textit{de valor}, frente a las \textit{de crecimiento} (FAMA y FRENCH, 1992). Para recogerlas construyeron dos factores. Clasificaron las acciones por tamaño, en pequeñas y grandes según la mediana de la Bolsa de Nueva York, y por la ratio entre valor contable y de mercado, en tres grupos, y formaron seis carteras con los cruces. El factor de tamaño, SMB (\textit{small minus big}), es la rentabilidad media de las tres carteras de empresas pequeñas menos la de las tres de grandes; el factor de valor, HML (\textit{high minus low}), la de las dos carteras de ratio alta menos la de las dos de ratio baja. Cada factor es así una cartera que no exige inversión neta y que mide la rentabilidad adicional de una característica, neutralizando la otra. Con el factor de mercado, la regresión es
$$r_i-\gamma=\alpha_i+b_i\,(r_M-\gamma)+s_i\,SMB+h_i\,HML+\varepsilon_i,$$
y el coste de los recursos propios es
$$r_E=\gamma+b_E\,\lambda_M+s_E\,\lambda_{SMB}+h_E\,\lambda_{HML},$$
con las primas estimadas por las rentabilidades medias de los factores.

Después se añadieron otros. CARHART (1997) incorporó el \textbf{momento}, la tendencia de las acciones que más han subido en el último año a seguir rindiendo más en los meses siguientes (JEGADEESH y TITMAN, 1993), con un factor que compra las ganadoras y vende las perdedoras. FAMA y FRENCH (2015) propusieron un modelo de cinco factores que añade la \textbf{rentabilidad} operativa, RMW (\textit{robust minus weak}), y la \textbf{inversión}, CMA (\textit{conservative minus aggressive}): las empresas más rentables y las que invierten menos rinden más, una relación que justifican con el modelo de descuento de dividendos, porque, con el precio y el valor contable dados, una rentabilidad mayor o una inversión menor implican una tasa de descuento mayor.

\subsubsection{Resultados}
En las carteras formadas por tamaño y por la ratio entre valor contable y de mercado, los tres factores explican la mayor parte de la variación de las rentabilidades y dejan constantes próximas a cero, cuando el CAPM dejaba grandes rentabilidades sin explicar (FAMA y FRENCH, 1993). Las acciones pequeñas y las de valor tienen exposiciones positivas a SMB y a HML, y las grandes y las de crecimiento, negativas. FAMA y FRENCH (2015) encontraron que su modelo de cinco factores mejora al de tres y que, con los factores de rentabilidad e inversión, el de valor se vuelve redundante en su muestra; su principal fallo son las acciones pequeñas que invierten mucho pese a tener una rentabilidad baja.

Sobre qué miden los factores hay dos lecturas. Para FAMA y FRENCH (1996), son primas por riesgo: las empresas de valor suelen atravesar dificultades, y su rentabilidad refleja un riesgo de insolvencia que la beta no recoge. Para LAKONISHOK, SHLEIFER y VISHNY (1994), son errores de valoración: los inversores extrapolan en exceso el crecimiento pasado, pagan de más por las empresas de crecimiento y de menos por las de valor, y la diferencia de rentabilidad es la corrección posterior de esos errores.

\subsubsection{Implicaciones para el coste del capital}
Frente al CAPM, el modelo eleva el coste de los recursos propios de las empresas pequeñas y de las de valor y lo reduce en las grandes y en las de crecimiento. Pero su uso para valorar proyectos plantea dos problemas. El primero es que el tamaño y la ratio entre valor contable y de mercado son características de la empresa, no del proyecto: un proyecto no tiene capitalización bursátil, y no es evidente qué exposición a SMB o a HML debe atribuírsele, salvo la de empresas comparables. El segundo depende de la lectura de los factores. Si miden riesgos, su prima es un coste de oportunidad y debe entrar en la tasa. Si miden errores de valoración, STEIN (1996) mostró que la respuesta depende de la empresa: si sus directivos miran al largo plazo y no tiene restricciones financieras, debe descontar sus proyectos con una medida del riesgo fundamental, para la que la beta del CAPM puede seguir siendo útil aunque no prediga las rentabilidades de las acciones; si su horizonte es corto o depende de emitir acciones a los precios del mercado, debe usar la rentabilidad que el mercado espera de sus acciones, con su error incluido.

A ello se añade la imprecisión. FAMA y FRENCH (1997) obtuvieron con su modelo de tres factores errores típicos tan grandes como con el CAPM, porque a la incertidumbre sobre las primas se suma la imprecisión con que se estiman las exposiciones de cada sector a los factores, y advirtieron que para empresas o proyectos concretos la estimación es aún menos precisa. En la práctica, pocas empresas usan estos factores: en la encuesta de GRAHAM y HARVEY (2001), solo el $13,1\%$ de los directores financieros tenía en cuenta la ratio entre valor contable y de mercado, y el $11,1\%$, el momento.

\subsubsection{Limitaciones}
Su debilidad es la contrapartida de su origen: los factores se eligieron porque explicaban las rentabilidades de un periodo, y no por una teoría que diga por qué deben tener prima. Con muchos investigadores probando variables sobre los mismos datos, algunas parecerán significativas por azar. HARVEY, LIU y ZHU (2016) contaron más de $300$ factores propuestos en la literatura y concluyeron que, para tener en cuenta esa búsqueda múltiple, un nuevo factor debería superar un estadístico $t$ de $3$, en lugar del $2$ habitual, umbral que muchos de los publicados no superan. McLEAN y PONTIFF (2016) mostraron que la rentabilidad de las carteras basadas en $97$ variables que predecían las rentabilidades era un $26\%$ menor fuera de la muestra en que se descubrieron y un $58\%$ menor tras su publicación, lo que sugiere que parte de las primas era fruto de la búsqueda en los datos y parte, errores de valoración que los inversores corrigieron al conocerlos. La propia prima de tamaño se ha debilitado desde su publicación. Para el coste del capital, la consecuencia es que las primas de los factores, que se estiman con medias históricas, pueden no reflejar las que se exigirán en el futuro.

\subsection{La rentabilidad implícita: el modelo de GORDON y SHAPIRO}
Los métodos anteriores deducen la rentabilidad exigida del riesgo. El modelo de descuento de dividendos, formulado por WILLIAMS (1938) y llevado a su forma más conocida, con crecimiento constante, por GORDON y SHAPIRO (1956), recorre el camino inverso: la despeja del precio de la acción.

\subsubsection{Objetivo}
Busca la rentabilidad que el mercado descuenta hoy en el precio de la acción, dados los dividendos que se esperan: la \textbf{rentabilidad implícita}, o coste del capital implícito. Aplicado al coste del capital, da una estimación del coste de los recursos propios que no depende de rentabilidades históricas ni de un modelo de riesgo, y que mira al futuro; aplicado al conjunto del mercado, da la prima de riesgo implícita que el CAPM necesita.

\subsubsection{Supuestos}
Son cuatro. (i) El precio de la acción es igual a su valor fundamental, el valor actual de los dividendos que se esperan, de modo que el mercado valora bien. (ii) La tasa de descuento, $r_E$, es constante en el tiempo. (iii) Los dividendos crecen a una tasa constante, $g$, indefinidamente. Y, (iv) esa tasa es menor que la de descuento, $g<r_E$; si no, el valor sería infinito. Para relacionar $g$ con la empresa se añade un quinto: la empresa reparte cada año una proporción constante de su beneficio y reinvierte el resto con una rentabilidad contable constante.

\subsubsection{Desarrollo}
El valor de la acción es el de los dividendos que se esperan cobrar, porque incluso quien la vende cobra un precio que es el valor de los dividendos posteriores:\footnote{%
\begin{cita}[W. BUFFETT][1994][Annual report to shareholders of Berkshire Hathaway]
    We define intrinsic value as the discounted value of the cash that can be taken out of a business during its remaining life. Anyone calculating intrinsic value necessarily comes up with a highly subjective figure that will change both as estimates of future cashflows are revised and as interest rates move. Despite its fuzziness, however, intrinsic value is all-important and is the only logical way to evaluate the relative attractiveness of investments and business.
\end{cita}
}
$$P_0=\sum_{t=1}^{\infty}\frac{D_t}{(1+r_E)^t}.$$
Con dividendos que crecen a una tasa constante, $D_t=D_1(1+g)^{t-1}$, la suma es una progresión geométrica de razón $(1+g)/(1+r_E)<1$:
$$P_0=\frac{D_1}{1+r_E}\sum_{t=1}^{\infty}\left(\frac{1+g}{1+r_E}\right)^{t-1}=\frac{D_1}{1+r_E}\cdot\frac{1}{1-\frac{1+g}{1+r_E}}=\frac{D_1}{r_E-g},$$
y, despejando la tasa,
$$r_E=\frac{D_1}{P_0}+g.$$
El coste de los recursos propios es la \textbf{rentabilidad por dividendo} más la \textbf{tasa de crecimiento} de los dividendos, que es también la del precio: con crecimiento constante, el precio crece al mismo ritmo que los dividendos, y la rentabilidad total del accionista es el dividendo más la ganancia de capital.\footnote{%
    Los dividendos y la tasa deben expresarse ambos en términos nominales o ambos en términos reales: si se usan dividendos reales, sin inflación, $r_E$ es el coste real.
}

La tasa $g$ no se observa, y el quinto supuesto permite relacionarla con la empresa. Si la empresa retiene una proporción $b$ de su beneficio, $B_t$, y la reinvierte con una rentabilidad contable $ROE$, el beneficio crece cada año en $b\cdot ROE\cdot B_t$, de modo que el \textbf{crecimiento sostenible} es
$$g=b\cdot ROE.$$
Sustituyendo, con $D_1=(1-b)\,B_1$,
$$P_0=\frac{(1-b)\,B_1}{r_E-b\cdot ROE}.$$
Esta expresión, que es la del modelo de WALTER y GORDON que se estudia con la política de dividendos ([\authorfont{Ver Tema 3.B.3}]), muestra cuándo el crecimiento crea valor. Si $ROE=r_E$, el precio es $B_1/r_E$ sea cual sea $b$: retener beneficios hace crecer los dividendos, pero no el valor, porque se reinvierte justo a la rentabilidad que exigen los accionistas. Solo si $ROE>r_E$ retener crea valor, y el precio supera $B_1/r_E$ en el \textbf{valor actual de las oportunidades de crecimiento}. Por eso la rentabilidad por beneficio, $B_1/P_0$, la inversa del PER, solo es el coste de los recursos propios si la empresa no tiene oportunidades de crecimiento rentable.

Cuando el crecimiento no es constante, como en una empresa joven, se usan \textbf{modelos de varias etapas}: dividendos previstos uno a uno durante $T$ años y, desde entonces, un crecimiento estable a largo plazo, $g_L$, que no puede superar indefinidamente el de la economía:
$$P_0=\sum_{t=1}^{T}\frac{D_t}{(1+r_E)^t}+\frac{1}{(1+r_E)^T}\cdot\frac{D_{T+1}}{r_E-g_L}.$$
La tasa ya no se despeja con una fórmula: es la que iguala el precio con el valor actual de los dividendos, es decir, la TIR de comprar la acción al precio actual y cobrar sus dividendos, y se obtiene por aproximaciones sucesivas, como la de un proyecto. Para las empresas que no reparten dividendos, o que remuneran al accionista con recompras de acciones, los dividendos se sustituyen por el total de la remuneración al accionista o por el flujo de caja disponible para el accionista. Y, como los analistas prevén beneficios y no dividendos, la literatura más reciente obtiene la rentabilidad implícita con el \textbf{modelo del beneficio residual} de OHLSON (1995), que expresa el precio como el valor contable de los recursos propios, $BV_0$, más el valor actual de los beneficios que superan la rentabilidad exigida a ese valor:
$$P_0=BV_0+\sum_{t=1}^{\infty}\frac{B_t-r_E\,BV_{t-1}}{(1+r_E)^t}.$$
Es equivalente al descuento de dividendos si el valor contable solo cambia por los beneficios y los dividendos.

\subsubsection{Resultados}
El modelo da una estimación del coste de los recursos propios que se lee en el precio: la rentabilidad por dividendo más el crecimiento. De él se siguen tres resultados. El primero es la \textbf{sensibilidad} al crecimiento: un punto de error en $g$ es un punto de error en $r_E$, y, como el precio depende de la diferencia $r_E-g$, pequeñas variaciones de cualquiera de las dos cambian mucho la valoración. El segundo es que el crecimiento solo crea valor si se reinvierte con una rentabilidad superior a la exigida. El tercero, aplicado a un índice bursátil, es la \textbf{prima de riesgo implícita} del mercado: la rentabilidad por dividendo del índice más el crecimiento esperado de sus dividendos, menos el tipo sin riesgo, $\lambda_M=D_{M,1}/P_{M,0}+g_M-\gamma$.

Con las previsiones de beneficios de los analistas y el modelo del beneficio residual, CLAUS y THOMAS (2001) obtuvieron primas implícitas en torno al $3\%$ o menos, en Estados Unidos y en otros grandes mercados, muy inferiores a las medias históricas. GEBHARDT, LEE y SWAMINATHAN (2001) encontraron que la rentabilidad implícita de cada empresa depende sobre todo de su sector, de su ratio entre valor contable y de mercado, del crecimiento a largo plazo que prevén los analistas y de la dispersión de sus previsiones, que explican en torno al $60\%$ de sus diferencias entre empresas.

\subsubsection{Implicaciones para el coste del capital}
El método aporta dos cosas. La primera es una estimación del coste de los recursos propios de la empresa que mira al futuro y no depende de una beta histórica, útil sobre todo en empresas maduras con dividendos estables, como las de servicios regulados, y como contraste del CAPM: si la rentabilidad implícita y la del CAPM difieren mucho, alguna de las dos estimaciones, o el precio, merece revisión. La segunda es la prima de mercado implícita, que puede usarse en el CAPM en lugar de la histórica y que se ajusta a las condiciones del momento: sube cuando los precios caen en una crisis y baja en las fases de euforia.

Pero da el coste de los recursos propios de la empresa en su conjunto, con su endeudamiento, y no el de un proyecto. Para un proyecto distinto de la actividad habitual, habría que aplicarlo a empresas comparables y corregir las diferencias de endeudamiento, como con la beta.

\subsubsection{Limitaciones}
La primera es el crecimiento: $g$ no se observa, sino que se estima con el crecimiento histórico de los dividendos de la empresa, con su crecimiento sostenible o con las previsiones de los analistas; el supuesto de crecimiento constante e indefinido es una aproximación pobre para la mayoría de las empresas, y el resultado depende tanto de él que el método traslada al crecimiento la incertidumbre que el CAPM concentra en la beta y en la prima. Si se usan las previsiones de los analistas, se hereda su optimismo: EASTON y SOMMERS (2007) estimaron que ese sesgo elevaba en unos $2,8$ puntos porcentuales la rentabilidad implícita media obtenida con ellas, una magnitud comparable a las primas implícitas de en torno al $3\%$ de los estudios anteriores. La segunda es el alcance: solo sirve para empresas cotizadas y, en su versión simple, para las que reparten dividendos con un crecimiento estable; no sirve si $g\geq r_E$. La tercera es conceptual: el método supone que el precio es correcto. Si el mercado valora mal, la rentabilidad implícita es la que se espera dado el precio, no la que corresponde al riesgo, y, como vimos al tratar los modelos factoriales, cuál de las dos debe usarse para valorar proyectos depende del horizonte y de las restricciones financieras de la empresa (STEIN, 1996). La cuarta es la circularidad: si se usa para valorar la propia empresa, la tasa se obtiene del precio que se pretende explicar.
