    % --- Imagen adaptativa ---
    % \imagenfit[alto]{fichero}{título}{fuente}: la imagen queda fija entre los párrafos donde se escribe (no flota), con su
    % título en el índice de figuras y la fuente debajo. Sin [alto]: las figuras PDF (TikZ) salen al tamaño con que se
    % dibujaron y solo se reducen si no caben en el ancho o en el alto de la página; las PNG/JPG ocupan el ancho del texto
    % hasta \imgaltoras del alto de la página. Con [alto] (fracción del alto de la página; [1] = página entera) se fija el
    % tamaño de esa imagen. Si no cabe en lo que queda de página, pasa entera a la siguiente.
    % \imagenfithorizontal{fichero}{título}{fuente}: la imagen en una página girada (apaisada), en ese mismo punto del texto.
    % Reglas: main/GUIA_TEMAS.md, §4 (Figuras).
    \newdimen\imgcap
    \newdimen\imgh
    \newdimen\imgw
    \newdimen\imgneed
    \newsavebox\imgcaja
    \newcommand{\imgaltoras}{0.30}% alto por defecto de las PNG/JPG (fracción del alto de la página)
    \makeatletter
    \newcommand{\imagenfit}[4][]{%
      \addvspace{10pt}%
      \par\begingroup
        \imgcap=3.5\baselineskip% alto aproximado de título + fuente
        \imgw=\linewidth
        \filename@parse{#2}\def\img@pdf{pdf}%
        \if\relax\detokenize{#1}\relax
          \ifx\filename@ext\img@pdf
            \sbox\imgcaja{\includegraphics{\detokenize{#2}}}% tamaño natural
            \ifdim\wd\imgcaja<\imgw \imgw=\wd\imgcaja \fi
            \imgh=\ht\imgcaja
          \else
            \imgh=\imgaltoras\textheight
          \fi
        \else
          \imgh=#1\textheight
        \fi
        \dimen@=\textheight \advance\dimen@ by -\imgcap
        \ifdim\imgh>\dimen@ \imgh=\dimen@ \fi
        \sbox\imgcaja{\includegraphics[width=\imgw,height=\imgh,keepaspectratio]{\detokenize{#2}}}%
        \imgneed=\ht\imgcaja \advance\imgneed by \imgcap
        \Needspace{\imgneed}%
        \noindent\begin{minipage}{\textwidth}
          \centering
          \captionof{figure}{\textemdash\ #3}\par\vspace{0.3em}% título arriba
          \usebox\imgcaja\par
          \vspace{0.3em}\small\textit{Fuente: }#4% fuente abajo
        \end{minipage}%
      \endgroup
      \par\addvspace{10pt}\RequestRestore
    }
    \newcommand{\imagenfithorizontal}[3]{%
      \begin{landscape}
        \centering
        \captionof{figure}{\textemdash\ #2}\par\vspace{0.3em}%
        \includegraphics[width=\linewidth,height=\dimexpr\textheight-4\baselineskip\relax,keepaspectratio]{\detokenize{#1}}\par
        \vspace{0.3em}{\small\textit{Fuente: }#3}%
      \end{landscape}%
    }
    \makeatother

    % \anexo{Título}: un anexo equivale a un \section (su título sale del mismo tamaño). Dentro se usan \subsection,
    % \subsubsection y \paragraph, nunca \section: se numeran I.1, I.1.1, I.1.1.1 y las páginas del anexo, I-1, I-2…
    % En el índice de contenido inicial solo figura el nombre del anexo. Sus apartados forman el índice del anexo, que sale
    % solo si el anexo los tiene, justo después de la introducción del anexo (antes del primer \subsection).
    % \anexo*{Título}: el mismo anexo, sin ese índice. Reglas: main/GUIA_TEMAS.md, §6.

    \newcommand{\anx@empezar}[2]{%
      \clearpage
      \refstepcounter{anexo}%
      \global\EnAnexotrue
      \setcounter{page}{1}\gdef\thepage{\theanexo-\arabic{page}}%
      \setcounter{section}{\numexpr\value{anexo}-1\relax}\stepcounter{section}% el anexo hace de \section
      \gdef\thesection{\Roman{section}}\gdef\theHsection{anexo.\Roman{section}}%
      \phantomsection
      \protected@xdef\leq@current@section{Anexo \theanexo.- \unexpanded{#2}}\gdef\leq@current@subsection{}% índice de ecuaciones
      \addtocontents{stc}{\protect\contentsline{section}{Anexo \theanexo.- #2}{}{\@currentHref}}% índice inicial: solo el nombre, sin página
      \addcontentsline{toc}{section}{Anexo \theanexo.- #2}% marcadores del PDF
      {\noindent\normalfont\Large\bfseries Anexo \theanexo.- #2\par}\nobreak\vspace{1.5ex \@plus .2ex}%
      #1\@ifundefined{anx@lista@\theanexo}{\global\anx@pendientefalse}{}%
    }
    % Índice del anexo: cada apartado escribe una línea en el fichero .atc (anexo, nivel, texto, página, enlace); al
    % empezar el documento se leen las de la compilación anterior y se guardan por anexo (dos pasadas, como cualquier índice).
    % En ese índice los apartados suben un nivel, porque el anexo hace de \section.
    \def\anx@entrada#1#2#3#4#5{\@ifundefined{anx@lista@#1}{\global\@namedef{anx@lista@#1}{}}{}%
      \expandafter\g@addto@macro\csname anx@lista@#1\endcsname{\contentsline{#2}{#3}{#4}{#5}}}
    \tcbuselibrary{breakable}
    \def\anx@indice{\ifanx@pendiente\global\anx@pendientefalse

            before upper={\parindent=0pt\parskip=2pt}]
          \small
          % números más anchos que en el índice inicial (I.1, I.1.1, I.1.1.1; páginas I-1)
          \def\@pnumwidth{2.6em}\def\@tocrmarg{3.4em}%
          \def\l@section##1##2{\addpenalty\@secpenalty\addvspace{1.0em \@plus\p@}\setlength\@tempdima{2.6em}%
            \begingroup\parindent\z@\rightskip\@pnumwidth\parfillskip-\@pnumwidth\leavevmode\bfseries
            \advance\leftskip\@tempdima\hskip-\leftskip##1\nobreak\hfil\nobreak\hb@xt@\@pnumwidth{\hss##2\kern-\p@\kern\p@}\par\endgroup}%

          \csname anx@lista@\theanexo\endcsname

      \fi}
    \def\anx@escribir#1#2#3{\ifEnAnexo

    % margen superior, así que no cambia las páginas del tema. Se enciende con \pagenumbering{arabic} y se apaga con

    \cs_new_protected:Npn \cab_guardar:n #1
      {
        \tl_gset:Nn \g_cab_titulo_tl {#1}
        \regex_replace_once:nnN { \A \s* Tema \s+ \S+ \s* \-\-? \s* } { } \g_cab_titulo_tl % sin «Tema X.Y.N --»
        \regex_replace_once:nnN { \.? \s* \c{\\} (\[ [^\]]* \])? \s* (\c{large} \s*)? } { .\  } \g_cab_titulo_tl % título. subtítulo
        \regex_replace_once:nnN { [\.\s]* \Z } { } \g_cab_titulo_tl % sin subtítulo: nada detrás
      }

\section{Decisiones de inversión: ¿cuándo crea valor un proyecto?}
\puntosclave{
    Toda inversión compara su rentabilidad con un coste de oportunidad que tiene dos componentes, el tiempo ($\gamma$) y el riesgo ($\lambda$). Con información perfecta y mercados de capitales perfectos, el VAN es el criterio correcto: mide el valor que crea el proyecto y todos los propietarios aprueban maximizarlo, sean cuales sean sus preferencias (separación de FISHER).

    El índice de rentabilidad tiene sentido con racionamiento de capital: ordena los proyectos por el VAN que crea cada euro, pero falla con proyectos indivisibles o con restricciones en varios periodos.

    La TIR mide la rentabilidad implícita del proyecto y el mayor coste del capital que soporta. Coincide con el VAN si el perfil del VAN es decreciente, pero puede no existir, no ser única u ordenar mal proyectos excluyentes (punto de FISHER); aplicada al proyecto diferencial, lleva a la misma decisión que el VAN. Que suponga reinvertir los flujos a la propia TIR es objeto de debate.

    El payback mide la liquidez y la exposición temporal del proyecto, no su valor: ignora lo que ocurre después del plazo, y el plazo máximo es arbitrario.

    Los tres criterios son estáticos: suponen que se decide ahora o nunca. Con información imperfecta, la flexibilidad tiene valor y el VAN estático infravalora los proyectos flexibles: con flexibilidad, más incertidumbre aumenta el valor del proyecto.

    Los árboles de decisión miden el valor de esperar a conocer la información; las opciones reales lo valoran con el descuento correcto (VAN ampliado). Su uso formal es limitado, y la competencia erosiona el valor de esperar.

    La sensibilidad, los escenarios y la simulación de Montecarlo miden, con profundidad creciente, la dispersión del resultado, con un plan fijo o con flexibilidad. Pero la dispersión total no es el riesgo que el mercado remunera, salvo que haya fricciones financieras: solo entonces importan las preferencias ante el riesgo.
}


Invertir es renunciar hoy a unos recursos a cambio de unos flujos futuros. Por eso toda decisión de inversión se somete a fuerzas que operan en sentido contrario. De un lado está la \textbf{rentabilidad}, que permite evaluar si el capital comprometido en un proyecto generará un rendimiento capaz de justificar la inversión. De otro, dos fuerzas que la contrapesan. La \textbf{intertemporalidad} exige tener en cuenta el coste de oportunidad de comprometer un determinado volumen de capital en un proyecto concreto durante un periodo determinado, es decir, el rendimiento al que se renuncia en la mejor alternativa disponible.\footnote{%
    Que un euro futuro valga menos que uno presente puede parecer trivial, pero para un economista no lo es: es el \textbf{valor temporal del dinero}, que depende tanto de la inflación como de la existencia de inversiones alternativas. Unas oportunidades de inversión con alto rendimiento o una inflación elevada llevan a descontar los flujos de caja lejanos con más intensidad que los que el inversor recibe en fechas cercanas. Para que el descuento sea coherente, los flujos nominales se descuentan con tipos nominales y los reales con tipos reales, que se relacionan por la ecuación de FISHER (1930): $(1+i)=(1+i_r)(1+\pi)$, con $i$ el tipo nominal, $i_r$ el real y $\pi$ la inflación.
} La \textbf{información} exige considerar la forma en que las contingencias futuras pueden alterar el curso del proyecto y, sobre todo, su rentabilidad.

Ambas fuerzas se condensan en la tasa con que se descuentan los flujos del proyecto, que mide el coste de oportunidad del capital, $r=\gamma+\lambda$: $\gamma$ remunera el tiempo (es el tipo de interés sin riesgo) y $\lambda$ remunera el riesgo (es una prima). Pero la información no actúa solo contra la rentabilidad. La incertidumbre resta valor al proyecto cuando es un riesgo que los inversores no pueden eliminar diversificando y exigen que se les remunere; y se lo añade cuando la empresa puede esperar a ver cómo se resuelve y adaptar sus decisiones, porque esa flexibilidad vale más cuanto mayor es la incertidumbre. De ahí la tesis de este bloque: con información perfecta y mercados de capitales perfectos, el valor actual neto es el criterio correcto para decidir, y los demás criterios lo son solo en la medida en que coinciden con él; con información imperfecta, sigue siéndolo únicamente si incorpora el valor de la información que se obtendrá a lo largo de la vida del proyecto, y un valor actual neto que lo ignora infravalora los proyectos flexibles.

Por eso distinguimos dos contextos. Con información perfecta, la única fuerza que se opone a la rentabilidad es el tiempo, y los criterios clásicos de valoración, con sus virtudes y sus problemas, se ven con nitidez. Con información imperfecta aparece la dimensión informativa, que extiende esos criterios en dos direcciones: cuánto vale la información que falta y cómo medir el efecto de la incertidumbre sobre el resultado del proyecto. Cuál es la tasa de descuento adecuada, el coste del capital, se aborda en el bloque siguiente.


\subsection{Información perfecta: la rentabilidad frente al coste del tiempo}
Con información perfecta, quien decide conoce con certeza los flujos de caja del proyecto y los tipos de interés a los que puede prestar y endeudarse en cada periodo. El único coste de oportunidad es el del tiempo, y la tasa de descuento es el tipo de interés sin riesgo, $r_t=\gamma_t$. En este marco se presentan los tres criterios básicos de evaluación de proyectos: el valor actual neto (VAN), la tasa interna de retorno (TIR) y el periodo de recuperación o \textit{payback}. Los tres se aplican en las primeras etapas de la evaluación y persiguen el mismo fin, estudiar la viabilidad de una inversión, pero responden a preguntas distintas: el VAN mide cuánto valor crea el proyecto, en unidades monetarias; la TIR, qué rentabilidad implícita ofrece, en tanto por ciento; y el payback, cuánto tarda en devolver lo invertido, en unidades de tiempo. Por eso no siempre llegan al mismo resultado y se emplean de forma complementaria. No tienen, sin embargo, el mismo rango: como veremos, solo el VAN se deriva directamente del objetivo de maximizar el valor de la empresa para sus propietarios, y la TIR y el payback son útiles en la medida en que coinciden con él o aportan información que el VAN no recoge.\footnote{%
    En la práctica se usan los tres. En la encuesta de GRAHAM y HARVEY (2001) a $392$ directores financieros estadounidenses, el $74,9\%$ declaraba emplear siempre o casi siempre el VAN, el $75,7\%$ la TIR y el $56,7\%$ el payback\textcolor{orange}{\textsuperscript{[Probable]}}. Que la TIR se use tanto como el VAN, pese a sus problemas, se explica en parte porque una rentabilidad en tanto por ciento es más fácil de comunicar y de comparar con el coste de los fondos\textcolor{orange}{\textsuperscript{[Probable]}}.
}

Los tres criterios comparten su materia prima, los \textbf{flujos de caja} del proyecto, cuya obtención no es evidente en la práctica. Lo relevante son los flujos incrementales: la diferencia entre los cobros y pagos de la empresa con el proyecto y sin él.\footnote{%
    De ahí cuatro reglas prácticas: (i) los costes hundidos, ya incurridos e irrecuperables, no se cuentan, porque no dependen de la decisión; (ii) los recursos propios que el proyecto ocupa (un terreno, una nave) se valoran por su coste de oportunidad, aunque no supongan un pago; (iii) se incluyen los efectos sobre el resto de la empresa, como la pérdida de ventas de otros productos propios; y, (iv) no se incluyen los flujos de la financiación (intereses, devolución de la deuda, dividendos), porque su coste ya está en la tasa de descuento y contarlos dos veces sesgaría el resultado (BREALEY, MYERS y ALLEN, 2020).
} Al evaluar una inversión deben considerarse tres tipos: (i) flujos de explotación;\footnote{%
    La contribución de la inversión al resultado bruto de explotación (EBITDA, por sus siglas en inglés: beneficio antes de intereses, impuestos, depreciaciones y amortizaciones) es la diferencia entre los ingresos y los gastos adicionales que genera, sin contar depreciaciones ni amortizaciones. Del EBITDA se deduce el impuesto teórico sobre el resultado de explotación adicional, que se calcula multiplicando el tipo impositivo que soporta el proyecto por la variación del resultado de explotación (después de amortizaciones, que son fiscalmente deducibles) y teniendo en cuenta los créditos fiscales por pérdidas.
} (ii) flujos de inversión;\footnote{%
    La inversión se entiende en sentido amplio: incluye desde el fondo de maniobra (\textit{working capital}) hasta los activos fijos. El fondo de maniobra es la diferencia entre los activos corrientes de la empresa (efectivo, existencias y cuentas por cobrar) y sus pasivos corrientes (cuentas por pagar y deudas a corto plazo); representa los fondos disponibles para las operaciones diarias y es una medida importante de su liquidez y de su salud financiera ([\authorfont{Ver Tema 3.B.1}]). Un fondo de maniobra positivo indica que la empresa tiene recursos a corto plazo suficientes para cumplir sus obligaciones; uno negativo puede anticipar tensiones de liquidez. Las variaciones del fondo de maniobra que exige el proyecto se deducen del EBITDA. La inversión en activos fijos comprende la inversión en capacidad de producción y crecimiento, ya sea en activos tangibles (maquinaria, terrenos, edificios), intangibles (investigación y desarrollo, patentes y licencias, capital empresarial) o financieros (acciones de filiales) para el crecimiento externo.
} y, (iii) flujos extraordinarios.

\subsubsection{Valor actual neto (VAN): el valor que crea el proyecto}
El primer criterio, y el que sirve de referencia a los demás, es el \textbf{valor actual neto}. Se fundamenta en el valor temporal del dinero: para determinar si un proyecto es viable, trae al presente todos sus flujos de caja netos, descontándolos al coste de oportunidad del tiempo, y les resta la inversión inicial. Con un tipo de descuento constante, el criterio del valor actual de FISHER (1930) se escribe:\footnote{%
    Calibrar sus componentes puede resultar difícil en la práctica. Si los tipos difieren entre periodos, cada flujo se descuenta con el producto de los factores de los periodos que median hasta su cobro, $VAN=-I_0+\sum_{t=1}^{T}CF_t\big/\prod_{s=1}^{t}(1+r_s)$, o, de forma equivalente, con el tipo de contado a plazo $t$ de la curva de tipos de interés. Suponer un tipo constante equivale a suponer que esa curva es totalmente plana, algo poco realista que aquí se adopta por simplicidad ([\authorfont{Ver Tema 3.B.24}]). Un método básico para fijar la tasa de descuento es tomar la curva de tipos del activo sin riesgo y sumarle una prima que depende de las características del proyecto, como veremos al introducir la incertidumbre y en el bloque siguiente.

    En tiempo continuo, el valor actual en $t$ de un flujo $CF_s$ que se cobra en $s$ es $CF_s\,e^{-\gamma_{t,s}(s-t)}=CF_s\,B(t,s)$, donde $\gamma_{t,s}$ es el tipo sin riesgo para el periodo $(s-t)$ y $B(t,s)$, el precio en $t$ de un bono que paga $1$ en $s$. Con riesgo, el flujo se sustituye por el previsto, $\mathbb E_t[CF_s]$, y el descuento añade la prima de riesgo del proyecto: $CF_s\,e^{-(\gamma_{t,s}+\lambda)(s-t)}=CF_s\,B(t,s)\,e^{-\lambda(s-t)}\approx CF_s/(1+\gamma_{t,s}+\lambda)^{s-t}$.
}
$$VAN=-I_0+\sum_{t=1}^{T}\frac{CF_t}{(1+r)^t},$$
donde $I_0$ es la inversión inicial, $CF_t$ el flujo de caja neto del periodo $t$, $T$ la vida del proyecto y $r$ la tasa de descuento, que con información perfecta es el tipo de interés sin riesgo. El VAN es, por tanto, una \textbf{medida de rentabilidad absoluta}: en unidades monetarias, mide cuánto aumenta (o disminuye) el valor de la empresa si se acomete la inversión.\footnote{%
    Si además los flujos son constantes, $CF_t=CF$, la expresión se reduce a $VAN=-I_0+CF\,\frac{1-(1+r)^{-T}}{r}$. De este caso se derivan otros de interés: la renta perpetua, cuyo valor actual es $CF/r$, y los flujos que crecen a una tasa constante $g<r$, cuyo valor actual es $CF_1/(r-g)$. Para estas extensiones, véase el capítulo $16$ de VERNIMMEN, QUIRY y LE FUR (2022)\textcolor{orange}{\textsuperscript{[Probable]}}.
}

La regla de decisión se sigue de esa interpretación. Si el VAN es positivo, el proyecto rinde más que su coste de oportunidad: obtener sus flujos en el mercado de capitales exigiría invertir más de lo que cuesta el proyecto, y acometerlo aumenta la riqueza de los propietarios en esa diferencia. Si es negativo, el proyecto destruye valor, porque los mismos fondos rendirían más en el mercado. Si es nulo, rinde exactamente su coste de oportunidad: no crea ni destruye valor, y la empresa es indiferente.\footnote{%
    \textbf{NOTA.-} Un VAN nulo equivale al beneficio normal de la teoría microeconómica: el capital obtiene justo su remuneración de mercado. Un VAN positivo es, por tanto, una renta económica, y en un mercado competitivo no debería abundar: los proyectos con VAN positivo suelen apoyarse en alguna ventaja que los competidores no pueden replicar de inmediato, como una patente, una tecnología, una posición de mercado o un coste menor. Un VAN muy elevado sin una ventaja que lo explique es, más a menudo, señal de un error en la estimación de los flujos que de una oportunidad excepcional.
} Por la lógica del arbitraje, sería deseable acometer todo proyecto con VAN positivo.

¿Por qué maximizar el VAN, y no la satisfacción de unos propietarios que pueden tener preferencias muy distintas sobre cuándo consumir? La respuesta es el \textbf{teorema de separación} de FISHER (1930), que formalizó HIRSHLEIFER (1958). Supongamos un propietario que dispone hoy de unos recursos $Y_0$, que puede invertir en proyectos de rendimiento decreciente (la frontera de inversión) o prestar y pedir prestado sin límite al tipo $r$ (la recta de mercado). La decisión se separa en dos etapas. Primero, la empresa invierte hasta que el rendimiento del último euro invertido iguala el tipo de interés: es el punto $P^*$, en que la frontera es tangente a una recta de pendiente $-(1+r)$. Esa inversión, $I^*$, es la que maximiza el valor actual de la riqueza, $W_0=Y_0+VAN$. Después, cada propietario se desplaza a lo largo de la recta de mercado hasta el consumo que prefiere, prestando si es paciente ($a$) o endeudándose si es impaciente ($b$). Como la recta de mercado más alejada del origen es la mejor para todos, cualesquiera que sean sus preferencias, todos los propietarios aprueban por unanimidad la misma decisión: maximizar el VAN. La empresa puede delegarla en sus directivos sin consultar a los propietarios, con una regla que solo exige conocer los flujos y el tipo de interés.

\imagenfit{fisher_separacion.pdf}{Separación de FISHER: la inversión óptima no depende de las preferencias}{Elaboración propia a partir de FISHER (1930) y HIRSHLEIFER (1958).}

El VAN tiene, además, una propiedad que ningún otro criterio comparte: es aditivo. El VAN de dos proyectos independientes es la suma de sus VAN, de modo que cada proyecto puede evaluarse por separado y el valor de la empresa es la suma del de sus proyectos. En definitiva, la principal virtud del VAN es que reduce un proyecto de inversión a una suma de flujos de caja descontados, lo que facilita el análisis objetivo de las alternativas e introduce, de forma inequívoca, el criterio más relevante: la creación de valor para la empresa.

El teorema descansa, sin embargo, en un mercado de capitales perfecto,\footnote{%
    HIRSHLEIFER (1958) mostró que, si el tipo al que la empresa se endeuda es mayor que el tipo al que presta, la separación se rompe: ya no hay una única recta de mercado, y la inversión óptima pasa a depender de las preferencias de los propietarios.
} y sus fricciones limitan la validez del resultado: la empresa puede no tener acceso ilimitado al mercado de capitales, pueden existir incompatibilidades entre proyectos, o costes de gestión y restricciones de capacidad que impidan acometer todos los proyectos con VAN positivo. Para fijar ideas, seguiremos a lo largo del bloque un mismo ejemplo. Una empresa estudia dos proyectos. El proyecto A exige hoy una inversión $I_A$ y devuelve un único flujo $CF_A$ al cabo de un periodo; el proyecto B, mayor y más lento, exige $I_B>I_A$ y devuelve un único flujo $CF_B$ al cabo de dos periodos. Aplicando la expresión del VAN, sus valores son
$$VAN_A=-I_A+\frac{CF_A}{1+r},\qquad VAN_B=-I_B+\frac{CF_B}{(1+r)^2}.$$
Supongamos que, al tipo de interés vigente, los dos son positivos y $VAN_B>VAN_A$. Si los proyectos son independientes y la empresa puede financiarse sin límite, acomete los dos, porque cada uno suma su VAN al valor de la empresa. Si son \textbf{excluyentes} (dos usos alternativos del mismo solar, dos tecnologías para la misma planta), elige B aunque exija más inversión: si eligiese A, la diferencia $I_B-I_A$ que deja de invertir solo podría colocarse en el mercado, con VAN nulo, y el valor total sería $VAN_A<VAN_B$. Cuando el límite es el capital disponible, la pregunta cambia.

\paragraph{Índice de rentabilidad: el VAN por euro invertido}
El VAN mide cuánto valor crea un proyecto, pero no cuánto capital necesita para crearlo. Cuando el capital es escaso, esa segunda dimensión importa, y la recoge el \textbf{índice de rentabilidad}, una extensión del VAN que lo divide por la inversión inicial del proyecto $p$:
$$R_p=\frac{VAN_p}{I_0^p}.$$
Es positivo si y solo si lo es el VAN, y mide la riqueza creada por cada euro invertido: una suerte de multiplicador del desembolso inicial sobre el VAN.\footnote{%
    Algunos textos lo definen como el cociente entre el valor actual de los flujos y la inversión, $VA_p/I_0^p=1+R_p$; el umbral de aceptación es entonces $1$, y no $0$. En el anexo de ejemplos numéricos se calcula con cifras.
} Mientras el VAN mide la rentabilidad en términos absolutos, el índice la mide en términos relativos, por lo que es complementario del VAN y no su sustituto. En el ejemplo, si B exige mucha más inversión que A, puede ocurrir que $R_A>R_B$ aunque $VAN_B>VAN_A$: A crea más valor por euro, pero B crea más valor en total. Si los proyectos son excluyentes y no hay restricción de capital, se elige B: un índice más alto no implica una inversión mejor.

La utilidad propia del índice aparece con el \textbf{racionamiento de capital}. Si la empresa dispone de un presupuesto $K$ que no le permite acometer todos los proyectos con VAN positivo, su problema es maximizar el VAN total sin que la inversión supere $K$. Con una sola restricción y proyectos divisibles, basta ordenar los proyectos de mayor a menor índice y aceptarlos hasta agotar el presupuesto. El índice es el VAN por unidad del recurso escaso, como el margen por unidad de un factor escaso en la teoría de la empresa, y el índice del último proyecto aceptado es el precio sombra del capital: el VAN que añadiría un euro más de presupuesto. En la figura, el índice de cada proyecto es la pendiente del radio que lo une con el origen; con un presupuesto $K=I_B$, A más la fracción $\theta=(K-I_A)/I_B$ de B que cabe en el resto crean más valor que B solo, porque cada euro invertido en A rinde más que en B.

\imagenfit{indice_rentabilidad.pdf}{Índice de rentabilidad y racionamiento de capital}{Elaboración propia a partir de LORIE y SAVAGE (1955).}

Esa ordenación falla en dos casos, que estudiaron LORIE y SAVAGE (1955) y WEINGARTNER (1963)\textcolor{orange}{\textsuperscript{[Probable]}}. El primero son los proyectos \textbf{indivisibles}. En el ejemplo, si A y B no pueden fraccionarse y el presupuesto es $K=I_B$, el índice recomienda A, que deja sin invertir $K-I_A$; ese resto solo rinde el tipo de interés, con VAN nulo, y la empresa obtiene $VAN_A<VAN_B$. Salvo que exista otro proyecto que quepa en el resto y compense la diferencia, B es mejor pese a su menor índice: la elección deja de ser una ordenación y se convierte en un problema de programación entera, el problema de la mochila. El segundo son las restricciones en \textbf{varios periodos} o sobre varios recursos, en que un único cociente no puede ordenar los proyectos. WEINGARTNER (1963) formuló entonces la selección como un programa matemático cuyos precios duales miden el valor de relajar el presupuesto de cada periodo.

Que haya racionamiento es, en sí mismo, una desviación del mundo de FISHER: con un mercado de capitales perfecto, todo proyecto con VAN positivo encuentra financiación. El racionamiento puede ser externo, cuando la empresa no obtiene fondos al tipo de mercado por imperfecciones como la información asimétrica entre la empresa y quienes la financian ([\authorfont{Ver Tema 3.B.3}]); o interno, cuando la dirección fija presupuestos para disciplinar a los responsables de cada división o porque su capacidad de gestión es limitada\textcolor{orange}{\textsuperscript{[Probable]}}. En el primer caso, el índice es un criterio de segundo óptimo; en el segundo, el presupuesto es una restricción que la empresa se impone, y la pregunta relevante es si el coste de relajarla es menor que el VAN al que renuncia. El índice relativiza el VAN por la escala del proyecto; la TIR, el criterio siguiente, va más allá y prescinde también de la tasa de descuento.


\subsubsection{Tasa interna de retorno (TIR): la rentabilidad implícita del proyecto}
El segundo criterio, que en la práctica se usa tanto como el VAN, es la \textbf{tasa interna de retorno}, que mide la \textbf{rentabilidad implícita de una inversión}. Es la tasa de descuento que iguala el valor actual de los flujos de caja del proyecto con su inversión inicial, es decir, la que anula su VAN; es lo que KEYNES (1936) llamó eficiencia marginal del capital ([\authorfont{Ver Tema 3.A.34}]):\footnote{%
    La ecuación es un polinomio de grado $T$ en $x=1/(1+TIR)$, de modo que, salvo en proyectos de uno o dos periodos, no tiene solución explícita y se resuelve por aproximaciones sucesivas. En el ejemplo, $TIR_A=CF_A/I_A-1$ y $TIR_B=\sqrt{CF_B/I_B}-1$. El anexo de ejemplos numéricos la calcula para un proyecto de dos periodos.
}\textsuperscript{,}\footnote{%
    \textbf{OJO.-} KEYNES (1936) definió la eficiencia marginal del capital como la tasa que iguala el valor actual de los rendimientos esperados de un bien de capital con su precio de oferta, es decir, como la TIR de un proyecto. El tipo de rendimiento sobre el coste de FISHER (1930), con el que se la suele identificar, se define en cambio al comparar dos alternativas: es la tasa que iguala sus valores actuales. ALCHIAN (1955) mostró que coincide con la TIR del proyecto diferencial, que veremos al tratar el punto de FISHER.
}
$$-I_0+\sum_{t=1}^{T}\frac{CF_t}{(1+TIR)^t}=0.$$
Para un proyecto, la TIR es lo que la rentabilidad al vencimiento es para un bono ([\authorfont{Ver Tema 3.B.24}]): el rendimiento que obtiene quien paga hoy $I_0$ a cambio de la serie de flujos.

La regla de decisión compara esa rentabilidad con el coste de oportunidad: el proyecto se acepta si su TIR supera la tasa de descuento, $TIR>r$, y se rechaza si es inferior. La lógica es la del arbitraje: si el proyecto rinde más de lo que los mismos fondos obtendrían en el mercado, crea valor. La TIR es, así, el mayor coste del capital que el proyecto puede soportar sin destruir valor, y la diferencia $TIR-r$ mide su margen de seguridad frente a una subida de ese coste. Cuando el VAN decrece con la tasa de descuento, como en un proyecto en que primero se invierte y después se cobra, la regla coincide con la del VAN: $VAN(r)>0$ si y solo si $r<TIR$.

De ahí sus dos ventajas como complemento del VAN, con el que comparte el objetivo de comparar rentabilidades con el coste del capital. La primera es que expresa la creación de valor en unidades de tipo de interés y no en unidades monetarias, y que puede calcularse sin conocer la tasa de descuento, que solo hace falta para compararla: sortea así, en parte, la dificultad de medir el coste del capital, porque basta saber si es mayor o menor que la TIR. La segunda es que, como el índice de rentabilidad, permite comparar la rentabilidad de proyectos de distinta escala, entre los que el VAN, por construcción, favorece a los grandes aunque rindan menos por euro invertido. Esa segunda ventaja es también su principal riesgo, porque, cuando los proyectos son excluyentes, el de mayor TIR no es necesariamente el que más valor crea. Los dos resultados que siguen se obtienen al relacionar el VAN y la TIR a través del \textbf{perfil del VAN}, el VAN como función de la tasa de descuento.


\paragraph{El perfil del VAN: cuándo la TIR no existe o no es única}
La TIR es el punto en que el perfil del VAN, la función $VAN(r)$, corta el eje. Que ese punto exista, sea único y separe los tipos que crean valor de los que lo destruyen depende de la forma del perfil, y esta, del calendario de signos de los flujos. En un proyecto convencional, en que los flujos negativos se acumulan en los primeros periodos y los positivos en los últimos, el perfil decrece con el tipo de descuento, porque al subir $r$ pierden más valor los cobros, que son lejanos, que los pagos, que son próximos. Hay entonces una sola TIR, el VAN es positivo a su izquierda y negativo a su derecha, y la regla de la TIR coincide con la del VAN (panel a). Es el caso de A y de B en el ejemplo.

Si los flujos cambian de signo más de una vez, esa coincidencia se pierde. La regla de los signos de DESCARTES acota el número de TIR por el de cambios de signo en la serie de flujos: con dos cambios puede haber dos. Supongamos que A obliga a restaurar el terreno al cabo de dos periodos, con un coste $C$. Sus flujos pasan a ser $-I_A$, $+CF_A$ y $-C$, y, si la restauración es lo bastante costosa, el perfil crece primero y decrece después: el VAN es negativo con tipos bajos, que dan mucho peso al coste final, y con tipos altos, que restan valor al cobro, y es positivo entre sus dos TIR (panel b).\footnote{%
    Con $x=1/(1+r)$, $VAN_A=-I_A+CF_A\,x-C\,x^2$ es una parábola en $x$. Tiene dos TIR positivas si es negativa en $r=0$, $CF_A<I_A+C$, y positiva en su máximo, $CF_A^2>4\,I_A\,C$. LORIE y SAVAGE (1955) ilustraron el caso con una bomba que acelera la extracción de un pozo de petróleo: adelanta al primer año un ingreso que, sin ella, se obtendría en el segundo (el anexo de ejemplos numéricos da sus cifras).
} ¿Cuál es entonces la verdadera TIR? Ninguna de las dos mide la rentabilidad del proyecto, porque el proyecto no es una inversión pura: después del cobro, la empresa debe un pago futuro, como quien ha recibido un préstamo, y para esa parte un tipo de interés más alto es favorable. La TIR mezcla en un solo número una inversión y una financiación, y por eso deja de ordenar. La decisión correcta la sigue dando el VAN: el proyecto crea valor si el coste del capital está entre las dos TIR.

En el caso extremo, la TIR no existe. Un proyecto en que el cliente paga un anticipo, la empresa ejecuta la obra y cobra el resto al final tiene flujos $+$, $-$, $+$, y puede tener un VAN positivo a cualquier tipo de descuento sin cortar nunca el eje (panel c): la TIR no dice nada, mientras que el VAN dice que se acometa.\footnote{%
    La inexistencia de la TIR, y otras cuestiones de su cálculo, se tratan en el capítulo $17$ de VERNIMMEN, QUIRY y LE FUR (2022)\textcolor{orange}{\textsuperscript{[Probable]}}.
}

\imagenfit{perfil_van.pdf}{El perfil del VAN y la TIR según el calendario de signos de los flujos}{Elaboración propia a partir de LORIE y SAVAGE (1955).}

La enseñanza es que la TIR es una propiedad de la serie de flujos, no de la decisión: solo es un criterio fiable cuando el perfil del VAN es decreciente. Incluso entonces, como veremos a continuación, puede ordenar mal proyectos excluyentes.

\paragraph{El punto de FISHER: cuándo VAN y TIR ordenan distinto}
El segundo resultado aparece al comparar proyectos excluyentes, aunque ambos sean convencionales y tengan el mismo riesgo. En el ejemplo, A es pequeño y rápido y B, grande y lento. Supongamos que B crea más valor sin descontar, $CF_B-I_B>CF_A-I_A$, pero que A tiene mayor TIR. Sus perfiles se cortan: con tipos bajos, B tiene mayor VAN; con tipos altos, A. El tipo en que se cortan, en que los dos proyectos tienen el mismo VAN, es la \textbf{tasa de FISHER}, $r_F$, y la decisión correcta depende del lado en que quede el coste del capital: si $r<r_F$, se elige B, aunque su TIR sea menor; si $r>r_F$, se elige A. La TIR, en cambio, elige A siempre, y se equivoca en todo el tramo $r<r_F$. Por eso el punto de FISHER es útil cuando hay que elegir entre proyectos que no pueden acometerse a la vez: dice para qué tipos de descuento conviene cada uno.

\imagenfit{punto_fisher.pdf}{El punto de FISHER: VAN y TIR ordenan distinto}{Elaboración propia a partir de FISHER (1930).}

La tasa de FISHER es la TIR del \textbf{proyecto diferencial}, $B-A$, cuyos flujos son la diferencia entre los de B y los de A: es la rentabilidad de los fondos adicionales que exige elegir B en lugar de A.\footnote{%
    De ahí que coincida con el tipo de rendimiento sobre el coste de FISHER (1930), definido precisamente al comparar dos alternativas (ALCHIAN, 1955). En el ejemplo, los flujos del proyecto diferencial son $-(I_B-I_A)$, $-CF_A$ y $+CF_B$; como solo cambian de signo una vez, su TIR es única. La tasa de FISHER no existe cuando un proyecto tiene mayor VAN que el otro a cualquier tipo de descuento, y entonces VAN y TIR coinciden en la elección.
} Ese es el criterio que reconcilia ambos instrumentos. Elegir B equivale a acometer A más el proyecto diferencial, y este conviene si su TIR supera el coste del capital, es decir, si $r<r_F$. Aplicada a los incrementos, la TIR lleva a la misma decisión que el VAN.

¿Por qué se cortan los perfiles? Por dos diferencias entre los proyectos. La primera es la \textbf{escala}: B exige más inversión y, con tipos bajos, genera más valor en euros, aunque menos por euro. La segunda es el \textbf{calendario}: los flujos de B llegan más tarde, y el valor de los flujos lejanos es más sensible al tipo de descuento. La pendiente del perfil es proporcional al valor actual de los flujos y a su duración, el plazo medio ponderado en que se cobran, de modo que un proyecto lento se comporta como un bono largo: su valor cae más cuando sube el tipo de interés.\footnote{%
    Derivando, $dVAN/dr=-\sum_{t}t\,CF_t/(1+r)^{t+1}=-D\cdot VA/(1+r)$, donde $VA$ es el valor actual de los flujos y $D$, su duración de MACAULAY ([\authorfont{Ver Tema 3.B.24}]).
}

El punto de FISHER ilustra, así, la sensibilidad del VAN al tipo de descuento y la necesidad de emplear el VAN y la TIR como indicadores complementarios, pero la regla de selección debe aplicarse con cautela. Hereda los supuestos de los criterios que la componen, como la certeza y el reduccionismo con que resumen las características de los proyectos. Si los proyectos difieren en riesgo, compararlos con un mismo tipo de descuento no es correcto, porque cada uno tiene su coste del capital; y si difieren en duración, la comparación debe tener en cuenta qué ocurre cuando termina el más corto.\footnote{%
    Si A dura un periodo y B dos, elegir A deja abierta la posibilidad de invertir de nuevo al cabo de un periodo. Cuando los proyectos se repiten, se comparan cadenas de reemplazo de igual duración o el VAN expresado como anualidad equivalente: la renta constante que, durante la vida de cada proyecto, tiene su mismo valor actual (BREALEY, MYERS y ALLEN, 2020).
}

\paragraph{Limitaciones de la TIR}
Los supuestos de fondo de la TIR son los del VAN (certeza, flujos independientes y maximización del valor), y sus limitaciones comunes se tratan más adelante. Pero la TIR tiene, además, limitaciones propias, que nacen de resumir un proyecto en una tasa y no en un valor.

La primera, como vimos, es que puede no existir o no ser única cuando los flujos cambian de signo más de una vez, y entonces no ordena los proyectos. La segunda es que prescinde de la escala y del calendario: ordenar proyectos excluyentes por su TIR favorece a los pequeños y a los rápidos, y la decisión correcta exige calcular la TIR del proyecto diferencial, lo que en la práctica equivale a hacer el análisis del VAN. La tercera es que no es aditiva: la TIR de dos proyectos acometidos juntos no es la suma ni la media de sus TIR, de modo que no puede evaluarse cada proyecto por separado y agregar, y un proyecto malo puede esconderse dentro de uno bueno. La cuarta es que se compara con una sola tasa: si el coste del capital varía con el plazo, porque la curva de tipos no es plana, no hay un único umbral con el que comparar la TIR, mientras que el VAN descuenta cada flujo con su tipo.

La quinta es objeto de debate: la \textbf{tasa de reinversión}. SOLOMON (1956) atribuyó los conflictos entre VAN y TIR a que cada criterio supone implícitamente que los flujos intermedios se reinvierten a una tasa distinta: el VAN, al tipo de descuento, y la TIR, a la propia TIR. Si esa tasa es irreal, como en un proyecto con una TIR muy alta, la TIR exagera la rentabilidad. De ahí la \textbf{TIR modificada}, que capitaliza los flujos intermedios al coste del capital hasta el final del proyecto y calcula la tasa que iguala ese valor final con la inversión inicial, y que tiene siempre una única solución. DUDLEY (1972)\textcolor{orange}{\textsuperscript{[Probable]}} y KEANE (1979) objetaron que ninguno de los dos criterios supone nada sobre la reinversión: ambos se calculan solo con los flujos del proyecto, y la TIR es una propiedad de esa serie, como la rentabilidad al vencimiento de un bono, que no depende de lo que su tenedor haga con los cupones. La reinversión solo importa al comparar proyectos con calendarios distintos, y entonces la pregunta relevante es qué rinden los fondos que uno libera antes que el otro: en un mercado de capitales perfecto, el coste del capital, que es lo que supone el VAN. El debate no cambia, por tanto, la conclusión: los conflictos se resuelven a favor del VAN, y la TIR es útil como información complementaria sobre el margen de rentabilidad del proyecto. El criterio siguiente, el periodo de recuperación, renuncia a ambos: no pregunta cuánto rinde el proyecto, sino cuándo devuelve lo invertido.


\subsubsection{Periodo de recuperación (\textit{payback}): cuándo se devuelve la inversión}
El último criterio para evaluar los proyectos con información perfecta es el \textbf{periodo de recuperación} o payback, que mide el \textbf{tiempo necesario para recuperar la inversión} inicial. Se expresa en unidades de tiempo (años o meses, según la periodicidad de los flujos de caja) y, en su versión simple, es el primer periodo $P$ en que los flujos acumulados del proyecto igualan, al menos, la inversión inicial:\footnote{%
    Si los flujos son constantes, $CF_t=CF$, el payback es simplemente $P=I_0/CF$. Su versión descontada, $P_d$, acumula los flujos descontados, $\sum_{t=1}^{P_d}CF_t/(1+r)^t\geq I_0$: es el plazo a partir del cual el VAN del proyecto, si terminase en ese momento, deja de ser negativo. Con flujos constantes, $P_d=-\ln(1-r\,I_0/CF)/\ln(1+r)$, que solo existe si $CF/I_0>r$, es decir, si la renta perpetua que genera el proyecto cubre la inversión.
}
$$\sum_{t=1}^{P}CF_t\geq I_0.$$
La regla de decisión fija un plazo máximo, $P^*$, que actúa como punto de corte: el proyecto se acepta si se recupera antes, $P\leq P^*$, y se descarta en caso contrario.

A diferencia del VAN y de la TIR, el payback no mide la rentabilidad del proyecto, sino la rapidez con que devuelve los fondos, y eso explica que su uso esté extendido pese a sus defectos: lo empleaba siempre o casi siempre más de la mitad de los directores financieros encuestados por GRAHAM y HARVEY (2001), sobre todo en las empresas pequeñas. Es un método adecuado, en primer lugar, cuando la empresa tiene restricciones en su acceso al mercado de capitales (de liquidez, o un alto coste del endeudamiento): si los fondos que devuelve el proyecto son los que financiarán los siguientes, recuperarlos pronto tiene un valor que el VAN, que supone un mercado de capitales perfecto, no recoge. En segundo lugar, es una aproximación al riesgo, aunque tosca: los flujos lejanos suelen ser los más inciertos, y un proyecto que se recupera pronto mantiene menos tiempo expuesto el capital comprometido. En tercer lugar, es sencillo de calcular y de comunicar, lo que permite a una gran empresa filtrar muchos proyectos menores sin un análisis completo de cada uno. Por último, la preferencia por los proyectos que devuelven pronto responde también a los incentivos de los directivos: quien es evaluado por sus resultados a corto plazo tiene motivos para preferirlos aunque creen menos valor (NARAYANAN, 1985).\footnote{%
    McDONALD (2000) sostiene, además, que reglas prácticas como un plazo de recuperación exigente o una tasa mínima de rentabilidad superior al coste del capital pueden aproximar la decisión óptima cuando invertir hoy supone renunciar a la opción de esperar, que veremos con la información imperfecta\textcolor{orange}{\textsuperscript{[Probable]}}.
} Por eso el payback no tiene por fin determinar el proyecto más rentable, como el VAN o la TIR, sino complementar ese estudio en empresas con difícil acceso a la liquidez o en grandes empresas que afrontan proyectos menores.

Idealmente, además, debería calcularse descontando los flujos futuros. No hacerlo, aunque es más sencillo y habitual entre los analistas, supone omitir el valor temporal del dinero, y el payback simple solo vale como aproximación con inflación baja y coste de oportunidad reducido. El \textbf{payback descontado} corrige ese defecto y es siempre mayor que el simple. En el ejemplo, A devuelve su inversión en un periodo y B, en dos. Con un plazo máximo de un periodo, la regla rechaza B aunque cree más valor que A: como la TIR, el payback favorece a los proyectos rápidos, pero por una razón más simple, porque no mira más allá del plazo.\footnote{%
    El anexo de ejemplos numéricos calcula el payback simple y el descontado de un proyecto con flujos constantes.
}

\paragraph{Limitaciones del payback}
Sus limitaciones nacen de que no es un criterio de valor. La primera, y principal, es que no tiene en cuenta los flujos de caja que se generan después de recuperar la inversión, por lo que es una medida limitada de la rentabilidad y del riesgo a largo plazo y debe usarse junto con los demás instrumentos de valoración. Dos proyectos con el mismo payback pueden crear valores muy distintos, y la regla sesga la inversión contra los proyectos de maduración larga, como la investigación o las infraestructuras, que son a menudo los que más valor crean. Puede incluso aceptar proyectos con VAN negativo, si tienen pagos importantes después del plazo, como un coste de desmantelamiento o la restauración del terreno en el proyecto A.

La segunda es que el plazo máximo es arbitrario: el VAN y la TIR toman su umbral del coste de oportunidad del capital, mientras que $P^*$ no se deriva de ningún criterio económico. GORDON (1955) mostró que, en un proyecto de vida larga con flujos constantes, la inversa del payback aproxima la TIR, de modo que exigir un plazo de tres años equivale a exigir una rentabilidad cercana al $33\%$, muy superior al coste del capital de la mayoría de las empresas.\footnote{%
    Con flujos constantes durante $T$ periodos, la TIR resuelve $I_0/CF=[1-(1+TIR)^{-T}]/TIR$; si $T$ es grande, el término $(1+TIR)^{-T}$ tiende a cero y $TIR\approx CF/I_0=1/P$. La aproximación empeora cuanto más corta es la vida del proyecto.
}

La tercera es que el payback simple da el mismo peso a todos los flujos anteriores al plazo, sin descontarlos; el descontado corrige este defecto, pero no el primero. La cuarta es que, como aproximación al riesgo, solo recoge el que depende del tiempo, y no la dispersión de los resultados posibles, que es lo que miden los instrumentos que veremos con la información imperfecta. El payback es, en suma, un indicador de liquidez y de exposición temporal, útil como complemento, pero no un criterio para elegir entre proyectos.

\subsubsection{Limitaciones comunes}
Los tres criterios comparten unos supuestos y, con ellos, unas limitaciones que no se resuelven eligiendo uno u otro. 

La primera es la \textbf{tasa de descuento}. Determina el resultado del análisis y, sin embargo, no es observable directamente, sino que debe estimarla el analista. Debe reflejar el coste de oportunidad de la inversión ajustado por su riesgo, porque los proyectos con un elevado nivel de incertidumbre aumentan el riesgo global de la empresa.\footnote{%
    A esta cuestión dedicamos el último bloque del tema. Su efecto sobre el resultado es mayor cuanto más lejanos son los flujos: como vimos con el punto de FISHER, el VAN de un proyecto lento es muy sensible al tipo de descuento, de modo que un error en la tasa puede invertir la decisión. 
    
    Por eso conviene acompañar el VAN de un análisis de su sensibilidad a la tasa, que veremos con la información imperfecta.
}

La segunda es el \textbf{calendario y la reinversión de los flujos}. Los criterios suponen que los flujos ocurren a intervalos temporales regulares y al final de cada periodo, lo que permite comparar flujos de momentos distintos. Pero también, tácitamente, supone que los fondos que genera el proyecto pueden reinvertirse con una rentabilidad igual a la tasa de descuento. Como vimos con la TIR, este segundo supuesto es objeto de debate, pero importa cuando no se cumple: si la empresa sufre racionamiento de capital, los fondos que libera un proyecto valen más que el tipo de interés, y el VAN calculado con el coste del capital los infravalora.\footnote{%
    Si los flujos no son regulares, se descuentan por sus fechas exactas (en las hojas de cálculo, con funciones como VNA.NO.PER). 
    
    Si la tasa de reinversión difiere del coste del capital, puede hacerse explícita: se capitalizan los flujos intermedios a esa tasa hasta el final del horizonte y se descuenta el valor final, que es la lógica de la TIR modificada; y, si hay racionamiento en varios periodos, se resuelve el programa matemático de WEINGARTNER (1963), cuyos precios duales valoran los fondos de cada periodo.
}

La tercera es la \textbf{independencia de los flujos}. Los criterios evalúan cada proyecto por separado, como si sus flujos no dependieran de los de otros proyectos ni de las decisiones futuras de la empresa. Pero los proyectos interactúan, y sus flujos son endógenos a esas decisiones: uno puede restar ventas a otro, o abrir el camino a inversiones que sin él no serían posibles. Con interacciones, la estructura de los flujos no es lineal, y el valor de un conjunto de proyectos deja de ser la suma de sus valores.\footnote{%
    Las interacciones entre proyectos simultáneos se resuelven evaluando combinaciones: el VAN de cada alternativa es el del conjunto de proyectos que implica o, lo que es lo mismo, el del proyecto incremental. Las interacciones en el tiempo, en que un proyecto crea la posibilidad de otros futuros, son las opciones reales que veremos con la información imperfecta.
}

La cuarta es el \textbf{objetivo}. Los criterios suponen que la decisión persigue estrictamente maximizar el valor actual neto, y no consideran factores difíciles de monetizar, como las consideraciones estratégicas, la responsabilidad social corporativa o los problemas de información y de gestión que acompañan a un proyecto concreto, salvo que se incorporen explícitamente en los flujos o en la tasa de descuento. Detrás hay dos problemas distintos. Uno es de alcance: el valor para los propietarios no recoge los efectos del proyecto sobre terceros, salvo que la regulación o los precios los internalicen, y su evaluación desde el punto de vista social exige un análisis coste-beneficio. El otro es de incentivos y de juicio, porque quien estima los flujos no es neutral. Los directivos pueden perseguir objetivos propios, como el tamaño de la empresa (JENSEN y MECKLING, 1976; [\authorfont{Ver Tema 3.B.3}]), y quienes proponen un proyecto tienden a sobrestimar sus beneficios y a subestimar sus costes: es el sesgo optimista que KAHNEMAN y LOVALLO (1993) atribuyen a la «visión desde dentro», la de quien razona a partir de los detalles de su proyecto y no de lo que ocurrió en proyectos parecidos.\footnote{%
    El Eurotúnel, inaugurado en 1994, es un caso emblemático. Su construcción costó un $80\%$ más de lo presupuestado, y su financiación, un $140\%$ más; las previsiones anunciaban $15,9$ millones de pasajeros en los trenes Eurostar en el primer año de explotación, y en 1995, el primer año completo, fueron $2,9$ millones, un $18\%$ de lo previsto (FLYVBJERG, BRUZELIUS y ROTHENGATTER, 2003). El caso no es aislado: en los proyectos ferroviarios estudiados por estos autores, el sobrecoste medio fue del $44,7\%$ y el déficit medio de demanda, del $51,4\%$\textcolor{orange}{\textsuperscript{[Probable]}}, y lo atribuyen tanto al sesgo optimista como a la presentación interesada de las cifras para lograr la aprobación del proyecto. La corrección que proponen KAHNEMAN y LOVALLO (1993) es adoptar la «visión desde fuera»: ajustar la previsión con la distribución de los resultados de proyectos parecidos.
}

La quinta, que abre el siguiente apartado, es la \textbf{información}. Los tres criterios suponen que los flujos se conocen, o que su incertidumbre puede resumirse en un valor esperado y en una prima en la tasa de descuento, y que la decisión se toma ahora o nunca: invertir hoy o renunciar para siempre. Ninguno recoge que la información se revela de forma progresiva y que la empresa puede esperar, ampliar, reducir o abandonar el proyecto según lo que aprenda. Esa flexibilidad tiene valor, y medirlo exige abandonar el supuesto de información perfecta.

\subsection{Información imperfecta: el valor de la información y la medida del riesgo}
Hasta aquí, la información perfecta reducía la decisión a comparar la rentabilidad del proyecto con el coste del tiempo. A partir de ahora, la pregunta es cómo afectan al proyecto, a su rentabilidad y a su calendario, unas contingencias futuras que no se conocen de antemano. Como vimos al cerrar el apartado anterior, los criterios clásicos tienen un \textbf{carácter estático}: suponen implícitamente que la decisión de inversión se toma una vez y es irreversible, y que la información relevante está disponible \textit{ex ante}. En muchos proyectos reales, en cambio, la información se revela de forma progresiva y las decisiones óptimas pueden variar a lo largo del tiempo, porque una inversión rentable en un momento puede dejar de serlo después, al cambiar el contexto con el que se relaciona.\footnote{%
    \textbf{OJO.-} Este fenómeno no es la inconsistencia dinámica de KYDLAND y PRESCOTT (1977), en que un plan óptimo hoy deja de serlo mañana sin que llegue información nueva, porque cambian los incentivos de quien decide ([\authorfont{Ver Tema 3.A.36}]). Aquí el plan cambia porque llega información, y prever ese cambio forma parte de la decisión óptima.
} Es la consecuencia lógica de uno de los fenómenos que perturban el mundo real: la \textbf{información imperfecta}.

La información es imperfecta en dos dimensiones ([\authorfont{Ver Tema 3.A.13}]). Es \textbf{incompleta} cuando nadie conoce de antemano el estado de la naturaleza que se dará (la demanda, los precios, la tecnología), aunque todos compartan la misma información; y es \textbf{asimétrica} cuando unos agentes saben más que otros. La información asimétrica distorsiona las decisiones de inversión de formas propias: los directivos, que saben más que los inversores, pueden renunciar a proyectos rentables para no emitir acciones infravaloradas (MYERS y MAJLUF, 1984) o invertir los fondos sobrantes en proyectos que no crean valor (JENSEN, 1986). Esos problemas se estudian en el Tema 3.B.3 ([\authorfont{Ver Tema 3.B.3}]); aquí nos ocupamos de la información incompleta, la que afecta al propio proyecto.

La información incompleta entra en la valoración por tres vías, y por eso no hay un único método. La primera es el \textbf{precio del riesgo}: si los flujos son inciertos, se sustituyen por su valor esperado y el descuento añade una prima, $r=\gamma+\lambda$, que remunera el riesgo que los inversores no pueden diversificar. Es la forma habitual de incorporar la incertidumbre, y su medida es el objeto del bloque siguiente.\footnote{%
    La alternativa es descontar al tipo sin riesgo los \textbf{equivalentes ciertos} de los flujos, la cantidad segura que se valora igual que cada flujo incierto. ROBICHEK y MYERS (1966) mostraron que descontar con una tasa ajustada constante supone que el riesgo crece con el plazo a un ritmo determinado, lo que no es adecuado cuando la incertidumbre se resuelve en una fecha concreta del proyecto; los equivalentes ciertos permiten ajustar por separado el riesgo de cada flujo.
} La segunda es el \textbf{valor de la información}: si la incertidumbre se resuelve con el tiempo y la empresa puede reaccionar, poder esperar, ampliar o abandonar el proyecto tiene un valor que el VAN estático ignora. La tercera es la \textbf{dispersión de los resultados}: cuánto pueden desviarse la rentabilidad y el valor del proyecto de lo esperado, y qué variables lo explican. A medir la segunda dedicamos el primer apartado, y a medir la tercera, en cualquier contexto, el segundo.

\subsubsection{¿Cuánto vale la información? Árboles de decisión y opciones reales}
Para entender por qué hacen falta métodos que incorporen los resultados contingentes, hay que saber primero cuánto vale la información que falta. Volvamos al proyecto A y supongamos que su flujo es incierto: será $CF^+$ si la demanda es alta, con probabilidad $p$, y $CF^-$ si es baja, con probabilidad $1-p$, de modo que, al tipo de descuento $r$, el proyecto crea valor con demanda alta y lo destruye con demanda baja. La demanda se conocerá al cabo de un periodo, y la empresa puede invertir hoy o esperar a conocerla.

\paragraph{Árboles de decisión: el valor de esperar}
El \textbf{árbol de decisión} representa este problema de forma compacta y permite evaluar múltiples alternativas de inversión bajo diversos escenarios, con el consiguiente ahorro de tiempo y recursos en la selección de proyectos. Es un problema de un solo decisor frente a la naturaleza, no un juego entre agentes (RAIFFA, 1968): sus nodos son de decisión, en que el inversor elige entre las acciones disponibles; de azar, en que la naturaleza elige un estado con una probabilidad conocida; y terminales, con el resultado de cada trayectoria.\footnote{%
    Los nodos de azar no deben confundirse con los conjuntos de información de los juegos en forma extensiva, que agrupan los nodos que un jugador no puede distinguir ([\authorfont{Ver Tema 3.A.14}]). En el árbol de decisión, el decisor conoce en cada nodo de decisión todo lo ocurrido hasta entonces. Aunque aquí se emplea como instrumento gráfico, el árbol es por definición un objeto matemático recursivo, y sirve también para hacer simulaciones, resolver problemas de optimización dinámica o predecir resultados con técnicas de aprendizaje automático.
} Se resuelve hacia atrás, con la regla de inducción del análisis de decisiones (RAIFFA, 1968), que asigna a cada nodo $n$ un valor $u(n)$:
$$u(n)=\begin{cases}\text{su resultado,} & \text{si $n$ es terminal,}\\[0.3em] \sum_i p_i\,u(n_i), & \text{si es de azar,}\\[0.3em] \max_i u(n_i), & \text{si es de decisión,}\end{cases}$$
donde $n_i$ son los nodos que le siguen. El valor esperado del proyecto es el del nodo inicial. Es la misma inducción hacia atrás de la programación dinámica ([\authorfont{Ver Tema 3.A.29}]).

En el ejemplo, si la empresa invierte hoy (panel a), su VAN es el esperado, descontando los flujos al tipo $r$; si espera (panel b), conoce la demanda antes de decidir, invierte solo si es alta y obtiene el flujo un periodo más tarde. Aplicando la regla anterior, los dos valores son
$$VAN_{\text{hoy}}=-I_A+\frac{p\,CF^++(1-p)\,CF^-}{1+r},\qquad VAN_{\text{esperar}}=\frac{p}{1+r}\left(-I_A+\frac{CF^+}{1+r}\right).$$
Conviene esperar si $VAN_{\text{esperar}}>VAN_{\text{hoy}}$, y la diferencia es el \textbf{valor de esperar}, el de la información que se obtiene al hacerlo. Su origen está en la asimetría de la decisión: con la información, la empresa evita la pérdida del estado malo, porque no invierte, y conserva la ganancia del bueno. Su coste es el retraso de esa ganancia.\footnote{%
    Llamando $a=-I_A+CF^+/(1+r)>0$ y $b=-I_A+CF^-/(1+r)<0$ a los VAN de invertir hoy con demanda alta y baja, $VAN_{\text{hoy}}=p\,a+(1-p)\,b$ y $VAN_{\text{esperar}}=p\,a/(1+r)$, de modo que
    $$VAN_{\text{esperar}}-VAN_{\text{hoy}}=\underbrace{(1-p)\,|b|}_{\text{pérdida evitada}}-\underbrace{p\,a\,\frac{r}{1+r}}_{\text{coste del retraso}}.$$
    Si la información pudiera obtenerse sin esperar, su valor sería el primer término, el valor esperado de la información perfecta (RAIFFA y SCHLAIFER, 1961).
} De ahí un resultado que contradice la intuición del VAN con una prima de riesgo: si los flujos se vuelven más dispersos sin que cambie su media, el VAN de invertir hoy no varía, pero el de esperar aumenta, porque solo cuenta el estado bueno. Con flexibilidad, más incertidumbre aumenta el valor del proyecto.

\imagenfit{arbol_esperar.pdf}{Árbol de decisión: invertir hoy o esperar a conocer la demanda}{Elaboración propia a partir de RAIFFA (1968) y DIXIT y PINDYCK (1994).}

Este tipo de análisis ofrece una evaluación sistemática de múltiples factores y facilita la elección de la mejor alternativa teniendo en cuenta tanto su rentabilidad como su riesgo: las opciones que no son atractivas se descartan en cada nodo, y, una vez desestimadas, el inversor solo tiene que escoger entre las alternativas del nodo inicial. Pero el árbol tiene también limitaciones. Un árbol en que todas las decisiones se toman al principio, como el del panel (a), elige de antemano el sendero que seguirá el proyecto y trata las decisiones como irreversibles, cuando en la realidad, a medida que el proyecto avanza, aparece nueva información y, con ella, nuevas opciones, como incorporar nuevas tecnologías, cancelar el proyecto o aplazarlo. Representar esa flexibilidad exige nodos de decisión posteriores, como en el panel (b), y el número de ramas crece entonces rápidamente con los periodos y los estados posibles. Sobre todo, el árbol no dice con qué tasa descontar: la flexibilidad cambia el riesgo del proyecto, porque el valor de la oportunidad de esperar depende de forma no lineal del valor del proyecto, y una tasa ajustada única deja de ser correcta. Resolver ese problema es lo que aportan las opciones reales.\footnote{%
    El anexo de ejemplos numéricos resuelve un árbol con dos decisiones sucesivas, ampliar la planta o construir una nueva, y en cada caso hacerlo a escala grande o pequeña.
}

\paragraph{Opciones reales: el precio de la flexibilidad}
Incrementar u obtener cierta versatilidad en el transcurso de una inversión tiene un valor que el análisis convencional no refleja, y las \textbf{opciones reales} son la forma más extendida de incorporarlo a la decisión. MYERS (1977) acuñó el término para subrayar que muchas oportunidades de inversión son como opciones de compra sobre activos reales: representan el derecho, pero no la obligación, de modificar un proyecto de inversión cuando se dispone de nueva información. La oportunidad de acometer A es una opción de compra cuyo subyacente es el valor actual de los flujos del proyecto, $V$; cuyo precio de ejercicio es la inversión, $I_A$; y cuyo vencimiento es el plazo durante el que puede posponerse la decisión (LUEHRMAN, 1998). Al vencimiento, la opción vale $\max\{V-I_A,\,0\}$, porque solo se ejerce si el proyecto crea valor; antes, vale más que el VAN de invertir hoy, porque conserva la posibilidad de no invertir si las cosas van mal. Por eso el valor de un proyecto flexible es su \textbf{VAN ampliado}, que según TRIGEORGIS (1996) es
$$VAN_{\text{ampliado}}=VAN_{\text{estático}}+\text{valor de las opciones}.$$

La valoración de opciones es, en el fondo, el árbol de decisión con el descuento correcto. COX, ROSS y RUBINSTEIN (1979) mostraron que una opción puede valorarse con un árbol binomial en que, en cada nodo, su valor se replica con una cartera del subyacente y del activo sin riesgo; eso equivale a calcular el valor esperado con unas probabilidades ajustadas por el riesgo y descontarlo al tipo sin riesgo, lo que resuelve el problema de la tasa. Cuando los periodos se acortan, el árbol converge al modelo de BLACK y SCHOLES (1973) ([\authorfont{Ver Tema 3.B.25}]), que, adaptado a una opción real de posponer, da su valor como
$$C=S^*\,N(d_1)-V(I_0)\,N(d_2),\qquad d_1=\frac{\ln[S^*/V(I_0)]}{\sigma\sqrt{t}}+\frac{\sigma\sqrt{t}}{2},\qquad d_2=d_1-\sigma\sqrt{t}.$$
$S^*=S-V(CF_t)$ es el valor actual de los flujos del proyecto menos el de los que se pierden por esperar; $V(I_0)=I_t/(1+r)^t$, el valor actual de la inversión pospuesta; $\sigma$, la volatilidad del valor del proyecto, que suele aproximarse por la del sector; $t$, el plazo hasta la decisión final; y $N(\cdot)$, la función de distribución normal estándar. El primer término es el valor de posponer: los flujos que se obtendrán si se ejerce la opción, ponderados por $N(d_1)$, la sensibilidad del valor de la opción al del proyecto (su delta). El segundo es el coste de posponer: la inversión, ponderada por $N(d_2)$, la probabilidad, en un mundo neutral al riesgo, de que llegue a acometerse. La regla de decisión compara ese valor con el de invertir hoy: si el VAN de invertir hoy es mayor que el valor de la opción, $\mathbb E_0[VAN]>C$, no se pospone; si $C\geq\mathbb E_0[VAN]$, se pospone, porque invertir hoy destruye la opción.

De la analogía se siguen los determinantes del valor de una opción real, que son los de una opción financiera. Crece con la \textbf{volatilidad} del valor del proyecto: como la empresa solo invierte en los estados buenos, una mayor dispersión aumenta las ganancias posibles sin aumentar las pérdidas, que quedan limitadas a no invertir. Crece también con el \textbf{plazo} durante el que puede esperarse y con el tipo de interés, que reduce el valor actual del precio de ejercicio; y disminuye con los flujos que se pierden mientras se espera, que actúan como los dividendos de una acción. Por eso invertir solo compensa si el valor del proyecto supera la inversión en un margen que crece con la incertidumbre: el umbral de inversión ya no es $V=I$, sino un valor $V^*>I$ (McDONALD y SIEGEL, 1986; DIXIT y PINDYCK, 1994), que la figura muestra para dos niveles de volatilidad y cuyas implicaciones para la inversión agregada, el retraso óptimo y la histéresis, se estudian en el Tema 3.A.34 ([\authorfont{Ver Tema 3.A.34}]).

\imagenfit{opcion_valor.pdf}{Valor de la opción de invertir y VAN estático}{Elaboración propia a partir de DIXIT y PINDYCK (1994).}

Si bien muchos proyectos encierran opciones reales, estas son específicas de cada tipo de inversión, y presentar una teoría general queda fuera del alcance de este tema. En la práctica aparecen, sobre todo, tres tipos. La \textbf{opción de posponer} es característica de las industrias en que los proyectos exigen enormes inversiones y son muy sensibles a variables como los precios o los tipos de interés, como la explotación de petróleo: la empresa puede considerar óptimo retrasar el comienzo del proyecto hasta que las condiciones del mercado sean propicias. La \textbf{opción de crecer} aparece cuando los gestores deben elegir la capacidad con que responder a un posible aumento de la demanda: un proyecto previsto para un crecimiento muy alto expone al riesgo de una inversión excesiva, y uno pequeño, a altos costes si después hay que ampliar la capacidad; la opción permite comenzar con una planta pequeña y ampliarla si el mercado lo demanda. La \textbf{opción de abandonar} permite renunciar a la inversión en cualquier etapa del proyecto y recuperar parte de lo invertido.\footnote{%
    TRIGEORGIS (1996) añade la opción de reducir la escala, la de cambiar de factores o de productos (flexibilidad operativa) y las opciones compuestas de los proyectos por etapas, como la investigación y el desarrollo, en que cada fase da derecho a acometer la siguiente. El anexo de ejemplos numéricos valora una opción de este tipo en un proyecto agrícola.
} En suma, las opciones reales otorgan flexibilidad a los proyectos al permitir posponer, ampliar o abandonar una inversión, y están implícitas en casi cualquier proyecto.

Tienen, sin embargo, límites. La analogía con las opciones financieras descansa en que el proyecto podría replicarse con activos negociados, lo que rara vez ocurre: en la práctica se supone que el valor actual del propio proyecto, sin flexibilidad, es el mejor estimador de su valor de mercado (COPELAND y ANTIKAROV, 2001)\textcolor{orange}{\textsuperscript{[Probable]}}, y la volatilidad que se emplea es la de empresas del sector, que es solo una aproximación. Además, el valor de esperar supone que la oportunidad es exclusiva. Si un competidor puede adelantarse, esperar cuesta la cuota de mercado o la oportunidad misma, y la decisión pasa a ser un juego entre empresas en el que la competencia erosiona el valor de la opción y puede precipitar la inversión (GRENADIER, 2002; SMIT y TRIGEORGIS, 2004). Es aquí, y no en el árbol de decisión, donde entra la teoría de juegos. Por último, su uso formal es limitado: en la encuesta de GRAHAM y HARVEY (2001), en torno a una cuarta parte de los directores financieros declaraba incorporar opciones reales a la evaluación de proyectos\textcolor{orange}{\textsuperscript{[Probable]}}.

\subsubsection{Medir la incertidumbre en cualquier contexto: sensibilidad, escenarios y Montecarlo}
La otra forma de incorporar la información incompleta parte de la imposibilidad de anticipar correctamente todas las contingencias relevantes. En la práctica, los flujos de caja, las tasas de descuento e incluso la propia estructura del proyecto están sujetos a errores de estimación y a sucesos adversos no previstos. El problema no radica ya en la secuencia de decisiones, sino en la robustez del análisis ante desviaciones respecto del escenario central: cuánto pueden alejarse la rentabilidad y el valor del proyecto de lo previsto, y qué variables lo explican. Para abordarlo se recurre al análisis de sensibilidad, al de escenarios y a la simulación de Montecarlo, que evalúan cómo varían los resultados del proyecto ante perturbaciones en los supuestos iniciales y proporcionan así una medida más realista del riesgo de la inversión.

¿Por qué varios métodos, y no uno? Porque, a diferencia de los del apartado anterior, que miden cuánto vale reaccionar, estos responden a una misma pregunta, la dispersión del resultado, con una profundidad creciente: el análisis de sensibilidad varía una variable cada vez y no asigna probabilidades; el de escenarios combina varias de forma coherente; y la simulación de Montecarlo asigna a todas una distribución de probabilidad y obtiene la del resultado. Y se aplican en cualquier contexto: tanto al proyecto tal como se ha planteado, con un plan fijo, como al proyecto con flexibilidad, cuyo valor también depende de los supuestos.

\paragraph{Análisis de sensibilidad y de escenarios}
El \textbf{análisis de sensibilidad} permite identificar las variables críticas que influyen en el desempeño financiero de un proyecto, lo que mejora la gestión del riesgo y la toma de decisiones durante el proceso de inversión. Su propósito es reconocer los elementos que contribuyen al éxito o al fracaso de la inversión y medir cuánto lo hacen de manera marginal. Responde a preguntas como cuál es el escenario de costes que anula el VAN, cómo varía la TIR ante una caída de la demanda del $10\%$ o si los flujos de caja son sensibles a los precios. Para ello, se modifica una variable, o unas pocas, cada vez, manteniendo las demás en su valor central, lo que permite valorar con claridad el efecto de cada cambio sobre la viabilidad del proyecto.\footnote{%
    Es posible analizar la sensibilidad a dos o más variables a la vez, para examinar cómo su interacción afecta a la viabilidad del proyecto, pero el análisis es considerablemente más complejo que el de una variable y exige cuidado para obtener resultados fiables.
} Esa misma flexibilidad dificulta, sin embargo, su formulación general, porque las modificaciones pueden influir sobre el resultado de formas muy diversas, según la variable y según el criterio sobre el que se haga el análisis.

Para simplificar, puede hacerse sobre el VAN, teniendo en cuenta las variables de las que depende cada uno de sus componentes: los flujos de caja dependen de la demanda, de los costes, que varían con la cantidad producida, del gasto comercial y de la regulación; la tasa de descuento, del tipo sin riesgo y de la prima, $r=\gamma+\lambda$; y la inversión inicial, de su valor esperado y de un componente no esperado, los sobrecostes. Si, por ejemplo, una pequeña caída de la demanda hace negativo el VAN, el proyecto es muy sensible, y por tanto arriesgado, ante variaciones de la demanda. Una forma útil de expresarlo es el \textbf{punto muerto} de cada variable: el valor que anula el VAN, que dice cuánto margen tiene el proyecto frente a cada desviación. En el diagrama de tornado de la figura, cada barra recoge el intervalo del VAN al mover una variable entre sus valores pesimista y optimista; las que cruzan la línea de VAN nulo son las que pueden cambiar la decisión. El análisis de sensibilidad es, así, de vital importancia para una gestión adecuada del riesgo y, una vez tomada la decisión, el conocimiento de esos factores de riesgo resulta de gran utilidad para diseñar estrategias de cobertura ante cualquier eventualidad.\footnote{%
    Aunque, en muchas ocasiones, el peor escenario contemplado no es el más adverso que se produce en la práctica, como mostró el caso del Eurotúnel que vimos al tratar las limitaciones comunes de los criterios.
}

Tiene, sin embargo, dos limitaciones. La primera es que omite las correlaciones: al analizar por separado el efecto de cada variable, ignora las relaciones que suelen existir entre ellas.\footnote{%
    Por ejemplo, los cambios en el coste de los factores podrían estar asociados a cambios en el coste de la financiación o en la demanda.
} La segunda es que no proporciona probabilidades: las variaciones las fija el analista de forma discrecional, de modo que no dice cuán probable es que el proyecto destruya valor. El \textbf{análisis de escenarios} corrige la primera, porque mueve a la vez y de forma coherente las variables que van juntas (una recesión reduce la demanda y los precios y encarece la financiación) y calcula el VAN en unos pocos escenarios, típicamente pesimista, central y optimista; llevado al extremo, con escenarios adversos pero plausibles, es una prueba de resistencia. Pero sigue sin asignar probabilidades.\footnote{%
    El anexo de ejemplos numéricos analiza la sensibilidad de un proyecto a la tasa de descuento y al flujo de su último año.
}

\paragraph{Simulación de Montecarlo}
Para sortear las limitaciones del análisis de sensibilidad puede recurrirse a la \textbf{simulación de Montecarlo}, uno de los instrumentos más utilizados para modelizar y estimar en las ciencias sociales, y a menudo también en las ciencias puras. Es un método no determinista que aproxima problemas complejos y costosos de resolver con exactitud generando un gran número de realizaciones aleatorias.\footnote{%
    Para una introducción al método, \href{https://ocw.mit.edu/courses/6-0002-introduction-to-computational-thinking-and-data-science-fall-2016/resources/lecture-6-monte-carlo-simulation/}{Lecture 6: Monte Carlo Simulation (MIT OpenCourseWare)}; y, para su aplicación al VAN, \href{https://www.youtube.com/watch?v=L5uAPOE8lMM}{Monte Carlo Simulation to Assess Net Present Value (YouTube)}.
} HERTZ (1964) lo introdujo en la evaluación de proyectos: en lugar de un único valor para cada variable, el analista especifica su distribución de probabilidad y sus correlaciones, y obtiene la distribución del resultado. El procedimiento tiene cuatro pasos: (i) identificar las distribuciones de probabilidad que mejor se ajustan al comportamiento de las variables relevantes, como el precio de venta, los costes o la tasa de descuento, reuniendo toda la información disponible y con ayuda del análisis estadístico; (ii) generar, a partir de ellas, un gran número de sendas posibles de esas variables; (iii) calcular el VAN en cada uno de los escenarios simulados; y, (iv) analizar la distribución resultante y evaluarla en función de los objetivos de la empresa. A diferencia del análisis de sensibilidad, recoge el efecto conjunto de todas las variables, con sus correlaciones, y dice, por ejemplo, cuál es la probabilidad de que el proyecto no recupere la inversión (figura, panel b), lo que permite al analista centrar su atención en las variables que más impulsan el valor del proyecto.

La simulación no se limita al plan fijo. Si en cada fecha la empresa puede decidir, por ejemplo, abandonar el proyecto, la simulación puede incorporar esa regla: LONGSTAFF y SCHWARTZ (2001) mostraron cómo estimar en cada fecha, a partir de las sendas simuladas, el valor de continuar y decidir de forma óptima, lo que permite valorar opciones reales con varias fuentes de incertidumbre, para las que no hay fórmulas como la de BLACK y SCHOLES. Por eso cierra el bloque: es la herramienta que sirve en cualquier contexto.

Su aplicación tiene, no obstante, tres límites. El primero es práctico: la precisión del análisis depende de manera crucial de que las distribuciones de probabilidad y las correlaciones se ajusten a la realidad, y si no lo hacen los resultados pueden ser inexactos o sesgados; además, si el número de simulaciones es demasiado pequeño, la distribución obtenida puede no reflejar la verdadera, y las simulaciones grandes o complejas pueden ser costosas en tiempo y en cálculo, aunque existan programas especializados. El segundo es conceptual: qué tasa de descuento emplear. Si cada senda se descuenta con una tasa ajustada por el riesgo, el riesgo se cuenta dos veces, en la tasa y en la dispersión; si se descuenta al tipo sin riesgo, la distribución obtenida no es la de valores de mercado del proyecto. Por eso se recomienda usar la simulación para obtener los flujos esperados y descontarlos después con la tasa ajustada (BREALEY, MYERS y ALLEN, 2020), o, para valorar opciones, simular con probabilidades neutrales al riesgo y descontar al tipo sin riesgo.

El tercero es de fondo: qué riesgo importa. La dispersión del VAN mide el \textbf{riesgo total} del proyecto, pero los propietarios con carteras diversificadas solo exigen una prima por el riesgo que no pueden eliminar diversificando, el riesgo sistemático, que es el que recoge la tasa de descuento ([\authorfont{Ver Tema 3.B.23}]). El riesgo total importa, en cambio, cuando hay fricciones: si la financiación externa es cara, una caída de los fondos internos obliga a renunciar a inversiones rentables, y la empresa tiene motivos para comportarse como si fuera aversa al riesgo y para cubrirlo (FROOT, SCHARFSTEIN y STEIN, 1993); también importan los costes de insolvencia ([\authorfont{Ver Tema 3.B.3}]) y la aversión al riesgo de unos directivos que no pueden diversificar su empleo. Las preferencias ante el riesgo, que con mercados perfectos no afectan a la decisión, vuelven a importar cuando los mercados son imperfectos. En resumen, la simulación de Montecarlo es una herramienta útil para modelizar los escenarios y resultados que puede atravesar un proyecto y precisar sus riesgos, y, como el análisis de sensibilidad, permite reconocer los factores clave que impulsan su valor; pero la medida de ese riesgo que el mercado remunera, la tasa de descuento, es el objeto del bloque siguiente.

\section{Coste del capital: ¿a qué tasa se descuenta un proyecto?}
\puntosclave{
    El coste del capital es el coste de oportunidad del proyecto, la rentabilidad de inversiones de riesgo comparable, $r=\gamma+\lambda$. De la precisión con que se estime depende la fiabilidad del VAN como criterio de selección de proyectos.

    Es una propiedad del proyecto, no de la empresa: el WACC de la empresa solo debe emplearse si el proyecto tiene un riesgo similar al de la compañía. Una tasa única lleva a rechazar proyectos seguros que crean valor y a aceptar proyectos arriesgados que lo destruyen (falacia del WACC).

    El coste de los recursos propios no se observa: se infiere con un modelo de valoración (CAPM: tipo sin riesgo del plazo de los flujos, beta y prima de mercado) o de la rentabilidad implícita en el precio (GORDON y SHAPIRO). Los métodos no tienen por qué coincidir.

    El coste de la deuda es el esperado por los acreedores en la deuda nueva, a valor de mercado y después de impuestos.

    El WACC del proyecto se construye con empresas comparables: se desapalanca su beta y se reapalanca con la estructura objetivo del proyecto. Si el endeudamiento cambia con el tiempo, el APV separa el valor de la inversión del de su financiación.

    En la práctica, las empresas usan tasas únicas y tasas mínimas muy superiores al WACC, que reflejan más sus restricciones de capital y de gestión que el riesgo.
}


El primer bloque daba por conocida la tasa de descuento. Este responde a la pregunta que dejó abierta: ¿qué tasa es la apropiada para un proyecto? 

La tasa de descuento, o coste del capital, es el coste de oportunidad de comprometer fondos en el proyecto: la rentabilidad que sus financiadores podrían obtener en otra inversión de riesgo comparable. Por eso tiene los dos componentes que ya conocemos:

\eqblock{
  \begin{aligned}
    r=\gamma+\lambda
  \end{aligned}
}{Coste del capital: ¿cuál es el coste de oportunidad del capital?}

Donde $\gamma$ es la rentabilidad del activo sin riesgo, que remunera el tiempo, y $\lambda$, una prima que remunera el riesgo (es la descomposición del coste del capital que se sigue de cualquier modelo de valoración de activos con un activo sin riesgo). Sin la prima, cualquier activo sin riesgo con el mismo rendimiento sería una alternativa mejor que el proyecto, y esa ventaja quedaría oculta.

La consecuencia es que el coste del capital es una propiedad del proyecto, no de quien lo financia, y que la comparación entre alternativas depende, esencialmente, de su \textbf{riesgo}.\footnote{%
  Pensemos en dos alternativas de una empresa: mejorar el precario sistema de almacenamiento de su planta A, lo que evitaría el deterioro de sus existencias, o construir una nueva planta B, anticipándose a un posible aumento de la demanda. La mejora del almacenamiento apenas tiene riesgo, por lo que basta con que su VAN, descontado al tipo sin riesgo, $\gamma$, sea positivo; la nueva planta depende de una demanda incierta, y su VAN debe ser positivo descontado a $\gamma+\lambda$. Si la empresa usara la misma tasa para ambas, infravaloraría la primera y sobrevaloraría la segunda.
}

El objeto de este bloque es medir esa tasa y discutir los problemas teóricos y prácticos que plantea, que proceden, sobre todo, de la falta de datos precisos, de la elección del tipo sin riesgo, de la estimación de la prima y de la estructura de capital que debe suponerse. 

Como la prima de un proyecto no se observa, se infiere de la rentabilidad que exigen quienes financian activos de riesgo parecido, accionistas y acreedores. A partir de su media ponderada obtendremos el \textbf{coste medio ponderado del capital} (WACC, por sus siglas en inglés): la rentabilidad mínima que una empresa debe obtener para compensar a sus inversores.\footnote{%
    Entendemos el capital de forma similar al factor $K$ de la teoría microeconómica: las fuentes de financiación de la actividad de la empresa. Distinguimos únicamente entre recursos propios (\textit{equity}) y deuda, aunque existen otras fuentes, como la deuda bancaria, los beneficios no distribuidos o las acciones preferentes; por simplicidad, y sin pérdida de generalidad, las agrupamos en esas dos en sentido amplio ([\authorfont{Ver Tema 3.B.3}]).
}\textsuperscript{,}\footnote{%
    \textbf{OJO.-} Como advierten VERNIMMEN, QUIRY y LE FUR (2022), el coste del capital, que mide el WACC, no es la rentabilidad del capital, que miden indicadores como la rentabilidad financiera (ROE) o la relación entre el precio y el beneficio por acción (PER). El primero recoge la rentabilidad exigida o esperada por los inversores, \textit{ex ante}; el segundo, la rentabilidad obtenida, \textit{ex post}. En mercados sin fricciones, el arbitraje tiende a igualarlos, pero son conceptos distintos, y usar los segundos como medida del coste del capital, como hacen a menudo los analistas, solo es una aproximación que introduce ruido y sesgo en la estimación.
}

\subsection{El coste de oportunidad del proyecto: la falacia del WACC}
La regla que se sigue de lo anterior es sencilla: cada proyecto se descuenta con el coste del capital que corresponde a su riesgo. 

El WACC de la empresa es la tasa adecuada si y solo si el proyecto tiene el mismo riesgo que la empresa en su conjunto, como una ampliación de su actividad habitual. Si la empresa aplica su WACC a todos los proyectos, se equivoca de forma sistemática. Para verlo, representemos la rentabilidad que el mercado exige a cada ivel de riesgo sistemático $\beta$ mediante la línea del mercado de valores (SML)

\imagenfit{falacia_wacc.pdf}{La falacia del WACC: una tasa única para proyectos de distinto riesgo}{Elaboración propia a partir de BREALEY, MYERS y ALLEN (2020).}

Un proyecto seguro, como S, que rinde más de lo que el mercado exige a su riesgo pero menos que el WACC de la empresa, se rechaza aunque cree valor; y uno arriesgado, como R, que rinde más que el WACC pero menos de lo que exige su riesgo, se acepta aunque pueda destruirlo. De esta forma, resulta evidente que la empresa que usa una tasa única invierte de más en sus actividades arriesgadas y de menos en las seguras: acaba siendo más arriesgada de lo que sus propietarios eligieron.\footnote{%
  No es un error teórico sin consecuencias. KRÜGER, LANDIER y THESMAR (2015) mostraron que las empresas diversificadas invierten más en las divisiones cuyo riesgo sistemático supera al de su actividad principal y menos en las que lo tienen inferior, como si descontaran todas con la misma tasa; el sesgo es mayor cuando el coste de equivocarse es menor, como en las divisiones pequeñas, lo que sugiere que se trata de una regla práctica más que de una decisión meditada.
} 

El remedio no es fácil, pues consiste en estimar, como veremos, el coste del capital de cada proyecto a partir del de empresas \textbf{comparables}, que se dedican solo a la actividad del proyecto, corrigiendo las diferencias de endeudamiento, como veremos al tratar el WACC del proyecto.

Dos precisiones completan la regla. La primera es que \textbf{el riesgo relevante es el sistemático}, como vimos al tratar la simulación de Montecarlo: un proyecto muy incierto puede tener una beta baja si su riesgo es específico, como una exploración petrolífera cuyo éxito no depende de la marcha de la economía, y su coste del capital es entonces bajo, aunque su dispersión sea grande. La segunda es que los \textbf{riesgos específicos de un proyecto} (un retraso, un sobrecoste) \textbf{se recogen mejor en los flujos esperados que en la tasa de descuento}: añadir puntos a la tasa para cubrirlos penaliza más a los flujos lejanos, sin que haya razón para ello (BREALEY, MYERS y ALLEN, 2020).

\subsection{Recursos propios: ¿cuánto cuestan los recurss propios?}
El coste de los recursos propios es la parte más compleja de la estimación, porque la rentabilidad que esperan los accionistas no se observa: hay que inferirla. 

Para ello se recurre a dos métodos, un modelo de valoración de activos, como el CAPM o la teoría de valoración por arbitraje (APT), o la rentabilidad implícita en el precio de la acción.

\subsubsection{Modelo de valoración de activos de capital (CAPM)}
El modelo de valoración de activos de capital de explica la relación entre riesgo y rentabilidad que existe en el mercado ([\authorfont{Ver Tema 3.B.23}]). 

Esta establece que, si todos los inversores toman sus decisiones según la media y la varianza de la rentabilidad, comparten el mismo horizonte de inversión y las mismas distribuciones de probabilidad y no hay fricciones en los mercados de capitales, en equilibrio existe una relación lineal entre el riesgo sistemático de un activo y su rentabilidad esperada. 

Aplicado a las acciones de la empresa, el CAPM da el coste de los recursos propios:

\eqblock{
  \begin{aligned}
    \mathbb E[r_E]=\gamma+\beta_E\,(\mathbb E[r_M]-\gamma)
  \end{aligned}
}{Recursos propios: ¿cuánto cuestan? Coste del capital: Modelo CAPM}

Donde $\beta_E$ mide el riesgo sistemático de las acciones, su sensibilidad a la rentabilidad del mercado, y $\mathbb E[r_M]$ es la rentabilidad esperada de la cartera de mercado, que no es observable.\footnote{%
    La derivación del modelo se estudia en el Tema 3.B.23 ([\authorfont{Ver Tema 3.B.23}]). \textbf{OJO.-} Bajo el CAPM, el riesgo relevante es el sistemático o no diversificable: es el único que el mercado remunera, porque el riesgo no sistemático puede eliminarse diversificando la cartera.
} 

\textcolor{magenta}{Es decir, la rentabilidad esperada de cualquier activo es la del activo sin riesgo más una prima, que es la beta del activo multiplicada por la prima de riesgo del mercado, $\lambda_M=\mathbb E[r_M]-\gamma$: en la notación del bloque, $\lambda=\beta_E\,\lambda_M$. 

El modelo se aplica por analogía al proyecto, con dos cautelas. Con la beta de las acciones, el CAPM da el coste de los recursos propios, el que entra en el WACC; con la beta de los activos, $\beta_A$, que mide el riesgo del negocio sin el efecto del endeudamiento, da el coste del capital de los activos, que es el de la empresa si no tuviera deuda. Y la beta que importa es la del proyecto, no la de la empresa.

$\beta_E$ es del activo o las acciones? En la fórmula se describe como del activo, pero luego se llama a la beta del activo $\beta_A$. Vigila la notación, NO PUEDE SER CONFUSA}

Aplicarlo exige estimar tres magnitudes. El \textbf{tipo sin riesgo} debe corresponder al plazo de los flujos que se descuentan: para un proyecto a largo plazo, el rendimiento de la deuda pública a un plazo parecido en la moneda de los flujos.\footnote{%
    En la zona del euro, suele tomarse el rendimiento de la deuda pública alemana, por ser la de menor riesgo de crédito; usar la de un país con prima de riesgo soberano incorpora ya parte de esa prima en $\gamma$\textcolor{orange}{\textsuperscript{[Probable]}}.
} La \textbf{beta} se estima con datos históricos, regresando la rentabilidad de la acción sobre la del mercado, y para las empresas cotizadas puede obtenerse hoy directamente de fuentes de libre acceso; pero la estimación de una empresa aislada es muy imprecisa, por lo que conviene usar la media de un sector, y las betas extremas tienden a acercarse a uno con el tiempo, lo que llevó a BLUME (1971) a proponer ajustarlas hacia la unidad.\footnote{%
    La corrección más extendida, que aplican algunos proveedores de datos, pondera la beta estimada por dos tercios y la unidad por un tercio\textcolor{orange}{\textsuperscript{[Probable]}}.
} La \textbf{prima de riesgo del mercado} es la magnitud más discutida. Puede estimarse con la media histórica del exceso de rentabilidad de las acciones sobre la deuda pública, que depende del periodo y de si se usa la media aritmética o la geométrica, o como prima implícita en los precios actuales, despejándola de un modelo de descuento de dividendos.\footnote{%
  Su nivel histórico es, además, difícil de explicar con los modelos habituales de comportamiento del inversor: es el enigma de la prima de las acciones de MEHRA y PRESCOTT (1985) ([\authorfont{Ver Tema 3.B.23}]).
}

\paragraph{Balance: ¿por qué es o no adecuado el CAMP?}
La principal fortaleza del método es que puede aplicarse con independencia de la política de dividendos de la empresa y que determina de forma explícita el nivel de riesgo. 

Como punto débil, el papel que tienen la calidad y la cantidad de los datos es evidente, y el empleo de datos históricos para situaciones futuras supone una cierta continuidad, o ergodicidad, hipotética. 

Además, la evidencia empírica añade una objeción de fondo: la relación entre la beta y la rentabilidad media de las acciones es más plana de lo que predice el modelo (FAMA y FRENCH, 1992), lo que ha llevado a modelos con varios factores de riesgo, \textcolor{magenta}{como la APT de ROSS

(desarrollar en anexo, aplicado al coste de los recursos propios)
} (1976) ([\authorfont{Ver Tema 3.B.23}]). 

Aun así, el CAPM sigue siendo el método más usado: en la encuesta de GRAHAM y HARVEY (2001), lo empleaban siempre o casi siempre tres de cada cuatro directores financieros que estimaban el coste de los recursos propios.

\subsubsection{La rentabilidad implícita en el precio: el modelo de GORDON y SHAPIRO}
El segundo método parte del modelo fundamental de valoración de acciones de GORDON y SHAPIRO (1956), según el cual el valor teórico de una acción está determinado por los flujos de caja que generará en el futuro, los dividendos.

Hay que especificar, por tanto, las variables que definen esos flujos y descontar esa corriente, en principio ilimitada, hasta el presente; si los dividendos crecen a una tasa constante $g<r_E$, la suma se simplifica, lo que es de gran utilidad para aislar el coste de los recursos propios:\footnote{%
    El precio de la acción es $p=\sum_{t=1}^{\infty}d_t/(1+r_E)^t$, donde $d_t$ es el dividendo del año $t$; con $d_t=d_1(1+g)^{t-1}$, la suma es la de una renta perpetua creciente, $p=d_1/(r_E-g)$. Los dividendos y la tasa deben expresarse ambos en términos nominales o ambos en términos reales: si se usan dividendos reales, sin inflación, $r_E$ es el coste real. 
    
    La misma expresión, con el crecimiento que generan los beneficios retenidos, es la del modelo de WALTER y GORDON que se estudia con la política de dividendos ([\authorfont{Ver Tema 3.B.3}]).
}

$$p=\frac{d_1}{r_E-g}\quad\Longrightarrow\quad r_E=\frac{d_1}{p}+g.$$

El coste de los recursos propios es, así, la rentabilidad por dividendo más su tasa de crecimiento. El origen del valor de la acción es el conjunto de pagos futuros.\footnote{%
\begin{cita}[W. BUFFETT][1994][Annual report to shareholders of Berkshire Hathaway]
    We define intrinsic value as the discounted value of the cash that can be taken out of a business during its remaining life. Anyone calculating intrinsic value necessarily comes up with a highly subjective figure that will change both as estimates of future cashflows are revised and as interest rates move. Despite its fuzziness, however, intrinsic value is all-important and is the only logical way to evaluate the relative attractiveness of investments and business.
\end{cita}
} 

El método es aplicable a las empresas cotizadas, cuyos dividendos y precio de mercado se observan directamente, aunque la tasa $g$ no: se estima con datos históricos de la empresa o con las previsiones publicadas por los analistas. Ese es su punto débil, porque el resultado depende mucho de $g$, y el crecimiento constante es una aproximación pobre para las empresas que no reparten dividendos o cuyo crecimiento va a cambiar.}


\subsection{Deuda: ¿cuál es el coste de contraer la obligación?}
El coste de la deuda es más sencillo de estimar, porque los acreedores tienen un derecho fijo y su rentabilidad se observa en el mercado. 

Es el coste en que la empresa incurre para financiarse con deuda, y lo que importa es el de la deuda \textbf{nueva}, no el tipo que figura en los préstamos antiguos. 

Puede obtenerse por tres vías. En primer lugar, y de manera directa, observando directamente el \textbf{tipo de interés} que la empresa debe pagar por nuevos préstamos. En segundo lugar, de manera indirecta, si la empresa ha emitido bonos que siguen en circulación, por su \textbf{rendimiento al vencimiento} ([\authorfont{Ver Tema 3.B.24}]). Por último, o si no los tiene y quiere hacerlo de forma indirecta, sumando al tipo sin riesgo el diferencial que el mercado exige a las empresas de su misma calificación crediticia: por \textbf{comparación circulante}.\footnote{%
  \textbf{NOTA.-} Podría estimarse también con el CAPM, con la beta de la deuda, pero rara vez se hace, porque esa beta es pequeña y difícil de medir.
}

\textcolor{magenta}{(Se pueden ilustrar una comparativa - insights de las tres vías?)}

Dos precisiones. La primera es que el rendimiento al vencimiento es la rentabilidad \textbf{prometida}, no la esperada: si la deuda puede impagarse, la rentabilidad que esperan los acreedores es menor, porque descuenta la probabilidad de impago y la pérdida en ese caso. Para la deuda de buena calidad la diferencia es pequeña, pero para la de alto riesgo es importante, y usar el rendimiento prometido sobrestima el coste de la deuda.\footnote{%
    El diferencial sobre el tipo sin riesgo depende de la solvencia del prestatario, y en particular de su patrimonio neto, que cae en las recesiones: es la prima de financiación externa que estudia el Tema 3.B.3 y que, en el plano macroeconómico, origina el acelerador financiero ([\authorfont{Ver Tema 3.B.3, Tema 3.A.34, Tema 3.A.37}]).
} 

La segunda es que los intereses son fiscalmente deducibles, de modo que el coste de la deuda para la empresa es el coste después de impuestos, $r_D(1-T)$, donde $T$ es el tipo marginal del impuesto sobre sociedades.\footnote{%
    \textbf{NOTA.-} En España, la deducción de los gastos financieros netos está limitada al $30\%$ del beneficio operativo del ejercicio, con un mínimo deducible de $1$ millón de euros (art. $16$ de la Ley 27/2014, del Impuesto sobre Sociedades). Por encima de ese límite, el ahorro fiscal de la deuda desaparece y el coste efectivo de la deuda adicional es $r_D$, no $r_D(1-T)$.
}

\subsection{Coste del capital promedio: ¿cuándo cuesta financiar el capital de un proyecto?}
Con el coste de cada fuente, el coste medio ponderado del capital es el promedio de esos costes, ponderados por la composición del capital (MODIGLIANI y MILLER, 1963):\footnote{%
  Donde $E$ es el valor de los recursos propios, $D$ el de la deuda y $V=E+D$; sin impuestos, desaparece el factor $(1-T)$. 
}

\eqblock{
  \begin{aligned}
    r_{\text{WACC}}=r_E\,\frac{E}{V}+r_D\,(1-T)\,\frac{D}{V}
  \end{aligned}
}{}

Calcularlo a partir del coste de las principales fuentes de financiación exige disponer de datos y menos hipótesis, que estimar directamente el riesgo de los activos: reduce el riesgo de una ceguera inducida por la teoría. Sin embargo, el coste de los recursos propios sigue necesitando uno de los modelos del apartado anterior. 

Además, debemos volver a recordar que las ponderaciones deben ser respecto a la estructura de capital \textbf{objetivo} del proyecto, la que se mantendrá durante su vida, y medirse a \textbf{valor de mercado}: el valor de los recursos propios es el precio de la acción multiplicado por el número de acciones en circulación, y el de la deuda, el valor actual de sus pagos al tipo de mercado, no su importe original, que solo es una aproximación aceptable si los tipos de interés y la solvencia de la empresa apenas han cambiado desde su emisión.

\subsubsection{Riesgo de los activos: desapalancar y reapalancar la beta}
\textcolor{magenta}{Esto último nos deja con un amargo resultado: aplicar el WACC a un proyecto exige conocer la beta de sus recursos propios, que no se observa. 

Para ello, y como anticipamos, se pueden utilizar métodos comparativos para obtenerla a partir de la de empresas comparables, con un procedimiento que separa el riesgo del negocio del que añade el endeudamiento. 

Para empezar, debemos encontrar la beta de los activos ($\beta_A$), que mide el riesgo sistemático del negocio. Este riesgo sistemático del negocio es equivalente al de una cartera formada por los recursos propios y la deuda de la empresa, ponderados por su peso en el valor, porque la beta de una cartera es la media ponderada de las betas de los activos que la componen:\footnote{%
    Despejando, $\beta_A=(\beta_E+\beta_D\,D/E)/(1+D/E)$. La beta de los activos es la que el Tema 3.B.3 denota $\beta_U$, la de las acciones de la empresa si no tuviera deuda, y $r_A$, el coste del capital de los activos, es el coste del capital sin deuda, $r_U$.
}
\eqblock{
  \begin{aligned}
    \beta_A=\beta_E\,\frac{E}{V}+\beta_D\,\frac{D}{V}
  \end{aligned}
}{}

Si la deuda es poco arriesgada, lo que suele ocurrir cuando el endeudamiento es moderado y la empresa es solvente, su beta será casi nula, y se obtiene $\beta_A=\beta_E/(1+D/E)$. 

No obstante, si los intereses son deducibles y la empresa mantiene constante el importe de su deuda, el ahorro fiscal reduce el riesgo de los propietarios, y la relación pasa a ser la de HAMADA (1972):\footnote{%
    Es la proposición II de MODIGLIANI y MILLER expresada en betas ([\authorfont{Ver Tema 3.B.3}]). Supone una deuda sin riesgo y de importe constante; si la empresa ajusta su deuda para mantener constante la proporción $D/V$, el ahorro fiscal tiene el mismo riesgo que los activos y la corrección por impuestos desaparece, de modo que vuelve a aplicarse $\beta_A=\beta_E/(1+D/E)$\textcolor{orange}{\textsuperscript{[Probable]}}.
} $\beta_A=\frac{\beta_E}{1+(1-T)\,D/E}.$

Por tanto, el procedimiento consiste en tomar las betas de las acciones de empresas que se dedican solo a la actividad del proyecto; \textbf{desapalancarlas} con sus propias ratios de endeudamiento para obtener la beta de los activos; promediarlas, lo que reduce el error de estimación; \textbf{reapalancar} la beta media con la estructura objetivo del proyecto; y, con ella, estimar el coste de los recursos propios con el CAPM y el WACC del proyecto. 

Aplicado a la beta de los activos, el CAPM da directamente el coste del capital de los activos, $r_A$, que coincide con el WACC si no hay impuestos.

\subsubsection{¿Depende el WACC de la estructura de capital?}
Que el coste de los recursos propios y el de la deuda dependan de la estructura financiera es una pregunta fundamental, que se estudia con detalle en el Tema 3.B.3 ([\authorfont{Ver Tema 3.B.3}]) y que es uno de los puntos de mayor divergencia entre la teoría y el análisis empírico. Si dependen, el cálculo del WACC incorpora dos variables endógenas, los dos costes; si no, el WACC no es sensible a los cambios en la estructura financiera. La \textbf{visión tradicional} sostiene que, como la deuda es más barata que los recursos propios, endeudarse reduce primero el WACC, pero que, a partir de cierto punto, el mayor riesgo eleva la prima que exigen accionistas y acreedores y el WACC vuelve a subir, de modo que existe una estructura que lo minimiza (panel a). Para MODIGLIANI y MILLER (1958), en cambio, con un mercado de capitales perfecto y perfecta fungibilidad del dinero, el coste del capital depende exclusivamente de la capacidad del proyecto para generar valor y de su riesgo, cualquiera que sea la estructura con que se financie: el valor de una empresa es el valor de mercado de los flujos que generan sus activos, y la rentabilidad que exigen los accionistas sube con el endeudamiento lo justo para que el WACC permanezca constante (panel b).

\imagenfit{wacc_estructura.pdf}{¿Depende el coste del capital de la estructura de capital?}{Elaboración propia a partir de DURAND (1952) y MODIGLIANI y MILLER (1958). Curvas ilustrativas.}

Con impuestos, el ahorro fiscal de los intereses hace que el WACC disminuya con el endeudamiento (MODIGLIANI y MILLER, 1963), hasta que los costes de insolvencia lo compensan. Para valorar proyectos, la consecuencia práctica es que el efecto de la financiación entra en el WACC a través del coste de la deuda después de impuestos, lo que solo es correcto si la estructura de capital se mantiene constante a lo largo del proyecto. Cuando no es así (una compra apalancada que reduce su deuda con los años, un proyecto con un préstamo subvencionado o con una financiación que se amortiza según un calendario), es preferible el \textbf{valor actual ajustado} (APV) de MYERS (1974):
$$APV=VAN_{\text{base}}+VA(\text{efectos de la financiación}),$$
donde el VAN base es el del proyecto financiado solo con recursos propios, descontado a $r_A$, y el segundo término suma el valor actual de los efectos de la financiación, como el ahorro fiscal de los intereses, las subvenciones o los costes de emisión. El APV separa lo que el WACC mezcla en la tasa: el valor que crea la inversión y el que crea, o destruye, su financiación, como hacía la separación de FISHER con la que empezamos.

\subsection{La práctica: tasas únicas, tasas mínimas y otros problemas}
La práctica se aleja de estas recomendaciones en dos direcciones. La primera ya la hemos visto: muchas empresas aplican una tasa única a todos sus proyectos; en la encuesta de GRAHAM y HARVEY (2001), más de la mitad de los directores financieros declaraba usar siempre o casi siempre el coste del capital de la empresa en su conjunto, más que el de cada proyecto\textcolor{orange}{\textsuperscript{[Probable]}}. La segunda es que las tasas mínimas de rentabilidad que exigen a los proyectos superan con mucho su WACC: JAGANNATHAN, MATSA, MEIER y TARHAN (2016) encontraron que la diferencia es mayor en las empresas con más restricciones de capital y de capacidad de gestión, y no en las que afrontan más riesgo\textcolor{orange}{\textsuperscript{[Probable]}}. La tasa mínima actúa, así, como un mecanismo de racionamiento: recoge el coste de oportunidad de unos fondos y de una capacidad de gestión escasos, como el índice de rentabilidad del primer bloque, y quizá el valor de la opción de esperar (McDONALD, 2000).

Otros dos problemas son frecuentes. Uno es la \textbf{prima de riesgo país} de los proyectos en economías emergentes: suele sumarse a la tasa el diferencial de la deuda soberana del país, pero, si el riesgo del país es diversificable para un inversor internacional, debería recogerse en los flujos esperados (la probabilidad de una expropiación, de una devaluación o de una crisis) y no en la tasa\textcolor{orange}{\textsuperscript{[Probable]}}. El otro es la \textbf{coherencia}: los flujos nominales se descuentan con tasas nominales y los reales con tasas reales, como vimos al empezar, y los flujos en una moneda, con la tasa de esa moneda; mezclar unos y otros, por ejemplo flujos reales con un WACC nominal, sesga el VAN tanto más cuanto mayor es la inflación.

\anexo{Ejemplos numéricos}\label{anx:ejemplos}
Este anexo reúne los ejemplos numéricos del primer bloque. Van fuera del cuerpo porque el bloque razona con un ejemplo en letras, que permite cambiar el orden de las magnitudes y ver cuándo cambia la decisión; aquí se comprueban esas conclusiones con cifras concretas.

\subsection{Valor actual neto e índice de rentabilidad}
Dos proyectos, A y B, son mutuamente excluyentes, y la empresa dispone de un presupuesto de $80$ mil euros. Ambos duran cuatro periodos (por ejemplo, cuatro años) y se descuentan al $8\%$, que incluye un tipo sin riesgo del $2\%$ y una prima de $6$ puntos (el riesgo se introduce en el segundo apartado del bloque). Sus flujos de caja netos, en miles de euros, son:

\begin{center}
\begin{tabular}{|p{0.25\linewidth}|p{0.25\linewidth}|p{0.25\linewidth}|}
\hline
\textbf{Año} & \textbf{Proyecto A} & \textbf{Proyecto B} \\
\hline
0 & $-50$ & $-80$ \\
\hline
1 & $20$ & $30$ \\
\hline
2 & $22$ & $32$ \\
\hline
3 & $25$ & $36$ \\
\hline
4 & $30$ & $40$ \\
\hline
\end{tabular}
\end{center}

Sus valores actuales netos, en miles de euros, son
$$VAN_A=-50+\frac{20}{1,08}+\frac{22}{1,08^2}+\frac{25}{1,08^3}+\frac{30}{1,08^4}\approx 29,28,\qquad VAN_B=-80+\frac{30}{1,08}+\frac{32}{1,08^2}+\frac{36}{1,08^3}+\frac{40}{1,08^4}\approx 33,19.$$
Al $8\%$, los dos tienen un VAN positivo, de modo que cualquiera de ellos aumentaría el valor de la empresa. Sus índices de rentabilidad son
$$R_A=\frac{29,28}{50}\approx 0,586,\qquad R_B=\frac{33,19}{80}\approx 0,415:$$
cada euro invertido en A crea unos $59$ céntimos de valor, y cada euro invertido en B, unos $41$. A tiene mejor índice, pero, como los proyectos son excluyentes e indivisibles, elegir A deja sin invertir $30$ mil euros del presupuesto, que solo rendirían el tipo de interés. La empresa debe elegir B, que crea $3,92$ mil euros más de valor, salvo que pueda invertir esos $30$ mil euros en otro proyecto con un VAN superior a esa diferencia, es decir, con un índice de rentabilidad superior a $3,92/30\approx 0,13$.

Dos casos breves ilustran lo mismo. Un proyecto con un VAN de $10$ euros y una inversión de $50$ tiene un índice de $0,2$: crea $20$ céntimos de valor por cada euro invertido. Y un proyecto con una inversión y un VAN de $2$ euros (índice $1$) es peor, si los proyectos son excluyentes, que otro que exige $100$ euros y tiene un VAN de $60$ (índice $0,6$).

\subsection{Tasa interna de retorno}
Un proyecto exige una inversión de $1$ mil euros y genera $600$ y $700$ euros al final del primer y del segundo año. Su TIR, en miles de euros, resuelve
$$0=-1+\frac{0,6}{1+TIR}+\frac{0,7}{(1+TIR)^2},$$
una ecuación de segundo grado en $x=1/(1+TIR)$ cuya raíz positiva es $x\approx 0,841$, de modo que $TIR\approx 18,9\%$. El proyecto se acepta si el coste del capital es inferior a esa tasa.

La bomba de LORIE y SAVAGE (1955) exige $1,6$ mil dólares y adelanta al primer año $10$ mil dólares de petróleo que, sin ella, se extraerían en el segundo: sus flujos son $-1,6$, $+10$ y $-10$ mil dólares. Su VAN se anula con tipos del $25\%$ y del $400\%$, es negativo fuera de ese intervalo y positivo dentro de él. Un proyecto con flujos $+1$, $-3$ y $+2,5$ no tiene TIR: su VAN es positivo a cualquier tipo de descuento, con un mínimo de $0,1$ en torno al $67\%$.

En el ejemplo del índice de rentabilidad, $TIR_A\approx 30,3\%$ y $TIR_B\approx 24,5\%$, de modo que la TIR prefiere A. El proyecto diferencial, B menos A, tiene flujos $-30$, $10$, $10$, $11$ y $10$ mil euros, y su TIR, la tasa de FISHER, es de un $13,7\%$. Al $8\%$, por debajo de esa tasa, B tiene mayor VAN ($33,19$ frente a $29,28$ mil euros), y la TIR ordena mal; al $15\%$, por encima, A tiene mayor VAN ($17,62$ frente a $16,82$), y ambos criterios coinciden.

\subsection{Periodo de recuperación}
Una empresa estudia instalar una máquina que exige una inversión de $100$ mil unidades monetarias (UM) y genera unos flujos de caja de $30$ mil UM al año durante cinco años. Los flujos acumulados al final de cada año son $30$, $60$, $90$, $120$ y $150$ mil UM, de modo que la inversión se recupera durante el cuarto año: si los flujos se reparten de forma uniforme a lo largo del año, el payback es de $3+10/30\approx 3,33$ años, que es también $I_0/CF=100/30$. Con un plazo máximo de cinco años, el proyecto se acepta, siempre que su VAN y su TIR sean también satisfactorios. Descontando al $8\%$, los flujos descontados acumulados son $27,8$, $53,5$, $77,3$, $99,4$ y $119,8$ mil UM, de modo que el payback descontado es de unos $4,03$ años, y el VAN del proyecto, de $19,8$ mil UM.

\subsection{Árboles de decisión y opciones reales}
Un inversor debe decidir entre ampliar su planta actual o construir una nueva planta en otra localización, y en ambos casos puede optar por una inversión grande o pequeña. Cada alternativa tiene dos resultados posibles, con los VAN, en UM, y las probabilidades del árbol. Para determinar qué opción conviene, se estudia cada rama de atrás hacia delante. En la de la ampliación, una ampliación grande, que lleva a un nodo de azar, tiene un VAN de $150$ UM con probabilidad $0,5$ y de $-50$ con la misma probabilidad, es decir, un VAN esperado de $50$; una pequeña, de $0,5\cdot 100+0,5\cdot 50=75$. Puesto a ampliar, el inversor debería hacerlo a pequeña escala. En la rama de la nueva planta, una grande tiene un VAN esperado de $0,5\cdot 350+0,5\cdot(-350)=0$, y una pequeña, de $0,75\cdot 250+0,25\cdot 150=225$. En el nodo inicial, el inversor compara las mejores alternativas de cada rama, $75$ y $225$, y elige construir una nueva planta pequeña.

\begin{center}\begin{minipage}{\linewidth}\centering {\small\textbf{Árbol de decisión: ampliar la planta o construir una nueva (VAN en UM)}}\par\vspace{0.3em} \includegraphics{anexo_arbol_planta.pdf}\par\vspace{0.3em} {\small\textit{Fuente: }Elaboración propia.}\end{minipage}\end{center}

Un proyecto consiste en sembrar $500$ hectáreas de trigo. Antes de empezar, el inversor puede no invertir, con un valor nulo. Si invierte, desembolsa $1$ mil UM y obtiene $2,5$ mil UM si el clima es bueno, con probabilidad $0,8$, o pierde $0,2$ mil UM adicionales si es desfavorable, con probabilidad $0,2$, porque tiene que seguir cuidando el cultivo. Con un tipo de interés nulo, de modo que el valor intertemporal del dinero no cambia, el VAN esperado es, en miles de UM, $-1+0,8\cdot 2,5-0,2\cdot 0,2=0,96$, y, de no existir restricciones a la financiación o de otro tipo, el proyecto debería acometerse. Supongamos ahora que el inversor puede no ir al campo a cuidar el cultivo una vez que conoce el clima: tras invertir $1$ mil UM y ver que el clima es desfavorable, puede quedarse en casa y evitar la pérdida adicional. También puede hacerlo con buen clima, pero no le interesa. El árbol incorpora entonces dos nodos de decisión, y el VAN esperado del proyecto con flexibilidad es $-1+0,8\cdot 2,5=1$ mil UM. La diferencia, $0,04$ mil UM ($40$ UM), es el valor de la opción de no cuidar el cultivo: lo máximo que un inversor neutral al riesgo pagaría por ella.

\subsection{Análisis de sensibilidad}
Una inversión exige un desembolso inicial de $1$ mil euros y genera unos flujos de caja netos de $300$, $500$ y $600$ euros en los tres años siguientes. Con una tasa de descuento del $10\%$, su VAN, en miles de euros, es
$$VAN=-1+\frac{0,3}{1,1}+\frac{0,5}{1,1^2}+\frac{0,6}{1,1^3}\approx 0,137,$$
es decir, unos $137$ euros: el proyecto es rentable, porque el valor actual de sus flujos supera la inversión inicial. Para saber cuán sensible es a la tasa de descuento, se calcula su VAN con tasas mayores:

\begin{center}
\begin{tabular}{|p{0.35\linewidth}|p{0.35\linewidth}|}
\hline
\textbf{Tasa de descuento} & \textbf{VAN (euros)}\\
\hline
$10\%$ & $136,74$ \\
\hline
$13\%$ & $72,89$ \\
\hline
$14\%$ & $52,87$ \\
\hline
$15\%$ & $33,45$ \\
\hline
$16\%$ & $14,60$ \\
\hline
$17\%$ & $-3,71$ \\
\hline
$18\%$ & $-21,49$ \\
\hline
$19\%$ & $-38,77$ \\
\hline
\end{tabular}
\end{center}

El VAN decrece con la tasa de descuento: es positivo hasta el $16\%$ y negativo desde el $17\%$, porque la TIR del proyecto es del $16,79\%$. Como el coste de oportunidad es hoy del $10\%$, el riesgo ante subidas del tipo de interés es bajo en condiciones normales: tendría que subir $6,8$ puntos, casi un $70\%$ de su nivel actual, para que el proyecto dejara de ser rentable. El analista puede estudiar también la sensibilidad al flujo del tercer año. Aplicando la definición del VAN, el valor mínimo de ese flujo para que el proyecto sea viable es, en miles de euros,
$$CF_3\geq\left[1-\frac{0,3}{1,1}-\frac{0,5}{1,1^2}\right]1,1^3\approx 0,418,$$
de modo que el proyecto solo tiene un VAN positivo si $CF_3$ supera los $418$ euros: ese es el punto muerto del flujo del tercer año, el valor de corte o de referencia con el que comparar las previsiones.
