    % --- Imagen adaptativa ---
    % \imagenfit[alto]{fichero}{título}{fuente}: la imagen queda fija entre los párrafos donde se escribe (no flota), con su
    % título en el índice de figuras y la fuente debajo. Sin [alto]: las figuras PDF (TikZ) salen al tamaño con que se
    % dibujaron y solo se reducen si no caben en el ancho o en el alto de la página; las PNG/JPG ocupan el ancho del texto
    % hasta \imgaltoras del alto de la página. Con [alto] (fracción del alto de la página; [1] = página entera) se fija el
    % tamaño de esa imagen. Si no cabe en lo que queda de página, pasa entera a la siguiente.
    % \imagenfithorizontal{fichero}{título}{fuente}: la imagen en una página girada (apaisada), en ese mismo punto del texto.
    % Reglas: main/GUIA_TEMAS.md, §4 (Figuras).
    \newdimen\imgcap
    \newdimen\imgh
    \newdimen\imgw
    \newdimen\imgneed
    \newsavebox\imgcaja
    \newcommand{\imgaltoras}{0.30}% alto por defecto de las PNG/JPG (fracción del alto de la página)
    \makeatletter
    \newcommand{\imagenfit}[4][]{%
      \addvspace{10pt}%
      \par\begingroup
        \imgcap=3.5\baselineskip% alto aproximado de título + fuente
        \imgw=\linewidth
        \filename@parse{#2}\def\img@pdf{pdf}%
        \if\relax\detokenize{#1}\relax
          \ifx\filename@ext\img@pdf
            \sbox\imgcaja{\includegraphics{\detokenize{#2}}}% tamaño natural
            \ifdim\wd\imgcaja<\imgw \imgw=\wd\imgcaja \fi
            \imgh=\ht\imgcaja
          \else
            \imgh=\imgaltoras\textheight
          \fi
        \else
          \imgh=#1\textheight
        \fi
        \dimen@=\textheight \advance\dimen@ by -\imgcap
        \ifdim\imgh>\dimen@ \imgh=\dimen@ \fi
        \sbox\imgcaja{\includegraphics[width=\imgw,height=\imgh,keepaspectratio]{\detokenize{#2}}}%
        \imgneed=\ht\imgcaja \advance\imgneed by \imgcap
        \Needspace{\imgneed}%
        \noindent\begin{minipage}{\textwidth}
          \centering
          \captionof{figure}{\textemdash\ #3}\par\vspace{0.3em}% título arriba
          \usebox\imgcaja\par
          \vspace{0.3em}\small\textit{Fuente: }#4% fuente abajo
        \end{minipage}%
      \endgroup
      \par\addvspace{10pt}\RequestRestore
    }
    \newcommand{\imagenfithorizontal}[3]{%
      \begin{landscape}
        \centering
        \captionof{figure}{\textemdash\ #2}\par\vspace{0.3em}%
        \includegraphics[width=\linewidth,height=\dimexpr\textheight-4\baselineskip\relax,keepaspectratio]{\detokenize{#1}}\par
        \vspace{0.3em}{\small\textit{Fuente: }#3}%
      \end{landscape}%
    }
    \makeatother

    % \anexo{Título}: un anexo equivale a un \section (su título sale del mismo tamaño). Dentro se usan \subsection,
    % \subsubsection y \paragraph, nunca \section: se numeran I.1, I.1.1, I.1.1.1 y las páginas del anexo, I-1, I-2…
    % En el índice de contenido inicial solo figura el nombre del anexo. Sus apartados forman el índice del anexo, que sale
    % solo si el anexo los tiene, justo después de la introducción del anexo (antes del primer \subsection).
    % \anexo*{Título}: el mismo anexo, sin ese índice. Reglas: main/GUIA_TEMAS.md, §6.

    \newcommand{\anx@empezar}[2]{%
      \clearpage
      \refstepcounter{anexo}%
      \global\EnAnexotrue
      \setcounter{page}{1}\gdef\thepage{\theanexo-\arabic{page}}%
      \setcounter{section}{\numexpr\value{anexo}-1\relax}\stepcounter{section}% el anexo hace de \section
      \gdef\thesection{\Roman{section}}\gdef\theHsection{anexo.\Roman{section}}%
      \phantomsection
      \protected@xdef\leq@current@section{Anexo \theanexo.- \unexpanded{#2}}\gdef\leq@current@subsection{}% índice de ecuaciones
      \addtocontents{stc}{\protect\contentsline{section}{Anexo \theanexo.- #2}{}{\@currentHref}}% índice inicial: solo el nombre, sin página
      \addcontentsline{toc}{section}{Anexo \theanexo.- #2}% marcadores del PDF
      {\noindent\normalfont\Large\bfseries Anexo \theanexo.- #2\par}\nobreak\vspace{1.5ex \@plus .2ex}%
      #1\@ifundefined{anx@lista@\theanexo}{\global\anx@pendientefalse}{}%
    }
    % Índice del anexo: cada apartado escribe una línea en el fichero .atc (anexo, nivel, texto, página, enlace); al
    % empezar el documento se leen las de la compilación anterior y se guardan por anexo (dos pasadas, como cualquier índice).
    % En ese índice los apartados suben un nivel, porque el anexo hace de \section.
    \def\anx@entrada#1#2#3#4#5{\@ifundefined{anx@lista@#1}{\global\@namedef{anx@lista@#1}{}}{}%
      \expandafter\g@addto@macro\csname anx@lista@#1\endcsname{\contentsline{#2}{#3}{#4}{#5}}}
    \tcbuselibrary{breakable}
    \def\anx@indice{\ifanx@pendiente\global\anx@pendientefalse

            before upper={\parindent=0pt\parskip=2pt}]
          \small
          % números más anchos que en el índice inicial (I.1, I.1.1, I.1.1.1; páginas I-1)
          \def\@pnumwidth{2.6em}\def\@tocrmarg{3.4em}%
          \def\l@section##1##2{\addpenalty\@secpenalty\addvspace{1.0em \@plus\p@}\setlength\@tempdima{2.6em}%
            \begingroup\parindent\z@\rightskip\@pnumwidth\parfillskip-\@pnumwidth\leavevmode\bfseries
            \advance\leftskip\@tempdima\hskip-\leftskip##1\nobreak\hfil\nobreak\hb@xt@\@pnumwidth{\hss##2\kern-\p@\kern\p@}\par\endgroup}%

          \csname anx@lista@\theanexo\endcsname

      \fi}
    \def\anx@escribir#1#2#3{\ifEnAnexo

    % margen superior, así que no cambia las páginas del tema. Se enciende con \pagenumbering{arabic} y se apaga con

    \cs_new_protected:Npn \cab_guardar:n #1
      {
        \tl_gset:Nn \g_cab_titulo_tl {#1}
        \regex_replace_once:nnN { \A \s* Tema \s+ \S+ \s* \-\-? \s* } { } \g_cab_titulo_tl % sin «Tema X.Y.N --»
        \regex_replace_once:nnN { \.? \s* \c{\\} (\[ [^\]]* \])? \s* (\c{large} \s*)? } { .\  } \g_cab_titulo_tl % título. subtítulo
        \regex_replace_once:nnN { [\.\s]* \Z } { } \g_cab_titulo_tl % sin subtítulo: nada detrás
      }

\section{Crecimiento externo: ¿por qué crecer comprando y cuándo crea valor?}\label{b1}
\puntosclave{
    Comprar una empresa es una inversión. Crea valor para el comprador si la \textbf{sinergia}, lo que la unión vale por encima de las dos empresas por separado, supera la \textbf{prima}, lo que se paga por encima del valor de la empresa comprada.

    Se compra en lugar de construir cuando el crecimiento interno es lento, porque la capacidad de dirección y la financiación solo se acumulan con el tiempo, o cuando lo que se busca no se vende por separado: capacidades ligadas a una organización. Aliarse es la vía intermedia cuando la sinergia no exige integrarlo todo.

    La relación entre los negocios dice de dónde sale la sinergia: escala, alcance y poder de mercado en las operaciones horizontales; costes de transacción y doble margen en las verticales; solo ventajas financieras en los conglomerados. Diversificar, por sí solo, no crea valor para el accionista.

    Lo que gana el accionista puede ser valor creado o valor trasladado desde acreedores, trabajadores, clientes o Hacienda, y solo lo primero es una ganancia para la economía. Comprar capacidad que ya existe, además, no añade capacidad a la economía.

    El directivo compra si $\alpha(\hat S-\pi)+B>0$: está dispuesto a pagar hasta $\hat S+B/\alpha$, por encima de la sinergia, por los \textbf{beneficios privados} de comprar (tamaño, flujo de caja libre, menos riesgo para él) y por la \textbf{arrogancia}, que lleva a sobrestimar la sinergia, porque gana quien más la sobrestima.

    En promedio, las operaciones crean un valor modesto, del que el vendedor se lleva casi todo, y el comprador no gana nada. Le va mejor cuando tiene una $q$ alta y la empresa comprada una baja, cuando paga en efectivo, cuando compra una empresa no cotizada y cuando es pequeño. Cuánto de la ganancia es eficiencia y cuánto poder de mercado sigue en debate.

    Las operaciones llegan en \textbf{olas}, que siguen a shocks sectoriales cuando sobra financiación y la bolsa está alta, por reasignación del capital o por errores de valoración; los picos reúnen las peores operaciones.
}

Crecer es aumentar la escala de los recursos que la empresa controla. En las finanzas de la empresa no es un fin, sino un medio: el crecimiento solo crea valor si los recursos nuevos rinden más de lo que cuestan, y crecer con una rentabilidad inferior al coste del capital puede aumentar las ventas y el beneficio, pero a la vez destruir valor ([\authorfont{Ver Tema 3.B.1}]).

La empresa puede crecer invirtiendo en capacidad nueva, el \textbf{crecimiento interno} u orgánico, cuyas decisiones estudiamos al tratar la inversión ([\authorfont{Ver Tema 3.B.2}]), o incorporando la que ya existe en otra empresa, el \textbf{crecimiento externo} o inorgánico, sea comprándola o aliándose con ella.

Las operaciones de crecimiento externo se anuncian casi siempre con la misma promesa: \textbf{reducir costes}. Unir dos empresas permite cerrar instalaciones duplicadas, unificar las compras y los servicios centrales y prescindir de parte de la plantilla, y es la promesa que más cree el mercado, porque es más fácil de comprobar que la de vender más.\footnote{%
    En las mayores fusiones bancarias estadounidenses entre 1985 y 1996, el valor conjunto de las acciones del comprador y de la empresa comprada aumentó al anunciarse la operación, aunque menos que el valor actual de las mejoras que preveía la dirección, y ese aumento procedía sobre todo de los ahorros de costes previstos, no de los aumentos de ingresos (HOUSTON, JAMES y RYNGAERT, 2001).
} Pero el culto a la reducción de costes encierra dos cosas muy distintas. Un recorte puede ahorrar recursos que la economía empleará en otra parte, o puede limitarse a trasladar rentas a los accionistas desde los trabajadores, los proveedores o los clientes. De esa diferencia depende que lo que gana el accionista sea también una ganancia para la economía.

Como veremos, comprar una empresa es una inversión como cualquier otra, cuyo valor para el comprador es la sinergia que crea la unión menos la prima que paga por la empresa comprada. Que esa sinergia puede ser creación de valor o traslado de rentas. Que los directivos pueden comprar aunque no la haya, por motivos propios o por sobrestimarla. Y que la evidencia muestra que las operaciones crean, en conjunto, un valor modesto, del que casi todo se queda el vendedor, y que se concentran en olas cuyos picos reúnen las peores operaciones.

\subsection{Vías: ¿construir, comprar o aliarse?}
La empresa que quiere una capacidad que no tiene, sea una tecnología, una red comercial o una escala mayor, puede construirla, comprársela a quien ya la tiene o aliarse con él. Las tres vías llevan al mismo sitio con costes y plazos distintos, y la elección depende de cuál cuesta menos para el uso que se le quiere dar.

\imagenfit{b1_vias.pdf}{Tres vías hacia una misma capacidad}{Elaboración propia. Esquema ilustrativo.}

Supongamos que la empresa A diseña e instala almacenes y quiere ofrecerlos automatizados. La empresa B, mucho más pequeña, ha desarrollado los robots que lo permiten. A puede crear su propia división de robótica, comprar B o pactar con B un desarrollo conjunto, y cada vía tiene un coste distinto: una inversión y un plazo, un precio o el reparto del resultado con B.

\subsubsection{Crecimiento interno: ¿qué limita su velocidad?}
Si A construye su división de robótica, crea valor siempre que la inversión rinda más de lo que cuesta, con los criterios que vimos al estudiar la inversión ([\authorfont{Ver Tema 3.B.2}]). El problema no es tanto si crea valor como a qué velocidad puede hacerlo, porque el crecimiento interno está limitado por recursos que solo se acumulan con el tiempo.

\textcolor{magenta}{Ilustrar esto sobre el gráfico inicial?}

El primero es la \textbf{financiación}. Sin emitir acciones ni cambiar su endeudamiento, la empresa solo puede crecer al ritmo que le permiten sus beneficios retenidos, la tasa de crecimiento sostenible de HIGGINS (1977) ([\authorfont{Ver Tema 3.B.3}]).

El segundo, más profundo, es la \textbf{capacidad de dirección}. PENROSE (1959) observó que cada expansión necesita directivos que conozcan la empresa, sus recursos y su forma de trabajar, y que esa experiencia no se compra en el mercado: los directivos nuevos tardan en adquirirla y los antiguos solo pueden dedicar a planear la expansión el tiempo que les deja la gestión ordinaria. La capacidad de dirección limita así cuánto puede crecer la empresa por sí misma en cada periodo, y ese límite solo retrocede a medida que cada expansión se completa y libera atención directiva para la siguiente.\footnote{%
    Es el llamado \textit{efecto PENROSE}. LOCKETT, WIKLUND, DAVIDSSON y GIRMA (2011) lo contrastaron con empresas suecas entre 1987 y 1996\textcolor{orange}{\textsuperscript{[Probable]}}: el crecimiento interno de un periodo reduce el del siguiente, como predice la teoría, mientras que el crecimiento por adquisiciones lo aumentaba. Una réplica con más de cien mil empresas neerlandesas confirma lo primero, pero encuentra que las adquisiciones pasadas reducen al principio el crecimiento interno posterior, y que el efecto se vuelve positivo con el tiempo, sobre todo cuando lo adquirido está moderadamente relacionado con el negocio propio (LUCINI-PAIONI, DESYLLAS, MARSILI y CEFIS, 2026)\textcolor{orange}{\textsuperscript{[Probable]}}.
}

\subsubsection{Comprar: ¿por qué adquirir lo que podría construirse?}
Comprar B permite a A saltarse esos plazos. No compra solo unos activos, sino una \textbf{organización que ya funciona}: un equipo que sabe hacer los robots, unos clientes, una marca, unas patentes y una forma de trabajar que ningún mercado vende por separado y que construir desde dentro llevaría años, con un resultado incierto.\footnote{%
    Es la complementariedad habitual entre empresas grandes y pequeñas en los sectores intensivos en tecnología, como el farmacéutico. La pequeña aporta un producto prometedor, un equipo de investigación o un capital humano difícil de reunir. La grande, la financiación, la capacidad de producción y la red comercial que la pequeña necesita para explotarlo. La compradora entra así en un nicho rentable y la comprada obtiene el capital que le faltaba.
} 

La propia PENROSE (1959) vio en la adquisición una forma de obtener recursos y experiencia directiva que la empresa no podría crear a tiempo\textcolor{orange}{\textsuperscript{[Probable]}}. El límite, sin embargo, no desaparece, sino que se desplaza, porque integrar B consume la misma atención directiva que habría consumido crecer desde dentro.

\textcolor{magenta}{Ilustrar esto sobre el gráfico inicial?}

Desde la teoría de la inversión, comprar es una forma más de invertir, y la decisión entre construir y comprar depende del precio relativo del capital nuevo y del ya instalado. La $q$ de TOBIN, el cociente entre el valor de mercado del capital de una empresa y su coste de reposición, indica cuánto vale invertir en ella ([\authorfont{Ver Tema 3.A.34}]). JOVANOVIC y ROUSSEAU (2002) la extendieron a las adquisiciones: las empresas que mejor uso hacen del capital, con una $q$ alta, lo compran a las que peor uso hacen de él, y la adquisición \textbf{reasigna el capital existente} hacia quien más rinde con él. Como comprar exige un coste fijo elevado (buscar, negociar e integrar) pero apenas costes de ajuste por cada unidad adicional, la inversión mediante adquisiciones responde a la $q$ del comprador mucho más que la inversión interna.\footnote{%
    Según sus estimaciones con datos de empresas estadounidenses, $2{,}6$ veces más (JOVANOVIC y ROUSSEAU, 2002). Las mismas estimaciones sugieren que el comprador típico malgasta en adquisiciones parte de su liquidez, pero no en su inversión interna, el problema de agencia que veremos más adelante, aunque esto solo explica una pequeña parte de las operaciones.
}

\paragraph{}
La compra tiene, además, dos atractivos que no son de eficiencia. En un mercado maduro, construir capacidad nueva aumenta la oferta y presiona los precios a la baja, mientras que comprarla no añade capacidad y, si la empresa comprada era un competidor, lo elimina. Y la adquisición puede pagarse con acciones propias en lugar de con efectivo, lo que sortea la restricción de financiación del crecimiento interno. 

El primero plantea la cuestión del interés público y el segundo, la de la forma de pago, que estudiaremos al tratar las operaciones.

\paragraph{}
Lo que crece la empresa, en cambio, no lo crece necesariamente la economía. Cuando A construye la división por su cuenta, la inversión agregada aumenta por la nueva capacidad creada. Cuando A compra B, solo cambia de manos una capacidad que ya existía, y el dinero que paga A va a los antiguos accionistas de B. 

\textcolor{magenta}{(Ilustrar esto sobre el gráfico inicial?)}

En la contabilidad nacional, la compra de acciones es una operación financiera y no forma parte de la formación bruta de capital.\footnote{%
    ALFAGEME (2025) documenta, con empresas no financieras estadounidenses entre 1980 y 2020, que las que más gastaron en adquisiciones fueron las que más redujeron su inversión en capital fijo, que el crecimiento mediante adquisiciones coincidió con una mayor concentración de los mercados y que las empresas más adquirentes reunían en torno al $40\%$ de los beneficios, con márgenes más altos. Son correlaciones, no efectos causales, pero señalan la tensión entre el crecimiento de cada empresa y la inversión agregada ([\authorfont{Ver Tema 3.A.15}]).
}

\subsubsection{Aliarse: ¿por qué no integrarse del todo?}
Entre construir y comprar hay una tercera vía. A y B pueden pactar un desarrollo conjunto, crear entre ambas una empresa nueva o cruzarse participaciones, sin que ninguna compre a la otra. La elección entre las tres vías es la misma que estudia la teoría de la empresa al preguntarse dónde termina cada empresa ([\authorfont{Ver Tema 3.A.15}]).

\imagenfit{b1_integracion.pdf}{Del mercado a la jerarquía: grados de integración con otra empresa}{Elaboración propia a partir de WILLIAMSON (1991). Esquema ilustrativo.}

A medida que se avanza hacia la derecha, el comprador gana control sobre los recursos de la otra parte y pierde flexibilidad para deshacer la operación. Integrar una actividad da \textbf{control} para coordinarla y para proteger las inversiones que cada parte hace pensando en la otra, pero exige pagar la prima y unir organizaciones distintas. Coordinarla mediante contratos ahorra ambas cosas, pero obliga a gobernar una relación entre partes independientes que pueden comportarse de forma oportunista. Las \textbf{alianzas} son formas híbridas que se sitúan entre los dos extremos (WILLIAMSON, 1991).

Qué forma conviene depende, por tanto, de cuánta coordinación exige la sinergia que se busca y de cuánto cuesta conseguirla por contrato: cuanto más específicos son los activos implicados y más difícil es prever las contingencias, más se acerca la solución a la adquisición. Cuándo resulta preferible la alianza y cómo se gobierna lo veremos al tratar las operaciones.

\subsection{Valor: ¿de dónde sale y quién se lo queda?}
Comprar es invertir: el comprador paga hoy un precio a cambio de los flujos de la empresa comprada y de los que añade la unión. Para los accionistas del comprador, la operación crea valor si lo que añade la unión supera lo que pagan de más. Para la economía, solo si ese valor no sale del bolsillo de otros.

\subsubsection{Sinergia y prima: ¿cuándo gana el comprador?}
Llamemos $V_A$ y $V_B$ al valor de A y de B como empresas independientes, es decir, al valor actual de los flujos de caja que cada una generaría por su cuenta, y $V_{AB}$ al de la empresa resultante.\footnote{%
    Para simplificar, suponemos que ninguna de las dos tiene deuda, de modo que estos valores son los de sus fondos propios. Con deuda, el razonamiento es el mismo con el valor de las empresas, y el precio de las acciones se obtiene restando la deuda que asume el comprador, como veremos al valorar. 
    
    La sinergia se mide, además, neta de los costes de la operación: asesores, reestructuración e integración.
} La \textbf{sinergia} es lo que la unión vale por encima de la suma de las partes (definición de la ganancia de una fusión de BREALEY, MYERS y ALLEN, 2020):

\eqblock{
  \begin{aligned}
    S=V_{AB}-\left(V_A+V_B\right)
  \end{aligned}
}{\textcolor{magenta}{rellenar, buscar referencias en otros temas sobre cómo hacerlo, 3.B.1}}

En el ejemplo, la sinergia procede sobre todo de vender los robots de B a los clientes de A, una sinergia de ingresos (panel a). Si A paga en efectivo un precio $P$ por B, la \textbf{prima} es lo que paga por encima del valor de B como empresa independiente, $\pi=P-V_B$, y el valor actual neto de la operación para los accionistas de A es lo que obtienen: la empresa B y la sinergia, menos lo que pagan:
$VAN_A=\left(V_B+S\right)-P=S-\pi$

\imagenfit{b1_valor.pdf}{Sinergia y prima: el valor que crea la unión y cómo se reparte}{Elaboración propia a partir de BREALEY, MYERS y ALLEN (2020). Esquema ilustrativo.}

La sinergia se reparte, por tanto, entre los dos grupos de accionistas: los de B se llevan la prima, $\pi$, y los de A, el resto, $S-\pi$ (panel b). De ahí se siguen dos consecuencias.

La primera es que una operación puede crear valor, $S>0$, y empobrecer al comprador, si la prima supera la sinergia: es el caso en que A paga de más. La segunda es que, en la negociación, el comprador no debería pagar más que $V_B+S$, lo que vale B en sus manos, ni el vendedor aceptar menos que $V_B$. 

Entre ambos límites, la prima decide quién se queda la sinergia. Si A paga con acciones propias en lugar de en efectivo, el coste depende además de lo que acabe valiendo la empresa unida, porque los accionistas de B pasan a serlo de ella, como veremos al tratar la forma de pago.

\subsubsection{Fuentes: ¿de dónde sale la sinergia?}
Para que la sinergia sea positiva, la empresa unida tiene que generar más flujos de caja que las dos por separado. ¿Cómo puede esto suceder? De forma general, porque: venda más, gaste menos o necesite invertir menos, o reduzca un riesgo que los inversores no puedan eliminar por su cuenta. 

\imagenfit{b1_direcciones.pdf}{La dirección de la operación y sus fuentes de sinergia}{Elaboración propia. Esquema ilustrativo.}

De qué fuente se trate depende de la relación entre los negocios, y por eso la clasificación tradicional de las operaciones en horizontales, verticales y conglomerados es, ante todo, un mapa de las fuentes de sinergia.

\paragraph{Horizontales: ...}
En una operación \textbf{horizontal} la sinergia procede de las \textbf{economías de escala y de alcance} ([\authorfont{Ver Tema 3.A.11, Tema 3.A.12}]): los costes fijos se reparten entre más unidades, se cierran las instalaciones, redes y servicios centrales duplicados y se negocia con los proveedores desde una posición más fuerte.\footnote{%
    Las economías de alcance y las sinergias de ingresos aparecen también cuando se unen empresas con productos distintos pero relacionados, que comparten clientes, tecnología o red de distribución: la empresa unida ofrece una gama más completa, con productos complementarios, y aprovecha mejor sus redes comerciales. La clasificación tradicional llama a estas operaciones conglomerados de extensión de producto o de mercado, para distinguirlas de los conglomerados puros, entre negocios sin relación alguna\textcolor{orange}{\textsuperscript{[Probable]}}.
} Es el terreno del culto a la reducción de costes. 

Pero la misma operación elimina un competidor y puede dar \textbf{poder de mercado}, que permite subir los precios: una ganancia para los accionistas que pagan los clientes.\footnote{%
    Sin ganancias de eficiencia, una fusión horizontal puede no ser rentable. En un oligopolio de COURNOT con producto homogéneo y costes marginales constantes, las empresas que no se fusionan responden aumentando su producción, y la fusión solo resulta rentable si reúne a casi todo el mercado, en torno al $80\%$ de las empresas en el caso lineal (SALANT, SWITZER y REYNOLDS, 1983). Con productos diferenciados y competencia en precios, en cambio, los rivales responden subiendo también los suyos y la fusión es siempre rentable (DENECKERE y DAVIDSON, 1985). En estos modelos, una fusión horizontal rentable descansa en ahorros de costes o en un mercado en el que se compite en precios.
} La otra sinergia que suele invocarse, la diversificación, merece un examen aparte.

\paragraph{Vertical: ...}
En una operación \textbf{vertical}, entre un proveedor y su cliente, la sinergia procede de lo que cuesta coordinarse por contrato. Si una parte ha hecho inversiones que solo valen en su relación con la otra, queda expuesta a que esta renegocie a su favor una vez hechas, y la integración elimina ese riesgo y la protección costosa que exige ([\authorfont{Ver Tema 3.A.15}]). Además, si las dos tienen poder de mercado, cada una añade su propio margen al precio; integradas, se fija uno solo, el precio final baja y el beneficio conjunto sube (SPENGLER, 1950), de modo que esta sinergia beneficia también a los clientes. Su contrapartida es que la empresa integrada puede negar a sus rivales el acceso al suministro o a la distribución.

Cualquiera que sea la relación entre los negocios, la sinergia puede proceder también de la propia \textbf{gestión}. Si los directivos de B no aprovechan sus recursos, B vale más en manos de quien la gestione mejor, y la compra es la forma de sustituirlos cuando fallan los mecanismos internos: los directivos no van a destituirse a sí mismos, y los accionistas dispersos apenas influyen en la dirección (MANNE, 1965; JENSEN y RUBACK, 1983). Cómo funciona ese mercado del control de las empresas lo veremos al estudiar las operaciones.

\paragraph{Conglomerado: ...}
En un \textbf{conglomerado}, entre negocios sin relación, no hay ahorros de costes ni coordinación que ganar, y las sinergias posibles, aparte de la gestión, son financieras. 

La más sólida es el \textbf{mercado interno de capital}: la dirección central conoce mejor sus divisiones que los inversores externos y puede trasladar fondos de las que los generan a las que tienen mejores proyectos (WILLIAMSON, 1975; STEIN, 1997). Pero el mismo mecanismo permite que las divisiones peor gestionadas negocien su parte de los fondos y que la central reparta el capital para contentarlas, no para invertir mejor (SCHARFSTEIN y STEIN, 2000; RAJAN, SERVAES y ZINGALES, 2000).\footnote{%
    Las demás sinergias financieras son fiscales: aprovechar las pérdidas pendientes de compensar de la empresa comprada o una capacidad de endeudamiento que no se usaba, o aflorar al comprar un valor que no figuraba en el balance de la empresa comprada, el fondo de comercio, cuya amortización reduce los impuestos donde la ley la admite. Su valor depende de la legislación de cada país, que suele limitar la compensación de las pérdidas de las sociedades adquiridas y en algunos países la ha restringido en los últimos años, lo que resta atractivo a estas operaciones. Se estudia en el anexo dedicado al marco institucional de las operaciones.
} 

\paragraph{Diversificación: ¿crea valor reducir el riesgo?}
Si los flujos de A y de B no se mueven a la vez, la empresa unida es menos arriesgada que cada una por separado, y es tentador concluir que vale más. No es así para sus accionistas, por dos razones.

La primera es que los accionistas pueden diversificar por su cuenta, comprando acciones de A y de B, a un coste mucho menor que el de una adquisición, y el mercado solo remunera el riesgo que no pueden eliminar así, el riesgo sistemático ([\authorfont{Ver Tema 3.B.2}]).\footnote{%
    El argumento supone mercados de capitales profundos, como el estadounidense, en los que el inversor encuentra empresas cotizadas de todos los sectores. En mercados más estrechos, como el español, con pocas empresas cotizadas en muchos sectores, la diversificación de la empresa podría ofrecer al inversor algo que no consigue por su cuenta, aunque la posibilidad de invertir en otros países reduce mucho esa ventaja.
} La beta de la empresa unida es la media de las betas de A y de B, ponderada por su valor (beta de una cartera en el CAPM de SHARPE, 1964):
$$\beta_{AB}=\frac{V_A}{V_A+V_B}\,\beta_A+\frac{V_B}{V_A+V_B}\,\beta_B$$
de modo que la rentabilidad que exigen los inversores a la empresa unida es la media de las que exigían a cada una y su valor, sin otras sinergias, la suma de los valores (panel a; LEVY y SARNAT, 1970). La diversificación reduce el riesgo total, pero solo porque elimina el riesgo específico, que nadie remuneraba.

\imagenfit{b1_diversificacion.pdf}{Diversificar: ni baja la beta ni crea valor para el accionista}{Elaboración propia a partir de LEVY y SARNAT (1970) y GALAI y MASULIS (1976). Curvas ilustrativas.}

La segunda es que el menor riesgo beneficia a los acreedores. Si las dos empresas tienen deuda, es menos probable que no puedan pagarla unidas que por separado, porque los resultados de una cubren las pérdidas de la otra: es el \textbf{efecto coaseguro} (LEWELLEN, 1971). La deuda vale entonces más y, como el valor total no ha cambiado, lo que ganan los acreedores lo pierden los accionistas, cuyas acciones son una opción de compra sobre los activos de la empresa que vale menos cuanto menor es su riesgo (panel b; HIGGINS y SCHALL, 1975; GALAI y MASULIS, 1976) ([\authorfont{Ver Tema 3.B.3}]). Los accionistas solo recuperan ese valor si la empresa unida se endeuda más, aprovechando su mayor capacidad de endeudamiento, y lo que ganan entonces procede del ahorro fiscal de la nueva deuda, no de la diversificación.\footnote{%
    KIM y McCONNELL (1977) no encontraron ganancias anormales de los obligacionistas en las fusiones de conglomerados, pero sí que las empresas fusionadas aumentaban su endeudamiento\textcolor{orange}{\textsuperscript{[Probable]}}, lo que es coherente con que los accionistas neutralicen el coaseguro endeudándose más.
}

Diversificar sí reduce el riesgo que soportan los directivos, cuyo empleo y reputación dependen de una sola empresa, y por eso puede ser un motivo para comprar aunque no cree valor, como veremos. El mercado parece descontarlo, porque los conglomerados suelen cotizar por debajo de lo que valdrían sus negocios por separado, un descuento cuyo alcance discutiremos al valorar la suma de las partes.\footnote{%
    Lo mismo ocurre con la diversificación geográfica. Los compradores estadounidenses que adquieren empresas extranjeras obtienen peores rentabilidades y resultados de explotación que los que compran empresas nacionales, y su rentabilidad es menor cuanto más diversifica la operación la empresa entre países y sectores (MOELLER y SCHLINGEMANN, 2005)\textcolor{orange}{\textsuperscript{[Probable]}}. La excepción son las compras que llevan al comprador a un país en el que no estaba: le permiten internalizar ventajas que las fricciones de los mercados internacionales impedirían explotar de otro modo, y el mercado las recibe con ganancias (DOUKAS y TRAVLOS, 1988)\textcolor{orange}{\textsuperscript{[Probable]}}.
}

\subsubsection{Valor creado y valor trasladado: ¿gana la economía lo que gana el accionista?}
La sinergia que hemos medido es la que ven los accionistas, y no tiene por qué coincidir con la que gana la economía. Una parte puede ser \textbf{valor creado}: recursos que se ahorran, producción que aumenta, capital que pasa a manos de quien mejor lo usa. Otra puede ser \textbf{valor trasladado} a los accionistas desde otros grupos con derechos sobre la empresa: desde los acreedores, cuando la operación hace más arriesgada su deuda; desde los clientes, cuando sube los precios; desde Hacienda, cuando reduce los impuestos; y desde los trabajadores y los proveedores.

\imagenfit{b1_traslados.pdf}{Valor creado y valor trasladado}{Elaboración propia. Esquema ilustrativo.}

El traslado desde los trabajadores es el menos visible. SHLEIFER y SUMMERS (1988) observaron que muchas relaciones laborales descansan en \textbf{contratos implícitos}: el trabajador invierte en conocimientos que solo valen en su empresa y acepta salarios bajos al principio a cambio de una promesa no escrita de estabilidad y de salarios que crecen con la antigüedad. Un nuevo propietario, que no ha hecho esa promesa, puede romperla, despidiendo a los trabajadores veteranos o recortando sus salarios, y quedarse con las rentas que la promesa les reservaba. La subida de la cotización mediría entonces un traslado de rentas, no una ganancia de eficiencia, y la confianza rota encarecería en el futuro cualquier acuerdo de este tipo\textcolor{orange}{\textsuperscript{[Probable]}}.

Es la otra cara del culto a la reducción de costes: el mismo recorte puede eliminar una sede duplicada, que ahorra recursos, o romper una promesa, que solo traslada rentas. La evidencia sugiere que lo segundo existe pero no lo explica todo: en las tomas de control hostiles de los años ochenta, los despidos explican entre el $10$ y el $20\%$ de la prima media y los ahorros fiscales algo menos, mientras que la mayor parte de los activos acabó en manos de empresas del mismo sector.\footnote{%
    BHAGAT, SHLEIFER y VISHNY (1990) estudian las $62$ tomas de control hostiles disputadas entre 1984 y 1986 por un importe de al menos $50$ millones de dólares. Los despidos afectaron sobre todo al personal directivo y administrativo de las sedes, y en unos pocos casos explican toda la prima; los ahorros fiscales son grandes en las compras de la empresa por sus propios directivos. El $72\%$ de los activos que cambiaron de manos acabó en empresas con activos similares: las tomas de control deshicieron los conglomerados de las décadas anteriores y devolvieron las empresas a su especialización, lo que apunta a los ahorros de la explotación conjunta o al poder de mercado como fuentes principales de la ganancia.
}

Los efectos del recorte van, además, más allá de la empresa: el cierre de una planta arrastra a la industria auxiliar y a los proveedores que trabajaban para ella, y la operación puede llevarse a otro lugar el centro de decisión. La distinción entre valor creado y valor trasladado, junto con estos efectos sobre terceros, es la que justifica que los poderes públicos examinen las operaciones, como veremos al estudiar el interés público.

\subsection{Motivos de los directivos: ¿por qué se compra sin crear valor?}\label{b1_motivos}
Hasta aquí hemos supuesto que quien decide la compra busca el valor de los accionistas del comprador y estima bien la sinergia, y en ese caso solo se compraría cuando la sinergia supera la prima. Pero la decisión la toman los directivos, que pueden tener motivos propios para comprar y pueden equivocarse al estimar lo que vale la unión. Cualquiera de las dos cosas basta para que se hagan operaciones que empobrecen al comprador.\footnote{%
    Los directivos no son los únicos que intervienen con intereses propios. Los bancos de inversión que asesoran al comprador cobran buena parte de sus honorarios solo si la operación se cierra, y su cuota de mercado depende de cuántas operaciones cierran, no de lo buenas que resultan para sus clientes (RAU, 2000)\textcolor{orange}{\textsuperscript{[Probable]}}. Los gobiernos, por su parte, pueden favorecer o frenar una operación por razones ajenas a su valor, como veremos al estudiar el interés público.
}

Las dos desviaciones caben en una sola condición. Sea $\alpha$ la parte del valor de la empresa que corresponde al directivo, a través de sus acciones y de una remuneración ligada a la cotización, $B$ el beneficio privado que obtiene de la compra y $\hat S$ la sinergia que él estima. El directivo soporta solo la fracción $\alpha$ de lo que la empresa gane o pierda con la operación, pero disfruta entero de su beneficio privado, y compra si (condición de compra del directivo, de elaboración propia a partir de la participación del directivo de JENSEN y MECKLING, 1976, y del error de estimación de ROLL, 1986):
$$\alpha\left(\hat S-\pi\right)+B>0$$
Es decir, está dispuesto a pagar una prima de hasta $\hat S+B/\alpha$, cuando a los accionistas solo les conviene pagar hasta $S$. La diferencia tiene dos componentes, que son los dos motivos que estudiamos: el beneficio privado, $B/\alpha$, tanto mayor cuanto menor es la participación del directivo, y el error de estimación, $\hat S-S$.

\subsubsection{Agencia: ¿qué gana el directivo con comprar?}
El primer motivo es un problema de agencia: los accionistas delegan la gestión en directivos que tienen intereses propios y a los que no pueden vigilar del todo ([\authorfont{Ver Tema 3.B.3}]). Comprar da al directivo dos beneficios que no comparte con los accionistas.

El primero es el \textbf{tamaño}. La remuneración, el prestigio y el poder de los directivos crecen con el tamaño de la empresa más que con su rentabilidad, y comprar es la forma más rápida de crecer. El problema es más grave cuando la empresa genera más fondos de los que puede invertir con rentabilidad: en lugar de repartir ese \textbf{flujo de caja libre}, los directivos tienden a invertirlo en crecer, y una adquisición permite invertir mucho de una sola vez (JENSEN, 1986).\footnote{%
    Las empresas con más liquidez de la que necesitan son más propensas a comprar, sobre todo empresas de otros sectores; sus compras atraen a menos competidores, la cotización las recibe con caídas y van seguidas de un deterioro anormal de los resultados de explotación. Por cada dólar de liquidez sobrante, destruyen siete centavos de valor (HARFORD, 1999).
}

El segundo es un \textbf{riesgo menor para el propio directivo}. Su empleo, su reputación y buena parte de su patrimonio dependen de una sola empresa, y, a diferencia de los accionistas, no puede diversificar ese riesgo. Comprar negocios distintos del propio estabiliza los resultados y reduce la probabilidad de que la empresa atraviese dificultades que le cuesten el puesto, aunque, como vimos, no crea valor para los accionistas (AMIHUD y LEV, 1981).\footnote{%
    AMIHUD y LEV (1981) encontraron que las empresas controladas por sus directivos, sin accionistas de referencia que los vigilen, hacían más adquisiciones de conglomerado que las controladas por sus propietarios\textcolor{orange}{\textsuperscript{[Probable]}}. En la misma línea, la cotización del comprador reacciona peor cuando la compra diversifica la empresa, cuando la empresa comprada crece deprisa y cuando los directivos del comprador venían obteniendo malos resultados (MORCK, SHLEIFER y VISHNY, 1990).
}

La condición de compra señala los dos remedios. El primero es aumentar $\alpha$, ligando la riqueza del directivo al valor de la empresa: las compras son mejor recibidas cuanto mayor es la participación de los directivos del comprador en su capital.\footnote{%
    LEWELLEN, LODERER y ROSENFELD (1985) encontraron que la rentabilidad anormal de los compradores, entre el anuncio de la operación y su aprobación por los accionistas, aumenta con el porcentaje de acciones de la empresa en manos de sus principales directivos. La remuneración puede jugar, sin embargo, en sentido contrario: tras una adquisición, los consejeros delegados de los compradores suelen recibir nuevas acciones y opciones, y su riqueza aumenta en tres de cada cuatro operaciones en las que los accionistas pierden (HARFORD y LI, 2007).
} El segundo es que comprar mal le cueste algo al propio directivo: la empresa que compra mal baja de cotización y se convierte a su vez en objetivo de quien espera gestionarla mejor, una disciplina que estudiaremos al tratar el mercado del control de las empresas.\footnote{%
    Las empresas que acabaron siendo objeto de una toma de control habían hecho adquisiciones que redujeron significativamente su valor, mientras que las que no lo fueron habían hecho adquisiciones que lo aumentaron; y las peores de esas adquisiciones se vendieron después, en la desmembración que siguió a la toma de control o en la reestructuración con la que la empresa intentó evitarla (MITCHELL y LEHN, 1990).
}

\subsubsection{Arrogancia: ¿por qué se paga de más sin pretenderlo?}
El segundo motivo no exige que el directivo persiga su propio interés: basta con que se equivoque. ROLL (1986) mostró que habría compras aunque ninguna creara valor. Cada posible comprador estima con error lo que valdría la empresa B en sus manos, y solo hace una oferta si su estimación supera el precio de mercado de B. El que acaba comprando es, por tanto, el que más ha sobrestimado su valor: es la \textbf{maldición del ganador} de las subastas en las que el objeto vale lo mismo para todos pero nadie conoce su valor ([\authorfont{Ver Tema 3.A.14}]). Los directivos no corrigen ese sesgo porque confían en su propia estimación, la \textbf{arrogancia} que da nombre a la hipótesis, y porque cada uno hace pocas adquisiciones a lo largo de su carrera y apenas puede aprender de la experiencia\textcolor{orange}{\textsuperscript{[Probable]}}.

En el caso extremo en que ninguna operación cree valor, $S=0$, lo que gana el vendedor es exactamente lo que pierde el comprador, $VAN_A=-\pi$, y el valor conjunto de las dos empresas cae en lo que cuesta la operación. A diferencia de la agencia, la arrogancia no se corrige con incentivos, porque el directivo cree estar creando valor: aunque tenga mucho que perder, comprará igual si sobrestima la sinergia.\footnote{%
    MALMENDIER y TATE (2008) identificaron a los consejeros delegados con exceso de confianza porque mantenían hasta su vencimiento opciones sobre acciones de su propia empresa que les convenía ejercer para diversificar, y por cómo los describía la prensa. Esos consejeros hacen más adquisiciones, sobre todo de empresas de otros sectores y cuando disponen de liquidez o de capacidad de endeudamiento sin usar, y el mercado recibe sus ofertas peor que las de los demás. BUFFETT (1982), en su carta a los accionistas de Berkshire Hathaway sobre 1981, lo describió con una fábula: muchos directivos parecen haber oído de niños demasiadas veces el cuento del príncipe encerrado en el cuerpo de un sapo al que libera el beso de una princesa, y están seguros de que su beso hará maravillas con la rentabilidad de la empresa que compran; se han visto muchos besos, pero muy pocos milagros.
}

\subsubsection{Contraste: ¿cómo distinguir los motivos?}
Los tres motivos, sinergia, arrogancia y agencia, predicen lo mismo para el vendedor, que gana. Se distinguen por lo que predicen para la ganancia total de la operación y, sobre todo, por cómo se relaciona con ella lo que gana el vendedor (BERKOVITCH y NARAYANAN, 1993).

\imagenfit{b1_motivos.pdf}{Sinergia, arrogancia y agencia: la ganancia total y la de la empresa comprada}{Elaboración propia a partir de BERKOVITCH y NARAYANAN (1993). Esquema ilustrativo.}

Si se compra por la sinergia, la ganancia total es positiva y el vendedor se lleva más cuanto mayor es, porque una sinergia mayor atrae a más compradores y refuerza su posición en la negociación: la correlación es positiva (panel a). Si se compra por arrogancia, la ganancia total es nula, cualquiera que sea la prima, y la correlación también (panel b). Si se compra por motivos de agencia, la ganancia total es negativa, y cuanto mayor es el beneficio privado que persigue el directivo, más valor está dispuesto a destruir y más está dispuesto a pagar: las primas más altas acompañan a las mayores pérdidas, y la correlación es negativa (panel c). En los dos últimos casos, todas las operaciones quedan por encima de la recta $\pi=S$, donde el comprador pierde.

BERKOVITCH y NARAYANAN (1993) encontraron que en las operaciones con ganancia total positiva predomina la sinergia, aunque con indicios de arrogancia, y que en las que la tienen negativa predomina la agencia. Los tres motivos coexisten, y el resultado medio de las operaciones, que veremos a continuación, es su suma.

\subsection{Evidencia: ¿cuándo y para quién crean valor?}\label{b1_evidencia}
Medir la sinergia no es sencillo, porque no se observa lo que habrían valido las dos empresas sin la operación. La evidencia la aproxima por dos vías. Los \textbf{estudios de eventos} miden cómo cambia la cotización de cada empresa en los días que rodean el anuncio, descontado lo que se habría movido de todos modos con el mercado: si el mercado valora bien, esa rentabilidad anormal estima el valor actual neto de la operación para cada parte, y la de las dos empresas juntas, la sinergia. Los \textbf{estudios contables} comparan los resultados de explotación de la empresa unida con los que obtenían las dos por separado, en relación con los de su sector, y miden la ganancia que de verdad se materializa ([\authorfont{Ver Anexo \ref{anx:ev_medicion}}]).

Las dos vías coinciden en lo esencial: las operaciones crean, en conjunto, un valor modesto, del que se queda casi todo el vendedor y que no siempre es una ganancia para la economía; las ganancias varían como predicen las teorías anteriores, según quién compra a quién y cómo se paga; y las operaciones llegan en olas, cuyos picos, con cotizaciones altas y financiación abundante, reúnen las peores.

\subsubsection{Reparto: ¿quién gana con la operación?}
Los estudios de eventos dan tres resultados que se repiten en muchas muestras y periodos. Los accionistas de la empresa comprada ganan mucho al anunciarse la operación, porque reciben la prima. Los del comprador no ganan nada en promedio, o pierden algo. Y las dos empresas juntas ganan poco, pero ganan ([\authorfont{Ver Anexo \ref{anx:ev_resultados}}]).

\imagenfit{b1_eventos.pdf}{La reacción de las cotizaciones al anuncio de la operación}{Elaboración propia a partir de ANDRADE, MITCHELL y STAFFORD (2001). Curvas ilustrativas.}

En términos de la sinergia y la prima, la prima absorbe casi toda la sinergia, $\pi\approx S$, y el valor actual neto del comprador es aproximadamente nulo. Es lo que cabe esperar si varios compradores compiten por la misma empresa y los accionistas de la comprada pueden negarse a vender por menos de lo que valdrá en manos del comprador, como veremos al tratar las operaciones. Para el comprador medio, crecer comprando es una inversión de valor actual neto nulo, pero ese promedio esconde diferencias.

Esas diferencias siguen el patrón que predicen las teorías anteriores. La ganancia es mayor cuando una empresa bien gestionada, con una $q$ de TOBIN alta, compra otra con una $q$ baja: es la \textbf{reasignación del capital} que vimos al comparar construir y comprar. El comprador pierde cuando paga con acciones, porque ofrecerlas indica que sus directivos las consideran caras, la misma \textbf{señal} que da cualquier emisión de acciones ([\authorfont{Ver Tema 3.B.3}]); gana cuando compra una empresa no cotizada, cuyos dueños, que no pueden vender sus acciones con facilidad, aceptan un descuento por su falta de liquidez; y gana más cuanto más pequeño es, lo que apunta a que la agencia y la arrogancia pesan más en las grandes empresas ([\authorfont{Ver Anexo \ref{anx:ev_resultados}}]).

\subsubsection{Origen de las ganancias: ¿valor creado o trasladado?}
Que la cotización conjunta suba no prueba que gane la economía: puede reflejar un traslado de rentas o un error del mercado. Los estudios contables permiten distinguirlo, pero sus resultados están divididos.

Una parte de la evidencia encuentra \textbf{ganancias de eficiencia}. Las mayores fusiones mejoran la productividad de sus activos en relación con su sector, sobre todo cuando los negocios se solapan, sin recortar la inversión ni la investigación, y las que más suben en bolsa al anunciarse son las que más mejoran después, de modo que el mercado anticipa ganancias que se materializan. Pero la mejora desaparece cuando se compara a las empresas que se fusionan con otras que ya tenían antes los mismos resultados, porque las que se fusionan suelen ser las que mejor iban.\footnote{%
    HEALY, PALEPU y RUBACK (1992), con las $50$ mayores fusiones estadounidenses entre 1979 y mediados de 1984: la mejora de la productividad de los activos eleva la rentabilidad del flujo de caja de explotación, y su relación con la reacción de la cotización al anuncio es positiva y fuerte. GHOSH (2001) no encuentra mejora cuando las empresas de control se eligen por su tamaño y sus resultados antes de la operación. La mejora de la gestión es una de las fuentes de ganancia: la sinergia es mayor cuanto más protege el comprador los derechos de sus accionistas en comparación con la empresa comprada, y de ella se benefician los accionistas de las dos (WANG y XIE, 2009).
}

Otra parte apunta al \textbf{poder de mercado}. Desde los años ochenta, los márgenes sobre el coste han aumentado en muchos sectores, sobre todo en las empresas que ya los tenían más altos, según unas estimaciones que han abierto un debate (DE LOECKER, EECKHOUT y UNGER, 2020) ([\authorfont{Ver Tema 3.A.15}]). Las fusiones contribuyen a ello: en la industria manufacturera estadounidense se asocian a márgenes más altos sin apenas efecto sobre la productividad de las plantas, y las evaluaciones de fusiones ya realizadas encuentran con frecuencia subidas de precios.\footnote{%
    BLONIGEN y PIERCE (2016), con datos de plantas manufactureras estadounidenses de 1997 a 2007, tampoco encuentran, en promedio, ganancias por la reasignación de la producción hacia las plantas más eficientes ni por el recorte de los servicios administrativos. Las evaluaciones posteriores de fusiones, como las de KWOKA (2015) para Estados Unidos y STIEBALE y SZÜCS (2026) para las fusiones revisadas por la Comisión Europea, se estudian en el Tema 3.A.15.
}

La evidencia no permite, por tanto, una conclusión general: unas operaciones crean valor real y otras trasladan rentas desde los clientes, y lo que gana el accionista no mide por sí solo lo que gana la economía.

\subsubsection{Olas: ¿por qué se concentran en el tiempo?}
Las operaciones no se reparten de forma regular en el tiempo: llegan en \textbf{olas}, en las que el número de operaciones en el máximo multiplica varias veces el del mínimo, y que se observan a la vez en muchos sectores y países. Desde finales del siglo XIX se han sucedido seis en Estados Unidos, y cada una ha tenido un tipo dominante de operación, que dependió en buena medida de lo que permitía la legislación de competencia ([\authorfont{Ver Anexo \ref{anx:ev_olas}}]). En la fase alcista se hacen cada vez más operaciones, a precios y múltiplos cada vez más altos, hasta que una caída de la bolsa o del crédito las paraliza: el crac de 1929, el estallido de la burbuja tecnológica en 2000 o la crisis financiera de 2008.

\imagenfit{b1_olas.pdf}{Las olas de fusiones en Estados Unidos y su tipo dominante}{Elaboración propia a partir de MARTYNOVA y RENNEBOOG (2008). Esquema ilustrativo: fechas aproximadas, que varían según los autores; la altura de las olas no refleja su tamaño.}

Que las olas coincidan con las subidas de la bolsa es, a primera vista, un enigma. Si comprar una empresa es una alternativa a invertir en capital nuevo, debería comprarse más cuando las empresas están baratas en relación con lo que costaría reponer su capital, es decir, cuando la $q$ de TOBIN es baja y la bolsa está baja. Los datos dicen lo contrario: la actividad aumenta con la $q$ y disminuye con los tipos de interés reales (GOLBE y WHITE, 1988). Hay dos explicaciones, que no se excluyen.

La primera ve en las olas una \textbf{reasignación del capital}. Un shock tecnológico, regulatorio o de precios cambia la escala o la organización eficientes de un sector, y sus empresas se reorganizan comprándose y vendiéndose unas a otras: por eso las operaciones se agrupan por sectores, y la intensidad y la duración de cada ola dependen de cuántos sectores sufren shocks a la vez. Esa reorganización necesita financiación, y las olas de los distintos sectores coinciden en el tiempo cuando el crédito es abundante y barato, las mismas condiciones que suben la bolsa. La $q$ alta del comprador indica entonces que sabe usar mejor el capital, no que esté caro, y antes de cada ola, salvo la de los sesenta, aumentó la dispersión de la $q$ entre empresas, la señal de que había capital que reasignar (MITCHELL y MULHERIN, 1996; JOVANOVIC y ROUSSEAU, 2002; HARFORD, 2005).

La segunda atribuye las olas a los \textbf{errores de valoración} del mercado (SHLEIFER y VISHNY, 2003). Si la bolsa sobrevalora algunas empresas, sus directivos pueden convertir esas acciones caras en activos reales comprando con ellas empresas menos sobrevaloradas. La operación no necesita sinergia para beneficiar a los accionistas que el comprador tendrá a largo plazo, porque lo que ganan lo pierden los de la empresa comprada, que reciben acciones que valen menos de lo que parece; sus directivos lo aceptan porque piensan a corto plazo o porque se les compensa. La explicación predice olas en las subidas de la bolsa, pagos con acciones y malos resultados de los compradores a largo plazo, y la evidencia muestra que las empresas más sobrevaloradas compran a las menos sobrevaloradas, pagando con acciones (RHODES-KROPF, ROBINSON y VISWANATHAN, 2005).

Las dos explicaciones resuelven el enigma, porque ambas predicen más operaciones cuando la bolsa sube, y la evidencia sugiere que se complementan: los shocks desencadenan las olas, y la liquidez y las valoraciones altas las amplifican. Lo que sí parece claro es que los picos reúnen las peores operaciones: las que se anuncian con valoraciones altas reciben mejor acogida en bolsa, pero obtienen peores resultados después, como si los directivos se imitaran unos a otros, y las del final de cada ola obedecen más a los motivos de los directivos que a la sinergia. El desarrollo de las dos explicaciones, su evidencia y los indicadores con que se sigue el ciclo están en el anexo ([\authorfont{Ver Anexo \ref{anx:ev_causas}}]).

\section{Valoración: ¿cuánto vale la empresa y cuánto pagar por ella?}\label{b2}
\puntosclave{
    Para comprar hay que estimar dos valores: el de la empresa por sí sola, $V_B$, que es lo mínimo que aceptará el vendedor, y el que tiene en manos del comprador, $V_B+S$, lo máximo que este debería pagar. La valoración fija esos límites; el precio sale de la negociación, $P=V_B+\beta S$, y con varios compradores sube hasta lo que vale la empresa para el segundo.

    Se valora primero la empresa, para todos los que la financian, y se resta la deuda neta para llegar al valor de las acciones, $E=V-D$.

    El descuento de flujos es la referencia: el valor actual de los flujos de caja libres, a la tasa del riesgo de la empresa comprada y no a la del comprador, más un valor terminal que suele ser la mayor parte del valor. Crecer solo aumenta el valor si la nueva inversión rinde más que el coste del capital, $RONIC>k$.

    La sinergia se valora aparte, con sus propios flujos y su propio riesgo: los costes de integración llegan antes que las ganancias, y los ahorros de costes son más seguros que los aumentos de ingresos.

    Un múltiplo es un descuento de flujos comprimido, que resume crecimiento, rentabilidad y riesgo. Las comparables cotizadas estiman el valor por sí sola; las transacciones incluyen la prima que pagaron otros compradores, que solo conviene pagar si la sinergia propia es al menos igual.

    El valor de liquidación es el suelo del valor de la empresa, y el valor contable solo es su valor si rinde exactamente lo que exigen sus accionistas. La suma de partes revela el descuento de los conglomerados, cuya causa se discute, y sus precios de mercado pueden fallar, como en 3Com y Palm.

    Comprar por etapas convierte la adquisición en una opción, que solo se ejerce si crea valor y vale más cuanto mayor es la incertidumbre; el comprador no debe pagar al vendedor las opciones que solo él puede ejercer.
}

El bloque anterior mostró que comprar crea valor para el comprador si la sinergia supera la prima, $VAN_A=S-\pi$. Para decidir si compra y cuánto ofrece, A necesita, por tanto, dos estimaciones: lo que vale B por sí sola, $V_B$, y lo que añade la unión, $S$. Los métodos de valoración son las herramientas para obtenerlas.

Todos estiman lo mismo, el valor actual de los flujos que la empresa generará, y difieren en el camino.\footnote{%
    Los manuales los agrupan en seis familias: los métodos de balance; los de la cuenta de resultados, que son los múltiplos; los mixtos, que suman al balance un fondo de comercio; los de descuento de flujos; los de creación de valor; y los de opciones (FERNÁNDEZ, 2007). Los de creación de valor, que descuentan el beneficio económico, son otra forma de descontar flujos y, con supuestos coherentes, dan el mismo resultado ([\authorfont{Ver Tema 3.B.1}]).
} El descuento de flujos los proyecta y los descuenta uno a uno; los múltiplos toman la valoración que el mercado da a empresas parecidas; los métodos de patrimonio parten de lo que la empresa tiene; y las opciones reales añaden el valor de poder decidir más adelante.

La tesis del bloque es que la valoración no fija el precio, sino sus límites, el mínimo que acepta el vendedor y el máximo que debe pagar el comprador, y que el error más caro no es de método, sino de supuestos: atribuir a la empresa comprada una sinergia que solo puede aportar el comprador, sobrestimarla o descontarla a una tasa que no corresponde a su riesgo.

\subsection{Valor y precio: ¿qué se valora y cuánto se paga?}\label{b2_precio}
Antes de calcular hay que precisar qué se valora y para quién. Lo que se valora es el negocio, y de su valor se deduce el de las acciones. Y la misma empresa vale distinto para cada uno de sus posibles dueños, de modo que el precio al que cambia de manos no lo da ninguna valoración, sino la negociación entre ellos.

\subsubsection{Valor: ¿de la empresa o de sus acciones?}
Lo que A compra es el negocio de B, sus activos de explotación, y lo que vale ese negocio es el \textbf{valor de la empresa}, $V$: el valor actual de los flujos que generará para todos los que lo financian, accionistas y acreedores. Los métodos lo estiman con independencia de cómo se financie la empresa, y el \textbf{valor de las acciones}, que es lo que A paga a los accionistas de B, se obtiene restando la deuda financiera neta que asume (relación entre el valor de la empresa y el de sus acciones, KOLLER, GOEDHART y WESSELS, 2020):
$$E=V-D$$
La deuda neta es la deuda con coste menos la tesorería y las inversiones financieras que la explotación no necesita, y se le suman otros compromisos que también son deuda, como las pensiones comprometidas y no cubiertas.\footnote{%
    El valor de la empresa es el que tiene para todos los que aportan capital. Los acreedores aportan fondos con un coste explícito, fijado por contrato, y los accionistas cobran lo que queda; por eso, en las operaciones, el valor de la empresa es la magnitud de referencia, y el valor contable o el de liquidación solo sirven de contraste. Se valora la actividad ordinaria aislada de la deuda, de la tesorería y de las oscilaciones transitorias del fondo de maniobra: en la práctica, el precio pactado se ajusta si al cierre la empresa tiene más o menos clientes por cobrar o proveedores por pagar de lo habitual.
}

\subsubsection{Precio: ¿dónde queda entre el mínimo y el máximo?}
La misma empresa vale distinto según quién la tenga. Para sus accionistas actuales vale $V_B$, lo que generará gestionada como hasta ahora, y es lo mínimo que aceptarán por ella; para A vale $V_B+S$, y es lo máximo que debería pagar. Es la diferencia entre el valor de mercado de un activo, el que le daría cualquier comprador, y su \textbf{valor de inversión} para un comprador concreto, que incluye la sinergia que solo él puede obtener.\footnote{%
    La contabilidad hace la misma distinción. El valor razonable de las normas internacionales de información financiera es el precio que se recibiría al vender un activo en una transacción ordenada entre participantes en el mercado; es una medida de mercado, no propia de una entidad concreta, y no incluye por tanto las sinergias de un comprador determinado (NIIF 13, párrafos 2 y 9, que en la Unión recoge el Reglamento (UE) 2023/1803).
} La valoración fija esos dos límites, y el precio, $P=V_B+\pi$, sale de la negociación entre ellos.

Si hay un solo comprador, el reparto depende del poder negociador de cada parte. Con el mismo poder, la solución de negociación de NASH reparte el excedente a partes iguales y la prima es la mitad de la sinergia; en general, si $\beta$ mide el poder negociador del vendedor (solución de negociación de NASH, 1950, en su versión asimétrica):
$$P=V_B+\beta S,\qquad 0\leq\beta\leq1$$
El poder de cada parte depende de sus alternativas ([\authorfont{Ver Tema 3.A.14}]): A puede construir la capacidad que busca o aliarse con B, como vimos, y B puede seguir sola o buscar otro comprador (panel a).

Si hay varios compradores, decide la competencia. Cada uno valora B según su propia sinergia, y en una subasta el precio sube hasta que solo queda el que más la valora, que paga lo que vale B para el segundo: con sinergias $S_1>S_2$, el precio se acerca a $V_B+S_2$, y el ganador solo conserva la diferencia, $S_1-S_2$ (panel b). Cuanto más se parecen los compradores, más se acerca la prima a toda la sinergia, que es lo que muestra la evidencia: el vendedor se queda casi todo.

\imagenfit{b2_precio.pdf}{El precio, entre el valor para el vendedor y el valor para el comprador}{Elaboración propia a partir de NASH (1950). Esquema ilustrativo.}

El vendedor tiene, además, una ventaja de información: conoce mejor que nadie lo que vale B. Por eso el comprador la revisa antes de pagar, con una \textit{due diligence} de sus cuentas, sus contratos, sus impuestos, su plantilla y sus riesgos medioambientales, y se protege con garantías del vendedor, con pagos aplazados que dependen de los resultados futuros o pagando con sus propias acciones, que trasladan al vendedor parte del riesgo, como veremos al tratar las operaciones.\footnote{%
    La revisión suele encargarse a consultores y abogados externos y busca contingencias y pasivos ocultos, para reducir la asimetría de información entre las partes. Es mucho más completa en las operaciones amistosas, en las que el comprador accede a la información interna de la empresa, que en las hostiles, en las que solo dispone de la pública.
}

Desde el punto de vista del vendedor, lo que el comprador paga de más, la \textbf{prima de control}, es lo que cobra por ceder el control. Para el comprador, solo se justifica por la sinergia o por los beneficios privados que el control da a quien lo ejerce, como los que vimos al estudiar los motivos de los directivos, que no comparten los demás accionistas.\footnote{%
    DYCK y ZINGALES (2004) los estimaron con $393$ ventas de paquetes de control en $39$ países: el valor del control equivale de media al $14\%$ del valor de las acciones, con diferencias entre países que van del $-4$ al $65\%$, y es mayor donde los mercados de capitales están menos desarrollados y la propiedad más concentrada.
}

\subsection{Descuento de flujos: ¿cuánto vale B por sí sola y cuánto vale la sinergia?}\label{b2_dcf}
El descuento de flujos es la referencia de todos los métodos, porque aplica a la empresa el criterio con que se valora cualquier inversión: vale el valor actual de los flujos que generará, descontados a la rentabilidad que exigen las inversiones de su mismo riesgo ([\authorfont{Ver Tema 3.B.2}]). En una adquisición se aplica dos veces, a B por sí sola y a la sinergia, y conviene no mezclarlas.

\subsubsection{Valor por sí sola: flujos, tasa y valor terminal}
La valoración tiene dos etapas. En la primera se analizan el negocio y el sector, sus ventajas competitivas, sus barreras de entrada y su regulación, y con ello se construyen las hipótesis sobre los ingresos, los costes y las inversiones; en la segunda, un modelo financiero las convierte en flujos.\footnote{%
    Los ingresos se proyectan por producto, canal o zona; los costes variables, con ellos, y los gastos generales, por separado, a partir de previsiones de inflación y de tipos de interés. Las inversiones son de dos clases: en inmovilizado, de mantenimiento, que repone lo que se amortiza, y de expansión; y en circulante, que se estima con el balance proyectado a partir de las cuentas a cobrar, las existencias y las deudas con proveedores. Es la parte en que más pesan la experiencia y el criterio del analista. El procedimiento completo se desarrolla en el anexo dedicado a la valoración.
}

El valor de la empresa es el valor actual de sus \textbf{flujos de caja libres}, los que quedan después de pagar los impuestos y de financiar las inversiones que necesita la explotación, y que pueden repartirse entre accionistas y acreedores. Como nadie puede prever esos flujos para siempre, se proyectan año a año durante un horizonte de unos cinco años, y lo que viene después se resume en un \textbf{valor terminal} (descuento de flujos de caja libres, KOLLER, GOEDHART y WESSELS, 2020):
$$V_B=\sum_{t=1}^{T}\frac{FCL_t}{\left(1+k_B\right)^t}+\frac{VT_T}{\left(1+k_B\right)^T},\qquad FCL_t=BAII_t\left(1-\tau\right)+A_t-I_t-\Delta NOF_t$$
donde $BAII_t\left(1-\tau\right)$ es el resultado de explotación después de impuestos, $A_t$ las amortizaciones, que no son un pago, $I_t$ la inversión en inmovilizado y $\Delta NOF_t$ el aumento de las necesidades operativas de fondos (panel a).\footnote{%
    Los intereses no se restan: el flujo es el de la empresa, independiente de cómo se financie, y el ahorro fiscal de la deuda se recoge en la tasa de descuento, para no contarlo dos veces. Si se descuentan los flujos para el accionista, después de intereses y de los cambios en la deuda, con el coste de los recursos propios, se obtiene directamente el valor de las acciones. Y cuando el endeudamiento va a cambiar mucho, como en las compras apalancadas, es mejor separar el valor del negocio del de su financiación, con el valor actual ajustado ([\authorfont{Ver Tema 3.B.2}]).
}

La tasa de descuento, $k_B$, es el coste del capital del negocio de B, el que corresponde a su riesgo, y no el del comprador. Descontar los flujos de B con el coste del capital de A cuando sus negocios tienen riesgos distintos es la falacia del WACC aplicada a una adquisición: un comprador con un negocio seguro sobrevaloraría una empresa arriesgada, y uno con un negocio arriesgado infravaloraría una segura ([\authorfont{Ver Tema 3.B.2}]).\footnote{%
    En la práctica, los fondos de capital riesgo fijan a menudo una rentabilidad mínima, de hasta el $25$-$30\%$, que hace de tasa de descuento: refleja su mandato y el coste de oportunidad de sus inversores, no solo el riesgo de la empresa.
}

El valor terminal suele ser la mayor parte del valor, y por eso es donde más importan los supuestos. Si desde $T$ los flujos crecen a una tasa constante $g<k_B$, su valor es el de una renta perpetua creciente, como en el modelo de GORDON y SHAPIRO (1956). Pero el crecimiento no es gratis: para crecer, la empresa tiene que reinvertir parte de su resultado, y la parte necesaria es menor cuanto más rinde la nueva inversión, $RONIC$. Teniéndolo en cuenta (fórmula de los inductores de valor de KOLLER, GOEDHART y WESSELS, 2020, con crecimiento constante):
$$VT_T=\frac{FCL_{T+1}}{k_B-g}=\frac{BAII_{T+1}\left(1-\tau\right)\left(1-\dfrac{g}{RONIC}\right)}{k_B-g}$$
El crecimiento solo aumenta el valor si la nueva inversión rinde más de lo que cuesta, $RONIC>k_B$. Si rinde exactamente su coste, crecer no añade nada, y el valor terminal es el resultado de explotación después de impuestos dividido por el coste del capital, cualquiera que sea $g$; y si rinde menos, cuanto más crece la empresa, menos vale (panel b). Es la condición con que empezamos el tema, ahora dentro de la fórmula del valor.\footnote{%
    Por eso la tasa de crecimiento a largo plazo suele ser inferior a la de los primeros años: las ventajas de una sinergia, de una nueva oportunidad de negocio o de una reducción de costes son transitorias, porque el entorno competitivo cambia y el sector evoluciona, y la rentabilidad de la nueva inversión tiende hacia el coste del capital. A largo plazo, $g$ no puede superar el crecimiento nominal de la economía. El valor terminal se calcula a veces aplicando al último resultado un múltiplo de empresas comparables, lo habitual en el capital riesgo, lo que supone que el mercado valorará entonces la empresa como hoy valora a sus comparables.
}

\subsubsection{Sinergia: flujos propios y riesgo propio}
La sinergia se valora aparte, con el mismo método, como el valor actual de los flujos adicionales que generará la unión, netos de los costes de integración (la sinergia como valor actual de los flujos incrementales, BREALEY, MYERS y ALLEN, 2020):
$$S=\sum_{t=1}^{\infty}\frac{\Delta FCL_t}{\left(1+k_S\right)^t}$$
donde $\Delta FCL_t$ es la diferencia entre los flujos de la empresa unida y la suma de los de A y B por separado. Separarla de $V_B$ tiene tres ventajas.

La primera es que evita pagar al vendedor la sinergia del comprador. $V_B$ debe medir lo que vale B gestionada como hasta ahora; si incorpora las mejoras que solo A puede introducir, A estaría pagando por adelantado lo que solo él puede crear.

La segunda es que permite descontar la sinergia a la tasa de su propio riesgo, $k_S$. Si procede del negocio de B, su riesgo es el de B, y por eso se descuenta con el coste del capital de la empresa comprada, no con el del comprador. Pero no toda la sinergia es igual de segura: los ahorros de costes dependen sobre todo de decisiones del comprador, mientras que los aumentos de ingresos dependen de los clientes y de la reacción de los competidores, y el mercado cree más en los primeros, como vimos al tratar la promesa de reducir costes.

La tercera es que obliga a fechar las ganancias y los costes. Los costes de integración llegan al principio y con seguridad, y las ganancias, más tarde y con dudas, de modo que la sinergia vale menos de lo que sugiere su cuantía anual.\footnote{%
    Los analistas ajustan sus proyecciones con las sinergias que se esperan y con los planes anunciados para la empresa unida: la duplicación de recursos puede llevar a vender una división de la empresa comprada, o los costes de explotación pueden revisarse a la baja por las economías de escala.
}

\subsubsection{Alcance: ¿cuánto fiarse del resultado?}
El descuento de flujos tiene la ventaja de hacer explícitos todos los supuestos, lo que permite estudiar cómo cambia el valor con cada uno, algo útil en la negociación, y modelizar los cambios que el comprador piensa introducir. Su debilidad es la otra cara de esa ventaja: el valor depende de supuestos inciertos sobre flujos lejanos, el valor terminal es muy sensible a la tasa de descuento y al crecimiento, y la tasa puede cambiar con los mercados o con el sector. En las empresas que crecen deprisa, con flujos negativos durante años porque invierten mucho, casi todo el valor procede del valor terminal y es, por tanto, el más difícil de estimar. Por eso el resultado se acompaña de análisis de sensibilidad y de escenarios, y se contrasta con los múltiplos del mercado.\footnote{%
    Bien aplicado, funciona: con $51$ compras apalancadas y recapitalizaciones entre 1983 y 1989, el valor actual de las previsiones de flujos que la dirección facilitó a los inversores se desvió de media menos de un $10\%$ del valor de mercado de la operación, y lo hizo al menos tan bien como los métodos basados en empresas y transacciones comparables (KAPLAN y RUBACK, 1995).
}

\subsection{Múltiplos: ¿cuánto paga el mercado por empresas parecidas?}\label{b2_multiplos}
Los múltiplos valoran B por comparación. Se toma lo que el mercado paga por empresas parecidas por cada unidad de una magnitud, como el resultado o las ventas, y se aplica a la de B: si las comparables valen $m$ veces su resultado de explotación, B vale $m$ veces el suyo. No son una alternativa al descuento de flujos, sino un atajo: el múltiplo resume lo que el mercado espera del crecimiento, la rentabilidad y el riesgo de las comparables, y aplicarlo a B supone que de ella se espera lo mismo.

\subsubsection{Contenido: ¿qué resume un múltiplo?}
Si una empresa crece a una tasa constante desde hoy, su valor es el de la fórmula de los inductores de valor con que calculamos el valor terminal, y al dividirlo por el resultado de explotación del año siguiente se obtiene el múltiplo que le corresponde (múltiplo implícito en la fórmula de los inductores de valor de KOLLER, GOEDHART y WESSELS, 2020):
$$\frac{V}{BAII_1}=\frac{\left(1-\tau\right)\left(1-\dfrac{g}{RONIC}\right)}{k-g}$$
El múltiplo es, por tanto, mayor cuanto menor es el riesgo, que reduce el coste del capital, $k$, y cuanto más rinde la nueva inversión, $RONIC$; y el crecimiento solo lo eleva si esa rentabilidad supera el coste del capital. Si la iguala, el múltiplo es el mismo cualquiera que sea el crecimiento, y si es inferior, crecer más lo reduce. Dos empresas del mismo sector pueden merecer, así, múltiplos muy distintos, y una que crece más solo merece uno mayor si su crecimiento crea valor.

\imagenfit{b2_multiplos.pdf}{El múltiplo según la rentabilidad de la nueva inversión}{Elaboración propia a partir de KOLLER, GOEDHART y WESSELS (2020). Curvas ilustrativas.}

La magnitud a la que se refiere el múltiplo decide qué diferencias entre empresas neutraliza. Los múltiplos del \textbf{valor de la empresa}, sobre el resultado de explotación, sobre el resultado antes de amortizaciones (EBITDA), sobre las ventas o sobre el flujo de caja libre, apenas dependen de cómo se financie, porque el numerador incluye la deuda y el denominador es anterior a los intereses. Los de las \textbf{acciones}, como la relación entre el precio y el beneficio (PER) o el precio sobre el flujo de caja para el accionista, sobre las ventas o sobre el valor contable, cambian con el endeudamiento aunque el negocio sea el mismo, y por eso los primeros son los habituales para comparar empresas con estructuras de capital distintas.\footnote{%
    Numerador y denominador deben referirse a los mismos financiadores: el valor de la empresa, a una magnitud anterior a los intereses; el de las acciones, a una posterior. El resultado antes de amortizaciones evita las diferencias en las políticas de amortización, a costa de ignorar las distintas necesidades de inversión, y las ventas solo se usan cuando el resultado es negativo o poco significativo, porque suponen márgenes iguales. Cada sector tiene sus múltiplos habituales, convenciones de analistas, bancos de inversión e inversores de capital riesgo, que también se usan para estimar el valor terminal.
}\textsuperscript{,}\footnote{%
    Con empresas estadounidenses, los múltiplos que mejor explican los precios son los basados en las previsiones de beneficios de los analistas, con errores inferiores al $15\%$ del precio en la mitad de los casos; les siguen los basados en los beneficios pasados; los flujos de caja y el valor contable empatan en tercer lugar, y las ventas son el peor, con un orden que se repite en casi todos los sectores (LIU, NISSIM y THOMAS, 2002). Funcionan mejor los que miran al futuro porque el valor depende de los resultados que vendrán.
}

\subsubsection{Comparables: ¿qué estiman las cotizadas y las transacciones?}
Como el múltiplo resume crecimiento, rentabilidad y riesgo, las comparables deben parecerse a B en los tres, y no basta con que sean del mismo sector. Se elige un grupo de empresas del sector de B y de sectores próximos, tantas como sea posible y de tamaño y estructura de capital parecidos, y si son pocas o distintas se corrigen las diferencias, con criterios como el tamaño, los mercados geográficos, el número de empleados o las ventas, o con regresiones que relacionan el múltiplo con el crecimiento y el riesgo de cada empresa.\footnote{%
    En la práctica se calculan la media, la mediana y el rango de cada múltiplo del grupo, se aplican a la magnitud de B y se obtiene un intervalo de valores, porque las condiciones del mercado y la escasez de comparables impiden una cifra precisa. Cada múltiplo da una estimación distinta: si convergen, la confianza aumenta; si divergen, hay que juzgar cuál refleja mejor el negocio, y a veces se combinan en un múltiplo sintético, una media simple o ponderada de varios. La distancia entre B y sus comparables puede medirse también con métodos de frontera, como el análisis envolvente de datos, que compara cada empresa con las que mejor convierten sus recursos en resultados.
}

Las comparables pueden ser de dos clases, que estiman cosas distintas. Las \textbf{cotizadas} se valoran a precios de participaciones minoritarias, que no dan el control: su múltiplo estima lo que vale B por sí sola, $V_B$, y para llegar al precio hay que sumarle una prima, que se estima con las pagadas en operaciones parecidas y recientes, porque las primas cambian con el tiempo y con el ciclo. Las \textbf{transacciones comparables}, adquisiciones de empresas parecidas, incluyen ya la prima que pagaron sus compradores, $\pi_c$, y su múltiplo estima directamente un precio, $V_B+\pi_c$. Como ese precio queda respaldado por operaciones del mercado, usarlas reduce además el riesgo de que los consejos de las dos empresas sean demandados por él.\footnote{%
    Los datos de las transacciones proceden de bases de datos especializadas, que buscan un equilibrio entre la confidencialidad de las operaciones y el valor que su información tiene para todo el mercado; la muestra debe ser amplia, pero de empresas del mismo sector o de uno muy próximo, y si no hay transacciones suficientes se recurre, como segunda opción, a las de sectores parecidos. El método supone que el mercado de adquisiciones valoró bien a las empresas compradas, y conviene tener en cuenta las condiciones financieras y la estructura del sector en que se cerraron.
}

\imagenfit{b2_comparables.pdf}{Lo que estiman las comparables cotizadas y las transacciones comparables}{Elaboración propia. Esquema ilustrativo.}

Pero la prima de las transacciones es la que justificaban las sinergias de otros compradores en las condiciones de su momento, no la de A. Para A, el máximo sigue siendo $V_B+S$, y pagar el múltiplo de las transacciones solo le conviene si su sinergia es al menos tan grande como la prima que pagaron otros, $S\geq\pi_c$.

Si B cotiza, el mercado da además su propio precio. Los métodos son los mismos que para una empresa que no cotiza; lo que añade la cotización es una estimación del mercado con la que contrastarlos: la capitalización bursátil es lo que valen las acciones de B por sí sola, para inversores que solo compran participaciones minoritarias, y sumándole la deuda neta se obtiene $V_B$. La prima se mide sobre esa cotización antes de que se conozca la operación.\footnote{%
    La cotización de la empresa comprada suele subir antes del anuncio, por los rumores y por las compras de quienes lo anticipan, y por eso la prima se mide sobre el precio de varias semanas antes. Esa subida previa no reduce lo que el comprador paga después del anuncio, sino que se suma a ello: es un coste adicional para el comprador (SCHWERT, 1996).
}

\subsubsection{Alcance: ¿cuándo fiarse del mercado?}
La ventaja de los múltiplos es que los datos están disponibles y que el valor procede del mercado, no de supuestos del analista que pueden alejarse de lo que alguien está dispuesto a pagar. Su debilidad es la misma: suponen que el mercado valora bien a las comparables, y si las sobrevalora, el múltiplo traslada el error al precio de B. Es lo que ocurre en los máximos de las olas, cuando las cotizaciones altas justifican precios altos y el comprador puede pagar con acciones sobrevaloradas, como vimos al estudiar por qué se concentran las operaciones. Además, un múltiplo resume la empresa en una cifra y no permite modelizar lo que cambiará en manos de A, que es precisamente la sinergia. Por eso los múltiplos sirven sobre todo para contrastar el descuento de flujos: una valoración muy alejada de la de las comparables obliga a explicar qué tiene B que no tengan ellas.

\subsection{Patrimonio y suma de partes: ¿qué valen los activos y las divisiones?}\label{b2_partes}
El descuento de flujos y los múltiplos valoran la empresa por lo que generará; los métodos de patrimonio, por lo que tiene. No miden el valor de un negocio en marcha, salvo cuando sus activos son el negocio, como en las sociedades de cartera o las inmobiliarias, que se valoran por el valor de mercado de lo que poseen, pero marcan el suelo de su valor. La suma de partes aplica los métodos anteriores negocio a negocio: valora por separado cada división de una empresa diversificada y permite ver si el conjunto vale lo que ellas.

\subsubsection{Patrimonio: ¿qué suelo pone lo que la empresa tiene?}
El \textbf{valor contable} de las acciones, el patrimonio neto, recoge lo que costaron los activos, menos las deudas y corregido por amortizaciones y deterioros, pero no lo que rinden. Si los activos y las deudas se llevan a su valor de mercado se obtiene el \textbf{valor contable ajustado}, y si se calcula lo que se obtendría vendiendo los activos por separado, una vez pagadas las deudas y los costes del cierre, como las indemnizaciones a la plantilla, el \textbf{valor de liquidación}.

El puente entre el valor contable y el valor de las acciones es el beneficio residual: la empresa vale su patrimonio neto más el valor actual de los beneficios que espera obtener por encima de lo que exigen sus accionistas ([\authorfont{Ver Tema 3.B.1}]). Si la rentabilidad financiera es constante, la empresa no crece y reparte todo su resultado, el valor de las acciones es (modelo del beneficio residual de OHLSON, 1995, con rentabilidad constante, sin crecimiento y con reparto de todo el resultado):
$$E=PN+\frac{\left(ROE-k_E\right)PN}{k_E}=PN\,\frac{ROE}{k_E}$$
donde $PN$ es el patrimonio neto, $ROE$ la rentabilidad financiera y $k_E$ la que exigen los accionistas. La empresa solo vale su valor contable si rinde exactamente lo que exigen sus accionistas; si rinde más, vale más, y si rinde menos, vale menos, hasta llegar al valor de liquidación, por debajo del cual a sus dueños les conviene más liquidarla que seguir.

\imagenfit{b2_patrimonio.pdf}{Valor de las acciones, valor contable y valor de liquidación}{Elaboración propia a partir de OHLSON (1995). Esquema ilustrativo.}

El valor de liquidación es, por tanto, el \textbf{suelo} del valor de B: el de una empresa que sigue funcionando es mayor, porque si no lo fuera convendría liquidarla, y ningún vendedor aceptará menos (FERNÁNDEZ, 2007). Quien compra una empresa que vale poco más que sus activos por separado compra, en el fondo, esos activos, y puede ganar vendiéndolos a quien les saque más rendimiento.\footnote{%
    Lo que el comprador paga por encima del valor razonable de los activos identificables de B, netos de sus deudas, se registra en sus cuentas consolidadas como \textbf{fondo de comercio} (NIIF 3, párrafo 32, que en la Unión recoge el Reglamento (UE) 2023/1803): recoge el valor de B como negocio en marcha que no figuraba en su balance, la sinergia y lo que se haya pagado de más. Su tratamiento contable y fiscal se estudia en el anexo dedicado al marco institucional.
}\textsuperscript{,}\footnote{%
    Los métodos mixtos, que se usaron mucho y hoy apenas se emplean, suman al activo neto corregido, $A$, un fondo de comercio igual al valor actual de unos pocos años de superbeneficio: el beneficio, $B$, menos lo que rendiría el activo neto a un tipo de interés de referencia, $i$. En el método abreviado de la Unión Europea de Expertos Contables, $V=A+a_n\left(B-iA\right)$, donde $a_n$ es el valor actual de una renta de $n$ años, entre cinco y ocho. Son una versión del beneficio residual con un horizonte y unos tipos fijados de forma arbitraria (FERNÁNDEZ, 2007).
}

\subsubsection{Suma de partes: ¿vale el conglomerado lo que sus divisiones?}
Cuando una empresa reúne negocios distintos, valorarla como un todo, con unos flujos consolidados y una sola tasa de descuento, mezcla riesgos y crecimientos que no se parecen. La \textbf{suma de partes} valora cada división por separado, con el método y la tasa que corresponden a su negocio, como exige la falacia del WACC que vimos al descontar los flujos, y suma los resultados; al total se le restan el valor actual de los costes de la sede central, que no pertenecen a ninguna división, y la deuda neta (panel a; suma de partes, elaboración propia a partir de KOLLER, GOEDHART y WESSELS, 2020):\footnote{%
    El procedimiento tiene cuatro pasos: identificar los componentes con peso en el valor, sean divisiones, unidades de negocio o activos concretos; analizar los flujos, el riesgo y el crecimiento de cada uno; valorarlos, por descuento de flujos o por múltiplos, con su propia tasa y las condiciones de su mercado; y sumarlos. No es un método distinto de los anteriores, sino su aplicación por partes.
}
$$E_{SP}=\sum_{j=1}^{n}V_j-V_{CC}-D$$
Comparar la suma de partes con lo que el mercado paga por la empresa, $E$, mide el \textbf{descuento del conglomerado}, que adelantamos al tratar la diversificación. Las empresas diversificadas cotizan de media por debajo de lo que valdrían sus divisiones valoradas con los múltiplos de empresas especializadas, aunque menos cuando sus negocios están relacionados. Parte del descuento puede deberse a que los inversores, que reciben menos información de cada división que de una empresa especializada, no perciben bien su valor; y parte, a cómo reparten el capital, invirtiendo de más en las divisiones con malas oportunidades con los fondos de las buenas, el riesgo del mercado interno de capital que vimos.\footnote{%
    BERGER y OFEK (1995) valoraron cada división de las empresas estadounidenses diversificadas con el múltiplo mediano de las empresas especializadas de su sector: entre 1986 y 1991, las diversificadas valían de media entre un $13$ y un $15\%$ menos que la suma de sus partes, con un descuento menor cuando sus divisiones pertenecían al mismo sector. La sobreinversión y las subvenciones cruzadas entre divisiones explicaban parte de la pérdida, y los ahorros fiscales de la diversificación solo la compensaban en pequeña medida.
}

Que el descuento mida valor destruido por diversificar está, sin embargo, en discusión. Las empresas que se diversifican no son una muestra al azar: si lo hacen porque sus negocios han perdido oportunidades, ya cotizarían con descuento sin diversificarse, y el descuento mediría el motivo de la diversificación, no su efecto. Al tener en cuenta esa elección, el descuento disminuye y en algunas estimaciones desaparece o se convierte en prima; y con datos de los establecimientos de cada empresa, en lugar de los segmentos que ella misma declara, una muestra que con estos daba descuento muestra una prima.\footnote{%
    CAMPA y KEDIA (2002) tienen en cuenta la decisión de diversificarse con datos de 1978 a 1996 y tres métodos econométricos distintos; VILLALONGA (2004) usa los datos de establecimientos del censo estadounidense (\textit{Business Information Tracking Series}).
}

Para un comprador, el descuento, sea cual sea su causa, es una oportunidad: quien compra el conglomerado a su precio de mercado y vende sus divisiones a empresas especializadas se queda la diferencia, como hicieron las tomas de control de los años ochenta ([\authorfont{Ver Anexo \ref{anx:ev_olas}}]). Para el propio conglomerado es una razón para desinvertir, vendiendo divisiones o separándolas en sociedades independientes, como veremos al tratar las operaciones. Pero la suma de partes valora las divisiones con precios de mercado, y estos pueden equivocarse, incluso en la dirección contraria, como muestra el caso de 3Com y Palm (panel b).

\begin{specialblock}{\textbf{Material complementario:} 3Com y Palm, una suma de partes negativa}
    La suma de partes tiene un límite inferior que no depende de ningún modelo. Si una matriz va a repartir entre sus accionistas $x$ acciones de una filial por cada acción propia, el precio de su acción, $p_M$, es el de esas acciones de la filial, $x\,p_F$, más el del resto de sus negocios, $R$; y como la responsabilidad de los accionistas es limitada, el resto no puede valer menos que cero, de modo que $p_M\geq x\,p_F$. Si no se cumple, el mercado viola la ley del precio único: hay dos formas de comprar la filial y una es más cara que la otra.

    El 2 de marzo de 2000, en plena euforia por las empresas tecnológicas, 3Com, una empresa rentable de equipos y servicios de redes, sacó a bolsa una pequeña parte de su filial Palm, que fabricaba ordenadores de bolsillo, con la idea de que el mercado reconociera su valor\textcolor{orange}{\textsuperscript{[Probable]}}. Conservó el $95\%$ y anunció que antes de fin de año repartiría esas acciones entre sus accionistas, a razón de unas $1{,}5$ acciones de Palm por cada acción de 3Com. Al cierre del primer día, Palm cotizaba a $95{,}06$ dólares, de modo que cada acción de 3Com debía valer al menos $145$, con la proporción exacta de $1{,}525$; pero cotizaba a $81{,}81$, después de cerrar a $104{,}13$ la víspera. El resto de 3Com, que además tenía más de $10$ dólares por acción en efectivo y valores, valía para el mercado $-63$ dólares por acción, unos $-22$ mil millones de dólares en total.

    No era un error difícil de ver, y la prensa lo explicó al día siguiente, pero duró meses. Aprovecharlo exigía comprar 3Com y vender al descubierto acciones de Palm, y las que podían tomarse prestadas para venderlas eran pocas y caras. Mientras tanto, los inversores pagaron más de $2{,}5$ mil millones de dólares por acciones de Palm que podían obtener más baratas comprando 3Com. No fue un caso aislado: de $18$ salidas a bolsa parciales seguidas de un reparto anunciado entre 1996 y 2000, seis dieron un resto negativo, todas de empresas tecnológicas (LAMONT y THALER, 2003; THALER, 2016). El mercado puede valorar mal las partes, y el error puede persistir si corregirlo es caro: una advertencia para quien valora con precios de mercado.
\end{specialblock}

\subsection{Opciones reales: ¿cuánto vale poder decidir más tarde?}\label{b2_opciones}
Los métodos anteriores valoran la empresa con un plan dado. Pero quien la compra puede adaptar sus decisiones a lo que vaya sabiendo: ampliar lo que funciona, abandonar lo que no o esperar. Esa flexibilidad tiene un valor que el descuento de flujos con un plan fijo no recoge, y que las opciones reales miden ([\authorfont{Ver Tema 3.B.2}]). En una adquisición aparece en dos lugares: en la empresa comprada, cuyo valor puede depender de oportunidades que aún no ha aprovechado, y en la propia operación, que puede hacerse por etapas.

\subsubsection{Opciones: ¿qué flexibilidad hay en una adquisición?}
Muchas empresas valen sobre todo por sus \textbf{opciones de crecimiento}. Una empresa de robótica como B vale en buena parte por los productos que podrá desarrollar si su tecnología funciona, que hoy no generan flujos, y descontar los flujos esperados las infravalora, porque trata como un compromiso lo que es un derecho: B solo invertirá en ellos si las cosas van bien. Al valorarlas hay que preguntarse, como con la sinergia, de quién son. Las que podría ejercer cualquier propietario forman parte de $V_B$, y el vendedor las cobrará; las que solo puede ejercer A, porque necesitan sus clientes o sus instalaciones, forman parte de $S$, y A no debería pagarlas.

La propia compra puede ser también una opción. En lugar de comprar B hoy, A puede tomar una participación en ella o crear con ella una empresa conjunta, con derecho a comprar el resto más adelante a un precio pactado, $P$. Si entretanto se confirma que la unión crea valor, compra; si no, renuncia, y solo pierde lo que pagó por la participación, $c$. Comprar hoy da el valor actual neto de la compra, sea positivo o negativo (panel a); comprar por etapas da, en $T$, solo su parte positiva (panel b; la empresa conjunta como opción de compra, KOGUT, 1991):
$$C_T=\max\left\{V_{B,T}+S_T-P,\;0\right\}$$
La opción vale más cuanto mayor es la incertidumbre sobre lo que vale la unión, porque A solo compra en los estados favorables: la incertidumbre, que en el descuento de flujos resta valor, aquí lo añade. Por eso la compra por etapas conviene más cuanto más incierta es la operación, como la entrada en un sector o en un país que el comprador no conoce.\footnote{%
    KOGUT (1991) lo contrastó con $92$ empresas conjuntas de la industria manufacturera: un crecimiento inesperado del mercado de sus productos aumentaba la probabilidad de que uno de los socios comprara la empresa conjunta, mientras que una caída inesperada no aumentaba la de disolverla. Es la asimetría propia de una opción de ampliar, que solo se ejerce cuando las noticias son buenas.
}

Lo que se pierde no son solo flujos: si otro comprador puede adelantarse, esperar puede costar la operación misma, y la competencia entre compradores que vimos al estudiar el precio reduce el valor de la opción.

Para calcularla, el modelo de BLACK y SCHOLES (1973) supone que el valor del subyacente sigue una distribución lognormal, es decir, que sus rentabilidades son normales, y que la opción solo puede ejercerse al vencimiento. Las opciones de una adquisición pueden ejercerse a menudo en cualquier momento y se encadenan unas con otras, y por eso en la práctica se valoran con árboles binomiales, que admiten el ejercicio anticipado ([\authorfont{Ver Tema 3.B.2}]).

La analogía tiene, además, límites propios. El precio de ejercicio de una opción financiera es fijo, y el de una real puede variar, con los precios de los proveedores o la disponibilidad de mano de obra en una inversión, o por una renegociación en una compra; el valor del subyacente no se observa en un mercado, sino que se estima; y su volatilidad no es ajena a la empresa, que puede reducirla, cubriendo por ejemplo el riesgo de tipo de cambio, o alterarla con su conducta ante los competidores. Las opciones reales son, por eso, más una forma de razonar que un cálculo preciso: obligan a preguntarse qué decisiones quedan abiertas y cuánto vale poder tomarlas más tarde, y en ellas pesa, como en los demás métodos, el juicio del analista.

\section{Crecimiento inórganico: Modificaciones estructurales}
\puntosclave{
    El crecimiento inorgánico se puede definir en contraposición al orgánico: busca alcanzar el tamaño óptimo de la empresa a través de capacidades y recursos externos y mediante operaciones corporativas de fusión o adquisición.
    
    Las operaciones de M\&A se pueden clasificar como horizontales, verticales o conglomerados (según la relación entre los negocios) y como hostiles o amistosas (según la actitud de los equipos directivos). 
    
    Las fusiones y adquisiciones tienden a producirse en oleadas: influyen factores microeconómicos (shocks específicos) y macro-financieros (valoraciones del mercado bursátil). 
    
    Entre los principales determinantes microeconómicos encontramos la necesidad de crecer más rápido de lo que el crecimiento orgánico permite, las sinergias fruto de una mejora en la eficiencia corporativa y los motivos de agencia (incentivos personales de los directivos) 
    
    Las operaciones de fusión y adquisición entre compañías se pueden estructurar a través de un canje de acciones o mediante un pago en efectivo (con la consiguiente necesidad de financiación). En la elección entre ambas alternativas influyen elementos corno las valoraciones bursátiles o el reparto de riesgo y rentabilidad entra los accionistas de la empresa adquirida y la adquirente 
    
    La cooperación y las alianzas estratégicas se constituyen como una alternativa al crecimiento externo a través del M\&A. Pemliten una mayor flexibilidad para la cooperación empresarial pero a su vez pueden enfrentar problemas de gobernanza
}

Para acompañar adecuadamente este tema, lo mejor es imaginarse que somos el gerente de una empresa. Como gerentes tendremos diversos objetivos, pero, indudablemente, uno de ellos será o estará de forma indirecta relacionado con aumentar el valor de la empresa:

\eqblock{
\begin{aligned}
    &\text{OBJETIVO}: \text{Crecimiento}\equiv g\quad \Longrightarrow\quad \underset{\text{\scriptsize Las empresas se financiancian para \textbf{crecer}}}{\boxed{g=\underbrace{\left[\text{ROA}+(\text{ROA}-i)\frac{D}{E}\right]}_{= \text{ROE}}(1-d)}}
\end{aligned}
}{Objetivo: Crecimiento de la empresa y descomposición. Fuentes de financiación}

Una de las formas que tenemos para aumentar el valor de nuestra empresa es a través de la inversión en nuestro propio ecosistema. Es decir, llevar a cabo operaciones que aumenten el tamaño de nuestra empresa a través de la expansión de sus operaciones existentes. Es lo que se conoce como crecimiento orgánico.

En contraposición, el crecimiento inorgánico busca alcanzar los mismos objetivos pero a través de la adquisición completa de otras empresas, la adquisición de divisiones de negocio y/o de activos específicos. Los motivos que llevan a optar por el crecimiento inorgánico (externo) son variados como veremos después, pero la diferencia clave con el crecimiento interno es la \textbf{velocidad en su ejecución}.

En este tipo de operaciones, por tanto, no nos interesará tanto el rendimiento esperado de la inversión (puesto que responde al plan estratégico individualizado de cada empresa) sino la valoración de la empresa objetivo. Pues será determinante para asegurar el rendimiento de la operación y, en todo caso, escoger entre alternativas de financiación.\footnote{%
    En el lenguaje de las normas internacionales de contabilidad (NIIF). se denominan combinaciones de negocios y de manera genérica en el ámbito del análisis financiero, se denomina actividad de M\&A (Mergers \& Acquisitions).
}

\textbf{¿Qué queremos decir?} -- Por tanto, sabiendo cuál es nuestro ojetivo como gerentes, y el objeto del análisis que emprendemos (el crecimiento inorgánico), lo primero que debemos saber es las opciones de las que disponemos, tanto desde un punto de vista jurídico como económico. Esto es esencial puesto que conocer el tipo de operaciones y sus particularidades permite tomar mejores decisiones atendiendo a las circunstancias que nos ocupan.

\subsection{Operaciones con alteración Patrimonial: Fusiones, Adquisiciones y Reestructuraciones}
El derecho mercantil y fiscal en las distintas jurisdicciones tiene una influencia clave en los negocios jurídicos que se utilizan para estructurar estas operaciones. Sin embargo, optamos por un enfoque estríctamente económico-financiero, más adecuado en nuestro caso. Estríctamente hablando, crecer externamente puede realizarse por dos vías: a través de la fusión con otras empresas y a través de la adquisición de estas.

\subsubsection{Fusiones: Consolidación de Patrimonios}
Una fusión es una operacion de concentración por la cual dos sociedades independientes consolidan sus patrimonios y actividades bajo una sola entidad. Se trata de operaciones en las que habitualmente el tamaño de ambas compañías es similar o al menos muy significativo en términos relativos de una frente a la otra. 

En casi todas las jurisdicciones hay dos clases posibles de fusiones (incluido en el caso de España):
\begin{la}
    \item \textbf{Fusión por creación}. En la que nace una nueva empresa con personalidad jurídica propia extinguiéndose las de las anteriores; y,
    \item \textbf{Fusión por absorción}. En la que se mantiene la personalidad jurídica de la absorbente y se extingue la de la absorbida. 
\end{la}

\subsubsection{Adquisiciones: Compra Patrimonial}  
Por otro ldo, las adquisiciones concisten en la compra de todos o parte de los activos de otra compañía. Por tanto, se suele hablar de adquisición cuando el tamaño es clarámente desigual: una compañía grande adquiere a una más pequeña.\footnote{%
    En ocasiones, la diferencia con una fusión por absorción puede no estar clara, pero se trata de diferencias jurídicas en la forma de la transacción. En el caso español por ejemplo, mientras que las fusiones por absorción se denominan con ese nombre, la adquisición, entendida como la compra de todo el patrimonio de una compaía por otra, se regula en la figura de la cesión global de activo y pasivo.
} De acuerdo con la actitud de los equipos directivos las operaciones pueden categorizadas como:
\begin{la}
    \item \textbf{Carácter hostil}. Intentos de adquirir una empresa en contra de la opinión de sus directivos. Esta reacción puede deberse al precio ofertado o a razones de otra índole (en ocasiones vinculadas al futuro personal de los. propios empleados);
    \item \textbf{Carácter amistoso}. Cuando cuenta con el respaldo de los directivos de ambas empresas. No obstante, esto no es una garantía de que la operación vaya a llevarse a cabo, dado que los accionistas pueden no estar de acuerdo con las condiciones. ofertadas o con la operación en términos generales.
\end{la}

\paragraph{Empresas cotizadas: Oferta Pública de Adquisición}
Cuando la compañía adquirida es una empresa cotizada, el adquirente tiene que llevar a cabo una oferta pública de adquisición (OPA), cumpliendo con los requisitos que establecen sobre estas ofertas las distintas legislaciones mercantiles, de competencia y supervisión del mercado. 

En las cotizadas, las principales diferencias entre la fusión y la adquisición radica en que en una adquisición los accionistas pueden ser apelados directamente por la empresa adquirente, sin necesidad del consentimiento de los directivos de la empresa objetivo y siempre de acuerdo con la legislación correspondiente. En las fusiones sin embargo, primero hay una negociación entre los directivos y una vez alcanzado el acuerdo este debe ser confirmado por los accionistas.

\paragraph{Conglomerados empresariales: \textit{Holdings}}
Un tipo particular de operación de adquisición que puede resultar en patrimonios diferenciados son los conglomerados empresariales o holdings. Es cada vez más recuente encontrarnos con empresas que pertenecen a una empresa matriz cuyo objeto social está tremendamente diseminado: una misma empresa es propietaria de una serie de empresas con procesos productivos distintos y no (necesariamente) relacionados ni con el negocio principal de la compañía ni con el negocio de las subempresas que la conforman. 

Las operaciones financieras de un holding son dificilmente encuadrables dentro del marco tradicional de la Teoría de la Organización Indutrial o de la empresa ya que este holding, que puede llevar a cabo las operaciones con o sin ánimo de control, persigue más bien el objetivo de tenencia de una cartera de empresas, no de una actividad productiva concreta.\footnote{%
    Las operaciones en las que entra un conglomerado o holding suelen ser más baratas que las fusiones o adquisiciones porque no suele existir una prima de control y no es necesario hacerse con la mayoría o totalidad del capital. El sentido económico de estos conglomerados en ocasiones es puesto en duda, al tratarse de instrumentos de concentración de poder político-económico que no necesariamente responden a razones de eficiencia financiera y operativa. Por lo general, cuando una empresa con un "core business" bien definido, comienza a adquirir paquetes de acciones y en el mercado se percibe que se está convirtiendo en un holding, pasa a cotizar con un descuento por la percepción de que el inversor es más eficiente diversificando.
} 


\subsubsection{Crecimiento negativo: Desinversión}
Los procesos de crecimiento empresarial pueden tener ambas direcciones y en ocasiones las empresas buscan reducir su tamaño, dando lugar a procesos de desinversión. 

Estos procesos pueden ser necesarios por diversas razones, como cambios en la estrategia corporativa, necesidades de liquidez, reestructuración o eliminación de divisiones no rentables, entre otros. En estas circunstancias, las empresas pueden decidir eliminar operaciones completas o cerrar divisiones específicas que no son viables, lo que podría implicar la liquidación de activos y costosos procesos de destrucción de empleo (liquidación o restructuración). Por ello, en estos casos es mucho más habitual recurrir al (de)crecimiento externo, por la facilidad para deshacerse de divisiones de negocio completas mediante \textbf{operaciones corporativas}. 

Además de procesos de venta, también se recurre a \textit{spin-offs} (escisiones de activo y pasivo). Además de ser un medio para alcanzar el tamaño óptimo empresarial y reasignar capital, estas operaciones de desinversión también pueden ser una fuente de información para el conjunto del mercado. \subsection{Operaciones sin alteración Patrimonial: Alianzas y cooperaciones}
También podemos observar operaciones sin alteración patrimonial que se llevan a cabo con el objetivo de aumentar el crecimiento de la empresa: las alianzas estratégicas o las cooperaciones. Al no haber alteraciones en el Patrimonio este se mantiene claramente diferenciado. 

Se trata de una parada intermedia entre crecimiento orgánico e inorgánico, en la que dos o más empresas deciden actuar conjuntamente para lograr determinados objetivos, compartiendo recursos, capacidades o actividades. Sin ánimo de ser exhaustivos, las razones que conducen a estas situaciones suelen tener que ver con: (i) cambios tecnológicos; (ii) acceso (recíproco, por ejemplo) a mercados tanto nacionales como internacionales; y, (iii) financiación compartida de grandes proyectos de inversión (UTE's, en España). 


Los acuerdos de cooperación pueden adoptar una personalidad jurídica propia que formaliza el compromiso entre los cooperantes: 
• Consorcio: Alianza entre varias empresas que mediante un contrato formaliza una relación a largo plazo entre cada una de las empresas y una organización conjuntaintegrada por ellas mismas. El objetivo es llevar a cabo una gran inversión que no podría efectuarse por las dos partes por separado.
• Joint-Venture: Dos o más empresas crean una tercera empresa independiente, concapital conjunto. Otra forma muy habitual de sellar estas alianzas estratégicas es mediante la adquisición cruzada de participaciones accionariales. La forma de estructurar estos acuerdos es variada pero suele implicar algún tipo de pacto en el que se refleje la gobernanza conjunta y los acuerdos de colaboración a nivel económico, técnico y financiero.

\begin{specialblock}{}
    Son muchos los ejemplo que podríamos poner en este campo, pero uno particularmente importante (e ilustrativo) se encuentra en el mundo de la aviación. 

    [Extraído de GAY DE LIEBANA, 2007] Son conocidos los corsés jurídicos sobre posibles fusiones entre compañías aéreas, donde de un lado existen normas restrictivas del transporte aéreo internacional que ponen barreras para que se lleven a cabo fusiones, integraciones y adquisiciones transfronterizas, a lo que se tiene que añadir el hecho de que sean bastantes los países que restringen la propiedad extranjera de las líneas aéreas. A pesar de ello el secreto para aumentar los ingresos y reducir los costes es muy sencillo: alianzas. Así, hoy, Oneworld donde participa la compañía de bandera española Iberia, entre otras, compite con SkyTeam, con Air France y Alitalia entre otras compañías, y Star Alliance, que cuenta, también entre otras, con la alemana Lufthansa, SAS y la española Spanair. ¿Las claves de esas tres grandes alianzas del tráfico áereo? El número de empleados -260.500, 186.000 y 284.475, respectivamente -, los destinos en total - 570, 512 y 700, respectivamente - el control de aeropuertos que constituyen referentes en el mundo de la aviación - Heathrow, Zurich, Madrid - y son hub, descollantes centros de distribución, con lo cual se planta cara al auge de las aerolíneas de vuelos baratos, a la vez que estas integraciones no causan la pérdida de "slots", es decir, de derechos de vuelo que suponen uno de los intangibles más apetecibles en el sector de la aviación comercial. El hecho de poder compartir estructuras, infraestructuras, servicios de handling, catering..., entre las compañías integradas en una misma alianza implica una mejora económica sustancial, aprovechando vuelos y recorridos así como las centrales de reservas.

    
\end{specialblock}



\paragraph{Ventajas: Riesgo y autonomía}
Por lo que respecta a las ventajas que estas operaciones suponen frente a las convencionales adquisiciones y/o fusiones son:
\begin{lnum}
    \item \textbf{Compartir riesgos}. Al cooperar o formar alianzas estratégicas, las empresas pueden compartir los riesgos y costos asociados con nuevas iniciativas o proyectos conjuntos;
    \item \textbf{Recursos}. Mediante estas operaciones, las empresas pueden acceder a los recursos, capacidades y conocimientos técnicos de sus socios, lo que les permite ampliar su alcance y competir de maneramás efectiva en el mercado;
    \item \textbf{Sinergias}. Estas operaciones pueden conducir a la creación de nuevos productos o servicios innovadores y a la expansión de su base de clientes; y,
    \item \textbf{Flexibilidad y autonomía}. Sobretodo, a diferencia de las fusiones y adquisiciones, permiten a las empresas mantener su independencia y autonomía jurídico-financiera, al conservar su identidad y estructura organizativa, mientras colaboran en áreas específicas de interés común.
\end{lnum}


\paragraph{Inconvenientes: }
Por lo que respecta a los inconvenientes, estas operaciones suelen tener un impacto desigual para los socios. Además, como cualquier otra interacción social, estas sujetas a riesgos inherentes a la conducta, como por ejemplo:
\begin{lnum}
    \item \textbf{Complejidad de la colaboración}. La colaboración entre empresas implica la gestión de relaciones complejas. Las diferencias en las culturas organizativas, los objetivos estratégicos y las prácticas comerciales pueden dificultar la cooperación efectiva y requerir una atención constante que puede drenar demasiados recursos de ambas organizaciones;
    \item \textbf{Rendimiento y expectativas}. En muchas ocasiones el éxito depende del desempeño y compromiso de los socios involucrados. Si uno de los socios no cumple con sus responsabilidades o no aporta valor esperado, puede afectar negativamente la alianza en su conjunto; y,
    \item \textbf{Compromisos}. Llegar a un acuerdo mutuamente beneficioso puede ser un desafio en las alianzas. Las empresas deben negociar y acordar términos y condiciones que satisfagan a ambas partes, lo que puede llevar tiempo y esfuerzo. Además, los intereses pueden cambiar a lo largo del tiempo, generando una tensión para diverger o diferenciarse de nuevo.
\end{lnum}

\subsection{Clasificaciones accesorias}
Además de las dos grandes categorías presentadas, existen otras formas de clasificar las operaciones de crecimiento externo. Sobre este aspecto, presentaremos principalmente dos clasificaciones: por las formas de adquisición y por el sector operativo de la empresa adquirida.

\subsubsection{Cadena de valor: Relación de negocios}
Desde una óptica de Política Económica y la valoración de las consecuencias de una fusión es importante tener en cuenta la posición relativa en la cadena de valor.

\paragraph{Intrasectoriales: Horizontal}
De acuerdo con esta relación, una operación se considera intrasectorial, u de integración horizontal, si se realiza entre empresas que pertenecen al mismo proceso productivo, habitualmente competidores. Las razones que llevan a este tipo de operaciones se analizan posteriormente en los determinantes del crecimiento inorgánico. 

\paragraph{Intersectoriales: Vertical}
Si la operación se lleva a cabo entre empresas con un proceso productivo diferente, u objeto social diferente, hablamos de integración vertical u intersectorial (a lo largo de la cadena de valor). Por ejemplo, un proveedor o un distribuidor. 

De esta forma, si el adquirente compra un objetivo que está por delante de él en la cadena de valor (un proveedor), se denomina integración hacia atrás (backward) y cuando un adquirente compra una empresa que está más abajo en la cadena de valor (un distribuidor), se denomina integración hacia delante (forward).

\subsubsection{Métodos de pago y financiación: Operativización}
Además, las operaciones pueden financiarse con efectivo, valores o una combinación de ambos en lo que se denomina una oferta mixta.

\paragraph{Oferta de efectivo}
En una oferta en efectivo, el efectivo puede proceder de los activos existentes de la empresa adquirente o de una emisión de deuda. De esta forma, el principal método para realizar una operación en efectivo es mediante préstamos puente, considerando como estos:\footnote{%
    Se trata de un préstamo temporal cuyo objeto es cubrir la necesidad inminente de financiación, pero sin una vocación de pemianencia, siendo necesario la búsqueda de una financiación a largo plazo definitiva. Permite conseguir los fondos con agilidad pero requiere una refinanciación.
}
\begin{lnum}
    \item \textbf{Ampliación de capital}. La empresa adquirente realiza una emisión de nuevas acciones. Es la opción con más riesgo para los bancos que participan en el préstamo puente y es habitual un sondeo previo del interés por participar en dicha ampliación;
    \item \textbf{Financiación a largo plazo}. En el menor plazo posible se estructura una financiación normalmente sindicada, que permita amortizar el préstamo puente. Los bancos que participan en la financiación sindicada a largo no tienen porque coincidir con los del préstamo puente. Como alternativa a la financiación bancaria la empresa puede emitir bonos, pero esto suele ser menos frecuente en operaciones de gran volumen; o,
    \item \textbf{Venta de activos}. Es un mecanismo cada vez más habitual por el que parte del préstamo puente se repaga con la venta de activos o filiales no consideradas estratégicas por la nueva dirección. También se puede estructurar mediante spin-offs que sacan a bolsa o venden parcialmente a terceros una participación en negocios estratégicos. Ha sido un mecanismo recurrente en el sector concesionario y constructor español con sus filiales de energías renovables.
\end{lnum}

\paragraph{Canje de acciones: Ecuación de canje}
La magnitud de las empresas cotizadas conlleva que la mayoría de estas operaciones se instrumentalicen a través de una fusión o canje de acciones.

En este contexto, es necesario determinar la \textbf{ecuación de canje}, es decir, la relación entre el número de acciones a emitir por cada acción de la otra empresa que se va a recibir. Esta ecuación de canje determinará a su vez el valor relativo de una compañía frente a otra, fruto del ejercicio de valoración realizado y del proceso de negociación de la operación. La ecuación de canje y el valor relativo están relacionados por la siguiente fónnula:

\eqblock{
\begin{aligned}
    \underset{\text{\scriptsize(Nº de accioens $A$ x 1 acción $B$)}}{\text{Ecuación de canje}}=\underset{\text{\scriptsize ($B$ vale $x$ veces $A$: bursátil)}}{\text{Ratio valor relativo}}\cdot\frac{\text{Nº acciones $A$ (pre-fusión)}}{\text{Nº acciones $B$ (pre-fusión)}}
\end{aligned}
}{}

Como se puede observar claramente, la ecuación de canje nos dará directamente el valor de mercado de las acciones de $A$ por cada acción de $B$. No obstante, cabe suponer que la ecuación de canje finalmente pactada (u ofertada) será mayor por la prima de control exigible por la empresa objetivo. De esta forma, podremos hallar la valoración relativa finalmente acordada a raíz de la ecuación de canje pactada.

\paragraph{Leverage Buy Out}
Además, es necesario destacar el papel cada vez más importante de las operaciones por compra apalancada. 

Muchas operaciones corporativas están protagonizadas por inversores financieros y fondos de capital riesgo, que aportan capital e impulsan cambios en la estrategia y dirección de las compañías. El mecanismo habitual al que recurren este tipo de agentes para financiar las fusiones y adquisiciones es el Leveraged Buyout (LBO) o compra apalancada. El objetivo de esta estructura es que la adquisición se financie endeudando (apalancando) a la propia compañía comprada, de tal modo que sean los propios recursos generados por la compañía comprada los que permitan repagar la financiación. Es frecuente que, una vez repagada la mayoría de la deuda, el inversor busque salir de la inversión (vendiendo la compañía o sacándola la bolsa, entre otras estrategias de salida). Al margen de la estrategia corporativa del adquirente, en última instancia se busca maximizar la rentabilidad del capital del inversor financiero.



\subsection{Empresas cotizadas: Capitalización bursátil}
Hasta aquí hemos visto las diferentes opciones metodológicas que tenemos para evaluar el enterprise value de una empresa no cotizadas. Esto nos permite, como ya hemos visto, alcanzar el equity value restando al enterprise value el valor correspondiente a los acreedores financieros.

En el caso de empresas cotizadas, el criterio de valoración es radicalmente diferente. ¿Por qué? Porque el valor de la empresa que le corresponde a los accionistas viene determinado por los mercados financieros. De esta forma, para observar este valor, analizar la rentabilidad de la operación y, en caso de que esta sea rentable, realizar la Oferta Pública de Adquisición, deberemos fijarnos en la \textbf{capitalización bursátil} de la empresa.

\subsubsection{Ecuación de canje: Valor relativo}
La magnitud de las empresas cotizadas conlleva que la mayoría de estas operaciones se instrumentalicen a través de una fusión o canje de acciones. 

En este contexto, es necesario determinar la \textbf{ecuación de canje}, es decir, la relación entre el número de acciones a emitir por cada acción de la otra empresa que se va a recibir. Esta ecuación de canje determinará a su vez el valor relativo de una compañía frente a otra, fruto del ejercicio de valoración realizado y del proceso de negociación de la operación. La ecuación de canje y el valor relativo están relacionados por la siguiente fónnula: 

\eqblock{
\begin{aligned}
    \underset{\text{\scriptsize(Nº de accioens $A$ x 1 acción $B$)}}{\text{Ecuación de canje}}=\underset{\text{\scriptsize ($B$ vale $x$ veces $A$: bursátil)}}{\text{Ratio valor relativo}}\cdot\frac{\text{Nº acciones $A$ (pre-fusión)}}{\text{Nº acciones $B$ (pre-fusión)}}
\end{aligned}
}{}

Como se puede observar claramente, la ecuación de canje nos dará directamente el valor de mercado de las acciones de $A$ por cada acción de $B$. No obstante, cabe suponer que la ecuación de canje finalmente pactada (u ofertada) será mayor por la prima de control exigible por la empresa objetivo. De esta forma, podremos hallar la valoración relativa finalmente acordada a raíz de la ecuación de canje pactada.

\anexo{Olas de fusiones y evidencia empírica}\label{anx:evidencia}
Este anexo desarrolla la evidencia que el primer bloque resume: cómo se mide si una operación crea valor y para quién, qué encuentran los estudios y de qué dependen sus resultados, cómo han sido las olas de fusiones desde finales del siglo XIX, por qué se producen y con qué indicadores se sigue el ciclo. Va fuera del cuerpo porque el bloque solo necesita de ella lo que contrasta sus teorías; aquí se ve de dónde salen esas conclusiones, con qué métodos y con qué datos.

\subsection{La medición: ¿cómo se sabe si una operación crea valor?}\label{anx:ev_medicion}
La sinergia de una operación no se observa, porque no sabemos qué habrían valido las dos empresas si no se hubieran unido. Hay dos formas de estimarla. La primera usa las cotizaciones: si el mercado valora bien, la reacción de las acciones al anuncio descuenta todo lo que se espera de la operación. La segunda usa las cuentas: compara los resultados de explotación de la empresa unida con los que obtenían las dos por separado. Cada una tiene su punto débil, porque el mercado puede equivocarse y las cuentas tardan en reflejar las ganancias y las mezclan con todo lo demás, y por eso se usan juntas.

\subsubsection{Estudios de eventos: ¿cómo reacciona el mercado al anuncio?}
Un estudio de eventos mide cuánto cambia el valor de una empresa por un hecho concreto, aislándolo de lo que habría ocurrido de todos modos (MACKINLAY, 1997). Se procede en tres pasos.

El primero es estimar la rentabilidad normal de cada acción, la que habría tenido sin el anuncio. Lo habitual es el modelo de mercado, estimado con las rentabilidades diarias de un periodo anterior que no incluya la operación, por ejemplo el año previo (modelo de mercado de los estudios de eventos, MACKINLAY, 1997):
$$R_{it}=\alpha_i+\beta_i\,R_{mt}+\varepsilon_{it}$$
donde $R_{it}$ es la rentabilidad de la acción $i$ el día $t$ y $R_{mt}$ la del índice de mercado.

El segundo es calcular la rentabilidad anormal de cada día de la ventana del evento, la diferencia entre la rentabilidad observada y la normal, $AR_{it}=R_{it}-\left(\hat\alpha_i+\hat\beta_i R_{mt}\right)$, y sumarla a lo largo de la ventana para obtener la rentabilidad anormal acumulada (misma referencia):
$$CAR_i\left(t_1,t_2\right)=\sum_{t=t_1}^{t_2}AR_{it}$$
Las ventanas cortas, de tres días en torno al anuncio, aíslan la noticia; las largas, desde unas semanas antes del anuncio hasta el cierre de la operación, recogen también la filtración previa y lo que el mercado aprende después.

El tercero es agregar. La ganancia en dinero de cada parte es su rentabilidad anormal por su valor antes del anuncio, y la suma de las dos, la rentabilidad del conjunto ponderada por el valor, estima la sinergia: es la contrapartida empírica de $S$, como la ganancia de la empresa comprada lo es de $\pi$ y la del comprador, de $VAN_A$. Con muchas operaciones se promedian los resultados y se contrasta si la media es distinta de cero.

El método descansa en tres supuestos, y cada uno tiene su problema. El primero es que el mercado valora bien: si sobrevalora en conjunto, como sostienen las explicaciones de las olas basadas en errores de valoración, la reacción mide lo que el mercado cree, no lo que la operación vale. El segundo es que el anuncio es una sorpresa: si el mercado esperaba la operación, la reacción solo recoge la parte inesperada, y en los compradores que hacen muchas adquisiciones el valor de todo el programa se incorpora a la cotización cuando se anuncia, no operación a operación (SCHIPPER y THOMPSON, 1983). El tercero es que el anuncio solo informa sobre la operación. En realidad revela también algo sobre el comprador, como que considera caras sus acciones cuando paga con ellas, y la reacción mezcla la sinergia con esa revisión de lo que vale el comprador por sí solo.\footnote{%
    Por eso las grandes pérdidas de los compradores entre 1998 y 2001 pueden reflejar en parte que los inversores revisaron a la baja el valor que atribuían a esas empresas por sí solas, y no solo una sinergia negativa, como reconocen MOELLER, SCHLINGEMANN y STULZ (2005).
}

A largo plazo, los problemas se multiplican. Las rentabilidades anormales de varios años dependen mucho del modelo de rentabilidad normal y de las empresas con que se comparan, y pequeños errores en la referencia se acumulan hasta producir anomalías aparentes que desaparecen al cambiarla (FAMA, 1998). Además, las operaciones se concentran en el tiempo y sus rentabilidades no son independientes; cuando se tiene en cuenta, buena parte de las rentabilidades anormales a largo plazo de los compradores deja de ser significativa (MITCHELL y STAFFORD, 2000)\textcolor{orange}{\textsuperscript{[Probable]}}.

\subsubsection{Estudios contables: ¿mejoran los resultados de explotación?}
Los estudios contables miden la ganancia que se materializa. Comparan el rendimiento de explotación de la empresa unida en los años posteriores a la operación con el que obtenían, sumadas, las dos empresas en los años anteriores, y corrigen ambos por el de su sector, para descontar lo que habría ocurrido de todos modos.

La medida más usada es el flujo de caja de explotación sobre el valor de los activos, porque no depende de cómo se contabilice la operación ni de cómo se financie: la amortización del fondo de comercio o los intereses de la deuda contraída para comprar cambian el beneficio sin que cambie la explotación (HEALY, PALEPU y RUBACK, 1992). El rendimiento anormal es la diferencia con la mediana del sector o con una empresa de control, y se compara su media en los años posteriores, normalmente tres, con la de los anteriores.

El problema es la referencia. Las empresas que se fusionan no son una muestra al azar: suelen ser las que mejor iban antes de la operación, y los rendimientos extremos tienden a volver hacia la media con fusión o sin ella. Si se comparan solo con su sector, esa vuelta a la media se confunde con el efecto de la operación; por eso la empresa de control debe elegirse también por su rendimiento anterior y por su tamaño (BARBER y LYON, 1996). Con esa corrección desaparece la mejora que habían encontrado los primeros estudios, como vimos al tratar el origen de las ganancias (GHOSH, 2001).

\subsubsection{De qué dependen las ganancias}
Las teorías del bloque predicen qué operaciones deberían salir mejor, y la evidencia permite contrastarlas en cuatro dimensiones.

\paragraph{Quién compra a quién: la reasignación del capital}
Si las adquisiciones trasladan el capital hacia quien mejor lo usa, la ganancia debería ser mayor cuando una empresa bien gestionada compra otra mal gestionada. La $q$ de TOBIN sirve de aproximación: una $q$ alta indica que el mercado valora mucho lo que la empresa hace con sus activos. LANG, STULZ y WALKLING (1989), con ofertas públicas de adquisición que salieron adelante, encontraron que los accionistas de los compradores con $q$ alta ganan bastante más que los de los compradores con $q$ baja, y que las empresas compradas con $q$ baja ganan más que las de $q$ alta; la ganancia conjunta es máxima cuando un comprador bien gestionado compra una empresa mal gestionada. SERVAES (1991) lo extendió a las fusiones, con $704$ operaciones entre 1972 y 1987: las ganancias de la empresa comprada, del comprador y del conjunto son mayores cuando la comprada tiene una $q$ baja y el comprador una $q$ alta. Y así suele ocurrir: en más de dos tercios de las fusiones estadounidenses desde 1973, la $q$ del comprador superaba a la de la empresa comprada (ANDRADE, MITCHELL y STAFFORD, 2001).

\paragraph{Cómo se paga: la señal de las acciones}
Cuando un comprador paga con sus acciones, le dice al mercado que las considera caras, igual que cuando emite acciones para financiarse ([\authorfont{Ver Tema 3.B.3}]); cuando paga en efectivo, transmite lo contrario. TRAVLOS (1987) encontró que la cotización de los compradores cae al anunciar ofertas pagadas con acciones y se mantiene en las pagadas en efectivo, tanto en las fusiones como en las ofertas públicas y con independencia de su resultado, y atribuyó la diferencia a la señal. La diferencia se agranda a largo plazo: con $947$ adquisiciones entre 1970 y 1989, los compradores que pagaron con acciones obtuvieron en los cinco años siguientes una rentabilidad anormal del $-25\%$, y los que compraron en efectivo mediante una oferta pública, del $61{,}7\%$ (LOUGHRAN y VIJH, 1997), con las cautelas sobre las rentabilidades a largo plazo que vimos. El patrón encaja también con los errores de valoración: quien paga con acciones suele ser quien las tiene más caras.\footnote{%
    Un caso conocido es la compra de Compaq por Hewlett-Packard, anunciada en septiembre de 2001 y pagada con acciones. Hasta la junta de marzo de 2002, la cotización de Hewlett-Packard cayó un $21{,}6\%$, y sus accionistas la aprobaron por un estrecho margen, frente a la oposición de las familias de los fundadores (CNNMONEY, 2002).
}

\paragraph{A quién se compra: el descuento por iliquidez}
FULLER, NETTER y STEGEMOLLER (2002) estudiaron empresas que hicieron al menos cinco adquisiciones en poco tiempo, de modo que las diferencias pueden atribuirse a la empresa comprada y a la oferta, y no al comprador. Los compradores ganan cuando compran una empresa no cotizada o la filial de otra, y pierden cuando compran una cotizada; la ganancia es mayor cuanto mayor es la empresa comprada y cuando se paga con acciones. Los autores lo atribuyen a un descuento por iliquidez, porque los dueños de una empresa no cotizada no pueden vender sus acciones con facilidad y aceptan menos, y a efectos fiscales y de control: cuando cobran en acciones, pasan a ser accionistas relevantes del comprador, con interés en vigilarlo\textcolor{orange}{\textsuperscript{[Probable]}}.

\paragraph{Quién compra: el tamaño del comprador}
Con más de doce mil adquisiciones de empresas cotizadas estadounidenses entre 1980 y 2001, la rentabilidad de los compradores pequeños en el anuncio supera en unos dos puntos porcentuales a la de los grandes, con cualquier forma de pago y tanto si la empresa comprada cotiza como si no (MOELLER, SCHLINGEMANN y STULZ, 2004), lo que es coherente con que la agencia y la arrogancia pesen más en las grandes empresas. Entre 1998 y 2001, los accionistas de los compradores perdieron $12$ centavos por cada dólar pagado, $240$ mil millones de dólares en total, frente a $7$ mil millones en toda la década de 1980; la pérdida se concentró en unas pocas operaciones de empresas con valoraciones muy altas, sin las cuales los compradores habrían ganado (MOELLER, SCHLINGEMANN y STULZ, 2005).

\subsubsection{El contraste de los motivos fuera de Estados Unidos}
El contraste de BERKOVITCH y NARAYANAN (1993) que vimos en el cuerpo se ha aplicado en otros mercados. En las adquisiciones parciales entre empresas cotizadas en la bolsa española, la ganancia de la empresa comprada aumenta con la total en las operaciones con ganancia positiva, como predice la sinergia, y no se relaciona con ella en las de ganancia negativa, como predice la arrogancia; la agencia, a diferencia de lo que ocurre en Estados Unidos, no aparece con claridad (FERNÁNDEZ y BAIXAULI, 2003).

\subsection{Las olas: ¿cómo han sido y qué dicen los datos?}\label{anx:ev_olas}
No hay una serie única de fusiones que cubra todo el siglo. Para Estados Unidos, NELSON (1959) reconstruyó las operaciones en la industria y la minería entre 1895 y 1920 y reprodujo otra serie de 1919 a 1939, que la Comisión Federal de Comercio (FTC) continuó desde 1940 hasta 1979; desde entonces solo hay series privadas, como las de las bases de datos comerciales, que difieren en su cobertura (GOLBE y WHITE, 1988). La serie pública que hoy permite comparar países, la de la UNCTAD, recoge solo las fusiones transfronterizas, desde 1990.

Con estas series, GOLBE y WHITE (1988) contrastaron si la actividad sigue un patrón de olas o se mueve al azar, como se había sostenido. Contaron las rachas, los periodos seguidos por encima o por debajo de la tendencia: si la actividad fuera aleatoria, en la serie anual de 1940 a 1985 habría que esperar unas $23$ rachas, y hubo $5$. Las olas existen, y son largas. Midieron, además, su tamaño en relación con la economía, dividiendo el número y el valor de las operaciones por el PNB real: la ola de finales del siglo XIX fue mucho mayor, en proporción, que la de los años sesenta o la de los ochenta.

En España, el número de compras en el extranjero de empresas españolas siguió las olas mundiales hasta 2007 y cayó con la crisis, mientras que las compras de empresas españolas por compradores extranjeros han crecido con fuerza desde 2014.

\subsubsection{Seis olas en Estados Unidos}
Cada ola tuvo un tipo dominante de operación, que respondió a las oportunidades de su tiempo y a lo que permitía la legislación de competencia. El cuadro resume sus rasgos, que se explican a continuación.

\paragraph{La primera ola: del cártel a la fusión}
Entre 1895 y 1904, el ferrocarril había creado mercados nacionales y las nuevas técnicas de producción en masa exigían grandes plantas, y en muchas industrias las fusiones reunieron a la mayoría de los productores en una sola empresa: la mayor fue United States Steel, en 1901. La Sherman Act de 1890 había prohibido los cárteles, pero no las fusiones, que se convirtieron en la forma legal de lograr lo mismo hasta que la sentencia del caso Northern Securities, en 1904, las sometió también a la ley\textcolor{orange}{\textsuperscript{[Probable]}}. La consolidación no era inevitable ni duradera: la mayoría de las nuevas empresas resultaron menos eficientes que los rivales que surgieron casi de inmediato y perdieron pronto su posición dominante (LAMOREAUX, 1985), lo que sugiere que buscaban sobre todo poner fin a la competencia en precios\textcolor{orange}{\textsuperscript{[Probable]}}.

\paragraph{La segunda ola: del monopolio al oligopolio}
Perseguidos los monopolios, las fusiones de los años veinte no reunieron a toda una industria, sino que formaron oligopolios de unas pocas empresas grandes, y muchas integraron verticalmente sus proveedores y sus redes de distribución: STIGLER (1950) la describió como una fusión para el oligopolio, frente a la fusión para el monopolio de la primera. Se alimentó del auge bursátil de la década y terminó con el crac de 1929.

\paragraph{La tercera ola: conglomerados y el juego del PER}
Desde la Celler-Kefauver Act de 1950, las autoridades perseguían las fusiones horizontales y verticales, y crecer comprando empresas de otros sectores se convirtió en la vía abierta. La ola de los conglomerados tiene dos lecturas. La primera es de eficiencia: en una época de mercados de capitales menos desarrollados, el conglomerado era un mercado interno de capital, y las ganancias de los compradores eran mayores cuando una empresa sin restricciones financieras compraba otra que las tenía (HUBBARD y PALIA, 1999). La segunda es de valoración: muchas de esas compras se pagaron con acciones de empresas con un PER alto que compraban otras con un PER bajo, lo que hacía crecer el beneficio por acción del comprador sin sinergia alguna, el llamado juego del PER (BREALEY, MYERS y ALLEN, 2020). Si A, con beneficios $E_A$ y un PER más alto, $PER_A$, compra con acciones, a su precio de mercado, a B, con beneficios $E_B$ y $PER_B$, el beneficio por acción de A sube; pero, si el mercado no se deja engañar, la empresa unida debe cotizar al PER medio ponderado por los beneficios (juego del PER, BREALEY, MYERS y ALLEN, 2020):
$$PER_{AB}=\frac{PER_A\,E_A+PER_B\,E_B}{E_A+E_B}$$
de modo que el precio de la acción no cambia. Si el mercado sigue aplicando $PER_A$ a toda la empresa, en cambio, la cotización sube y el comprador puede repetir la jugada con la siguiente compra, mientras el crecimiento aparente se mantenga. Es la sinergia percibida que formaliza el modelo de SHLEIFER y VISHNY (2003) que veremos, y la única ola que, según JOVANOVIC y ROUSSEAU (2002), no fue precedida por un aumento de la dispersión de la $q$ entre empresas. Terminó con la caída de la bolsa de 1969 y 1970 y con la reforma contable de 1970, que limitó la contabilización de las fusiones como simple agregación de los balances de las dos empresas, con la que el comprador evitaba reflejar el precio pagado\textcolor{orange}{\textsuperscript{[Probable]}}.

\paragraph{La cuarta ola: tomas de control y desmembración}
Los años ochenta deshicieron lo que habían hecho los sesenta. Inversores financiados con deuda, a menudo con bonos de alto rendimiento, compraban conglomerados que valían menos que la suma de sus negocios, los dividían y vendían las partes a empresas especializadas: como vimos, la mayoría de los activos acabó en empresas del mismo sector (BHAGAT, SHLEIFER y VISHNY, 1990). Las compras apalancadas llevaron al extremo la disciplina de la deuda ([\authorfont{Ver Tema 3.B.3}]), y la mayor, la de RJR Nabisco por KKR en 1989, simbolizó el final de la ola\textcolor{orange}{\textsuperscript{[Probable]}}, que llegó con el hundimiento del mercado de bonos de alto rendimiento.

\paragraph{La quinta ola: estrategia, globalización y acciones}
La de los noventa fue una ola de operaciones entre empresas del mismo sector, a menudo de países distintos, pagadas con acciones y concentradas en los sectores que se desregulaban, como las telecomunicaciones, la banca y la energía: la desregulación explica casi la mitad de la actividad desde finales de los ochenta (ANDRADE, MITCHELL y STAFFORD, 2001). Fue la primera ola verdaderamente europea, y sus mayores operaciones, como la compra de Time Warner por AOL o la de Mannesmann por Vodafone, ambas en 2000, coincidieron con el máximo de la burbuja tecnológica\textcolor{orange}{\textsuperscript{[Probable]}}.

\paragraph{La sexta ola y las siguientes}
Entre 2003 y 2007, la liquidez abundante y el crédito barato impulsaron las compras de los fondos de capital riesgo, financiadas con deuda, y las compras en efectivo. Los compradores estaban menos sobrevalorados en relación con sus objetivos, pagaron primas más bajas y hubo menos competencia por el control que en los noventa; aun así, destruyeron para sus accionistas tanto valor como entonces (ALEXANDRIDIS, MAVROVITIS y TRAVLOS, 2012). La crisis financiera la cortó en seco. Tras la recuperación, la actividad volvió a crecer, y en 2021, con tipos de interés muy bajos y bolsas en máximos, el valor mundial de las operaciones anunciadas alcanzó un récord en torno a $5{,}8$ billones de dólares, un $64\%$ más que el año anterior (REUTERS, 2021).

\subsubsection{Europa y España}
Las olas europeas siguieron a las estadounidenses con retraso y menor intensidad hasta los noventa, cuando el mercado único y el euro integraron los mercados de capitales\textcolor{orange}{\textsuperscript{[Probable]}}. En la quinta ola, las empresas compradas obtuvieron en el anuncio una rentabilidad anormal del $9{,}13\%$ y los compradores, del $0{,}53\%$, y la hostilidad de la oferta aumentaba la ganancia de la comprada y reducía la del comprador (MARTYNOVA y RENNEBOOG, 2011).

En España, la concentración bancaria de los años ochenta y noventa y las privatizaciones dieron paso, en los dos mil, a la expansión internacional de las grandes empresas, que compraron en el extranjero hasta la crisis, como muestra el panel (c) de la figura. Tras ella, la reestructuración de las cajas de ahorros redujo las $45$ que existían a finales de 2009 a $11$ entidades o grupos a finales de 2014, mediante fusiones e integraciones, en buena parte apoyadas por el Fondo de Reestructuración Ordenada Bancaria (FROB), creado en junio de 2009; a finales de 2014, las cajas habían reducido un $36{,}7\%$ sus oficinas y un $33{,}4\%$ su plantilla, la cara más visible del ajuste de capacidad que buscaban las integraciones (CECA, 2015). Desde 2014, las compras de empresas españolas por extranjeros crecen con fuerza.

\subsubsection{¿Concentran las olas la economía?}
Que las olas reúnan cada vez más actividad en menos empresas no se sigue de los datos. GOLBE y WHITE (1988) mostraron que la parte del valor añadido de la industria estadounidense que generaban las $200$ mayores empresas pasó del $30\%$ en 1947 al $41\%$ en 1963, sobre todo por el crecimiento interno de las grandes empresas de los sectores en expansión, y apenas cambió después, hasta el $44\%$ de 1982, pese a las olas de los sesenta y los ochenta. En el conjunto del sector privado, la parte del empleo de las $100$ mayores empresas bajó del $18{,}8\%$ en 1970 al $15{,}1\%$ en 1984.

A muy largo plazo, sin embargo, la concentración agregada ha crecido sin interrupción: la parte de los activos de las sociedades estadounidenses que corresponde al $1\%$ mayor pasó de alrededor del $70\%$ a principios de los años treinta al $97\%$ en 2018, y la del $0{,}1\%$ mayor, del $47\%$ al $88\%$, primero en la industria y la minería y, desde los setenta, en los servicios y el comercio. Los autores no pueden atribuirlo a una causa concreta y señalan fuerzas persistentes más que episodios como las olas de fusiones (KWON, MA y ZIMMERMANN, 2024). La concentración que preocupa a la política de competencia es otra, la de cada mercado, y su relación con las fusiones es la del debate sobre los márgenes que vimos en el cuerpo ([\authorfont{Ver Tema 3.A.15}]).

\subsection{Las causas de las olas: ¿reasignación o errores de valoración?}\label{anx:ev_causas}
Que la actividad aumente con la bolsa fue, para GOLBE y WHITE (1988), un enigma. Si comprar una empresa es una alternativa a invertir en capital nuevo, debería comprarse más cuando las empresas están baratas en relación con lo que costaría reponer su capital, es decir, cuando la $q$ de TOBIN es baja: es la hipótesis de las gangas, que predice una relación negativa entre la $q$ y la actividad. Estimaron la relación con datos trimestrales y anuales de la posguerra y encontraron que la actividad depende sobre todo de la del periodo anterior, aumenta con el tamaño de la economía, disminuye con los tipos de interés reales y aumenta con la $q$, en contra de su hipótesis. Las dos explicaciones de las olas resuelven el enigma de manera distinta.

\subsubsection{Reasignación: shocks, dispersión de la q y liquidez}
La explicación neoclásica parte de GORT (1969): los shocks económicos, como un cambio técnico o una variación brusca de los precios, alteran lo que vale cada empresa y crean diferencias de valoración entre quienes podrían usar sus activos, que se resuelven comprando y vendiendo empresas. Si los shocks afectan a sectores concretos, las operaciones deben agruparse por sectores. MITCHELL y MULHERIN (1996) lo comprobaron con $51$ sectores estadounidenses entre 1982 y 1989: la actividad de cada sector se explica por los shocks que sufrió, como la desregulación, los cambios en el coste de los factores y las innovaciones financieras.

JOVANOVIC y ROUSSEAU (2002) le dieron forma con la teoría de la $q$. Si las empresas con mejores oportunidades, de $q$ alta, compran los activos de las que tienen peores, de $q$ baja, las fusiones deben aumentar cuando aumentan las diferencias de $q$ entre empresas, y la $q$ alta del comprador indica que sabe usar mejor el capital, no que esté caro. Con datos desde 1890, la dispersión de la $q$ aumentó antes de cada ola, salvo la de los sesenta, y la inversión mediante adquisiciones responde a la $q$ del comprador $2{,}6$ veces más que la inversión interna.

Falta explicar por qué las olas de los distintos sectores coinciden en el tiempo. HARFORD (2005) mostró que un shock solo desencadena una ola si hay liquidez suficiente para financiar la reasignación: la abundancia de crédito, que sube también la bolsa, sincroniza las olas sectoriales, y las variables de valoración apenas añaden capacidad explicativa a un modelo de shocks y liquidez. La sexta ola encaja en esa explicación, porque la impulsó sobre todo la liquidez (ALEXANDRIDIS, MAVROVITIS y TRAVLOS, 2012).

\subsubsection{Errores de valoración: el modelo de SHLEIFER y VISHNY}
La explicación alternativa supone que el mercado se equivoca y que los directivos lo saben. SHLEIFER y VISHNY (2003) la formalizaron con un modelo sencillo, en el que las fusiones no crean ningún valor real y aun así se producen en olas.

Hay dos empresas, el comprador A, con un capital $K_A$ que el mercado valora a $Q_A$ por unidad, y la empresa B, con un capital $K_B$ valorado a $Q_B<Q_A$. A largo plazo, todo el capital vale lo mismo, $q$ por unidad, esté o no unido: no hay sinergia real. A corto plazo, el mercado valora la empresa unida a $sQ_A+(1-s)Q_B$ por unidad de capital, donde $s$ es la sinergia percibida, la historia que el mercado se cuenta sobre la operación, y A paga por cada unidad de capital de B un precio $P=pQ_A+(1-p)Q_B$, donde $p$ mide la fuerza negociadora de B.

Si A paga en efectivo, lo que gana a largo plazo es lo que paga de menos por un capital que acabará valiendo $q$, $K_B\left(q-P\right)$: solo le conviene comprar empresas infravaloradas. Si paga con acciones, los accionistas de B reciben una parte de la empresa unida, y lo que ganan a largo plazo los accionistas de A es lo que pierden los de B (proposición 3 del modelo de SHLEIFER y VISHNY, 2003, con pago en acciones):
$$\Delta V_A=K_B\,q\left(1-r\right),\qquad r=\frac{P}{sQ_A+(1-s)Q_B}$$
donde $r$ es el precio pagado en relación con el valor que el mercado da a la empresa unida. Si el precio es inferior a ese valor, $p<s$, los accionistas de A acaban dueños de una parte mayor del capital a cambio de unas acciones sobrevaloradas, y como a largo plazo solo importa el capital, ganan.

El modelo predice todo lo que vimos: compras con acciones por empresas sobrevaloradas de otras menos sobrevaloradas; compras en efectivo de empresas infravaloradas; olas cuando las cotizaciones están altas; y rentabilidades negativas de los compradores a largo plazo que, sin embargo, son menos negativas de lo que habrían sido sin la compra, porque convertir unas acciones caras en capital real amortigua su caída. Son adquisiciones defensivas, y la diversificación hacia sectores menos valorados es una de ellas. Los directivos de B aceptan porque tienen un horizonte corto, porque quieren vender o porque se les compensa, y RHODES-KROPF y VISWANATHAN (2004) añadieron que incluso unos directivos racionales pueden aceptar acciones sobrevaloradas, porque en las fases de euforia los errores al valorar la sinergia están correlacionados con el error de valoración del mercado.

La evidencia apoya el modelo en lo que predice sobre quién compra a quién y cómo se paga. RHODES-KROPF, ROBINSON y VISWANATHAN (2005) descompusieron el cociente entre el valor de mercado y el valor contable de cada empresa en un error de valoración propio, otro común a su sector y un valor a largo plazo. Los compradores tienen errores de valoración propios mayores que las empresas que compran, y la sobrevaloración del sector se asocia a más operaciones y a más pagos con acciones; pero, en contra de la intuición, son las empresas con menor valor a largo plazo en relación con su valor contable las que compran a las que lo tienen mayor.

\subsubsection{¿Qué explicación tiene más apoyo?}
Las dos explicaciones predicen más operaciones cuando la bolsa sube, y por eso resuelven el enigma de GOLBE y WHITE (1988): la $q$ alta es la señal de quién debe comprar, en la primera, y de quién puede pagar con acciones caras, en la segunda. La evidencia sugiere que se complementan. Los shocks y la liquidez explican qué sectores viven una ola y cuándo (HARFORD, 2005); los errores de valoración explican quién compra a quién y con qué paga.

En lo que coinciden todas las revisiones es en el final de las olas. Todas siguieron a shocks técnicos o sectoriales, en un entorno económico y político favorable, con expansión del crédito y bolsas al alza, y las operaciones del final de cada ola obedecen más a decisiones poco racionales y a menudo interesadas de los directivos (MARTYNOVA y RENNEBOOG, 2008). Las adquisiciones anunciadas con valoraciones altas obtienen mejores rentabilidades en el anuncio, pero peores rentabilidades anormales y resultados de explotación a largo plazo que las anunciadas con valoraciones bajas, lo que sus autores atribuyen a que los directivos se imitan unos a otros (BOUWMAN, FULLER y NAIN, 2009). Y las grandes pérdidas de los compradores entre 1998 y 2001 se concentraron precisamente en unas pocas operaciones de empresas con valoraciones muy altas (MOELLER, SCHLINGEMANN y STULZ, 2005).

\subsection{Los indicadores del ciclo: ¿en qué punto de la ola estamos?}\label{anx:ev_ciclo}
Los ciclos de fusiones suelen coincidir con mercados de capitales al alza y con la recuperación de la actividad económica, y los analistas los siguen con indicadores de valoración, de crédito y de la propia actividad. CRETIN, DIEUDONNÉ y BOUACHA (2015) propusieron cuatro para Estados Unidos. El primero compara el valor de las operaciones anunciadas en los doce meses anteriores con la capitalización bursátil total: se sitúa entre el $2$ y el $4\%$ en los mínimos del ciclo y entre el $9$ y el $12\%$ en los máximos. El segundo es la encuesta de la Reserva Federal a los responsables de crédito de los bancos, que mide cuántos endurecen las condiciones de los préstamos a las empresas: un nivel alto refleja una financiación difícil, y uno bajo, una financiación fácil. El tercero mide las mejoras de la oferta que espera el mercado: cuando la cotización de la empresa objetivo supera el precio ofrecido, los inversores esperan que el comprador lo mejore o que aparezca otro, y sus máximos y mínimos coinciden con los del ciclo. El cuarto es la tasa de fracaso, el número de operaciones canceladas en relación con las completadas en los doce meses anteriores, que puede deberse a la oposición de los accionistas de la empresa objetivo o del comprador, a la de las autoridades de competencia, a problemas de financiación o al deterioro de los mercados: su media ronda el $7\%$, y los periodos de menor actividad coinciden con tasas de fracaso elevadas, que reflejan una mayor aversión al riesgo.

Los dos primeros pueden reconstruirse con datos públicos, aunque solo para las fusiones transfronterizas, las únicas de las que hay una serie pública y comparable.
