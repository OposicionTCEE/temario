    % --- Imagen adaptativa ---
    % \imagenfit[alto]{fichero}{título}{fuente}: la imagen queda fija entre los párrafos donde se escribe (no flota), con su
    % título en el índice de figuras y la fuente debajo. Sin [alto]: las figuras PDF (TikZ) salen al tamaño con que se
    % dibujaron y solo se reducen si no caben en el ancho o en el alto de la página; las PNG/JPG ocupan el ancho del texto
    % hasta \imgaltoras del alto de la página. Con [alto] (fracción del alto de la página; [1] = página entera) se fija el
    % tamaño de esa imagen. Si no cabe en lo que queda de página, pasa entera a la siguiente.
    % \imagenfithorizontal{fichero}{título}{fuente}: la imagen en una página girada (apaisada), en ese mismo punto del texto.
    % Reglas: main/GUIA_TEMAS.md, §4 (Figuras).
    \newdimen\imgcap
    \newdimen\imgh
    \newdimen\imgw
    \newdimen\imgneed
    \newsavebox\imgcaja
    \newcommand{\imgaltoras}{0.30}% alto por defecto de las PNG/JPG (fracción del alto de la página)
    \makeatletter
    \newcommand{\imagenfit}[4][]{%
      \addvspace{10pt}%
      \par\begingroup
        \imgcap=3.5\baselineskip% alto aproximado de título + fuente
        \imgw=\linewidth
        \filename@parse{#2}\def\img@pdf{pdf}%
        \if\relax\detokenize{#1}\relax
          \ifx\filename@ext\img@pdf
            \sbox\imgcaja{\includegraphics{\detokenize{#2}}}% tamaño natural
            \ifdim\wd\imgcaja<\imgw \imgw=\wd\imgcaja \fi
            \imgh=\ht\imgcaja
          \else
            \imgh=\imgaltoras\textheight
          \fi
        \else
          \imgh=#1\textheight
        \fi
        \dimen@=\textheight \advance\dimen@ by -\imgcap
        \ifdim\imgh>\dimen@ \imgh=\dimen@ \fi
        \sbox\imgcaja{\includegraphics[width=\imgw,height=\imgh,keepaspectratio]{\detokenize{#2}}}%
        \imgneed=\ht\imgcaja \advance\imgneed by \imgcap
        \Needspace{\imgneed}%
        \noindent\begin{minipage}{\textwidth}
          \centering
          \captionof{figure}{\textemdash\ #3}\par\vspace{0.3em}% título arriba
          \usebox\imgcaja\par
          \vspace{0.3em}\small\textit{Fuente: }#4% fuente abajo
        \end{minipage}%
      \endgroup
      \par\addvspace{10pt}\RequestRestore
    }
    \newcommand{\imagenfithorizontal}[3]{%
      \begin{landscape}
        \centering
        \captionof{figure}{\textemdash\ #2}\par\vspace{0.3em}%
        \includegraphics[width=\linewidth,height=\dimexpr\textheight-4\baselineskip\relax,keepaspectratio]{\detokenize{#1}}\par
        \vspace{0.3em}{\small\textit{Fuente: }#3}%
      \end{landscape}%
    }
    \makeatother

    % \anexo{Título}: un anexo equivale a un \section (su título sale del mismo tamaño). Dentro se usan \subsection,
    % \subsubsection y \paragraph, nunca \section: se numeran I.1, I.1.1, I.1.1.1 y las páginas del anexo, I-1, I-2…
    % En el índice de contenido inicial solo figura el nombre del anexo. Sus apartados forman el índice del anexo, que sale
    % solo si el anexo los tiene, justo después de la introducción del anexo (antes del primer \subsection).
    % \anexo*{Título}: el mismo anexo, sin ese índice. Reglas: main/GUIA_TEMAS.md, §6.

    \newcommand{\anx@empezar}[2]{%
      \clearpage
      \refstepcounter{anexo}%
      \global\EnAnexotrue
      \setcounter{page}{1}\gdef\thepage{\theanexo-\arabic{page}}%
      \setcounter{section}{\numexpr\value{anexo}-1\relax}\stepcounter{section}% el anexo hace de \section
      \gdef\thesection{\Roman{section}}\gdef\theHsection{anexo.\Roman{section}}%
      \phantomsection
      \protected@xdef\leq@current@section{Anexo \theanexo.- \unexpanded{#2}}\gdef\leq@current@subsection{}% índice de ecuaciones
      \addtocontents{stc}{\protect\contentsline{section}{Anexo \theanexo.- #2}{}{\@currentHref}}% índice inicial: solo el nombre, sin página
      \addcontentsline{toc}{section}{Anexo \theanexo.- #2}% marcadores del PDF
      {\noindent\normalfont\Large\bfseries Anexo \theanexo.- #2\par}\nobreak\vspace{1.5ex \@plus .2ex}%
      #1\@ifundefined{anx@lista@\theanexo}{\global\anx@pendientefalse}{}%
    }
    % Índice del anexo: cada apartado escribe una línea en el fichero .atc (anexo, nivel, texto, página, enlace); al
    % empezar el documento se leen las de la compilación anterior y se guardan por anexo (dos pasadas, como cualquier índice).
    % En ese índice los apartados suben un nivel, porque el anexo hace de \section.
    \def\anx@entrada#1#2#3#4#5{\@ifundefined{anx@lista@#1}{\global\@namedef{anx@lista@#1}{}}{}%
      \expandafter\g@addto@macro\csname anx@lista@#1\endcsname{\contentsline{#2}{#3}{#4}{#5}}}
    \tcbuselibrary{breakable}
    \def\anx@indice{\ifanx@pendiente\global\anx@pendientefalse

            before upper={\parindent=0pt\parskip=2pt}]
          \small
          % números más anchos que en el índice inicial (I.1, I.1.1, I.1.1.1; páginas I-1)
          \def\@pnumwidth{2.6em}\def\@tocrmarg{3.4em}%
          \def\l@section##1##2{\addpenalty\@secpenalty\addvspace{1.0em \@plus\p@}\setlength\@tempdima{2.6em}%
            \begingroup\parindent\z@\rightskip\@pnumwidth\parfillskip-\@pnumwidth\leavevmode\bfseries
            \advance\leftskip\@tempdima\hskip-\leftskip##1\nobreak\hfil\nobreak\hb@xt@\@pnumwidth{\hss##2\kern-\p@\kern\p@}\par\endgroup}%

          \csname anx@lista@\theanexo\endcsname

      \fi}
    \def\anx@escribir#1#2#3{\ifEnAnexo

    % margen superior, así que no cambia las páginas del tema. Se enciende con \pagenumbering{arabic} y se apaga con

    \cs_new_protected:Npn \cab_guardar:n #1
      {
        \tl_gset:Nn \g_cab_titulo_tl {#1}
        \regex_replace_once:nnN { \A \s* Tema \s+ \S+ \s* \-\-? \s* } { } \g_cab_titulo_tl % sin «Tema X.Y.N --»
        \regex_replace_once:nnN { \.? \s* \c{\\} (\[ [^\]]* \])? \s* (\c{large} \s*)? } { .\  } \g_cab_titulo_tl % título. subtítulo
        \regex_replace_once:nnN { [\.\s]* \Z } { } \g_cab_titulo_tl % sin subtítulo: nada detrás
      }

\section{El Fondo Monetario Internacional: Perspectiva Institucional}
\puntosclave{
    Los recursos del FMI ascienden a 1 billón de DEG y provienen de los balances de los bancos centrales y de las reservas de divisas de los países miembros. 
    
    Las cuotas del FMI son asimilables al capital de la institución y determinan el poder de voto de cada país. Estados Unidos es el mayor accionista y cuenta con poder de veto para las decisiones esenciales sobre el gobierno de la institución.
    
    El personal del FMI (\textit{staff}) es quien negocia y acuerda los programas de asistenciafinanciera con los países miembros y la supervisión de las políticas. Los programas acordados con los países luego tienen que ser aprobados por el Directorio Ejecutivo. 
    
    El Directorio Ejecutivo del FMI reside en Washington D.C. (EEUU) y está compuesto por representantes de los países miembros (\textit{sillas}). Es donde se aprueban los programas de asistencia financiera.
}

El FMI es una institución internacional de cooperación monetaria y financiera. Su origen se puede datar en la firma del Convenio Constitutivo del Fondo, el 22 de julio 1944, que entraría en vigor el día 27 de diciembre de 1945 (BOE n.305, de 22 de diciembre de 1978), en la Conferencia de Bretton Woods, por parte de veintinueve países que representaban el $80\%$ de las cuotas aceptadas.\footnote{
Se puede consultar aquí: \href{https://www.imf.org/external/spanish/pubs/ft/aa/aas.pdf}{Convenio Constitutivo (español)}.

El Convenio Constitutivo ha sido enmendado varias veces entrando en vigor éstas en 1969, 1978, 1992 (BOE n.253, de 22 de octubre de 1998, pp. 34926-34928, y BOE n.9, de 22 de enero de 1999, p. 3074).
} 

Poco después, el día 8 de marzo de 1946, se celebró la sesión inaugural del Consejo de Gobernadores del Fondo en Savannah (Georgia), y el Fondo comenzó sus operaciones el día 1 de marzo de 1947. En septiembre de 1947, el Acuerdo mediante el cual pasó a ser Organismo Especializado de las Naciones Unidas fue aprobado por el Consejo de Gobernadores, y entró en vigor el 15 de noviembre de 1947 mediante su aprobación por la AGNU. Tiene su sede principal en Washington D.C y cuanta con pequeñas oficinas en París, Bruselas, Nueva York y Ginebra, así como con representantes residentes en países con programas importantes o de larga duración.\footnote{%
    El Convenio Constitutivo establece que la sede del FMI residirá en el país miembro con la mayor cuota. La sede en Washington D.C. está en la calle 19 esquina con Pennsylvania Avenue, a escasa distancia del Tesoro de los Estados Unidos y de la Casa Blanca.
} 

Para llevar a cabo sus operaciones cuenta con $3.100$ empleados (2025) de más de $162$ países distintos. Éstas operaciones son financiadas mediante un régimen dual: cuotas y préstamos. Como no se financia en los mercados financieros internacionales, sus recursos provienen de los Balances de los Bancos Centrales (vía cuotas) y de la emisión de Derechos Especiales de Giro como activo reserva (vía préstamos), de forma bilateral o multilateral. Además, se encuentra entre los acreedores preferentes en el orden de prelación de créditos, aunque de manera no institucionalizada (por no haber un régimen ejecutorio común, es evidente). De hecho, esta es una condición adquirida por costumbre y no prevista. 

Actualmente, cuenta con $191$ miembros. Es decir, prácticamente todas las naciones soberanas que son miembros de las Naciones Unidas son también miembros del FMI. ¿Cuáles no? Tengamos en cuenta que Naciones Unidas tiene 193 miembros de pleno derecho por tanto, parecería razonable suponer que los faltantes son Cuba y Corea del Norte. Sin embargo la realidad es más compleja. Por un lado, tanto Mónaco como Liechenstein son mimebros de pleno derecho de las Naciones Unidas pero no lo son del FMI. Por otro lado, tanto Kosovo como Taiwán son miembros del FMI sin ser miembros de las Naciones Unidas. Por tanto, debemos entender algo elemental: cualquier entidad territorial puede ser miembro del Fondo Monetario Internacional mientras los miembros lo acepten, no es necesario ser una entidad territorial soberana a los efectos del DIP. España se adhirió al Fondo en 1958.\footnote{
Las razones a las que se alude son diversas:
\begin{la}
    \item \textbf{Exigencias para la membresía}. En primer lugar, la imposibilidad de desembolsar las cuotas requeridas en divisa fuerte, entendiendo por divisas fuertes aquellas cuyo poder adquisitivo es relativamente estable (por baja inflación en términos relativos) y que son ampliamente aceptadas en pagos internacionales. Por ejemplo, el dólar, la libra esterlina o el yen, y ahora el Euro. El FMI, sin embargo, utiliza el término \textit{divisas de libre uso} para referirse a estas divisas fuertes, tomando como criterios objetivos: (i) que las divisas sean ampliamente utilizadas en el comercio y finanzas internacionales; y, (ii) que existan mercados profundos para la divisa. Una divisa puede ser fuerte y considerada de libre uso aunque no sea libremente convertible -es decir, aunque mantenga determinados controles de capitales-, pero normalmente las monedas fuertes y ampliamente negociadas suelen ser fácilmente convertibles, y estar sometidas a relativamente pocos controles de capitales;
    \item \textbf{Motivaciones políticas}. También se argumenta que España, y otros países, optaba por la exclusión voluntaria por razones ideológicas (eran países que en la época contaban con modelos autárquicos nacionalistas). Sin embargo, CAVALIERI muestra cómo el gobierno de España intentó sondear discretamente las posibilidades de entrada en las Organizaciones Internacionales al término de la II Guerra Mundial, pero se enfrentó al rechazo de los Estados Unidos e Inglaterra. Por otra parte, YARELA y MUNS explican que cuando se produjeron las iniciativas de apertura a España por parte de Estados Unidos a principios de los 50, se produjo un acercamiento muy gradual. MUNS explica que España intentó antes entrar en la OECE, pero ésta se mostró menos dispuesta que el FMI. De hecho, el jefe de misión para España del FMI, GABRIEL FERRAS, intercedió de forma determinante para facilitar la entrada de España en la OECE.
\end{la}
}

\subsection{Estructura Organizativa}
El FMI es la institución financiera multilateral con más recursos financieros disponibles. Es también la organización que orienta los principales debates de política económica internacional en sus Asambleas bianuales de ministros de economía y finanzas y en las reuniones del G20. Por estas razones, comprender quién gobierna o influye en la institución, y cómo toma sus decisiones, es particularmente importante.\footnote{%
    En las reuniones de Bretton Woods en 1944. las principales características del FMI venían negociadas de antemano entre Estados Unidos y el Reino Unido, y el principal contenido de las discusiones versó sobre las cuotas relativas de cada país en el Fondo, es decir, del reparto de influencia y poder. Las discusiones sobre el reparto de cuotas entre los miembros han sido una tónica desde entonces hasta hoy, y se encuentran entre las más sensibles a nivel político.
} 

El FMI se suele dividir entre:
\begin{la}
    \item \textbf{Órganos políticos}. En la cúspide de la estructura funcional encontramos los órganos de deicisón políticos. En estos, los países mimebros y sus respectivos intereses toman una espeial relevancia. Estan formados, esencialmente, por designaciones que siguen criterios político-técnicos en interés de sus representados. \paragraph{Junta de Gobernadores}
El Board of Governors está compuesta por los 190 ministros o gobernadores de los bancos centrales (1 por país, que elige si lo representa el ministro o el gobernador del Banco Central). Los "Gobernadores ante el FMI" de los países miembros son los Ministros de Economía o Hacienda (Finance Ministers) o los Gobernadores de los Bancos Centrales.\footnote{%
    Cada país decide quién es su Gobernador/a ante el Fondo, como máximo representante formal en la institución. En España en 2023 es su Vicepresidenta y Ministra de Asuntos Económicos y Transformación Digital, pero otros países como Alemania designan al Gobernador de su Banco Central.
} Teóricamente, los 190 Gobernadores ante el FMI, junto a otros 1 90 Gobernadores alternos, se reúnen una vez al afio en la Asamblea Anual del FMI, pero en la práctica esta reunión es puramente formal, con discursos de la Directora Gerente y algún otro m iembro selecto, sin discusión y sin asistencia dela gran mayoría de los ministros o gobernadores -puesto que organizar un debate entre 380dignatarios de alto nivel sería impracticable-. Los Gobernadores en l a práctica efectúan votaciones a distancia sin discusión. Las funciones principales de la Junta de Gobernadores son el refrendo formal de las decisiones aprobadas en su Directorio Ejecutivo, cuando así lo exige el Convenio Constitutivo. Suele seren materias que afecten a la estructura de la organización, tanto organizativa como financiera.Por ejemplo, las decisiones sobre aumentos de cuotas y asignaciones de Derechos Especiales de Giro, cambios importantes en la estructura organizativa, la remuneración del Directorio, o la admisión de nuevos miembros.

La mayor parte de las decisiones estratégicas que se someten a votación se toman por mayoría reforzada del 85\% de los votos, lo que confiere poder de veto a los Estados Unidos en todos los temas determinantes de la organización. Algunas decisiones de especial calado, como la modificación del Convenio Constitutivo, que es un Tratado Internacional, requieren la ratificación de los parlamentos nacionales. Este proceso puede dilatar la aplicación de algunas decisiones durante más de un año.

\paragraph{Directorio Ejecutivo}
El Directorio Ejecutivo (la Junta) es responsable de la gestión cotidiana de las actividades del FMI. Está compuesto por 25 directores, que son elegidos por los países miembros o por grupos de países, y por el Director Gerente, que actúa como presidente del órgano. La Junta suele reunirse varias veces por semana. Desarrolla su trabajo principalmente sobre la base de documentos preparados por la dirección y el personal técnico del FMI.

El International Monetary Fund cuenta así con una estructura en la que la Junta de Gobernadores define las grandes orientaciones institucionales, mientras que el Directorio Ejecutivo supervisa la administración y las decisiones operativas diarias del organismo.


El Executive Board lo componen 24 representantes de los países miembros, residentes en Washington.\footnote{%
    Existen otras organizaciones internacionales cuyo Directorio es no residente. Suele ser en casos en los que se otorga gran autonomía a la gerencia.
} 

Hay elecciones cada dos años para que los 190 países, agrupados en \textit{sillas} (constituencies), designan a sus representantes. 

Los Estados Unidos, Japón, China, Alemania, Reino Unido, Francia y Arabia Saudí, como representantes de los principales accionistas de la institución, tienen silla propia que no necesitan compartir con otros países. Otros países como Italia o Brasil cuentan con otros países en sus sillas pero siempre ostentan la Dirección Ejecutiva. Finalmente, países como España y México, cuentan con acuerdos de rotación sobre su silla. 

El Directorio Ejecutivo decide semanalmente el día a día de la institución. Por ejemplo, aprobación sobre programas financieros.\footnote{%
    Una anécdota que reseña MUNS es que ULLASTRES, ministro de Comercio de España, presentó el programa de estabilización de 1959 ante el Directorio Ejecutivo del FMI.
} 

Las decisiones se toman por mayoría simple de votos, pero en la práctica se busca alcanzar un amplio consenso. Algunas materias estratégicas se elevan a la Junta de Gobernadores, que suelen requerir mayorías reforzadas.


\paragraph{Comité Monetario y Financiero Internacional (CMFI)}
El CMFI (Comité Monetario y Financiero Internacional) asesora e informa a la Junta de Gobernadores del FMI sobre la supervisión y gestión del sistema monetario y financiero internacional, incluidas las respuestas ante acontecimientos que puedan alterar dicho sistema. También examina propuestas del Directorio Ejecutivo para modificar los Artículos del Convenio Constitutivo y asesora sobre cualquier otra cuestión que le sea remitida por la Junta de Gobernadores. Aunque el CMFI no posee poderes formales de decisión, en la práctica se ha convertido en un instrumento clave para proporcionar orientación estratégica al trabajo y a las políticas del Fondo.

El CMFI suele reunirse dos veces al año, durante las Reuniones Anuales y de Primavera del Banco Mundial y el FMI. Para cada reunión, la Directora Gerente prepara un proyecto de agenda que es debatido por el Directorio Ejecutivo, aprobado por el presidente del CMFI y adoptado formalmente por el Comité durante la reunión. Al finalizar las reuniones, el Comité emite una declaración que resume sus posiciones. Estas declaraciones proporcionan orientación para el programa de trabajo del FMI durante el semestre siguiente, hasta las próximas Reuniones de Primavera o Anuales.

El tamaño y la composición del CMFI reflejan los del Directorio Ejecutivo. El Comité cuenta con 25 miembros, que son gobernadores de bancos centrales, ministros u otros cargos de rango equivalente, normalmente seleccionados entre los gobernadores de los 191 países miembros del FMI, siendo Liechtenstein el miembro más reciente tras incorporarse al FMI en octubre de 2024. Cada país miembro y cada grupo de países miembros que elige a un director ejecutivo designa a un miembro del CMFI.

Actualmente, el grupo está presidido por Mohammed Aljadaan, ministro de Finanzas de Saudi Arabia. Fue seleccionado para dirigir el Comité por un período de tres años a partir del 4 de enero de 2024.

El CMFI opera por consenso, incluida la elección de su presidente. Aunque no existen reglas formales sobre límites de mandato, desde 2007 los presidentes del CMFI son nombrados por períodos de tres años. Varias instituciones internacionales, entre ellas el World Bank Group, participan como observadores en las reuniones del CMFI.


El CMFI es una versión ejecutiva de la Junta de Gobernadores al estar compuesta por Ministros y Gobernadores de Bancos Centrales, pero con una composición similar a la del Directorio de 24 sillas. Normalmente, los países miembros establecen reglas de rotación conjunta entre el Directorio y el CMFI. En el CMFI es donde se adoptan las principales líneas estratégicas de trabajo de la institución, plasmadas en sus Comunicados semestrales, que se negocian por los miembros en las Asambleas de Primavera (abril) y Anuales, de Otoño (octubre). El FMI participa también formalmente en el Comité de Desarrollo junto con el Banco Mundial, pero se suele considerar un foro independiente del FMI. 

Comentario : El CMFI es un foro importante para el diálogo entre ministros de los principales países para discutir los principales problemas económicos internacionales. No obstante, se ha visto desplazado en esta función en gran medida por el G7 y desde 2009 por el G20, y en el plano puramente monetario por el Banco Internacional de Pagos (BIS). La principal razón para esta pérdida de peso fue, en los 80 y 90, la combatividad del bloque de los países en desarrollo, que derivaba en peticiones de trato diferenciado, con el consiguiente desplazamiento de las discusiones clave al G5 y al G7. Con un G20 menos dividido en bloques Norte-Sur (menores peticiones de trato diferenciado que en el CMFI) y con menores inercias en su composición 10 , éste mantiene todavía la condición de 'principal foro de cooperación económica' porque cuenta con la presencia prominente de China.

Las variables PIB, Apertura, Variabilidad y Reservas se miden en porcentajes del total de los miembros del FMI. La principal variable de la fórmula es el PIB, que es la variable más utilizada habitualmente para medir el peso de un país en la economía mundial, y se mide como una mezcla (blend) de: 

PIB Blend = 60\%. PIB en dólares medido a tipos de cambio de mercado + 40\% PIB a Paridad de Poder Adquisitivo. 

El PIB en PPA beneficia más a los países en desarrollo y se introdujo como una concesión a los países que tenían fuertes perspectivas de crecimiento, como China y la India. La Apertura de un país es la suma de sus ingresos y pagos por cuenta corriente en los últimos 5 años. Esta variable tiene gran tradición en el FMI puesto que la institución originalmente financiaba situaciones de desequilibrio comercial y reconoce en positivo la apertura exterior, como recogen los objetivos del Fondo. No obstante, prima mucho a los países desarrollados pequeños muy abiertos. La Variabilidad es la desviación típica de los ingresos por cuenta corriente y de los flujos de capitales en los últimos 3 años. Se introdujo para conceder más cuotas a los países que sufrían más la volatilidad en los flujos comerciales y de cuenta corriente, es decir, los potenciales prestatarios. Las reservas internacionales tienen ahora un peso pequeño, pero anteriom1ente tenían un mayor peso, porque reflejaban la capacidad de contribución mejor que el PIB (salvo para los países que emiten moneda de reserva, que pueden aportar la moneda que crean). Se at1ade también un "factor de compresión" matemático que eleva al 0,95 el resultado de la fórmula, de forma que 'comprime' las cuotas, reduciendo las de los países más grandes y aumentando las de los países pequeños. Junto a los votos básicos, son elementos que tratan de dar más peso a los países con economías más pequeñas, atenuando el principio de accionista.

Las cuotas han sido objeto de intensas negociaciones desde el nacimiento del Fondo. La fómrnla de cuotas ha sido objeto de negociación intensa, puesto que es una de las claves más importantes para el reparto de poder.\footnote{%

Pero otros países en desarrollo están adecuadamente representados, o incluso sobrerrepresentados en cuotas o en sillas (Argentina, por ejemplo, tiene más presencia en el Directorio que España, siendo una economía mucho más pequeña). Más aún, el problema de definición de país en desarrollo es cada vez mayor, al existir numerosos países que han pasado el umbral de renta alta definido por el Banco Mundial, pero desean seguir contando con el trato favorable conferido a los países en desarrollo.\footnote{%

\subsection{Estructura Financiera del FMI (I): Presentación}
Ahora que conocemos la estructura organizativa del FMI, podemos presentar brevemente su estructura financiera. 

Para ello, a continuación seguiremos la clasificación que se deriva directamente de su último \href{https://www.imf.org/external/pubs/ft/ar/2025/pdfs/imf-annual-report-2025-financial-statements.pdf}{\textit{IMF Annual Report} - Financial Statements (2025)}. Nos daremos cuenta que, por la definición de sus \textit{cuentas}, parece que estemos ante una organización con características de grupo. No obstante, el FMI no es un grupo en sentido institucional. Se trata de una Organización Internacional jurídica e institucionalmente unitaria, pero que financieramente realiza una distinción contable y operativa interna entre: (i) las cuentas administradas directamente por el Departamento General, que son el núcleo financiero del FMI y donde se registran las operaciones ordinarias financiadas con cuotas; y, (ii) las cuentas administradas por \textit{entidades administratas por el FMI}. No son organizaciones separadas, sino mecanismos jurídicos y financieros administrados por el propio FMI, algunos con personalidad contable diferenciada porque tienen recursos separados, objetivos específicos y reglas propias de financiación y riesgo. (Como veremos más adelante).\footnote{%
    Al preparar los estados financieros, el Departamento General evalúa el control que ejerce sobre otras entidades administradas por el FMI -Fondos Fiduciarios y Departamento de DEG- para determinar si deben consolidarse. Se considera que existe control cuando el Departamento General: (i) tiene poder sobre una entidad; (ii) está expuesto a rendimientos variables derivados de su participación en ella; y, (iii) tiene capacidad para influir sobre dichos rendimientos. La evaluación del control requiere un juicio significativo y la formulación de supuestos mediante una evaluación multifactorial. Una consideración clave es la estructura de gobernanza del International Monetary Fund y las responsabilidades del Directorio Ejecutivo respecto de las distintas entidades. Al dirigir las actividades de una entidad, el Directorio Ejecutivo debe ejercer su autoridad de manera independiente respecto de su influencia sobre otras entidades y proteger los intereses de las partes interesadas de dichas entidades. Además, la evaluación debe tener en cuenta las limitaciones y salvaguardias procedimentales que regulan la gestión de recursos, normalmente definidas en los documentos constitutivos de cada entidad. Otro factor importante es la intención de los países miembros, reflejada en decisiones de política y marcos acordados caso por caso. Los principales accionistas del FMI (y principales contribuyentes en divisas) no han querido otorgar un papel significativo al DEG precisamente por su naturaleza incondicional. Han preferido que el uso de los recursos del FMI sea condicional. No obstante, con la voluminosa asignación de 202 1 los miembros del FMI han querido potenciar el uso del DEG. El PRGT se nutre de préstamos voluntarios de los bancos centrales y de aportaciones fiscales de los países miembros más desarrollados, para financiar el tipo de interés concesional, más bajo que el de mercado.

\subsubsection{Cuentas: Dependiente del Departamento General}
El FMI lleva a cabo sus operaciones y transacciones a través del Departamento General (Gerencia). Por tanto, los estados financieros del Departamento General comprenden la posición financiera y las actividades de la Cuenta de Recursos Generales (GRA), la Cuenta de Actividades de Inversión (AI) y la Cuenta Especial de Desembolsos (SDA).

Cada cuenta mantiene sus propios registros financieros y desempeña funciones específicas dentro de la arquitectura financiera general del FMI. 

\paragraph{Cuenta de Recursos Generales (GRA)}
La \textbf{mayor parte} de las finanzas del Fondo Monetario Internacional se produce a través de su \textbf{Cuenta de Recursos Generales}, a través de la cual el Fondo Monetario puede prestar a \textbf{todos sus miembros}. 

Los activos y pasivos de la GRA reflejan el pago de las cuotas de suscripción de los miembros, el uso y reembolso del crédito de la CRA, los préstamos y reembolsos a prestamistas, el cobro de cargos a los prestatarios (países deudores), el pago de remuneraciones sobre las posiciones acreedoras de los miembros y los intereses pagados a los prestamistas. En la terminologla técnica del Fondo, cuando efectúa un programa con un país, en esta cuenta el FMI efectúa compras de las monedas de sus miembros -a cambio de divisas fuertes-, y recompras -cuando los países deudores devuelven esas divisas en el momento en que se producen los vencimientos de los programas-. 

\paragraph{Cuenta de Actividades de Inversión (IA)}
Paralelamente, la Cuenta de Inversión se mantiene con el fin de destinar desde la Cuenta de Recursos Generales (GRA), invertirlos y ampliar la base de ingresos del FMI.\footnote{%
    Las Reglas y Reglamentos de la IA adoptados por el Directorio Ejecutivo proporcionan el marco para implementar la ampliación de las facultades de inversión autorizadas por la Quinta Enmienda de los Artículos del Convenio Constitutivo del FMI, que entró en vigor en febrero de 2011.
} La IA comprende dos subcuentas: 
\begin{la}
    \item \textbf{Subcuenta de Renta Fija} (\textit{Fixed-Income Subaccount}). La Subcuenta de Renta Fija mantiene recursos transferidos desde la CRA que no están relacionados con beneficios procedentes de ventas de oro. Con el objetivo de generar ingresos al tiempo que se protege el Balance del FMI, el objetivo de inversión de esta subcuenta es obtener, a lo largo del tiempo, \textbf{rendimientos que superen la tasa de interés de los Derechos Especiales de Giro} (DEG) en \textbf{50 puntos básicos}, minimizando simultáneamente la frecuencia y magnitud de rendimientos negativos y de resultados inferiores al objetivo durante un horizonte de inversión de entre tres y cuatro años; y,
    \item \textbf{Subcuenta Patrimonial} (\textit{Endowment Subaccount}). El objetivo de inversión de la Subcuenta Patrimonial es lograr, en el largo plazo, un rendimiento real del 3\% medido en dólares estadounidenses, con el fin de contribuir a financiar los gastos administrativos del FMI, preservando al mismo tiempo el valor real de largo plazo de estos activos. El FMI acreditó, a la Subcuenta Patrimonial, 4.400 millones de DEG provenientes de beneficios obtenidos por ventas de oro durante los ejercicios financieros 2010 y 2011. 
\end{la}

\paragraph{Cuentas Especial de Desembolsos (SDA)}
La última de las cuentas administradas directamente por el Departamente General es la Cuenta Especial de Desembolsos (\textit{Special Disbursement Account}, SDA). Se trata del mecanismo utilizado para recibir los beneficios procedentes de la venta del oro que el FMI mantenía en el momento de la entrada en vigor de la II Enmienda de los Artículos del Convenio Constitutivo del FMI (abril de 1978). 

Los recursos de la SDA pueden utilizarse para diversos fines, incluidos transferencias a la Cuenta de Recursos Generales (GRA) para su uso inmediato en operaciones y transacciones, transferencias a la Cuenta de Inversión (IA), u operaciones y transacciones que no estén autorizadas por otras disposiciones de los Artículos del Convenio Constitutivo pero que sean coherentes con el mandato del FMI, especialmente para proporcionar asistencia de balanza de pagos en condiciones especiales a países miembros de bajos ingresos.

\subsubsection{Cuentas: Entidades Administradas por el FMI}
El FMI mantiene bajo su paraguas una serie de entidades administrativas independientes del Departamento General. Se trata principalmente de tres: (i) el Departamento de Derechos Especiales de Giro (SDR Department); (ii) los Fondos Fiduciarios para Préstamos Concesionales y Alivio de Deuda; y, (iii) el Fondo Fiduciario para la Resiliencia y la Sostenibilidad (Resilience and Sustainability Trust, RST) (denominados conjuntamente \textit{los fondos fiduciarios} (trusts)).

Dado que el Departamento General no tiene control sobre estas entidades, sus estados financieros se presentan por separado pero dentro del Balance Consolidado del Grupo.

\paragraph{Departamento de DEG}
Los recursos del Departamento de Derechos Especiales de Giro (DEG) se mantienen separados de los activos de todas las demás cuentas pertenecientes o administradas por el FMI. Estos recursos no pueden utilizarse para cubrir pasivos, obligaciones o pérdidas derivadas de las operaciones del Departamento General (ni viceversa), salvo que los gastos administrativos derivados de la gestión del Departamento de DEG son inicialmente pagados por el Departamento General y posteriormente reembolsados por el propio Departamento de DEG.

\paragraph{Fondos fiduciarios: Crecimiento y Reducción de la Pobreza y Resiliencia y Sostenibilidad}
Los recursos de los fondos fiduciarios y de las cuentas administradas son aportados por países miembros, otras instituciones financieras o, en algunos casos, por el propio FMI (a través de la \textbf{Cuenta Especial de Desembolsos}). Los activos y pasivos de los fondos fiduciarios y de las cuentas administradas están separados de los activos y pasivos del Departamento General. Los activos de estas entidades no pueden utilizarse para cubrir pasivos, obligaciones o pérdidas derivados de las operaciones del Departamento General. Sin embargo, el Departamento General puede ser reembolsado por los gastos incurridos en la administración de los fondos fiduciarios.

Dentro de los fondos fiduciarios bajo el paraguas del FMI, destacan esencialmente dos:
\begin{lnum}
    \item \textbf{Cuenta del Fondo para el Crecimiento y Reducción de la Pobreza} (\textit{Poverty Reduction and Growth Trust}, PRGT). Los países pobres pueden obtener financiación del FMI en términos concesionales (es decir, por debajo de los tipos de mercado) a través de esta cuenta. De este modo, la carga del repago de la deuda de los programas se aligera, y se incentiva que recurran a esta financiación. No obstante, es necesario que los países miembros donantes (normalmente, los países desarrollados) hagan periódicamente aportaciones específicas para préstamos (puesto que las cuotas y los NAB financian los préstamos de la Cuenta de Recursos Generales GRA) y de contribuciones fiscales para financiar los elementos de concesionalidad de los préstamos. La capacidad del PRGT en 2023 es cercana a los 40.000 M de DEG (53.000 M\$);
    \item \textbf{Cuenta del Fondo para la Resiliencia y la Sostenibilidad} (\textit{Resilience and Sustainability Trust}). En 2022, el FMI creó esta cuenta para suministrar financiación en términos concesionales tanto a los países pobres elegibles para el PRGT como a países de renta media vulnerables, específicamente para apoyar programas que incluyan reformas en materia de cambio climático y pandemias. La financiación bajo esta cuenta también es en ténninos concesionales y requiere aportaciones de los países miembros tanto para préstamos como aportaciones fiscales para financiar los elementos de concesionalidad de los préstamos.
\end{lnum}

\begin{specialblock}{Bonos -- ¿Podría utilizarse como instrumento de financiación? ¿Por qué?}
    El FMI, según su convenio, podría teóricamente emitir bonos para financiarse (como hacen el Banco Mundial o el MEDE), pero en la práctica se financia exclusivamente mediante las cuotas o préstamos que le conceden los bancos centrales de sus países miembros. Por ello decimos que el Fondo se nutre exclusivamente de "recursos monetarios", entendiendo como tales los provenientes de la creación monetaria de los bancos centrales. (Por contraste, el MEDE no cuenta con financiación de los bancos centrales, sino que se financia mediante emisiones de deuda avaladas por el capital aportado por los gobiernos de los países miembros). ¿Cuánto esfuerzo le cuesta a un país aportar recursos al FMI? Principalmente, depende de si su moneda es una divisa fuerte o no. Para un país como los Estados Unidos, que emite dólares, el coste es casi nulo, puesto que en la práctica la Reserva Federal crea los dólares que desembolsa al FMl cuando éste le solicita el importe de sus cuotas, por ejemplo. En cambio, para países con problemas para acumular reservas por la debilidad de su moneda, la aportación del tramo de reservas en divisas fuertes es costosa.
    
    Fue la situación de España cuando entró en el FMI en 1958: Esprula disponía de muy pocas reservas de divisas. y tenia problemas en su balanza de pagos, por la sobrevaloración del tipo de cambio y la poca credibilidad de nuestra divisa.
\end{specialblock}

Las monedas son activos financieros que consisten en monedas de los países miembros mantenidas por la Cuenta de Recursos Generales (GRA) en depositarios designados (ya sea en forma de saldos de cuenta o de pagarés no remunerados convertibles a demanda por la GRA). ¿Qué significa esta última clasificación? Desde la óptica del FMI, el crédito pendiente es un activo financiero que representa la financiación proporcionada a los países miembros bajo las distintas facilidades financieras de la Cuenta de Recursos Generales: los miembros reciben financiación en esta cuenta comprando DEGs o monedas utilizables a cambio de sus propias monedas y reembolsan el crédito recomprando sus monedas utilizando DEG o monedas utilizables.\footnote{%
    En este contexto, cabría esperar que, al igual que cualquier entidad financiera, también cabría observar minusvaloraciones por deterioros del crédito pendiente. No es así. El Departamento General no ha reconocido pérdidas por deterioro desde su creación y justifica semejante anomalía de la siguiente forma: (i) el FMI mantiene una relación única con sus miembros, todos ellos accionistas de la institución; (ii) la financiación del FMI bajo acuerdos de la GRA está vinculada a revisiones periódicas del desempeño dentro de programas de política económica que el país miembro se compromete a implementar para superar problemas de balanza de pagos, restaurar la viabilidad externa y reembolsar al FMI; (iii) el FMI emplea un amplio conjunto de medidas para mitigar el riesgo de crédito; y, (iv) el FMI también posee un estatus de acreedor preferente de facto, reconocido por la comunidad oficial y generalmente aceptado por los acreedores privados. En conjunto, dice, estos factores reducen significativamente la probabilidad de pérdidas crediticias para el Departamento General.
}

Tradicionalmente, para preservar su condición de fondo puramente monetario y con recursos disponibles para emergencias, el FMI no puede asumir pérdidas (puesto que serían los bancos centrales de los miembros los que las asumirían), pero ha efectuado condonaciones de deuda en el marco de la iniciativa de Países Pobres Altamente Endeudados (HIPC, por sus siglas en inglés), lanzada en 1999, y reforzada por la Iniciativa de Alivio de Deuda Multilateral (Multilateral Debt Relief lnitiative, MDRJ). \section{El Fondo Monetario Internacional: Visión operativa}
\puntosclave{
    La Supervisión del FMI consiste principalmente en revisiones anuales de sus políticas económicas mediante documentos llamados \textbf{consultas del Artículo IV} orientadas a prevenir crisis. 
    
    La supervisión financera, mediante Programas de Evaluación del Sector Financiero (FSAP) ha cobrado mayor protagonismo con el tiempo, mientras que la supervisión de tipos de cambio la ha perdido, pese a ser el mandato original de la institución. 
    
    La financiación del FMI a países con problemas de balanza de pagos, o crisis en un sentido más amplio, es la función que le confiere mayor importancia a la institución. 
    
    Las facilidades financieras del FMI han evolucionado con las circunstancias. Sus elementos clave son el volumen de acceso (expresado en porcentaje de la cuota de cada país en el FMI), el plazo de repago, la condicionalidad y los recargos y comisiones. 
    
    Las facilidades más utilizadas son los Acuerdos de Derecho de Giro (Stand-By Arrangements) y los Acuerdos Extendidos (Extended Fund Faci lity). El FMI cuenta con programas precautorios específicos para prevención de crisis. El más importante es la Línea de Crédito Flexible (Flexible Credit Line).
}



\subsection{Supervisión: Asesoramiento del FMI}
Una de las funciones centrales del FMI es realizar una labor de supervisión sobre el sistema monetario y financiero internacional. Esta actividad suele descomponerse en dos: (i) la \textbf{supervisión bilateral} de los miembros; y, (ii) la \textbf{supervisión multilateral} al conjunto de los miembros.\footnote{%
    La labor incinial de supervisión pretendía velar por el cumplimiento de los acuerdos del Fondo Monetario, en particular en materia cambiaria. Dada la pérdida del objeto, ha evolucionado para ser un instrumento orientado a la prevención de crisis, señalando desequilibrios y vulnerabilidades y proporcionando recomendaciones de políticas económicas al respecto. 

No obstante, cabe resaltar que esta actividad de supervisión ya se realizaba antes bajo el mandato del Articulo XIV. De hecho, la primera misión del FMI a España (1959) fue bajo el paraguas del Artículo XIV.
}



\subsubsection{Supervisión bilateral: \textit{ex} Art. IV}
La supervisión bilateral del FMI se articula a través de dos instrumentos: (i) la supervisión de la Política Macroeonómica; y, (ii) la supervisión del Sector Financiero mediante Programas de Evaluación del Sector Financiero (FSAP). 

\paragraph{Desempeño: Políticas Macroprudenciales}
La primera de ellas viene directamente consagragada en el Convenio Constitutivo, Art. IV; por ello es habitual referirse a este proceso como \textit{llamar el Artículo IV de ...}. Esta labor de supervisión consiste en el envío de misiones anuales (o cada dos años, para países pequeños) del personal del FMI para entablar un diálogo con los países miembros sobre el desempeño de su economía y las posibles recomendaciones de política económica.

Como busca identificar posibles riesgos y recomendar ajustes de política económica adecuados para sustentar el crecimiento económico y promover la estabilidad financiera, podemos identificar (informalmente) tres \textit{corpus} distintos:
\begin{la}
    \item \textbf{Políticas Económicas}. Donde se evalua la sostenibilidad de la Política Económica del país miembro bajo una interpretación amplia del Art. IV las decisiones del Directorio Ejecutivo;
    \item \textbf{Obligaciones cambiarias}. Centrando el objeto en las obligaciones cambiarias contraídas por el estado miembro, que se desprende directamente de la literalidad del Art. IV;
    \item \textbf{Obligaciones en materia de convertibilidad}. Donde se supervisa la ejecución de las obligaciones contraídas en materia de convertibilidad, suministro de información y cumplimiento de estándares (Art. VIII y XIV del CC);
    \item \textbf{Supervisión del Sector Financiero}. 
\end{la}

Las conclusiones quedan plasmadas en un informe (\textit{staff report}), se dicute en el Directorio Ejecutivo y, finalmente, se publica.\footnote{%
    Salvo que un país objete a su publicación. Por ejemplo, Brasil, hasta 2011 no autorizaba la publicación de los informes completos del personal del FMI. Sin embargo, para todos los países se publica una reseña abreviada (\textit{Public Information Notice}) del informe de las consultas, que incluye el cuadro macroeconómico y la valoración del Directorio Ejecutivo.
} Con el fin de garantizar la comparabilidad entre países, el informe contiene cuadros estandarizados. De esta forma, suele tener un gran impacto en previsiones sobre crecimiento económico y déficit público. Son ampliamente consideradas independientes y están, además, sujetas a contrastación con las respectivas nacionales. Resultan particularmente útiles cuando un país necesite un programa, ya que el trabajo previo permite atender con rapidez y eficacia las solicitudes de asistencia financiera, teniendo identificadas las principales prioridades.

Las recomendaciones formuladas suelen tenerse en cuenta por los Gobiernos. Y, el ámbito técnico, se considera que someterse a una supervisión anual genera incentivos para la recapitulación de datos, internalización de impacto y reflexión sobre la Política Económica. Además, sirven de vehículo para fortalecer y mejorar las relaciones entre el personal del FMI y las autoridades, generando el hábito de trabajar juntos. 

\paragraph{FSAPs y FSSAs: Sector Financiero}
Esta labor de supervisión era la piedra angular del sistema pero, a raíz de la crisis financiera de 2008, una de las tareas de supervisión más relevante recae sobre el Sector Financiero. Como hemos visto, esta actividad se realiza paralelamente a la supervisión anterior. Sin embargo, también de forma mucho más específica e intensiva mediante los Programas de Evaluación del Sector Financiero (\textit{Financial Sector Assessment Program}, conocidos como FSAPs por sus siglas en inglés), de los que hablaremos brevemente. 

Los programas FSAPs permiten efectuar un diagnóstico en profundidad de los problemas del sector y se ven complementados con el análisis sobre su estabilidad (\textit{Financial Sector Stability Assessment}, FSSA). En conjunto, persigue la mitigación de los riesgos sistémicos, tanto nacionales como internacionales, mediante la identificación de riesgos e implementación de mejoras. 

Si bien los objetivos nobles y loables que persiguen los FSAPs y FSSAs son indiscutibles, los métodos utilizados no están exentos de crítica. Una de sus columnas vertebradoras son los análisis de sensibilidad o \textbf{prueba de tensión} (\textit{stress test}). Estos métodos, excesivamente dependientes de la situación coyuntural, los datos y las hipótesis de partida han sido ampliamente criticados. Especial incidencia se ha hecho sobre la sensibilidad al mercado e intensidad (restricción de hipótesis), puesto que puede conducir a resultados sobredimensionados e impiden realizarlos con elevado frecuencia (por las señales que envía al mercado). 

No obstante, sus limitaciones vienen notablemente compensadas por la fácil interpretación que ofrecen. Por ejemplo, un stress test puede indicarnos elevadas necesidades de capital para un abanico amplio de esceneraios, por lo que debiera haber un esfuerzo de saneamiento por pate de las entidades y, en casos extremos, requerimientos de ayuda financiera. Los FSAPs y FSSAs no se realizan únicamente sobre intermediarios de crédito (sector bancario) si no que también se realizan diagnósticos y recomendaciones dirigidas al sector seguros/reasegurs y, en algunos casos, los mercados financieros. En definitiva, abarcan todo el conjunto de servicios financieros y su exposición/resiliencia ante escenerios adversos. 

Su éxito ha sido tal que muchos países, sobretodo aquellos en los que el sector financiero tenga especial importancia (sistémica, como España), se han comprometido a efectuar FSAPs cada cinco años. Este compromiso se adoptó tras la crisis financiera de 2008, a la vista de la importancia crítica del sector financiero para la supervisión.

\subsubsection{Multilateral: Art. IV}
Si bien esta actividad de supervisión es, hoy en día, la más relevante y conocida, el mandato legal más fuerte y claro de supervisión del FMI es en materia cambiaria (art. IV): \textbf{firme supervisión sobre los tipos de cambio y sus regímenes}, que los miembros están obligados a declarar junto con la información que solicite el Fondo para este fin. 

No obstante, como ya sabemos, desde 1978 (cuando entró en vigor el Acuerdo de Jamaica, 1976), el FMI permite a sus miembros cualquier régimen cambiario o monetario menos un tipo de cambio fijo con el oro, por lo que, en la práctica, la supervisión cambiaría se ha enfrentado a serias dificultades. 

\paragraph{Supervisión: Informes periódicos}
Para hacer frente a esta obligación de supervisión dentro del contexto actual el FMI elabora anualmente varios informes de supervisión, siendo el \href{https://www.imf.org/en/publications/sprolls/annual-report-on-exchange-arrangements-and-exchange-restrictions}{Informe Anual sobre Regímenes y Restricciones Cambiarias} el dedicado a esta labora, pues recoge y compila la información proporcionada por los miembros sobre su régimen cambiario.

Pero, como suele decirse, del dicho al hecho hay un trecho... Y, a menudo, los regímenes declarados no coinciden con los observados (\textit{de facto}) o resultan inconsistentes con las intervenciones realizadas por sus Bancos Centrales en los mercados cambiarios.\footnote{%
    El caso más emblemático de este debate fue China. El país declaró un Tipo de Cambio de flotación con bandas hasta 2015, pero en la práctica, se le acusó en reiteradas ocasinoes de intervenir para mantener un Tipo de Cambio fijo respecto al dólar hasta 2009 y permitir únicamente la apreciación gradual de su moneda hasta 2015.
} En este contexto, la supervisión del mercado cambiario se limita a la recapitulación de los compromisos decalrados y una lista de movimientos observados en los que los Bancos Centrales parecen actuar \textit{en contra de los compromisos adquiridos}, lo que permite al FMI estimar el régimen de facto de cada moneda y su correspondiente grado de cumplimiento.

Paralelamente, elabora otros informes de supervisión multilateral. Por ejemplo, los informes \href{https://www.imf.org/en/publications/weo}{\textit{Perspectivas Económicas Mundiales}} (\textit{World Economic Outlook}, WEO), de \href{https://www.imf.org/en/publications/gfsr}{Estabilidad Financiera Global} (Global Financia! Stability Report, GFSR), y el \href{https://www.imf.org/en/publications/fm}{Fiscal Monitor}. A menudo tienen un impacto mediático importante y se presentan justo antes de las Asambleas anuales para estimular y dirigir las discusiones de Política Económica.

Uno de los debates más importantes sobre la supervisión del Fondo es si la suma de las políticas dedicadas a objetivos nacionales constituye el óptimo global, o si, por el contrario, pueden existir efectos negativos indeseados derivados de las políticas domésticas. En particular, las políticas monetarias expansivas en países sistémicos pueden generar expansiones de crédito internacionales y flujos de capitales volátiles.\footnote{%
    Tradicionalmente, el Fondo había considerado que las políticas nacionales deben dirigirse solamente a objetivos nacionales.

En el sistema de Bretton Woods se reconocían abiertamente las interdependencias y la responsabilidad de la moneda central de cumplir con reglas (principalmente de cumplir con la convertibilidad en oro), pero la crítica de autores como RUEFF o TRIFFIN era que confería demasiada autonomía a la moneda central del sistema para escapar la disciplina internacional requerida, generando efectos externos sobre el resto de países (un privilegio exorbitante, en palabras de Giscard d'Estaing y De Gaulle).
} Teniendo esto en cuenta, el FMI, reconociendo que determinadas políticas tienen consecuencias más allá de sus fronteras, intenta reforzar el conocimieto técnico al respecto mediante los informes \href{https://www.imf.org/en/publications/sprolls/spillover-reports}{\textit{Spillover Reports} (última publicación en 2015)}.

\paragraph{Armonización: Estándares y supervisión estadística}
Además de esta labor de supervisión y asistencia multilateral del FMI, también cabe destacar la labor de armonización de estándares y metodología estadística. El FMI se encuentra a la cabeza de los organismos armonizadores en las siguientes materias:
\begin{lnum}
    \item \textbf{Sistema de Cuentas Nacionales: Sector exterior, mercados financieros}. El FMI publica los manuales para la elaboración de las cuentas del sector exterior (\textit{latest}: \href{https://www.imf.org/external/pubs/ft/bop/2007/bopman6.htm}{VI Manual - BPM6}), las \href{https://www.imf.org/external/pubs/ft/gfs/manual/esl/pdf/compils.pdf}{Estadísticas de Finanzas Públicas} (\textit{Government Finance Statistics Manual}) y estadísticas sobre \href{https://www.imf.org/external/pubs/ft/mfs/manual/esl/index.htm}{mercados de valores}:
    \item \textbf{Estandarización de datos}. El FMI tiene una Iniciativa de Estándares de Datos, que ofrece recomendaciones y un sello a las estadísticas nacionales que cumplan unos parámetros determinados de metodología y de publicidad (lo que se conoce como diseminación). Los niveles de exigencia son: (i) Enhanced Global Data Dissemination Standards (e-GDDS), nivel básico que se limita a recomendaciones metodológicas y publicación de los datos esenciales; (ii) Special Data Dissemination Standards (SDDS), nivel de exigencia bastante por encima del e-GDDS, pensado para países que desean tener acceso a mercados financieros internacionales; y, (iii) Special Data Dissemination Standards Plus (SDDS+), introducido en 2013 tras el examen de las carencias observadas en la crisis financiera global de 2008 y busca un nivel adicional de profundidad en los datos.
\end{lnum}

Además, el FMI supervisa el cumplimiento de estándares estadísticos tanto propios como los de las Naciones Unidas (Sistema de Cuentas Nacionales 2008, SCN-93 y SCN-68) amparándose de nuevo bajo la cobertura del art. IV, así como en los FSAPs sobre el sector financiero y en ROSCs específicos sobre estadísticas. 

Dejará entonces constancia en el Informe elaborado mediante \textbf{apéndices estadísticos} (\textit{Statistical Issues Appendixes}), en los que emite un juicio sobre la calidad de las estadísticas: (A) adecuados para la supervisión; (B) generalmente adecuados pero con limitaciones; y, (C) inadecuados para la supervisión.

En algunos casos particulannente graves de manipulación estadística detectada en la supervisión, el FMI puede hacer declaraciones públicas de censura o incluso sancionar a un país mediante la suspensión de acceso a Programas del FMI. (In extremis, cabría la expulsión por incumplimiento de las obligaciones del Convenio Constitutivo). Un caso particularmente grave fue el de Argentina en 2013. Sus estadísticas de PIS e IPC recibieron una declaración formal de censura por demostrar gravísimas carencias y de forma persistente. 

Aunque, desde nuestra perspectiva (España), el caso de manipulación más grave, por sus implicaciones y cercanía, bien puede ser el de las estadísticas fiscales de Grecia. Aunque el FMI no llegó a declarar la censura porque se adoptaron medidas drásticas: el déficit público griego inicialmente reportado en 3.7\% del PIB (febrero 2009), fue revisado al 12,5\% y posteriormente a un 15,8\% en menos de un año (finales de 2009). Tanto el FMI como EUROSTAT llevaban tiempo solicitando mejoras en los IE griegos, pero su supervisión demostró ser insuficientemente estricta. (En el Art. IV (2007) figuraba en portada: \textit{Data are adequate for surveillance, but improvements are needed. In particular, the authorities concur that national accounts revisions should be more timely, and, to booster its reputation, that the statistical service should be made independent}).\footnote{%
    El escándalo tenía causas directamente institucionales y trajo consigo graves consecuencias fiscales, generando un efecto desbordamiento a otras economías de la UE, por lo que se cerrí el Servicio Estadístico Griego (Ministerio de Finanzas) y fundó un Instituto Estadístico Independiente, ELSTAT.

El nuevo instituto pasó a ser dirigido por un funcionario del FMI del Departamento de Estadística, A GEORGIU, y sus datos pasaron a ser aceptados como fiables por las instituciones europeas y el FMI. No obstante, en un giro paradójico del destino, un fiscal asociado a los partidos opuestos al rescate mantuvo abierto un proceso judicial contra GEORGIU, acusándolo de traición por inflar los datos de déficit, que se expandió durante dos años (2013-15). Un titular de la prensa canadiense resumió bastante bien la situación: \textit{Greek statistics chief probed 'for not cooking the books'}. Los cargos contra GEORGIU fueron retirados pero el proceso judicial sometió a una dura presión política a ELSTAT.
} Otros países que han manipulado estadísticas durante la financiación de un programa, por ejemplo, han tenido que adoptar medidas restrictivas adicionales.


\subsection{Fortalecimiento de capacidades: Aistencia técnica y capacitación}
Por último, entre sus funciones principales el FMI proporciona asistencia técnica y capacitación. Es un servicio a disposición de todos los países miembros que lo solicitan y se adapta a las necesidades específicas de cada país. Principalmente, se proporciona de dos maneras: (i) mediante asistencia ténica; y/o, (ii) centros de capacitación.

En conjunto, representan, aproximadamente, una \textbf{tercera parte del gasto anual} del FMI y buscan ayudar a los países a mejorar la recaudación de impuestos y respaldar las finanzas públicas, modernizar las políticas monetaria y cambiaria, establecer ordenamientos jurídicos, reforzar el buen gobierno o, incluso, a la hora de recopilar y divulgar datos para fundamentar la toma de decisiones.

Más de la mitad de las actividades de fortalecimiento de las capacidades del FMI están financiadas por socios externos, en forma de apoyo a los centros regionales de capacitación, fondos temáticos y proyectos bilaterales. Varios países miembros son donantes importantes, como Japón o la Unión Europea, y financian la mitad del presupuesto de asistencia técnica del FMI. Estas donaciones computan como ayuda al desarrollo, puesto que los destinatarios son en su práctica totalidad países en desarrollo -los desarrollados se suelen pagar su fonnación.

\subsubsection{Ad hoc: Asistencia técnica del FMI}
El primero de los instrumentos/programas, la asistencia ténica, debe ser solicitado por el país miembro y buscar la colaboración/asistencia ténica del FMI en: (i) el diseño práctico y concreto de reformas y políticas económicas; y, (ii) en la formación de los funcionarios de los países receptores (\textit{capacity development}). En cualquier caso, será proporcionada bien por el personal técnico de la sede central del FMI, que puede reunirse con funcionarios públicos del país, ya sea en persona o virtualmente, bien por trabajares en el país en calidad de asesores residente. 

Lo más común es que este tipo de solicitudes se produzcan a raíz de un informe de consultas (ex art. IV). Por ejemplo, si una misión del FMI identifica alguna carencia, junto a la necesidad de emprender alguna reforma. En estas circunstancias, si un país desea reforzar la calidad de sus reformas o mejorar su credibilidad, podría solicitar la asistencia. Dos ejemplos concretos que podemos citar son: 
\begin{lnum}
    \item \textbf{España, 2012}. España y varias instituciones europeas recibieron asistencia en julio de 2012, para la supervisión y el diseño del proceso de recapitalización del sector financiero. El FMI contaba con importante información fruto de la supervisión del sector financiero (FSAP) presentada en abril de 2012, donde formuló recomendaciones de refuerzo del sector en profundidad. En esta importante asistencia técnica, el FMI contribuyó al diseño y estrategia de las medidas adoptadas, y elaboró informes trimestrales de seguimiento;\footnote{%
        La Asistencia Técnica suministrada por el FMI en España en 2012 y 2013 para la refoma del sector financiero tuvo un impacto muy positivo: consiguió aportar elementos importantes en la estrategia de refonna, y reforzar la credibilidad de las medidas de recapitalización del sector. Los informes de supervisión elaborados por la asistencia técnica mostraron un cumplimiento rápido y completo de las medidas recomendadas -una buena muestra de apropiación local (ownership) de las recomendaciones.
    }
    \item \textbf{Italia, 2012}. La asistencia técnica suministrada a Italia para la orientación estratégica de su reforma fiscal. 
\end{lnum}

El Fondo suele requerir un diálogo en profundidad, mediante el envio de misiones al país, para adaptar sus recomendaciones a las circunstancias concretas. Además, suele aprovechar su experiencia en la supervisión y asistencia a otros países para recomendar las mejores prácticas. La utilidad para los países suele ser muy alta, pero la asistencia no siempre está disponible por la escasez de recursos humanos y financieros. El FMI suele priorizar el suministro de asistencia técnica a los países éstratégicos y con mayores vulnerabilidades. En esos casos no cobra tarifas.

\subsubsection{Permanente: Centros de Capacitación Regional}
Por otro lado, y con el mismo objetivo, el FMI mantiene una red global cuasi permanente de centros de capacitación regional, destinada esencialmente a los funcionarios de los países del área de influencia.

\imagenfit{Fig4.png}{Red de Centros de Capacitación Regional}{\href{https://www.imf.org/-/media/files/capacity-developement/brochures/brochure-regional-capacity-development-center-september-2022.pdf}{\textit{Regional Capacity Development Centers}. IMF (2022)}}

En los países en desarrollo, donde la formación de los funcionarios suele tener importantes lagunas, estos cursos son particularmente valiosos. El FMI, en algunos casos de graves carencias institucionales en ámbitos clave de la gestión financiera pública, por ejemplo en la gestión de la deuda pública en países con falta de control de su endeudamiento, suministra tanto los sistemas informáticos y de gestión de la deuda como la formación a los funcionarios. 

La prestación de asistencia técnica mediante la formación de funcionarios es de particular valor en países que han sufrido conflictos o con graves problemas de inestabilidad institucional. En estos países, la rápida reconstrucción de unas estructuras para el buen funcionamiento de la gestión pública es determinante para evitar problemas fiscales. Y el FMI tiene una ventaja comparativa, respecto a los bancos de desarrollo y las agencias de desarrollo en sus ámbitos de especialización, al ser también la institución internacional que genera los estándares. Los programas de formación suelen ser gratuitos salvo para los países desarrollados, que a veces acerrean con el coste. 


\subsection{Financiera: Préstamos/Programas del FMI}
Ahora que conocemos las labores que abarcan el mandato de supervisión que ostenta el FMI, debemos presentar la función financiera que ostenta, que ha sido esencial desde los planes de su creación.\footnote{%
    La experiencia de los años 20's y 30's mostró que los países que no podían financiar sus déficits por cuenta corriente o las presiones sobre sus tipos de cambio recurrían a medidas dañinas para la prosperidad nacional e internacional, como las devaluaciones competitivas desordenadas, los controles de cambios, el proteccionismo comercial y los impagos de sus deudas.

EICHCNGREEN e IRWIN (\textit{The slide to protectionism in the Great Depression: Who succumbed and Why?}) muestran el ciclo que se generó: La devaluación de un país importante provocaba devaluaciones en otros países, o controles de cambios y subidas de aranceles y cuotas de importación. Las medidas proteccionistas eran más intensas en los países que no seguían la devaluación. Los eventos de 1931 son un ejemplo claro: Alemania en los años precedentes había sufrido un parón prolongado en las entradas de crédito exterior y, pese a las salidas de capitales tras la quiebra del Crédit Anstalt austriaco en mayo, y el temor a la extensión del pánico y sin quere devaluar por cuestiones reputacionales, estableció controles de cambios y un impago generalizado de las obligaciones exteriores. En particular de las reparaciones de guerra en julio de 1931. En medio de este shock financiero internacional, el Reino Unido, ante la negativa de los EEUU y Francia a suministrarle financiación exterior, devaluó fuertemente la libra esterlina en septiembre. Tras la devaluación de la libra, que todavía era la moneda central del SMI, los países que no siguieron la senda de la devaluación reforzaron sus medidas proteccionistas. La peor fase de la Gran Depresión comenzó a partir de ese momento. De haber existido entonces un prestamista internacional de última instancia como el FMI, podemos aventurar que las devaluaciones hubieran sido ordenadas y no se habría recurrido a medidas tan proteccionistas. Los impagos de instituciones bancarias grandes (cuando no podían recibir fondos de su banco central) y en particular los impagos de los soberanos demostraron ser mecanismos enormemente nocivos de transmisión de crisis.
} El objetivo era claro: los países que sufrieran perturbaciones financieras debían disponer de financiación internacional para mantener sus tipos de cambio (si eran sostenibles) o efectuar ajustes ordenados más suaves sin generar impagos exteriores. 

Tanto el Plan White como el Plan Keynes preveían el establecimiento de un fondo que ejerciera de prestamista de última instancia y garantizara la estabilidad financiera y cambiaria. Pero, como sabemos, diferían tanto en su tamaño como en la condicionalidad de los préstamos:\footnote{%
    Aparte de que diferían en la moneda de los préstamos. El Plan Keynes establecía que los préstamos fueran en bancor, mientras que White quería que fueran en dólares.
} KEYNES deseaba un Fondo (o \textit{Clearing Union} en su plan) mayor e incondicional (Era la posición natural de un país muy deudor con el exterior, como el Reino Unido en 1943); mientras que WHITE defendía la condicionalidad, para facilitar ajustes exteriores, y, en consecuencia, un menor tamaño. 

Como nueva potencia mundial, el Plan White venció finalmente y fueron sus condiciones las que prevalecieron. En este contexto, se busca impedir muy en particular la financiación de devaluaciones competitivas. Quizá compense aclarar el término de devaluación competitiva en este contexto. Todas las devaluaciones sirven para ganar competitividad, y el FMI original permitía la devaluación cuando la moneda estaba sobrevalorada. De hecho, típicamente se exige una devaluación cuando una moneda está sobrevalorada. Sin embargo, no se admite que las devaluaciones busquen obtener ventajas competitivas mediante políticas de \textit{empobrecimiento del vecino}. Es decir, excesivas. Como vemos, la valoración contiene un margen muy grande de discrecionalidad.


La necesidad de la financiación exterior de emergencia (controlada por la institución) y la obligación de consulta es lo que impide que las devaluaciones sean desordenadas. 

La función financiera del FMI se justifica por la volatilidad de las cuentas corrientes y de los flujos de financiación (por posibles shocks reales o financieros), y la consideración de que la disponibilidad de financiación de un prestamista internacional pennite suministrar confianza en la capacidad de ajuste y mantener un sistema financiero internacional abierto. 

Algunos autores actuales justifican la financiación del FMI por \textbf{fallos de mercado}, pero este razonamiento en concreto no fue el que se produjo en la creación del Fondo ni forma parte del acervo normativo del FMI. Los shocks pueden tener muy diversas causas, y los principales shocks que se querían abordar en el momento de la creación del Fondo provenían de decisiones políticas, generalmente de otros países, como los impagos soberanos o las subidas de aranceles. En los años 40 los flujos oficiales eran predominantes internacionalmente, y las restricciones alos flujos de capitales privados eran la nonna. De hecho, los estatutos del FMI pennitencontroles de capitales (a los flujos de la cuenta financiera), mientras que prohíben los controles de cambios y controles a los pagos por transacciones de la cuenta corriente. Y las dificultades de balanza de pagos que se han financiado tradicionalmente son tanto de cuenta corriente (por problemas 'reales' o desajustes cambiarios) como de cuenta financiera. La volatilidad de los flujos de capitales ha sido elevada tanto en los flujos privados como en los flujos oficiales (incluyendo la Ayuda al Desarrollo).\footnote{%
    En economías de planificación centralizada, la función de prestamista de última instancia también cumple una función para evitar racionamientos ante la volatilidad de los ingresos y pagos por cuenta corriente, y ante la volatilidad de la financiación exterior oficial. Por ejemplo, puede resultar ilustrativo el caso de Cuba: Ha sufrido intensa volatilidad en su cuenta corriente por las oscilaciones de las cosechas agrícolas, y volatilidad en la ayuda en especie y en la financiación exterior que recibía de la Unión Soviética. Por otra parte, los países de la OCÓE que liberalizaron por completo sus movimientos de capitales han sido los que menos recurso han tenido que hacer del FMI, hasta el punto de que el FMI se llegó a considerar como una institución cuya función financiera era exclusivamente para los países en desarrollo.
} 

Modernamente, en un entorno de liberalización de flujos de capitales, un debate importante ha sido sobre la posible función de aseguramiento del FMI ante la volatilidad de flujos de capitales. Los mayores accionistas del FMI suelen ser reacios a este concepto, puesto que el FMI no protege ante contingencias, sino que facilita ajustes. Y la financiación del FMI se somete a adecuadas salvaguardas para preservar sus recursos, principalmente mediante la condicionalidad. (No obstante, el FMI ha avanzado hacia facilidades financieras similares a seguros mediante sus líneas precautorias como la Línea de Crédito Flexible.) 

En los primeros Stand-By Arrangements, se fueron elaborando las condiciones por las que el FMI iba a conceder financiación, fomentando el ajuste de los países cuando se debiera a un desequilibrio fundamental. 

El debate sobre la condicionalidad es uno de los más importantes y centrales en las políticas del Fondo. El plan original de Keynes era una cooperativa de crédito que suministrara financiación esencialmente incondicional y casi automática, pero el principal acreedor, los Estados Unidos, no se mostró dispuesto a conceder financiación sin garantías suficientes de repago. Por ello la condicionalidad fom1ó parte del diseño del Fondo, y se fue refinando con el tiempo. Una parte de la condicionalidad que se ha asociado al FMI no proviene solamente de sus programas, sino también proviene de las obligaciones estatutarias de los miembros. 

En los años 80 y 90, tanto los países en desarrollo como los países en transición mostraron grandes necesidades de reforma estructural, y solicitaron apoyo al FMI, así como la inclusión en los programas -también como una manera de suavizar los requisitos de ajuste macroeconómico, que como veremos en el modelo, pueden considerarse menores s i las reformas estructurales pem1iten aumentar la oferta lo suficiente. Al mismo tiempo, el FMI fue considerando deseable la inclusión de profundas reformas y el resultado fue un gran aumento del volumen de la condicionalidad. En los años 90 y 2000 a los países de baja renta se les llegó a exigir documentos de "estrategias de reducción de la pobreza'' cuyo diseño era extremadamente complejo, puesto que se les llegaba a requerir extremos como una ·negociación inclusiva' de las medidas con participación de la ·sociedad civir (ONG). Con la reducción de la demanda de los programas del FMI en la década de 2000, los requisitos de reforma estructural se fueron suavizando, e incluso en muchos casos obviando. La práctica desde 2000 ha sido tratar de centrarse en la condicionalidad estructural "macro-crítica" y reducir el número de condiciones a lo más manejable y menos sensible políticamente. No obstante, el Fondo ha reintroducido una mayor condicionalidad estructural en materia de medio ambiente bajo la Facilidad de Resiliencia y Sostenibilidad en 2022. 

El tipo de condicionalidad más fuerte son las actuaciones previas (\textit{prior actions}) que tiene que acometer un país antes de recibir la financiación. Estas condiciones previas pueden ser tanto macroeconómicas como estructurales, en función de lo que valore el personal del Fondo. 

La financiación del Fondo pues se dedica pues a fomentar el ajuste, a facilitar que sea más suave. En su ausencia, el ajuste debería, o bien ser más brusco, o producirse vía racionam iento de cantidades de divisas, mediante controles de cambios y de capitales, medidas que son muy dañinas para la prosperidad nacional e internacional. La necesidad de ajuste no es provocada por el FMl, sino que se debe a problemas preexistentes que el FMI ayuda a gestionar. 

La colaboración con acuerdos financieros regionales es un tema de gran importancia desde la crisis financiera europea. En la crisis de la Zona Euro de 2010-12, el FMI consideró que no contaba con recursos- suficientes para abordar la escala de los problemas, y exigir la cofinanciación por parte de los socios europeos. Éstos montaron la European Financia) Stability Facility (EFSF), que luego fue reforzada y subsumida en el Mecanismo Europeo De Estabilización (MEDE-ESM). El FMI contribuyó con un tercio (o menos, en los programas griegos) de la financiación total. Normalmente el suministro de menor financiación confiere un rol secundario a una institución, pero el FMI tuvo un papel mayor que el de un subordinado, en virtud de su experiencia de gestión de crisis. En todo caso, concedió las decisiones sobre tipos de cambio y método de ajuste a sus socios europeos -que optaron por la devaluación interna en lugar de la salida del euro, que habría sido potencialmente incontrolable-. Pero ha llegado a criticar el esquema de decisiones conjuntas ('troika') entre el FMI, la Comisión Europea y el BCE principalmente por estimar · que el BCE debería haber estado sometido también a condicionalidad, al ser el banco central de los países receptores de programas.

\subsubsection{Directorio General: Cuenta General de Recursos (GRA)}
En linea con esta responsabilidad central y a petición de un país miembro, el FMI dispone los recursos de la Cuenta Gneral de Recursos, administrada directamente por el Directorio General. para ponerse a disposición mediante un acuerdo de financiación (\textit{financing arrangement}) o en forma de compras directas (\textit{disbursements}). 

Nos contraremos en los acuerdos de financiación instrumentados a través de acuerdos. En este contexto, un acuerdo es una decisión del Directorio Ejecutivo del FMI por la cual se garantiza a un país miembro que el FMI está dispuesto a proporcionarle recursos durante un período determinado y hasta una cantidad específica, conforme a las condiciones del instrumento financiero correspondiente. 

De entre los acueros de financiación disponibles, podemos distinguir tres grandes categorías informales: (i) estandar; (ii) precautorias; y, (iii) de emergencia.

\paragraph{Facilidades estándar: Stand-By Arrangements (SBA)}
Los acuerdos estándar por excelencia son los \textit{Stand-By Arrangement} (SBA), contemplados desde los inicios del Fondo y diseñados para países miembros con desequilibrios que, se espera, pueda resolver en el corto o mediano plazo. Tienen una duración normal de entre 1 y 2 años, ampliables a 3. 

No obstante, también se cuenta con un programa a más largo plazo: el \textit{Extended Fund Facility} (EFF). Si se considera que el desequilibrio exterior requiere ajustes/reformas estructurales y económicas más amplias,que se prolongaran en el tiempo, este es el mejor acuerdo. Su duración normal es de tres años años, ampliables a cuatro. 

Independientemnte del progama estándar, que será consensuado con el Directorio General, este tendrá un costes: el tipo de interés del DEG (que es variable, y esencialmente es la media ponderada de los tipos de interés de las monedas de la cesta del DEG), más recargos previstos que pueden o no ser aplicables.\footnote{%

\begin{lnum}

    \item \textbf{Por duración}. Si hay un retraso de más de 51 meses en repagar el monto total, se aplica un recargo de hasta 300 puntos básicos;
    \item \textbf{Comisión por disposición}. Cuando se produce un desembolso, se aplican 50 puntos básicos;
    \item \textbf{Comisión por compromiso de fondos}. Cuando se concede la linea de crédito, se aplica una comisión de entre 15 y 60 puntos básicos, en función del nivel de acceso. En el momento en que se produce un desembolso, se sustituye por la comisión por disposición.

\paragraph{Facilidades precautorias: Flexible Credit Line (FCL)}
Las facilidades precautorias se diseñaron como instrumento de prevención de crisis, y van orientadas a dar una señalización positiva para un país cuyas políticas son consideradas sólidas. 

La Línea de Crédito Flexible (Flexible Credit Line, FCL) está disponible para países con fundamentos económicos, políticas y antecedentes de implementación de políticas muy sólidos, y está destinada tanto a la prevención como a la resolución de crisis. En la crisis de 2009, el FMI introdujo una novedad importante en su instrumental de préstamo: la FCL, que eliminaba la condicionalidad tradicional (ex post) para conceder financiación esencialmente incondicional a los países que previamente (ex ante) vinieran desarrollando políticas muy sólidas. Estaba pensada para el apoyo a países como México en 2009. Solamente la han tenido México, Polonia, y Colombia. o Duración: 1-2 años, prorrogable.


La Línea de Precaución y Liquidez (Precautionary and Liquidity Line, PLL) está disponible para países con fundamentos económicos sólidos pero con ciertas vulnerabilidades remanentes que les impiden cumplir los requisitos de elegibilidad de la FCL y la SLL. Es Una línea precautoria similar a la FCL, pero para países con políticas relativamente sólidas. Marruecos ha sido el beneficiario más significativo.

La Línea de Liquidez de Corto Plazo (Short-term Liquidity Line, SLL) está diseñada para proporcionar apoyo de liquidez y presenta varias características innovadoras, incluido el acceso renovable. Tiene los mismos criterios de elegibilidad que la FCL, pero solo está disponible para países que enfrentan necesidades moderadas y potenciales de balanza de pagos a corto plazo derivadas de presiones sobre la cuenta de capital. Se trata, por tanto, de un instrumento precautorio por un valor máximo del 145\% sobre la cuota. Fue puesto en marcha durante la crisis covid-19 en 2020 y es incondicional ni revisiones periódicas. La fortaleza de las políticas se establece a través de la selección previa: se ofrece solamente a países con políticas muy sólidas, aquellos que pueden acceder a la Línea de Crédito Flexible, y se diferencia de ésta en que tiene menor volumen de acceso, mayor automaticidad y menor coste.

\paragraph{Facilidades de emergencia: Rapid Financig Intrument (RFI)}
Las facilidades de emergencia se diseñan como herramientas de concesión de financiación más ágiles y casi automáticas, con menos condicionalidad y un nivel de acceso más limitado.

El Instrumento de Financiamiento Rápido (Rapid Financing Instrument, RFI) es un instrumento de financiación mediante desembolso directo, utilizado por países que enfrentan una necesidad urgente de balanza de pagos sin requerir —o sin tener capacidad para implementar— un programa económico completo. En esencia, un país puede solicitar hasta un 150\% sobre la cuota (50\% anual de su cuota) bajo este instrumento de financiación y sin un calendario de condicionalidad (aunque el Fondo puede solicitar alguna actuación previa -prior action). El periodo de repago es de 3,25-5 años. o Una subcategoría es la Ventanilla para emergencias alimentarias (Food Shock Window), bajo la que se puede obtener hasta un 175\% de cuota.

\section{Fondo Monetaria Internacional: Implementación práctica}

\puntosclave{
    El Modelo tradicional de programación financiera del FMI está orientado al ajuste de un país con problemas de competitividad exterior, de disciplina fiscal y/o monetaria. 
    
    Los prpgramas del FMI han venido incorporando como elementos fundamentales la sostenibilidad de deuda, el equilibrio exterior y la reestructuración del sector financiero.
    
    Los principales debates sobre el contenido y enfoque de los programas del FMI residen en la composición y el gradualismo en el ajuste, el volumen de la financiación a suministrar, la propiedad local de los programas, y el contenido de las reformas estructurales asociadas a un programa.
}


En función de las causas por las que se produzca la solicitud de apoyo financiero, el tipo de ajuste será diferente. 

Al mismo tiempo, el FMI adapta sus programas a las circunstancias específicas de cada pais, y sus recomendaciones sobre el tipo de ajuste evolucionan a la par que la teoría económica, la disponibilidad de asistencia financiera por parte de otras fuentes, y las preferencias políticas de cada país receptor. 


En este apartado vamos a desarrollar los elementos importantes de la programación financiera del FMI. 

En primer lugar analizamos el marco analítico y el sustrato teórico de la programación financiera del Fondo, que es necesario para entender su fundamentación. Posteriormente analizaremos en detalle algunas de las herramientas clave de los programas del FMI: los análisis de sostenibilidad de deuda, el equilibrio del sector exterior y la reestructuración del sector financiero. El último apartado analiza las consecuencias habituales de los programas del FMI, y cómo influyen algunas decisiones clave de diseño y aplicación sobre los resultados.

\subsection{Modelo de Programación Financiera: Marco Teoríco}

El modelo de programación financiera del FMI tiene sus orígenes en los trabajos de POLAK (1957) y aplica, entre otros, el modelo absorción (ALEXANDER, 1952) y el enfoque monetario de la Balanza de Pagos. 

¿Por dónde empezamos? Lo más sencillo será imaginar que somos técnicos del FMI y se nos solicita, por parte de un país en dificultades, la evaluación de la situación para formalizar un programa de asistencia financiera.

\subsubsection{Primera etapa: Diagnóstico}

Lo primero que deberemos hacer será ejecutar un diagnóstico. Lo más habitual es que una misión del FMI se dirija al país para llevar a cabo labores de diligencia debida (\textit{due diligence}) y verificar los datos, su calidad y metodología.\footnote{%
    Por ejemplo, en el caso de Ucrania en 1996-98, la visita de ta misión del FMI al país permitió descubrir que las reservas internacionales utilizables (300 M\$) eran menos de ta mitad de las reportadas oficialmente (700 M\$ de los casi 1.000 M\$ de reservas declarados eran ilíquidos). Es un ejemplo, entre muchos, de cómo la diligencia debida o auditoría de las variables estadísticas claves revela aspectos fundamentales. Otro ejemplo dramático de diligencia debida en el diseño de un programa, aunque algo riguroso, fue la reclasificación de la deuda pública de Portugal en su programa con el FMI de 2011 por sus Partenariados Público-Privados (PPPs, por ejemplo, concesiones de hospitales y autopistas) y empresas públicas, que incrementó la deuda pública portuguesa en 10 puntos de PIB.

La práctica de la diligencia debida requiere gran experiencia y olfato para detectar dónde se producen las ocultaciones o manipulaciones de datos (por parte de las mismas autoridades que solicitan ayuda).
} Esto se hará en todo caso manteniendo un diálogo constante con las autoridades y con otros actores económicos y políticos pues, además de necesario, nos permite obtener más y mejores pistas sobre el fondo del asunto. En definitiva, el diálogo revela preferencias políticas por distintos tipos de actuación, pero también puede revelar deficiencias institucionales y dónde residen las principales resistencias políticas.

Con esto, estaremos en disposición de formular el cuadro macroeconómico, indentificar los desequilibrios y fijar los objetivos del programa.

\paragraph{Objetivos: Identificación y contraste}

¿Cómo podemos identificar los objetivos que debería tener un programa? Esta labor no es sencilla y está sujeta a constante cambio. Podemos decir, sin pérdida de generalidad, que en el diagnóstico podemos llegar a obtener suficiencite información como para poder identificar los objetivos adecuados para el programa dada la naturaleza de los desequilibrios observados; siendo los más clásicos: (i) restaurar la competitividad exterior; (ii) restaurar la disciplina monetaria y fiscal (incluyendo problemas de deuda); (iii) reducir la inflación; y, (iv) restaurar el crecimiento.

Una vez identificados los objetivos debemos proceder a contrastarlos y verificar que estos son adecuados. Esto, como podemos suponer, resulta aún más complicado de realizar. No obstante, podemos considerar el arquetipo de contrastación con la siguiente lógica:\footnote{%
    No obstante, el FMI también se reserva la potestad de, una vez identificados los objetivos generales e instrumentos, traducirlos a un marco estandarizado de objetivos cuantitativos (\textit{quantitative performance criteria}):
\begin{la}
    \item \textbf{Objetivos mínimos de Reservas Internacionales Netas} (Net lnternational Reserves, NIR). Es decir, la estabilización de las reservas del país;
    \item \textbf{Objetivos máximos de Activos Domésticos Netos}, del Banco Central o de las entidades de depósito (Net Domestic Assets). Es decir, limitar el crecimiento de los agregados monetarios para estabilizar la cuenta corriente y la inflación;
    \item \textbf{Objetivos máximos de Déficit}. Generalmente, establecer techos de déficit público (generalmente déficit primario, antes del pago de intereses) o de gasto público primario.
\end{la}

Para conseguir alcanzar estos grandes objetivos, se establecen otros objetivos intermedios típicos importantes como la limitación en la masa salarial pública, de atrasos en los pagos del sector público (\textit{domestic arrears of the public sector}) para restaurar la confianza en los contratos y el funcionamiento del sistema de pagos, o restaurar la viabilidad del sector energético (es muy habitual que, en países con problemas fiscales, los precios de la energía como la gasolina, electricidad o gas estén muy por debajo del valor internacional de mercado).
}
\begin{lnum}
    \item \textbf{Construcción del cuadro macroeconómico}. En primer lugar, el FMI desarrolla el cuadro macroeconómico del país sobre la base de cuatro sectores principales y sus interrelaciones: (i) Sector Real; (ii) Sector Fiscal; (iii) Sector Monetario; y, (iv) Sector Exterior;
    \item \textbf{Escenarios: proyección}. Una vez tenemos el cuadro elaborado, el FMI proyecta escenarios de diferente tomando como calibración el cuadro macroeconómico elaborado. Un ejemplo básico sería la proyección de dos escenarios polares: (i) un escenario \textbf{sin cambio} en las políticas actuales (\textit{no policy change scenario}); y, (ii) un escenario base (\textit{baseline scenario} o \textit{program scenario}) con la aplicación de todos los cambios adecuadamente implementados, dados los desequilibrios observados. 
\end{lnum} 

\paragraph{Instrumentos: ¿Cuáles utilizar?}
Una vez identificados los objetivos y contrastada su adecuidad surge otro problema:: ¿Cómo alcanzarlos? Como sabemos, existe una amplia gama de medidas/instrumentos que permitirían alcanzar objetivos de Política Económica; por tanto, ¿cuál deberíamos escoger? Para soretar estas dificultades, los instrumentos son inicialmente acotados a aquellos que se consideran: controlables de forma directa o indirecta por las autoridades del país, lo que se denominan \textbf{variables de política} (\textit{policy variables}). Por ejemplo, las variables fiscales primarias (antes del pago de intereses) son más controlables que los pagos de intereses, que en muchos casos dependen de los tipos de interés internacionales. Una vez tenemos este subconjunto, es normal tener especial consideración a las variables/instrumentos/medidas preferidas por las autoridades del país, puesto que serán estas las que deban implementarlas y podemos, de esta forma, garantizar o suponer mayor compromiso: cada medida tendrá consecuencias (económicas, políticas, sociales, etc.) diferentes y algunas serán menos adecuadas que otras dado el contexto nacional, que conocen las autoridades del país. 

\paragraph{Condicionalidad: Acceso a las facilidades}

Es frecuente que se introduzcan también objetivos de reformas estructurales (\textit{structural benchmarks}), que pueden afectar a las previsiones. Estos objetivos son, en realidad, criterios de condicionalidad estructural y no objetivos económicos para remediar la situación observada. De esta forma, se encuentran fuertemente personalizados a las situaciones de cada país y contienen mucho (quizás excesivo) grado de discrecionalidad por parte de la misión.

\subsubsection{Segunda etapa: Implementación y evaluación}

Ahora que conocemos el marco teórico que guía la definición, valuración e implementación de los programas del FMI, veamos cómo se ejecutan de manera más realista atendiendo a tres de los grandes objetivos que suele perseguir cualquier programa del FMI. 
La ayuda financiera del FMI se produce por tres tipos de motivos:
• presiones sobre la cuenta exterior (sobre las reservas y tipo de cambio),
• crisis macroeconómica (inflacionaria o fiscal) y
• crisis en el sector financiero.
En muchos casos se combinan los tipos de crisis, con problemas simultáneos de reservas, inflación, fiscales y del sector financiero. En este apartado vamos a examinar el modelo de referencia utilizado por el Fondo para el ajuste macroeconómico y exterior. 


\subsection{Objetivo específico (I): Desequilibrios por deuda}

El Fondo define la sostenibilidad de la deuda como la \textbf{capacidad de mantener el servicio de la deuda sin una corrección excesiva y poco realista de ingresos o gastos}. Como vemos, es una definición algo distinta de la condición matemática de sostenibilidad, pues tiene en cuenta tanto el nivel de la deuda como la \textbf{viabilidad del ajuste}.\footnote{%
    Permite una comprensión sofisticada de las dinámicas de la deuda de un país y sus vulnerabilidades. Sin embargo, sigue siendo una aproximación puesto que todo el análisis depende de proyecciones sobre el comportamiento futuro de las variables clave y, en consecuencia, existe un elemento discrecional muy importante.
}

En los programas del Fondo, los desequilibrios por deuda juegan un papel cada vez más importante dado el grado de movilidad de capitales alcanzado. Por eso, cuando un país solicita un programa del FMI, es necesario identificar, con mucho rigor, si dentro de los objetivos del programa debemos o no establecer la reestructuración o no de la deuda, decisión de mucha trascendencia e impacto económico, político y administrativo.\footnote{%
    ¿Por qué? Como sabemos, los contratos de deuda pública sirven de referencia básica para el sistema financiero de un país, por lo que cualquier incumplimiento o modificación puede enviar señales al mercado muy negativos y considerarse un evento de primera magnitud, equivalente a una bancarrota de escala nacional.
} 

En este contexto, ¿cómo puede el FMI identificar este desequilibrio y establecer los objetivos estríctamente necesarios? Es decir, ¿cómo puedo el FMI establecer idóneamente sus objetivos respecto a la deuda?

\subsubsection{Enfoque adoptado: Nivel de desarrollo}

Lo primero importante será adecuar el análisis al contexto del país: ¿país desarrollado o país en vías de desarrollo? Las razones para adaptar el enfoque son varias, pero resultan especialmente relavantes dos:
\begin{la}
    \item \textbf{Tolerancia}. En primer lugar, se considera un hecho contrastado que los países con mayor renta suelen tener capacidad de asumir niveles mayores de deuda sin sufrir una crisis, lo que se conoce como mayor \textbf{tolerancia al endeudamiento} (REINHART, ROGOFF y SAVASTANO, 2003). Esta mayor tolerancia puede deberse a múltiples factores (instituciones más sólidas, mayor capacidad de recaudación fiscal, mercados financieros más profundos, mayor capital doméstico acumulado, etc.) pero es muy importante a la hora de ajustar las recomendaciones a este hecho: sin pasarnos de frenada;
    \item \textbf{Exposición externa}. En segundo lugar, los países de baja renta suelen presentar mucha mayor exposición exterior debido a la volatilidad de sus ingresos por exportaciones (muy dependientes de las materias primas, en muchos casos). Por el contrario, la deuda externa suele ser de cuantía relativamente inferior en los países más desarrollados, dado que pueden financiarse emitiendo en sus mercados y en su moneda. De esta forma, mientras que los análisis de sostenibilidad sostenibilidad de la deuda (pública y privada) para países en vías de desarrollo suele centrarse en la deuda externa y tomando en especial consideración el papel del tipo de cambio; para los países con mercados financieros más desarrollados, el foco del análisis se suele centrar en la deuda pública. 
\end{la}

Una vez hemos centrado el marco de análisis, se procede a evaluar la posición y sostenibilidad de las variables objetivos.

\subsubsection{Conclusiones: Acceso a programas y condicionalidad}

Una vez realizadas las proyecciones y el test de sensibilidad, el \textit{staff} tomará sus conclusiones en base a los umbrales definidos por el FMI. Estos son estándares que se consideran razonables y adecuados para evitar desequilibrios por deuda que acaben menoscabando el rendimiento de la economíca en otras areas. Si son o no coherentes, consistentes o verdaderamente fundadoa, es otro tema. Se trata de convenciones generalmente aceptadas y con poco consenso teórico. Simplemente es la costumbre. 

El FMI justifica la evaluzación de estos riesgos en base a sus estimaciones empíricas, distribuciones de probabilidad de crisis y niveles de deuda.
Los umbrales son sustancialmente diferentes en función de si se trata de un pais desarrollado o un país en vías de desarrollo:\footnote{%

No obstante, como vemos los umbrales distan mucho de lo que observamos en la práctica. ¿Qué implicaciones tiene esta evaluación? El dictamen de un país respecto a la sostenibilidad de su deuda tendrá implicaciones muy fuertes en la condicionalidad de la financiación y los objetivos que persiga el programa:
\begin{la}
    \item \textbf{Reestructuración}. Cuando la trayectoria de la deuda supera los niveles de riesgo de forma persistente es probable que se exstablezca como objetivo una reestructuración de la deuda. Ahora bien, condicionar a un país desarrollado a algo tan grave ha hecho que los umbrales de riesgo en la práctica se sitúen en niveles en torno al 120\% del PIB. Para los países de renta baja, la reestructuración de deuda es menos traumática ya que los tenedores suelen ser acreedores oficiales, que se coordinan en el Club de París, y otorgar tratamientos de deuda de forma rápida y bastante flexible (coordinada con el FMI);\footnote{%
        De hecho, muchos de estos países han recibido alivios de su deuda en la Iniciativa de Países Pobres Altamente Endeudados (HIPC, por sus siglas en inglés).
    
    No obstante, un nuevo reto para los países de baja renta es que su principal acreedor actual es China, que no forma parte del Club de París (y ha anunciado que no lo acepta como foro de negociación). Este país ha anunciado que por ser país en desarrollo no puede efectuar alivios de deuda de la misma forma que d Club los ha realizado en el pasado.
    } 
    \item \textbf{Límites de endeudamiento exterior}. Si la situación no es tan grave pero sigue siendo problemática, el FMI puede establecer como objetivo limitar el endeudamiento exterior, en particular al endeudamiento en términos no concesionales, a los países menos desorrollados. En caso de un país desarrollado, no se establecería ningún objetivo concreto.
\end{la}

En general, el establecimiento de estos objetivos pretende reducir la exposición del propio FMI, no prestando si la deuda es insostenible. Además, debe tenerse en cuenta que el FMI no emite \textit{rating} de riesgo, sino que se trata de una evaluación interna totalmente condicionada a la solicitud de un programa.\footnote{%
    No obstante, sí lo hará si el programa solicitado es de una cuantía importante. En esas circunstancias, la evaluación se rige por la Política de Acceso Excepcional, 2003/mod. 2016:
\begin{la}
    \item \textbf{Sostenible con alta probabilidad} (zona verde). En esta circunstancia no se le exigirá una reestructuración;
    \item \textbf{Sostenible, pero no alta probabilidad} (zona gris). Se le puede pedir una reestructuración de sus vencimientos a corto plazo (re-profiling) salvo que cuente con financiación de otros países u otro tipo de involucración del sector privado;
    \item \textbf{Deuda insostenible} (zona roja). Se le exigirá una reestructuración para acceder a la financiación del FMI 
\end{la}

\subsection{Objetivo específico (II): Desequilirbios exteriores}
Como hemos visto, una de las situaciones típicas en las que un país solicita apoyo financiero al FMI es una situación de falta de competitividad exterior. 

\subsubsection{Metodología EBA: Tipo de Cambio}

En estas circunstancias, para orientar las recomendaciones sobre tipos de cambio y competitividad, el FMI utiliza su metodología EBA, complementada por las valoraciones de los economistas encargados del país en cuestión (country desk) que suelen tener en cuenta otros indicadores más informales. 

El \href{https://www.imf.org/external/np/res/eba/pdf/080913.pdf}{Enfoque de Balance Externo} (EBA), desarrollado por el Fondo Monetario Internacional (FMI), es una metodología analítica utilizada para evaluar las posiciones externas de un país (como la cuenta corriente y el tipo de cambio real). Su objetivo principal es identificar si estas variables están alineadas con los fundamentos económicos a mediano plazo y si las políticas internas son adecuadas.

Una versión simplificada de esta metodología aplicada a un caso real se puede encontrar aquí: \href{https://www.bccr.fi.cr/investigaciones-economicas/DocIE/2017-DT-03.pdf}{\textit{Enfoque de Balance Externo (EBA) para aproximar un tipo de cambio real}. Banco Central de Costa Rica (2017)}






\paragraph{Conclusiones: Menú de opciones (Política Monetaria)}
Esta metodología suministra una orientación sobre las necesidades de devaluación del tipo de cambio real y del tipo de cambio nominal. 



Para los países con divisa propia y tipo de cambio fijo ajustable, existen diversas estrategias: una devaluación calculada (utilizando las estimaciones EBA y otros indicadores cambiarios), o bien dejar flotar la moneda durante un período de descubrimiento de precios para posteriormente estabilizarla una vez que el mercado ha dejado de ejercer presiones depreciatorias -típicamente mediante el uso de un ancla nominal (como la fijación al dólar o euro) o, en los países con capacidad institucional media y alta, mediante la adopción de un régimen de objetivos de inflación combinado con garantías de independencia para el banco central. 

El FMI suele ofrecer un menú de opciones internamente consistentes a los gobiernos sobre tipo de cambio, que eligen en función de sus preferencias políticas. 

Pero en general, en países con tipos de cambio sobrevalorados, suele exigir una devaluación suficiente como condición previa al suministro de financiación. En los países que forman parte de una unión monetaria, están dolarizados, o no quieren  abandonar su tipo de cambio fijo, las estimaciones de sobrevaloración del tipo de cambio son importantes para orientar las medidas estructurales para tratar de efectuar una devaluación interna que permita reducciones de precios y salarios domésticos, que tienen alto coste político, y coste económico, sobre todo si en el país hay importantes rigideces en precios y salarios (baja competencia, negociación colectiva a nivel sectorial, contratos difíciles de renegociar) y la deuda es elevada (el esfuerzo de servicio relativo de la deuda aumenta puesto que el valor nominal de la deuda se mantiene y el valor nominal de ingresos para repagarla disminuye).

\subsection{Objetivo específico (III): Desequilibrios financieros}

Sin ánimo de exhaustividad, también es necesario conocer los elementos básicos de una estrategia de saneamiento del sector financiero. 

Muchos programas han tenido su epicentro en el sector financiero, con necesidades importantes de reestructuración. La desconfianza sobre la salud del sector financiero ha influido sobre las variables macroeconómicas, por ejemplo, generando presiones depreciatorias sobre los tipos de cambio o presiones sobre el mercado de deuda pública.\footnote{%
    Fue el caso de Ecuador en los arlos 97-99, con una gravísima crisis financiera que implicó el bloqueo de los depósitos (un 'corralito bancario') durante I ano. Los problemas macroeconómicos interactuaron con los problemas del sector financiero. Trás varios rescates bancarios, el desencadenante final del pánico fue el establecimiento de un impuesto a las transacciones financieras por importe de un 1\% de cada transacción, que motivó peticiones masivas deretirada de fondos para eludir el impuesto.
}

De nuevo, es fundamental partir de un diagnóstico bien informado de la situación del sistema financiero, mediante una due diligence en el país, en colaboración con el supervisor nacional. En algunos casos en los cuales se ha producido una pérdida de credibilidad de la supervisión manifestada en una desconfianza de los mercados, y el equipo del FMI para el país (que suele estar compuesto de un máximo de 15 personas) se revela insuficiente para un diagnóstico adecuado en profundidad, el personal del Fondo puede recomendar una auditoría o diagnóstico a una agencia extema.\footnote{%
    En el caso de los programas de Irlanda y Grecia, se encargó la valoración del diagnóstico a la fimia pr ivada BlackRock. A Portugal le asesoró la consultora Oliver Wyman. En el MOU de rccapitalización del sector financiero (en el cual el FMI suministró as istencia técnica), España contrató a las consultoras Oliver Wyman y Roland Berger, así como a las principales auditoras, con la colaboración de otras instituciones internacionales públicas como el BCE e incluso otros supervisores europeos.
} 

Esta externalización permite ganancias de credibilidad pero requiere vigilancia por posibles conflictos de interés de la agencia externa. El diagnóstico inicial debe hacer estimaciones sólidas de los posibles activos dañados o fallidos (morosidad), así como de las posiciones de liquidez.\footnote{%
    Un problema habitual es la infravaloración de los problemas de morosidad por la ocultación por parte de los bancos con problemas En el momento en el que desvelen la extensión de su morosidad corren el riesgo de ser vendidos, nacionalizados o declarados en resolución o concurso de acreedores.
} capital y rentabilidad, normalmente mediante análisis de escenarios (stress test) que sean suficientemente creíbles. También es necesario examinar los controles de riesgo y operativos del sistema.\footnote{%
    Un caso significativo es el de Moldavia en 2014, cuando se produjo la desaparición de 1.000M\$ (un 12.5\% del PIB) del sistema bancario. por diversos fraudes que explotaron las carencias de sistemas de control y las deficiencias institucionales de los bancos locales. Las diferencias entre el país y el FMI sobre qué hacer con los bancos afectados (reformarlos o resolverlos) imposibilitaron el acuerdo para un programa de ayuda financiera que habría podido estabilizar la situación económica del país.
} Una de las recomendaciones del FMI respecto a los stress tests es que para ser creíbles requieren estar respaldados por recursos financieros suficientes para la recapitalización, resolución y garantía de depósitos. El apoyo financiero del FMI o de acuerdos regionales puede ser detem1inante para obtener esa credibilidad. 

\subsubsection{Metodología: Test de sensibilidad}

Los objetivos a corto plazo de una estrategia de reestructuración del sector financiero en los programas del FMI suelen ser, a corto plazo: 

• Recuperación de la estabilidad del sistema: solvencia, liquidez y continuidad de la intermediación financiera • Rapidez en las actuaciones, para evitar el contagio a otras partes del sistema, o a otros países. 
• Restablecer la confianza y prevenir salidas de depósitos o de capitales (pánicos bancarios o financieros). Que se complementan con objetivos a largo plazo, que se suelen introducir también en el programa: 
• Refuerzo del gobierno de las instituciones financieras y del supervisor\footnote{%
    Es común que las instituciones financieras de países con problemas sufran intervenciones políticas que instalen o mantengan a gestores inadecuados. Este tipo de problemas también se pueden producir en el supervisor.
} 
• Mejora del marco legal y de la infraestructura financiera;\footnote{%
    Una ilustración de problemas en el marco legal es el caso del programa de Chipre (2013). El saneamiento del sistema financiero se hacia imposible por la persistencia de una morosidad a niveles estratosféricos (60\% de los préstamos brutos y 1 50\% del PIB en 201 5). Un impedimento clave para la reducción de la morosidad era la imposibilidad de efectuar desahucios o ejecuciones de garantías, que provocaban impagos estratégicos. La aprobación de la ley para remediar el problema fue impopular y se produjo con retrasos, al final del programa.
}
• Introducción de competencia, muchas veces mediante la autorización de inversiones extranjeras en el sector financiero
• Mantener el acceso a los servicios financieros por parte de la población


\paragraph{Conclusiones: Recomendaciones de Política Económica}
Las principales herramientas para una estrategia del sector bancario son:

• Identificación de los activos dañados actuales y potenciales, y en su caso segregación de los mismos en vehículos específicos de gestión de activos dañados ('Bancos malos', Asset Management Companies) que compran los activos con descuento y permiten aflorar las pérdidas.

• Recapitalizaciones bancarias siguiendo el diagnóstico de un stress test, que pueden ser privadas (recomendaciones de ampliaciones de capital) o públicas 

• Resolución o venta de las entidades no viables, incluyendo mediante la posibilidad de 'bancos puente' (bridge banks), que reciben los activos y pasivos de la entidad en resolución y cumplen con sus compromisos para evitar contagios por el incumplimiento de contratos o la incertidumbre sobre la posibilidad de incumplimientos 

• Suministro de liquidez a las entidades solventes o recapitalizadas por parte del banco central o del fondo de garantía de depósitos 

• En su caso, aplicar pérdidas (\textit{bail-in}) a los distintos tenedores de pasivos de una entidad financiera dañada, en primer lugar los accionistas, seguidos de los tenedores de deuda senior, de deuda subordinada, y finalmente, en los casos más graves, los depositantes. La decisión sobre la aplicación de pérdidas a los acreedores es una de las más delicadas, puesto que puede haber una importante masa de perdedores, e incluso contagio sobre otras partes del sistema. Por otra parte, el bail-in tiene la ventaja de que puede ahorrar costes fiscales al contribuyente. 

• En casos extremos de incertidumbre o de gravedad de la situación y cuando los fondos públicos y externos disponibles son escasos, una opción es gestionar unos controles a las salidas de capitales para parar el pánico. Los controles pueden ser de distinta intensidad.\footnote{%
    Dos casos imponantes de controles de capitales bajo programas del FMI conjuntos con Europa han sido los controles de capitales en Chipre en 2013, que duraron dos arios aunque se fueron liberalizando rápidamente. y Grecia en 2015. Los controles de capitales en Chipre se establecieron a la par que un bail-in muy duro a los depositantes de más de 1 00.000€. En Grecia en julio de 2015 se establecieron controles de capitales ante la incertidumbre sobre la continuidad de un programa. los unpagos del Estado griego al FMI. y la permanencia del país en el euro. El control de capitales fue duro. con limitaciones severas a la retirada de efectivo y la prohibición total de transferencias al exterior. Pero no estableció restricciones para los pagos domésticos con tarjeta -puesto que los pagos se mantenían dentro del sistema bancario nacional-. Esta característica pem1itió que las distorsiones a la economía derivadas de los controles de capitales no fueran dramáticas. en un contexto ya deprimido. de alta incertidumbre y de baja inversión desde la recaída económica de finales de 2014.
} Los más suaves son impuestos a las salidas, y los más duros son la restricción total a las retiradas de depósitos o transferencias al exterior ('corralitos'). Estas medidas son las más impopulares de todas, y las más distorsionantes para la economía, aunque pueden ser inevitables si la situación está fuera de control. El objetivo es diseñar las medidas para que sean temporales y limitadas, y con una estrategia de salida


\subsection{Las intervenciones del Fondo Monetario Internacional: ¿cómo son valoradas? ¿qué consecuencias tienen?}

Las consecuencias sobre los países receptores de un programa con el FMl dependen del tipo de programa solicitado, de las circunstancias y de las elecciones efectuadas por las autoridades y por el personal del Fondo. El efecto típico de un programa de ajuste tradicional del FMl puede ser el siguiente:

A corto plazo (fase de ajuste):
• Reducción del crecimiento del PIB por la reducción de la absorción interna derivada del ajuste fiscal y la contracción del crédito interno
• Aumento de la ratio Deuda / PIB, por el reconocimiento de pasivos ocultos o contingentes, así como por el efecto de la reducción del crecimiento.
• Reducción de la inflación
• Reducción del déficit primario y de los atrasos del sector público con otros agentes
• Mejora de la cuenta corriente por la reducción del crédito interno y la absorción
• Depreciación / devaluación del tipo de cambio a niveles más cercanos al equilibrio, que suele ser de forrna previa o inmediatamente posterior a la firma de un programa. En cuanto se ha producido la devaluación inicial, el tipo de cambio suele estabilizarse rápidamente por la disponibilidad de la financiación exterior del FMI.

A medio plazo (fase de estabilización, al cabo de 6 meses o 2 años, en función de la senda de ajuste)
• Recuperación del crecimiento
• Estabilización de la inflación en niveles menores
• Aumento de reservas
• Reducción del déficit público y estabilización progresiva de la senda de la deuda pública

Las variables más importantes que determinan los efectos de un programa son:
• El gradualismo en el ajuste: Si el ajuste se concentra al principio del programa (front/o aded adjustment, 'shock therapy '), el efecto de la fase de ajuste es más intenso y rápido.\footnote{%
    El Banco de España describía en ténninos similares su Informe correspondiente a 1959 ,el efecto inicial del exitoso y muy importante programa de España con el FMI en ese ailo. (Como detalla MUNS en su libro \textit{Historia de las relaciones entre España y el FMI 1958-1982}. página 37).
} El coste político y económico a corto plazo suele ser más evidente y hay menos tiempo para el diseño de las medidas, pero el programa gana en credibilidad. La alternativa es un ajuste espaciado y gradual o incluso retardado (backlo aded), que permite suavizar los efectos de la fase de ajuste y suministrar un tiempo de adaptación a los agentes. En circunstancias en las que el ajuste necesario es muy voluminoso o complejo, puede ser casi imposible concentrarlo al principio del programa. Pero tiene el principal inconveniente de que pierde credibilidad y que puede generarse 'fatiga en el ajuste'. En términos de economía política, puede resultar poco viable administrar dosis continuas de medidas impopulares o mantener el capital político para superar la presiónde los grupos de interés contrarios a las reformas. Por eso el FMI ha establecido una limitación tradicional de la duración máxima de sus programas estándar (los SBA) a 3 años, que ya es una duración elevada para un ajuste si el ciclo electoral es de 4 años, por ejemplo.
• El volumen de la financiación: o Un programa de gran volumen permite suministrar w1 'muro de dinero' (wall of money) para disuadir la especulación cuando los problemas son de liquidez y la necesidad de ajuste es limitada, pero requiere de buenas políticas previas o una cond•i cionalidad reforzada para evitar el riesgo moral. Un problema que se ha apreciado en programas muy grandes es el temor a la subordinación de su deuda por parte de los acreedores privados. Siendo el FMI un acreedor preferente que no acepta quitas, en caso de una reestructuración de la deuda del país con programa, tanto mayor deberá ser la quita para los tenedores privados de deuda. o Por otra parte, programas pequeños del FMI suministran menores garantías de que un país va a estar suficientemente financiado pero suelen darse en condiciones de gran propiedad local de la reforma, que además permiten reforzar el "papel cata/ític:o " de la financiación del Fondo. En esas condiciones, el FMI suele tener que hacer esfuerzos informales para que otros agentes suministren financiación (países amigos, o bancos forzados por las circunstancias a refinanciar al país para evitar una quiebra desordenada). 
• Propiedad local de un programa ('ownership '): Los programas solicitados sin convicción interna de la necesidad de ajuste o reforma y sin la conciencia de que el papel del Fondo es esencialmente de apoyo, suelen tener mayores índices de fracaso. Un programa del FMI como el que concedió a España en 1959 es un ejemplo de muy elevada propiedad local, tanta que los libros de historia suelen considerar el Plan de Estabilización corno una iniciativa interna e incluso omitir el papel fundamental del FMI en un programa típico de la institución. La propiedad local se produjo sobre todo en la adopción de numerosas medidas de reforma estructural. En general, la propiedad local permite reducir los riesgos de reversión de reformas o de ajustes incompletos e ineficaces.
o Un aspecto interesante es el de la condición de "chivo expiatorio" del Fondo respecto a las reformas. Para justificar ajustes o reformas, los gobiernos suelen argumentar que les vienen 'impuestos' por el FMI -incluso en casos en los que los consideren necesarios. Para los acreedores o financiadores, la condicionalidad independiente del FMI permite reducir el desgaste político bilateral.\footnote{%
    A este respecto, resulta interesante comprobar como China ha considerado útil la condicionalidad del FMI en la creación de nuevas instituciones multilaterales. El Acuerdo de Reservas Contingentes (CRA) de los BRICS condiciona los préstamos de un tamrulo mínimo a que el país cuente con un programa del FMI. Esta condición se estableció por exigencia de China. que es el mayor tinanciador del CRA
} No obstante, el uso de esta función de 'chivo expiatorio' conlleva un fuerte desgaste reputacional para la institución y reduce la propiedad local.
• Contenido de reforma estructural: La introducción de reformas estructurales en los programas del FMI ha cumplido varias funciones. Por un lado, en los años 80 y 90, los programas fueron muy intensivos en condicionalidad estructural como sustitutiva del ajuste macroeconómico, que se consideraba más impopular. No obstante, lacondicionalidad estructural también se enfrenta a grupos de interés y también puede ser impopular. El FMI en los últimos años ha tendido a reducir el volumen de la condicionalidad estructural para permitir a los gobiernos concentrarse en las refonnas estructurales críticas (o 'macro-críticas' en la terminología actual del Fondo). Por otra parte, en términos de economía política, en las situaciones de crisis los grupos de interés tienen menos fuerza porque existe una conciencia general de la necesidad de reforma. Por eso suele tener interés incorporar las reformas estructurales más difíciles dentro de los programas de ajuste.
• La inclusión de reestructuración de deuda: Este punto es posiblemente el más difícil y delicado en el diseño de un programa. Una reestructuración puede ser necesaria si un país está en una situación de clara insostenibilidad de deuda, y prolongar una situación insostenible es costoso y contraproducente. En esas circunstancias, es deseable una reestructuración temprana, ordenada y suficientemente profunda. No obstante, en la mayoría de las situaciones en que se · solicita apoyo financiero un país está en una  situación de frontera. El suministro de una 'segunda oportunidad' para así evitar una traumática quiebra e incumplimiento de las obligaciones de deuda y el contagio internacional es precisamente una de las funciones que cumple el FMI. Si, como pretenden algunos accionistas del FMI que desean minimizar el volumen de los préstamos del Fondo, las políticas de préstamo pasan a exigir el recurso sistemático a lareestructuración de deuda, la implicación del Fondo puede ser asociada a un duro estigma en los mercados financieros de deuda, y evitada con mayor intensidad por los gobiernos.
• La con:iposición de los ajustes: Idealmente, los ajustes de gasto público deberían efectuarse principalmente sobre los gastos más ineficientes o innecesarios, con criterios de eficiencia y equidad. Algo similar debería ocurrir con eventuales aumentos impositivos. No obstante, en condiciones de crisis y máxima tensión es difícil identificar el ajuste óptimo, en muchos casos por el limitado tiempo disponible, los limitados recursós· humanos en el país receptor y en el FMI, y la limitada información. La situación ideal es que un país tenga preparado un plan de contingencia y de reforma diseñado con anticipación y que pueda aplicar, pero normalmente existen carencias institucionales y dificultad de previsión que hacen que sea poco frecuente. El arreglo más efectivo es la intemalización de los grandes objetivos del programa por el receptor, y su desarrollo y adaptación por la administración nacional a las circunstancias para garantizar eficacia, equidad y solidez técnica a nivel económico y legal de las medidas de un programa. En definitiva, de nuevo la propiedad local.

\anexo{La institución por dentro: Convenio, gobierno y recursos\label{anx:institucion}}

El Fondo Monetario Internacional es una institución de tratado: casi todo lo que puede y lo que no puede hacer está escrito, y lo que no está escrito no lo puede hacer. Este anexo reconstruye esa maquinaria. Primero, el Convenio Constitutivo: qué fines le impone, qué obligaciones contrae un país al ingresar, con qué forma jurídica entrega el dinero y qué artículos, poco citados, explican los límites de su poder. Después, el gobierno: cómo se calcula la cuota de cada miembro, cómo se convierte esa cuota en votos y los votos en un veto, y dónde se decide realmente. Por último, los recursos: de qué está hecho el balance del Fondo, cuánto puede prestar y qué son exactamente los Derechos Especiales de Giro.

El cuerpo del tema solo necesita cuatro o cinco piezas de este aparato, las que explican por qué el Fondo puede hacer lo que hace y por qué su reforma está bloqueada. Aquí está entero, desde el artículo que crea la obligación hasta la aritmética que convierte $9{.}535{,}5$ millones de DEG en un $1{,}92\%$ del poder de voto. Quien lo haya trabajado debería poder responder a cualquier pregunta sobre el gobierno y las cuentas de la institución sin volver al Convenio.

\subsection{El Convenio Constitutivo: ¿qué puede y qué no puede hacer el Fondo?}\label{anx:institucion:convenio}

El Fondo no tiene competencias generales sobre la economía mundial: tiene las que le atribuye su Convenio Constitutivo, firmado en Bretton Woods el 22 de julio de 1944 y en vigor desde el 27 de diciembre de 1945.\footnote{%
    BOE núm. 305, de 22 de diciembre de 1978, que publica el texto español tras la adhesión. El Convenio es un tratado internacional y sus enmiendas requieren ratificación parlamentaria en los Estados que la exijan, lo que explica que su modificación tarde años.
} Esta observación, que parece formal, resuelve buena parte de los reproches que se le dirigen. Cuando se le acusa de no obligar a Estados Unidos a corregir su déficit, o de no haber forzado a China a revaluar su moneda, se le está reprochando que no ejerza un poder que nunca se le dio: sus miembros le reconocieron la facultad de vigilar y de condicionar su dinero, no la de mandar. \textbf{El Fondo no tiene jurisdicción sobre las políticas de un país; tiene jurisdicción sobre su propio dinero}, y toda su influencia deriva de ahí.

\subsubsection{Los fines del artículo I: ¿para qué se creó?}\label{anx:institucion:fines}

El artículo I enumera seis fines: promover la cooperación monetaria internacional mediante una institución permanente; facilitar la expansión y el crecimiento equilibrado del comercio internacional, contribuyendo con ello a altos niveles de empleo y de renta real; fomentar la estabilidad cambiaria, mantener regímenes de cambios ordenados y evitar depreciaciones competitivas; coadyuvar a establecer un sistema multilateral de pagos y a eliminar las restricciones cambiarias que dificulten el comercio; infundir confianza a los miembros poniendo a su disposición temporalmente, y con las garantías adecuadas, los recursos generales del Fondo, dándoles así oportunidad de corregir los desequilibrios de sus balanzas de pagos sin recurrir a medidas perniciosas para la prosperidad nacional o internacional; y, de acuerdo con lo anterior, acortar la duración y aminorar el grado de desequilibrio de las balanzas de pagos.

Conviene detenerse en el quinto, porque contiene en una sola frase toda la doctrina del crédito del Fondo y las tres condiciones que la limitan. \textbf{Temporalmente}: el dinero se entrega para un plazo, no para financiar un desequilibrio permanente, lo que convierte la distinción entre problema de liquidez y problema de solvencia en la frontera de su actuación. \textbf{Con las garantías adecuadas}: la condicionalidad no es un añadido posterior de los acreedores, sino una exigencia del propio Convenio, que obliga al Fondo a proteger el carácter rotatorio de sus recursos. Y \textbf{sin recurrir a medidas perniciosas}: el criterio para juzgar un programa no es si duele, sino si duele menos que la alternativa de no tenerlo, que en los años treinta fueron el control de cambios, la devaluación competitiva y el impago.\footnote{%
    \textbf{NOTA.-} La comparación relevante no es entre el programa y la situación previa a la crisis, sino entre el programa y el ajuste que habría impuesto la ausencia de financiación. Es el argumento con que el Fondo responde a la crítica habitual de que sus programas son contractivos, y es correcto en su lógica; lo que no resuelve es si el ajuste exigido fue el mínimo necesario, que es donde se juega realmente la discusión ([\authorfont{Ver Anexo \ref{anx:institucion}}], y la evidencia empírica sobre los resultados de los programas).
}

La última frase del artículo I no es retórica: \textit{el Fondo se guiará en todas sus políticas y decisiones por los fines enunciados en este Artículo}. Es una norma operativa, y de ella se sirve el Fondo para justificar la expansión de su actuación hacia materias que el Convenio no menciona. Si la estabilidad financiera, la desigualdad o el clima afectan al crecimiento equilibrado y a la balanza de pagos de un miembro, entran en el mandato por la puerta del artículo I. Ese es el fundamento jurídico del concepto de \textbf{macrocriticidad} con el que el Fondo delimita hoy lo que es de su incumbencia, y también el origen de la acusación de que se extralimita.

\subsubsection{Las obligaciones del miembro: artículos IV, VIII y XIV}\label{anx:institucion:obligaciones}

Un país que ingresa en el Fondo contrae obligaciones de tres clases, y la diferencia entre ellas explica por qué la vigilancia funciona en unas materias y es inoperante en otras.

La primera es la del \textbf{artículo IV}, reescrito por la segunda enmienda en 1978 para adaptarlo al mundo que siguió al derrumbe de las paridades. Su apartado 1 obliga a cada miembro a colaborar con el Fondo y con los demás para establecer regímenes de cambios ordenados y promover un sistema estable de tipos de cambio, y concreta esa colaboración en cuatro compromisos: orientar su política económica hacia el crecimiento ordenado con razonable estabilidad de precios; procurar la estabilidad fomentando condiciones económicas y financieras ordenadas y un sistema monetario que no tienda a producir perturbaciones erráticas; evitar manipular los tipos de cambio o el sistema monetario internacional para impedir el ajuste de la balanza de pagos u obtener ventajas competitivas desleales; y seguir políticas cambiarias compatibles con los compromisos anteriores.

La redacción es decisiva y conviene leerla con atención. Los tres primeros compromisos son \textbf{obligaciones de conducta}: el país debe \textit{orientar}, \textit{procurar}, \textit{fomentar}. No hay resultado exigible ni incumplimiento verificable. Solo el tercero contiene una verdadera prohibición, la de manipular el tipo de cambio para obtener una ventaja desleal, y es inaplicable en la práctica porque exige probar la intención: cualquier gobierno puede sostener que su intervención cambiaria persigue acumular reservas o suavizar la volatilidad, y no ganar cuota de mercado.\footnote{%
    De ahí que la Decisión sobre la Supervisión Bilateral de 2007, que intentó objetivar el concepto introduciendo la noción de \textit{desalineamiento fundamental} del tipo de cambio, fuera recibida con una resistencia tan fuerte por los países con superávit y acabara sustituida por la Decisión sobre Supervisión Integrada de 2012, que renunció a esa vía.\textcolor{orange}{\textsuperscript{[Probable]}}
} El apartado 2 remata la paradoja: cada miembro puede adoptar \textbf{el régimen cambiario que quiera}, con la única excepción de vincular su moneda al oro. Y el apartado 3 encarga al Fondo supervisar el sistema monetario internacional y ejercer una \textbf{firme supervisión} sobre las políticas de tipo de cambio de los miembros, adoptando principios concretos que los orienten.

El resultado es que \textbf{el mandato jurídicamente más fuerte del Fondo recae sobre la materia en que menos puede exigir}: supervisa con firmeza un régimen que cada país elige libremente y respecto del cual solo ha prometido esforzarse. Esta tensión, que se analiza en el cuerpo del tema, no es un defecto de diseño sino el precio de 1978: para que el sistema sobreviviera a la desaparición de las paridades fijas hubo que sustituir una obligación de resultado, mantener la paridad declarada, por una obligación de comportamiento.

La segunda clase de obligaciones, las del \textbf{artículo VIII}, sí son exigibles y tienen contenido preciso: no imponer restricciones a los pagos y transferencias por transacciones corrientes sin aprobación del Fondo; no recurrir a prácticas de tipos de cambio múltiples ni discriminatorias; comprar los saldos en moneda propia que otro miembro haya acumulado por transacciones corrientes; y suministrar la información que el Fondo requiera. Son obligaciones de resultado, su incumplimiento es verificable y su infracción activa el procedimiento sancionador. El \textbf{artículo XIV} permitía mantener transitoriamente las restricciones existentes al ingresar, y casi todos los miembros han ido aceptando las obligaciones del artículo VIII, España entre ellos.\footnote{%
    La primera consulta del Fondo con España, en 1959, se celebró todavía bajo el artículo XIV, que obliga al Fondo a examinar periódicamente las restricciones que el país mantiene y a recomendar su eliminación. La supervisión es, por tanto, anterior al artículo IV de 1978: lo que cambió entonces no fue la práctica de la consulta anual, sino su fundamento y su objeto.
}

\subsubsection{El crédito: el artículo V y por qué el Fondo no presta}\label{anx:institucion:articulov}

La tercera pieza es la que regula el dinero. El artículo V establece que un miembro puede obtener recursos generales del Fondo cuando \textit{declare que tiene necesidad} de realizar la compra por su posición de balanza de pagos o de reservas, o por la evolución de estas. Esa \textbf{necesidad de balanza de pagos} es el requisito de entrada, y no es una fórmula vacía: excluye la financiación de un déficit presupuestario sin contrapartida exterior y es la razón jurídica por la que el Fondo no financia proyectos ni desarrollo, cometido que corresponde al Banco Mundial y a la banca multilateral ([\authorfont{Ver Tema 3.B.30}]).

La forma jurídica de la operación es el detalle técnico con más consecuencias de todo el Convenio, y casi nunca se explica. \textbf{El Fondo no presta: compra y vende monedas.} Cuando un país recibe financiación, lo que ocurre es que el Fondo \textbf{compra} su moneda nacional entregándole a cambio DEG o monedas de libre uso; cuando devuelve, el país \textbf{recompra} su propia moneda entregando DEG o divisas utilizables. De esa construcción se derivan cuatro consecuencias que recorren todo el tema:

\begin{lnum}
    \item \textbf{No hay impago, hay atrasos}. Como la operación es un intercambio de activos y no un préstamo, el incumplimiento no se articula como una quiebra sino como un \textit{atraso} (\textit{arrears}) que activa un procedimiento gradual de sanciones;
    \item \textbf{El balance no reconoce deterioro}. El Fondo no ha dotado nunca provisiones por insolvencia sobre su crédito vivo, algo impensable en cualquier entidad financiera, y lo justifica por su condición de acreedor preferente de hecho, por la relación con unos deudores que son a la vez sus accionistas y por la vinculación del desembolso a revisiones periódicas del programa;
    \item \textbf{La cuota es un pasivo, no un fondo propio}. La suscripción figura en el pasivo del Fondo porque incorpora una obligación incondicional de devolución si el miembro se retira, de modo que la asimilación de la cuota al capital social es exacta en lo político y equivocada en lo contable; y,
    \item \textbf{El coste no es un tipo de interés, son cargos}. Lo que paga el deudor es un \textit{cargo} sobre la tenencia de su moneda por encima de la cuota, y lo que cobra el acreedor es una \textit{remuneración} sobre su posición en el tramo de reserva, una terminología que no es un capricho sino el reflejo de la naturaleza de la operación.
\end{lnum}

El propio artículo V fija los límites cuantitativos clásicos del acceso y, sobre todo, faculta al Fondo para \textit{adoptar políticas sobre el uso de sus recursos generales}, que es la base jurídica de todo el catálogo de facilidades y de la condicionalidad asociada a cada una.

\subsubsection{Los artículos que casi nunca se citan: VI, VII y XXVI}\label{anx:institucion:olvidados}

Tres artículos poco recordados delimitan el poder real del Fondo mejor que los anteriores.

El \textbf{artículo VI} dice lo contrario de lo que casi todo el mundo supone. Su apartado 3 reconoce expresamente que los miembros \textbf{podrán ejercer los controles necesarios para regular los movimientos internacionales de capital}, siempre que no restrinjan con ello los pagos por transacciones corrientes; y su apartado 1 prohíbe al Fondo permitir que sus recursos se destinen a atender una \textbf{salida de capital considerable o continua}, pudiendo incluso exigir al miembro que establezca controles. Es decir: el Convenio protege los controles de capitales y restringe los controles de cambios, exactamente al revés de lo que el Fondo predicó en los años noventa. Esta asimetría, deliberada en 1944 porque los negociadores daban por supuesto que la movilidad de capitales era incompatible con tipos fijos y autonomía monetaria, es el punto de partida obligado para entender el giro doctrinal que condujo a la Visión Institucional sobre los flujos de capital de 2012 y a su revisión de 2022, que se analiza en el cuerpo.

El \textbf{artículo VII} contiene la única huella del plan derrotado en Bretton Woods. Si el Fondo no puede atender la demanda de una moneda porque se le agotan sus tenencias, puede declararla \textbf{moneda escasa} y autorizar a los demás miembros a imponer restricciones cambiarias contra el país que la emite. Es el único mecanismo del Convenio que hace pagar parte del ajuste al país con superávit, en lugar de cargarlo entero sobre el deficitario, y traduce a letra pequeña la preocupación de KEYNES por la asimetría del ajuste ([\authorfont{Ver Tema 3.B.12, Tema 3.B.20}]). Nunca se ha invocado.\textcolor{orange}{\textsuperscript{[Probable]}} Que la cláusula exista y jamás se haya aplicado resume mejor que ningún otro dato dónde está el poder dentro de la institución.

El \textbf{artículo XXVI} regula la salida y la expulsión, y organiza la sanción en tres peldaños sucesivos: la declaración de \textbf{inhabilitación} para usar los recursos generales; la \textbf{suspensión del derecho de voto}, introducida por la tercera enmienda en 1992 precisamente porque el salto directo a la expulsión resultaba inservible por desproporcionado; y la \textbf{separación obligatoria}, acordada por la Junta de Gobernadores. La retirada voluntaria, en cambio, es libre y surte efecto por la simple notificación.\footnote{%
    Checoslovaquia fue separada en 1954 y Cuba se retiró en 1964; Polonia lo había hecho en 1950.\textcolor{orange}{\textsuperscript{[Probable]}} El único caso de suspensión del derecho de voto por la vía de la tercera enmienda ha sido el de Zimbabue, en 2003, restablecido en 2010.\textcolor{orange}{\textsuperscript{[Probable]}}
}

Queda una pieza que conviene no pasar por alto, porque marca la diferencia de fondo con la otra gran institución económica multilateral. El \textbf{artículo XXIX} atribuye la interpretación del Convenio al Directorio Ejecutivo, con recurso ante un Comité de Interpretación de la Junta de Gobernadores. \textbf{El Fondo interpreta su propio tratado y no hay tribunal que lo revise.} Frente a la Organización Mundial del Comercio, cuyo pilar jurisdiccional convierte los compromisos en obligaciones exigibles ante un órgano independiente ([\authorfont{Ver Tema 3.B.33}]), el Fondo carece de solución de diferencias: su única sanción efectiva es el acceso al dinero. Las dos instituciones nacieron del mismo impulso de 1944 y resolvieron de manera opuesta el problema de hacer cumplir lo pactado, y las dos están hoy en crisis por motivos distintos.

\subsubsection{Las siete enmiendas: ¿cómo ha cambiado el Convenio?}\label{anx:institucion:enmiendas}

El Convenio se ha enmendado siete veces, y la lista es un resumen de la historia de la institución:

\begin{lnum}
    \item \textbf{Primera} (en vigor el 28 de julio de 1969): crea el Derecho Especial de Giro y el Departamento que lo administra, como respuesta al problema de liquidez que había planteado TRIFFIN;
    \item \textbf{Segunda} (1 de abril de 1978): da efecto a los acuerdos de Jamaica. Suprime el sistema de paridades y el precio oficial del oro, permite cualquier régimen cambiario y reescribe el artículo IV en términos de supervisión. Es la enmienda que cambia el objeto del Fondo;
    \item \textbf{Tercera} (11 de noviembre de 1992): introduce la suspensión del derecho de voto como peldaño intermedio del régimen sancionador;
    \item \textbf{Cuarta} (10 de agosto de 2009): permite una asignación especial de DEG, de carácter único, para corregir la situación de los países que ingresaron después de las asignaciones de los años setenta y ochenta. Fue aprobada en 1997 y tardó doce años en entrar en vigor, a la espera del Congreso de los Estados Unidos;
    \item \textbf{Quinta} (18 de febrero de 2011): amplía las facultades de inversión del Fondo, base de la actual Cuenta de Inversiones;
    \item \textbf{Sexta} (3 de marzo de 2011): da efecto a la reforma de voz y participación de 2008. Triplica los votos básicos y fija su cuantía en el $5{,}502\%$ del total de votos; y,
    \item \textbf{Séptima} (26 de enero de 2016): suprime los directores ejecutivos nombrados y convierte el Directorio en un órgano íntegramente electivo. Formaba parte de la reforma de cuotas y gobierno de 2010 y, como la cuarta, estuvo seis años esperando la ratificación estadounidense.
\end{lnum}

El procedimiento explica esas esperas. El artículo XXVIII exige que la enmienda sea aprobada por la Junta de Gobernadores y aceptada después por \textbf{tres quintos de los miembros que reúnan el $85\%$ del poder de voto}; cumplido eso, obliga a todos. Como ningún conjunto de miembros alcanza el $85\%$ sin Estados Unidos, \textbf{el Convenio solo se modifica al ritmo del Congreso estadounidense}, y dos de las siete enmiendas lo acreditan.

\subsection{El gobierno: ¿quién decide y con cuántos votos?}\label{anx:institucion:gobierno}

Todo el gobierno del Fondo responde a una sola pregunta: quién pone el dinero del fondo común. A diferencia de las organizaciones internacionales de un Estado un voto, el Fondo se rige por el \textbf{principio del accionista}, y el instrumento que lo encarna es la cuota. Entender la cuota es entender a la vez el reparto del poder, el bloqueo de su reforma y la geografía política de la institución.

\subsubsection{La cuota: cuatro funciones en un solo número}\label{anx:institucion:cuota}

La cuota de un miembro es una cantidad única, expresada en DEG, que desempeña simultáneamente cuatro funciones distintas.

\begin{center}\begin{minipage}{\linewidth}\centering
{\small\textbf{Las cuatro funciones de la cuota y por qué bloquean la reforma del Fondo}}\par\vspace{0.3em}
\includegraphics{anx1_cuota.pdf}\par\vspace{0.3em}
{\small\textit{Fuente: }elaboración propia a partir del Convenio Constitutivo, arts. III, V, XII y XVIII.}
\end{minipage}\end{center}

La primera es la \textbf{suscripción}: la cuota es lo que el país aporta al fondo común, y lo aporta en dos tramos de naturaleza muy distinta. El $25\%$ se desembolsa en activos de reserva, esto es, en DEG o en monedas de libre uso, y constituye dinero efectivamente utilizable por el Fondo. El $75\%$ restante se pone a disposición en moneda nacional del propio país, normalmente mediante un pagaré no remunerado depositado en su banco central. Esta construcción tiene una consecuencia que conviene fijar desde ahora: \textbf{la mayor parte de los recursos nominales del Fondo no es utilizable}, porque de poco sirven los pesos argentinos o las liras turcas depositadas cuando lo que necesita un país en crisis son dólares o euros. De los recursos totales, solo son movilizables las monedas de los miembros con posición externa suficientemente sólida, y por eso el Fondo publica trimestralmente un \textit{plan de transacciones financieras} con la lista de monedas utilizables.

La segunda función es el \textbf{voto}, que se examina en el apartado \ref{anx:institucion:voto}. La tercera es el \textbf{acceso}: todos los límites de financiación del Fondo, sin excepción, se expresan en porcentaje de la cuota del país, de modo que la cuota determina a la vez lo que se aporta y lo que se puede pedir. La cuarta es la participación en las \textbf{asignaciones generales de DEG}, que se reparten en proporción a las cuotas, razón por la cual la asignación de 2021 entregó la mayor parte de los nuevos derechos a los países que menos los necesitaban.

De la superposición de las cuatro funciones nace el problema de gobierno más persistente de la institución. \textbf{No se puede aumentar la voz de un país sin aumentar a la vez su contribución, su acceso al crédito y su participación en las asignaciones de DEG}, porque son el mismo número. Y como el realineamiento de las participaciones relativas exige que alguien baje, y ningún miembro ha aceptado nunca una reducción de su cuota nominal, la única manera de recomponer el reparto es aumentar el total. Por eso toda reforma del gobierno del Fondo pasa por una revisión general de cuotas, y toda revisión general de cuotas pasa por el Congreso de los Estados Unidos.

\subsubsection{La fórmula de cuotas: ¿cuánto le corresponde a cada país?}\label{anx:institucion:formula}

Para orientar esa discusión, el Fondo dispone de una fórmula que calcula la cuota que \textit{correspondería} a cada país según su peso en la economía mundial. La vigente se acordó en la reforma de 2008 y combina cuatro variables, todas ellas medidas como porcentaje del total de los miembros:

$$CQS_i=\bigl(0{,}50\,Y_i+0{,}30\,O_i+0{,}15\,V_i+0{,}05\,R_i\bigr)^{0{,}95}$$

donde $CQS_i$ es la participación calculada del país $i$ (\textit{calculated quota share}). El \textbf{producto interior bruto}, $Y_i$, se mide como una mezcla de dos valoraciones, $Y_i=0{,}60\,Y_i^{\text{mercado}}+0{,}40\,Y_i^{\text{PPA}}$; la \textbf{apertura}, $O_i$, es la media de los pagos e ingresos corrientes de los últimos cinco años; la \textbf{variabilidad}, $V_i$, es la desviación típica de los ingresos corrientes y de los flujos netos de capital respecto de una tendencia centrada de tres años, medida sobre un periodo de trece\textcolor{orange}{\textsuperscript{[Probable]}}; y las \textbf{reservas}, $R_i$, son la media de doce meses de las reservas oficiales. El exponente $0{,}95$ es el \textbf{factor de compresión}, que acerca todas las participaciones a la media y, por tanto, recorta las de los países grandes en beneficio de los pequeños.

Cada variable responde a una concepción distinta de qué debe medir una cuota, y de ahí que la fórmula sea un campo de batalla permanente.

El \textbf{producto}, con la mitad del peso, es la única variable que todos aceptan como pertinente, pero su medición no es neutral: valorado a tipos de cambio de mercado favorece a las economías avanzadas, y valorado en paridad de poder adquisitivo favorece a las emergentes, de modo que el reparto $60$-$40$ no es una decisión estadística sino el resultado de una negociación. La \textbf{apertura} procede directamente de 1944, cuando el Fondo financiaba desequilibrios comerciales y la vocación exterior era el mejor indicador de la exposición de un país; hoy premia desproporcionadamente a las economías avanzadas pequeñas y muy abiertas, y computa como apertura el comercio interior de la zona del euro, que para sus miembros ya no es comercio exterior en sentido económico.\footnote{%
    \textbf{OJO.-} Es la crítica técnica más sólida contra la fórmula vigente y la que más beneficia a los países europeos. Corregirla exigiría consolidar el comercio intracomunitario, lo que reduciría las cuotas europeas de manera apreciable y es una de las razones por las que la revisión de la fórmula no avanza.\textcolor{orange}{\textsuperscript{[Probable]}}
} La \textbf{variabilidad} se incorporó para dar más cuota a quienes sufren mayores oscilaciones en sus ingresos exteriores, es decir, a los prestatarios potenciales: es coherente con la cuota entendida como límite de acceso, pero contradictoria con la cuota entendida como aportación, porque da más peso a quien previsiblemente sacará dinero del fondo. Las \textbf{reservas}, en fin, fueron en su día la mejor medida de la capacidad de contribuir y hoy son residuales, porque el país que emite la moneda de reserva puede aportar sin tenerlas.

Dos observaciones cierran el apartado. La primera es que \textbf{la fórmula no es normativa ni vinculante}: no mide nada que la teoría económica defina, sino que pondera cuatro magnitudes según un acuerdo político, y su resultado solo informa la negociación. La segunda es que, pese a ello, cumple una función real, porque convierte la discusión sobre el poder en una discusión sobre ponderaciones, que es políticamente más manejable. La distancia entre la participación calculada y la efectiva es el indicador estándar de infrarrepresentación, y es grande y persistente en el caso de las grandes economías emergentes, muy en particular de China, cuyo crecimiento ha sido siempre más rápido que el de su cuota.

\subsubsection{El voto: la aritmética del veto}\label{anx:institucion:voto}

El poder de voto de un miembro suma dos componentes. Los \textbf{votos por cuota}, a razón de un voto por cada $100{.}000$ DEG, y los \textbf{votos básicos}, iguales para todos los miembros, cuya cuantía conjunta la sexta enmienda fijó en el $5{,}502\%$ del total de votos, con ajuste automático en cada aumento de cuotas. Los votos básicos son el único elemento igualitario del sistema: compensan en parte el principio del accionista y pesan tanto más cuanto menor es el país.

La aritmética es sencilla y conviene saber hacerla. Con unas cuotas totales del orden de $476{.}400$ millones de DEG y $191$ miembros, los votos por cuota suman unos $4{,}76$ millones; como estos representan el $94{,}498\%$ del total, los votos totales ascienden a unos $5{,}04$ millones y los votos básicos, a unos $277{.}400$, esto es, $1{.}452$ por miembro.\textcolor{orange}{\textsuperscript{[Probable]}} Aplicado a dos casos:

\begin{center}
\begin{tabular}{|l|r|r|r|r|}
\hline
 & \textbf{Cuota} & \textbf{\% de} & \textbf{Votos} & \textbf{\% de} \\
 & (millones de DEG) & \textbf{las cuotas} & (básicos $+$ por cuota) & \textbf{los votos} \\
\hline
Estados Unidos & $82{.}994{,}2$ & $17{,}42$ & $1{.}452+829{.}942=831{.}394$ & $16{,}49$ \\
\hline
España & $9{.}535{,}5$ & $2{,}00$ & $1{.}452+95{.}355=96{.}807$ & $1{,}92$ \\
\hline
País con cuota de $10$ & $10{,}0$ & $0{,}002$ & $1{.}452+100=1{.}552$ & $0{,}031$ \\
\hline
\end{tabular}
\end{center}

La última fila muestra el efecto igualador en estado puro: un miembro con una cuota de diez millones de DEG tiene una participación en los votos quince veces mayor que su participación en las cuotas. El efecto, sin embargo, es pequeño en el agregado, porque el conjunto de los votos básicos no llega al $6\%$ del total.

Sobre esa base operan tres mayorías. La \textbf{simple} es la regla general y cubre casi todo lo que decide el Directorio Ejecutivo. La del \textbf{$70\%$} se reserva a decisiones financieras como la fijación del tipo de cargo y de la remuneración. Y la del \textbf{$85\%$} se exige para lo que altera la institución: modificar las cuotas, asignar DEG, vender oro, enmendar el Convenio o acordar una separación obligatoria. La primera enmienda sometió dieciocho tipos de decisiones a mayoría cualificada y la segunda añadió unas treinta y nueve más, de modo que hoy el núcleo de la actividad estructural del Fondo está sujeto a estos umbrales.

La consecuencia es la pieza central del gobierno de la institución. \textbf{Con el $16{,}49\%$ de los votos, Estados Unidos puede bloquear por sí solo todas las decisiones que requieren el $85\%$}, y ningún otro miembro se acerca a ese umbral. Conviene añadir dos precisiones que suelen omitirse. La primera es que el veto no es un accidente aritmético: cuando la segunda enmienda elevó del $80\%$ al $85\%$ las mayorías reforzadas, la participación estadounidense estaba descendiendo, de manera que el umbral se ajustó al poder y no al revés.\textcolor{orange}{\textsuperscript{[Probable]}} La segunda es que los Estados miembros de la Unión Europea superan conjuntamente ese porcentaje, pero votan repartidos entre varias sillas y sin mandato común, de modo que disponen de un veto potencial que no ejercen: es el argumento de fondo del debate sobre una silla europea única, que liberaría asientos para las economías emergentes pero crearía un segundo poder de bloqueo.

\subsubsection{Los órganos: ¿dónde se decide de verdad?}\label{anx:institucion:organos}

El reparto formal de competencias entre los órganos del Fondo no coincide con el reparto real de influencia, y conviene tener presentes los dos.

\begin{center}\begin{minipage}{\linewidth}\centering
{\small\textbf{El circuito de una decisión del Fondo: quién propone, quién decide y con qué mayoría}}\par\vspace{0.3em}
\includegraphics{anx1_decision.pdf}\par\vspace{0.3em}
{\small\textit{Fuente: }elaboración propia a partir del Convenio Constitutivo, arts. XII y XXVIII, y de la información institucional del FMI (2026).}
\end{minipage}\end{center}

La \textbf{Junta de Gobernadores} es el órgano soberano: un gobernador y un suplente por miembro, normalmente el ministro de Economía o Hacienda o el gobernador del banco central, según decida cada país.\footnote{%
    España designa a la titular del ministerio competente en economía; Alemania, al presidente del Bundesbank. La elección revela si el país concibe al Fondo como una institución financiera o como un foro de política económica.
} Se reúne una vez al año en la Asamblea Anual, pero esa reunión es formal: un debate entre casi cuatrocientos dignatarios es impracticable y los gobernadores votan por correo y sin discusión. Su competencia es, sin embargo, exclusiva en lo que el Convenio le reserva, que es justamente lo estructural: las cuotas, las asignaciones de DEG, la admisión y separación de miembros, las enmiendas y la propia organización del Directorio.

El \textbf{Directorio Ejecutivo} es el órgano que gobierna de verdad. Desde el 1 de noviembre de 2024 lo integran \textbf{veinticinco directores}, todos electivos desde la séptima enmienda, residentes en Washington y reunidos varias veces por semana. El vigesimoquinto asiento se creó por resolución de la Junta de Gobernadores de 16 de julio de 2024 para reforzar la representación del África subsahariana, cuyos cuarenta y cinco miembros pasaron de dos a tres circunscripciones. Los miembros se agrupan en \textbf{sillas} (\textit{constituencies}) que eligen a su director cada dos años; las mayores economías disponen de silla propia, otras encabezan de forma permanente una silla compartida y otras, como España, acuerdan con sus socios reglas de rotación.\footnote{%
    La silla de España la integran, además, México, Colombia, Costa Rica, El Salvador, Guatemala y Honduras. Pertenecer a una silla con países que son o pueden ser prestatarios obliga al director a representar intereses que no siempre coinciden, lo que es una característica general del sistema y no una anomalía española.
} El Directorio decide por \textbf{consenso}, no por votación formal: el presidente constata el \textit{sentir de la reunión}, ponderado por los votos que cada director representa. El voto, por tanto, casi nunca se ejerce, pero disciplina permanentemente el consenso, que es la forma habitual en que opera el poder en esta institución.

El \textbf{Comité Monetario y Financiero Internacional} reproduce la composición del Directorio en el plano ministerial: veinticinco miembros, uno por cada silla, que se reúnen dos veces al año con ocasión de las Reuniones de Primavera y de la Asamblea Anual. Carece de poder formal de decisión, pero su comunicado fija las orientaciones estratégicas y el programa de trabajo del semestre siguiente. Su historia es, en todo caso, la de un desplazamiento: primero el G7 y, desde 2009, el G20 le han arrebatado la función de foro donde se acuerda la respuesta a una crisis, entre otras razones porque el enfrentamiento entre bloques Norte-Sur hacía del Comité un lugar poco operativo y porque el G20 incorpora a China con un peso que el reparto de cuotas todavía no le reconoce.

\subsubsection{La gerencia y el personal: la convención de las nacionalidades}\label{anx:institucion:gerencia}

Por encima del Directorio, que preside, está la \textbf{Directora Gerente}, nombrada por cinco años renovables. Un acuerdo tácito desde 1946 reserva el puesto a un europeo, como contrapartida de la presidencia del Banco Mundial, reservada a un estadounidense; el cargo lo ocupa desde 2019 Kristalina GEORGIEVA, reelegida para un segundo mandato que comenzó el 1 de octubre de 2024. La convención se extiende al número dos: \textbf{el primer subdirector gerente es siempre estadounidense}, y desde el 6 de octubre de 2025 lo es Dan KATZ, procedente del Tesoro de los Estados Unidos, que sucedió a Gita GOPINATH. Completan la gerencia otros tres subdirectores, hoy Bo LI, Kenji OKAMURA y Nigel CLARKE.

El reparto por nacionalidades no es anecdótico. Que el principal accionista no presida la institución pero controle de forma permanente su segundo puesto ejecutivo, además de disponer del único veto, es una solución de compromiso que concentra la influencia estadounidense en los puestos donde se filtra lo que llega al Directorio. El nombramiento de 2025 reavivó precisamente ese debate.\textcolor{orange}{\textsuperscript{[Probable]}}

Por debajo están los directores de departamento, entre los que destaca el \textbf{economista jefe}, hoy Pierre-Olivier GOURINCHAS, que firma las \textit{Perspectivas de la Economía Mundial} y es la cara pública del análisis del Fondo. El personal, en torno a $3{.}100$ personas de más de ciento sesenta nacionalidades,\textcolor{orange}{\textsuperscript{[Probable]}} presenta una diversidad geográfica notable y una homogeneidad de formación extrema: la gran mayoría de los economistas procede de un número reducido de programas de doctorado estadounidenses y británicos. Esa homogeneidad tiene una virtud, la coherencia doctrinal y la posibilidad de aplicar un marco común a ciento noventa y un países, y un defecto que la propia Oficina de Evaluación Independiente documentó al examinar la actuación del Fondo antes de 2008: el \textbf{pensamiento de grupo}, la dificultad para tomar en serio hipótesis ajenas al consenso profesional del momento.

La consecuencia práctica es que \textbf{la iniciativa está en el personal técnico}. Es el \textit{staff} quien diagnostica, negocia el programa con las autoridades y redacta el documento que el Directorio aprueba o rechaza, de modo que quien influye sobre el Fondo no es solo quien tiene votos, sino quien participa en la elaboración técnica. Es la razón por la que los países con administraciones capaces de dialogar con las misiones obtienen resultados distintos de los que solo pueden aceptar o rechazar lo que se les presenta.

\subsubsection{Las revisiones generales: ¿por qué está bloqueado el realineamiento?}\label{anx:institucion:revisiones}

El Convenio obliga a revisar las cuotas al menos cada cinco años. Se han completado dieciséis revisiones generales, y varias concluyeron sin aumento alguno. Las tres últimas describen con precisión el estado del gobierno de la institución.

La \textbf{reforma de 2010}, que correspondió a la decimocuarta revisión, fue la más ambiciosa: duplicó las cuotas totales hasta unos $477{.}000$ millones de DEG y trasladó más de seis puntos de participación hacia las economías emergentes y en desarrollo dinámicas, con China entre las principales beneficiarias y también España, que arrastraba una infrarrepresentación antigua. Entró en vigor en enero de 2016, seis años después de acordarse, cuando el Congreso estadounidense la ratificó.

La \textbf{decimoquinta revisión} concluyó en 2020 \textbf{sin ningún aumento}, un desenlace que en la práctica equivale a un fracaso: el Fondo renunció a ampliar sus recursos permanentes en vísperas de la mayor crisis desde su fundación y hubo de cubrir la diferencia con recursos prestados.

La \textbf{decimosexta revisión} se aprobó el 15 de diciembre de 2023 y su diseño es revelador. Acordó un aumento \textbf{equiproporcional del $50\%$}, esto es, $238{.}600$ millones de DEG que elevarían el total a $715{.}700$ millones, y lo hizo \textbf{sin realineamiento alguno}: todas las cuotas crecen el mismo porcentaje, de modo que las participaciones relativas, y con ellas el poder de voto, quedan exactamente donde estaban en 2010. El realineamiento se aplazó a unos \textit{enfoques} que deberían guiar la decimoséptima revisión. España pasaría de $9{.}535{,}5$ a $14{.}303{,}3$ millones de DEG.

El aumento no está en vigor. Requiere el consentimiento de miembros que reúnan el $85\%$ de las cuotas y, además, la reducción correlativa de los Nuevos Acuerdos de Préstamo, y el plazo para prestar esos consentimientos se ha prorrogado en dos ocasiones, la última hasta el 15 de noviembre de 2026.\footnote{%
    Decisión del Directorio Ejecutivo de 8 de mayo de 2026, que prorrogó por seis meses el plazo que vencía el 15 de mayo. El comunicado del Comité Monetario y Financiero Internacional de octubre de 2026 instaba a los miembros a completar sus procedimientos internos sin más demora, lo que indica que los consentimientos seguían incompletos en esa fecha.\textcolor{orange}{\textsuperscript{[Probable]}} El estado final de la revisión debe verificarse antes de exponer el tema.
}

El conjunto permite formular la tesis con la que se cierra el apartado y que el cuerpo recoge. \textbf{El diseño equiproporcional de la decimosexta revisión es la confesión de que el realineamiento se ha vuelto imposible}: para evitar la discusión sobre quién sube y quién baja, se ha hecho crecer a todos por igual. El Fondo podrá así disponer de más recursos propios y depender menos de los prestados, pero su estructura de poder seguirá reflejando la economía mundial de 2010, con una China cuya participación en el producto mundial casi duplica la que le reconocen sus cuotas. Esa distancia entre el peso económico y el peso institucional es el primero de los tres frentes de cuestionamiento que se examinan en el último bloque del tema.

\subsection{Los recursos: ¿con qué dinero opera el Fondo?}\label{anx:institucion:recursos}

El Fondo no es un banco: es una \textbf{cooperativa de crédito entre bancos centrales}. No capta depósitos, no se financia en los mercados y no crea dinero; todo lo que presta sale de los balances de los bancos centrales de sus miembros, bien porque lo han aportado como cuota, bien porque se lo prestan.\footnote{%
    Jurídicamente podría endeudarse con cualquier fuente, y de hecho en 2009 emitió pagarés denominados en DEG suscritos por China, Brasil, Rusia e India para reforzar sus recursos ante la crisis.\textcolor{orange}{\textsuperscript{[Probable]}} Pero nunca ha emitido deuda en los mercados, a diferencia del Banco Mundial o del Mecanismo Europeo de Estabilidad, que se financian con emisiones avaladas por el capital de sus socios. La diferencia no es menor: los recursos del Fondo son \textbf{monetarios}, creados o inmovilizados por bancos centrales, y por eso aportar al Fondo no genera déficit ni deuda pública en el país que aporta.
} De ahí se sigue el rasgo que lo define: \textbf{el esfuerzo de aportar depende de la moneda que se emita}. Para Estados Unidos, cuyo banco central crea los dólares que desembolsa, el coste es prácticamente nulo; para un país con moneda débil, reunir el tramo pagadero en activos de reserva es un sacrificio real. Fue la situación de España en 1958, con reservas exiguas y una peseta sobrevalorada.

\subsubsection{Las tres capas de recursos}\label{anx:institucion:capas}

Los recursos del Fondo se ordenan en tres capas que se usan en un orden determinado y tienen una solidez muy distinta.

\begin{center}\begin{minipage}{\linewidth}\centering
{\small\textbf{Las tres capas de recursos del Fondo y lo que cambiaría la 16.ª revisión}}\par\vspace{0.3em}
\includegraphics{anx1_recursos.pdf}\par\vspace{0.3em}

La primera son las \textbf{cuotas}, unos $476$ mil millones de DEG, que son el recurso propio y permanente: están disponibles sin ninguna decisión adicional y pertenecen a los $191$ miembros. Su limitación, ya señalada, es que solo resulta utilizable la parte depositada en monedas de países con posición externa sólida, más el $25\%$ pagado en activos de reserva.

La segunda son los \textbf{Nuevos Acuerdos de Préstamo} (NAP), unas líneas de crédito permanentes que una cuarentena de miembros mantiene abiertas a favor del Fondo, por unos $364$ mil millones de DEG. Nacieron en 1998, tras las crisis de México y Asia, cuando se comprobó que las cuotas no bastaban para una crisis de cuenta financiera, y se duplicaron en 2021. Son recursos prestados: se activan por periodos determinados y su activación exige el consentimiento de los participantes.\footnote{%
    Subsisten además los \textbf{Acuerdos Generales de Préstamo}, de 1962, suscritos con el Grupo de los Diez para atender las tensiones del sistema de paridades; su importe, unos $17$ mil millones de DEG, es hoy residual, pero su existencia recuerda que la insuficiencia de las cuotas es un problema tan antiguo como la institución.
}

La tercera son los \textbf{acuerdos bilaterales de préstamo}, unos $137$ mil millones de DEG, negociados uno a uno y renovables por plazos cortos. Es la capa más frágil, porque depende de que cada acreedor quiera renovar, y la que en teoría debería ser excepcional, aunque lleva activa desde 2012.

Sumadas, dan al Fondo una capacidad de préstamo cercana al \textbf{billón de dólares}. Pero la cifra agregada oculta lo importante, que es la \textbf{composición}. Como las revisiones de cuotas se han ido frustrando, el Fondo ha cubierto el hueco con recursos prestados, hasta el punto de que hoy algo más de la mitad de su capacidad no es suya. Eso tiene dos consecuencias: la institución depende de un grupo reducido de acreedores que renuevan sus líneas, lo que les da una influencia que no se refleja en sus votos; y su capacidad es revisable, de modo que la seguridad que ofrece a un país en crisis es menor de lo que sugiere el total. \textbf{El sentido profundo de la 16.ª revisión está ahí}: no agranda el Fondo, sino que convierte en permanente y de todos lo que hoy es prestado y de unos pocos.

\subsubsection{El balance: por qué el Fondo tiene las cuentas separadas}\label{anx:institucion:balance}

La estructura financiera del Fondo desconcierta al principio porque parece la de un grupo de sociedades, con cuentas que se presentan por separado. No lo es: es una organización internacional unitaria que mantiene una separación contable estricta entre compartimentos con reglas y recursos distintos.

El \textbf{Departamento General} es el núcleo y comprende tres cuentas. La \textbf{Cuenta de Recursos Generales} (CRG) es la que importa: por ella pasan las cuotas, los préstamos recibidos, las compras y recompras de los miembros y los cargos y remuneraciones. La \textbf{Cuenta de Inversiones} gestiona los recursos no comprometidos en dos subcuentas, una de renta fija, cuyo objetivo es superar el tipo del DEG en $50$ puntos básicos, y una patrimonial, dotada con los beneficios de las ventas de oro de 2009-2010, que busca un rendimiento real del $3\%$ a largo plazo para contribuir a los gastos administrativos. La \textbf{Cuenta Especial de Desembolsos} recibe los beneficios de aquellas ventas de oro y es el vehículo por el que el propio Fondo financia operaciones que sus recursos ordinarios no pueden atender.

Junto al Departamento General están las \textbf{entidades administradas}: el \textbf{Departamento de DEG}, cuya separación impone el propio Convenio; el \textbf{Fondo Fiduciario para la Reducción de la Pobreza y el Crecimiento} (PRGT), que presta en condiciones concesionales a los países de renta baja; el \textbf{Fondo Fiduciario para la Resiliencia y la Sostenibilidad} (RST), creado en 2022; y los fondos de alivio de deuda.

La razón de la separación es financiera y no organizativa, y conviene entenderla porque explica una tensión permanente de la institución. \textbf{La CRG no puede prestar barato}: sus recursos son monetarios, tienen que rotar y han de remunerarse a tipos relacionados con el mercado, porque son los bancos centrales acreedores quienes los ponen. Prestar al $0\%$ a un país de renta baja exige, por tanto, sacarlo de la CRG y montar un fondo fiduciario con dos fuentes: préstamos voluntarios de miembros, que cobran el tipo de mercado, y \textbf{contribuciones a fondo perdido} que subvencionan la diferencia. De ahí que la capacidad del Fondo para ayudar a los países más pobres no dependa de su tamaño, sino de la generosidad presupuestaria de sus miembros, y que el PRGT viva en una crisis de financiación recurrente que la demanda posterior a la pandemia agravó.

En el activo, el Fondo tiene monedas de sus miembros, su crédito vivo, tenencias de DEG, inversiones y \textbf{oro}: unos $90{,}5$ millones de onzas, alrededor de $2{.}814$ toneladas, que lo convierten en el tercer tenedor oficial del mundo.\textcolor{orange}{\textsuperscript{[Probable]}} Ese oro está contabilizado a su coste histórico, muy por debajo del precio de mercado, de modo que el balance esconde una plusvalía latente enorme que no figura como recurso. Venderlo exige el $85\%$ de los votos, y la última vez que se hizo, entre 2009 y 2010, los beneficios dotaron la subcuenta patrimonial y subvencionaron los préstamos concesionales. Esa plusvalía es, en la práctica, el colchón último de la institución y la respuesta recurrente de quienes proponen financiar el PRGT sin pedir dinero nuevo a los Tesoros.

En el pasivo están las cuotas suscritas, los préstamos recibidos y los \textbf{saldos precautorios}, que son las reservas acumuladas con los márgenes de intermediación para cubrir pérdidas. El Directorio elevó su objetivo a medio plazo a $25$ mil millones de DEG en 2022 y su suelo a $20$ mil millones en 2024; el objetivo se alcanzó al cierre del ejercicio de 2024, antes de lo previsto, y la revisión de abril de 2026 decidió mantenerlo.

\subsubsection{De la cuota al crédito: el tramo de reserva y los tramos de crédito}\label{anx:institucion:tramos}

Queda por explicar cómo se traduce todo esto en una operación concreta, y aquí la terminología del Fondo, heredada de 1944, oscurece una mecánica sencilla. La clave es que el Fondo lleva la cuenta de lo que ha hecho cada país mediante \textbf{sus tenencias de la moneda de ese país}, medidas en porcentaje de su cuota.

\begin{center}\begin{minipage}{\linewidth}\centering
{\small\textbf{De la cuota al crédito: tramo de reserva, tramos de crédito y recargos}}\par\vspace{0.3em}
\includegraphics{anx1_tramos.pdf}\par\vspace{0.3em}
{\small\textit{Fuente: }elaboración propia a partir del Convenio Constitutivo, art. V, y de la política de cargos y recargos del FMI vigente desde el 1 de noviembre de 2024.}
\end{minipage}\end{center}

Al suscribir, el país pagó el $25\%$ de su cuota en activos de reserva y el $75\%$ en moneda propia, de modo que las tenencias del Fondo en su moneda son el $75\%$ de la cuota. La diferencia entre la cuota y esas tenencias es su \textbf{posición en el tramo de reserva}: un derecho a obtener divisas \textbf{sin condiciones, sin cargos y sin obligación de devolución}, que por eso computa como reservas internacionales del país y no como crédito del Fondo. Es una precisión que se pasa por alto con frecuencia y que tiene una consecuencia práctica importante: pagar la cuota no cuesta reservas, porque lo que se entrega reaparece como posición disponible.

Cuando el país agota ese tramo, las tenencias del Fondo alcanzan el $100\%$ de su cuota y todo lo que compre por encima es \textbf{crédito vivo}: exige un programa, devenga cargos y debe recomprarse en los plazos pactados. Por eso el crédito del Fondo a un país se mide siempre como el exceso de sus tenencias sobre la cuota, y por eso todos los límites de acceso se expresan en porcentaje de esta.

\subsubsection{El modelo de ingresos: ¿quién paga el Fondo?}\label{anx:institucion:ingresos}

El Fondo se financia a sí mismo con el margen de su intermediación, igual que un banco, aunque con una diferencia decisiva: sus deudores son pocos y muy grandes.

El deudor paga un \textbf{cargo básico} igual al tipo de interés del DEG más un \textbf{margen}, que el Directorio fija cada dos años para cubrir los costes de intermediación y acumular reservas, y que desde el 1 de noviembre de 2024 es de $60$ puntos básicos, frente a los $100$ anteriores. Paga además una comisión por cada desembolso y una comisión de compromiso sobre los importes no dispuestos, que se devuelve si se disponen. Y, si su uso es grande o prolongado, paga \textbf{recargos}: $200$ puntos básicos sobre la parte del crédito que excede del $300\%$ de su cuota, y $75$ puntos básicos adicionales a partir de los $36$ meses de uso, o de los $51$ en los acuerdos ampliados. Del otro lado, el Fondo \textbf{remunera} al tipo del DEG las posiciones acreedoras de los miembros y paga intereses por los recursos prestados.

La reforma de octubre de 2024 rebajó el coste conjunto del margen y los recargos en torno a un $36\%$ y redujo de veinte a trece el número de países que pagan recargos, al elevar el umbral desde el $187{,}5\%$ hasta el $300\%$ de la cuota y recortar el recargo por tiempo. Fue el resultado de una discusión larga y revela bien la tensión interna de la institución. Los recargos tienen una justificación defendible: desincentivan el uso excesivo y prolongado de un dinero que es de todos y financian el colchón que protege a los acreedores. Pero tienen un efecto perverso evidente, y es que \textbf{cobran más caro justo a los países que peor están}, en el momento en que menos pueden pagarlo, de modo que funcionan de manera procíclica; y son una fuente de ingresos que depende de que haya países en dificultades graves, lo que no es una base sana para financiar una institución.\footnote{%
    \textbf{NOTA.-} Organizaciones como el Center for Economic and Policy Research han sostenido que los recargos deberían suprimirse, y estiman que, incluso después de la reforma, el Fondo ingresará unos $5{,}2$ mil millones de dólares por este concepto entre 2025 y 2030.\textcolor{orange}{\textsuperscript{[Probable]}} El argumento contrario, que es el de la mayoría del Directorio, es que los saldos precautorios protegen a todos los miembros frente a un impago y que su coste debe recaer sobre quien genera el riesgo.
}

\subsubsection{¿Cuánto presta y a quién?}\label{anx:institucion:exposicion}

Conviene contrastar la capacidad con el uso. Frente a una capacidad cercana al billón de dólares, el \textbf{crédito vivo} del Fondo era de unos $121$ mil millones de DEG a 31 de marzo de 2026. Es decir, la institución emplea una fracción pequeña de su potencia de fuego, lo que admite dos lecturas opuestas que el cuerpo del tema retoma: o bien está sobredimensionada, o bien su función es precisamente estar disponible para el día en que haga falta.

Más revelador es el reparto. \textbf{Argentina concentra por sí sola en torno a un tercio de todo el crédito vivo}, unos $41{,}8$ mil millones de DEG a comienzos de 2026, seguida a gran distancia por Ucrania, Egipto, Pakistán y Ecuador.\textcolor{orange}{\textsuperscript{[Probable]}} Una concentración así tiene tres implicaciones. La primera es de riesgo: la solidez financiera del Fondo depende de un puñado de deudores, y de uno en particular. La segunda es de ingresos: los recargos los pagan esos mismos países, de modo que el presupuesto de la institución se sostiene sobre sus programas más problemáticos. Y la tercera es política: cuando un deudor es tan grande, la decisión sobre su programa deja de ser técnica, porque un impago no solo dañaría al Fondo, sino que destruiría la convención en que se apoya su condición de acreedor preferente.

\subsubsection{El acreedor preferente: la garantía que no está escrita}\label{anx:institucion:preferente}

Esa condición merece un apartado propio, porque es a la vez el activo más valioso del Fondo y el más frágil. \textbf{La preferencia del Fondo como acreedor no está en el Convenio, ni en los contratos, ni en norma alguna}: es una convención aceptada por la comunidad oficial y tolerada por los acreedores privados. Se sostiene sobre una lógica sencilla: el Fondo presta cuando ya no presta nadie, y un país que no le pagara quedaría excluido del único mecanismo de financiación disponible en la próxima crisis.

De ahí que el Fondo no reconozca pérdidas ni acepte quitas, y que el incumplimiento se trate como \textbf{atrasos} sometidos a un procedimiento gradual de sanciones. Las únicas condonaciones que ha practicado, en la iniciativa para los Países Pobres Muy Endeudados y en la Iniciativa de Alivio de la Deuda Multilateral, las financiaron los donantes con aportaciones presupuestarias, no el propio Fondo ([\authorfont{Ver Tema 3.B.29}]).

La fragilidad aparece cuando la exposición crece. Si el Fondo presta mucho a un país cuya deuda acaba reestructurándose, su preferencia implica que \textbf{toda la quita recae sobre los acreedores privados}, que la anticipan y la cobran por adelantado en forma de prima de riesgo. Los programas muy grandes, por tanto, no solo comprometen recursos: subordinan a los demás acreedores y pueden encarecer la financiación del país al que pretenden ayudar. Es uno de los argumentos de fondo de la discusión sobre el acceso excepcional, que se examina en el anexo siguiente.

\subsection{El Derecho Especial de Giro: ¿puede el Fondo crear reservas?}\label{anx:institucion:deg}

El DEG es la pieza más incomprendida de la institución y la única que responde a una pregunta distinta de todas las anteriores. Hasta aquí el Fondo ha aparecido como un intermediario que reparte una liquidez que otros crean; con el DEG, en cambio, el sistema se dio en 1969 la capacidad de \textbf{crear un activo de reserva de la nada, por acuerdo colectivo}, que era justamente lo que TRIFFIN había echado en falta ([\authorfont{Ver Tema 3.B.20, Tema 3.B.21}]).

\subsubsection{Qué es y qué no es}\label{anx:institucion:deg_que}

Conviene empezar por las negaciones, porque casi todos los errores sobre el DEG vienen de ahí. \textbf{El DEG no es una moneda}, porque no sirve para pagar: ningún contrato se liquida en DEG y ningún particular puede tenerlos, ya que solo pueden poseerlos el propio Fondo, los bancos centrales de los miembros y un corto número de tenedores oficiales autorizados. \textbf{Tampoco es un crédito contra el Fondo}: el Fondo no entrega nada cuando un país usa sus DEG. Lo que el DEG confiere es un \textbf{derecho a obtener monedas de libre uso de otros miembros}, que están obligados a entregarlas.

La mecánica se entiende mejor viendo que una asignación crea simultáneamente un activo y un pasivo del mismo importe.

\begin{center}\begin{minipage}{\linewidth}\centering
{\small\textbf{La mecánica del DEG: asignación, tenencias y posición neta}}\par\vspace{0.3em}
\includegraphics{anx1_deg.pdf}\par\vspace{0.3em}
{\small\textit{Fuente: }elaboración propia a partir del Convenio Constitutivo, arts. XV a XXV.}
\end{minipage}\end{center}

El país recibe una \textbf{asignación}, que es un pasivo permanente y fijo, y unas \textbf{tenencias} por el mismo importe, que son un activo que puede usar. Sobre la asignación paga el tipo de interés del DEG y sobre las tenencias lo cobra, de modo que, mientras no las toque, su posición neta es nula y la asignación le ha aumentado las reservas \textbf{sin coste alguno}. Si vende DEG a otro miembro para obtener divisas, sus tenencias caen por debajo de su asignación y paga el tipo del DEG sobre la diferencia, sin plazo de devolución ni condiciones. Dicho de otro modo: \textbf{el DEG funciona como una línea de crédito perpetua, incondicional y automática entre bancos centrales}, al tipo de interés más bajo del mercado oficial.

Esa caracterización explica por sí sola la historia política del instrumento. Lo que para un país con dificultades es la financiación ideal, para los principales accionistas del Fondo es exactamente lo que llevan ochenta años negándose a conceder: dinero sin condiciones.

\subsubsection{La valoración y el tipo de interés}\label{anx:institucion:deg_valor}

Desde 1974 el DEG se valora como una cesta de monedas, lo que lo hace más estable que cualquiera de ellas. La composición se revisa cada cinco años y la vigente, en vigor desde el 1 de agosto de 2022, es de cinco monedas con estos pesos: dólar estadounidense, $43{,}38\%$; euro, $29{,}31\%$; renminbi, $12{,}28\%$; yen, $7{,}59\%$; y libra esterlina, $7{,}44\%$. La próxima revisión debe concluir antes del final de julio de 2027.

Los criterios para entrar en la cesta son dos: que el país emisor sea de los mayores exportadores y que la moneda sea \textbf{de libre uso}, es decir, ampliamente utilizada en pagos internacionales y ampliamente negociada en los mercados de divisas. Conviene distinguir ese concepto del de convertibilidad plena: una moneda puede ser de libre uso manteniendo controles de capitales, y ese fue precisamente el punto discutido en 2015, cuando el renminbi entró en la cesta. Su incorporación tuvo un componente técnico, el peso comercial de China, y otro claramente político, el de acercar a Pekín a la institución; la revisión de 2022 elevó además su peso.

El \textbf{tipo de interés del DEG} es la media ponderada de los tipos de instrumentos del mercado monetario a tres meses de esas cinco monedas, con un suelo que impide que sea negativo.\textcolor{orange}{\textsuperscript{[Probable]}} No es un tipo arbitrario: es el coste de oportunidad medio de las reservas mundiales, y por eso sirve de referencia para remunerar a los acreedores del Fondo y para cobrar a sus deudores.

\subsubsection{Las asignaciones: cuándo y cuánto}\label{anx:institucion:deg_asignacion}

Crear DEG no es fácil, y ahí está la segunda clave política. Una \textbf{asignación general} exige que el Directorio constate una \textbf{necesidad mundial de complementar los activos de reserva existentes a largo plazo} y que la Junta de Gobernadores la apruebe por el $85\%$ de los votos, lo que devuelve la decisión al veto estadounidense. Se reparte en proporción a las cuotas.

Ha habido cuatro asignaciones generales y una especial, por un total acumulado de $660{,}7$ mil millones de DEG. Las dos primeras fueron modestas y propias de su época, en 1970-1972 y en 1979-1981. Las dos grandes son de crisis: $161{,}2$ mil millones en agosto de 2009, tras el colapso financiero, y \textbf{$456{,}5$ mil millones en agosto de 2021}, equivalentes a unos $650$ mil millones de dólares, la mayor de la historia, acordada para dar reservas a los países golpeados por la pandemia. La especial, de $21{,}5$ mil millones, se hizo en septiembre de 2009 para corregir la situación de los países que habían ingresado después de los repartos anteriores, y requirió la cuarta enmienda del Convenio, que tardó doce años en entrar en vigor.

El patrón es claro: \textbf{el sistema solo acepta crear reservas cuando la alternativa es peor}. Fuera de una crisis sistémica, la propuesta de una asignación tropieza siempre con el mismo argumento, el de que no existe una necesidad mundial de largo plazo, y con la misma sospecha, la de que se trata de ayuda encubierta.

\subsubsection{Por qué no llegó a ser el centro del sistema}\label{anx:institucion:deg_limites}

El DEG se creó para resolver el dilema de TRIFFIN, esto es, para que el sistema dejara de depender de que el país emisor de la moneda de reserva incurriera en déficits. Más de cincuenta años después representa una fracción muy pequeña de las reservas mundiales. Las razones son cinco y conviene tenerlas ordenadas, porque son las mismas que explican por qué ninguna propuesta de reforma monetaria internacional prospera.

La primera es el \textbf{reparto}. Al asignarse en proporción a las cuotas, el DEG entrega la mayor parte a quien menos lo necesita: de la asignación de 2021, la mayoría fue a parar a economías avanzadas que no tenían ningún problema de reservas, y solo una parte pequeña a los países de renta baja.\textcolor{orange}{\textsuperscript{[Probable]}} Un activo pensado para dar liquidez a quien no la tiene se reparte según el criterio contrario.

La segunda es que \textbf{no circula}. Sin tenedores privados no hay mercado, y sin mercado no se puede facturar, invertir ni endeudarse en DEG. Un activo de reserva lo es, en última instancia, porque se puede usar en el mercado donde hay que intervenir, y el DEG hay que cambiarlo antes por dólares.

La tercera es que \textbf{su creación está vetada}: exige el $85\%$, de modo que el sistema no puede emitir reservas cuando las necesita, sino cuando Estados Unidos lo consiente.

La cuarta es la \textbf{preferencia de los acreedores por la condicionalidad}. Un DEG es financiación incondicional, y los grandes accionistas han querido siempre que el dinero del Fondo llegue acompañado de un programa. Potenciar el DEG equivaldría a renunciar al instrumento con el que ejercen su influencia.

Y la quinta es que \textbf{no resuelve la asimetría del ajuste}. Aquí la comparación con el \textit{bancor} de KEYNES es obligada y la desarrolla el cuerpo del tema: en el plan de 1943 la unidad de cuenta se creaba dentro de una cámara de compensación que penalizaba tanto al deudor como al acreedor persistente, de manera que el sistema empujaba a corregir los dos lados del desequilibrio. El DEG es solo un activo que se reparte; no lleva aparejado mecanismo alguno que obligue al país con superávit a hacer nada ([\authorfont{Ver Tema 3.B.12}]).

\subsubsection{La recanalización: un uso nuevo para un activo viejo}\label{anx:institucion:deg_recanalizacion}

La asignación de 2021 dejó planteado un problema evidente: cientos de miles de millones de DEG ociosos en los bancos centrales de países que no los necesitaban, junto a países que los necesitaban y habían recibido poco. De ahí nació la \textbf{recanalización}, que consiste en que un país preste sus DEG, o su contravalor, a un vehículo que los haga llegar a quien los precisa.

Las dos vías principales son los propios fondos fiduciarios del Fondo, el PRGT y el RST, que se nutren así de préstamos de miembros a tipo de mercado y de contribuciones que subvencionan el tipo final. Una tercera vía, más discutida, es la aportación de DEG como \textbf{capital híbrido de los bancos multilaterales de desarrollo}, que permite apalancarlos y multiplicar la financiación disponible.

El límite de todas ellas es técnico y no político, y conviene entenderlo porque es lo que acota el debate. Para el país que los presta, los DEG tienen que seguir siendo \textbf{un activo de reserva}: disponibles de inmediato y sin riesgo apreciable de crédito. Si se prestaran a un vehículo que los inmovilizara o los expusiera a pérdidas, dejarían de computar como reservas y el banco central del país prestamista, que necesita autorización legal para disponer de ellas, no podría participar. Todo el diseño de la recanalización consiste, por tanto, en encontrar estructuras que trasladen el dinero sin que pierda esa condición, y esa es la razón de que los importes efectivamente movilizados hayan sido muy inferiores a los anunciados.

\clearpage
\anexo{El instrumental: facilidades, condicionalidad y deuda\label{anx:instrumental}}

El cuerpo del tema sostiene que el Fondo suministra el tercer margen del ajuste y le pone precio. Este anexo desarrolla ese precio en todos sus términos. Primero, el catálogo completo de facilidades: qué puede pedir un país según su renta y según la naturaleza de su problema, cuánto, a qué plazo y con qué coste. Después, la condicionalidad: de qué está hecha, cómo se reparte a lo largo de la vida de un programa, cuánta ha habido en cada época y qué criterio decide hoy lo que es asunto del Fondo y lo que no. Por último, las dos cuestiones que deciden si un programa es siquiera posible: qué ocurre cuando la deuda del país no es sostenible, y quién ejecuta materialmente las medidas que el programa exige.

Nada de lo que sigue es accesorio, pero casi nada cabe en el cuerpo: un tribunal espera que el opositor sepa qué es un acuerdo de derecho de giro y en qué se diferencia de un acuerdo ampliado, y espera también que no convierta la exposición en una lista de siglas. El reparto es, por tanto, el de siempre: en el cuerpo, la lógica que ordena el instrumental; aquí, el instrumental entero.

\subsection{Las facilidades: ¿qué puede pedir un país y en qué condiciones?}\label{anx:instrumental:facilidades}

El catálogo del Fondo parece inabarcable y en realidad obedece a un orden sencillo, porque todas las facilidades son combinaciones de las respuestas a cuatro preguntas: \textbf{quién} puede pedirla, que depende de la renta del país; \textbf{para qué}, que depende de la naturaleza del desequilibrio; \textbf{cuánto y a qué plazo}, que se mide siempre en porcentaje de la cuota; y \textbf{con qué condiciones}, que es el precio en política económica.

\subsubsection{Cómo se ordena el catálogo}\label{anx:instrumental:mapa}

Las dos primeras preguntas son independientes entre sí y conviene no confundirlas. La renta decide la \textbf{ventanilla}: la Cuenta de Recursos Generales está abierta a todos los miembros y presta a tipos de mercado; el Fondo Fiduciario para la Reducción de la Pobreza y el Crecimiento presta en condiciones concesionales a los países de renta baja que reúnen los requisitos; y los países en la frontera entre ambos grupos reciben una \textbf{mezcla} de las dos. La naturaleza del problema decide la \textbf{facilidad}, y por eso cada instrumento de la Cuenta tiene su gemelo concesional, con el mismo diseño y distinto precio.

\subsubsection{Los acuerdos ordinarios: derecho de giro y servicio ampliado}\label{anx:instrumental:ordinarios}

El \textbf{Acuerdo de Derecho de Giro} (\textit{Stand-By Arrangement}, SBA) es el instrumento clásico, existente desde 1952, y sigue siendo el patrón con el que se comparan los demás. Se dirige a un país con un desequilibrio que puede corregirse en un plazo corto o medio; dura entre uno y tres años y se devuelve entre los tres años y cuarto y los cinco desde cada desembolso. Su nombre describe bien lo que es: el Fondo no entrega el dinero, sino que \textbf{se compromete a estar disponible} durante un periodo y hasta un importe, contra el cumplimiento de un programa.

El \textbf{Servicio Ampliado} (\textit{Extended Fund Facility}, EFF) nació en 1974, cuando se comprobó que muchos desequilibrios no eran coyunturales sino estructurales y no se corregían en dos años. Dura tres años, prorrogables a cuatro, y se devuelve entre los cuatro años y medio y los diez. A cambio del plazo más largo exige una condicionalidad estructural más intensa: es el instrumento con el que el Fondo entra en el diseño de reformas de fondo.

La diferencia entre ambos, que parece técnica, encierra un juicio sobre la naturaleza del problema del país, y equivocarlo tiene consecuencias. Tratar como coyuntural lo que es estructural produce programas que se renuevan una y otra vez sin resolver nada, el fenómeno del \textbf{uso prolongado} de los recursos del Fondo que su propia Oficina de Evaluación Independiente ha criticado en repetidas ocasiones.

\subsubsection{Las líneas precautorias: asegurar en lugar de rescatar}\label{anx:instrumental:precautorias}

Las líneas precautorias son la innovación conceptual más importante del Fondo en las últimas décadas y la que mejor encarna la idea de prevención. Invierten la lógica del crédito tradicional: en lugar de condicionar el dinero a lo que el país \textbf{hará} (condicionalidad posterior), lo condicionan a lo que el país \textbf{ha hecho} (admisión previa). Quien reúne los requisitos obtiene un derecho de giro que puede usar sin más trámite, y lo normal es que no lo use nunca.

La \textbf{Línea de Crédito Flexible} (FCL), creada en marzo de 2009, es la principal. Se concede por uno o dos años a países con fundamentos económicos, marco de políticas y antecedentes de ejecución muy sólidos, sin condicionalidad posterior de ningún tipo y con acceso elevado. México la ha renovado de forma ininterrumpida desde 2009 y en noviembre de 2025 firmó su undécimo acuerdo, por unos $24$ mil millones de dólares, reducido respecto de los anteriores por la mejora de sus propios colchones. Colombia la mantiene desde la misma fecha, y la han utilizado además Polonia, Perú, Chile, Marruecos y Costa Rica.\textcolor{orange}{\textsuperscript{[Probable]}}

La \textbf{Línea de Liquidez a Corto Plazo} (SLL), de 2020, es una versión menor y renovable, pensada para presiones de liquidez de corto plazo originadas en la cuenta financiera, con los mismos requisitos de admisión que la FCL, menor acceso y mayor automatismo. La \textbf{Línea de Precaución y Liquidez} (PLL) ocupa el escalón intermedio: se dirige a países con políticas sólidas pero con vulnerabilidades que impiden admitirlos en la FCL, y por eso conserva una condicionalidad ligera. Marruecos fue su usuario más significativo antes de pasar a la FCL.

El problema de las líneas precautorias no es su diseño sino su demanda, y conviene enunciarlo con claridad porque reaparece en el cuerpo del tema. \textbf{Apenas se usan}: un puñado de países en más de quince años, y ninguno con saldo dispuesto. La razón es el \textbf{estigma}: solicitar una línea del Fondo, aunque sea para no usarla, se interpreta en los mercados como el reconocimiento de una debilidad, de modo que el seguro resulta más caro que el riesgo. A ello se suma que, para que la línea sea creíble, el Fondo tiene que declarar que el país cumple unos requisitos exigentes, y retirar después esa declaración sería en sí mismo un acontecimiento desestabilizador. El resultado es que los países prefieren \textbf{autoasegurarse} acumulando reservas, que es mucho más caro para ellos y para el sistema, pero no manda ninguna señal.

\subsubsection{La financiación de emergencia}\label{anx:instrumental:emergencia}

Cuando la urgencia impide negociar un programa, o cuando la capacidad institucional del país está rota, el Fondo dispone de \textbf{desembolsos directos}, sin acuerdo, sin tramos y sin revisiones: el \textbf{Instrumento de Financiación Rápida} (RFI) en la Cuenta de Recursos Generales y el \textbf{Servicio de Crédito Rápido} (RCF) en la ventanilla concesional. El acceso es limitado y la condicionalidad se reduce, en su caso, a alguna actuación previa.

Su momento de gloria fue la pandemia: en pocas semanas de 2020 el Fondo atendió a más de setenta países por esta vía, una velocidad imposible con los instrumentos ordinarios. Ese episodio también mostró su límite: dinero rápido sin programa es apropiado para un \textit{shock} exógeno y transitorio, y peligroso para un desequilibrio de fondo, porque financia sin corregir.

\subsubsection{La ventanilla concesional y su problema de financiación}\label{anx:instrumental:prgt}

Los países de renta baja acceden a los mismos instrumentos con otro precio y otros plazos a través del PRGT: el \textbf{Servicio de Crédito Ampliado} (ECF) para problemas prolongados, el \textbf{Servicio de Crédito \textit{Stand-By}} (SCF) para los de corto plazo, y el ya citado RCF para emergencias. Los límites de acceso quedaron fijados desde el 1 de enero de 2025 en el $200\%$ de la cuota al año y el $600\%$ acumulado, después de una subida temporal desde el $145\%$, y el tipo de interés es del \textbf{cero por ciento} para los países del tramo de renta más baja.

Prestar al cero por ciento exige que alguien pague la diferencia, y ahí está la fragilidad estructural de esta ventanilla, ya anticipada al describir el balance ([\authorfont{Ver Anexo \ref{anx:institucion:balance}}]). El PRGT necesita simultáneamente \textbf{préstamos} de miembros, que cobran tipo de mercado, y \textbf{subvenciones} presupuestarias que financien la concesionalidad, y la demanda posterior a la pandemia agotó ambas. La reforma aprobada en octubre de 2024 buscó precisamente poner el fondo sobre una base sostenible combinando recursos internos de la institución con nuevas contribuciones.\textcolor{orange}{\textsuperscript{[Probable]}} El fondo de la cuestión no es técnico: \textbf{la capacidad del Fondo para ayudar a los países más pobres no depende de su tamaño, sino de la disposición presupuestaria de sus miembros ricos}, y esa es una dependencia que la institución no controla.

\subsubsection{El Fondo Fiduciario para la Resiliencia y la Sostenibilidad}\label{anx:instrumental:rst}

Creado en 2022, el RST es la novedad más discutida del instrumental porque rompe dos reglas de la casa. La primera es el plazo: su facilidad presta a veinte años, con más de diez de carencia,\textcolor{orange}{\textsuperscript{[Probable]}} cuando todo lo demás en el Fondo se devuelve en menos de diez. La segunda es el objeto: no financia un desequilibrio actual de balanza de pagos, sino la prevención de \textbf{riesgos de largo plazo} para la estabilidad, señaladamente el cambio climático y la preparación ante pandemias. Está abierto a países de renta baja y también a países de renta media vulnerables, exige un programa ordinario en vigor como condición y su acceso es moderado.

Las objeciones son de dos clases y conviene conocerlas porque resumen el debate sobre el mandato de la institución. La primera es \textbf{competencial}: financiar la transición climática a veinte años es la función de los bancos multilaterales de desarrollo, no de un fondo monetario ([\authorfont{Ver Tema 3.B.30}]), y el Fondo entra en un terreno donde no tiene ventaja comparativa. La segunda es \textbf{financiera}: comprometer recursos a veinte años es difícilmente compatible con el carácter rotatorio y de emergencia que exige el artículo I del Convenio. La defensa del Fondo es la macrocriticidad: si un riesgo afecta previsiblemente a la balanza de pagos de un país, entra en su mandato por mucho que su horizonte sea largo.

\subsubsection{Los instrumentos sin dinero}\label{anx:instrumental:sinfinanciacion}

El Fondo dispone, por último, de instrumentos que no entregan nada y cuyo valor es la \textbf{señal}. El \textbf{Instrumento de Coordinación de Políticas} (PCI), creado en julio de 2017 y abierto a todos los miembros, somete al país a un programa con revisiones del Directorio sin financiación asociada, de modo que el aval del Fondo puede desbloquear la financiación de otros. Sustituyó en la práctica al antiguo Instrumento de Apoyo a las Políticas, que se suprimió en octubre de 2023. Por debajo está el \textbf{programa supervisado por el personal} (SMP), un acuerdo informal con el \textit{staff}, sin intervención del Directorio, que suele emplearse para reenganchar a países con atrasos o con instituciones dañadas.

Estos instrumentos interesan por lo que revelan: \textbf{una parte del valor que el Fondo añade no es su dinero, sino su certificación}. Si eso es así, su influencia no depende solo de sus recursos, lo que matiza el diagnóstico habitual de que un Fondo relativamente más pequeño es necesariamente un Fondo irrelevante.

\subsubsection{Los límites de acceso y el acceso excepcional}\label{anx:instrumental:acceso}

Todo lo anterior está acotado por los \textbf{límites de acceso}, que la revisión general de diciembre de 2024 mantuvo en el $200\%$ de la cuota al año y el $600\%$ acumulado, un $38\%$ por encima de los fijados en 2016. Dos precisiones importan. La primera es que \textbf{no son techos}, sino umbrales que, al superarse, activan un escrutinio reforzado. La segunda es aritmética y revela otra vez la centralidad de la cuota: como los límites se expresan en porcentaje de esta, cuando entre en vigor la 16.ª revisión y las cuotas suban un $50\%$, los porcentajes bajarán en torno al $135\%$ y el $405\%$ para mantener constantes los importes en DEG.\textcolor{orange}{\textsuperscript{[Probable]}} Dicho de otro modo, \textbf{la revisión de cuotas no aumenta lo que un país puede pedir}: solo cambia de dónde sale el dinero.

Por encima de esos umbrales opera el \textbf{marco de acceso excepcional}, que exige cumplir cuatro criterios acumulativos: que las presiones sobre la cuenta corriente o la financiera sean excepcionales; que \textbf{la deuda sea sostenible con alta probabilidad}; que sea razonable esperar la recuperación del acceso a los mercados en el plazo del programa; y que existan perspectivas razonables de éxito, atendiendo a la capacidad institucional y política del país.

El segundo criterio es el decisivo y tiene historia propia. En mayo de 2010, para poder prestar a Grecia pese a no poder afirmar que su deuda fuera sostenible con alta probabilidad, el Directorio introdujo una \textbf{exención sistémica}: bastaba con que el impago amenazara con contagio internacional. La excepción se usó y fue muy criticada, porque convertía la gravedad del riesgo en una razón para prestar más en lugar de para reestructurar antes, y se \textbf{suprimió en enero de 2016}. En su lugar se admitió una vía intermedia para los casos dudosos: cuando la deuda no es claramente insostenible pero tampoco sostenible con alta probabilidad, el Fondo puede prestar si obtiene garantías de financiación de otros acreedores, o una \textbf{reperfilación} de los vencimientos próximos, sin exigir una quita. Esa zona gris, y lo que ocurre cuando se entra en ella, es el objeto del apartado siguiente de este anexo.

\subsection{La condicionalidad: ¿qué pide el Fondo a cambio?}\label{anx:instrumental:condicionalidad}

La condicionalidad es, con la cuota, el otro concepto que hay que dominar para entender la institución, y el que más malentendidos acumula. No es una imposición política añadida por los acreedores: es una exigencia del artículo V del Convenio, que obliga al Fondo a prestar \textit{con las garantías adecuadas}. Y no es tampoco un conjunto de buenos consejos: es el calendario del dinero.

\subsubsection{Dos funciones, no una}\label{anx:instrumental:funciones}

Conviene distinguir dos funciones que suelen confundirse, porque la mayoría de las críticas y de las defensas se dirigen solo a una de ellas.

La primera es la \textbf{salvaguardia de los recursos}. El Fondo presta sin garantías reales a un país insolvente de hecho, y lo único que asegura la devolución es que el país corrija el problema que lo llevó allí. Desde este punto de vista la condicionalidad protege a los demás miembros, cuyos bancos centrales han puesto el dinero.

La segunda es menos evidente y más interesante: la condicionalidad funciona como un \textbf{dispositivo de compromiso} al servicio del propio gobierno. Un gobierno que anuncia un ajuste no resulta creíble, porque siempre podrá abandonarlo; el mismo ajuste inscrito en un programa del Fondo, con revisiones semestrales de las que depende el dinero, sí lo es, porque incumplirlo tiene un coste inmediato y verificable. El país no compra solo financiación: \textbf{compra credibilidad que no puede fabricar por sí mismo}.

Las dos funciones no siempre apuntan en la misma dirección, y de esa tensión nacen los peores programas. Lo que protege el dinero del Fondo es el ajuste rápido y verificable; lo que da credibilidad duradera a un gobierno es la reforma que su sociedad acepta. Cuando la primera lógica se impone sobre la segunda, el programa se cumple sobre el papel y se revierte en cuanto termina.

\subsubsection{De qué está hecha y cómo se reparte en el tiempo}\label{anx:instrumental:tipos}

La condicionalidad de un programa se compone de cinco clases de elementos, que se distinguen por el momento en que operan y por la consecuencia de incumplirlos.

\begin{center}\begin{minipage}{\linewidth}\centering
{\small\textbf{La anatomía de un programa: condicionalidad, revisiones y desembolsos}}\par\vspace{0.3em}
\includegraphics{anx2_programa.pdf}\par\vspace{0.3em}

\end{minipage}\end{center}

Las \textbf{acciones previas} son medidas que el país debe haber ejecutado \textbf{antes} de que el Directorio apruebe el acuerdo: una devaluación, la aprobación de un presupuesto, la subida de un precio administrado. Son la forma más dura de condicionalidad porque no admiten promesa, y se emplean cuando la credibilidad del gobierno es escasa o cuando la medida es indispensable para que el programa tenga sentido.

Los \textbf{criterios de ejecución cuantitativos} son los que gobiernan los desembolsos. Se miden en fechas de corte, normalmente semestrales, y son siempre variables que las autoridades controlan: un suelo de reservas internacionales netas, un techo de activos internos netos del banco central, un techo de déficit o de gasto primario y un límite al nuevo endeudamiento exterior no concesional. No es casual que sean justamente las variables del modelo de programación financiera ([\authorfont{Ver Anexo \ref{anx:instrumental}}], y el desarrollo completo en el anexo del programa): \textbf{la condicionalidad cuantitativa es el modelo convertido en contrato}.

Las \textbf{metas indicativas} miden lo mismo pero no bloquean el desembolso; suelen usarse para variables de calidad incierta o para periodos aún lejanos. Los \textbf{parámetros estructurales de referencia} son reformas concretas con fecha, como aprobar una ley de insolvencia o publicar una auditoría, y su incumplimiento no detiene automáticamente el dinero, aunque pesa en la valoración de la revisión. Y las \textbf{cláusulas de consulta} obligan al país a discutir con el Fondo cuando una variable, típicamente la inflación, se desvía de una banda, sin consecuencia automática.

El mecanismo se completa con las \textbf{revisiones}. Si en la fecha de corte los criterios se han cumplido, la revisión se completa y se libera el tramo. Si hay una desviación menor, o ya corregida, el Directorio puede conceder una \textbf{dispensa}. Si no, la revisión no se completa, el desembolso no se produce y el programa \textbf{descarrila}, situación mucho más frecuente de lo que sugiere el discurso oficial. De aquí sale la formulación más exacta de todo este apartado: \textbf{la condicionalidad no es una promesa, es el calendario del desembolso}, y ese escalonamiento es todo el poder coercitivo que tiene el Fondo.

\subsubsection{Cuánta condicionalidad: una historia de excesos y correcciones}\label{anx:instrumental:volumen}

El número de condiciones de un programa no es un dato técnico, sino el mejor indicador de lo que el Fondo ha creído en cada época que estaba a su alcance.

En los años ochenta y noventa, con los países en desarrollo y las economías en transición demandando reformas de fondo, la condicionalidad estructural creció sin medida. Los programas de la crisis asiática de 1997 son el caso extremo: el acuerdo con Indonesia llegó a incluir más de un centenar de condiciones estructurales, algunas tan alejadas del mandato como el régimen del mercado del clavo.\textcolor{orange}{\textsuperscript{[Probable]}} La justificación era doble, y parte de ella sigue siendo válida: las reformas de oferta permiten un ajuste de demanda menor, de modo que más reforma estructural puede significar menos recesión. Pero el resultado fue un intervencionismo detallado que el Fondo no estaba en condiciones de diseñar ni de vigilar, y que erosionó gravemente su legitimidad.

La corrección llegó con las \textbf{Directrices sobre Condicionalidad de 2002}, que introdujeron el criterio que sigue vigente: solo es admisible la condición \textbf{macrocrítica}, esto es, la que resulta necesaria para alcanzar los objetivos macroeconómicos del programa o para salvaguardar los recursos del Fondo. El volumen cayó y volvió a subir después de 2009; la revisión de 2018-2019 constató que seguía siendo elevado en muchos programas, y que el exceso de condiciones estructurales está asociado a peores resultados, probablemente porque dispersa el esfuerzo político en vez de concentrarlo.\textcolor{orange}{\textsuperscript{[Probable]}}

\subsubsection{La macrocriticidad: el criterio que decide qué es asunto del Fondo}\label{anx:instrumental:macrocritica}

Ese criterio merece atención propia porque es la bisagra entre la condicionalidad y el debate sobre el mandato. Macrocrítico es lo que afecta de manera apreciable a la estabilidad macroeconómica o a la balanza de pagos; y, como casi cualquier cosa puede afectarlas por alguna vía, el criterio se ha ido extendiendo desde el presupuesto y el tipo de cambio hacia el sector financiero, el gobierno corporativo, la corrupción, la desigualdad, el clima y la igualdad de género.

La ampliación tiene una justificación analítica sólida en cada caso concreto, y una objeción agregada muy difícil de rebatir: \textbf{un criterio que puede justificar cualquier cosa no delimita nada}. Si todo es macrocrítico, el Fondo acaba opinando sobre todo, con un personal formado para otra cosa, en países que no le han dado ese mandato, y con el coste reputacional de dispersarse mientras sus funciones centrales se cuestionan. Es el argumento del Fondo concentrado frente al Fondo amplio, que el tercer bloque del cuerpo retoma.

\subsubsection{Las salvaguardias: lo que el Fondo exige para proteger su dinero}\label{anx:instrumental:salvaguardias}

Junto a la condicionalidad, y a menudo confundidas con ella, operan tres exigencias técnicas que condicionan igualmente la aprobación de un programa.

La primera es la \textbf{capacidad de repago}: el Directorio debe convencerse de que el país podrá devolver, lo que en la práctica vincula el tamaño del programa a la sostenibilidad de su deuda. La segunda son las \textbf{garantías de financiación}: el programa debe estar \textbf{íntegramente financiado} al menos durante los doce meses siguientes, lo que obliga al Fondo a verificar que otros acreedores, oficiales o privados, aportan lo que se espera de ellos; esta exigencia convierte al Fondo, quiera o no, en coordinador de los demás. La tercera son las \textbf{evaluaciones de salvaguardias} de los bancos centrales, introducidas en 2000 tras varios episodios graves de información falseada, que examinan la auditoría, los controles internos y la publicación de cuentas de la autoridad monetaria que va a recibir y manejar los fondos.\textcolor{orange}{\textsuperscript{[Probable]}}

A ellas se añade la \textbf{política sobre información errónea}, que permite al Fondo exigir la devolución anticipada de lo desembolsado cuando el país ha obtenido el dinero sobre la base de datos falsos. No es un supuesto teórico: la verificación de las reservas ucranianas en 1996-1998 reveló que menos de la mitad de las declaradas eran utilizables.

\subsubsection{La propiedad local y el chivo expiatorio}\label{anx:instrumental:ownership}

Toda la discusión anterior desemboca en el único factor que la evidencia señala de forma consistente como determinante del éxito o del fracaso: la \textbf{propiedad local} del programa (\textit{ownership}), es decir, que el gobierno y la administración consideren suyas las medidas en lugar de padecerlas.

La razón es mecánica y no moral. El Fondo puede verificar el cumplimiento de un techo de déficit, pero no puede ejecutar una reforma tributaria, ni mantener una política cuando el programa termina, ni impedir que el siguiente gobierno la revierta. Todo lo que el programa logra depende de una administración que lo aplique y de una coalición política que lo sostenga. El programa español de 1959 es el ejemplo clásico de propiedad local extrema: tanto, que la historiografía suele presentar el Plan de Estabilización como una iniciativa enteramente interna y omitir el papel del Fondo, cuando fue un programa típico de la institución.

De ahí la ambivalencia del uso del Fondo como \textbf{chivo expiatorio}. A los gobiernos les resulta cómodo presentar como impuestas desde fuera medidas que consideran necesarias, y a los acreedores les conviene que la condicionalidad la fije un tercero técnico en lugar de ellos mismos, porque evita el desgaste bilateral. El Fondo lo tolera porque facilita la aprobación de los programas. Pero el precio es doble: \textbf{erosiona su reputación}, ya que carga con la impopularidad de decisiones que no siempre son suyas, y \textbf{destruye precisamente la propiedad local} de la que depende el éxito. Un gobierno que explica que el ajuste se lo impone el Fondo está anunciando que lo abandonará en cuanto pueda.

\subsection{La deuda: ¿y si el país no puede pagar?}\label{anx:instrumental:deuda}

Todo lo anterior presupone que el país puede devolver lo que recibe. Cuando no puede, el programa deja de ser un problema de diseño y se convierte en otro muy distinto: hay que decidir quién asume la pérdida antes de que el Fondo ponga un euro. Es la decisión más difícil de la institución, y la que mejor muestra que no es un actor técnico y neutral.

\subsubsection{La decisión que precede a todas las demás: las tres zonas}\label{anx:instrumental:zonas}

El Fondo valora la sostenibilidad con dos marcos distintos según quién sea el país. Para los que se financian en los mercados utiliza desde 2021 el \textbf{Marco de Riesgo Soberano y Sostenibilidad de la Deuda} (SRDSF), que sustituyó al antiguo análisis para países con acceso a mercados y evalúa el riesgo en tres horizontes, de corto, medio y largo plazo. Para los países de renta baja con acceso limitado a los mercados rige el \textbf{Marco de Sostenibilidad de la Deuda} conjunto con el Banco Mundial, construido sobre umbrales que dependen de la capacidad institucional del país y que atiende al valor actual del servicio de la deuda, porque buena parte de ella es concesional. La mecánica de ambos se desarrolla en el anexo siguiente; lo que importa aquí es lo que el dictamen obliga a hacer.

Conviene subrayar dos cosas que la figura resume. La primera es que \textbf{el dictamen no describe un hecho, sino una proyección}: la sostenibilidad depende de supuestos sobre crecimiento, tipos de interés y saldo primario a cinco o diez años, que elabora el propio Fondo. La segunda, que de ahí nace un sesgo bien documentado hacia el optimismo, porque declarar insostenible una deuda obliga a una reestructuración que nadie quiere firmar: ni el gobierno, que teme el estigma; ni los acreedores, que prefieren cobrar; ni el Fondo, que tendría que admitir que su préstamo anterior fue un error.\textcolor{orange}{\textsuperscript{[Probable]}}

\subsubsection{Quién tiene que estar de acuerdo}\label{anx:instrumental:acreedores}

Si hay que reestructurar, aparece el problema que ninguna institución internacional ha resuelto: \textbf{no existe un procedimiento concursal para los Estados}. No hay tribunal que imponga un convenio a los acreedores disidentes ni autoridad que suspenda los pagos. Lo que hay es un conjunto de foros, convenciones y cláusulas contractuales que funcionan cuando todos quieren que funcionen.

\begin{center}\begin{minipage}{\linewidth}\centering
{\small\textbf{La arquitectura de una reestructuración: tres clases de acreedor y las reglas que los enlazan}}\par\vspace{0.3em}
\includegraphics{anx2_acreedores.pdf}\par\vspace{0.3em}
{\small\textit{Fuente: }elaboración propia a partir de la práctica del Club de París, del Marco Común del G20 y de la política de garantías de financiación del FMI.}
\end{minipage}\end{center}

Dos reglas enlazan los tres grupos. La \textbf{comparabilidad de trato} obliga al país que ha pactado con los acreedores oficiales a pedir un sacrificio equivalente a los privados, y es la pieza que impide que el dinero oficial sirva para pagar a los bonistas. Las \textbf{garantías de financiación} obligan al Fondo a no desembolsar hasta que el reparto está cerrado, y convierten a la institución, quiera o no, en el coordinador de una negociación que no preside.

Falta una pieza que el Fondo intentó construir y no consiguió, y que conviene conocer porque es la gran discusión doctrinal sobre esta materia. Entre 2001 y 2003, la entonces primera subdirectora gerente, Anne KRUEGER, propuso un \textbf{mecanismo de reestructuración de deuda soberana} (SDRM): un procedimiento estatutario, inscrito en el Convenio, que habría permitido imponer un acuerdo a los acreedores disidentes por mayoría cualificada, con suspensión de ejecuciones, igual que un concurso de acreedores. La propuesta fue rechazada por Estados Unidos y por los mercados, que temían encarecer la deuda emergente y entregar al Fondo un poder cuasi judicial. En su lugar prevaleció la \textbf{vía contractual}: generalizar en los bonos soberanos las \textbf{cláusulas de acción colectiva}, reforzadas en 2014 con agregación en una sola votación de todas las series, de modo que una minoría no pueda bloquear el acuerdo. La vía contractual funciona, pero solo para los bonos nuevos: el problema del acreedor que litiga sigue vivo mientras quede deuda antigua, como mostraron los pleitos de Argentina en los tribunales de Nueva York.

\subsubsection{El Marco Común del G20 y el problema chino}\label{anx:instrumental:marcocomun}

La arquitectura anterior se diseñó para un mundo en el que los acreedores oficiales eran un club de democracias ricas que se ponían de acuerdo en París. Ese mundo ya no existe: \textbf{el mayor acreedor bilateral de los países de renta baja es hoy China}, que no pertenece al Club de París, prefiere negociar caso por caso y sostiene que algunos de sus préstamos son comerciales y no oficiales.\textcolor{orange}{\textsuperscript{[Probable]}}

La respuesta fue la \textbf{Iniciativa de Suspensión del Servicio de la Deuda} de 2020, que aplazó pagos durante la pandemia, y, sobre todo, el \textbf{Marco Común para el Tratamiento de la Deuda} acordado por el G20 en noviembre de 2020, cuya novedad es sentar en un mismo Comité de Acreedores Oficiales a los miembros del Club de París y a los que no lo son. Su balance, cinco años después, es el que cabía esperar de un mecanismo sin obligatoriedad: cuatro países lo han solicitado, Chad, Zambia, Ghana y Etiopía; Zambia y Ghana han cerrado su tratamiento; Chad lo cerró en 2022 sin reducción nominal alguna de la deuda; y Etiopía quedó bloqueada años por un litigio con sus tenedores de bonos.\textcolor{orange}{\textsuperscript{[Probable]}}

Las tres causas del atasco se repiten en todos los casos y merecen retenerse. La primera es la \textbf{lentitud}, que destruye valor mientras se negocia. La segunda es la \textbf{indefinición de quién es acreedor oficial}, que estalló cuando instituciones como el Afreximbank reclamaron la preferencia de los multilaterales en Ghana y Zambia. Y la tercera es que \textbf{el acreedor privado no está obligado}: la comparabilidad de trato es una exigencia que el país debe perseguir, no un poder que alguien pueda ejercer.

Para el tema importa, además, una consecuencia que el cuerpo recoge. La condición de acreedor preferente del Fondo, que nunca estuvo escrita, empieza a discutirse desde el lado de los acreedores emergentes, que no ven por qué han de asumir quitas de las que los organismos multilaterales quedan exentos. Si esa convención se rompiera, el Fondo tendría que prestar como cualquier otro acreedor, con garantías y provisiones, y dejaría de poder hacer lo único que justifica su existencia: prestar cuando ya nadie presta.

\subsubsection{Los atrasos: prestar a quien ya no paga}\label{anx:instrumental:atrasos}

Queda un supuesto extremo. Si un país deja de pagar a alguno de sus acreedores, ¿puede el Fondo seguir prestándole? Durante décadas la respuesta fue negativa, lo que entregaba a cualquier acreedor un derecho de veto sobre el programa: le bastaba con negarse a negociar para que el Fondo no pudiera desembolsar.

La política de \textbf{préstamo con atrasos} corrigió eso en dos tiempos. Desde 1989, refinada en 1999, el Fondo puede prestar a un país con atrasos frente a \textbf{acreedores privados} si el país negocia de buena fe y el programa lo requiere. Y desde diciembre de 2015 puede hacerlo también frente a \textbf{acreedores bilaterales oficiales}, cuando el acreedor se niega a un tratamiento razonable y el resto de la comunidad oficial respalda la operación; la reforma se aprobó días antes del impago ucraniano de un bono de tres mil millones de dólares en poder de Rusia, y no es casualidad.\textcolor{orange}{\textsuperscript{[Probable]}}

El principio que late debajo es el mismo que ordena todo este anexo: \textbf{el Fondo no puede dejar que el interés de un acreedor bloquee el ajuste de un país}, porque su función no es cobrar deudas ajenas, sino evitar que un impago desordenado se propague.

\subsubsection{Por qué las reestructuraciones llegan tarde}\label{anx:instrumental:tarde}

Todos los estudios sobre episodios de deuda soberana coinciden en un punto: las reestructuraciones se hacen \textbf{tarde y por poco}, y la demora resulta cara. Se destruye producto mientras la incertidumbre dura, se agotan las reservas, el ajuste se hace mayor y la quita final acaba siendo más profunda que la que habría bastado al principio.

Los incentivos explican por qué ocurre siempre. El gobierno aplaza porque la reestructuración es un acontecimiento político de primer orden y porque siempre cabe esperar un golpe de suerte. Los acreedores aplazan porque cada mes que pasa sin quita es un mes en el que cobran, y porque el dinero oficial que entra les permite salir sin pérdidas. Y el Fondo aplaza porque declarar insostenible una deuda equivale a reconocer que su propio análisis anterior fue demasiado benévolo.

El caso canónico es Grecia. El programa de mayo de 2010 se aprobó sin reestructuración, al amparo de la exención sistémica creada para la ocasión, y la reestructuración llegó en marzo de 2012, casi dos años después. En ese intervalo una parte sustancial de la deuda privada venció y se cobró íntegra, de modo que el ajuste posterior recayó sobre quienes se habían quedado. La evaluación posterior del propio Fondo reconoció que la demora benefició a los acreedores privados y agravó la contracción.\textcolor{orange}{\textsuperscript{[Probable]}} De ahí la supresión de la exención en 2016 y la regla que hoy rige: \textbf{si la deuda no es sostenible, la quita va antes que el préstamo}.

\subsection{El desarrollo de capacidades: ¿quién ejecuta el programa?}\label{anx:instrumental:capacidades}

El último elemento del instrumental no es dinero ni condicionalidad, y suele tratarse como un apéndice benévolo del Fondo. Es un error de perspectiva: sin él, nada de lo anterior se ejecuta.

\subsubsection{Qué es y cuánto pesa}\label{anx:instrumental:cd_que}

El desarrollo de capacidades consiste en asesorar técnicamente a las administraciones de los países miembros y formar a sus funcionarios, a petición del país. Pesa alrededor de \textbf{un tercio del presupuesto administrativo} del Fondo, unos cuatrocientos millones de dólares al año, y más de la mitad lo financian socios externos, principalmente donantes bilaterales y la Unión Europea.\textcolor{orange}{\textsuperscript{[Probable]}}

Se presta por tres canales. El primero son las \textbf{misiones de asistencia técnica} y los asesores residentes, que trabajan sobre un problema concreto: diseñar una reforma tributaria, reconstruir la gestión de la deuda pública, montar un marco de política monetaria. El segundo es la red de \textbf{centros regionales}, diecisiete en 2024, que dan asistencia continuada y cercana a los países de su área. El tercero es la \textbf{formación} de funcionarios, presencial y en línea.

Las materias son las que cabe esperar de la institución: administración y política tributarias, gestión de las finanzas públicas, política monetaria y cambiaria, supervisión financiera, marcos jurídicos y, muy destacadamente, \textbf{estadística}, donde el Fondo no solo asesora sino que fija los estándares internacionales con sus manuales de balanza de pagos, de finanzas públicas y de estadísticas monetarias.

\subsubsection{Por qué es la condición de posibilidad de todo lo demás}\label{anx:instrumental:cd_porque}

El argumento analítico es sencillo y rara vez se formula. \textbf{Un programa es un contrato cuya ejecución depende de una administración}. La condicionalidad puede fijar un techo de déficit, pero no puede construir una agencia tributaria capaz de recaudar lo que ese techo supone; puede exigir un objetivo de reservas, pero no puede dotar al banco central de la contabilidad que permita medirlas con fiabilidad.

De ahí que el desarrollo de capacidades y la condicionalidad sean complementarios y no alternativos, y que el orden entre ambos importe: donde la capacidad administrativa es la restricción efectiva, añadir condiciones no mejora el resultado, solo multiplica los incumplimientos. Es también la explicación menos ideológica del fracaso de muchos programas en países frágiles, y conecta directamente con la propiedad local: una administración que ha trabajado con el Fondo en el diseño de una reforma la considera suya con mucha más facilidad que una que la recibe hecha.

El caso español de 2012 lo ilustra bien. La asistencia técnica para la reestructuración del sector financiero se apoyó en el diagnóstico que el propio Fondo había hecho meses antes en el programa de evaluación del sector financiero, y su resultado fue bueno precisamente porque las autoridades hicieron suyas las recomendaciones y las ejecutaron con rapidez.

\subsubsection{Sus límites}\label{anx:instrumental:cd_limites}

Tres objeciones acotan el valor de esta función. La primera es que, al financiarla mayoritariamente terceros, \textbf{la agenda tiende a seguir las prioridades de los donantes} más que las necesidades de los receptores, y las materias de moda atraen fondos que no siempre están donde más falta hacen. La segunda es que \textbf{su eficacia es muy difícil de medir}: entre el asesoramiento y el resultado median años y mil factores, de modo que la evaluación se queda casi siempre en indicadores de actividad. Y la tercera es la ya conocida tentación de dispersarse: si todo es macrocrítico, también lo es toda la asistencia técnica imaginable, y la institución acaba repartiendo un recurso escaso entre demasiadas materias.

Conviene cerrar con una observación que no suele hacerse. El desarrollo de capacidades forma, año tras año, a los cuadros técnicos de los ministerios y bancos centrales de los países miembros en los marcos analíticos del Fondo. \textbf{Es un canal de influencia que no aparece en ningún recuento de votos}, y probablemente explica mejor que las cuotas por qué el diagnóstico del Fondo se ha convertido en el lenguaje común de la política macroeconómica en buena parte del mundo.

\clearpage
\anexo{El programa: programación financiera y marcos de diagnóstico\label{anx:programa}}

Hasta aquí sabemos qué puede pedir un país y qué se le exige a cambio. Falta lo más técnico y lo menos conocido: \textbf{cómo se construye materialmente un programa}. Cuando una misión del Fondo llega a un país en crisis no trae un modelo econométrico ni un juicio improvisado: trae un procedimiento contable con setenta años de antigüedad que obliga a que las cuentas del Estado, del banco central, del sector exterior y de la economía real encajen entre sí, y que convierte un objetivo de reservas en un techo de gasto público por una cadena de pasos que no admite discusión una vez aceptados los supuestos.

Este anexo desarrolla ese procedimiento en dos partes. Primero, el método: de dónde viene, qué cuentas enlaza y con qué supuestos se cierra. Después, un \textbf{ejemplo numérico completo}, de un país imaginario con los problemas de todos, que recorre la cadena entera y permite ver, con números, lo que el cuerpo del tema sostiene en palabras: que la financiación del Fondo no sustituye al ajuste, sino que lo reparte en el tiempo. Los dos apartados siguientes se ocupan de los tres marcos de diagnóstico con los que el Fondo decide hoy sobre la deuda, el sector exterior y el sector financiero.

\subsection{La programación financiera: ¿cómo se construye un programa?}\label{anx:programa:metodo}

\subsubsection{De dónde viene: un modelo para países sin datos}\label{anx:programa:origen}

La programación financiera nace del trabajo de POLAK (1957) en el propio Fondo, y su mérito original fue de diseño más que de teoría. En los años cincuenta la institución trataba con países cuyas estadísticas eran escasas o inexistentes, y hacía falta un marco que permitiera diagnosticar y programar un ajuste con lo poco que siempre hay: \textbf{el balance del banco central} y una estimación de la demanda de dinero.

El razonamiento parte de dos igualdades. La oferta de dinero es la suma de las reservas internacionales y del crédito interno del banco central, $M^{s}=R+CI$; y la demanda de dinero es proporcional a la renta nominal, $M^{d}=P\,Y/V$, con una velocidad $V$ estable. Igualando ambas y tomando diferencias se obtiene la ecuación central del enfoque monetario de la balanza de pagos:

$$\Delta R=\Delta M^{d}-\Delta CI$$

cuya derivación y supuestos se estudian en [\authorfont{Ver Tema 3.B.12}] y no se repiten aquí. Lo que interesa en este tema es el paso siguiente, que es el que convirtió una ecuación explicativa en un instrumento de política. \textbf{La ecuación se invierte}: fijado el objetivo de reservas que el programa persigue, $\Delta R^{*}$, y estimada la demanda de dinero, lo único que queda por determinar es el crédito que el banco central puede conceder:

$$\Delta CI\;\le\;\Delta M^{d}-\Delta R^{*}$$

Esa desigualdad es el \textbf{techo de crédito interno}, y de ella cuelga todo lo demás. Nótese lo que acaba de ocurrir: una identidad contable se ha transformado en una restricción política que ningún gobierno puede discutir sin discutir antes sus supuestos.

\subsubsection{Las cuatro cuentas y sus ligaduras}\label{anx:programa:cuentas}

El marco moderno amplía aquel esquema monetario a cuatro conjuntos de cuentas que se proyectan simultáneamente y que han de ser mutuamente coherentes.

\begin{center}\begin{minipage}{\linewidth}\centering
{\small\textbf{El cuadro macroeconómico: las cuatro cuentas y las ligaduras que las obligan a cuadrar}}\par\vspace{0.3em}
\includegraphics{anx3_sectores.pdf}\par\vspace{0.3em}
{\small\textit{Fuente: }elaboración propia a partir de la programación financiera del FMI (POLAK, 1957, y desarrollos posteriores).}
\end{minipage}\end{center}

Conviene subrayar qué clase de artefacto es esto, porque de ahí se siguen tanto sus virtudes como sus críticas. \textbf{No es un modelo de comportamiento, sino un marco de coherencia contable}: no predice qué hará el consumo si suben los tipos, sino que impide afirmar a la vez que el déficit será del $2\%$, que el crédito crecerá un $15\%$ y que las reservas aumentarán, cuando esas tres cosas no caben juntas. Su valor está en que \textbf{obliga a hacer explícito el coste de cada objetivo}, y en que cualquier promesa política queda inmediatamente traducida a una cifra en otra cuenta.

Para cerrarlo hacen falta, eso sí, tres supuestos de comportamiento, y los tres son discutibles: que la \textbf{velocidad del dinero} es estable y por tanto la demanda de dinero predecible; que las \textbf{importaciones} dependen de la absorción con una propensión conocida; y que los \textbf{precios internos} están anclados, bajo tipo de cambio fijo, por los precios mundiales y por la paridad declarada.

\subsubsection{Objetivos, instrumentos y supuestos}\label{anx:programa:taxonomia}

Antes de proyectar nada, la misión clasifica todas las variables en tres grupos, y esa clasificación es media negociación.

Las \textbf{variables objetivo} son las que el programa quiere mover: reservas, inflación, crecimiento, saldo exterior. Las \textbf{variables de política} son las que las autoridades controlan de manera directa o suficientemente indirecta: el crédito interno, el saldo primario, el tipo de cambio, los tipos de interés, los precios administrados. Y las \textbf{variables exógenas} son las que el país padece: los precios mundiales de sus exportaciones, los tipos de interés internacionales, la demanda externa, las remesas.

La frontera entre las tres no es técnica. Que el gasto en intereses sea exógeno y el gasto primario endógeno, o que las remesas se den por dadas, son decisiones que determinan dónde recaerá el ajuste; y, como recordó TINBERGEN (1952), con menos instrumentos que objetivos alguien tendrá que renunciar a alguno ([\authorfont{Ver Tema 3.B.12}]). De ahí que la elección de instrumentos se acuerde con las autoridades, que son quienes los van a usar y quienes conocen lo que su sistema político soporta.

\subsubsection{Los dos escenarios y la brecha entre ellos}\label{anx:programa:escenarios}

Con el cuadro construido se proyectan dos trayectorias. La primera es el \textbf{escenario sin cambio de políticas}, que prolonga lo que el país viene haciendo y suele terminar en el agotamiento de las reservas o en una inflación explosiva: su función no es predecir, sino \textbf{mostrar lo que ocurre si no se hace nada}, y es el argumento más eficaz de cualquier misión. La segunda es el \textbf{escenario de programa}, que incorpora las medidas acordadas y alcanza los objetivos.

La distancia entre ambas es exactamente el ajuste que el programa pide, y el procedimiento consiste en encontrar la combinación de medidas que cubra esa distancia sin que ninguna de las cuatro cuentas deje de cuadrar. Como no hay solución analítica, \textbf{el cuadro se resuelve por iteración}: se prueba una combinación, se comprueba la coherencia, se ajusta y se vuelve a empezar.

\subsubsection{De los objetivos a los criterios de ejecución}\label{anx:programa:criterios}

El último paso es el que enlaza este anexo con el anterior. Los resultados del cuadro se convierten en las tres obligaciones que el país firma:

\begin{lnum}
    \item el objetivo de reservas se escribe como un \textbf{suelo de reservas internacionales netas};
    \item el techo de crédito se escribe como un \textbf{techo de activos internos netos} del banco central; y,
    \item el déficit compatible se escribe como un \textbf{techo de déficit} o de gasto primario del sector público.
\end{lnum}

Es decir: \textbf{los tres criterios de ejecución cuantitativos de cualquier programa del Fondo son las tres variables del modelo de programación financiera}. La condicionalidad cuantitativa no es una lista de exigencias políticas, sino el modelo convertido en contrato ([\authorfont{Ver Anexo \ref{anx:instrumental:tipos}}]).

La columna del año 0 describe una situación que el lector reconocerá: un déficit del $6\%$ del PIB financiado en sus dos terceras partes por el banco central, un crédito interno que crece deprisa, una inflación del $24\%$ con el tipo de cambio fijo y una cuenta corriente del $-5\%$ del PIB que ya solo se cubre vendiendo reservas. Estas han caído en diez y quedan en treinta, apenas \textbf{mes y medio de importaciones}.

\subsubsection{Por qué el país llama a la puerta}\label{anx:programa:sincambio}

La columna central proyecta lo que ocurre si nada cambia, y lo importante es por dónde llega el desenlace. No llega por el lado monetario, porque la inflación alta eleva la demanda nominal de dinero y absorbe una parte del crédito: llega \textbf{por el sector exterior}. Con una inflación interna del $24\%$ frente al $3\%$ mundial y una paridad fija, el tipo de cambio real se aprecia un quinto más, la cuenta corriente se deteriora hasta el $-80$ y, sobre todo, \textbf{el capital privado se da la vuelta}: quien ve venir una devaluación no espera a que llegue. Con una salida neta de diez y la financiación oficial reducida a la mitad, las reservas tendrían que caer setenta, cuando solo hay treinta. \textbf{El país se queda sin reservas hacia el quinto mes}, y a partir de ahí el ajuste lo impone la realidad en forma de control de cambios, devaluación desordenada o impago.

Este escenario es el que lleva al gobierno a Washington, y conviene reparar en que ya contiene una lección: cuando un país pide un programa, el ajuste no está por decidir, \textbf{ya ha empezado}. Lo único que se discute es si se hará de forma ordenada y con financiación, o desordenada y sin ella.

\subsubsection{La cadena del programa}\label{anx:programa:cadena}

El programa fija dos objetivos: reducir la inflación del $24\%$ al $10\%$ y detener la pérdida de reservas, con una recuperación modesta de cinco en el primer año. A partir de ahí, la aritmética hace el resto.

Merece la pena detenerse en el paso 4, porque es donde se ve el poder del método. Nadie ha discutido el gasto público; nadie ha decidido qué partida recortar; nadie ha hablado de política. Se ha fijado un objetivo de reservas, se ha proyectado la demanda de dinero y se ha repartido el crédito disponible, y de esas tres operaciones ha salido que \textbf{el déficit no puede pasar del $2{,}4\%$ del PIB cuando el año anterior fue del $6{,}0\%$}. El ajuste de tres puntos y medio del PIB no es una exigencia ideológica del Fondo: es lo que queda después de restar.

\subsubsection{Lo que cambia la financiación del Fondo}\label{anx:programa:finfondo}

Ahora puede responderse con números a la pregunta que vertebra el tema. ¿Qué habría pasado sin los veinticinco del Fondo?

Sin financiación exterior adicional, el país no podría conformarse con recuperar cinco de reservas: con mes y medio de importaciones en caja y el capital privado en fuga, necesitaría llegar a un nivel defendible, digamos veinte más, en el propio año. Recorriendo la misma cadena con $\Delta R=+20$: el techo de crédito interno baja de $19{,}4$ a $4{,}4$; si el crédito al sector privado sigue creciendo con el PIB nominal, \textbf{el crédito al Gobierno tendría que caer en $7{,}8$ en vez de aumentar en $7{,}2$}; y el déficit máximo pasaría de $27{,}2$ a $12{,}2$, esto es, del $2{,}4\%$ al $1{,}1\%$ del PIB. El ajuste fiscal del primer año sería de \textbf{casi cinco puntos del PIB en lugar de tres y medio}, y la mejora exigida a la cuenta corriente, de treinta en vez de quince.

Esa diferencia es exactamente lo que compra el dinero del Fondo. Y conviene precisar \textbf{cómo} lo compra, porque hay aquí un detalle técnico que casi nunca se explica y que resuelve una objeción evidente. Si el Fondo presta veinticinco, ¿no bastaría con eso para cumplir el objetivo de reservas sin ajustar nada? No, y la razón está en la definición del criterio: las reservas que el programa vigila son las \textbf{netas}, de las que se deduce el crédito del propio Fondo. Un desembolso aumenta a la vez las reservas brutas y los pasivos de reserva, de modo que \textbf{deja las reservas netas exactamente donde estaban}. El criterio de ejecución está construido, deliberadamente, para que el dinero del Fondo no pueda satisfacerlo.

Lo que el desembolso sí hace es duplicar las \textbf{reservas brutas}, de treinta a sesenta, y ese colchón utilizable es lo que permite tolerar una posición neta más baja mientras el ajuste surte efecto. De ahí la formulación exacta, que es la misma a la que llega el estudio de los mecanismos de ajuste: \textbf{financiar no sustituye al ajuste, lo reparte en el tiempo} ([\authorfont{Ver Tema 3.B.12}]). La mejora de treinta en la cuenta corriente sigue siendo necesaria; lo que cambia es que se hará en tres años en lugar de en uno, y que al final habrá además que devolver los veinticinco, lo que obliga a que el ajuste se complete.

\subsubsection{Qué enseña el ejemplo y qué esconde}\label{anx:programa:limites}

El ejercicio enseña tres cosas que ninguna descripción verbal transmite igual de bien. La primera, que \textbf{la aritmética es vinculante}: fijados los objetivos y los supuestos, el ajuste no se negocia, se calcula. La segunda, que los tres criterios de ejecución de un programa no son arbitrarios, sino las tres incógnitas del cuadro. Y la tercera, que financiación y ajuste son \textbf{sustitutivos dentro del año y complementarios a lo largo del programa}.

Pero conviene ser igual de explícito con lo que el ejemplo esconde, porque ahí están las críticas serias al método y lo que el opositor debe poder contestar.

El primer punto débil es la \textbf{velocidad del dinero}. Todo el cálculo depende de que $V$ se mantenga en cinco, y en una crisis la velocidad es justamente lo que no se queda quieto: si la gente huye del dinero nacional, la demanda cae, el techo de crédito calculado resulta demasiado holgado y el programa fracasa; si la desinflación es más rápida de lo previsto, ocurre lo contrario y el ajuste pedido es innecesariamente duro. El segundo es la \textbf{circularidad de la inflación}: para proyectar la demanda de dinero hay que suponer una inflación que es, a la vez, uno de los objetivos del programa.

El tercero es que se trata de un marco de \textbf{un solo periodo y de demanda}: no hay respuesta de la oferta, ni multiplicadores, ni expectativas, ni efectos de la contracción sobre la propia recaudación, de modo que el ajuste fiscal calculado tiende a quedarse corto cuando la caída de la actividad reduce los ingresos. El cuarto es que \textbf{la composición no aparece}: el cuadro dice que el déficit debe bajar treinta y tres, y es indiferente a que se consiga subiendo el impuesto sobre el consumo o recortando la inversión pública, que es precisamente lo que decide el coste social del programa y su reversibilidad. Y el quinto es que el \textbf{crecimiento es un supuesto, no un resultado}, lo que explica buena parte del sesgo optimista que las evaluaciones posteriores encuentran de forma sistemática en las proyecciones de los programas.

La defensa del método, con todo, es sólida y conviene conocerla. Fue concebido para países sin datos ni modelos, impone una disciplina de coherencia que ningún marco alternativo ha mejorado, y en setenta años nadie ha propuesto algo que lo sustituya. Lo que sí ha ocurrido es que se le han añadido las piezas que le faltaban, y a ellas se dedica el resto de este anexo: un marco para juzgar si la deuda aguanta, otro para juzgar si el tipo de cambio está donde debe, y un tercero para el sector financiero.
